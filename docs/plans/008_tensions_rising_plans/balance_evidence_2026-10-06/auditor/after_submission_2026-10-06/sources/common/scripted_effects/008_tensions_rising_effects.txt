tensions_rising_prepare_random_event_fire = {
	if = {
		limit = {
			has_variable = automatic_event_firing_context
			check_variable = { automatic_event_firing_context > 0 }
			NOT = { is_tensions_rising_automatic_event_available = yes }
		}
		set_temp_variable = { event_single_fire_allowed = 0 }
	}
}

get_tensions_rising_stage = {
	set_temp_variable = { tensions_rising_stage = 0 }
	if = {
		limit = {
			OR = {
				has_global_flag = { flag = chaos_tier value = 4 }
				has_global_flag = { flag = chaos_tier value = 5 }
			}
		}
		set_temp_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_4 }
	}
	else_if = {
		limit = {
			has_global_flag = { flag = chaos_tier value = 3 }
		}
		set_temp_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_3 }
	}
	else_if = {
		limit = {
			has_global_flag = { flag = chaos_tier value = 2 }
		}
		set_temp_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_2 }
	}
	else_if = {
		limit = {
			has_global_flag = { flag = chaos_tier value = 1 }
		}
		set_temp_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_1 }
	}
	if = {
		limit = { check_variable = { tensions_rising_stage > 0 } }
		set_temp_variable = { events_log_evolution_event_id = constant:tensions_rising_event_log.event_id }
		set_temp_variable = { events_log_evolution_type = constant:tensions_rising_event_log.evolution_type }
		set_temp_variable = { events_log_evolution_stage = tensions_rising_stage }
		set_temp_variable = { events_log_evolution_tier = tensions_rising_stage }
		if = {
			limit = { NOT = { is_current_evolution_enabled = yes } }
			set_temp_variable = { tensions_rising_stage = 0 }
		}
	}
}

# A no-effect first firing may share a pending report with a later effective
# shock. Keep its explanation until the first packet is actually applied.
capture_tensions_rising_first_effectful_report_stage = {
	if = {
		limit = { NOT = { has_country_flag = tensions_rising_report_effectful } }
		set_country_flag = tensions_rising_report_effectful
		set_variable = { tensions_rising_report_stage = tensions_rising_stage }
		clr_country_flag = tensions_rising_report_terminal
		clr_country_flag = tensions_rising_report_baseline_full_world_tension
	}
}

# An automatic shock has a month to develop before another automatic entry.
# Manual requests retain their immediate report and coalescing behavior.
mark_tensions_rising_recent_shock = {
	set_temp_variable = { tensions_rising_shock_cooldown_days = constant:tensions_rising_reporting.automatic_cooldown_days }
	set_global_flag = { flag = tensions_rising_recent_shock days = tensions_rising_shock_cooldown_days }
}

apply_tensions_rising_event_effect = {
	set_temp_variable = { tensions_rising_chaos_before = global.chaos_meter_value }
	set_temp_variable = { tensions_rising_applied_world_tension = 0 }
	set_temp_variable = { tensions_rising_stage = 0 }
	get_tensions_rising_stage = yes
	set_temp_variable = { tensions_rising_new_report_batch = 0 }
	if = {
		limit = { NOT = { has_country_flag = tensions_rising_report_pending } }
		set_temp_variable = { tensions_rising_new_report_batch = 1 }
		clear_variable = tensions_rising_report_stage
		clear_variable = tensions_rising_report_world_tension
		clear_variable = tensions_rising_report_chaos
		set_variable = { tensions_rising_report_world_tension = 0 }
		set_variable = { tensions_rising_report_chaos = 0 }
		clr_country_flag = tensions_rising_report_terminal
		clr_country_flag = tensions_rising_report_baseline_full_world_tension
		clr_country_flag = tensions_rising_report_followup_attempted
		clr_country_flag = tensions_rising_report_headlines_counted
		clr_country_flag = tensions_rising_report_effectful
		set_country_flag = tensions_rising_report_pending
	}
	if = {
		limit = { has_global_flag = world_end }
		set_temp_variable = { tensions_rising_stage = 0 }
		if = {
			limit = { check_variable = { tensions_rising_new_report_batch > 0 } }
			set_variable = { tensions_rising_report_stage = tensions_rising_stage }
			set_country_flag = tensions_rising_report_terminal
		}
	}
	else_if = {
		limit = {
			check_variable = { tensions_rising_stage < constant:tensions_rising_event_log.stage_1 }
			is_tensions_rising_world_tension_full = yes
		}
		if = {
			limit = { check_variable = { tensions_rising_new_report_batch > 0 } }
			set_variable = { tensions_rising_report_stage = tensions_rising_stage }
			set_country_flag = tensions_rising_report_baseline_full_world_tension
		}
	}
	else_if = {
		limit = { check_variable = { tensions_rising_stage < constant:tensions_rising_event_log.stage_1 } }
		capture_tensions_rising_first_effectful_report_stage = yes
		set_temp_variable = { tensions_rising_world_tension_packet = constant:tensions_rising_world_tension_gain.baseline }
		apply_tensions_rising_distributed_world_tension = yes
		add_to_variable = { tensions_rising_report_world_tension = tensions_rising_applied_world_tension }
		mark_tensions_rising_recent_shock = yes
		tensions_rising_sync_world_tension_tracker = yes
	}
	else = {
		capture_tensions_rising_first_effectful_report_stage = yes
		record_tensions_rising_evolution_if_needed = yes
		apply_tensions_rising_stage_world_tension = yes
		apply_tensions_rising_stage_chaos = yes
		apply_tensions_rising_timer_pulse = yes
		select_tensions_rising_relation_pairs = yes
		apply_tensions_rising_ai_posture = yes
		if = {
			limit = { NOT = { has_country_flag = tensions_rising_report_followup_attempted } }
			set_country_flag = tensions_rising_report_followup_attempted
			schedule_tensions_rising_followup = yes
		}
		update_tensions_rising_achievement_tracking = yes
		mark_tensions_rising_recent_shock = yes
	}
	set_temp_variable = { tensions_rising_chaos_delta = global.chaos_meter_value }
	subtract_from_temp_variable = { tensions_rising_chaos_delta = tensions_rising_chaos_before }
	add_to_variable = { tensions_rising_report_chaos = tensions_rising_chaos_delta }
	if = {
		limit = { check_variable = { tensions_rising_new_report_batch > 0 } }
		country_event = { id = chaosx.nr8.16 }
	}
}

# Country scope. Input: tensions_rising_world_tension_packet, read only.
# Output: caller-owned tensions_rising_applied_world_tension (actual meter points).
# Other incident callers may ignore the output; this helper never changes the
# pending main-report snapshot. Recipient count derives from the shared packet
# and minimum slice. Meta injection supplies the native static iterator field.
apply_tensions_rising_distributed_world_tension = {
	set_temp_variable = { tensions_rising_world_tension_before = global.threat }
	set_temp_variable = { tensions_rising_applied_world_tension = 0 }
	set_temp_variable = { tensions_rising_world_tension_country_count = 0 }
	every_country = {
		limit = { is_tensions_rising_world_tension_source_country = yes }
		add_to_temp_variable = { tensions_rising_world_tension_country_count = 1 }
	}
	if = {
		limit = {
			check_variable = { tensions_rising_world_tension_country_count > 0 }
			check_variable = { tensions_rising_world_tension_packet > 0 }
		}
		set_temp_variable = { tensions_rising_world_tension_source_cap = tensions_rising_world_tension_packet }
		divide_temp_variable = { tensions_rising_world_tension_source_cap = constant:tensions_rising_world_tension_gain.minimum_country_slice }
		set_temp_variable = { tensions_rising_world_tension_unrounded_cap = tensions_rising_world_tension_source_cap }
		round_temp_variable = tensions_rising_world_tension_source_cap
		if = {
			limit = {
				check_variable = { var = tensions_rising_world_tension_source_cap value = tensions_rising_world_tension_unrounded_cap compare = greater_than }
			}
			subtract_from_temp_variable = { tensions_rising_world_tension_source_cap = 1 }
		}
		clamp_temp_variable = { var = tensions_rising_world_tension_source_cap min = 1 }
		set_temp_variable = { tensions_rising_world_tension_selected_count = tensions_rising_world_tension_country_count }
		clamp_temp_variable = { var = tensions_rising_world_tension_selected_count max = tensions_rising_world_tension_source_cap }
		set_temp_variable = { tensions_rising_world_tension_slice = tensions_rising_world_tension_packet }
		divide_temp_variable = { tensions_rising_world_tension_slice = tensions_rising_world_tension_selected_count }
		meta_effect = {
			text = {
				every_country = {
					limit = { is_tensions_rising_world_tension_source_country = yes }
					random_select_amount = [RECIPIENT_COUNT]
					add_named_threat = {
						threat = tensions_rising_world_tension_slice
						name = tensions_rising_global_panic_threat
					}
				}
			}
			RECIPIENT_COUNT = "[?tensions_rising_world_tension_selected_count|.0]"
		}
	}
	set_temp_variable = { tensions_rising_applied_world_tension = global.threat }
	subtract_from_temp_variable = { tensions_rising_applied_world_tension = tensions_rising_world_tension_before }
	multiply_temp_variable = { tensions_rising_applied_world_tension = constant:tensions_rising_reporting.world_tension_scale }
	clamp_temp_variable = { var = tensions_rising_applied_world_tension min = 0 }
}

					check_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_1 }
					NOT = { has_global_flag = tensions_rising_stage_1_recorded }
				}
				AND = {
					check_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_2 }
					NOT = { has_global_flag = tensions_rising_stage_2_recorded }

					check_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_3 }

					check_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_4 }
					NOT = { has_global_flag = tensions_rising_stage_4_recorded }

			limit = { check_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_4 } }
			set_global_flag = tensions_rising_stage_4_recorded

		}
	}

# Starts or replaces a pulse using its configured duration and exact expiry date.
tensions_rising_begin_timer_pulse = {
	set_variable = { global.tensions_rising_timer_pulse_strength = tensions_rising_pulse_strength }
	set_variable = { global.tensions_rising_timer_pulse_days = tensions_rising_pulse_days }
	tensions_rising_set_timer_pulse_deadline = yes
	refresh_tensions_rising_timer_pulse_active_flag = yes
}

# Rebuilds a pulse deadline when active state predates the exact expiry variable.
tensions_rising_migrate_timer_pulse_deadline = {
	if = {
		limit = {
			has_global_flag = tensions_rising_timer_pulse_active
			NOT = { has_variable = global.tensions_rising_timer_pulse_expiry_date }
			has_variable = global.tensions_rising_timer_pulse_days
		}
		tensions_rising_set_timer_pulse_deadline = yes
	}
}

# Converts the exact remaining-day budget into a campaign-date expiry.
tensions_rising_set_timer_pulse_deadline = {
	set_temp_variable = { chaosx_date_days = global.tensions_rising_timer_pulse_days }
	set_temp_variable = { chaosx_date_offset = 0 }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { global.tensions_rising_timer_pulse_expiry_date = global.date }
	add_to_variable = { global.tensions_rising_timer_pulse_expiry_date = chaosx_date_offset }
}

# Reads exact remaining time before a same-stage extension.
tensions_rising_update_timer_pulse_remaining_days = {
	set_temp_variable = { chaosx_date_offset = global.tensions_rising_timer_pulse_expiry_date }
	subtract_from_temp_variable = { chaosx_date_offset = global.date }
	set_temp_variable = { chaosx_date_days = 0 }
	chaosx_convert_date_offset_to_days = yes
	set_variable = { global.tensions_rising_timer_pulse_days = chaosx_date_days }
}

apply_tensions_rising_relation_pair = {
	set_temp_variable = { tensions_rising_relation_actor_found = 0 }
	set_temp_variable = { tensions_rising_relation_target_found = 0 }

	random_country = {
		limit = {
			is_tensions_rising_priority_relation_actor = yes
			NOT = { has_country_flag = tensions_rising_recent_relation_actor }
		}
		save_event_target_as = tensions_rising_first_actor
		ROOT = { set_temp_variable = { tensions_rising_relation_actor_found = 1 } }
	}
	if = {
		limit = { check_variable = { tensions_rising_relation_actor_found < 1 } }
		random_country = {
			limit = {
				is_tensions_rising_valid_diplomatic_country = yes
				NOT = { has_country_flag = tensions_rising_recent_relation_actor }
			}
			save_event_target_as = tensions_rising_first_actor
			ROOT = { set_temp_variable = { tensions_rising_relation_actor_found = 1 } }
		}
	}
	if = {
		limit = { check_variable = { tensions_rising_relation_actor_found < 1 } }
		random_country = {
			limit = { is_tensions_rising_valid_diplomatic_country = yes }
			save_event_target_as = tensions_rising_first_actor
			ROOT = { set_temp_variable = { tensions_rising_relation_actor_found = 1 } }
		}
	}

	if = {
		limit = { check_variable = { tensions_rising_relation_actor_found > 0 } }
		event_target:tensions_rising_first_actor = {
			random_country = {
				limit = {
					is_tensions_rising_relation_target_for_prev = yes
					is_neighbor_of = PREV
					NOT = { has_country_flag = tensions_rising_recent_relation_actor }
				}
				save_event_target_as = tensions_rising_second_actor
				ROOT = { set_temp_variable = { tensions_rising_relation_target_found = 1 } }
			}
			if = {
				limit = { ROOT = { check_variable = { tensions_rising_relation_target_found < 1 } } }
				random_country = {
					limit = {
						is_tensions_rising_relation_target_for_prev = yes
						OR = {
							is_major = yes
							is_faction_leader = yes
						}
						NOT = { has_country_flag = tensions_rising_recent_relation_actor }
					}
					save_event_target_as = tensions_rising_second_actor
					ROOT = { set_temp_variable = { tensions_rising_relation_target_found = 1 } }
				}
			}
			if = {
				limit = { ROOT = { check_variable = { tensions_rising_relation_target_found < 1 } } }
				random_country = {
					limit = {
						is_tensions_rising_relation_target_for_prev = yes
						NOT = { has_country_flag = tensions_rising_recent_relation_actor }
					}
					save_event_target_as = tensions_rising_second_actor
					ROOT = { set_temp_variable = { tensions_rising_relation_target_found = 1 } }
				}
			}
			if = {
				limit = { ROOT = { check_variable = { tensions_rising_relation_target_found < 1 } } }
				random_country = {
					limit = { is_tensions_rising_relation_target_for_prev = yes }
					save_event_target_as = tensions_rising_second_actor
					ROOT = { set_temp_variable = { tensions_rising_relation_target_found = 1 } }
				}
			}
		}
	}

	if = {
		limit = { check_variable = { tensions_rising_relation_target_found > 0 } }
		apply_tensions_rising_relation_modifier_to_targets = yes
		add_to_variable = { global.tensions_rising_relation_pair_count = 1 }
		set_global_flag = { flag = tensions_rising_recent_relation_damage days = @TENSIONS_RISING_RELATION_FOLLOWUP_WINDOW_DAYS }
	}
}

run_tensions_rising_incident_border_war_roll = {
	if = {
		limit = {
			check_variable = {
				var = tensions_rising_stage
				value = constant:tensions_rising_incident_border_war.min_stage
				compare = greater_than_or_equals
			}
			NOT = { has_global_flag = tensions_rising_recent_border_war_incident }
			any_country = { is_tensions_rising_border_war_actor = yes }
		}
		if = {
			limit = { check_variable = { tensions_rising_stage = constant:tensions_rising_event_log.stage_4 } }
			set_temp_variable = { tensions_rising_border_war_chance = tensions_rising_border_war_stage_4_chance }
		}
		else = {
			set_temp_variable = { tensions_rising_border_war_chance = tensions_rising_border_war_stage_3_chance }
		}
		set_temp_variable_to_random = {
			var = tensions_rising_border_war_roll
			min = 1
			max = 100
			integer = yes
		}
		if = {
			limit = {
				check_variable = {
					var = tensions_rising_border_war_roll
					value = tensions_rising_border_war_chance
					compare = less_than_or_equals
				}
			}
			set_temp_variable = { tensions_rising_border_actor_found = 0 }
			set_temp_variable = { tensions_rising_border_target_found = 0 }
			select_tensions_rising_border_war_pair = yes
			try_start_tensions_rising_border_war_from_pair = yes
		}
	}
}

select_tensions_rising_border_war_pair = {
	set_temp_variable = { tensions_rising_border_actor_found = 0 }
	set_temp_variable = { tensions_rising_border_target_found = 0 }
	random_country = {
		limit = { is_tensions_rising_border_war_actor = yes }
		save_event_target_as = tensions_rising_border_attacker
		ROOT = { set_temp_variable = { tensions_rising_border_actor_found = 1 } }
	}
	if = {
		limit = { check_variable = { tensions_rising_border_actor_found > 0 } }
		event_target:tensions_rising_border_attacker = {
			random_country = {
				limit = {
					is_tensions_rising_border_war_target_for_prev = yes
					is_neighbor_of = PREV
				}
				save_event_target_as = tensions_rising_border_defender
				ROOT = { set_temp_variable = { tensions_rising_border_target_found = 1 } }
			}
		}
	}
}

try_start_tensions_rising_border_war_from_pair = {
	if = {
		limit = { check_variable = { tensions_rising_border_target_found > 0 } }
		set_temp_variable = { tensions_rising_border_attacker_state_found = 0 }
		set_temp_variable = { tensions_rising_border_defender_state_found = 0 }
		event_target:tensions_rising_border_attacker = {
			random_owned_controlled_state = {
				limit = {
					impassable = no
					NOT = { has_border_war = yes }
					any_neighbor_state = {
						impassable = no
						NOT = { has_border_war = yes }
						is_owned_and_controlled_by = event_target:tensions_rising_border_defender
					}
				}
				set_variable = { ROOT.tensions_rising_border_attacker_state = id }
				ROOT = { set_temp_variable = { tensions_rising_border_attacker_state_found = 1 } }
			}
		}
		if = {
			limit = { check_variable = { tensions_rising_border_attacker_state_found > 0 } }
			event_target:tensions_rising_border_defender = {
				random_owned_controlled_state = {
					limit = {
						impassable = no
						NOT = { has_border_war = yes }
						any_neighbor_state = { state = var:ROOT.tensions_rising_border_attacker_state }
					}
					set_variable = { ROOT.tensions_rising_border_defender_state = id }
					ROOT = { set_temp_variable = { tensions_rising_border_defender_state_found = 1 } }
				}
			}
		}
		if = {
			limit = { check_variable = { tensions_rising_border_defender_state_found > 0 } }
			start_border_war = {
				change_state_after_war = no
				combat_width = @TENSIONS_RISING_BORDER_WAR_COMBAT_WIDTH
				minimum_duration_in_days = @TENSIONS_RISING_BORDER_WAR_MINIMUM_DURATION_DAYS
				attacker = {
					state = var:tensions_rising_border_attacker_state
					num_provinces = @TENSIONS_RISING_BORDER_WAR_NUM_PROVINCES
					on_win = chaosx.nr8.13
					on_lose = chaosx.nr8.14
					on_cancel = chaosx.nr8.15
				}
				defender = {
					state = var:tensions_rising_border_defender_state
					num_provinces = @TENSIONS_RISING_BORDER_WAR_NUM_PROVINCES
					on_win = chaosx.nr8.13
					on_lose = chaosx.nr8.14
					on_cancel = chaosx.nr8.15
				}
			}
			record_tensions_rising_border_war_started = yes
		}
		clear_variable = tensions_rising_border_attacker_state
		clear_variable = tensions_rising_border_defender_state
	}
}

}

update_tensions_rising_achievement_tracking = {
	if = {
		limit = {
			check_variable = {
				var = tensions_rising_report_stage
				value = constant:tensions_rising_event_log.stage_1
				compare = greater_than
			}
			is_tensions_rising_world_at_peace = yes
			NOT = { has_country_flag = tensions_rising_report_headlines_counted }
			NOT = { has_global_flag = achievement_tensions_only_headlines_ready }
			NOT = { has_global_flag = tensions_rising_only_headlines_broken }
		}
		set_country_flag = tensions_rising_report_headlines_counted
		set_global_flag = tensions_rising_only_headlines_tracking
		add_to_variable = { global.tensions_rising_only_headlines_count = 1 }
		if = {
			limit = {
				check_variable = {
					var = global.tensions_rising_only_headlines_count
					value = constant:tensions_rising_achievement.only_headlines_required
					compare = greater_than_or_equals
				}
			}
			set_global_flag = achievement_tensions_only_headlines_ready
		}
	}

	if = {
		limit = {
			is_ai = no
			check_variable = { tensions_rising_stage > constant:tensions_rising_event_log.stage_2 }
			is_tensions_rising_world_tension_full = yes
			has_war = no
			NOT = { has_country_flag = achievement_tensions_thin_wire_ready }
			NOT = { has_country_flag = tensions_rising_thin_wire_watch }
		}
		set_country_flag = { flag = tensions_rising_thin_wire_watch days = @TENSIONS_RISING_THIN_WIRE_FLAG_DAYS }
		clr_country_flag = tensions_rising_thin_wire_broken
		set_temp_variable = { chaosx_date_days = constant:tensions_rising_achievement.thin_wire_watch_days }
		set_temp_variable = { chaosx_date_offset = 0 }
		chaosx_convert_days_to_date_offset = yes
		set_variable = { tensions_rising_thin_wire_expiry_date = global.date }
		add_to_variable = { tensions_rising_thin_wire_expiry_date = chaosx_date_offset }
		if = {
			limit = { NOT = { has_country_flag = tensions_rising_thin_wire_callback_pending } }
			set_country_flag = tensions_rising_thin_wire_callback_pending
			set_temp_variable = { tensions_rising_thin_wire_step_days = constant:tensions_rising_achievement.thin_wire_step_days }
			country_event = { id = chaosx.nr8.20 days = tensions_rising_thin_wire_step_days }
		}
	}

	update_tensions_rising_blackout_achievement_tracking = yes
}

			clr_country_flag = tensions_rising_thin_wire_broken

			NOT = { has_global_flag = achievement_tensions_only_headlines_ready }
		}
		set_global_flag = tensions_rising_only_headlines_broken
	}
	if = {
		limit = {
			has_global_flag = tensions_rising_recent_relation_damage
			NOT = { has_global_flag = achievement_tensions_one_denial_ready }
			OR = {
				ROOT = { has_country_flag = tensions_rising_one_denial_eligible }

			}
		}

	}
