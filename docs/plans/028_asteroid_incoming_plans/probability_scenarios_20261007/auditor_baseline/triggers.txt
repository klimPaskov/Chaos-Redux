# ============================================================================
# CHAOS REDUX - EVENT 028 ASTEROID INCOMING TRIGGERS
# ============================================================================
# Event-owned validation and ledger predicates for Event 028. These triggers
# validate exact targets and provide a read-only country-count preflight.
# They never choose or change a target. The parent event owns
# candidate weighting, evolution availability, decision definitions, and the
# final report surface.
#
# Target-country classification reuses the shared
# is_actual_nonhuman_country classifier. Special Chaos countries remain valid
# when they are still human-populated and use normal civilian systems.
# ============================================================================

# Native any-scope count fields use literal mirrors here. Keep these equal to
# asteroid_incoming_runtime.minimum_target_pair_count and
# asteroid_incoming_runtime.target_country_minimum_usable_states.
@asteroid_target_pair_count = 3
@asteroid_target_country_state_count = 2
# Mirror of asteroid_incoming_recovery.active_mission_limit for count_triggers.
@asteroid_recovery_active_mission_limit = 2

asteroid_incoming_state_has_meaningful_settlement = {
	hidden_trigger = {
		scope_exists = yes
		impassable = no
		state_population_k > constant:asteroid_incoming_target.minimum_population_k
		OR = {
			has_state_category = city
			has_state_category = large_city
			has_state_category = metropolis
			has_state_category = megalopolis
			industrial_complex > constant:asteroid_incoming_target.minimum_building_level
			arms_factory > constant:asteroid_incoming_target.minimum_building_level
			dockyard > constant:asteroid_incoming_target.minimum_building_level
			infrastructure > constant:asteroid_incoming_target.minimum_building_level
			rail_way > constant:asteroid_incoming_target.minimum_building_level
			supply_node > constant:asteroid_incoming_target.minimum_building_level
			air_base > constant:asteroid_incoming_target.minimum_building_level
			naval_base > constant:asteroid_incoming_target.minimum_building_level
			state_strategic_value > constant:asteroid_incoming_target.minimum_strategic_value
		}
	}
}

asteroid_incoming_state_is_valid_target = {
	hidden_trigger = {
		scope_exists = yes
		impassable = no
		NOT = { has_state_category = wasteland }
		NOT = { has_state_flag = asteroid_incoming_main_crater }
		NOT = { has_state_flag = asteroid_incoming_fragment_site }
		asteroid_incoming_state_has_meaningful_settlement = yes
		OWNER = { exists = yes }
		CONTROLLER = { exists = yes }
	}
}

# COUNTRY scope. One center and one distinct surviving capital candidate.
# The same predicate is used before automatic selection and by the builder.
asteroid_incoming_country_has_viable_target_pair = {
	hidden_trigger = {
		exists = yes
		is_actual_nonhuman_country = no
		NOT = { tag = CXT }
		any_owned_state = {
			count = @asteroid_target_country_state_count
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = PREV
		}
		any_owned_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = PREV
			is_fully_controlled_by = PREV
		}
	}
}

# STATE scope. The center itself can be contested, but a separate owned
# fully controlled state must be available for the surviving government.
asteroid_incoming_state_is_valid_main_center = {
	hidden_trigger = {
		asteroid_incoming_state_is_valid_target = yes
		is_owned_and_controlled_by = OWNER
		OWNER = {
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				is_fully_controlled_by = PREV
				NOT = { state = PREV.PREV }
			}
		}
	}
}

# STATE scope inside the capital-relocation country's every_owned_state or
# every_controlled_state. A fallback capital can be any usable state other than
# the impact centre, wasteland, an impassable state or an already struck crater.
asteroid_incoming_state_is_fallback_capital_site = {
	hidden_trigger = {
		scope_exists = yes
		impassable = no
		NOT = { has_state_category = wasteland }
		NOT = { has_state_flag = asteroid_incoming_main_crater }
		NOT = { has_state_flag = asteroid_incoming_state_transaction_applied }
		NOT = { state = event_target:asteroid_incoming_locked_state }
	}
}

asteroid_incoming_country_is_valid_target = {
	hidden_trigger = {
		asteroid_incoming_country_has_viable_target_pair = yes
		has_event_target = asteroid_incoming_chooser
		NOT = { tag = event_target:asteroid_incoming_chooser }
	}
}

# ACTING COUNTRY scope. Counts countries, so the three possible centers and
# their controlled owned backup states are necessarily distinct. No target,
# random draw, persistent variable, flag, or array is written by this check.
asteroid_incoming_target_pool_can_exist = {
	hidden_trigger = {
		exists = yes
		any_country = {
			count = @asteroid_target_pair_count
			NOT = { tag = PREV }
			asteroid_incoming_country_has_viable_target_pair = yes
		}
	}
}

		NOT = { has_global_flag = asteroid_incoming_transaction_resolved }

		asteroid_incoming_target_pool_can_exist = yes
	}
}

asteroid_incoming_country_has_backup_state = {
	hidden_trigger = {
		exists = yes
		has_event_target = asteroid_incoming_candidate_state
		any_owned_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = PREV
			is_fully_controlled_by = PREV
			NOT = { state = event_target:asteroid_incoming_candidate_state }
		}
	}
}

# Called after the parent has saved candidate country and state targets. The
# state must still be owned and controlled by that same country at the point
# where the pair is accepted.
asteroid_incoming_target_pair_is_valid = {
	hidden_trigger = {
		has_event_target = asteroid_incoming_candidate_country
		has_event_target = asteroid_incoming_candidate_state
		event_target:asteroid_incoming_candidate_country = {
			asteroid_incoming_country_is_valid_target = yes
			asteroid_incoming_country_has_backup_state = yes
		}
		event_target:asteroid_incoming_candidate_state = {
			asteroid_incoming_state_is_valid_main_center = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_candidate_country
		}
	}
}

# The locked pair preserves the accepted coordinates even if the original
# country is annexed or the state's owner, controller, category, or identity
# changes during the two-day trajectory. Only an unusable state scope or the
# absence of a live country to receive the physical transaction fails closed.
asteroid_incoming_locked_target_pair_is_valid = {
	hidden_trigger = {
		has_event_target = asteroid_incoming_locked_state
		event_target:asteroid_incoming_locked_state = {
			scope_exists = yes
			impassable = no
			NOT = { has_state_category = wasteland }
			OWNER = { exists = yes }
			CONTROLLER = { exists = yes }
		}
	}
}

asteroid_incoming_locked_transaction_is_valid = {
	hidden_trigger = {
		has_global_flag = asteroid_incoming_transaction_active
		has_global_flag = asteroid_incoming_target_locked
		NOT = { has_global_flag = asteroid_incoming_transaction_resolved }
		check_variable = {
			var = global.asteroid_incoming_transaction_id
			value = constant:asteroid_incoming_runtime.zero
			compare = greater_than
		}
		# The monthly watchdog marks a lock whose resolver callbacks were all
		# lost as overdue, so the late resolver still strikes.
		OR = {
			check_variable = {
				var = global.asteroid_incoming_lock_expires_day
				value = global.num_days
				compare = greater_than_or_equals
			}
			has_global_flag = asteroid_incoming_lock_overdue
		}
		asteroid_incoming_locked_target_pair_is_valid = yes
	}
}

# COUNTRY scope, with the chooser in ROOT. Protected diplomatic relationships
# never receive a positive strike weight without a separately implemented and
# audited betrayal strategy; Event 028 does not invent one. A co-belligerent
# fighting on ROOT's side is protected even outside ROOT's faction.
asteroid_incoming_ai_target_is_protected = {
	hidden_trigger = {
		OR = {
			tag = ROOT
			is_subject_of = ROOT
			ROOT = { is_subject_of = PREV }
			AND = { is_in_faction_with = ROOT NOT = { has_war_with = ROOT } }
			has_war_together_with = ROOT
			has_non_aggression_pact_with = ROOT
			has_guaranteed = ROOT
			is_guaranteed_by = ROOT
		}
	}
}

# COUNTRY scope, with the chooser in ROOT. ROOT itself, a faction partner, or
# a country fighting on ROOT's side.
asteroid_incoming_ai_country_is_friendly_ground_holder = {
	hidden_trigger = {
		OR = {
			tag = ROOT
			AND = { is_in_faction_with = ROOT NOT = { has_war_with = ROOT } }
			has_war_together_with = ROOT
		}
	}
}

# STATE scope, a candidate target state, with the chooser in ROOT. True when
# the state itself, a neighbouring state, or a state two land steps away is
# controlled by ROOT, a faction partner or a co-belligerent, so the inner
# impact rings would devastate the chooser's own or friendly ground.
asteroid_incoming_ai_strike_would_hit_friendly_ground = {
	hidden_trigger = {
		OR = {
			OWNER = { asteroid_incoming_ai_country_is_friendly_ground_holder = yes }
			CONTROLLER = { asteroid_incoming_ai_country_is_friendly_ground_holder = yes }
			any_neighbor_state = {
				OR = {
					OWNER = { asteroid_incoming_ai_country_is_friendly_ground_holder = yes }
					CONTROLLER = { asteroid_incoming_ai_country_is_friendly_ground_holder = yes }
					any_neighbor_state = {
						OR = {
							OWNER = { asteroid_incoming_ai_country_is_friendly_ground_holder = yes }
							CONTROLLER = { asteroid_incoming_ai_country_is_friendly_ground_holder = yes }
						}
					}
				}
			}
		}
	}
}

# Ring planning treats every adjacent state scope as a graph carrier and only
# excludes states already visited by this breadth-first search. Empty or
# impassable land states therefore carry the shortest path outward, while the
# later resolvable-transaction trigger prevents them from receiving losses or
# damage. A state in another profile remains traversable so the profile writer
# can deduplicate it by strongest applicable profile after discovery.
asteroid_incoming_state_is_valid_ring_candidate = {
	hidden_trigger = {
		scope_exists = yes
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_planner_seen_states
				value = THIS
			}
		}
	}
}

# COUNTRY scope. Fragment centers use human civilian systems and exclude the
# dedicated system country; human-populated Special Chaos countries remain valid.
asteroid_incoming_country_can_host_fragment_center = {
	hidden_trigger = {
		exists = yes
		uses_normal_civilian_systems = yes
		NOT = { tag = CXT }
	}
}

asteroid_incoming_state_is_valid_fragment_site = {
	hidden_trigger = {
		has_global_flag = asteroid_incoming_fragmentation_active
		check_variable = {
			var = global.asteroid_incoming_fragment_site_count
			value = constant:asteroid_incoming_runtime.maximum_fragment_sites
			compare = less_than
		}
		asteroid_incoming_state_is_valid_target = yes
		OWNER = { asteroid_incoming_country_can_host_fragment_center = yes }
		CONTROLLER = { asteroid_incoming_country_can_host_fragment_center = yes }
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_main_footprint_states
				value = THIS
			}
		}
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_fragment_site_states
				value = THIS
			}
		}
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_fragment_reserved_first_ring_states
				value = THIS
			}
		}
	}
}

# A fragment center should have a usable land neighbor whenever the currently
# available global candidate pool contains one. The selector owns the
# fail-open pool check; this predicate only describes the preferred local
# geometry and deliberately does not require the neighbor to be unsettled.
asteroid_incoming_fragment_site_has_valid_land_neighbor = {
	hidden_trigger = {
		any_neighbor_state = {
			scope_exists = yes
			impassable = no
			NOT = { has_state_category = wasteland }
			OWNER = { exists = yes }
			CONTROLLER = { exists = yes }
			NOT = { has_state_flag = asteroid_incoming_main_crater }
			NOT = { has_state_flag = asteroid_incoming_fragment_site }
		}
	}
}

asteroid_incoming_state_is_valid_fragment_selection = {
	hidden_trigger = {

		OR = {

		}
	}
}

# STATE scope, one applied fragment centre. True when a continent entry saved
# by asteroid_incoming_record_fragment_continents lies on the same continent.

				has_event_target = asteroid_incoming_fragment_continent_2_state

				has_event_target = asteroid_incoming_fragment_continent_3_state
				is_on_same_continent_as = event_target:asteroid_incoming_fragment_continent_3_state
			}

				has_event_target = asteroid_incoming_fragment_continent_4_state
				is_on_same_continent_as = event_target:asteroid_incoming_fragment_continent_4_state

				has_event_target = asteroid_incoming_fragment_continent_5_state
				is_on_same_continent_as = event_target:asteroid_incoming_fragment_continent_5_state
			}

				has_event_target = asteroid_incoming_fragment_continent_6_state
				is_on_same_continent_as = event_target:asteroid_incoming_fragment_continent_6_state
			}
		}
	}
}

# Crater competition opens once the country's own emergency and
# reconstruction work is closed. Global dust can last for years, so the dust
# phase does not hold back crater actions; both phases may then be visible.

# ----------------------------------------------------------------------------
# Recovery target ranking
# ----------------------------------------------------------------------------
# Each state-targeted recovery action lists one state per country: the
# highest-ranked state that is open for that action now. The ranking is read
# live in target_trigger, so once a state is used, resolved or lost, the next
# state takes its place on the following daily check.
# Emergency work ranks an active rescue burden first, then the strongest impact
# profile, then the larger current population. Reconstruction ranks the
# strongest impact profile first, then the larger infrastructure plus railway
# gap against the pre-impact snapshot (the Stabilize the Damaged Zone order).
# Equal states fall back to the lower state id.
# Both comparisons read and write only their own unscoped temporaries inside
# one scripted trigger, so no temporary crosses a scripted-trigger boundary.

# STATE scope, candidate state; FROM is the decision's target state.
asteroid_incoming_state_outranks_target_for_emergency = {
	hidden_trigger = {
		set_temp_variable = { asteroid_incoming_rank_state_id = THIS }
		set_temp_variable = { asteroid_incoming_rank_state_profile = asteroid_incoming_state_profile }
		set_temp_variable = { asteroid_incoming_rank_state_population = state_population_k }
		FROM = {
			set_temp_variable = { asteroid_incoming_rank_target_id = THIS }
			set_temp_variable = { asteroid_incoming_rank_target_profile = asteroid_incoming_state_profile }
			set_temp_variable = { asteroid_incoming_rank_target_population = state_population_k }
		}
		OR = {
			AND = {
				has_state_flag = asteroid_incoming_rescue_burden_active
				FROM = { NOT = { has_state_flag = asteroid_incoming_rescue_burden_active } }
			}
			AND = {
				# The target does not lead on the rescue burden. NOT is a NOR,
				# so the pair is grouped in AND.
				NOT = {
					AND = {
						FROM = { has_state_flag = asteroid_incoming_rescue_burden_active }
						NOT = { has_state_flag = asteroid_incoming_rescue_burden_active }
					}
				}
				OR = {
					check_variable = { asteroid_incoming_rank_state_profile > asteroid_incoming_rank_target_profile }
					AND = {
						NOT = { check_variable = { asteroid_incoming_rank_state_profile < asteroid_incoming_rank_target_profile } }
						OR = {
							check_variable = { asteroid_incoming_rank_state_population > asteroid_incoming_rank_target_population }
							AND = {
								NOT = { check_variable = { asteroid_incoming_rank_state_population < asteroid_incoming_rank_target_population } }
								check_variable = { asteroid_incoming_rank_state_id < asteroid_incoming_rank_target_id }
							}
						}
					}
				}
			}
		}
	}
}

# STATE scope, candidate state; FROM is the decision's target state.
asteroid_incoming_state_outranks_target_for_reconstruction = {
	hidden_trigger = {
		set_temp_variable = { asteroid_incoming_rank_state_id = THIS }
		set_temp_variable = { asteroid_incoming_rank_state_profile = asteroid_incoming_state_profile }
		set_temp_variable = { asteroid_incoming_rank_state_gap = asteroid_incoming_preimpact_infrastructure }
		subtract_from_temp_variable = { asteroid_incoming_rank_state_gap = building_level@infrastructure }
		add_to_temp_variable = { asteroid_incoming_rank_state_gap = asteroid_incoming_preimpact_railway }
		subtract_from_temp_variable = { asteroid_incoming_rank_state_gap = building_level@rail_way }
		FROM = {
			set_temp_variable = { asteroid_incoming_rank_target_id = THIS }
			set_temp_variable = { asteroid_incoming_rank_target_profile = asteroid_incoming_state_profile }
			set_temp_variable = { asteroid_incoming_rank_target_gap = asteroid_incoming_preimpact_infrastructure }
			subtract_from_temp_variable = { asteroid_incoming_rank_target_gap = building_level@infrastructure }
			add_to_temp_variable = { asteroid_incoming_rank_target_gap = asteroid_incoming_preimpact_railway }
			subtract_from_temp_variable = { asteroid_incoming_rank_target_gap = building_level@rail_way }
		}
		OR = {
			check_variable = { asteroid_incoming_rank_state_profile > asteroid_incoming_rank_target_profile }
			AND = {
				NOT = { check_variable = { asteroid_incoming_rank_state_profile < asteroid_incoming_rank_target_profile } }
				OR = {
					check_variable = { asteroid_incoming_rank_state_gap > asteroid_incoming_rank_target_gap }
					AND = {
						NOT = { check_variable = { asteroid_incoming_rank_state_gap < asteroid_incoming_rank_target_gap } }
						check_variable = { asteroid_incoming_rank_state_id < asteroid_incoming_rank_target_id }
					}
				}
			}
		}
	}
}

# The dispatcher uses this as the fire-once gate. It is deliberately based on
# the six persisted targets rather than on a random-country count alone, so a
# partially populated pool can never expose a fourth or invalid target.
asteroid_incoming_has_three_valid_target_pairs = {
	hidden_trigger = {
		check_variable = {
			var = global.asteroid_incoming_target_pool_count
			value = constant:asteroid_incoming_runtime.minimum_target_pair_count
			compare = equals
		}
		has_event_target = asteroid_incoming_target_1_country
		has_event_target = asteroid_incoming_target_1_state
		has_event_target = asteroid_incoming_target_2_country
		has_event_target = asteroid_incoming_target_2_state
		has_event_target = asteroid_incoming_target_3_country
		has_event_target = asteroid_incoming_target_3_state
		event_target:asteroid_incoming_target_1_country = {
			asteroid_incoming_country_has_viable_target_pair = yes
			NOT = { tag = event_target:asteroid_incoming_chooser }
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				is_fully_controlled_by = PREV
				NOT = { state = event_target:asteroid_incoming_target_1_state }
			}
		}
		event_target:asteroid_incoming_target_1_state = {
			asteroid_incoming_state_is_valid_main_center = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_target_1_country
		}
		event_target:asteroid_incoming_target_2_country = {
			asteroid_incoming_country_has_viable_target_pair = yes
			NOT = { tag = event_target:asteroid_incoming_chooser }
			NOT = { tag = event_target:asteroid_incoming_target_1_country }
			NOT = { tag = event_target:asteroid_incoming_target_3_country }
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				is_fully_controlled_by = PREV
				NOT = { state = event_target:asteroid_incoming_target_2_state }
			}
		}
		event_target:asteroid_incoming_target_2_state = {
			asteroid_incoming_state_is_valid_main_center = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_target_2_country
			NOT = { state = event_target:asteroid_incoming_target_1_state }
			NOT = { state = event_target:asteroid_incoming_target_3_state }
		}
		event_target:asteroid_incoming_target_3_country = {
			asteroid_incoming_country_has_viable_target_pair = yes
			NOT = { tag = event_target:asteroid_incoming_chooser }
			NOT = { tag = event_target:asteroid_incoming_target_1_country }
			NOT = { tag = event_target:asteroid_incoming_target_2_country }
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				is_fully_controlled_by = PREV
				NOT = { state = event_target:asteroid_incoming_target_3_state }
			}
		}
		event_target:asteroid_incoming_target_3_state = {
			asteroid_incoming_state_is_valid_main_center = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_target_3_country
			NOT = { state = event_target:asteroid_incoming_target_1_state }
			NOT = { state = event_target:asteroid_incoming_target_2_state }
		}
	}
}
