rf_transport_cost_debit = {
	set_temp_variable = { rf_transport_cost_paid = constant:rf_transport_cost_model.none }
	# Allocate quote storage in this frame before the nested trigger returns.
	set_temp_variable = { rf_transport_cost_route = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_civilian = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_pp = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_manpower = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_motorized = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_transport = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_mine_seq = constant:rf_transport_cost_model.none }
	if = {
		limit = { rf_transport_cost_can_pay = yes }
		rf_transport_cost_store_receipt = yes
		if = {
			limit = { check_variable = { rf_transport_cost_pp > constant:rf_transport_cost_model.none } }
			set_temp_variable = { rf_transport_cost_signed = rf_transport_cost_pp }
			multiply_temp_variable = { rf_transport_cost_signed = constant:rf_transport_cost_model.negative_one }
			add_political_power = var:rf_transport_cost_signed
		}
		if = {
			limit = { check_variable = { rf_transport_cost_manpower > constant:rf_transport_cost_model.none } }
			set_temp_variable = { rf_transport_cost_signed = rf_transport_cost_manpower }
			multiply_temp_variable = { rf_transport_cost_signed = constant:rf_transport_cost_model.negative_one }
			add_manpower = var:rf_transport_cost_signed
		}
		if = {
			limit = { check_variable = { rf_transport_cost_motorized > constant:rf_transport_cost_model.none } }
			set_temp_variable = { rf_transport_cost_signed = rf_transport_cost_motorized }
			multiply_temp_variable = { rf_transport_cost_signed = constant:rf_transport_cost_model.negative_one }
			add_equipment_to_stockpile = { type = motorized_equipment_1 amount = rf_transport_cost_signed }
		}
		set_temp_variable = { rf_transport_cost_signed = rf_transport_cost_transport }
		multiply_temp_variable = { rf_transport_cost_signed = constant:rf_transport_cost_model.negative_one }
		if = {
			limit = { check_variable = { rf_transport_cost_route = constant:rf_transport_cost_model.land } }
			add_equipment_to_stockpile = { type = train_equipment_1 amount = rf_transport_cost_signed }
		}
		else = {
			add_equipment_to_stockpile = { type = convoy_1 amount = rf_transport_cost_signed }
		}
		set_temp_variable = { rf_transport_cost_paid = constant:rf_transport_cost_model.one }
	}
}

rf_transport_cost_refund = {
	set_temp_variable = { rf_transport_cost_closed = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_route = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_civilian = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_pp = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_manpower = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_motorized = constant:rf_transport_cost_model.none }
	set_temp_variable = { rf_transport_cost_transport = constant:rf_transport_cost_model.none }
	if = {
		limit = { rf_transport_cost_load_receipt = yes }
		# Close before crediting. No closure path can refund these fields twice.
		rf_transport_cost_clear_loaded_receipt = yes
		if = {
			limit = { check_variable = { rf_transport_cost_pp > constant:rf_transport_cost_model.none } }
			add_political_power = var:rf_transport_cost_pp
		}
		if = {
			limit = { check_variable = { rf_transport_cost_manpower > constant:rf_transport_cost_model.none } }
			add_manpower = var:rf_transport_cost_manpower
		}
		if = {
			limit = { check_variable = { rf_transport_cost_motorized > constant:rf_transport_cost_model.none } }
			add_equipment_to_stockpile = { type = motorized_equipment_1 amount = rf_transport_cost_motorized }
		}
		if = {
			limit = { check_variable = { rf_transport_cost_route = constant:rf_transport_cost_model.land } }
			add_equipment_to_stockpile = { type = train_equipment_1 amount = rf_transport_cost_transport }
		}
		else = {
			add_equipment_to_stockpile = { type = convoy_1 amount = rf_transport_cost_transport }
		}
		set_temp_variable = { rf_transport_cost_closed = constant:rf_transport_cost_model.one }
	}
}

rf_transport_cost_clear_receipt = {
	set_temp_variable = { rf_transport_cost_closed = constant:rf_transport_cost_model.none }
	if = {
		limit = { rf_transport_cost_load_receipt = yes }
		rf_transport_cost_clear_loaded_receipt = yes
		set_temp_variable = { rf_transport_cost_closed = constant:rf_transport_cost_model.one }
	}
}
