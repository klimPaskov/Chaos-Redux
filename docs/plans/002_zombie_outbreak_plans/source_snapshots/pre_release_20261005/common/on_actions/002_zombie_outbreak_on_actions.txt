# ============================================================================
# CHAOS REDUX - ZOMBIE OUTBREAK ON ACTIONS
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				chaos_unit_family_provider_501_register = yes
				zombie_register_event19_variant_provider = yes
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = { fallout_world_rewrite_callbacks_are_allowed = yes }
				if = {
					limit = {
						ROOT = { is_zombie_outbreak_country = yes }
					}
					FROM.FROM = {
						zombie_on_state_captured_by_outbreak = yes
					}
				}
			}
		}
	}

	on_join_faction = {
		effect = {
			if = {
				limit = {
					has_global_flag = anti_zombie_league_formed
					FROM = {
						has_country_flag = azl_leader
						is_faction_leader = yes
					}
					is_valid_anti_zombie_league_country = yes
				}
				prepare_anti_zombie_league_member = yes
				grant_anti_zombie_league_membership_benefits = yes
			}
			if = {
				limit = {
					has_global_flag = anti_zombie_league_formed
					FROM = {
						has_country_flag = azl_leader
						is_faction_leader = yes
					}
					NOT = { is_valid_anti_zombie_league_country = yes }
				}
				if = {
					limit = { is_in_faction = yes }
					leave_faction = yes
				}
				remove_anti_zombie_league_membership_package = yes
			}

				if = {
					limit = {
						has_anti_zombie_league_membership_idea = yes
						NOT = { is_in_anti_zombie_league_faction = yes }
					}
					remove_anti_zombie_league_membership_package = yes
				}
		}
	}

	on_leave_faction = {
		effect = {
				if = {
					limit = {
						fallout_world_rewrite_callbacks_are_allowed = yes
						has_anti_zombie_league_membership_idea = yes
						NOT = { has_country_flag = azl_disbanding_league }
					}
					remove_anti_zombie_league_membership_package = yes
				}
		}
	}

	on_assume_faction_leadership = {
		effect = {
			if = {
				limit = {
					has_global_flag = anti_zombie_league_formed
					FROM = {
						has_country_flag = azl_leader
					}
					is_valid_anti_zombie_league_country = yes
				}
				FROM = {
					clr_country_flag = azl_leader
				}
				set_country_flag = azl_leader
				save_global_event_target_as = azl_faction_leader
				prepare_anti_zombie_league_member = yes
				grant_anti_zombie_league_membership_benefits = yes
			}
			if = {
				limit = {
					has_global_flag = anti_zombie_league_formed
					FROM = {
						has_country_flag = azl_leader
					}
					NOT = { is_valid_anti_zombie_league_country = yes }
				}
				if = {
					limit = { is_in_faction = yes }
					leave_faction = yes
				}
				remove_anti_zombie_league_membership_package = yes
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					OR = {
						ROOT = { is_zombie_outbreak_country = yes }
						FROM = { is_zombie_outbreak_country = yes }
					}
				}
				mark_zombie_war_participants = yes
			}
		}
	}

	# Zombie Outbreak
	on_daily = {
		effect = {
			if = {
				limit = {
					is_zombie_outbreak_country = yes
					exists = yes
					has_global_flag = has_zombie_outbreak
					NOT = { has_global_flag = zombie_system_disabled }
				}
				zombie_process_controlled_state_decay = yes
			}

			if = {
				limit = {
					is_weaponized_zombie_outbreak_country = yes
					exists = yes
					num_of_controlled_states > 0
					NOT = { has_capitulated = yes }
				}
				weaponized_zombie_turn_hostile_to_humans_if_at_war = yes
				weaponized_zombie_resolve_canonical_standard_neighbor_merges = yes
				weaponized_zombie_expand_against_neighbors = yes
			}

			if = {
				limit = {
					original_tag = ZZZ
					exists = yes
					has_global_flag = has_zombie_outbreak
					NOT = { has_global_flag = zombie_system_disabled }
				}
				refresh_zombie_world_threat_source = yes
				check_main_zombie_continent_rejoin_pressure = yes
				weaponized_zombie_expand_against_neighbors = yes
				annex_neighboring_dynamic_zombie_outbreaks_to_main = yes

				every_state = {
					limit = {
						is_controlled_by = ROOT
						NOT = { is_core_of = ROOT }
					}
					add_core_of = ROOT
				}

				relocate_isolated_zombie_capital = yes

				if = {
					limit = {
						tag = ZZZ
						NOT = { has_country_flag = cannibalism_wendigo_hannibal_country }
						NOT = {
							has_country_leader = {
								ruling_only = yes
								character = ZZZ_leader
							}
						}
					}
					refresh_dynamic_zombie_country_leader = yes
				}

				if = {
					limit = {
						has_country_flag = zombie_outbreak_dynamic_country
						NOT = { has_country_flag = cannibalism_wendigo_hannibal_country }
						NOT = { is_weaponized_zombie_outbreak_country = yes }
						NOT = {
							has_country_leader = {
								ruling_only = yes
								character = ZZZ_leader
							}
						}
					}
					refresh_dynamic_zombie_country_leader = yes
				}

				# Stop Monroe Doctrine white peace logic from trapping the outbreak.
				if = {
					limit = {
						any_country = {
							has_war_with = ROOT
							OR = {
								any_owned_state = { is_capital = yes is_on_continent = south_america }
								any_owned_state = { is_capital = yes is_on_continent = north_america }
							}
						}
						NOT = { has_war_with = USA }
					}
					declare_war_on = {
						target = USA
						type = annex_everything
					}
				}
			}

			if = {
				limit = {
					has_country_flag = azl_leader
					num_faction_members > 1
					NOT = { has_global_flag = azl_super_event_fired }
				}
				set_global_flag = {
					flag = super_event_visible
					days = 30
					value = 2
				}
				set_variable = { global.current_super_event_audio_id = 2 }
				play_current_super_event_audio = yes
				set_global_flag = azl_super_event_fired
			}
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = {
					tag = ZZZ
					has_global_flag = has_zombie_outbreak
					NOT = { has_global_flag = zombie_system_disabled }
				}
				update_all_zombie_outbreak_chances = yes
				attempt_dynamic_zombie_outbreak = yes
			}
			if = {
				limit = {
					has_global_flag = has_zombie_outbreak
					NOT = { has_global_flag = zombie_system_disabled }
					can_request_zombie_evacuation = yes
				}
				attempt_zombie_mass_evacuation = yes
			}
			if = {
				limit = {
					original_tag = ZZZ
					has_global_flag = has_zombie_outbreak
					NOT = { has_global_flag = anti_zombie_league_formed }
					NOT = { has_global_flag = azl_dissolved }
					NOT = { has_global_flag = zombie_system_disabled }
					NOT = {
						any_country = {
							has_country_flag = azl_formation_in_progress
						}
					}
					check_variable = {
						var = global.chaos_meter_value
						value = constant:anti_zombie_league.emergency_chaos
						compare = greater_than_or_equals
					}
				}
				calculate_total_zombie_strength = yes
				calculate_zombie_continent_presence = yes
				if = {
					limit = {
						check_variable = { total_zombie_divisions > constant:anti_zombie_league.emergency_divisions }
						check_variable = { zombie_controls_continent > 0 }
						any_country = {
							is_emergency_anti_zombie_league_founder_candidate = yes
						}
					}
					force_emergency_anti_zombie_league_formation = yes
				}
			}
			if = {
				limit = {
					has_global_flag = anti_zombie_league_formed
					NOT = { has_global_flag = zombie_system_disabled }
					has_country_flag = azl_leader
					is_faction_leader = yes
				}
				review_anti_zombie_league_membership_pressure = yes
			}
			if = {
				limit = {
					has_global_flag = anti_zombie_league_formed
					NOT = { has_global_flag = zombie_system_disabled }
					has_country_flag = azl_leader
					is_faction_leader = yes
					num_faction_members > 1
				}
				calculate_total_zombie_strength = yes
				if = {
					limit = {
						check_variable = {
							var = total_zombie_divisions
							value = constant:zombie_threat.very_high_divisions
							compare = greater_than_or_equals
						}
						check_variable = {
							var = total_zombie_controlled_states
							value = constant:zombie_threat.very_high_controlled_states
							compare = greater_than_or_equals
						}
					}
					azl_absorb_smaller_factions = yes
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			# ZZZ
			if = {
				limit = {
					original_tag = ZZZ
					NOT = { has_country_flag = cannibalism_wendigo_hannibal_country }
				}

				every_core_state = {
					remove_core_of = PREV
				}

				if = {
					limit = {
						tag = ZZZ
						has_global_flag = has_zombie_outbreak
						NOT = { has_global_flag = zombie_system_disabled }
					}
					add_to_variable = { global.zombie_main_capitulation_count = 1 }
					set_temp_variable = { zombie_main_collapse_current_day = global.num_days }
					set_temp_variable = { zombie_main_outbreak_days_alive = 99999 }
					if = {
						limit = { has_variable = global.zombie_main_outbreak_restore_day }
						set_temp_variable = { zombie_main_outbreak_days_alive = zombie_main_collapse_current_day }
						subtract_from_temp_variable = { zombie_main_outbreak_days_alive = global.zombie_main_outbreak_restore_day }
					}
					set_temp_variable = { zombie_main_collapse_is_quick = 0 }
					if = {
						limit = {
							check_variable = {
								var = global.zombie_main_capitulation_count
								value = 1
								compare = equals
							}
							check_variable = {
								var = zombie_main_outbreak_days_alive
								value = constant:zombie_threat.first_quick_collapse_days
								compare = less_than_or_equals
							}
						}
						set_temp_variable = { zombie_main_collapse_is_quick = 1 }
					}
					else_if = {
						limit = {
							check_variable = {
								var = global.zombie_main_capitulation_count
								value = 2
								compare = equals
							}
							check_variable = {
								var = zombie_main_outbreak_days_alive
								value = constant:zombie_threat.second_quick_collapse_days
								compare = less_than_or_equals
							}
						}
						set_temp_variable = { zombie_main_collapse_is_quick = 1 }
					}
					if = {
						limit = {
							check_variable = {
								var = global.zombie_main_capitulation_count
								value = constant:zombie_threat.main_collapse_shutdown_limit
								compare = greater_than_or_equals
							}
						}
						disable_zombie_outbreak_system = yes
					}
					else_if = {
						limit = {
							check_variable = {
								var = zombie_main_collapse_is_quick
								value = 1
								compare = less_than
							}
						}
						promote_strongest_dynamic_zombie_successor_to_main = yes
						if = {
							limit = {
								check_variable = {
									var = zombie_main_successor_promoted
									value = 1
									compare = less_than
								}
							}
							disable_zombie_outbreak_system = yes
						}
					}
					else = {
						promote_strongest_dynamic_zombie_successor_to_main = yes
					}
				}
			}

			if = {
				limit = {
					has_global_flag = has_zombie_outbreak
					NOT = { has_global_flag = zombie_system_disabled }
					FROM = {
						original_tag = ZZZ
						NOT = { has_country_flag = cannibalism_wendigo_hannibal_country }
					}
				}
				every_owned_state = {
					limit = { is_core_of = ROOT }
					transfer_state_to = FROM
				}
			}

			if = {
				limit = {
					is_zombie_outbreak_country = yes
					NOT = { has_country_flag = cannibalism_wendigo_hannibal_country }
					zombie_threat_fully_defeated = yes
				}
				on_zombie_threat_defeated = yes
			}
		}
	}
}
