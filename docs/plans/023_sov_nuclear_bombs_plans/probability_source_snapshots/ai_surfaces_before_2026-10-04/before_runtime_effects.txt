# Scratch temporaries are unscoped and expire with the enclosing effect chain.
# Each operation overwrites its inputs; callers consume persisted ledger outputs.
# ============================================================================
# CHAOS REDUX - EVENT 023 SOVIET NUCLEAR BOMBS RUNTIME EFFECTS
# ============================================================================
# This file is the reusable, event-neutral runtime boundary for the Event 023
# nuclear ledger. Event 023 owns authorization and action context. This file
# owns ledger transactions, the native launch_nuke boundary, and the Event 005
# custody snapshot/reconciliation bridge.
#
# The native on_nuke_drop callback owns detonation consequences. Accepted
# no-yield test accidents call the separate shared non-yield endpoint once.
# This adapter never calls chaos_meter_on_nuke_drop or a Fallout effect directly.
# ============================================================================

sov_nuclear_bombs_initialize_ledger = {
	if = {
		limit = { NOT = { has_global_flag = sov_nuclear_bombs_ledger_active } }
		set_variable = { global.sov_nuclear_bombs_ledger_total_registered = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_operational = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_assigned = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_transferred = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_dismantled = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_missing = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_expended = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_physical_custody = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_accounted_total = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.clean }
		set_variable = { global.sov_nuclear_bombs_ledger_transaction_sequence = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_evolution_last_credited_stage = constant:sov_nuclear_bombs_evolution.stage_baseline }
		set_variable = { global.sov_nuclear_bombs_event5_snapshot_generation = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_event5_snapshot_state_count = constant:sov_nuclear_bombs_ledger.zero }
		clr_global_flag = sov_nuclear_bombs_event5_reconciliation_pending
		clr_global_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
		set_variable = { global.sov_nuclear_bombs_soviet_usable_devices = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_native_stockpile_snapshot = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_major_receipt_count = constant:sov_nuclear_bombs_ledger.zero }
		clear_variable = global.sov_nuclear_bombs_major_receipt_first_actor_id
		clear_variable = global.sov_nuclear_bombs_major_receipt_first_target_id
		clear_variable = global.sov_nuclear_bombs_major_receipt_second_actor_id
		clear_variable = global.sov_nuclear_bombs_major_receipt_second_target_id
		set_variable = { global.sov_nuclear_bombs_breakaway_actor_count = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { global.sov_nuclear_bombs_breakaway_operational_actor_count = constant:sov_nuclear_bombs_ledger.zero }
		clear_array = global.sov_nuclear_bombs_breakaway_actor_entries
		set_global_flag = sov_nuclear_bombs_ledger_active
	}
	# Memorandum subsets also initialize when recovering an existing ledger.
	if = { limit = { NOT = { has_variable = global.sov_nuclear_bombs_ledger_in_transit } } set_variable = { global.sov_nuclear_bombs_ledger_in_transit = constant:sov_nuclear_bombs_ledger.zero } }
	if = { limit = { NOT = { has_variable = global.sov_nuclear_bombs_ledger_active_dismantlement } } set_variable = { global.sov_nuclear_bombs_ledger_active_dismantlement = constant:sov_nuclear_bombs_ledger.zero } }
}

# Country scope. Read the installed native country stockpile without replacing
# it. The snapshot is presentation evidence and is refreshed at bounded Event
# 023 transaction points plus on_daily_SOV; ledger reservations are subtracted
# only for the displayed unreserved operational total.
sov_nuclear_bombs_refresh_native_stockpile_snapshot = {
	if = {
		limit = { tag = SOV exists = yes }
		set_variable = { sov_nuclear_bombs_native_stockpile_snapshot = num_of_nukes }
		set_variable = { global.sov_nuclear_bombs_native_stockpile_snapshot = num_of_nukes }
	}
}

sov_nuclear_bombs_refresh_ledger_reconciliation = {
	sov_nuclear_bombs_initialize_ledger = yes
	set_temp_variable = { sov_nuclear_bombs_physical_custody_total = global.sov_nuclear_bombs_ledger_operational }
	add_to_temp_variable = { sov_nuclear_bombs_physical_custody_total = global.sov_nuclear_bombs_ledger_assigned }
	add_to_temp_variable = { sov_nuclear_bombs_physical_custody_total = global.sov_nuclear_bombs_ledger_reserved }
	add_to_temp_variable = { sov_nuclear_bombs_physical_custody_total = global.sov_nuclear_bombs_ledger_transferred }
	set_variable = { global.sov_nuclear_bombs_ledger_physical_custody = sov_nuclear_bombs_physical_custody_total }

	set_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_operational }
	add_to_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_assigned }
	add_to_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_reserved }
	add_to_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_transferred }
	add_to_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_dismantled }
	add_to_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_missing }
	add_to_temp_variable = { sov_nuclear_bombs_accounted_total = global.sov_nuclear_bombs_ledger_expended }
	set_variable = { global.sov_nuclear_bombs_ledger_accounted_total = sov_nuclear_bombs_accounted_total }

	if = {
		limit = {
			check_variable = {
				var = global.sov_nuclear_bombs_ledger_accounted_total
				value = global.sov_nuclear_bombs_ledger_total_registered
				compare = equals
			}
		}
		set_variable = { global.sov_nuclear_bombs_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.clean }
	}
	else = {
		set_variable = { global.sov_nuclear_bombs_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.disputed }
	}
	if = {
		limit = {
			OR = {
				has_global_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
				has_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
				check_variable = { var = global.sov_nuclear_bombs_ledger_operational value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
				check_variable = { var = global.sov_nuclear_bombs_ledger_assigned value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
				check_variable = { var = global.sov_nuclear_bombs_ledger_reserved value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
				check_variable = { var = global.sov_nuclear_bombs_ledger_transferred value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
				check_variable = { var = global.sov_nuclear_bombs_ledger_dismantled value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
				check_variable = { var = global.sov_nuclear_bombs_ledger_missing value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
				check_variable = { var = global.sov_nuclear_bombs_ledger_expended value = constant:sov_nuclear_bombs_ledger.zero compare = less_than }
			}
		}
		set_variable = { global.sov_nuclear_bombs_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.disputed }
	}
	set_variable = { global.sov_nuclear_bombs_soviet_usable_devices = constant:sov_nuclear_bombs_ledger.zero }
	if = {
		limit = { SOV = { exists = yes } }
		SOV = {
			sov_nuclear_bombs_initialize_actor_ledger = yes
			sov_nuclear_bombs_refresh_native_stockpile_snapshot = yes
			set_temp_variable = { sov_nuclear_bombs_soviet_usable_devices = sov_nuclear_bombs_native_stockpile_snapshot }
			subtract_from_temp_variable = { sov_nuclear_bombs_soviet_usable_devices = sov_nuclear_bombs_actor_reserved }
			subtract_from_temp_variable = { sov_nuclear_bombs_soviet_usable_devices = sov_nuclear_bombs_actor_in_transit }
			clamp_temp_variable = { var = sov_nuclear_bombs_soviet_usable_devices min = constant:sov_nuclear_bombs_ledger.zero }
			set_variable = { global.sov_nuclear_bombs_soviet_usable_devices = sov_nuclear_bombs_soviet_usable_devices }
			# Native production and pre-event bombs may exceed this allocation.
			# A pending native launch has its own exact callback/delta contract.
			if = {
				limit = { NOT = { has_country_flag = sov_nuclear_bombs_native_delivery_pending } }
				set_temp_variable = { sov_nuclear_bombs_native_required = sov_nuclear_bombs_actor_operational }
				add_to_temp_variable = { sov_nuclear_bombs_native_required = sov_nuclear_bombs_actor_assigned }
				add_to_temp_variable = { sov_nuclear_bombs_native_required = sov_nuclear_bombs_actor_reserved }
				if = {
					limit = { check_variable = { var = sov_nuclear_bombs_native_stockpile_snapshot value = sov_nuclear_bombs_native_required compare = less_than } }
					set_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
					set_variable = { global.sov_nuclear_bombs_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.disputed }
				}
			}
		}
	}
	sov_nuclear_bombs_refresh_transit_registry = yes
	sov_nuclear_bombs_refresh_dismantlement_registry = yes
}

# Country scope. Record the first major-power receipt and then require a
# reciprocal actor/target pair before opening the single nonterminal exchange
# super-event. Two Soviet strikes against different majors are not an exchange
# receipt: the second receipt must answer the first strike.
sov_nuclear_bombs_record_major_exchange_receipt = {
	if = {
		limit = {
			tag = SOV
			sov_nuclear_bombs_event_actor_is_valid = yes
			sov_nuclear_bombs_event_has_evolution_iii = yes
			check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.major_exchange_chaos compare = greater_than_or_equals }
			has_variable = sov_nuclear_bombs_major_receipt_candidate_actor_id
			has_variable = sov_nuclear_bombs_major_receipt_candidate_target_id
		}
		if = {
			limit = {
				NOT = { has_variable = global.sov_nuclear_bombs_major_receipt_count }
			}
			set_variable = { global.sov_nuclear_bombs_major_receipt_count = constant:sov_nuclear_bombs_ledger.zero }
		}
		if = {
			limit = {
				check_variable = { var = global.sov_nuclear_bombs_major_receipt_count value = constant:sov_nuclear_bombs_ledger.zero compare = equals }
			}
			set_variable = { global.sov_nuclear_bombs_major_receipt_first_actor_id = sov_nuclear_bombs_major_receipt_candidate_actor_id }
			set_variable = { global.sov_nuclear_bombs_major_receipt_first_target_id = sov_nuclear_bombs_major_receipt_candidate_target_id }
			set_variable = { global.sov_nuclear_bombs_major_receipt_count = constant:sov_nuclear_bombs_ledger.one }
			set_variable = { global.sov_nuclear_bombs_exchange_major_count = constant:sov_nuclear_bombs_ledger.one }
			set_global_flag = sov_nuclear_bombs_major_detonation_confirmed
			set_global_flag = sov_nuclear_bombs_evolution_iii_world_gate_opened
			set_global_flag = sov_nuclear_bombs_major_exchange_active
			set_country_flag = sov_nuclear_bombs_serious_crisis
		}
		else_if = {
			limit = {
				check_variable = { var = global.sov_nuclear_bombs_major_receipt_count value = constant:sov_nuclear_bombs_tuning.major_receipt_minimum compare = less_than }
				AND = {
					check_variable = { var = global.sov_nuclear_bombs_major_receipt_first_actor_id value = sov_nuclear_bombs_major_receipt_candidate_target_id compare = equals }
					check_variable = { var = global.sov_nuclear_bombs_major_receipt_first_target_id value = sov_nuclear_bombs_major_receipt_candidate_actor_id compare = equals }
				}
			}
			set_variable = { global.sov_nuclear_bombs_major_receipt_second_actor_id = sov_nuclear_bombs_major_receipt_candidate_actor_id }
			set_variable = { global.sov_nuclear_bombs_major_receipt_second_target_id = sov_nuclear_bombs_major_receipt_candidate_target_id }
			set_variable = { global.sov_nuclear_bombs_major_receipt_count = constant:sov_nuclear_bombs_tuning.major_receipt_minimum }
			set_variable = { global.sov_nuclear_bombs_exchange_major_count = constant:sov_nuclear_bombs_tuning.major_receipt_minimum }
			set_global_flag = sov_nuclear_bombs_major_detonation_confirmed
			set_global_flag = sov_nuclear_bombs_major_exchange_active
			set_global_flag = sov_nuclear_bombs_major_exchange_answered
			set_country_flag = sov_nuclear_bombs_serious_crisis
			sov_nuclear_bombs_open_major_exchange = yes
		}
		clear_variable = sov_nuclear_bombs_major_receipt_candidate_actor_id
		clear_variable = sov_nuclear_bombs_major_receipt_candidate_target_id
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# Country scope. The caller supplies the Event 023 opening amount in
# sov_nuclear_bombs_registration_amount. The one-shot flag prevents replaying
# the opening grant while evolution credits use the separate helper below.
sov_nuclear_bombs_register_event_owned_devices = {
	sov_nuclear_bombs_initialize_ledger = yes
	sov_nuclear_bombs_initialize_actor_ledger = yes
	if = {
		limit = {
			NOT = { has_country_flag = sov_nuclear_bombs_event_registration_complete }
			has_variable = sov_nuclear_bombs_registration_amount
			check_variable = { var = sov_nuclear_bombs_registration_amount value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
		}
		add_to_variable = { global.sov_nuclear_bombs_ledger_total_registered = sov_nuclear_bombs_registration_amount }
		add_to_variable = { global.sov_nuclear_bombs_ledger_operational = sov_nuclear_bombs_registration_amount }
		add_to_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_registration_amount }
		set_country_flag = sov_nuclear_bombs_actor_registered
		set_country_flag = sov_nuclear_bombs_event_registration_complete
		clear_variable = sov_nuclear_bombs_registration_amount
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# Country scope. The caller supplies a strictly increasing stage id in
# sov_nuclear_bombs_evolution_stage and the matching incremental device count
# in sov_nuclear_bombs_evolution_credit_amount.
sov_nuclear_bombs_credit_evolution_devices = {
	sov_nuclear_bombs_initialize_ledger = yes
	sov_nuclear_bombs_initialize_actor_ledger = yes
	if = {
		limit = {
			has_country_flag = sov_nuclear_bombs_actor_registered
			has_variable = sov_nuclear_bombs_evolution_stage
			has_variable = sov_nuclear_bombs_evolution_credit_amount
			check_variable = { var = sov_nuclear_bombs_evolution_stage value = global.sov_nuclear_bombs_evolution_last_credited_stage compare = greater_than }
			check_variable = { var = sov_nuclear_bombs_evolution_stage value = constant:sov_nuclear_bombs_evolution.stage_iv compare = less_than_or_equals }
			check_variable = { var = sov_nuclear_bombs_evolution_credit_amount value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
		}
		# The event-owned ledger and the native stockpile move together. This is
		# the only incremental grant route after the opening package.
		add_nuclear_bombs = sov_nuclear_bombs_evolution_credit_amount
		add_to_variable = { global.sov_nuclear_bombs_ledger_total_registered = sov_nuclear_bombs_evolution_credit_amount }
		add_to_variable = { global.sov_nuclear_bombs_ledger_operational = sov_nuclear_bombs_evolution_credit_amount }
		add_to_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_evolution_credit_amount }
		set_variable = { global.sov_nuclear_bombs_evolution_last_credited_stage = sov_nuclear_bombs_evolution_stage }
		clear_variable = sov_nuclear_bombs_evolution_stage
		clear_variable = sov_nuclear_bombs_evolution_credit_amount
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# State scope. The caller supplies the exact site amount in
# sov_nuclear_bombs_site_registration_amount. Registration moves devices from
# the current owner's operational ledger into the site's assigned ledger.
sov_nuclear_bombs_register_storage_site = {
	sov_nuclear_bombs_initialize_ledger = yes
	if = {
		limit = {
			NOT = { has_state_flag = sov_nuclear_bombs_registered_storage_site }
			ROOT = { has_variable = sov_nuclear_bombs_site_registration_amount }
			check_variable = { var = ROOT.sov_nuclear_bombs_site_registration_amount value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
			check_variable = { var = global.sov_nuclear_bombs_ledger_operational value = ROOT.sov_nuclear_bombs_site_registration_amount compare = greater_than_or_equals }
			OWNER = {
				exists = yes
				has_variable = sov_nuclear_bombs_actor_operational
				check_variable = { var = sov_nuclear_bombs_actor_operational value = ROOT.sov_nuclear_bombs_site_registration_amount compare = greater_than_or_equals }
			}
		}
		set_variable = { sov_nuclear_bombs_site_operational = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_assigned = ROOT.sov_nuclear_bombs_site_registration_amount }
		set_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_transferred = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_dismantled = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_missing = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_expended = constant:sov_nuclear_bombs_ledger.zero }

		}
		ROOT = { clear_variable = sov_nuclear_bombs_site_registration_amount }

	}

# Country scope. Event 023 writes all authorization/context fields before this
# call. A reservation is exactly one device and is returned by the release
# helper when an action is cancelled or fails validation.
sov_nuclear_bombs_reserve_device_for_action = {
	sov_nuclear_bombs_initialize_ledger = yes
	sov_nuclear_bombs_initialize_actor_ledger = yes
	if = {
		limit = {
			sov_nuclear_bombs_action_context_is_valid = yes
			NOT = { has_country_flag = sov_nuclear_bombs_action_reservation_active }
		}
		clear_variable = sov_nuclear_bombs_action_reservation_source
		if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_actor_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				check_variable = { var = global.sov_nuclear_bombs_ledger_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				has_variable = sov_nuclear_bombs_action_source_state
				var:sov_nuclear_bombs_action_source_state = {
					sov_nuclear_bombs_state_is_registered = yes
					is_owned_by = ROOT
					is_controlled_by = ROOT
					has_variable = sov_nuclear_bombs_site_assigned
					check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				}
			}
			set_variable = { sov_nuclear_bombs_action_reservation_source = constant:sov_nuclear_bombs_ledger.reservation_source_assigned }
			subtract_from_variable = { global.sov_nuclear_bombs_ledger_assigned = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }
			subtract_from_variable = { sov_nuclear_bombs_actor_assigned = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
			var:sov_nuclear_bombs_action_source_state = {
				subtract_from_variable = { sov_nuclear_bombs_site_assigned = constant:sov_nuclear_bombs_ledger.one }
				add_to_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.one }
			}
		}
		else_if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_actor_operational value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				check_variable = { var = global.sov_nuclear_bombs_ledger_operational value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				var:sov_nuclear_bombs_action_source_state = { is_owned_by = ROOT is_controlled_by = ROOT }
			}
			set_variable = { sov_nuclear_bombs_action_reservation_source = constant:sov_nuclear_bombs_ledger.reservation_source_operational }
			subtract_from_variable = { global.sov_nuclear_bombs_ledger_operational = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }
			subtract_from_variable = { sov_nuclear_bombs_actor_operational = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
		}
		else_if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals }
				check_variable = { var = global.sov_nuclear_bombs_ledger_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				has_variable = sov_nuclear_bombs_action_source_state
				var:sov_nuclear_bombs_action_source_state = {
					sov_nuclear_bombs_state_is_registered = yes
					check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				}
				OR = {
					AND = { NOT = { has_country_flag = sov_nuclear_bombs_action_remote_recovery_authorized } check_variable = { var = sov_nuclear_bombs_actor_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals } }
					AND = {
						has_country_flag = sov_nuclear_bombs_action_remote_recovery_authorized
						var:sov_nuclear_bombs_action_source_state = {
							has_variable = sov_nuclear_bombs_site_holder
							var:sov_nuclear_bombs_site_holder = {
								exists = yes
								check_variable = { var = sov_nuclear_bombs_actor_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
							}
						}
					}
				}
			}
			set_variable = { sov_nuclear_bombs_action_reservation_source = constant:sov_nuclear_bombs_ledger.reservation_source_transferred }
			subtract_from_variable = { global.sov_nuclear_bombs_ledger_transferred = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }
			if = {
				limit = { has_country_flag = sov_nuclear_bombs_action_remote_recovery_authorized }
				var:sov_nuclear_bombs_action_source_state = {
					set_variable = { ROOT.sov_nuclear_bombs_action_reservation_holder = sov_nuclear_bombs_site_holder }
				}
				var:sov_nuclear_bombs_action_reservation_holder = {
					sov_nuclear_bombs_initialize_actor_ledger = yes
					subtract_from_variable = { sov_nuclear_bombs_actor_transferred = constant:sov_nuclear_bombs_ledger.one }
					add_to_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
				}
			}
			else = {
				subtract_from_variable = { sov_nuclear_bombs_actor_transferred = constant:sov_nuclear_bombs_ledger.one }
				add_to_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
			}
			var:sov_nuclear_bombs_action_source_state = {
				subtract_from_variable = { sov_nuclear_bombs_site_transferred = constant:sov_nuclear_bombs_ledger.one }
				add_to_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.one }
			}
		}
		if = {
			limit = { has_variable = sov_nuclear_bombs_action_reservation_source }
			add_to_variable = { global.sov_nuclear_bombs_ledger_transaction_sequence = constant:sov_nuclear_bombs_ledger.one }
			set_variable = { sov_nuclear_bombs_action_transaction_sequence = global.sov_nuclear_bombs_ledger_transaction_sequence }
			set_country_flag = sov_nuclear_bombs_action_reservation_active
			set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.rejected }
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.reservation_missing }
			sov_nuclear_bombs_refresh_ledger_reconciliation = yes
		}
	}
}

# Country scope. This is the only rollback for a live reservation.
sov_nuclear_bombs_release_reserved_device = {

	sov_nuclear_bombs_initialize_actor_ledger = yes
	if = {
		limit = {
			has_country_flag = sov_nuclear_bombs_action_reservation_active
			NOT = { has_country_flag = sov_nuclear_bombs_native_delivery_pending }
			sov_nuclear_bombs_action_reservation_is_valid = yes
		}
		if = {
			limit = { check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_assigned compare = equals } }
			add_to_variable = { global.sov_nuclear_bombs_ledger_assigned = constant:sov_nuclear_bombs_ledger.one }
			subtract_from_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { sov_nuclear_bombs_actor_assigned = constant:sov_nuclear_bombs_ledger.one }
			subtract_from_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
			if = {
				limit = { has_variable = sov_nuclear_bombs_action_source_state }
				var:sov_nuclear_bombs_action_source_state = {
					if = { limit = { sov_nuclear_bombs_state_is_registered = yes } add_to_variable = { sov_nuclear_bombs_site_assigned = constant:sov_nuclear_bombs_ledger.one } subtract_from_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.one } }

			}
		}

			limit = { check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_transferred compare = equals } }
			add_to_variable = { global.sov_nuclear_bombs_ledger_transferred = constant:sov_nuclear_bombs_ledger.one }
			subtract_from_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }

				limit = { has_variable = sov_nuclear_bombs_action_reservation_holder }
				var:sov_nuclear_bombs_action_reservation_holder = {
					sov_nuclear_bombs_initialize_actor_ledger = yes
					add_to_variable = { sov_nuclear_bombs_actor_transferred = constant:sov_nuclear_bombs_ledger.one }
					subtract_from_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
				}
			}
			else = {
				add_to_variable = { sov_nuclear_bombs_actor_transferred = constant:sov_nuclear_bombs_ledger.one }
				subtract_from_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
			}
		if = {
			limit = { has_variable = sov_nuclear_bombs_action_source_state }
			var:sov_nuclear_bombs_action_source_state = {
				if = { limit = { sov_nuclear_bombs_state_is_registered = yes } add_to_variable = { sov_nuclear_bombs_site_transferred = constant:sov_nuclear_bombs_ledger.one } subtract_from_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.one } }

# Country scope. A validated demolition or non-yield test may commit, or an
# exact native on_nuke_drop confirmation may commit a detonation. The native
# callback is the only detonation commit boundary; this helper applies no
# direct Chaos, Fallout, death, contamination, or condemnation consequence.
sov_nuclear_bombs_commit_reserved_device = {
	sov_nuclear_bombs_initialize_ledger = yes
	sov_nuclear_bombs_initialize_actor_ledger = yes
	if = {
		limit = {
			OR = {
				sov_nuclear_bombs_action_can_commit = yes
				sov_nuclear_bombs_native_delivery_country_is_valid = yes
			}
		}
		if = {
			limit = {
				sov_nuclear_bombs_action_test_non_yield_can_consume = yes
				NOT = { check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_transferred compare = equals } }
			}
			# Non-detonating tests and preparation accidents do not reach the native
			# launch effect, so spend one native device at the ledger commit boundary.
			add_nuclear_bombs = constant:sov_nuclear_bombs_action.native_bomb_spend
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_assigned compare = equals }
					check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_transferred compare = equals }
				}
			}
			if = {
				limit = { has_variable = sov_nuclear_bombs_action_source_state }
				var:sov_nuclear_bombs_action_source_state = {
					if = {
						limit = { sov_nuclear_bombs_state_is_registered = yes }
						subtract_from_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.one }
						if = {
							limit = { ROOT = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } } }
							add_to_variable = { sov_nuclear_bombs_site_dismantled = constant:sov_nuclear_bombs_ledger.one }
						}
						else = { add_to_variable = { sov_nuclear_bombs_site_expended = constant:sov_nuclear_bombs_ledger.one } }
						}
					}
				}
			}
			else_if = {
				limit = {
					check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals }
					has_variable = sov_nuclear_bombs_action_source_state
					var:sov_nuclear_bombs_action_source_state = {
						sov_nuclear_bombs_state_is_registered = yes
						check_variable = { var = sov_nuclear_bombs_site_operational value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
					}
				}
				var:sov_nuclear_bombs_action_source_state = {
					subtract_from_variable = { sov_nuclear_bombs_site_operational = constant:sov_nuclear_bombs_ledger.one }
					add_to_variable = { sov_nuclear_bombs_site_dismantled = constant:sov_nuclear_bombs_ledger.one }
				}
			}
			else = {
				# Operational custody was already moved into the reserved bucket by
				# sov_nuclear_bombs_reserve_device_for_action.
		}
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }
		if = {
			limit = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } }
			# The source-specific branch below removes one native bomb for an
			# assigned or operational demolition. A transferred device is physical
			# custody only and therefore never changes the Soviet native stockpile.
			add_to_variable = { global.sov_nuclear_bombs_ledger_dismantled = constant:sov_nuclear_bombs_ledger.one }
		}
		else = { add_to_variable = { global.sov_nuclear_bombs_ledger_expended = constant:sov_nuclear_bombs_ledger.one } }
		if = {
			limit = { has_variable = sov_nuclear_bombs_action_reservation_holder }
			var:sov_nuclear_bombs_action_reservation_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				subtract_from_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
				if = {
					limit = { check_variable = { var = ROOT.sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } }
					add_to_variable = { sov_nuclear_bombs_actor_dismantled = constant:sov_nuclear_bombs_ledger.one }
				}
				else = { add_to_variable = { sov_nuclear_bombs_actor_expended = constant:sov_nuclear_bombs_ledger.one } }
			}
		}
		else = {
			subtract_from_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
			if = {
				limit = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } }
				add_to_variable = { sov_nuclear_bombs_actor_dismantled = constant:sov_nuclear_bombs_ledger.one }
			}
			else = { add_to_variable = { sov_nuclear_bombs_actor_expended = constant:sov_nuclear_bombs_ledger.one } }
		}
		if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals }
				NOT = { check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_transferred compare = equals } }
			}
			# Assigned and operational devices are represented in the native
			# Soviet stockpile, while a transferred device is only physical
			# custody until it is recovered or dismantled.
			add_nuclear_bombs = constant:sov_nuclear_bombs_action.native_bomb_spend
		}
		add_to_variable = { global.sov_nuclear_bombs_ledger_transaction_sequence = constant:sov_nuclear_bombs_ledger.one }
		set_variable = { sov_nuclear_bombs_action_transaction_sequence = global.sov_nuclear_bombs_ledger_transaction_sequence }
		clr_country_flag = sov_nuclear_bombs_action_reservation_active
		set_variable = { sov_nuclear_bombs_action_reservation_source = constant:sov_nuclear_bombs_ledger.zero }
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# State scope. The shared action contract has already moved exactly one
# reserved device into the site's dismantled bucket. This helper only closes
# the state when no intact device remains; it never consumes another device or
# applies a second ledger transaction.
sov_nuclear_bombs_finalize_one_device_dismantlement = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
			NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
			check_variable = { var = sov_nuclear_bombs_site_operational value = constant:sov_nuclear_bombs_ledger.zero compare = equals }
			check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.zero compare = equals }
			check_variable = { var = sov_nuclear_bombs_site_reserved value = constant:sov_nuclear_bombs_ledger.zero compare = equals }
			check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.zero compare = equals }
		}
		clear_variable = sov_nuclear_bombs_site_holder
		set_state_flag = sov_nuclear_bombs_site_dismantled
		clr_state_flag = sov_nuclear_bombs_site_missing
		set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.destroyed_dismantled }
		set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.reconciled }
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# Country scope. The Event 023 caller must set the action context and reserve a
# device first. Detonating actions issue the installed native route and hold
# their reservation until the documented on_nuke_drop callback confirms the
# exact state. Non-yield tests and accidents commit synchronously; an accident
# then calls the shared non-yield consequence endpoint once. No second native
# detonation or terminal Fallout consequence is applied here.
sov_nuclear_bombs_execute_shared_action = {
	sov_nuclear_bombs_initialize_ledger = yes
	sov_nuclear_bombs_initialize_actor_ledger = yes
	clr_country_flag = sov_nuclear_bombs_last_action_accepted
	clr_country_flag = sov_nuclear_bombs_last_action_rejected
	if = {
		limit = { NOT = { has_country_flag = sov_nuclear_bombs_native_delivery_pending } }
		clr_country_flag = sov_nuclear_bombs_shared_test_context
		clr_country_flag = sov_nuclear_bombs_shared_partial_yield_context
		clr_country_flag = sov_nuclear_bombs_shared_accident_context
		clr_country_flag = sov_nuclear_bombs_shared_demonstration_context
	}
	if = {
		limit = { has_country_flag = sov_nuclear_bombs_native_delivery_pending }
		# An unresolved native delivery cannot be refunded or replaced.
		set_country_flag = sov_nuclear_bombs_last_action_rejected
		set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.rejected }
		set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.delivery_unresolved }
	}
	else_if = {
		limit = { sov_nuclear_bombs_actor_can_execute_shared_action = yes }
		if = {
			limit = { sov_nuclear_bombs_action_test_non_yield_can_consume = yes }
			sov_nuclear_bombs_commit_reserved_device = yes
			if = {
				limit = { NOT = { has_country_flag = sov_nuclear_bombs_action_reservation_active } }
				set_country_flag = sov_nuclear_bombs_last_action_accepted
				set_variable = { sov_nuclear_bombs_last_action_nonce = sov_nuclear_bombs_action_nonce }
				set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.accepted }
				set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_ledger.zero }
				if = {
					limit = {
						check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.test compare = equals }
						check_variable = { var = sov_nuclear_bombs_action_outcome value = constant:sov_nuclear_bombs_tuning.test_result_accident compare = equals }
						has_country_flag = sov_nuclear_bombs_action_non_detonating
					}
					chaos_meter_on_non_yield_test_accident = yes
				}
			}
		}
		else_if = {
			limit = { sov_nuclear_bombs_action_can_commit = yes }
			sov_nuclear_bombs_commit_reserved_device = yes
			if = {
				limit = { NOT = { has_country_flag = sov_nuclear_bombs_action_reservation_active } }
				set_country_flag = sov_nuclear_bombs_last_action_accepted
				set_variable = { sov_nuclear_bombs_last_action_nonce = sov_nuclear_bombs_action_nonce }
				set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.accepted }
				set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_ledger.zero }
			}
		}
		else_if = {
			limit = {
				sov_nuclear_bombs_native_delivery_request_is_valid = yes
			}
			if = {
				limit = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.test compare = equals } }
				set_country_flag = sov_nuclear_bombs_shared_test_context
				if = {
					limit = { check_variable = { var = sov_nuclear_bombs_action_outcome value = constant:sov_nuclear_bombs_tuning.test_result_partial compare = equals } }
					set_country_flag = sov_nuclear_bombs_shared_partial_yield_context
				}
				if = {
					limit = { check_variable = { var = sov_nuclear_bombs_action_outcome value = constant:sov_nuclear_bombs_tuning.test_result_accident compare = equals } }
					set_country_flag = sov_nuclear_bombs_shared_accident_context
				}
			}
			else_if = { limit = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demonstration compare = equals } } set_country_flag = sov_nuclear_bombs_shared_demonstration_context }
			else_if = { limit = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.accident compare = equals } } set_country_flag = sov_nuclear_bombs_shared_accident_context }
			# launch_nuke detonates; it does not select an air wing or start a raid.
			# Event 23 keeps one pending detonation request until on_nuke_drop returns.
			# A verified real bomber-delivery adapter remains an acceptance requirement.
			set_variable = { sov_nuclear_bombs_action_native_stockpile_before = num_of_nukes }
			set_variable = { sov_nuclear_bombs_action_native_stockpile_expected_after = sov_nuclear_bombs_action_native_stockpile_before }
			subtract_from_variable = { sov_nuclear_bombs_action_native_stockpile_expected_after = constant:sov_nuclear_bombs_ledger.one }
			set_variable = { sov_nuclear_bombs_native_delivery_age_days = constant:sov_nuclear_bombs_ledger.zero }
			# The shared consequence hook consumes this context to keep the verified
			# Event 023 route atomic even when SOV independently has thermonuclear tech.
			set_country_flag = sov_nuclear_bombs_atomic_delivery_context
			set_country_flag = sov_nuclear_bombs_native_delivery_pending
			if = {
				limit = { has_variable = sov_nuclear_bombs_action_state }
				var:sov_nuclear_bombs_action_state = {
					set_state_flag = sov_nuclear_bombs_native_delivery_state_pending
					set_variable = { sov_nuclear_bombs_native_delivery_nonce = ROOT.sov_nuclear_bombs_action_nonce }
				}
			}
			launch_nuke = {
				state = var:sov_nuclear_bombs_action_state_id
				use_nuke = yes
				nuke_type = nuclear_bomb
			}
		}
		else = {
			# A detonation without a documented callback boundary is rejected before
			# launch_nuke so a failed request remains refundable.
			set_country_flag = sov_nuclear_bombs_last_action_rejected
			set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.rejected }
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.confirmation_unavailable }
			sov_nuclear_bombs_release_reserved_device = yes
		}
	}
	else = {
		set_country_flag = sov_nuclear_bombs_last_action_rejected
		if = {
			limit = { NOT = { has_global_flag = sov_nuclear_bombs_ledger_active } }
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.invalid_actor }
		}
		else_if = {
			limit = { NOT = { has_country_flag = sov_nuclear_bombs_actor_registered } }
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.invalid_actor }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_country_flag = sov_nuclear_bombs_action_actor_proven }
					NOT = { has_country_flag = sov_nuclear_bombs_action_authorized }
					NOT = { has_variable = sov_nuclear_bombs_action_type }
					NOT = { has_variable = sov_nuclear_bombs_action_weapon }
					NOT = { has_variable = sov_nuclear_bombs_action_route }
					NOT = { sov_nuclear_bombs_action_yield_profile_is_valid = yes }
					NOT = { has_variable = sov_nuclear_bombs_action_nonce }
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.missing_context }
		}
		else_if = {
			limit = {
				NOT = {
					check_variable = {
						var = sov_nuclear_bombs_action_nonce
						value = sov_nuclear_bombs_last_action_nonce
						compare = greater_than
					}
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.duplicate_nonce }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_tech = atomic_research }
					NOT = { has_tech = nukes }
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.missing_native_capability }
		}
		else_if = {
			limit = {
				AND = {
					NOT = { num_of_nukes > constant:sov_nuclear_bombs_ledger.zero }
					OR = {
						NOT = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } }
						NOT = { check_variable = { var = sov_nuclear_bombs_action_reservation_source value = constant:sov_nuclear_bombs_ledger.reservation_source_transferred compare = equals } }
					}
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.missing_stockpile }
		}
		else_if = {
			limit = { NOT = { sov_nuclear_bombs_action_target_is_valid = yes } }
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.invalid_target }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_action_weapon value = constant:sov_nuclear_bombs_action.nuclear_bomb compare = equals } }
					NOT = { has_country_flag = sov_nuclear_bombs_action_detonating }
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.unsupported_profile }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_action_route value = constant:sov_nuclear_bombs_action.native_launch_nuke compare = equals } }
					NOT = { sov_nuclear_bombs_action_delivery_route_is_valid = yes }
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.invalid_route }
		}
		else_if = {
			limit = {
				OR = {
					has_global_flag = soviet_collapse_terminal_collapse
					has_global_flag = soviet_collapse_terminal_collapse_in_progress
				}
			}
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.terminal_state }
		}
		else = {
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.reservation_missing }
		}
		set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.rejected }
		if = { limit = { has_country_flag = sov_nuclear_bombs_action_reservation_active } sov_nuclear_bombs_release_reserved_device = yes }
	}
}

# Country scope. The SOV-only maintenance hook calls this bounded recovery
# check once per day while Event 023 is active. If the native engine does not
# return the exact callback before the centralized timeout, the reservation is
# moved once into disputed/missing custody. This closes the ledger without
# refunding an uncertain device or replaying shared consequences.
sov_nuclear_bombs_reconcile_unreturned_native_delivery = {
	if = {
		limit = { has_country_flag = sov_nuclear_bombs_native_delivery_pending }
		if = {
			limit = { NOT = { has_variable = sov_nuclear_bombs_native_delivery_age_days } }
			set_variable = { sov_nuclear_bombs_native_delivery_age_days = constant:sov_nuclear_bombs_ledger.zero }
		}
		add_to_variable = { sov_nuclear_bombs_native_delivery_age_days = constant:sov_nuclear_bombs_ledger.one }
		if = {
			limit = {
				check_variable = {
					var = sov_nuclear_bombs_native_delivery_age_days
					value = constant:sov_nuclear_bombs_tuning.native_delivery_timeout_days
					compare = greater_than_or_equals
				}
			}
			sov_nuclear_bombs_commit_disputed_native_delivery = yes
		}
	}
	else = { clear_variable = sov_nuclear_bombs_native_delivery_age_days }
}

# Country scope, called only from the Event 023 branch of on_nuke_drop after
# the native engine has identified the launcher and exact struck state.
# Country scope. A native callback whose stockpile delta cannot be reconciled is
# a terminal disputed transaction. The reserved device moves once to `missing`
# so the conservation equation remains closed without claiming a successful
# Event 023 action or applying another consequence.
sov_nuclear_bombs_commit_disputed_native_delivery = {
	sov_nuclear_bombs_initialize_ledger = yes
	sov_nuclear_bombs_initialize_actor_ledger = yes
	if = {
		limit = {
			has_country_flag = sov_nuclear_bombs_native_delivery_pending
			has_country_flag = sov_nuclear_bombs_action_reservation_active
		}
		if = {
			limit = { has_variable = sov_nuclear_bombs_action_source_state }
			var:sov_nuclear_bombs_action_source_state = {
				if = {
					limit = { sov_nuclear_bombs_state_is_registered = yes }
					subtract_from_variable = { sov_nuclear_bombs_site_reserved = constant:sov_nuclear_bombs_ledger.one }
					add_to_variable = { sov_nuclear_bombs_site_missing = constant:sov_nuclear_bombs_ledger.one }
					set_state_flag = sov_nuclear_bombs_site_missing
					set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.missing }
					set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed }
				}
			}
		}
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_reserved = constant:sov_nuclear_bombs_ledger.one }
		add_to_variable = { global.sov_nuclear_bombs_ledger_missing = constant:sov_nuclear_bombs_ledger.one }
		subtract_from_variable = { sov_nuclear_bombs_actor_reserved = constant:sov_nuclear_bombs_ledger.one }
		add_to_variable = { sov_nuclear_bombs_actor_missing = constant:sov_nuclear_bombs_ledger.one }
		add_to_variable = { global.sov_nuclear_bombs_ledger_transaction_sequence = constant:sov_nuclear_bombs_ledger.one }
		set_variable = { sov_nuclear_bombs_last_action_nonce = sov_nuclear_bombs_action_nonce }
		set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.rejected }
		set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_action.delivery_unresolved }
		clr_country_flag = sov_nuclear_bombs_last_action_accepted
		set_country_flag = sov_nuclear_bombs_last_action_rejected
		set_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
		if = {
			limit = { has_variable = sov_nuclear_bombs_action_state }
			var:sov_nuclear_bombs_action_state = {
				clr_state_flag = sov_nuclear_bombs_native_delivery_state_pending
				clear_variable = sov_nuclear_bombs_native_delivery_nonce
			}
		}
		clr_country_flag = sov_nuclear_bombs_native_delivery_pending
		clear_variable = sov_nuclear_bombs_native_delivery_age_days
		clr_country_flag = sov_nuclear_bombs_action_reservation_active
		sov_nuclear_bombs_clear_action_context = yes
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

sov_nuclear_bombs_confirm_native_delivery = {
	if = {
		limit = { sov_nuclear_bombs_native_delivery_country_is_valid = yes }
		set_variable = { sov_nuclear_bombs_action_native_stockpile_after = num_of_nukes }
		if = {
			limit = { sov_nuclear_bombs_native_stockpile_delta_is_valid = yes }
			sov_nuclear_bombs_commit_reserved_device = yes
			if = {
				limit = { NOT = { has_country_flag = sov_nuclear_bombs_action_reservation_active } }
				clr_country_flag = sov_nuclear_bombs_last_action_rejected
				set_country_flag = sov_nuclear_bombs_last_action_accepted
				set_variable = { sov_nuclear_bombs_last_action_nonce = sov_nuclear_bombs_action_nonce }
			set_variable = { sov_nuclear_bombs_last_action_result = constant:sov_nuclear_bombs_action.accepted }
			set_variable = { sov_nuclear_bombs_last_action_rejection_reason = constant:sov_nuclear_bombs_ledger.zero }
			if = {
				limit = { has_variable = sov_nuclear_bombs_action_state }
				var:sov_nuclear_bombs_action_state = {
					clr_state_flag = sov_nuclear_bombs_native_delivery_state_pending
					clear_variable = sov_nuclear_bombs_native_delivery_nonce
				}
			}
			clr_country_flag = sov_nuclear_bombs_native_delivery_pending
			clear_variable = sov_nuclear_bombs_native_delivery_age_days
			set_country_flag = sov_nuclear_bombs_native_delivery_confirmed
			}
		}
		else = { sov_nuclear_bombs_commit_disputed_native_delivery = yes }
	}
}

# Country scope. Event 023 calls this on cancellation and on target/side-change
# cleanup so no stale state or target pointer survives the action context.
sov_nuclear_bombs_clear_action_context = {
	# A native pending marker is unresolved evidence, never a refundable request.
	if = {
		limit = { NOT = { has_country_flag = sov_nuclear_bombs_native_delivery_pending } }
		if = { limit = { has_country_flag = sov_nuclear_bombs_action_reservation_active } sov_nuclear_bombs_release_reserved_device = yes }
		if = {
			limit = { NOT = { has_country_flag = sov_nuclear_bombs_action_reservation_active } }
			clr_country_flag = sov_nuclear_bombs_action_actor_proven
			clr_country_flag = sov_nuclear_bombs_action_authorized
			clear_variable = sov_nuclear_bombs_action_type
			clear_variable = sov_nuclear_bombs_action_weapon
			clear_variable = sov_nuclear_bombs_action_route
			clr_country_flag = sov_nuclear_bombs_action_detonating
			clr_country_flag = sov_nuclear_bombs_action_non_detonating
			clear_variable = sov_nuclear_bombs_action_nonce
			clear_variable = sov_nuclear_bombs_action_state
			clear_variable = sov_nuclear_bombs_action_state_id
			clear_variable = sov_nuclear_bombs_action_source_state
			clear_variable = sov_nuclear_bombs_action_target
			clear_variable = sov_nuclear_bombs_action_target_country_id
			clear_variable = sov_nuclear_bombs_action_reservation_holder
			clr_country_flag = sov_nuclear_bombs_action_remote_recovery_authorized
			clear_variable = sov_nuclear_bombs_action_native_stockpile_before
			clear_variable = sov_nuclear_bombs_action_native_stockpile_after
			clear_variable = sov_nuclear_bombs_action_native_stockpile_expected_after
			clear_variable = sov_nuclear_bombs_native_delivery_age_days
			clear_variable = sov_nuclear_bombs_action_outcome
			clr_country_flag = sov_nuclear_bombs_action_reservation_active
			clr_country_flag = sov_nuclear_bombs_action_reservation_disputed
		}
	}
}

# State scope. Record one distinct breakaway holder for the collapse ledger.
# The array is historical evidence, while the current actor variables remain
# the authority for physical custody and operational status.
sov_nuclear_bombs_record_breakaway_actor_from_state = {
	if = {
		limit = {
			OR = {
				owner = { has_country_flag = soviet_collapse_breakaway }
				controller = { has_country_flag = soviet_collapse_breakaway }
			}
			NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
			NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
		}
		if = {
			limit = { controller = { has_country_flag = soviet_collapse_breakaway } }
			controller = { save_event_target_as = sov_nuclear_bombs_current_breakaway_actor }
		}
		else = { owner = { save_event_target_as = sov_nuclear_bombs_current_breakaway_actor } }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.sov_nuclear_bombs_breakaway_actor_entries
						value = event_target:sov_nuclear_bombs_current_breakaway_actor
					}
				}
			}
			add_to_array = {
				array = global.sov_nuclear_bombs_breakaway_actor_entries
				value = event_target:sov_nuclear_bombs_current_breakaway_actor
			}
			add_to_variable = { global.sov_nuclear_bombs_breakaway_actor_count = constant:sov_nuclear_bombs_ledger.one }
			if = { limit = { SOV = { exists = yes } } SOV = { add_to_variable = { sov_nuclear_bombs_breakaway_crises = constant:sov_nuclear_bombs_ledger.one } } }
		}
	}
}

# State scope. Convert a physically transferred breakaway site into a limited
# operational actor only after the Event 23 staged technical, command, and
# delivery decisions have completed. No Soviet technology, reactor grant, or
# native stockpile is copied to the breakaway.
sov_nuclear_bombs_operationalize_breakaway_site = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
			NOT = { has_state_flag = sov_nuclear_bombs_breakaway_site_operationalized }
			NOT = { has_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed }
			OR = {
				owner = { has_country_flag = soviet_collapse_breakaway }
				controller = { has_country_flag = soviet_collapse_breakaway }
			}
			# Native custody must already have been removed from SOV or a
			# prior holder before a breakaway can become operational. SOV
			# disappearance never substitutes for the native stockpile
			# transfer receipt.
			has_state_flag = sov_nuclear_bombs_native_custody_removed
			sov_nuclear_bombs_event_state_has_local_command_receipt = yes
			var:sov_nuclear_bombs_site_holder = {
				has_country_flag = sov_nuclear_bombs_breakaway_operationalization_mission_active
				check_variable = { var = sov_nuclear_bombs_breakaway_progress_stage value = constant:sov_nuclear_bombs_tuning.breakaway_delivery_stage compare = equals }
				has_variable = sov_nuclear_bombs_local_custody_state_id
				check_variable = { var = sov_nuclear_bombs_local_custody_state_id value = PREV.id compare = equals }
			}
			sov_nuclear_bombs_event_state_has_breakaway_delivery_capability = yes
		}
		set_temp_variable = { sov_nuclear_bombs_operationalized_amount = sov_nuclear_bombs_site_transferred }
		if = {
			limit = { has_variable = sov_nuclear_bombs_site_holder }
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				if = { limit = { check_variable = { var = sov_nuclear_bombs_actor_transferred value = sov_nuclear_bombs_operationalized_amount compare = greater_than_or_equals } } subtract_from_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_operationalized_amount } }
				add_to_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_operationalized_amount }
				if = {
					limit = { NOT = { has_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational } }
					set_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
					add_to_variable = { global.sov_nuclear_bombs_breakaway_operational_actor_count = constant:sov_nuclear_bombs_ledger.one }
				}
			}
		}
		else_if = {
			limit = { controller = { exists = yes NOT = { tag = SOV } } }
			set_variable = { sov_nuclear_bombs_site_holder = CONTROLLER }
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				add_to_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_operationalized_amount }
				if = {
					limit = { NOT = { has_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational } }
					set_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
					add_to_variable = { global.sov_nuclear_bombs_breakaway_operational_actor_count = constant:sov_nuclear_bombs_ledger.one }
				}
			}
		}
		else = {
			set_variable = { sov_nuclear_bombs_site_holder = OWNER }
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				add_to_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_operationalized_amount }
				if = {
					limit = { NOT = { has_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational } }
					set_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
					add_to_variable = { global.sov_nuclear_bombs_breakaway_operational_actor_count = constant:sov_nuclear_bombs_ledger.one }
				}
			}
		}
		if = {
			limit = {
				has_state_flag = sov_nuclear_bombs_native_custody_removed
				has_variable = sov_nuclear_bombs_site_holder
				var:sov_nuclear_bombs_site_holder = {
					exists = yes
					NOT = { tag = SOV }
				}
			}
			var:sov_nuclear_bombs_site_holder = {
				set_temp_variable = { sov_nuclear_bombs_native_custody_delta = sov_nuclear_bombs_operationalized_amount }
				add_nuclear_bombs = sov_nuclear_bombs_native_custody_delta
			}
			set_state_flag = sov_nuclear_bombs_native_custody_with_holder
		}
		add_to_variable = { global.sov_nuclear_bombs_ledger_operational = sov_nuclear_bombs_operationalized_amount }
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_transferred = sov_nuclear_bombs_operationalized_amount }
		set_variable = { sov_nuclear_bombs_site_operational = sov_nuclear_bombs_operationalized_amount }
		set_variable = { sov_nuclear_bombs_site_transferred = constant:sov_nuclear_bombs_ledger.zero }
		set_state_flag = sov_nuclear_bombs_breakaway_site_operationalized
		set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.breakaway_physical }
		set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.reconciled }
		sov_nuclear_bombs_record_breakaway_actor_from_state = yes
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# State scope. Move all currently intact site custody into the transferred
# bucket without granting operational access to the new holder.
sov_nuclear_bombs_mark_site_transferred = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			OR = {
				check_variable = { var = sov_nuclear_bombs_site_operational value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { sov_nuclear_bombs_transfer_operational = sov_nuclear_bombs_site_operational }
		set_temp_variable = { sov_nuclear_bombs_transfer_assigned = sov_nuclear_bombs_site_assigned }
		# A live native request remains in the reserved bucket until its exact
		# callback resolves. Moving it here would detach the callback from its
		# source site and allow a second or negative ledger commit.
		set_temp_variable = { sov_nuclear_bombs_transfer_reserved = constant:sov_nuclear_bombs_ledger.zero }
		set_temp_variable = { sov_nuclear_bombs_transfer_total = constant:sov_nuclear_bombs_ledger.zero }
		add_to_temp_variable = { sov_nuclear_bombs_transfer_total = sov_nuclear_bombs_transfer_operational }
		add_to_temp_variable = { sov_nuclear_bombs_transfer_total = sov_nuclear_bombs_transfer_assigned }
		add_to_temp_variable = { sov_nuclear_bombs_transfer_total = sov_nuclear_bombs_transfer_reserved }
		add_to_variable = { global.sov_nuclear_bombs_ledger_transferred = sov_nuclear_bombs_site_operational }
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_operational = sov_nuclear_bombs_site_operational }
		add_to_variable = { global.sov_nuclear_bombs_ledger_transferred = sov_nuclear_bombs_site_assigned }
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_assigned = sov_nuclear_bombs_site_assigned }
		add_to_variable = { sov_nuclear_bombs_site_transferred = sov_nuclear_bombs_site_operational }
		add_to_variable = { sov_nuclear_bombs_site_transferred = sov_nuclear_bombs_site_assigned }
		set_variable = { sov_nuclear_bombs_site_operational = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_assigned = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_native_transfer_amount = sov_nuclear_bombs_transfer_total }
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_native_custody_with_holder }
			if = {
				limit = {
					has_variable = sov_nuclear_bombs_native_transfer_amount
					check_variable = { var = sov_nuclear_bombs_native_transfer_amount value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
					has_variable = sov_nuclear_bombs_site_holder
					var:sov_nuclear_bombs_site_holder = {
						exists = yes
						NOT = { tag = SOV }
						NOT = { num_of_nukes < PREV.sov_nuclear_bombs_native_transfer_amount }
					}
				}
				var:sov_nuclear_bombs_site_holder = {
					set_temp_variable = { sov_nuclear_bombs_native_transfer_delta = PREV.sov_nuclear_bombs_native_transfer_amount }
					multiply_temp_variable = { sov_nuclear_bombs_native_transfer_delta = -1 }
					add_nuclear_bombs = sov_nuclear_bombs_native_transfer_delta
				}
				clr_state_flag = sov_nuclear_bombs_native_custody_with_holder
				set_state_flag = sov_nuclear_bombs_native_custody_removed
			}
			else = {
				set_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
				set_state_flag = sov_nuclear_bombs_site_native_custody_disputed
			}
		}
		else_if = {
			limit = {
				has_variable = sov_nuclear_bombs_native_transfer_amount
				check_variable = { var = sov_nuclear_bombs_native_transfer_amount value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
				SOV = {
					exists = yes
					NOT = { num_of_nukes < PREV.sov_nuclear_bombs_native_transfer_amount }
				}
			}
			SOV = {
				set_temp_variable = { sov_nuclear_bombs_native_transfer_delta = PREV.sov_nuclear_bombs_native_transfer_amount }
				multiply_temp_variable = { sov_nuclear_bombs_native_transfer_delta = -1 }
				add_nuclear_bombs = sov_nuclear_bombs_native_transfer_delta
			}
			set_state_flag = sov_nuclear_bombs_native_custody_removed
		}
		else_if = {
			limit = { SOV = { exists = yes } }
			set_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
			set_state_flag = sov_nuclear_bombs_site_native_custody_disputed
		}
		clr_state_flag = sov_nuclear_bombs_site_returned_to_soviet
		if = {
			limit = { has_variable = sov_nuclear_bombs_site_recorded_owner }
			var:sov_nuclear_bombs_site_recorded_owner = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				if = { limit = { check_variable = { var = sov_nuclear_bombs_transfer_assigned value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } } subtract_from_variable = { sov_nuclear_bombs_actor_assigned = sov_nuclear_bombs_transfer_assigned } }
				if = { limit = { check_variable = { var = sov_nuclear_bombs_transfer_reserved value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } } subtract_from_variable = { sov_nuclear_bombs_actor_reserved = sov_nuclear_bombs_transfer_reserved } }
				if = {
					limit = { PREV = { NOT = { has_state_flag = sov_nuclear_bombs_breakaway_site_operationalized } } }
					if = { limit = { check_variable = { var = sov_nuclear_bombs_transfer_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } } subtract_from_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_transfer_operational } }
					add_to_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_transfer_total }
				}
				else = { add_to_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_transfer_assigned } add_to_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_transfer_reserved } }
			}
		}
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_breakaway_site_operationalized has_variable = sov_nuclear_bombs_site_holder }
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				if = { limit = { check_variable = { var = sov_nuclear_bombs_transfer_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } } subtract_from_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_transfer_operational } }
				add_to_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_transfer_operational }
				if = {
					limit = {
						has_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
						NOT = { check_variable = { var = sov_nuclear_bombs_actor_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } }
					}
					clr_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
					if = {
						limit = {
							check_variable = {
								var = global.sov_nuclear_bombs_breakaway_operational_actor_count
								value = constant:sov_nuclear_bombs_ledger.zero
								compare = greater_than
							}
						}
						subtract_from_variable = { global.sov_nuclear_bombs_breakaway_operational_actor_count = constant:sov_nuclear_bombs_ledger.one }
					}
				}
			}
		}
		set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed }
		if = { limit = { NOT = { has_variable = sov_nuclear_bombs_site_holder } has_variable = sov_nuclear_bombs_site_recorded_owner } set_variable = { sov_nuclear_bombs_site_holder = sov_nuclear_bombs_site_recorded_owner } }
		# Operational attribution above must see the old holder's live marker.
		# Once custody is transferred, a successor starts from physical custody,
		# while this separate history flag preserves the site's prior status.
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_breakaway_site_operationalized }
			set_state_flag = sov_nuclear_bombs_site_previously_operationalized
			clr_state_flag = sov_nuclear_bombs_breakaway_site_operationalized
		}
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
}

# State scope. This marker belongs only to the Event 005 release bridge. An
# ordinary foreign control change must not create a collapse episode or satisfy
# the scattered-arsenal achievement by passing through the shared transfer

	if = {

	}
}

# State scope. Rebind the transferred ledger to the current owner/controller
# after Event 005 has completed its state change. Physical custody follows this
# pointer, but the new holder still has no technical, command, or delivery
# capability until its own staged conditions are met.
sov_nuclear_bombs_rebind_event5_site_holder = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
		}
		set_temp_variable = { sov_nuclear_bombs_holder_devices = sov_nuclear_bombs_site_transferred }
		set_temp_variable = { sov_nuclear_bombs_previous_holder = constant:sov_nuclear_bombs_ledger.zero }
		if = {
			limit = { has_variable = sov_nuclear_bombs_site_holder }
			set_temp_variable = { sov_nuclear_bombs_previous_holder = sov_nuclear_bombs_site_holder }
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				if = {
					limit = { check_variable = { var = sov_nuclear_bombs_actor_transferred value = sov_nuclear_bombs_holder_devices compare = greater_than_or_equals } }
					subtract_from_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_holder_devices }
				}
			}
		}
		clear_variable = sov_nuclear_bombs_site_holder
		if = {
			limit = { CONTROLLER = { exists = yes NOT = { tag = SOV } } }
			set_variable = { sov_nuclear_bombs_site_holder = CONTROLLER }
		}
		else_if = {
			limit = { OWNER = { exists = yes NOT = { tag = SOV } } }
			set_variable = { sov_nuclear_bombs_site_holder = OWNER }
		}
		if = {
			limit = { has_variable = sov_nuclear_bombs_site_holder }
			# Reclassify only a real holder change. A same-holder control
			# callback must not dissolve a surviving joint-custody agreement.
			if = {
				limit = { NOT = { check_variable = { var = sov_nuclear_bombs_site_holder value = sov_nuclear_bombs_previous_holder compare = equals } } }
				if = {
					limit = { var:sov_nuclear_bombs_site_holder = { has_country_flag = soviet_collapse_breakaway } }
					set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.breakaway_physical }
				}
				else = { set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.foreign } }
			}
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				add_to_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_holder_devices }
			}
		}
	}
}

# State scope. Return physical devices and their ledger custody to SOV without
# changing the state's territorial owner. This is the concrete return-device
# settlement path; it is also reused by collapse recovery so native stockpile
# movement and actor attribution cannot diverge between the two routes.
sov_nuclear_bombs_return_site_custody_to_soviet = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			NOT = { has_state_flag = sov_nuclear_bombs_site_native_custody_disputed }
			OR = {
				check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
				check_variable = { var = sov_nuclear_bombs_site_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
			}
			NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
			NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
			SOV = { exists = yes }
			OR = {
				NOT = { has_state_flag = sov_nuclear_bombs_native_custody_with_holder }
				AND = {
					has_state_flag = sov_nuclear_bombs_native_custody_with_holder
					has_variable = sov_nuclear_bombs_site_holder
					has_variable = sov_nuclear_bombs_native_transfer_amount
					check_variable = { var = sov_nuclear_bombs_native_transfer_amount value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
					var:sov_nuclear_bombs_site_holder = {
						exists = yes
						NOT = { num_of_nukes < PREV.sov_nuclear_bombs_native_transfer_amount }
					}
				}
			}
		}
		set_temp_variable = { sov_nuclear_bombs_return_operational = sov_nuclear_bombs_site_operational }
		set_temp_variable = { sov_nuclear_bombs_return_transferred = sov_nuclear_bombs_site_transferred }
		set_temp_variable = { sov_nuclear_bombs_return_total = constant:sov_nuclear_bombs_ledger.zero }
		add_to_temp_variable = { sov_nuclear_bombs_return_total = sov_nuclear_bombs_return_operational }
		add_to_temp_variable = { sov_nuclear_bombs_return_total = sov_nuclear_bombs_return_transferred }
		add_to_variable = { global.sov_nuclear_bombs_ledger_assigned = sov_nuclear_bombs_return_total }
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_operational = sov_nuclear_bombs_return_operational }
		subtract_from_variable = { global.sov_nuclear_bombs_ledger_transferred = sov_nuclear_bombs_return_transferred }
		if = {
			limit = { has_variable = sov_nuclear_bombs_site_holder }
			var:sov_nuclear_bombs_site_holder = {
				sov_nuclear_bombs_initialize_actor_ledger = yes
				if = { limit = { check_variable = { var = sov_nuclear_bombs_return_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } } subtract_from_variable = { sov_nuclear_bombs_actor_operational = sov_nuclear_bombs_return_operational } }
				if = { limit = { check_variable = { var = sov_nuclear_bombs_return_transferred value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } } subtract_from_variable = { sov_nuclear_bombs_actor_transferred = sov_nuclear_bombs_return_transferred } }
				if = {
					limit = {
						has_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
						NOT = { check_variable = { var = sov_nuclear_bombs_actor_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } }
					}
					clr_country_flag = sov_nuclear_bombs_breakaway_nuclear_operational
					if = {
						limit = { check_variable = { var = global.sov_nuclear_bombs_breakaway_operational_actor_count value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } }
						subtract_from_variable = { global.sov_nuclear_bombs_breakaway_operational_actor_count = constant:sov_nuclear_bombs_ledger.one }
					}
				}
			}
		}
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_native_custody_with_holder }
			var:sov_nuclear_bombs_site_holder = {
				set_temp_variable = { sov_nuclear_bombs_native_return_delta = sov_nuclear_bombs_return_total }
				multiply_temp_variable = { sov_nuclear_bombs_native_return_delta = -1 }
				add_nuclear_bombs = sov_nuclear_bombs_native_return_delta
			}
		}
		SOV = { add_nuclear_bombs = sov_nuclear_bombs_return_total }
		add_to_variable = { sov_nuclear_bombs_site_assigned = sov_nuclear_bombs_return_total }
		set_variable = { sov_nuclear_bombs_site_operational = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_transferred = constant:sov_nuclear_bombs_ledger.zero }
		set_variable = { sov_nuclear_bombs_site_holder = SOV }
		set_state_flag = sov_nuclear_bombs_site_returned_to_soviet
		clr_state_flag = sov_nuclear_bombs_site_joint_custody
		clr_state_flag = sov_nuclear_bombs_breakaway_site_operationalized
		clr_state_flag = sov_nuclear_bombs_native_custody_removed
		clr_state_flag = sov_nuclear_bombs_native_custody_with_holder
		set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.central_soviet }
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_site_native_custody_disputed }
			set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed }
		}
		else = { set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.reconciled } }
		clear_variable = sov_nuclear_bombs_native_transfer_amount
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
	else_if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			has_state_flag = sov_nuclear_bombs_native_custody_with_holder
		}
		set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed }
		set_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
	}
}

# State scope. Return a transferred site to Soviet assigned custody only after
# an explicit recovery/negotiation action and a current SOV owner or controller
# proof. This never grants a breakaway technical or delivery capability.
sov_nuclear_bombs_restore_site_to_soviet = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			OR = {
				check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
				AND = {
					has_state_flag = sov_nuclear_bombs_breakaway_site_operationalized
					check_variable = { var = sov_nuclear_bombs_site_operational value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
				}
			}
			OR = {
				is_owned_by = SOV
				is_controlled_by = SOV
			}
		}
		sov_nuclear_bombs_return_site_custody_to_soviet = yes
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_site_returned_to_soviet }
			set_variable = { sov_nuclear_bombs_site_recorded_owner = SOV }
			set_variable = { sov_nuclear_bombs_site_recorded_controller = SOV }
			clr_state_flag = sov_nuclear_bombs_site_technical_denial
			clr_state_flag = sov_nuclear_bombs_return_requested
		}
	}
}

# State scope. This is a terminal physical disposition and is idempotent after
# the current intact buckets have been moved.
sov_nuclear_bombs_mark_site_dismantled = {
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes

		set_temp_variable = { sov_nuclear_bombs_disposition_reserved = sov_nuclear_bombs_site_reserved }

		set_temp_variable = { sov_nuclear_bombs_disposition_total = constant:sov_nuclear_bombs_ledger.zero }

# Country scope. Close only the matching site mirror; retain the historical
# facility and closed nonce, then remove the active country pointer and target.
sov_nuclear_bombs_close_assembly_receipt = {
	if = {
		limit = {
			has_variable = sov_nuclear_bombs_assembly_state
			has_variable = sov_nuclear_bombs_assembly_state_id
			has_variable = sov_nuclear_bombs_assembly_nonce
			var:sov_nuclear_bombs_assembly_state = {
				check_variable = { var = THIS.id value = PREV.sov_nuclear_bombs_assembly_state_id compare = equals }
				OR = {
					AND = { has_state_flag = sov_nuclear_bombs_assembly_batch_active check_variable = { var = sov_nuclear_bombs_assembly_batch_nonce value = PREV.sov_nuclear_bombs_assembly_nonce compare = equals } }
					AND = { NOT = { has_state_flag = sov_nuclear_bombs_assembly_batch_active } has_variable = sov_nuclear_bombs_assembly_closed_nonce check_variable = { var = sov_nuclear_bombs_assembly_closed_nonce value = PREV.sov_nuclear_bombs_assembly_nonce compare = equals } }
				}
			}
		}
		var:sov_nuclear_bombs_assembly_state = {
			if = {
				limit = { has_state_flag = sov_nuclear_bombs_assembly_batch_active check_variable = { var = sov_nuclear_bombs_assembly_batch_nonce value = PREV.sov_nuclear_bombs_assembly_nonce compare = equals } }
				set_variable = { sov_nuclear_bombs_assembly_closed_nonce = sov_nuclear_bombs_assembly_batch_nonce }
				clr_state_flag = sov_nuclear_bombs_assembly_batch_active
				clear_variable = sov_nuclear_bombs_assembly_batch_nonce
			}
		}
		set_variable = { sov_nuclear_bombs_assembly_closed_nonce = sov_nuclear_bombs_assembly_nonce }
		clear_variable = sov_nuclear_bombs_assembly_state
		clear_variable = sov_nuclear_bombs_assembly_state_id
		clear_variable = sov_nuclear_bombs_assembly_nonce
		clear_global_event_target = sov_nuclear_bombs_assembly_state
		clr_country_flag = sov_nuclear_bombs_assembly_receipt_disputed
	}
	else_if = {
		limit = { OR = { has_variable = sov_nuclear_bombs_assembly_state has_variable = sov_nuclear_bombs_assembly_state_id has_variable = sov_nuclear_bombs_assembly_nonce } }
		# Preserve the only cleanup pointer on a mismatch. No closed receipt or
		# new batch may replace an unresolved state/actor mirror.
		set_country_flag = sov_nuclear_bombs_assembly_receipt_disputed
	}
}

# Source-state scope. Cancellation or an Event 005 endpoint loss returns one
# device to secure SOV custody, reclassifies captured physical custody through
# the existing native-aware transfer helper, or records an unresolved loss.
sov_nuclear_bombs_settle_transit_source = {
	if = {
		limit = { has_state_flag = sov_nuclear_bombs_transit_source_active check_variable = { var = sov_nuclear_bombs_site_in_transit value = constant:sov_nuclear_bombs_ledger.one compare = equals } }
		add_to_variable = { sov_nuclear_bombs_site_assigned = constant:sov_nuclear_bombs_ledger.one }
		if = {
			limit = { OR = { has_state_flag = sov_nuclear_bombs_site_missing has_state_flag = sov_nuclear_bombs_site_dismantled NOT = { OR = { owner = { exists = yes } controller = { exists = yes } } } } }
			subtract_from_variable = { sov_nuclear_bombs_site_assigned = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { sov_nuclear_bombs_site_missing = constant:sov_nuclear_bombs_ledger.one }
			subtract_from_variable = { global.sov_nuclear_bombs_ledger_assigned = constant:sov_nuclear_bombs_ledger.one }
			add_to_variable = { global.sov_nuclear_bombs_ledger_missing = constant:sov_nuclear_bombs_ledger.one }
			if = { limit = { SOV = { exists = yes } } SOV = { subtract_from_variable = { sov_nuclear_bombs_actor_assigned = constant:sov_nuclear_bombs_ledger.one } add_to_variable = { sov_nuclear_bombs_actor_missing = constant:sov_nuclear_bombs_ledger.one } } }
			set_state_flag = sov_nuclear_bombs_site_native_custody_disputed
			set_state_flag = sov_nuclear_bombs_transit_lost
		}
		else_if = {
			limit = { is_owned_by = SOV is_controlled_by = SOV }
			set_state_flag = sov_nuclear_bombs_transit_returned
		}
		else = {
			sov_nuclear_bombs_mark_site_transferred = yes
			sov_nuclear_bombs_rebind_event5_site_holder = yes
			set_state_flag = sov_nuclear_bombs_transit_captured
		}
		sov_nuclear_bombs_close_transit_receipt = yes
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
		sov_nuclear_bombs_refresh_transit_registry = yes
	}
}

# State scope, observed after an Event 005 transfer. The cancellation path

sov_nuclear_bombs_reconcile_dismantlement_state_change = {
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_active_dismantlement_site
			NOT = { AND = { is_owned_by = SOV is_controlled_by = SOV } }
			SOV = { exists = yes has_country_flag = sov_nuclear_bombs_dismantlement_registry_active }
		}
		SOV = { sov_nuclear_bombs_cancel_dismantlement = yes }
		if = {
			limit = { check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals } OR = { owner = { exists = yes } controller = { exists = yes } } }
			sov_nuclear_bombs_mark_site_transferred = yes
			sov_nuclear_bombs_rebind_event5_site_holder = yes
		}
		else_if = { limit = { check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals } } sov_nuclear_bombs_mark_site_missing = yes }
	}
	else_if = {
		limit = { has_state_flag = sov_nuclear_bombs_active_dismantlement_site SOV = { exists = no } }
		set_state_flag = sov_nuclear_bombs_site_native_custody_disputed
		set_global_flag = sov_nuclear_bombs_native_custody_reconciliation_disputed
	}
}

# Exact state scope. Territorial change marks existing roles contested but
# never transfers command, institutional access, or delivery capability.
sov_nuclear_bombs_reconcile_registry_state_change = {
	if = {
		limit = { has_state_flag = sov_nuclear_bombs_assembly_site NOT = { AND = { is_owned_by = SOV is_controlled_by = SOV } } }
		set_state_flag = sov_nuclear_bombs_assembly_site_contested
		if = {
			limit = { has_state_flag = sov_nuclear_bombs_assembly_batch_active SOV = { has_country_flag = sov_nuclear_bombs_production_mission_active has_variable = sov_nuclear_bombs_assembly_state_id check_variable = { var = sov_nuclear_bombs_assembly_state_id value = PREV.id compare = equals } } }
			SOV = { sov_nuclear_bombs_cancel_production_mission = yes }
		}
	}
	if = {
		limit = {
			OR = { has_state_flag = sov_nuclear_bombs_central_command_site has_state_flag = sov_nuclear_bombs_backup_command_site has_state_flag = sov_nuclear_bombs_production_reactor_site }
			NOT = { AND = { is_owned_by = SOV is_controlled_by = SOV } }
		}
		set_state_flag = sov_nuclear_bombs_soviet_registry_site_contested
	}
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_backup_command_site
			NOT = { AND = { is_owned_by = SOV is_controlled_by = SOV } }
			SOV = {
				exists = yes
				has_variable = sov_nuclear_bombs_backup_command_state
				var:sov_nuclear_bombs_backup_command_state = {
					check_variable = { var = THIS.id value = PREV.PREV.id compare = equals }
				}
			}
		}
		SOV = { clear_variable = sov_nuclear_bombs_backup_command_state }
	}
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_institutional_access_site
			OR = {
				NOT = { has_variable = sov_nuclear_bombs_site_holder }
				NOT = { check_variable = { var = sov_nuclear_bombs_site_holder value = sov_nuclear_bombs_institutional_access_actor compare = equals } }
			}
		}
		set_state_flag = sov_nuclear_bombs_institutional_access_contested
	}
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_breakaway_custody_secured
			OR = {
				NOT = { has_variable = sov_nuclear_bombs_site_holder }
				NOT = { has_variable = sov_nuclear_bombs_local_custody_actor }
				NOT = { check_variable = { var = sov_nuclear_bombs_site_holder value = sov_nuclear_bombs_local_custody_actor compare = equals } }
			}
		}
		clr_state_flag = sov_nuclear_bombs_breakaway_custody_secured
	}
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_breakaway_command_site
			OR = {
				NOT = { has_variable = sov_nuclear_bombs_site_holder }
				NOT = { check_variable = { var = sov_nuclear_bombs_site_holder value = sov_nuclear_bombs_breakaway_command_actor compare = equals } }
			}
		}
		set_state_flag = sov_nuclear_bombs_breakaway_command_site_contested
	}
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_delivery_force_site
			OR = {
				NOT = { has_variable = sov_nuclear_bombs_site_holder }
				NOT = { check_variable = { var = sov_nuclear_bombs_site_holder value = sov_nuclear_bombs_site_delivery_force_owner compare = equals } }
			}
		}
		set_state_flag = sov_nuclear_bombs_delivery_force_site_contested
	}
}

# Country scope before an Event 005 release operation.
# The release target is the existing soviet_collapse_dynamic_release_target
# event target. Vanilla release transfers only states owned by this host;
# the state helper filters that exact set to represented Event 23 context.

	if = {
		limit = {
			sov_nuclear_bombs_event5_bridge_is_active = yes

		}
		add_to_variable = { global.sov_nuclear_bombs_event5_snapshot_generation = constant:sov_nuclear_bombs_ledger.one }

			limit = {

			sov_nuclear_bombs_snapshot_event5_state = yes
		}
	}
}

# State scope. This stores exact owner/controller pointers, current site
# buckets where registered, and the represented wider Event 23 context before
# Event 005 changes the state. It never operationalizes a new holder.
sov_nuclear_bombs_snapshot_event5_state = {
	if = {
		limit = {
			NOT = { sov_nuclear_bombs_state_has_pending_event5_snapshot = yes }
			OR = {
				sov_nuclear_bombs_state_is_registered = yes
				has_state_flag = sov_nuclear_bombs_reactor_site_pending
				has_state_flag = sov_nuclear_bombs_reactor_event_owned

				has_state_flag = sov_nuclear_bombs_delivery_force_site

				has_state_flag = sov_nuclear_bombs_transit_destination_pending

				has_state_flag = sov_nuclear_bombs_disperse_source
				has_state_flag = sov_nuclear_bombs_disperse_destination

				AND = { has_event_target = sov_nuclear_bombs_test_state check_variable = { var = THIS.id value = event_target:sov_nuclear_bombs_test_state.id compare = equals } }
				AND = { has_event_target = sov_nuclear_bombs_selected_test_state check_variable = { var = THIS.id value = event_target:sov_nuclear_bombs_selected_test_state.id compare = equals } }
				AND = { has_event_target = sov_nuclear_bombs_selected_state check_variable = { var = THIS.id value = event_target:sov_nuclear_bombs_selected_state.id compare = equals } }
				AND = { has_event_target = sov_nuclear_bombs_pending_reactor_state check_variable = { var = THIS.id value = event_target:sov_nuclear_bombs_pending_reactor_state.id compare = equals } }
				AND = { has_event_target = sov_nuclear_bombs_disperse_source_state check_variable = { var = THIS.id value = event_target:sov_nuclear_bombs_disperse_source_state.id compare = equals } }
			}

			set_variable = { sov_nuclear_bombs_site_snapshot_operational = sov_nuclear_bombs_site_operational }

# State scope, called after the Event 005 command or by a later callback.
# Current owner and controller are read after the in-progress marker clears and
# classified without granting technical, command, or delivery access.
sov_nuclear_bombs_reconcile_event5_state = {
	if = {
		limit = {
			sov_nuclear_bombs_state_has_pending_event5_snapshot = yes
			NOT = { has_state_flag = sov_nuclear_bombs_event5_transfer_in_progress }
		}
		if = {
			limit = {
				sov_nuclear_bombs_state_is_registered = yes
				NOT = {
					AND = {
						check_variable = { var = sov_nuclear_bombs_site_operational value = sov_nuclear_bombs_site_snapshot_operational compare = equals }
						check_variable = { var = sov_nuclear_bombs_site_assigned value = sov_nuclear_bombs_site_snapshot_assigned compare = equals }
						check_variable = { var = sov_nuclear_bombs_site_reserved value = sov_nuclear_bombs_site_snapshot_reserved compare = equals }
						check_variable = { var = sov_nuclear_bombs_site_transferred value = sov_nuclear_bombs_site_snapshot_transferred compare = equals }
						check_variable = { var = sov_nuclear_bombs_site_dismantled value = sov_nuclear_bombs_site_snapshot_dismantled compare = equals }
						check_variable = { var = sov_nuclear_bombs_site_missing value = sov_nuclear_bombs_site_snapshot_missing compare = equals }
					}
				}
			}
			set_state_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
			set_global_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
		}
		sov_nuclear_bombs_reconcile_dismantlement_state_change = yes
		sov_nuclear_bombs_reconcile_transit_state_change = yes
		if = {
			limit = { sov_nuclear_bombs_state_is_registered = yes }
			if = {
				limit = {
					NOT = { OR = { owner = { exists = yes } controller = { exists = yes } } }
					NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
					NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
					NOT = { has_state_flag = sov_nuclear_bombs_native_delivery_state_pending }
				}
				sov_nuclear_bombs_mark_site_missing = yes
			}
			else = {
				if = {
					limit = {
						NOT = {
							OR = {
								is_owned_by = SOV
								is_controlled_by = SOV
							}
						}
						NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
						NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
					}
					sov_nuclear_bombs_mark_site_transferred = yes
					sov_nuclear_bombs_record_event5_custody_episode = yes
				}
				if = {
					limit = {
						NOT = {
							OR = {
								is_owned_by = SOV
								is_controlled_by = SOV
							}
						}
						NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
						NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
					}
					sov_nuclear_bombs_rebind_event5_site_holder = yes
				}
			}
			if = { limit = { OWNER = { exists = yes } } set_variable = { sov_nuclear_bombs_site_recorded_owner = OWNER } else = { clear_variable = sov_nuclear_bombs_site_recorded_owner } }
			if = { limit = { CONTROLLER = { exists = yes } } set_variable = { sov_nuclear_bombs_site_recorded_controller = CONTROLLER } else = { clear_variable = sov_nuclear_bombs_site_recorded_controller } }
			if = {
				limit = { has_state_flag = sov_nuclear_bombs_site_dismantled }
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.destroyed_dismantled }
			}
			else_if = {
				limit = { has_state_flag = sov_nuclear_bombs_site_missing }
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.missing }
			}
			else_if = {
				limit = { has_state_flag = sov_nuclear_bombs_site_technical_denial }
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.technical_denial }
			}
			else_if = {
				limit = { is_owned_by = SOV is_controlled_by = SOV }
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.central_soviet }
			}
			else_if = {
				limit = {
					OR = {
						is_owned_by = SOV
						is_controlled_by = SOV
					}
				}
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.soviet_siege }
			}
			else_if = {
				limit = {
					OR = {
						owner = { has_country_flag = soviet_collapse_breakaway }
						controller = { has_country_flag = soviet_collapse_breakaway }
					}
				}
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.breakaway_physical }
			}
			else_if = {
				limit = { OR = { owner = { exists = yes } controller = { exists = yes } } }
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.foreign }
			}
			else = {
				set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.missing }
			}
			if = {
				limit = {
					OR = {
						owner = { has_country_flag = soviet_collapse_breakaway }
						controller = { has_country_flag = soviet_collapse_breakaway }
					}
				}
				sov_nuclear_bombs_record_breakaway_actor_from_state = yes
			}
			if = {
				limit = {
					OR = {
						has_state_flag = sov_nuclear_bombs_site_native_custody_disputed
						has_state_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
					}
				}
				set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed }
			}
			else = { set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.reconciled } }
		}
		set_variable = { sov_nuclear_bombs_site_reconciliation_generation = sov_nuclear_bombs_site_snapshot_generation }
		clr_state_flag = sov_nuclear_bombs_event5_snapshot_pending
		if = {
			limit = { check_variable = { var = global.sov_nuclear_bombs_event5_snapshot_state_count value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } }
			subtract_from_variable = { global.sov_nuclear_bombs_event5_snapshot_state_count = constant:sov_nuclear_bombs_ledger.one }
		}
		if = {
			limit = { check_variable = { var = global.sov_nuclear_bombs_event5_snapshot_state_count value = constant:sov_nuclear_bombs_ledger.zero compare = equals } }
			clr_global_flag = sov_nuclear_bombs_event5_reconciliation_pending
		}
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
		sov_nuclear_bombs_refresh_transit_registry = yes
		sov_nuclear_bombs_refresh_dismantlement_registry = yes
		sov_nuclear_bombs_reconcile_registry_state_change = yes
		sov_nuclear_bombs_capture_event5_context_after = yes
		sov_nuclear_bombs_capture_event5_registry_after = yes
		if = {
			limit = {
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_event5_before_global_ledger_total_registered value = sov_nuclear_bombs_event5_after_global_ledger_total_registered compare = equals } }
					NOT = { check_variable = { var = sov_nuclear_bombs_event5_after_global_ledger_accounted_total value = sov_nuclear_bombs_event5_after_global_ledger_total_registered compare = equals } }
				}
			}
			set_state_flag = sov_nuclear_bombs_event5_context_total_mismatch
			set_global_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
			set_variable = { global.sov_nuclear_bombs_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.disputed }
			set_variable = { sov_nuclear_bombs_event5_after_global_ledger_reconciliation = constant:sov_nuclear_bombs_ledger.disputed }
			if = { limit = { sov_nuclear_bombs_state_is_registered = yes } set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed } }
		}
	}
}

# State scope, called by the existing state-control transaction after the
# Event 005 snapshot branch has had first refusal. This is deliberately
# event-driven: an ordinary transfer moves intact Event 023 custody into the
# transferred bucket, while a later foreign-to-foreign transfer only rebinds
# the physical holder. Technical access, command formation, and delivery
# integration are never granted by a territorial change.
sov_nuclear_bombs_reconcile_state_control_change = {
	if = {
		limit = { NOT = { sov_nuclear_bombs_state_has_pending_event5_snapshot = yes } }
		sov_nuclear_bombs_reconcile_transit_state_change = yes
		sov_nuclear_bombs_reconcile_dismantlement_state_change = yes
	}
	if = {
		limit = {
			sov_nuclear_bombs_state_is_registered = yes
			NOT = { has_state_flag = sov_nuclear_bombs_event5_snapshot_pending }
			NOT = { has_state_flag = sov_nuclear_bombs_site_dismantled }
			NOT = { has_state_flag = sov_nuclear_bombs_site_missing }
		}
		if = {
			limit = {
				NOT = { OR = { is_owned_by = SOV is_controlled_by = SOV } }
				OR = {
					check_variable = { var = sov_nuclear_bombs_site_operational value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
					check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
					check_variable = { var = sov_nuclear_bombs_site_reserved value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				}
			}
			sov_nuclear_bombs_mark_site_transferred = yes
		}
		if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_site_transferred value = constant:sov_nuclear_bombs_ledger.one compare = greater_than_or_equals }
				NOT = { OR = { is_owned_by = SOV is_controlled_by = SOV } }
				OR = { owner = { exists = yes NOT = { tag = SOV } } controller = { exists = yes NOT = { tag = SOV } } }
			}
			sov_nuclear_bombs_rebind_event5_site_holder = yes
		}
		if = {
			limit = { OWNER = { exists = yes } }
			set_variable = { sov_nuclear_bombs_site_recorded_owner = OWNER }
		}
		else = { clear_variable = sov_nuclear_bombs_site_recorded_owner }
		if = {
			limit = { CONTROLLER = { exists = yes } }
			set_variable = { sov_nuclear_bombs_site_recorded_controller = CONTROLLER }
		}
		else = { clear_variable = sov_nuclear_bombs_site_recorded_controller }
		if = {
			limit = {
				OR = {
					has_state_flag = sov_nuclear_bombs_site_native_custody_disputed
					has_state_flag = sov_nuclear_bombs_native_delivery_state_pending
					has_state_flag = sov_nuclear_bombs_event5_snapshot_bucket_mismatch
				}
			}
			set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.disputed }
		}
		else = { set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.reconciled } }
		sov_nuclear_bombs_refresh_ledger_reconciliation = yes
	}
	# Access and delivery markers compare against the final physical holder.
	if = { limit = { NOT = { sov_nuclear_bombs_state_has_pending_event5_snapshot = yes } } sov_nuclear_bombs_reconcile_registry_state_change = yes }
}
