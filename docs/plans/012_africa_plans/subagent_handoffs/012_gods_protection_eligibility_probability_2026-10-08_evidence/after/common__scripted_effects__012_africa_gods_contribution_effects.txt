				# The participant is credited with what it actually gave up, up to the
				# amount asked. Africa's own stock accounting reads the separate
				# receiver gain, and a lower gain keeps the mismatch flag as evidence.
				set_variable = { gods_of_africa_material_transfer_receipt_amount = gods_of_africa_material_transfer_receipt_requested_native }
				clamp_variable = { var = gods_of_africa_material_transfer_receipt_amount min = constant:gods_of_africa_value.zero max = gods_of_africa_material_transfer_source_debit }

					clr_country_flag = gods_of_africa_material_receipt_emergency
				}
				if = {
					limit = {
						check_variable = { gods_of_africa_material_transfer_source_debit = gods_of_africa_material_transfer_receipt_requested_native }
						check_variable = { gods_of_africa_material_transfer_receiver_credit = gods_of_africa_material_transfer_receipt_requested_native }

				}
				set_variable = { gods_of_africa_material_transfer_source_debit_native = gods_of_africa_material_transfer_source_debit }
				set_variable = { gods_of_africa_material_transfer_receiver_credit_native = gods_of_africa_material_transfer_receiver_credit }
				if = {
					limit = { gods_of_africa_contribution_kind_uses_k = yes }
					multiply_variable = { gods_of_africa_material_transfer_receipt_amount = constant:gods_of_africa_contribution_measure.native_per_k }
					clamp_variable = { var = gods_of_africa_material_transfer_source_debit max = gods_of_africa_material_transfer_receipt_requested_native }
					clamp_variable = { var = gods_of_africa_material_transfer_receiver_credit max = gods_of_africa_material_transfer_receipt_requested_native }
					multiply_variable = { gods_of_africa_material_transfer_source_debit = constant:gods_of_africa_contribution_measure.native_per_k }
					multiply_variable = { gods_of_africa_material_transfer_receiver_credit = constant:gods_of_africa_contribution_measure.native_per_k }

					clamp_variable = { var = gods_of_africa_material_transfer_receiver_credit max = gods_of_africa_material_transfer_receipt_requested_native }
				}

				set_country_flag = gods_of_africa_material_transfer_measured
				set_country_flag = gods_of_africa_material_transfer_receipt_pending
				# A short or unequal delivery is still measured: the recorder credits the
				# participant's measured loss once and keeps the mismatch flag as evidence.
				# The contract owner keeps the undelivered remainder owed, so a short
				# transfer never locks the participant out of later payments.
				gods_of_africa_record_paid_contribution = yes
			}
		}
		else = {
			set_country_flag = gods_of_africa_material_transfer_receipt_pending

	}
	else_if = {
		limit = { has_country_flag = gods_of_africa_material_transfer_prepared }

		clr_country_flag = gods_of_africa_material_transfer_emission_permitted
		set_country_flag = gods_of_africa_material_transfer_mismatch

					}
				}
				add_to_variable = { gods_of_africa_war_count = constant:gods_of_africa_value.one }
			}
			set_variable = { gods_of_africa_war_observation_host_id = gods_of_africa_conduct_host_id }
			set_country_flag = gods_of_africa_war_observation_active
			set_country_flag = gods_of_africa_prior_war
			set_country_flag = gods_of_africa_war_floor
			set_country_flag = gods_of_africa_unresolved_major_offense
		}
		else = {
			clr_country_flag = gods_of_africa_war_observation_active
			clr_country_flag = gods_of_africa_war_floor
		}
		if = {
			limit = {
				OR = {
					check_variable = { gods_of_africa_observed_host_core_count > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_owned_host_core_count > constant:gods_of_africa_value.zero }
				}
			}
			set_country_flag = gods_of_africa_territory_floor
			set_country_flag = gods_of_africa_unresolved_major_offense
		}
		else = { clr_country_flag = gods_of_africa_territory_floor }
		if = {
			limit = { check_variable = { gods_of_africa_observed_host_core_count > constant:gods_of_africa_value.zero } }
			set_country_flag = gods_of_africa_prior_occupation
		}
		if = {
			limit = {
				OR = {
					has_country_flag = gods_of_africa_defiant
					has_country_flag = gods_of_africa_permanent_opt_out
				}
			}
			set_country_flag = gods_of_africa_prior_defiance
			set_country_flag = gods_of_africa_unresolved_major_offense
		}
		# No treaty write: generic offender/protection markers prove no participant promise.
		gods_of_africa_measure_grading_wrath_floor = yes
	}
	else = { set_country_flag = gods_of_africa_grading_conduct_unresolved }
}

# PARTICIPANT: maximum, never sequentially overwrite a stronger applicable floor.
gods_of_africa_measure_grading_wrath_floor = {
	if = {
		limit = { gods_of_africa_grading_conduct_can_observe = yes }
		set_variable = { gods_of_africa_grading_wrath_floor = constant:gods_of_africa_value.minimum_wrath }
		if = {
			limit = { has_country_flag = gods_of_africa_war_floor }
			clamp_variable = { var = gods_of_africa_grading_wrath_floor min = constant:gods_of_africa_wrath.floor_war }
		}
		if = {
			limit = { has_country_flag = gods_of_africa_territory_floor }
			clamp_variable = { var = gods_of_africa_grading_wrath_floor min = constant:gods_of_africa_wrath.floor_territory }
		}
		if = {
			limit = { has_country_flag = gods_of_africa_unresolved_treaty_break }
			clamp_variable = { var = gods_of_africa_grading_wrath_floor min = constant:gods_of_africa_wrath.floor_broken_treaty }
		}
		if = {
			limit = { has_country_flag = gods_of_africa_defiant }
			clamp_variable = { var = gods_of_africa_grading_wrath_floor min = constant:gods_of_africa_wrath.floor_defiant }
		}
		if = {
			limit = { has_country_flag = gods_of_africa_recent_extreme_punishment }
			clamp_variable = { var = gods_of_africa_grading_wrath_floor min = constant:gods_of_africa_wrath.floor_recent_extreme }
		}
		if = {
			limit = { has_variable = gods_of_africa_wrath }
			clamp_variable = { var = gods_of_africa_wrath min = gods_of_africa_grading_wrath_floor max = constant:gods_of_africa_value.maximum_wrath }
		}
	}
}

# PARTICIPANT: normal input gods_of_africa_return_state holds the selected STATE.
gods_of_africa_begin_core_return = {
	clr_country_flag = gods_of_africa_core_return_begin_ready
	if = {
		limit = {
			gods_of_africa_grading_conduct_can_observe = yes
			OR = {
				AND = {
					gods_of_africa_contribution_contract_is_current = yes
					has_variable = gods_of_africa_payment_family
					check_variable = { gods_of_africa_payment_family = constant:gods_of_africa_demand_family.territory }
				}
				gods_of_africa_final_ultimatum_is_open = yes
			}
			has_variable = gods_of_africa_return_state
			NOT = { has_country_flag = gods_of_africa_core_return_pending }
			var:gods_of_africa_return_state = {
				is_on_continent = africa
				is_core_of = event_target:africa_host
				OR = { is_owned_by = PREV is_controlled_by = PREV }
			}
		}
		save_event_target_as = gods_of_africa_return_participant
		set_variable = { gods_of_africa_core_return_state = gods_of_africa_return_state }
		set_variable = { gods_of_africa_core_return_participant_id = THIS.id }
		clr_country_flag = gods_of_africa_core_return_ultimatum
		clear_variable = gods_of_africa_core_return_ultimatum_due_day
		if = {
			limit = { gods_of_africa_final_ultimatum_is_open = yes }
			set_country_flag = gods_of_africa_core_return_ultimatum
			set_variable = { gods_of_africa_core_return_host = event_target:africa_host }
			set_variable = { gods_of_africa_core_return_host_id = event_target:africa_host.id }
			set_variable = { gods_of_africa_core_return_ultimatum_due_day = gods_of_africa_final_ultimatum_due_day }
			clear_variable = gods_of_africa_core_return_contract_id
		}
		else = {
			set_variable = { gods_of_africa_core_return_host = gods_of_africa_demand_host }
			set_variable = { gods_of_africa_core_return_host_id = gods_of_africa_demand_host_id }
			set_variable = { gods_of_africa_core_return_contract_id = gods_of_africa_contract_ledger_id }
		}
		set_variable = { gods_of_africa_core_return_generation = event_target:africa_host.africa_host_generation }
		set_variable = { gods_of_africa_core_return_origin_id = event_target:africa_host.africa_origin_host_id }
		set_variable = { gods_of_africa_core_return_day = global.num_days }
		clr_country_flag = gods_of_africa_core_return_before_owned
		clr_country_flag = gods_of_africa_core_return_before_controlled
		var:gods_of_africa_core_return_state = {
			set_temp_variable = { gods_of_africa_return_state_id = THIS.id }
			if = {
				limit = { is_owned_by = event_target:gods_of_africa_return_participant }
				event_target:gods_of_africa_return_participant = { set_country_flag = gods_of_africa_core_return_before_owned }
			}
			if = {
				limit = { is_controlled_by = event_target:gods_of_africa_return_participant }
				event_target:gods_of_africa_return_participant = { set_country_flag = gods_of_africa_core_return_before_controlled }
			}
		}
		set_variable = { gods_of_africa_core_return_state_id = gods_of_africa_return_state_id }
		set_country_flag = gods_of_africa_core_return_pending
		set_country_flag = gods_of_africa_core_return_begin_ready
	}
}

# PARTICIPANT: selected STATE has actually become both owned and controlled by that host.
gods_of_africa_complete_core_return = {
	clr_country_flag = gods_of_africa_core_return_recorded
	clr_country_flag = gods_of_africa_core_return_begin_ready
	if = {
		limit = {
			gods_of_africa_grading_conduct_can_observe = yes
			has_country_flag = gods_of_africa_core_return_pending
			has_variable = gods_of_africa_core_return_state
			has_variable = gods_of_africa_core_return_state_id
			has_variable = gods_of_africa_core_return_host
			has_variable = gods_of_africa_core_return_participant_id
			check_variable = { gods_of_africa_core_return_participant_id = THIS.id }
			check_variable = { gods_of_africa_core_return_host = event_target:africa_host }
			check_variable = { gods_of_africa_core_return_host_id = event_target:africa_host.id }
			check_variable = { gods_of_africa_core_return_generation = event_target:africa_host.africa_host_generation }
			check_variable = { gods_of_africa_core_return_origin_id = event_target:africa_host.africa_origin_host_id }
			check_variable = { gods_of_africa_core_return_day = global.num_days }
			OR = {
				AND = {
					NOT = { has_country_flag = gods_of_africa_core_return_ultimatum }
					gods_of_africa_contribution_contract_is_current = yes
					check_variable = { gods_of_africa_core_return_contract_id = gods_of_africa_contract_ledger_id }
				}
				AND = {
					has_country_flag = gods_of_africa_core_return_ultimatum
					gods_of_africa_final_ultimatum_is_open = yes
					has_variable = gods_of_africa_core_return_ultimatum_due_day
					check_variable = { gods_of_africa_core_return_ultimatum_due_day = gods_of_africa_final_ultimatum_due_day }
				}
			}
			OR = {
				has_country_flag = gods_of_africa_core_return_before_owned
				has_country_flag = gods_of_africa_core_return_before_controlled
			}
			var:gods_of_africa_core_return_state = {
				check_variable = { PREV.gods_of_africa_core_return_state_id = THIS.id }
				is_on_continent = africa
				is_core_of = event_target:africa_host
				is_owned_by = event_target:africa_host
				is_controlled_by = event_target:africa_host
			}
		}
		clr_country_flag = gods_of_africa_core_return_pending
		if = {
			limit = { NOT = { is_in_array = { array = gods_of_africa_returned_core_state_ids value = gods_of_africa_core_return_state_id } } }
			add_to_array = { array = gods_of_africa_returned_core_state_ids value = gods_of_africa_core_return_state_id }
			set_country_flag = gods_of_africa_core_return_recorded
		}
		gods_of_africa_clear_core_return_work = yes
		gods_of_africa_refresh_grading_conduct = yes
	}
	else_if = {
		limit = { has_country_flag = gods_of_africa_core_return_pending }
		set_country_flag = gods_of_africa_grading_conduct_unresolved
		# Retain failed/stale return snapshots for explicit owner recovery.
	}
}

	clear_variable = gods_of_africa_core_return_participant_id
	clear_variable = gods_of_africa_core_return_ultimatum_due_day
	clr_country_flag = gods_of_africa_core_return_pending
	clr_country_flag = gods_of_africa_core_return_before_owned
	clr_country_flag = gods_of_africa_core_return_before_controlled
	clear_variable = gods_of_africa_core_return_state
	clear_variable = gods_of_africa_core_return_state_id
	clear_variable = gods_of_africa_core_return_host
	clear_variable = gods_of_africa_core_return_host_id
	clear_variable = gods_of_africa_core_return_generation
	clear_variable = gods_of_africa_core_return_origin_id
	clear_variable = gods_of_africa_core_return_contract_id
	clear_variable = gods_of_africa_core_return_day
}

# STATE scope. The caller freezes the returning participant as the regular
# gods_of_africa_territory_payer target. Keep proof around the exact native transfer.
gods_of_africa_transfer_current_core_return = {
	event_target:gods_of_africa_territory_payer = {

		}

		set_country_flag = gods_of_africa_territory_returned
	}
}

# PARTICIPANT, inside actual host-acceptance callback AFTER unique paid acceptance receipt.
# Parent must observe before acceptance as well; no current war/core land may be waived.
gods_of_africa_resolve_accepted_grading_conduct = {
	gods_of_africa_refresh_grading_conduct = yes
	if = {
		limit = {
			gods_of_africa_contribution_sender_is_current = yes
			ROOT = { tag = event_target:africa_host }
			has_country_flag = gods_of_africa_reparations_paid
			has_country_flag = gods_of_africa_reconciliation_offer_pending
			has_variable = gods_of_africa_reconciliation_offer_sequence
			has_variable = gods_of_africa_last_accepted_reconciliation_sequence
			has_variable = gods_of_africa_last_accepted_reconciliation_generation
			has_variable = gods_of_africa_last_accepted_reconciliation_origin_id
			has_variable = gods_of_africa_last_accepted_reconciliation_day
			check_variable = { gods_of_africa_last_accepted_reconciliation_sequence = gods_of_africa_reconciliation_offer_sequence }
			check_variable = { gods_of_africa_last_accepted_reconciliation_generation = event_target:africa_host.africa_host_generation }
			check_variable = { gods_of_africa_last_accepted_reconciliation_origin_id = event_target:africa_host.africa_origin_host_id }
			check_variable = { gods_of_africa_last_accepted_reconciliation_day = global.num_days }
			NOT = { has_war_with = event_target:africa_host }
			check_variable = { gods_of_africa_observed_host_core_count = constant:gods_of_africa_value.zero }
			check_variable = { gods_of_africa_owned_host_core_count = constant:gods_of_africa_value.zero }
			NOR = {
				has_country_flag = gods_of_africa_grading_conduct_unresolved
				has_country_flag = gods_of_africa_core_return_pending
			}
		}
		clr_country_flag = gods_of_africa_unresolved_major_offense
		clr_country_flag = gods_of_africa_unresolved_treaty_break
		clr_country_flag = gods_of_africa_broken_treaty
		clr_country_flag = gods_of_africa_war_floor
		clr_country_flag = gods_of_africa_territory_floor
		# Parent acceptance owner clears current defiance and resets recovery counters.
	}
}

# PARTICIPANT immediately after the native send, inside the same effect.
# Publishes the participant's measured loss in gods_of_africa_payment_delivered
# (the contract and contribution credit) and Africa's measured gain, never more
# than that loss, in gods_of_africa_payment_received (Africa's stock accounts).
# Both stay zero when nothing was measured, so no record is ever credited from
# a requested amount.
gods_of_africa_complete_payment_receipt = {
	gods_of_africa_complete_material_transfer = yes
	set_variable = { gods_of_africa_payment_delivered = constant:gods_of_africa_value.zero }
	set_variable = { gods_of_africa_payment_received = constant:gods_of_africa_value.zero }

		limit = { has_country_flag = gods_of_africa_material_transfer_completion_consumed }
		set_variable = { gods_of_africa_payment_delivered = gods_of_africa_material_transfer_actual_amount }
		set_variable = { gods_of_africa_payment_received = gods_of_africa_material_transfer_receiver_credit }
		clamp_variable = { var = gods_of_africa_payment_received min = constant:gods_of_africa_value.zero max = gods_of_africa_payment_delivered }
	}
}
