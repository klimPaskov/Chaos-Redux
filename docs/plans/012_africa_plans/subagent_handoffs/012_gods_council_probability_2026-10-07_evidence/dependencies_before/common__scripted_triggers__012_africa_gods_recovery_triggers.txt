# Country-object validators are independent of native ROOT. Action gates alone
# require host ROOT. Temporary recount outputs are unscoped and caller-owned.

gods_of_africa_authority_owner_object_is_current = {
	exists = yes
	has_event_target = africa_host
	tag = event_target:africa_host
	has_country_flag = africa_unifier_host
	has_country_flag = africa_host_commit_completed
	has_variable = africa_host_generation
	has_variable = africa_origin_host_id
	has_global_flag = africa_event_active
	has_global_flag = africa_is_one
	has_global_flag = africa_evolution_i_logged
	NOT = { has_global_flag = africa_event_resolved }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = world_end_africa_the_world }
	NOT = { has_global_flag = africa_host_transfer_pending }
	NOT = { has_global_flag = africa_host_transfer_suspended }
	NOT = { has_country_flag = africa_terminal_world_actor }
	NOT = { has_country_flag = event_012_incompatible_terminal_actor }
	NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
	# A live authorized epoch must still reach its one-use final settlement.
	# Secured-continent proof is excluded at new-grant gates, not here.
}

# Always true; initialized output storage is required when values escape.
# The source relationship registry is unique by its existing registration API.
# OR fallthrough ensures ineligible members do not stop all_of_scopes early.
gods_of_africa_recount_recovery_mandate = {
	set_temp_variable = { gods_of_africa_recovery_eligible_count = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_recovery_approved_count = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_recovery_majority_count = constant:gods_of_africa_value.zero }
	all_of_scopes = {
		array = africa_relationship_countries
		OR = {
			AND = {
				gods_of_africa_recovery_member_is_eligible = yes
				add_to_temp_variable = { gods_of_africa_recovery_eligible_count = constant:gods_of_africa_value.one }
				OR = {
					AND = {
						gods_of_africa_recovery_member_approval_is_current = yes
						add_to_temp_variable = { gods_of_africa_recovery_approved_count = constant:gods_of_africa_value.one }
					}
					always = yes
				}
			}
			always = yes
		}
	}
	set_temp_variable = { gods_of_africa_recovery_majority_count = gods_of_africa_recovery_approved_count }
	multiply_temp_variable = { gods_of_africa_recovery_majority_count = constant:gods_of_africa_recovery.majority_multiplier }
}
