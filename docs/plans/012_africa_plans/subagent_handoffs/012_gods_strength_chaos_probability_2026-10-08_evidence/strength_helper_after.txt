gods_of_africa_refresh_strength = {
	if = {
		limit = { africa_is_current_host = yes }
		set_temp_variable = { gods_of_africa_strength_military = num_divisions }
		divide_temp_variable = { gods_of_africa_strength_military = constant:gods_of_africa_strength.divisions_divisor }
		if = {
			limit = { has_variable = africa_continental_war_readiness }
			add_to_temp_variable = { gods_of_africa_strength_military = africa_continental_war_readiness }
			divide_temp_variable = { gods_of_africa_strength_military = constant:gods_of_africa_strength.component_average_divisor }
		}
		clamp_temp_variable = { var = gods_of_africa_strength_military min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		# Sustainment (decision L29): an army short of its rifles or of fuel
		# counts for less than a supplied one of the same size.
		set_temp_variable = { gods_of_africa_sustain_equipment = constant:gods_of_africa_strength_sustain.full }
		set_temp_variable = { gods_of_africa_sustain_target = num_target_equipment_in_armies_k@infantry_equipment }
		if = {
			limit = { check_variable = { gods_of_africa_sustain_target > constant:gods_of_africa_strength_sustain.none } }
			set_temp_variable = { gods_of_africa_sustain_equipment = num_equipment_in_armies_k@infantry_equipment }
			divide_temp_variable = { gods_of_africa_sustain_equipment = gods_of_africa_sustain_target }
		}
		clamp_temp_variable = { var = gods_of_africa_sustain_equipment min = constant:gods_of_africa_strength_sustain.none max = constant:gods_of_africa_strength_sustain.full }
		set_temp_variable = { gods_of_africa_sustain_fuel = constant:gods_of_africa_strength_sustain.full }
		set_temp_variable = { gods_of_africa_sustain_target = max_fuel_k }
		if = {
			limit = { check_variable = { gods_of_africa_sustain_target > constant:gods_of_africa_strength_sustain.none } }
			set_temp_variable = { gods_of_africa_sustain_fuel = fuel_k }
			divide_temp_variable = { gods_of_africa_sustain_fuel = gods_of_africa_sustain_target }
		}
		clamp_temp_variable = { var = gods_of_africa_sustain_fuel min = constant:gods_of_africa_strength_sustain.none max = constant:gods_of_africa_strength_sustain.full }
		multiply_temp_variable = { gods_of_africa_sustain_equipment = constant:gods_of_africa_strength_sustain.equipment_share }
		multiply_temp_variable = { gods_of_africa_sustain_fuel = constant:gods_of_africa_strength_sustain.fuel_share }
		set_temp_variable = { gods_of_africa_sustain_factor = gods_of_africa_sustain_equipment }
		add_to_temp_variable = { gods_of_africa_sustain_factor = gods_of_africa_sustain_fuel }
		multiply_temp_variable = { gods_of_africa_sustain_factor = constant:gods_of_africa_strength_sustain.span }
		add_to_temp_variable = { gods_of_africa_sustain_factor = constant:gods_of_africa_strength_sustain.floor }
		multiply_temp_variable = { gods_of_africa_strength_military = gods_of_africa_sustain_factor }
		set_variable = { gods_of_africa_strength_sustain_factor = gods_of_africa_sustain_factor }
		set_temp_variable = { gods_of_africa_strength_industry = num_of_civilian_factories }
		add_to_temp_variable = { gods_of_africa_strength_industry = num_of_military_factories }
		multiply_temp_variable = { gods_of_africa_strength_industry = constant:gods_of_africa_strength.factories_divisor }
		if = {
			limit = { has_variable = africa_central_capacity }
			add_to_temp_variable = { gods_of_africa_strength_industry = africa_central_capacity }
			divide_temp_variable = { gods_of_africa_strength_industry = constant:gods_of_africa_strength.component_average_divisor }
		}
		clamp_temp_variable = { var = gods_of_africa_strength_industry min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		set_temp_variable = { gods_of_africa_strength_continent = constant:gods_of_africa_strength.component_min }
		if = {
			limit = { has_variable = africa_cooperative_relationship_count }
			set_temp_variable = { gods_of_africa_strength_continent = africa_cooperative_relationship_count }
		}
		if = {
			limit = { has_variable = africa_charter_authority }
			add_to_temp_variable = { gods_of_africa_strength_continent = africa_charter_authority }
			divide_temp_variable = { gods_of_africa_strength_continent = constant:gods_of_africa_strength.component_average_divisor }
		}
		# Each recognising major power adds to the continent component, up to
		# the cap (decision L25).
		set_variable = { gods_of_africa_recognising_major_count = constant:gods_of_africa_value.zero }
		for_each_scope_loop = {
			array = gods_of_africa_participants
			if = {
				limit = {
					has_country_flag = gods_of_africa_recognition_granted
					is_major = yes
				}
				event_target:africa_host = { add_to_variable = { gods_of_africa_recognising_major_count = constant:gods_of_africa_value.one } }
			}
		}
		set_temp_variable = { gods_of_africa_strength_recognition = gods_of_africa_recognising_major_count }
		multiply_temp_variable = { gods_of_africa_strength_recognition = constant:gods_of_africa_strength.recognition_per_major }
		clamp_temp_variable = { var = gods_of_africa_strength_recognition min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.recognition_cap }
		add_to_temp_variable = { gods_of_africa_strength_continent = gods_of_africa_strength_recognition }
		clamp_temp_variable = { var = gods_of_africa_strength_continent min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		set_temp_variable = { gods_of_africa_strength_reach = constant:gods_of_africa_strength.component_min }
		if = { limit = { has_variable = africa_continental_reach } set_temp_variable = { gods_of_africa_strength_reach = africa_continental_reach } }
		set_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.component_min }
		if = { limit = { has_tech = radio } add_to_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.technology_radio } }
		if = { limit = { has_tech = radio_detection } add_to_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.technology_radio_detection } }
		if = { limit = { has_tech = construction5 } add_to_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.technology_construction } }
		if = { limit = { has_tech = advanced_machine_tools } add_to_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.technology_machine_tools } }
		if = { limit = { OR = { has_tech = fighter3 has_tech = advanced_small_airframe } } add_to_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.technology_fighter } }
		if = { limit = { has_tech = synth_oil_experiments } add_to_temp_variable = { gods_of_africa_strength_technology = constant:gods_of_africa_strength.technology_synthetic } }
		add_to_temp_variable = { gods_of_africa_strength_reach = gods_of_africa_strength_technology }
		divide_temp_variable = { gods_of_africa_strength_reach = constant:gods_of_africa_strength.component_average_divisor }
		divide_temp_variable = { gods_of_africa_strength_reach = constant:gods_of_africa_strength.reach_divisor }
		clamp_temp_variable = { var = gods_of_africa_strength_reach min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		set_temp_variable = { gods_of_africa_strength_internal = stability }
		multiply_temp_variable = { gods_of_africa_strength_internal = 100 }
		if = {
			limit = { check_variable = { var = surrender_progress value = constant:africa_action_threshold.near_capitulation compare = greater_than } }
			subtract_from_temp_variable = { gods_of_africa_strength_internal = constant:africa_measure.medium }
		}
		clamp_temp_variable = { var = gods_of_africa_strength_internal min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		set_temp_variable = { gods_of_africa_strength_chaos = constant:gods_of_africa_strength.component_min }
		set_temp_variable = { gods_of_africa_strength_chaos = global.chaos_meter_value }
		# Normalize the shared meter against its final-tier endpoint.
		divide_temp_variable = { gods_of_africa_strength_chaos = constant:chaos_meter_tier_range.tier_final.min }
		multiply_temp_variable = { gods_of_africa_strength_chaos = constant:gods_of_africa_strength.component_max }
		if = { limit = { has_global_flag = africa_evolution_ii_logged } add_to_temp_variable = { gods_of_africa_strength_chaos = constant:gods_of_africa_strength.evolution_ii_amplifier } }
		if = { limit = { has_global_flag = africa_evolution_iii_logged } add_to_temp_variable = { gods_of_africa_strength_chaos = constant:gods_of_africa_strength.evolution_iii_amplifier } }
		clamp_temp_variable = { var = gods_of_africa_strength_chaos min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		set_temp_variable = { gods_of_africa_strength_target = gods_of_africa_strength_military }
		multiply_temp_variable = { gods_of_africa_strength_target = constant:gods_of_africa_value.strength_military_weight }
		set_temp_variable = { gods_of_africa_strength_component = gods_of_africa_strength_industry }
		multiply_temp_variable = { gods_of_africa_strength_component = constant:gods_of_africa_value.strength_industry_weight }
		add_to_temp_variable = { gods_of_africa_strength_target = gods_of_africa_strength_component }
		set_temp_variable = { gods_of_africa_strength_component = gods_of_africa_strength_continent }
		multiply_temp_variable = { gods_of_africa_strength_component = constant:gods_of_africa_value.strength_continent_weight }
		add_to_temp_variable = { gods_of_africa_strength_target = gods_of_africa_strength_component }
		set_temp_variable = { gods_of_africa_strength_component = gods_of_africa_strength_reach }
		multiply_temp_variable = { gods_of_africa_strength_component = constant:gods_of_africa_value.strength_reach_weight }
		add_to_temp_variable = { gods_of_africa_strength_target = gods_of_africa_strength_component }
		set_temp_variable = { gods_of_africa_strength_component = gods_of_africa_strength_internal }
		multiply_temp_variable = { gods_of_africa_strength_component = constant:gods_of_africa_value.strength_internal_weight }
		add_to_temp_variable = { gods_of_africa_strength_target = gods_of_africa_strength_component }
		set_temp_variable = { gods_of_africa_strength_component = gods_of_africa_strength_chaos }
		multiply_temp_variable = { gods_of_africa_strength_component = constant:gods_of_africa_value.strength_chaos_weight }
		add_to_temp_variable = { gods_of_africa_strength_target = gods_of_africa_strength_component }
		divide_temp_variable = { gods_of_africa_strength_target = constant:gods_of_africa_strength.weighted_divisor }
		# A lost capital and a losing war cost Strength directly (decision L29).
		clr_country_flag = gods_of_africa_strength_capital_lost
		if = {
			limit = { capital_scope = { NOT = { is_controlled_by = PREV } } }
			set_country_flag = gods_of_africa_strength_capital_lost
			subtract_from_temp_variable = { gods_of_africa_strength_target = constant:gods_of_africa_strength_sustain.capital_loss_penalty }
			# The fall is felt at once the first time the capital is lost.
			if = {
				limit = { NOT = { has_country_flag = gods_of_africa_strength_capital_loss_recorded } }
				set_country_flag = gods_of_africa_strength_capital_loss_recorded
				set_country_flag = gods_of_africa_strength_immediate
			}
		}
		else = { clr_country_flag = gods_of_africa_strength_capital_loss_recorded }
		if = {
			limit = { check_variable = { var = surrender_progress value = constant:gods_of_africa_strength_sustain.surrender_severe compare = greater_than } }
			subtract_from_temp_variable = { gods_of_africa_strength_target = constant:gods_of_africa_strength_sustain.surrender_severe_penalty }
		}
		else_if = {
			limit = { check_variable = { var = surrender_progress value = constant:gods_of_africa_strength_sustain.surrender_heavy compare = greater_than } }
			subtract_from_temp_variable = { gods_of_africa_strength_target = constant:gods_of_africa_strength_sustain.surrender_heavy_penalty }
		}
		else_if = {
			limit = { check_variable = { var = surrender_progress value = constant:gods_of_africa_strength_sustain.surrender_light compare = greater_than } }
			subtract_from_temp_variable = { gods_of_africa_strength_target = constant:gods_of_africa_strength_sustain.surrender_light_penalty }
		}
		if = {
			limit = { check_variable = { var = surrender_progress value = constant:gods_of_africa_strength_sustain.surrender_immediate compare = greater_than } }
			set_country_flag = gods_of_africa_strength_immediate
		}
		clamp_temp_variable = { var = gods_of_africa_strength_target min = constant:gods_of_africa_strength.component_min max = constant:gods_of_africa_strength.component_max }
		if = {
			limit = {
				OR = {
					has_country_flag = gods_of_africa_strength_immediate
					NOT = { has_variable = gods_of_africa_strength }
				}
			}
			set_variable = { gods_of_africa_strength = gods_of_africa_strength_target }
		}
		else = {
			set_temp_variable = { gods_of_africa_strength_delta = gods_of_africa_strength_target }
			subtract_from_temp_variable = { gods_of_africa_strength_delta = gods_of_africa_strength }
			set_temp_variable = { gods_of_africa_strength_deficit = gods_of_africa_strength }
			subtract_from_temp_variable = { gods_of_africa_strength_deficit = gods_of_africa_strength_target }
			if = {
				limit = { check_variable = { gods_of_africa_strength_delta > constant:gods_of_africa_value.strength_smoothing_step } }
				add_to_variable = { gods_of_africa_strength = constant:gods_of_africa_value.strength_smoothing_step }
			}
			else_if = {
				limit = { check_variable = { gods_of_africa_strength_deficit > constant:gods_of_africa_value.strength_smoothing_step } }
				set_temp_variable = { gods_of_africa_strength_delta = constant:gods_of_africa_value.strength_smoothing_step }
				subtract_from_variable = { gods_of_africa_strength = gods_of_africa_strength_delta }
			}
			else = { set_variable = { gods_of_africa_strength = gods_of_africa_strength_target } }
		}
		clamp_variable = { var = gods_of_africa_strength min = constant:gods_of_africa_value.zero max = constant:gods_of_africa_value.maximum_strength }
		if = { limit = { check_variable = { gods_of_africa_strength < constant:gods_of_africa_value.limited_strength } } set_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.limited } }
		else_if = { limit = { check_variable = { gods_of_africa_strength < constant:gods_of_africa_value.regional_strength } } set_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.regional } }
		else_if = { limit = { check_variable = { gods_of_africa_strength < constant:gods_of_africa_value.continental_strength } } set_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.continental } }
		else_if = { limit = { check_variable = { gods_of_africa_strength < constant:gods_of_africa_value.global_strength } } set_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.global } }
		else = { set_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.extreme } }
		set_variable = { global.gods_of_africa_strength = gods_of_africa_strength }
		clr_country_flag = gods_of_africa_strength_immediate
		gods_of_africa_emit_global_credibility_super_event = yes
	}
}
