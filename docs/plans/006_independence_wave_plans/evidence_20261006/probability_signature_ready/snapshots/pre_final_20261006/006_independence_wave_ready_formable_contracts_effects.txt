# ============================================================================
# CHAOS REDUX - EVENT 006 READY FORMABLE CONTRACT EFFECTS
# ============================================================================
# Overview:
# - Registers the four accepted member contracts through the shared FORM
#   profile, bounded active-country registry, invitation, consent, and ledger
#   protocol.
# - Lets an existing admitted member become the carrier under a new X-ending
#   cosmetic identity; it never creates a sovereign tag or leader.
# - Leaves every member sovereign. A consenting member receives reciprocal
#   relations and may exchange an already-active package institution for the
#   compact; no substitute idea is granted when that institution is absent.
# - Grants a carrier charter only after the shared cost, prevalidation, exact
#   live roster, frozen consent, identity, and integration proofs pass.
# - Advances each installed charter through two paid, one-time family projects.
# - Rolls back a failed identity and removes generation-bound member relations
#   when the carrier or a member Event 006 origin ends.
# - Contains no periodic hook, world scan, state transfer, annexation, or core grant.
# ============================================================================

# Called by the generic selected-family readiness dispatcher after the shared
# profile is loaded. The member and anchor ledger reads only the aligned,
# current-generation Event 006 active-country registry.
independence_wave_ready_formable_register_readiness = {
	if = {
		limit = { can_independence_wave_ready_formable_register_readiness = yes }
		independence_wave_formable_build_member_and_anchor_ledgers = yes
		if = {
			limit = { has_independence_wave_ready_formable_live_roster_ledger = yes }
			set_variable = { independence_wave_formable_readiness_family = independence_wave_formable_profile_family }
			set_country_flag = independence_wave_formable_territory_adapter_ready
			set_country_flag = independence_wave_formable_x_tag_reserved
			set_country_flag = independence_wave_formable_flag_package_ready
			set_country_flag = independence_wave_formable_identity_adapter_ready
			set_country_flag = independence_wave_formable_integration_adapter_ready
			set_country_flag = independence_wave_formable_member_policy_audited
			if = { limit = { check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.north_caucasian_federation } }
				set_country_flag = independence_wave_form17_readiness_attested
				set_country_flag = independence_wave_formable_map_reveal_ready
			}
			if = { limit = { check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.indus_federation } }
				set_country_flag = independence_wave_form32_readiness_attested
				set_country_flag = independence_wave_formable_map_reveal_ready
			}
			if = { limit = { check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.maritime_federation } }
				set_country_flag = independence_wave_form36_readiness_attested
				set_country_flag = independence_wave_formable_league_reveal_ready
			}
			if = { limit = { check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.mainland_southeast_asian_federation } }
				set_country_flag = independence_wave_form37_readiness_attested
				set_country_flag = independence_wave_formable_map_reveal_ready
			}
		}
	}
}

# Candidate-country scope with ROOT as the carrier. AI replies are explicit
# consent or refusal. The accepting member pays no integration cost because
# it remains sovereign; the carrier pays the shared formation cost at commit.
independence_wave_ready_formable_resolve_ai_invitation_from_root = {
	if = {
		limit = {
			is_ai = yes
			has_pending_independence_wave_formable_founding_invitation = yes
			has_independence_wave_ready_formable_invited_member = yes
		}
		if = {
			limit = { should_independence_wave_ready_formable_ai_consent_to_root = yes }
			independence_wave_formable_declare_consent_for_selected_family = yes
		}
		else = { independence_wave_formable_withhold_consent_for_selected_family = yes }
	}
}

# Candidate-country scope with ROOT as the inviting carrier. This exact receipt
# is written before the event is fired and remains separate from the still-live
# invitation; the event consumes the delivery receipt before its popup appears.
independence_wave_ready_formable_deliver_human_invitation_from_root = {
	if = {
		limit = {
			is_ai = no
			has_pending_independence_wave_formable_founding_invitation = yes
			has_independence_wave_ready_formable_invited_member = yes
			NOT = { has_country_flag = independence_wave_ready_formable_popup_delivery_pending }
			NOT = { has_country_flag = independence_wave_ready_formable_popup_shown }
		}
		set_country_flag = independence_wave_ready_formable_popup_delivery_pending
		set_variable = { independence_wave_ready_formable_popup_delivery_sequence = independence_wave_formable_invitation_sequence }
		set_variable = { independence_wave_ready_formable_popup_delivery_family = independence_wave_formable_invitation_family }
		set_variable = { independence_wave_ready_formable_popup_delivery_carrier = independence_wave_formable_inviting_carrier }
		set_variable = { independence_wave_ready_formable_popup_delivery_carrier_generation = independence_wave_formable_inviting_carrier_generation }
		set_variable = { independence_wave_ready_formable_popup_delivery_member_generation = independence_wave_formable_invited_member_generation }
		country_event = { id = chaosx.nr6.8617 }
	}
}

# Country-event scope. A stale delivery receipt is discarded without changing
# the member's still-valid consent state. A valid receipt is consumed before
# the event becomes a visible popup.
independence_wave_ready_formable_consume_human_invitation_delivery = {
	if = {
		limit = { has_independence_wave_ready_formable_valid_delivery_request = yes }
		clr_country_flag = independence_wave_ready_formable_popup_delivery_pending
		set_country_flag = independence_wave_ready_formable_popup_shown
		set_variable = { independence_wave_ready_formable_popup_sequence = independence_wave_formable_invitation_sequence }
		set_variable = { independence_wave_ready_formable_popup_family = independence_wave_formable_invitation_family }
		set_variable = { independence_wave_ready_formable_popup_carrier_generation = independence_wave_formable_inviting_carrier_generation }
		set_variable = { independence_wave_ready_formable_popup_member_generation = independence_wave_formable_invited_member_generation }
	}
	else = { independence_wave_ready_formable_clear_human_invitation_popup = yes }
}

independence_wave_ready_formable_human_accept_invitation = {
	if = {
		limit = { has_independence_wave_ready_formable_human_invitation_reply = yes }
		independence_wave_formable_declare_consent_for_selected_family = yes
	}
	independence_wave_ready_formable_clear_human_invitation_popup = yes
}

independence_wave_ready_formable_human_refuse_invitation = {
	if = {
		limit = { has_independence_wave_ready_formable_human_invitation_reply = yes }
		independence_wave_formable_withhold_consent_for_selected_family = yes
	}
	independence_wave_ready_formable_clear_human_invitation_popup = yes
}

independence_wave_ready_formable_clear_human_invitation_popup = {
	clr_country_flag = independence_wave_ready_formable_popup_delivery_pending
	clr_country_flag = independence_wave_ready_formable_popup_shown
	clear_variable = independence_wave_ready_formable_popup_delivery_sequence
	clear_variable = independence_wave_ready_formable_popup_delivery_family
	clear_variable = independence_wave_ready_formable_popup_delivery_carrier
	clear_variable = independence_wave_ready_formable_popup_delivery_carrier_generation
	clear_variable = independence_wave_ready_formable_popup_delivery_member_generation
	clear_variable = independence_wave_ready_formable_popup_sequence
	clear_variable = independence_wave_ready_formable_popup_family
	clear_variable = independence_wave_ready_formable_popup_carrier_generation
	clear_variable = independence_wave_ready_formable_popup_member_generation
}
