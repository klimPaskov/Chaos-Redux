# Overview:
# - Registers the three accepted Level 3 country modules and owns their
#   package-specific route, project, formable-entry and cleanup effects.
# - State construction uses the current owned Event 006 anchor.

# carrier in the CXT extension registry. Applying it only revisits the three
# exact, live Event 006 package rows in the aligned active-country registry.
# It never grants country content or creates a released package.
independence_wave_signature_modules_register_cxt_test_content = {
	set_temp_variable = {
		var = chaosx_test_country_registration_extension_effect
		value = token:independence_wave_signature_modules_cxt_extension
	}
	chaosx_test_country_register_extension_effect = yes
}

independence_wave_signature_modules_cxt_extension_apply = {
	if = {
		limit = {
			tag = CXT
			has_country_flag = chaosx_test_country_initialized
			independence_wave_active_country_arrays_are_aligned = yes
		}
		for_each_scope_loop = {
			array = global.independence_wave_active_countries
			if = {
				limit = { is_independence_wave_signature_modules_exact_active_package = yes }
				if = {
					limit = {
						original_tag = DSX
						check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_097 }
					}
					independence_wave_dsx_signature_setup = yes
				}
				if = {
					limit = {
						original_tag = ATJ
						check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_150 }
					}
					independence_wave_atj_signature_setup = yes
				}
				if = {
					limit = {
						original_tag = IAX
						check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_161 }
					}
					independence_wave_iax_signature_setup = yes
				}
			}
		}
	}
}
