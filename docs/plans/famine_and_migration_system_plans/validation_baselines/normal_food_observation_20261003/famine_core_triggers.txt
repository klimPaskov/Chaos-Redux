# Chaos Redux famine scripted triggers.
# Food-state, blockade, profile, mortality, reserve, and famine retirement contracts.


# Each mechanic owns its own bounded mission budget. A famine mission never
# consumes or releases a migration slot, and vice versa.
famine_country_has_free_mission_slot = {
	OR = {
		NOT = { has_variable = famine_active_mission_count }
		check_variable = {
			var = famine_active_mission_count
			value = constant:famine_mission_contract.maximum_active_missions
			compare = less_than
		}
	}
}

famine_pressure_request_is_valid = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		check_variable = { var = famine_pressure_request_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = famine_pressure_request_amount value = constant:humanitarian_population.zero compare = greater_than }
		check_variable = { var = famine_pressure_request_source value = constant:humanitarian_pressure_source.unknown compare = greater_than }
		check_variable = { var = famine_pressure_request_actor_proven value = constant:humanitarian_runtime.zero compare = greater_than }
	}
}


# Every condition is binding. A caller must provide the explicit physical and
# political proof in addition to the engine's owner/controller war state.
famine_blockade_proof = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		OR = {
			OWNER = { has_war = yes }
			CONTROLLER = { has_war = yes }
		}
		check_variable = { var = famine_blockade_island_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = famine_blockade_isolation_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = famine_blockade_maritime_dependence_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		OR = {
			check_variable = { var = famine_blockade_route_disrupted_proven value = constant:humanitarian_runtime.zero compare = greater_than }
			check_variable = { var = famine_blockade_port_disrupted_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		}
		OR = {
			check_variable = { var = famine_blockade_convoy_shortage_proven value = constant:humanitarian_runtime.zero compare = greater_than }
			check_variable = { var = famine_blockade_escort_shortage_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		}
		check_variable = { var = famine_blockade_no_humanitarian_corridor_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = famine_blockade_local_food_insufficient_proven value = constant:humanitarian_runtime.zero compare = greater_than }
	}
}


# A state is eligible for the scheduled evaluator only while an owner has
# deliberately submitted a context or a pressure request. This keeps the
# registry sparse and avoids a world scan hidden inside a trigger.
famine_state_has_active_context = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		OR = {
			has_state_flag = famine_food_security_active
			has_state_flag = famine_surface_context_active
			has_state_flag = famine_profile_context_active
			check_variable = { var = famine_food_pressure value = constant:humanitarian_population.zero compare = greater_than }
		}
	}
}

		}
	}
}


famine_state_can_retire = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		OR = {
			NOT = { has_variable = famine_food_stage }
			check_variable = { var = famine_food_stage value = constant:famine_food_stage.stable compare = equals }
		}
		NOT = { check_variable = { var = famine_food_pressure value = constant:humanitarian_population.zero compare = greater_than } }
		NOT = { check_variable = { var = famine_reception_demand_population value = constant:humanitarian_population.zero compare = greater_than } }
		NOT = { has_state_flag = famine_blockade_candidate_active }
	}
}
