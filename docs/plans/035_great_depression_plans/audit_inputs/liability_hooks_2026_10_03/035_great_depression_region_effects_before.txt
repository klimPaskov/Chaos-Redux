# Identity validation uses the exact calling country. ID-only input is resolved
# as a literal state scope, never as a variable scope existence test.
great_depression_validate_inherited_region = {
	great_depression_reset_region_outputs = yes
	clr_country_flag = great_depression_region_validated
	clr_country_flag = great_depression_region_duplicate
	clr_country_flag = great_depression_region_state_verified

	set_variable = { great_depression_region_existing_center = constant:great_depression_region_contract.unavailable }

		limit = { NOT = { check_variable = { var = great_depression_region_call_contract_supplied value = constant:great_depression_region_contract.version compare = equals } } }
		set_variable = { great_depression_region_call_reject_reason = constant:great_depression_region_reject.missing_contract }
	}
	else_if = {
		limit = {
			NOT = {
				AND = {
					exists = yes
					uses_normal_civilian_systems = yes
					NOT = { is_special_chaos_country = yes }
					NOT = { is_actual_nonhuman_country = yes }
					has_variable = great_depression_region_call_country_id
					check_variable = { var = great_depression_region_call_country_id value = THIS.id compare = equals }
				}
			}
		}
		set_variable = { great_depression_region_call_reject_reason = constant:great_depression_region_reject.invalid_country }
	}
	else = {
		# A target-only contract may supply the exact state identity here.

			limit = {
				NOT = { has_variable = great_depression_region_call_state_id }

			event_target:great_depression_inherited_region_state = {
				if = {
					limit = { is_in_array = { array = global.states value = THIS } }

			limit = {
				has_variable = great_depression_region_call_state_id
				check_variable = { var = great_depression_region_call_state_id value = constant:great_depression_api.zero compare = greater_than }
				check_variable = { var = great_depression_region_call_state_id value = great_depression_region_integer_state compare = equals }
				# Built-in state membership is checked before literal scope construction.
				is_in_array = { array = global.states value = great_depression_region_call_state_id }
				OR = {
					NOT = { has_country_flag = great_depression_region_call_state_target_supplied }
					AND = {
						has_event_target = great_depression_inherited_region_state
						event_target:great_depression_inherited_region_state = {
							is_in_array = { array = global.states value = THIS }
							check_variable = { var = id value = PREV.great_depression_region_call_state_id compare = equals }
						}
					}
				}
			}
			meta_effect = {
				text = {
					[STATE] = {
						save_event_target_as = great_depression_inherited_region_state
						PREV = {
							set_variable = { great_depression_region_resolved_state = PREV }
							set_country_flag = great_depression_region_state_verified
						}
					}
				}
				STATE = "[?great_depression_region_call_state_id|.0]"
			}
		}
		if = {
			limit = { NOT = { has_country_flag = great_depression_region_state_verified } }
			set_variable = { great_depression_region_call_reject_reason = constant:great_depression_region_reject.invalid_state }
		}
		else_if = {
			limit = {
				NOT = {
					AND = {
						check_variable = { var = great_depression_region_call_episode_id value = constant:great_depression_api.minimum_episode_id compare = greater_than_or_equals }
						check_variable = { var = great_depression_region_call_event34_episode_id value = constant:great_depression_api.minimum_episode_id compare = greater_than_or_equals }
						check_variable = { var = great_depression_region_call_receipt_id value = constant:great_depression_api.minimum_receipt_id compare = greater_than_or_equals }
						check_variable = { var = great_depression_region_call_collapse_receipt_id value = constant:great_depression_api.minimum_receipt_id compare = greater_than_or_equals }
					}
				}
			}
			set_variable = { great_depression_region_call_reject_reason = constant:great_depression_region_reject.invalid_source }
		}

			}

			set_temp_variable = { great_depression_region_state_military = building_level@arms_factory }
			PREV = {
				set_variable = { great_depression_region_civilian_baseline = great_depression_region_state_civilian }
				set_variable = { great_depression_region_military_baseline = great_depression_region_state_military }
			}
			if = {
				limit = { is_owned_by = PREV }
				PREV = { set_variable = { great_depression_region_live_control = constant:great_depression_region_control.temporary_loss } }

					limit = { is_controlled_by = PREV }
					PREV = { set_variable = { great_depression_region_live_control = constant:great_depression_region_control.connected } }
				}
			}
		}
		great_depression_resolve_inherited_region_stage = yes
		if = {
			limit = { check_variable = { var = great_depression_region_call_role value = constant:great_depression_region_role.none compare = equals } }
			set_variable = { great_depression_region_disposition = constant:great_depression_region_result.ignored }
		}
		else_if = {
			limit = {
				event_target:great_depression_inherited_region_state = {
					is_owned_by = PREV
					is_controlled_by = PREV
					is_core_of = PREV
					impassable = no
					state_population_k > constant:great_depression_api.minimum_viable_state_population_k
					OR = { industrial_complex > 0 arms_factory > 0 dockyard > 0 }
					NOT = { has_state_flag = industrial_boom_region_conversion_forbidden }
					PREV = { NOT = { has_country_flag = great_depression_region_call_conversion_forbidden } }
					OR = {
						NOT = { has_state_flag = great_depression_center }
						check_variable = { var = great_depression_center_episode_id value = PREV.great_depression_active_episode_id compare = equals }
					}
				}
			}
			if = {
				limit = { is_in_array = { array = great_depression_center_state_ids value = great_depression_region_resolved_state } }
				for_loop_effect = {
					start = 0
					end = great_depression_center_state_ids^num
					value = great_depression_region_center_scan_index
					if = {
						limit = { check_variable = { var = great_depression_center_state_ids^great_depression_region_center_scan_index value = great_depression_region_resolved_state compare = equals } }
						set_variable = { great_depression_region_existing_center = great_depression_region_center_scan_index }
					}
				}
				set_variable = { great_depression_region_disposition = constant:great_depression_region_result.merged }
				if = {
					limit = { has_country_flag = great_depression_region_precommit_new }
					# The initial selector created this row in the same new transaction.
					set_variable = { great_depression_region_disposition = constant:great_depression_region_result.created }
				}
			}
			else_if = {
				limit = { check_variable = { var = great_depression_center_count value = great_depression_region_visible_cap compare = less_than } }
				set_variable = { great_depression_region_existing_center = great_depression_center_count }
				great_depression_append_inherited_center = yes
				set_variable = { great_depression_region_disposition = constant:great_depression_region_result.created }
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = great_depression_region_disposition value = constant:great_depression_region_result.created compare = equals }
					check_variable = { var = great_depression_region_disposition value = constant:great_depression_region_result.merged compare = equals }
				}
			}
			great_depression_apply_inherited_center_state = yes
			set_variable = { great_depression_region_call_center_index = great_depression_region_existing_center }
		}
		set_variable = { great_depression_region_call_result = great_depression_region_disposition }
		add_to_array = { array = great_depression_inherited_region_episode_id_entries value = great_depression_region_call_episode_id }
		add_to_array = { array = great_depression_inherited_region_event34_episode_id_entries value = great_depression_region_call_event34_episode_id }
		add_to_array = { array = great_depression_inherited_region_state_id_entries value = great_depression_region_call_state_id }
		add_to_array = { array = great_depression_inherited_region_role_entries value = great_depression_region_call_role }
		add_to_array = { array = great_depression_inherited_region_project_profile_entries value = great_depression_region_call_project_profile }
		add_to_array = { array = great_depression_inherited_region_project_stage_entries value = great_depression_region_call_project_stage }
		add_to_array = { array = great_depression_inherited_region_protection_entries value = great_depression_region_call_protection }
		add_to_array = { array = great_depression_inherited_region_fragility_entries value = great_depression_region_call_fragility }
		add_to_array = { array = great_depression_inherited_region_control_entries value = great_depression_region_call_control }
		add_to_array = { array = great_depression_inherited_region_damage_entries value = great_depression_region_call_damage }
		add_to_array = { array = great_depression_inherited_region_receipt_id_entries value = great_depression_region_call_receipt_id }
		add_to_array = { array = great_depression_inherited_region_collapse_receipt_id_entries value = great_depression_region_call_collapse_receipt_id }
		add_to_array = { array = great_depression_inherited_region_legacy_receipt_id_entries value = great_depression_region_call_legacy_receipt_id }
		add_to_array = { array = great_depression_inherited_region_corridor_receipt_id_entries value = great_depression_region_call_corridor_receipt_id }
		add_to_array = { array = great_depression_inherited_region_exhaustion_entries value = great_depression_region_call_exhaustion }
		add_to_array = { array = great_depression_inherited_region_result_entries value = great_depression_region_disposition }
		add_to_array = { array = great_depression_inherited_region_center_index_entries value = great_depression_region_call_center_index }
		add_to_array = { array = great_depression_inherited_region_stage_entries value = great_depression_region_resolved_stage }
		add_to_array = { array = great_depression_inherited_region_live_control_entries value = great_depression_region_live_control }
		add_to_array = { array = great_depression_inherited_region_civilian_baseline_entries value = great_depression_region_civilian_baseline }
		add_to_array = { array = great_depression_inherited_region_military_baseline_entries value = great_depression_region_military_baseline }
		if = {
			limit = { check_variable = { var = great_depression_region_disposition value = constant:great_depression_region_result.hidden compare = equals } }
			set_country_flag = great_depression_inherited_hidden_liability
		}
		if = {
			limit = { check_variable = { var = great_depression_region_live_control value = constant:great_depression_region_control.connected compare = greater_than } }
			set_variable = { great_depression_region_call_disconnected = constant:great_depression_api.one }
			set_country_flag = great_depression_inherited_disconnected_liability
		}
	}
	set_variable = { great_depression_region_call_accepted = constant:great_depression_api.one }
	set_variable = { great_depression_region_call_rejected = constant:great_depression_api.zero }
	if = { limit = { check_variable = { var = great_depression_region_call_result value = constant:great_depression_region_result.created compare = equals } } set_variable = { great_depression_region_call_created = constant:great_depression_api.one } }
	if = { limit = { check_variable = { var = great_depression_region_call_result value = constant:great_depression_region_result.merged compare = equals } } set_variable = { great_depression_region_call_merged = constant:great_depression_api.one } }

great_depression_apply_inherited_center_state = {
	# A precommit new center receives the exact mapped opening. A later merge
	# cannot erase an existing treatment, work receipt, or worse local outcome.
	if = {
		limit = {
			NOT = { has_country_flag = great_depression_region_precommit_new }
			check_variable = { var = great_depression_center_stage_entries^great_depression_region_existing_center value = great_depression_region_resolved_stage compare = greater_than }
			OR = {
				check_variable = { var = great_depression_center_stage_entries^great_depression_region_existing_center value = constant:great_depression_center_stage.abandoned compare = less_than_or_equals }
				check_variable = { var = great_depression_center_stage_entries^great_depression_region_existing_center value = constant:great_depression_center_stage.liquidated compare = equals }
				check_variable = { var = great_depression_center_stage_entries^great_depression_region_existing_center value = constant:great_depression_center_stage.scarred compare = equals }
			}
		}
		set_variable = { great_depression_region_resolved_stage = great_depression_center_stage_entries^great_depression_region_existing_center }
	}
	set_variable = { great_depression_center_stage_entries^great_depression_region_existing_center = great_depression_region_resolved_stage }
	event_target:great_depression_inherited_region_state = {
		if = {
			limit = {
				OR = { has_state_flag = great_depression_center_reopened has_state_flag = great_depression_center_restructured }
				check_variable = { var = PREV.great_depression_region_resolved_stage value = constant:great_depression_center_stage.shuttered compare = less_than_or_equals }
			}
			# A fresh proven collapse invalidates current reopening proof. Paid
			# project progress, completed work and physical-loss receipts survive.
			clr_state_flag = great_depression_center_positive_resolution_receipt
		}
		set_variable = { great_depression_center_registry_index = PREV.great_depression_region_existing_center }
		set_variable = { great_depression_center_episode_id = PREV.great_depression_active_episode_id }
		set_variable = { great_depression_center_stage = PREV.great_depression_region_resolved_stage }
		set_variable = { great_depression_center_ledger_stage = PREV.great_depression_region_resolved_stage }
		set_state_flag = great_depression_center
		set_state_flag = great_depression_inherited_center
		set_variable = { great_depression_inherited_region_episode_id = PREV.great_depression_region_call_episode_id }
		set_variable = { great_depression_inherited_region_event34_episode_id = PREV.great_depression_region_call_event34_episode_id }
		set_variable = { great_depression_inherited_region_state_id = PREV.great_depression_region_call_state_id }
		set_variable = { great_depression_inherited_region_role = PREV.great_depression_region_call_role }
		set_variable = { great_depression_inherited_region_project_profile = PREV.great_depression_region_call_project_profile }
		set_variable = { great_depression_inherited_region_project_stage = PREV.great_depression_region_call_project_stage }
		set_variable = { great_depression_inherited_region_protection = PREV.great_depression_region_call_protection }
		set_variable = { great_depression_inherited_region_fragility = PREV.great_depression_region_call_fragility }
		set_variable = { great_depression_inherited_region_control = PREV.great_depression_region_call_control }
		set_variable = { great_depression_inherited_region_damage = PREV.great_depression_region_call_damage }
		set_variable = { great_depression_inherited_region_receipt_id = PREV.great_depression_region_call_receipt_id }
		set_variable = { great_depression_inherited_region_collapse_receipt_id = PREV.great_depression_region_call_collapse_receipt_id }
		set_variable = { great_depression_inherited_region_legacy_receipt_id = PREV.great_depression_region_call_legacy_receipt_id }
		set_variable = { great_depression_inherited_region_corridor_receipt_id = PREV.great_depression_region_call_corridor_receipt_id }
		set_variable = { great_depression_inherited_region_exhaustion = PREV.great_depression_region_call_exhaustion }
		# Source protection is salvage memory, not a free Event 35 treatment.
		great_depression_clear_center_physical_stage = yes
		if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.distressed compare = equals } }
			set_state_flag = great_depression_center_distressed
			add_dynamic_modifier = { modifier = great_depression_center_distressed }
		}
		else_if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.idled compare = equals } }
			set_state_flag = great_depression_center_idled
			add_dynamic_modifier = { modifier = great_depression_center_idled }
		}
		else_if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.shuttered compare = equals } }
			set_state_flag = great_depression_center_shuttered
			add_dynamic_modifier = { modifier = great_depression_center_shuttered }
		}
		else_if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.abandoned compare = equals } }
			set_state_flag = great_depression_center_abandoned
			add_dynamic_modifier = { modifier = great_depression_center_abandoned }
		}
		else_if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.liquidated compare = equals } }
			set_state_flag = great_depression_center_liquidated
			add_dynamic_modifier = { modifier = great_depression_center_hollowed }
		}
		else_if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.reopened compare = equals } }
			set_state_flag = great_depression_center_reopened
			add_dynamic_modifier = { modifier = great_depression_center_reopened }
		}
		else_if = {
			limit = { check_variable = { var = great_depression_center_stage value = constant:great_depression_center_stage.restructured compare = equals } }
			set_state_flag = great_depression_center_restructured
		}
		else = { set_state_flag = great_depression_center_scarred }
	}
}
