# State. Fresh decision/tooltip eligibility for the same native living row used
# by the exact transfer. The effect keeps its own final preflight and receipt.
camp_occ_manual_transfer_live_source_ready = {
	camp_occ_manual_transfer_source_ready = yes
	set_temp_variable = { camp_occ_manual_source_margin_people = state_population_k }
	subtract_from_temp_variable = { camp_occ_manual_source_margin_people = constant:camp_occ_value.minimum_population_k }
	if = { limit = { has_variable = camp_admin_detainees_k } subtract_from_temp_variable = { camp_occ_manual_source_margin_people = camp_admin_detainees_k } }
	multiply_temp_variable = { camp_occ_manual_source_margin_people = constant:camp_occ_value.people_per_k }
	round_temp_variable = camp_occ_manual_source_margin_people
	subtract_from_temp_variable = { camp_occ_manual_source_margin_people = constant:camp_occ_value.one_person }
	check_variable = { var = camp_occ_manual_source_margin_people value = constant:camp_occ_value.manual_order_minimum_people compare = greater_than_or_equals }
	set_temp_variable = { camp_occ_manual_source_aggregate_people = camp_occ_remaining_k }
	multiply_temp_variable = { camp_occ_manual_source_aggregate_people = constant:camp_occ_value.people_per_k }
	check_variable = { var = camp_occ_manual_source_aggregate_people value = constant:camp_occ_value.manual_order_minimum_people compare = greater_than_or_equals }
	var:camp_occ_responsible_country = {
		check_variable = { camp_occ_row_root_ids^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_origins^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_hosts^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_authorities^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_profiles^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_amounts_k^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_deaths_k^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_removed_k^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_released_k^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_moved_out_k^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_moved_in_k^num = camp_occ_row_ids^num }
		check_variable = { camp_occ_row_statuses^num = camp_occ_row_ids^num }
		any_of = {
			array = camp_occ_row_ids
			value = camp_occ_manual_row_probe_id
			index = camp_occ_manual_row_probe_index
			check_variable = { var = camp_occ_manual_row_probe_id value = PREV.camp_occ_native_row_id compare = equals }
			check_variable = { camp_occ_row_statuses^camp_occ_manual_row_probe_index = constant:camp_occ_row_status.active }
			check_variable = { camp_occ_row_root_ids^camp_occ_manual_row_probe_index > 0 }
			var:camp_occ_row_hosts^camp_occ_manual_row_probe_index = { state = PREV.PREV }
			var:camp_occ_row_authorities^camp_occ_manual_row_probe_index = { tag = PREV }
			var:camp_occ_row_origins^camp_occ_manual_row_probe_index = { has_state_flag = camp_occ_registered }
			set_temp_variable = { camp_occ_manual_row_probe_people = camp_occ_row_amounts_k^camp_occ_manual_row_probe_index }
			check_variable = { camp_occ_manual_row_probe_people > 0 }
			multiply_temp_variable = { camp_occ_manual_row_probe_people = constant:camp_occ_value.manual_auschwitz_living_share }
			multiply_temp_variable = { camp_occ_manual_row_probe_people = constant:camp_occ_value.people_per_k }
			set_temp_variable = { camp_occ_manual_row_probe_ceiling = camp_occ_manual_row_probe_people }
			round_temp_variable = camp_occ_manual_row_probe_people
			if = {
				limit = { check_variable = { camp_occ_manual_row_probe_people > camp_occ_manual_row_probe_ceiling } }
				subtract_from_temp_variable = { camp_occ_manual_row_probe_people = constant:camp_occ_value.one_person }
			}
			check_variable = { var = camp_occ_manual_row_probe_people value = constant:camp_occ_value.manual_order_minimum_people compare = greater_than_or_equals }
		}
	}
}
