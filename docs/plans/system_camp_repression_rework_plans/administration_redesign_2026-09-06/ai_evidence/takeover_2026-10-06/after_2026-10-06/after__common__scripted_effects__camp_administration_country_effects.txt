# A one-time German detention-site establishment in the package's Krakow-area state.
# It charges the Event 026 sale price of genocide_decision_cost.build_concentration.
# The later killing-role conversion retains its separate radical mandate, quote and consequences.
camp_admin_ger_establish_krakow_site = {
	if = {
		limit = {
			camp_admin_ger_krakow_setup_eligible = yes
			check_variable = { var = political_power value = global.black_friday_price_genocide_decision_cost_build_concentration compare = greater_than_or_equals }
		}
		set_temp_variable = { camp_admin_krakow_setup_debit = global.black_friday_price_genocide_decision_cost_build_concentration }
		multiply_temp_variable = { camp_admin_krakow_setup_debit = constant:camp_admin_value.negative_one }
		add_political_power = camp_admin_krakow_setup_debit
		hidden_effect = { black_friday_credit_political_power_purchase = yes }
		88 = {
			set_variable = { genocide_responsible_country = ROOT }
			set_temp_variable = { genocide_requested_concentration_level = constant:genocide_camp_level.standard }
			genocide_ensure_concentration_camp_level = yes
			set_temp_variable = { camp_register_site_type = constant:camp_rework_site_type.detention }
			camp_rework_register_active_site = yes
			set_state_flag = camp_admin_ger_krakow_site_established
		}
		set_country_flag = camp_admin_ger_krakow_site_established
		set_temp_variable = { camp_admin_record_kind = constant:camp_admin_history_kind.site_established }
		set_temp_variable = { camp_admin_record_state_id = 88 }
		camp_admin_record_history = yes
		camp_admin_refresh_interface = yes
	}
}

camp_admin_country_start_program = {
	if = {
		limit = { NOT = { has_country_flag = camp_admin_institution_slot_active } NOT = { has_country_flag = camp_admin_country_engineer_release_pending } camp_admin_country_start_input_valid = yes }
		set_variable = { camp_admin_country_program = camp_admin_start_program_input }
		set_variable = { camp_admin_institution_progress = 0 }
		set_variable = { camp_admin_institution_target = constant:camp_admin_country.program_months }
		clear_variable = camp_admin_institution_state_id
		for_each_scope_loop = {
			array = camp_active_site_states
			if = {
				limit = {
					camp_admin_country_program_site_ready = yes
					OR = { PREV = { NOT = { original_tag = SOV } } gulag_labor_camp_network > 0 }
					PREV = { NOT = { has_variable = camp_admin_institution_state_id } }
				}
				PREV = { set_variable = { camp_admin_institution_state_id = PREV } }
			}
		}
		if = {
			limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.ger_auschwitz } }
			set_variable = { camp_admin_institution_state_id = 88 }
		}
		else_if = {
			limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_pingfang } }
			set_variable = { camp_admin_institution_state_id = 328 }
		}
		else_if = {
			limit = { original_tag = JAP }
			clear_variable = camp_admin_institution_state_id
			for_each_scope_loop = {
				array = camp_active_site_states
				if = {
					limit = {
						camp_admin_country_program_site_ready = yes
						camp_admin_country_china_geography = yes
						PREV = { NOT = { has_variable = camp_admin_institution_state_id } }
						OR = { PREV = { NOT = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_transport } } } camp_admin_country_census_jap_industry_host = yes }
					}
					PREV = { set_variable = { camp_admin_institution_state_id = PREV } }
				}
			}
		}
		if = {
			limit = { has_variable = camp_admin_institution_state_id var:camp_admin_institution_state_id = { camp_admin_country_program_site_ready = yes } }
			set_country_flag = camp_admin_institution_slot_active
			if = {
				limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.sov_engineer_release } }
				var:camp_admin_institution_state_id = {
					set_temp_variable = { camp_admin_engineer_cohort_k = camp_admin_detainees_k }
					clamp_temp_variable = { var = camp_admin_engineer_cohort_k min = 0 max = constant:camp_admin_country.engineer_cohort_k }
				}
				set_variable = { camp_admin_country_engineer_cohort_k = camp_admin_engineer_cohort_k }
			}
		}
		else = { set_variable = { camp_admin_country_program = constant:camp_admin_country_program.none } clear_variable = camp_admin_institution_state_id }
	}
	set_temp_variable = { camp_admin_start_program_input = constant:camp_admin_value.zero }
}

			OR = { NOT = { has_variable = camp_admin_country_ai_policy_next_month } check_variable = { var = camp_admin_month_key value = camp_admin_country_ai_policy_next_month compare = greater_than_or_equals } }
		}
		set_temp_variable = { camp_admin_ai_maintenance_k = 0 }
		for_each_scope_loop = {
			array = camp_active_site_states
			if = { limit = { camp_admin_site_is_operational = yes } add_to_temp_variable = { camp_admin_ai_maintenance_k = camp_admin_detainees_k } }
		}
		for_each_scope_loop = { array = camp_admin_civilian_states add_to_temp_variable = { camp_admin_ai_maintenance_k = camp_admin_civilian_workforce_k } }
		multiply_temp_variable = { camp_admin_ai_maintenance_k = constant:camp_admin_value.maintenance_support_per_k }
		set_temp_variable = { camp_admin_ai_low_stock = camp_admin_ai_maintenance_k }
		add_to_temp_variable = { camp_admin_ai_low_stock = constant:camp_admin_value.support_reserve }
		set_temp_variable = { camp_admin_ai_high_stock = camp_admin_ai_maintenance_k }
		add_to_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_value.project_support }
		add_to_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_country.monthly_support }
		multiply_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_country.ai_commitment_buffer_months }
		add_to_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_value.support_reserve }
		set_temp_variable = { camp_admin_ai_budget_before = camp_admin_budget }
		if = {
			limit = { check_variable = { camp_admin_budget > constant:camp_admin_budget.maintenance } check_variable = { num_equipment@support_equipment < camp_admin_ai_low_stock } }
			set_temp_variable = { camp_admin_control_requested = constant:camp_admin_budget.maintenance }
			camp_admin_set_budget = yes
		}
		else_if = {
			limit = {
				check_variable = { camp_admin_budget = constant:camp_admin_budget.maintenance }
				check_variable = { var = num_equipment@support_equipment value = camp_admin_ai_high_stock compare = greater_than_or_equals }
				OR = {
					camp_admin_country_program_available = yes
					camp_admin_country_archives_available = yes
					camp_admin_ger_krakow_setup_context = yes
					any_of_scopes = { array = camp_active_site_states camp_admin_has_supported_workforce = yes check_variable = { camp_admin_development_level < constant:camp_admin_value.max_development } }
				}
			}
			set_temp_variable = { camp_admin_control_requested = constant:camp_admin_budget.development }
			camp_admin_set_budget = yes
		}
		if = {
			limit = { NOT = { check_variable = { camp_admin_budget = camp_admin_ai_budget_before } } }
			set_variable = { camp_admin_country_ai_policy_next_month = camp_admin_month_key }
			add_to_variable = { camp_admin_country_ai_policy_next_month = constant:camp_admin_country.ai_policy_hold_months }
		}
	}
