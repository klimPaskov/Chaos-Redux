	camp_admin_country_regime_valid = yes
	NOT = { has_country_flag = camp_occ_country_closed }
	OR = {
		AND = { original_tag = JAP date > 1937.7.6 check_variable = { camp_admin_mandate > 1 } }
		AND = { camp_admin_country_nazi_authority = yes date > 1941.6.21 check_variable = { camp_admin_mandate = 3 } }
	}
}

camp_occ_state_has_current_authority = {
	has_state_flag = camp_occ_registered

	has_variable = camp_occ_responsible_country

# Country and selected-state gates for the paid German exact-cohort transfer.
# The transaction rechecks row alignment, physical headroom and destination bind.
camp_occ_manual_transfer_country_ready = {
	has_country_flag = camp_admin_initialized
	original_tag = GER
	camp_occ_authority_has_campaign = yes
	has_war = yes
	date > 1943.5.29
	NOT = { has_country_flag = camp_rework_expansion_frozen }
	OR = {
		germany_mengele_program_active = yes
		germany_mengele_restricted_program_active = yes
	}
	has_variable = mengele_permission_level
	check_variable = { mengele_permission_level > constant:germany_mengele_permission.rejected }
	has_variable = camp_occ_manual_source_state_id
	var:camp_occ_manual_source_state_id = { camp_occ_manual_transfer_source_ready = yes }
	88 = { camp_occ_manual_transfer_destination_ready = yes }
}

	state = 88

camp_occ_manual_transfer_source_ready = {
	camp_occ_state_has_current_authority = yes
	has_variable = camp_occ_native_row_id
	check_variable = { camp_occ_native_row_id > 0 }
	check_variable = { camp_occ_remaining_k > constant:camp_occ_value.minimum_population_k }
	NOT = { state = 88 }
	is_on_continent = europe
	NOT = { is_core_of = GER }
	NOT = { has_state_flag = camp_occ_custody_rollback_failed }
	NOT = { has_state_flag = camp_occ_transfer_rollback_failed }
	var:camp_occ_responsible_country = { original_tag = GER }
}
