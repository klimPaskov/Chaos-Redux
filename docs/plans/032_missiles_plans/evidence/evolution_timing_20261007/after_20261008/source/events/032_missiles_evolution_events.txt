add_namespace = chaosx.nr32

# Native Event 032 MTTH clocks. One living host evaluates stored observation
# data. Rejected attempts remain re-armable; successful global unlock flags,
# not fire_only_once, enforce the shared once-only log contract.
# The one-day base is the unit for the variable day scale below.
@missiles_mtth_unit_days = 1

country_event = {
	id = chaosx.nr32.201
	hidden = yes
	is_triggered_only = no
	trigger = {
		missiles_evolution_clock_host_is_current = yes
		has_global_flag = missiles_evolution_rogue_launch_commands_pending
		has_global_flag = missiles_evolution_rogue_launch_commands_native_pending
		NOT = { has_global_flag = missiles_evolution_rogue_launch_commands_short_pending }
		NOT = { has_variable = global.missiles_evolution_rogue_launch_commands_due_date }
		NOT = { has_global_flag = missiles_evolution_rogue_launch_commands_unlocked }
		NOT = { has_global_flag = events_log_disabled_evolution_32_32_4 }
		missiles_evolution_rogue_launch_commands_is_eligible = yes
	}
	mean_time_to_happen = {
		days = @missiles_mtth_unit_days
		modifier = { factor = global.missiles_evolution_rogue_mtth_days }
	}
	immediate = {
		set_temp_variable = { missiles_evolution_selection = constant:missiles_evolution.rogue_commands }
		missiles_attempt_native_evolution_unlock = yes
	}
}
