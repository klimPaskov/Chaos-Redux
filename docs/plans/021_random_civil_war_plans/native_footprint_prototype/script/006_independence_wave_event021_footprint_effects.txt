# Input: the same caller/package and a fresh matched blueprint query.
# Caller-owned outputs: native_status, native_count and temporary arrays
# native_states, prior_controllers, source_core_states (event021_release_ prefix).
# Every source-owned carrier core is recorded, including occupied states.
# Source-core membership is a sparse exact subset; prior controllers are
# aligned one-for-one with native_states. No eligibility/resource shortcut is
# inferred from ownership, and no state or country flags are written.
independence_wave_collect_event021_native_footprint = {
	set_temp_variable = { event021_release_native_status = constant:event021_release_query_status.identity_mismatch }
	set_temp_variable = { event021_release_native_count = constant:random_civil_war_value.zero }
	clear_temp_array = event021_release_native_states
	clear_temp_array = event021_release_prior_controllers
	clear_temp_array = event021_release_source_core_states
	if = {
		limit = {
			exists = yes
			has_variable = independence_wave_event021_adapter_package_id
			check_variable = { var = event021_release_query_status value = constant:event021_release_query_status.matched compare = equals }
			check_variable = { var = event021_release_queried_package_id value = independence_wave_event021_adapter_package_id compare = equals }
			has_event_target = event021_release_carrier_blueprint
			event_target:event021_release_carrier_blueprint = { check_variable = { var = id value = event021_release_carrier_id compare = equals } }
		}
		save_event_target_as = event021_release_footprint_source
		# Native release is a no-op for an already living static carrier.
		# Explicit living-recipient increments are planned separately by Event 021.
		if = {
			limit = { event_target:event021_release_carrier_blueprint = { exists = no } }
			every_owned_state = {
				limit = { is_core_of = event_target:event021_release_carrier_blueprint }
				add_to_temp_array = { array = event021_release_native_states value = THIS }
				controller = { add_to_temp_array = { array = event021_release_prior_controllers value = THIS } }
				if = {
					limit = { is_core_of = event_target:event021_release_footprint_source }
					add_to_temp_array = { array = event021_release_source_core_states value = THIS }
				}
			}
		}
		set_temp_variable = { event021_release_native_count = event021_release_native_states^num }
		set_temp_variable = { event021_release_native_status = constant:event021_release_query_status.snapshot_incomplete }
		if = {
			limit = { check_variable = { var = event021_release_prior_controllers^num value = event021_release_native_count compare = equals } }
			set_temp_variable = { event021_release_native_status = constant:event021_release_query_status.matched }
		}
	}
}



# COUNTRY scope on one living original-tag identity. The resolver caller owns
# live_matches and live_identity_statuses before this nested call. Record all
# occupied identities, including unattested or nonhuman ones, so none can be
# mistaken for a free dormant carrier. No actor setup or public alias is touched.
independence_wave_record_event021_live_carrier_match = {
	if = {
		limit = { NOT = { is_in_array = { array = event021_release_live_matches value = THIS } } }
		add_to_temp_array = { array = event021_release_live_matches value = THIS }
		set_temp_variable = { event021_release_match_identity_status = constant:event021_release_query_status.identity_mismatch }
		if = {
			limit = {
				has_variable = independence_wave_package_id
				check_variable = { var = independence_wave_package_id value = event021_release_queried_package_id compare = equals }
				NOT = { is_actual_nonhuman_country = yes }
			}
			set_temp_variable = { event021_release_match_identity_status = constant:event021_release_query_status.matched }
		}
		add_to_temp_array = { array = event021_release_live_identity_statuses value = event021_release_match_identity_status }
	}
}
