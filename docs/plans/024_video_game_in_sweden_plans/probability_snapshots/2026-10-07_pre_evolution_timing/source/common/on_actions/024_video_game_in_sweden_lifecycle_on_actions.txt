# ============================================================================
# CHAOS REDUX - EVENT 024 TRANSACTION-TIME LIFECYCLE RECEIPTS
# ============================================================================
# Bounded actor/state callbacks preserve continuous-loss disqualifiers and
# close an invalid Swedish program, and the capitulation callback records a
# Trust the Model opponent's defeat at the moment it happens. They never
# iterate the world or refire it.
# Native contexts: state transfer ROOT=new controller, FROM=old controller,
# FROM.FROM=state; capitulation ROOT=loser; annexation FROM=annexed country.
# ============================================================================

on_actions = {
	on_state_control_changed = {
		effect = {
			FROM = { video_game_in_sweden_record_control_loss = yes }
		}
	}
	on_capitulation = {
		effect = {
			video_game_in_sweden_record_actor_loss = yes
			video_game_in_sweden_record_opponent_capitulation = yes
		}
	}
	on_annex = {
		effect = { FROM = { video_game_in_sweden_record_actor_loss = yes } }
	}
}
