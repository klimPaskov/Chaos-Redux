# Country-scope emergency quote, shared by payment and numeric localisation.
# Outputs: video_game_in_sweden_restriction_political_power_quote,
# video_game_in_sweden_restriction_stability_quote and
# video_game_in_sweden_restriction_war_support_quote. Callers initialize each
# consumed output before entry so its storage survives the helper's return.
# Available-reserve scratch values are local. No persistent state changes.
video_game_in_sweden_quote_restriction = {
	# A failed field validation, audit or dual-track verification halves every
	# nominal charge through the stored fee factor.
	set_temp_variable = { video_game_in_sweden_restriction_fee_scale = constant:video_game_in_sweden_effect.restriction_full_fee_factor }
	OR = {
		NOT = { has_variable = video_game_in_sweden_restriction_fee_factor }
		set_temp_variable = { video_game_in_sweden_restriction_fee_scale = video_game_in_sweden_restriction_fee_factor }
	}
	set_temp_variable = { video_game_in_sweden_restriction_nominal_fee = constant:video_game_in_sweden_cost.restriction_political_power }
	multiply_temp_variable = { video_game_in_sweden_restriction_nominal_fee = video_game_in_sweden_restriction_fee_scale }
	set_temp_variable = { video_game_in_sweden_restriction_political_power_quote = has_political_power }
	clamp_temp_variable = {
		var = video_game_in_sweden_restriction_political_power_quote
		min = constant:video_game_in_sweden_value.zero
		max = video_game_in_sweden_restriction_nominal_fee
	}
	set_temp_variable = { video_game_in_sweden_restriction_available_stability = has_stability }
	clamp_temp_variable = { var = video_game_in_sweden_restriction_available_stability min = constant:video_game_in_sweden_value.zero max = constant:video_game_in_sweden_value.one }
	set_temp_variable = { video_game_in_sweden_restriction_stability_quote = constant:video_game_in_sweden_effect.restriction_stability_loss }
	multiply_temp_variable = { video_game_in_sweden_restriction_stability_quote = -1 }
	multiply_temp_variable = { video_game_in_sweden_restriction_stability_quote = video_game_in_sweden_restriction_fee_scale }
	clamp_temp_variable = {
		var = video_game_in_sweden_restriction_stability_quote
		min = constant:video_game_in_sweden_value.zero
		max = video_game_in_sweden_restriction_available_stability
	}
	set_temp_variable = { video_game_in_sweden_restriction_available_war_support = has_war_support }
	clamp_temp_variable = { var = video_game_in_sweden_restriction_available_war_support min = constant:video_game_in_sweden_value.zero max = constant:video_game_in_sweden_value.one }
	set_temp_variable = { video_game_in_sweden_restriction_war_support_quote = constant:video_game_in_sweden_effect.restriction_war_support_loss }
	multiply_temp_variable = { video_game_in_sweden_restriction_war_support_quote = -1 }
	multiply_temp_variable = { video_game_in_sweden_restriction_war_support_quote = video_game_in_sweden_restriction_fee_scale }
	clamp_temp_variable = {
		var = video_game_in_sweden_restriction_war_support_quote
		min = constant:video_game_in_sweden_value.zero
		max = video_game_in_sweden_restriction_available_war_support
	}
	always = yes
}
