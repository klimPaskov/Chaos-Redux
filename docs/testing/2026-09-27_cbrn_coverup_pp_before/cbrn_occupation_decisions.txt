# ============================================================================
# CHAOS REDUX - CBRN OCCUPATION DECISIONS
# ============================================================================
# Overview:
# - Authorizes protective occupation administration, delivers real
#   population-scaled respirator aid from a foreign stockpile, and resolves
#   exact-record concealment or disclosure choices.
# - The hidden coercive-authorization token is retained for occupation-law
#   compatibility. Sarin and Soman attacks are not decision actions.
# - Aid debits real stock, while record choices modify only their exact
#   historical chemical record and its consequences.
# ============================================================================

		}

		}
		custom_cost_trigger = { hidden_trigger = { cbrn_occupation_external_aid_payment_ready = yes } }
		custom_cost_text = cbrn_deploy_protective_aid_cost

		complete_effect = {
			custom_effect_tooltip = cbrn_deploy_protective_aid_begin_tt
			hidden_effect = {
				FROM = {
					set_temp_variable = { cbrn_mask_reservation_quote = constant:cbrn_math.zero }
					if = {
						limit = {
							ROOT = {
								cbrn_occupation_country_can_supply_external_protective_aid = yes
								NOT = { has_political_power < constant:cbrn_occupation_decision_cost.protective_aid_pp }
							}
							cbrn_occupation_state_can_receive_external_protective_aid = yes
							cbrn_quote_external_state_mask_reservation = yes
							check_variable = { cbrn_mask_reservation_quote > constant:cbrn_math.zero }
						}
						set_temp_variable = { cbrn_mask_escrow_requested = cbrn_mask_reservation_quote }
						set_temp_variable = { cbrn_mask_escrow_external = constant:cbrn_math.one }
						set_temp_variable = { cbrn_mask_escrow_completed = constant:cbrn_math.zero }
						cbrn_reserve_requested_masks_to_state = yes
						if = {
							limit = { check_variable = { cbrn_mask_escrow_completed > constant:cbrn_math.zero } }
							cbrn_occupation_register_external_aid_reservation = yes
							set_temp_variable = { cbrn_occupation_external_aid_days = constant:cbrn_occupation_timing.protective_aid_days }
							set_state_flag = { flag = cbrn_occupation_external_aid_project_active days = var:cbrn_occupation_external_aid_days }
							ROOT = {
								set_temp_variable = { cbrn_protection_pp_debit_requested = constant:cbrn_occupation_decision_cost.protective_aid_pp }
								cbrn_debit_protection_political_power = yes
							}
						}
					}
				}
			}
		}

		remove_effect = {
			custom_effect_tooltip = cbrn_deploy_protective_aid_complete_tt
			hidden_effect = {
				FROM = {
					cbrn_occupation_close_external_aid_reservation = yes

			}
		}

		cancel_trigger = {

				}
			}
		}

			}
		}

			}

			}

					}
				}
			}
		}
