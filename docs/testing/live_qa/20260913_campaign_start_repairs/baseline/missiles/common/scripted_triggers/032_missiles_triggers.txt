# ============================================================================
# CHAOS REDUX - EVENT 032 MISSILES SCRIPTED TRIGGERS
# ============================================================================
# This file contains the read-only side of the Event 032 contract. Country
# program values and state launch-site records are persistent; operation and
# incident receipts are only considered usable while their owning flags and
# scopes remain valid. The firing transaction is the only Event 032 helper
# that is allowed to enumerate all countries, and it lives in the companion
# effects file rather than in any recurring trigger.
# ============================================================================

# -----------------------------------------------------------------------------
# Recipient and program validity
# -----------------------------------------------------------------------------

missiles_country_has_controlled_land = {
	exists = yes
	any_owned_state = {
		impassable = no
	}
	any_controlled_state = {
		impassable = no
	}
}

missiles_country_has_usable_command = {
	OR = {
		uses_normal_civilian_systems = yes
		AND = {
			is_special_chaos_country = yes
			has_country_flag = missiles_special_industrial_actor
			has_country_flag = missiles_special_command_structure
			num_of_factories > constant:missiles_program_value.zero
		}
	}
}

missiles_is_valid_recipient = {
	exists = yes
	has_capitulated = no
	NOT = { has_country_flag = missiles_system_carrier }
	NOT = { has_country_flag = missiles_temporary_actor }
	NOT = { has_country_flag = missiles_permanently_ineligible }
	NOT = { has_country_flag = chaosx_test_country_carrier_only }
	missiles_country_has_usable_command = yes
	missiles_country_has_controlled_land = yes
	any_controlled_state = {
		impassable = no
	}
}

missiles_is_deferred_recipient = {
	exists = yes

		is_government_in_exile = yes

	NOT = { missiles_country_has_controlled_land = yes }

		infrastructure > constant:missiles_tuning.site_min_infrastructure

# The target-country trigger is deliberately country-scoped. State targeting
# is auto-selected by the operation helper after the country selector closes,
# because the ordinary decision system cannot safely expose a foreign state
# target and an enemy-country array in the same decision.
missiles_target_is_valid = {
	exists = yes
	NOT = { tag = ROOT }
	has_war_with = ROOT
	any_controlled_state = {
		impassable = no
	}
	NOT = { is_subject_of = ROOT }
}

# Ordinary operations may not select a faction partner. Retaliation and rogue
# incident paths continue to use missiles_target_is_valid because they model a
# warning or compromised command relationship rather than a normal strike.
missiles_ordinary_target_is_valid = {
	missiles_target_is_valid = yes
	NOT = { is_in_faction_with = ROOT }
}

		is_owned_by = var:ROOT.missiles_selected_target
	}
}

missiles_special_payload_policy_allows_operation = {
	has_variable = missiles_special_payload_policy
	OR = {
		AND = {
			check_variable = { var = missiles_special_payload_policy value = constant:missiles_special_payload_policy.retaliatory_only compare = equals }
			OR = {
				has_country_flag = missiles_warning_verified
				check_variable = { var = missiles_operation_type value = constant:missiles_operation.retaliation compare = equals }
			}
		}
		AND = {
			check_variable = { var = missiles_special_payload_policy value = constant:missiles_special_payload_policy.military_targets_only compare = equals }
			NOT = { check_variable = { var = missiles_operation_target_profile value = constant:missiles_target_profile.command compare = equals } }
			NOT = { check_variable = { var = missiles_operation_target_profile value = constant:missiles_target_profile.broad_strategic compare = equals } }
		}
		AND = {
			check_variable = { var = missiles_special_payload_policy value = constant:missiles_special_payload_policy.unrestricted_war compare = equals }
			has_war = yes
		}
		AND = {
			check_variable = { var = missiles_special_payload_policy value = constant:missiles_special_payload_policy.emergency_delegation compare = equals }
			OR = { has_country_flag = missiles_warning_active has_country_flag = missiles_rogue_command_active }
		}
		AND = {
			check_variable = { var = missiles_special_payload_policy value = constant:missiles_special_payload_policy.rogue_control compare = equals }
			has_country_flag = missiles_rogue_command_active
		}
	}
}
