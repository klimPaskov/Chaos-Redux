sov_nuclear_bombs_event_has_delivery_route = {
	has_tech = atomic_research
	has_tech = nukes
	OR = {
		has_tech = strategic_bomber1
		AND = {
			has_dlc = "By Blood Alone"
			has_tech = iw_large_airframe
		}
	}
	# Numeric fields are injected because direct constant support is undocumented.
	set_temp_variable = { sov_nuclear_bombs_delivery_bomber_floor = constant:sov_nuclear_bombs_delivery.deployed_bomber_floor }
	set_temp_variable = { sov_nuclear_bombs_delivery_airbase_floor = constant:sov_nuclear_bombs_delivery.airbase_level_floor }
	set_temp_variable = { sov_nuclear_bombs_delivery_fuel = constant:sov_nuclear_bombs_delivery.minimum_fuel }
	meta_trigger = {
		text = {
			has_deployed_air_force_size = { size > [BOMBERS] type = strategic_bomber }
			NOT = { has_fuel < [FUEL] }
			any_owned_controlled_state = { air_base > [AIRBASE] }
		}
		BOMBERS = "[?sov_nuclear_bombs_delivery_bomber_floor|.0]"
		FUEL = "[?sov_nuclear_bombs_delivery_fuel|.0]"
		AIRBASE = "[?sov_nuclear_bombs_delivery_airbase_floor|.0]"
	}
}

# State scope, PREV is the launcher country. Checks this exact endpoint.
# No persistent outputs. Range is an event ceiling, not an air-wing range query.
sov_nuclear_bombs_event_state_has_delivery_access = {
	exists = yes
	impassable = no
	OR = { owner = { exists = yes } controller = { exists = yes } }
	PREV = {
		sov_nuclear_bombs_event_has_delivery_route = yes
		set_temp_variable = { sov_nuclear_bombs_delivery_range = constant:sov_nuclear_bombs_delivery.maximum_state_distance }
		any_owned_controlled_state = {
			meta_trigger = {
				text = {
					air_base > [AIRBASE]
					distance_to = { target = PREV.PREV value < [RANGE] }
				}
				AIRBASE = "[?sov_nuclear_bombs_delivery_airbase_floor|.0]"
				RANGE = "[?sov_nuclear_bombs_delivery_range|.0]"
			}
		}
	}
}

# The presentation and retaliation gate can open after the first confirmed
# major receipt, but the one Event 23 exchange super-event requires two
# different observed major-to-major receipts. This prevents a test, a threat,
# or a single major strike on a minor from being presented as an exchange.
sov_nuclear_bombs_event_major_exchange_receipt_gate = {
	sov_nuclear_bombs_event_has_evolution_iii = yes
	check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.major_exchange_chaos compare = greater_than_or_equals }
	has_global_flag = sov_nuclear_bombs_major_detonation_confirmed
	has_global_flag = sov_nuclear_bombs_major_exchange_active
	has_variable = global.sov_nuclear_bombs_major_receipt_count
	check_variable = { var = global.sov_nuclear_bombs_major_receipt_count value = constant:sov_nuclear_bombs_tuning.major_receipt_minimum compare = greater_than_or_equals }
	sov_nuclear_bombs_event_has_other_nuclear_major = yes
}

# Country scope. This is the final settlement proof used by Event 23's
# verification event. Every accepted or partial response must leave a real,
# named term in the world before it can be recorded as durable compliance.
sov_nuclear_bombs_event_settlement_terms_are_valid = {
	sov_nuclear_bombs_event_actor_is_valid = yes
	has_country_flag = sov_nuclear_bombs_settlement_terms_applied
	has_variable = sov_nuclear_bombs_demand_kind
	has_variable = sov_nuclear_bombs_demand_state_id
	has_event_target = sov_nuclear_bombs_coercion_target
	has_event_target = sov_nuclear_bombs_selected_state
	event_target:sov_nuclear_bombs_coercion_target = {
		exists = yes
		NOT = { tag = SOV }
		is_subject = no
		NOT = { has_capitulated = yes }
	}
	event_target:sov_nuclear_bombs_selected_state = {
		exists = yes
		check_variable = { var = THIS.id value = ROOT.sov_nuclear_bombs_demand_state_id compare = equals }
	}
	if = {
		limit = { has_country_flag = sov_nuclear_bombs_partial_settlement_accepted }
		OR = {
			AND = {
				OR = {
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.cease_hostilities compare = equals }
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals }
				}
				has_country_flag = sov_nuclear_bombs_settlement_ceasefire_recorded
				event_target:sov_nuclear_bombs_coercion_target = { NOT = { has_war_with = SOV } }
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals } }
					has_country_flag = sov_nuclear_bombs_settlement_surrender_recorded
				}
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.withdraw_territory compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_withdrawal_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_demilitarized_zone }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.return_devices compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_return_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_site_joint_custody }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.soviet_protection compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_protection_recorded
				has_guaranteed = event_target:sov_nuclear_bombs_coercion_target
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.demilitarize_zone compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_demilitarization_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_demilitarized_zone }
			}
		}
	}
	else = {
		OR = {
			AND = {
				OR = {
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.cease_hostilities compare = equals }
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals }
				}
				has_country_flag = sov_nuclear_bombs_settlement_ceasefire_recorded
				event_target:sov_nuclear_bombs_coercion_target = { NOT = { has_war_with = SOV } }
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals } }
					has_country_flag = sov_nuclear_bombs_settlement_surrender_recorded
				}
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.withdraw_territory compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_withdrawal_recorded
				event_target:sov_nuclear_bombs_selected_state = { OR = { is_owned_by = SOV is_controlled_by = SOV } }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.return_devices compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_return_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_site_returned_to_soviet }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.soviet_protection compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_protection_recorded
				has_guaranteed = event_target:sov_nuclear_bombs_coercion_target
				has_non_aggression_pact_with = event_target:sov_nuclear_bombs_coercion_target
				has_military_access_to = event_target:sov_nuclear_bombs_coercion_target
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.demilitarize_zone compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_demilitarization_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_demilitarized_zone }
			}
		}
	}
