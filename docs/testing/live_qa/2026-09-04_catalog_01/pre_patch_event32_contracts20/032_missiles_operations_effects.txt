# ============================================================================
# CHAOS REDUX - EVENT 032 MISSILE OPERATIONS, EVOLUTIONS, AND SITE CONTROL
# ============================================================================
# Overview:
# - This file owns the bounded operation state machine behind Event 032's
#   ordinary decisions and missions.
# - A country owns one active operation receipt. The receipt freezes its actor,
#   target country, target state, launch sites, reserve payment, payload, reach,
#   guidance inputs, incident lineage, and commitment dates before resolution.
# - States own physical launch-site custody. Every site transition revalidates
#   controller, damage, capture, hardening, security, capacity, and stale
#   operation references before it can be used again.
# - Evolution unlocks are evaluated only from an Event 032 firing transaction or
#   an exact scenario setup. No recurring all-country scan is introduced here.
# - Payload consequences are submitted to the existing owning systems. The
#   conventional adapter below owns only conventional damage, shared Deaths,
#   evidence, attribution, and the Event 032 incident record.
# ============================================================================

missiles_update_firing_pressure_snapshot = {
	if = {
		limit = {
			check_variable = {
				var = missiles_program_stage
				value = constant:missiles_stage.guided
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { global.missiles_firing_guided_count = constant:missiles_event.one }
	}
	if = {
		limit = { has_country_flag = missiles_program_mature }
		add_to_variable = { global.missiles_firing_mature_count = constant:missiles_event.one }
	}
	if = {
		limit = { has_war = yes }
		add_to_variable = { global.missiles_firing_war_count = constant:missiles_event.one }
	}
	if = {
		limit = {
			missiles_country_has_payload_owner_capability = yes
		}
		add_to_variable = { global.missiles_firing_payload_owner_count = constant:missiles_event.one }
	}
	if = {
		limit = {
			OR = {
				check_variable = { var = missiles_command_control value = constant:missiles_evolution_pressure.minimum_vulnerable_control compare = less_than }
				has_country_flag = missiles_site_compromise_active
				has_country_flag = missiles_code_compromise_active
				has_country_flag = missiles_civil_war_command_split
			}
		}
		add_to_variable = { global.missiles_firing_vulnerable_count = constant:missiles_event.one }
	}
	if = {
		limit = {
			check_variable = { var = missiles_command_control value = constant:missiles_evolution_pressure.minimum_retaliation_control compare = greater_than_or_equals }
			missiles_country_has_active_site = yes
		}
		add_to_variable = { global.missiles_firing_retaliation_count = constant:missiles_event.one }
	}
}

# AI profile is a persistent preference record, not a second country selector.
# It is recomputed at package boundaries so scenario flags, war status, custody,
# and command failures can change how the ordinary operation selector behaves.
missiles_assign_ai_profile = {
	set_variable = { missiles_ai_profile = constant:missiles_ai_profile.restrained_deterrent }
	if = {
		limit = {
			has_country_flag = missiles_scenario_saturation_war
			has_country_flag = missiles_scenario_saturation_belligerent
		}
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.saturation_offender }
	}
	else_if = {
		limit = {
			has_country_flag = missiles_evolution_automatic_retaliation
			has_country_flag = missiles_retaliation_network_active
		}
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.retaliation_network }
	}
	else_if = {
		limit = { has_country_flag = missiles_rogue_command_active }
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.rogue_command }
	}
	else_if = {
		limit = {
			OR = {
				has_country_flag = missiles_site_compromise_active
				has_country_flag = missiles_code_compromise_active
				has_country_flag = missiles_civil_war_command_split
			}
		}
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.brittle_command }
	}
	else_if = {
		limit = {
			has_country_flag = missiles_evolution_special_warheads
			missiles_special_payload_available = yes
		}
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.special_warhead_escalator }
	}
	else_if = {
		limit = {
			has_country_flag = missiles_evolution_saturation_arsenals
			has_war = yes
		}
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.saturation_offender }
	}
	else_if = {
		limit = {
			has_war = yes
			check_variable = { var = missiles_guidance_score value = constant:missiles_guidance.accurate_score compare = greater_than_or_equals }
			check_variable = { var = missiles_command_control value = constant:missiles_program_value.secure_command_gate compare = greater_than_or_equals }
		}
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.counterforce_guardian }
	}
	else_if = {
		limit = { has_war = yes }
		set_variable = { missiles_ai_profile = constant:missiles_ai_profile.conventional_interdiction }
	}
	set_country_flag = missiles_ai_profile_assigned
}

missiles_evaluate_country_evolution_adoption = {
	if = {
		limit = {
			has_global_flag = missiles_evolution_saturation_unlocked
			NOT = { has_global_flag = event_evolution_disabled_32_1 }
			check_variable = { var = missiles_program_stage value = constant:missiles_evolution_pressure.minimum_guidance_stage compare = greater_than_or_equals }
			missiles_country_has_active_site = yes
			check_variable = { var = missiles_command_control value = constant:missiles_evolution_pressure.minimum_retaliation_control compare = greater_than_or_equals }
		}
		set_country_flag = missiles_evolution_saturation_arsenals
		set_country_flag = missiles_saturation_doctrine_adopted
	}
	if = {
		limit = {
			has_global_flag = missiles_evolution_unreliable_guidance_unlocked
			NOT = { has_global_flag = event_evolution_disabled_32_2 }
		}
		set_country_flag = missiles_evolution_unreliable_guidance
	}
	if = {
		limit = {
			has_global_flag = missiles_evolution_special_warheads_unlocked
			NOT = { has_global_flag = event_evolution_disabled_32_3 }
			missiles_country_has_payload_owner_capability = yes
			check_variable = { var = missiles_program_stage value = constant:missiles_evolution_pressure.minimum_guidance_stage compare = greater_than_or_equals }
		}
		set_country_flag = missiles_evolution_special_warheads
		set_country_flag = missiles_special_payload_supported
	}
	if = {
		limit = {
			has_global_flag = missiles_evolution_rogue_launch_commands_unlocked
			NOT = { has_global_flag = event_evolution_disabled_32_4 }
		}
		set_country_flag = missiles_evolution_rogue_launch_commands
	}
	if = {
		limit = {
			has_global_flag = missiles_evolution_automatic_retaliation_unlocked
			NOT = { has_global_flag = event_evolution_disabled_32_5 }
			check_variable = { var = missiles_command_control value = constant:missiles_evolution_pressure.minimum_retaliation_control compare = greater_than_or_equals }
			missiles_country_has_active_site = yes
		}
		set_country_flag = missiles_evolution_automatic_retaliation
		set_country_flag = missiles_retaliation_network_active
	}
}

missiles_calculate_command_pressure = {
	set_temp_variable = { missiles_command_pressure_component = constant:missiles_program_value.maximum }
	subtract_from_temp_variable = { missiles_command_pressure_component = missiles_command_control }
	set_variable = { missiles_command_pressure = missiles_command_pressure_component }
	set_temp_variable = { missiles_stability_pressure = stability }
	multiply_temp_variable = { var = missiles_stability_pressure value = constant:missiles_program_value.maximum }
	set_temp_variable = { missiles_stability_pressure_gap = constant:missiles_program_value.maximum }
	subtract_from_temp_variable = { missiles_stability_pressure_gap = missiles_stability_pressure }
	multiply_temp_variable = { var = missiles_stability_pressure_gap value = constant:missiles_rogue_tuning.stability_pressure_multiplier }
	add_to_variable = { missiles_command_pressure = missiles_stability_pressure_gap }
	if = { limit = { has_civil_war = yes } add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.civil_war_pressure } }
	if = { limit = { has_variable = missiles_site_count check_variable = { var = missiles_site_count value = constant:missiles_rogue_tuning.multiple_site_threshold compare = greater_than_or_equals } } add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.multiple_site_pressure } }
	if = { limit = { check_variable = { var = missiles_operational_reserve value = constant:missiles_rogue_tuning.large_reserve_threshold compare = greater_than_or_equals } } add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.large_reserve_pressure } }
	if = { limit = { missiles_special_payload_available = yes } add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.special_payload_pressure } }
	if = { limit = { has_country_flag = missiles_recent_code_rotation } subtract_from_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.recent_rotation_reduction } }
	clamp_variable = { var = missiles_command_pressure min = constant:missiles_program_value.zero max = constant:missiles_program_value.maximum }
}

missiles_select_rogue_incident_site = {
	clear_variable = missiles_incident_site
	random_scope_in_array = {
		array = missiles_launch_site_states
		limit = { missiles_site_can_host_rogue_incident = yes }
		ROOT = { set_variable = { missiles_incident_site = PREV } }
	}
	if = {
		limit = { has_variable = missiles_incident_site }
		var:missiles_incident_site = {
			if = { limit = { has_state_flag = missiles_site_captured } ROOT = { add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.captured_site_pressure } } }
			if = { limit = { has_state_flag = missiles_site_compromised } ROOT = { add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.compromised_site_pressure } } }
			if = { limit = { has_state_flag = missiles_site_isolated } ROOT = { add_to_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.isolated_site_pressure } } }
			if = { limit = { missiles_site_is_secure = yes } ROOT = { subtract_from_variable = { missiles_command_pressure = constant:missiles_rogue_tuning.secure_site_reduction } } }
		}
		clamp_variable = { var = missiles_command_pressure min = constant:missiles_program_value.zero max = constant:missiles_program_value.maximum }
	}
}

missiles_open_rogue_incident = {
	missiles_allocate_incident_id = yes
	if = { limit = { NOT = { has_variable = global.missiles_rogue_active_incidents } } set_variable = { global.missiles_rogue_active_incidents = constant:missiles_program_value.zero } }
	add_to_variable = { global.missiles_rogue_active_incidents = constant:missiles_program_value.one }
	set_country_flag = missiles_rogue_active_counted
	set_country_flag = missiles_rogue_command_active
	clr_country_flag = chaosx_032_keys_returned_initial_proof
	clr_country_flag = chaosx_032_rogue_resolution_receipt_written
	clr_country_flag = missiles_site_recovery_complete
	clr_country_flag = missiles_command_negotiation_complete
	clr_country_flag = missiles_rogue_launch_committed
	clr_country_flag = missiles_rogue_public_launch
	set_country_flag = missiles_incident_active
	set_variable = { missiles_current_incident_type = constant:missiles_incident_type.rogue_preparation }
	set_variable = { missiles_last_incident_id = missiles_current_incident_id }
	set_variable = { missiles_active_incident_id = missiles_current_incident_id }
	set_variable = { missiles_last_incident_type = constant:missiles_incident_type.rogue_preparation }
	set_variable = { missiles_last_incident_date = global.date }
	set_variable = { missiles_last_incident_actor_id = ROOT.id }
	set_variable = { missiles_last_incident_closed = constant:missiles_program_value.zero }
	set_variable = { missiles_rogue_crisis_deadline = global.date }
	add_to_variable = { missiles_rogue_crisis_deadline = constant:missiles_rogue_tuning.crisis_window_days }
	if = {
		limit = { has_variable = missiles_incident_site }
		var:missiles_incident_site = {
			set_state_flag = missiles_site_rogue
			set_state_flag = missiles_site_compromised
			clr_state_flag = missiles_site_active
			if = {
				limit = {
					OR = {
						check_variable = { var = missiles_site_reserve value = constant:missiles_program_value.zero compare = greater_than }
						check_variable = { var = missiles_site_operation_id value = constant:missiles_program_value.zero compare = greater_than }
					}
				}
				ROOT = { set_country_flag = chaosx_032_keys_returned_initial_proof }
			}
			ROOT = { set_variable = { missiles_last_incident_target_state_id = PREV.id } }
		}
	}
	if = { limit = { has_variable = missiles_rogue_foreign_actor } var:missiles_rogue_foreign_actor = { ROOT = { set_variable = { missiles_rogue_foreign_actor_id = PREV.id } } } }
	if = { limit = { has_variable = missiles_rogue_defection_recipient } var:missiles_rogue_defection_recipient = { ROOT = { set_variable = { missiles_rogue_defection_recipient_id = PREV.id } } } }
	set_country_flag = missiles_incident_history
	missiles_set_program_phases = yes
	country_event = { id = chaosx.nr32.60 hours = 1 }
	country_event = { id = chaosx.nr32.69 days = constant:missiles_rogue_tuning.crisis_window_days }
}

		}
		constant:missiles_rogue_tuning.weight_mutiny = {
			modifier = { factor = constant:missiles_rogue_tuning.factor_high has_civil_war = yes }
			set_variable = { missiles_rogue_incident_family = constant:missiles_rogue_incident_family.launch_site_mutiny }
			missiles_open_rogue_incident = yes
		}
		constant:missiles_rogue_tuning.weight_defection = {
			modifier = { factor = constant:missiles_rogue_tuning.factor_zero NOT = { has_variable = missiles_rogue_defection_recipient } }
			modifier = { factor = constant:missiles_rogue_tuning.factor_zero NOT = { missiles_rogue_incident_site_has_reserve = yes } }
			set_variable = { missiles_rogue_incident_family = constant:missiles_rogue_incident_family.crew_defection }

		}
		constant:missiles_rogue_tuning.weight_regional = {
			modifier = {
				factor = constant:missiles_rogue_tuning.factor_zero

					OR = {
						has_civil_war = yes

			modifier = { factor = constant:missiles_rogue_tuning.factor_zero NOT = { has_variable = missiles_rogue_foreign_actor } }
			modifier = { factor = constant:missiles_rogue_tuning.factor_high has_country_flag = missiles_code_compromise_active }
			set_variable = { missiles_rogue_incident_family = constant:missiles_rogue_incident_family.foreign_bribery }
			missiles_open_rogue_incident = yes
		}
		constant:missiles_rogue_tuning.weight_captured = {
			modifier = { factor = constant:missiles_rogue_tuning.factor_zero NOT = { var:missiles_incident_site = { has_state_flag = missiles_site_captured } } }
			set_variable = { missiles_rogue_incident_family = constant:missiles_rogue_incident_family.captured_arsenal }
			missiles_open_rogue_incident = yes
		}
		constant:missiles_rogue_tuning.weight_payload = {
			modifier = { factor = constant:missiles_rogue_tuning.factor_zero NOT = { missiles_rogue_incident_site_has_payload = yes } }
			modifier = { factor = constant:missiles_rogue_tuning.factor_zero NOT = { has_variable = missiles_rogue_launch_target } }
			modifier = { factor = constant:missiles_rogue_tuning.factor_severe var:missiles_incident_site = { has_state_flag = missiles_site_captured } }
			set_variable = { missiles_rogue_incident_family = constant:missiles_rogue_incident_family.special_payload_custody }
			missiles_open_rogue_incident = yes
		}
	}
}

			limit = { has_global_flag = event_evolution_disabled_32_1 }

missiles_evaluate_evolution_unlocks = {
	if = {
		limit = {
			has_global_flag = missiles_program_ever_initialized
			has_variable = global.missiles_firing_guided_count
		}
		set_global_flag = missiles_evolution_unlock_recorded_this_pulse
		# Rogue Launch Commands: a live vulnerable command structure is enough to
		# open the delayed track; no incident is manufactured at unlock time.
		if = {
			limit = {
				check_variable = { var = global.chaos_meter_value value = constant:missiles_evolution_pressure.minimum_chaos_rogue compare = greater_than_or_equals }
				check_variable = { var = global.missiles_firing_vulnerable_count value = constant:missiles_evolution_pressure.minimum_operational_programs compare = greater_than_or_equals }
			}
			set_temp_variable = { missiles_evolution_selection = constant:missiles_evolution.rogue_commands }
			set_temp_variable = { missiles_evolution_delay_days = constant:missiles_evolution_timing.rogue_days }
			if = { limit = { check_variable = { var = global.missiles_firing_vulnerable_count value = constant:missiles_evolution_pressure.minimum_mature_programs compare = greater_than_or_equals } } multiply_temp_variable = { var = missiles_evolution_delay_days value = constant:missiles_evolution_pressure.delay_shortening_multiplier } }
			missiles_schedule_evolution_track = yes
		}
		# Unreliable Guidance follows global strain after guided programs exist.
		if = {
			limit = {
				check_variable = { var = global.chaos_meter_value value = constant:missiles_evolution_pressure.minimum_chaos_guidance compare = greater_than_or_equals }
				check_variable = { var = global.missiles_firing_guided_count value = constant:missiles_evolution_pressure.minimum_operational_programs compare = greater_than_or_equals }
			}
			set_temp_variable = { missiles_evolution_selection = constant:missiles_evolution.unreliable_guidance }
			set_temp_variable = { missiles_evolution_delay_days = constant:missiles_evolution_timing.guidance_days }
			if = { limit = { check_variable = { var = global.missiles_firing_war_count value = constant:missiles_evolution_pressure.minimum_war_programs compare = greater_than_or_equals } } multiply_temp_variable = { var = missiles_evolution_delay_days value = constant:missiles_evolution_pressure.delay_shortening_multiplier } }
			missiles_schedule_evolution_track = yes
		}
		# Saturation Arsenals requires a meaningful guided and wartime pool.
		if = {
			limit = {
				check_variable = { var = global.chaos_meter_value value = constant:missiles_evolution_pressure.minimum_chaos_saturation compare = greater_than_or_equals }
				check_variable = { var = global.missiles_firing_guided_count value = constant:missiles_evolution_pressure.minimum_mature_programs compare = greater_than_or_equals }
				check_variable = { var = global.missiles_firing_war_count value = constant:missiles_evolution_pressure.minimum_war_programs compare = greater_than_or_equals }
			}
			set_temp_variable = { missiles_evolution_selection = constant:missiles_evolution.saturation }
			set_temp_variable = { missiles_evolution_delay_days = constant:missiles_evolution_timing.saturation_days }
			if = { limit = { check_variable = { var = global.missiles_firing_mature_count value = constant:missiles_evolution_pressure.minimum_mature_programs compare = greater_than_or_equals } } multiply_temp_variable = { var = missiles_evolution_delay_days value = constant:missiles_evolution_pressure.delay_shortening_multiplier } }
			missiles_schedule_evolution_track = yes
		}
		# Special Warheads requires an existing owner; the evolution never grants
		# payload stock, technology, or a delivery system by itself.
		if = {
			limit = {
				check_variable = { var = global.chaos_meter_value value = constant:missiles_evolution_pressure.minimum_chaos_special compare = greater_than_or_equals }
				check_variable = { var = global.missiles_firing_payload_owner_count value = constant:missiles_evolution_pressure.minimum_payload_owners compare = greater_than_or_equals }
			}
			set_temp_variable = { missiles_evolution_selection = constant:missiles_evolution.special_warheads }
			set_temp_variable = { missiles_evolution_delay_days = constant:missiles_evolution_timing.special_days }
			missiles_schedule_evolution_track = yes
		}
		# Automatic Retaliation has an explicit 900-chaos gate in addition to its
		# two-program support requirement. It only queues warnings after unlock.
		if = {
			limit = {
				check_variable = { var = global.chaos_meter_value value = constant:missiles_evolution_pressure.minimum_chaos_retaliation compare = greater_than_or_equals }
				check_variable = { var = global.missiles_firing_retaliation_count value = constant:missiles_evolution_pressure.minimum_retaliation_programs compare = greater_than_or_equals }
			}
			set_temp_variable = { missiles_evolution_selection = constant:missiles_evolution.automatic_retaliation }
			set_temp_variable = { missiles_evolution_delay_days = constant:missiles_evolution_timing.retaliation_days }
			missiles_schedule_evolution_track = yes
		}
		clr_global_flag = missiles_evolution_unlock_recorded_this_pulse
	}
}

missiles_initialize_scenario_package = {
	clr_country_flag = missiles_scenario_package_failed
	if = {
		limit = {
			NOT = { has_country_flag = missiles_program_initialized }
		}
		set_country_flag = missiles_program_initialized
		add_to_variable = { global.missiles_program_count = constant:missiles_event.one }
	}
	set_global_flag = missiles_program_ever_initialized
	if = {
		limit = {
			has_variable = missiles_scenario_package_stage
			check_variable = { var = missiles_program_stage value = missiles_scenario_package_stage compare = less_than }
		}
		set_variable = { missiles_program_stage = missiles_scenario_package_stage }
	}
	else_if = { limit = { NOT = { has_variable = missiles_program_stage } } set_variable = { missiles_program_stage = constant:missiles_stage.guided } }
	if = {
		limit = { check_variable = { var = missiles_program_stage value = constant:missiles_stage.experimental compare = greater_than_or_equals } }
		set_technology = { experimental_rockets = 1 }
	}
	if = {
		limit = { check_variable = { var = missiles_program_stage value = constant:missiles_stage.guided compare = greater_than_or_equals } }
		set_technology = { rocket_engines = 1 }
	}
	if = {
		limit = { check_variable = { var = missiles_program_stage value = constant:missiles_stage.advanced compare = greater_than_or_equals } }
		set_technology = { improved_rocket_engines = 1 }
	}
	if = {
		limit = { check_variable = { var = missiles_program_stage value = constant:missiles_stage.strategic compare = greater_than_or_equals } }
		set_technology = { advanced_rocket_engines = 1 }
	}
	if = {
		limit = { check_variable = { var = missiles_scenario_package_reserve value = constant:missiles_program_value.zero compare = greater_than } }
		set_variable = { missiles_operational_reserve = missiles_scenario_package_reserve }
	}
	else = {
		set_variable = { missiles_operational_reserve = constant:missiles_reserve.initial_base }
	}
	if = {
		limit = {
			has_variable = missiles_scenario_package_reserve
			check_variable = { var = missiles_reserve_cap value = missiles_scenario_package_reserve compare = less_than }
		}
		set_variable = { missiles_reserve_cap = missiles_scenario_package_reserve }
	}
	else_if = {
		limit = { check_variable = { var = missiles_reserve_cap value = constant:missiles_reserve.mature_cap compare = less_than } }
		set_variable = { missiles_reserve_cap = constant:missiles_reserve.mature_cap }
	}
	if = { limit = { has_variable = missiles_scenario_package_readiness } set_variable = { missiles_launch_readiness = missiles_scenario_package_readiness } }
	else = { set_variable = { missiles_launch_readiness = constant:missiles_program_value.initial_readiness } }
	if = { limit = { has_variable = missiles_scenario_package_control } set_variable = { missiles_command_control = missiles_scenario_package_control } }
	else = { set_variable = { missiles_command_control = constant:missiles_program_value.initial_command_control } }
	if = {
		limit = { check_variable = { var = missiles_program_stage value = constant:missiles_stage.complete compare = greater_than_or_equals } }
		set_country_flag = missiles_program_mature
		set_country_flag = missiles_supported_line_complete
	}
	missiles_create_primary_site = yes
	if = {
		limit = { has_variable = missiles_scenario_package_site_capacity }
		# The scenario input is a physical site level (1-4), not a raw missile
		# count. Derive the canonical 4/6/8/10 capacity ladder so setup cannot
		# create an impossible one-missile launch site.
		for_each_scope_loop = {
			array = missiles_launch_site_states
			if = {
				limit = { exists = yes is_controlled_by = ROOT }
				if = { limit = { NOT = { has_variable = missiles_site_level } } set_variable = { missiles_site_level = constant:missiles_site_level.initial } }
				if = {
					limit = { check_variable = { var = missiles_site_level value = ROOT.missiles_scenario_package_site_capacity compare = less_than } }
					set_variable = { missiles_site_level = ROOT.missiles_scenario_package_site_capacity }
				}
				if = {
					limit = { check_variable = { var = missiles_site_level value = constant:missiles_site_level.strategic compare = greater_than_or_equals } }
					set_variable = { missiles_site_capacity = constant:missiles_site_capacity.strategic }
				}
				else_if = {
					limit = { check_variable = { var = missiles_site_level value = constant:missiles_site_level.expanded compare = greater_than_or_equals } }
					set_variable = { missiles_site_capacity = constant:missiles_site_capacity.expanded }
				}
				else_if = {
					limit = { check_variable = { var = missiles_site_level value = constant:missiles_site_level.reinforced compare = greater_than_or_equals } }
					set_variable = { missiles_site_capacity = constant:missiles_site_capacity.reinforced }
				}
				else = { set_variable = { missiles_site_capacity = constant:missiles_site_capacity.initial } }
				missiles_refresh_site_flags = yes
			}
		}
	}
	missiles_recalculate_network = yes
	missiles_rebalance_site_custody = yes
	chaosx_032_achievement_initialize_country = yes
	if = {
		limit = { NOT = { missiles_country_has_active_site = yes } }
		set_country_flag = missiles_scenario_package_failed
	}
}

missiles_recover_site_capacity_loss = {
	# STATE scope. A repair receipt restores only the remaining physically lost
	# capacity and never exceeds the strategic capacity ceiling or creates a free reserve.
	set_temp_variable = { missiles_capacity_recovery_result = constant:missiles_result.rejected }
	if = {
		limit = {
			has_variable = missiles_site_capacity_loss
			check_variable = { var = missiles_site_capacity_loss value = constant:missiles_program_value.zero compare = greater_than }
			NOT = { has_state_flag = missiles_site_scuttled }
			has_variable = missiles_site_capacity
			check_variable = { var = missiles_site_capacity value = constant:missiles_site_capacity.strategic compare = less_than }
		}
		set_temp_variable = { missiles_capacity_recovery_amount = constant:missiles_natural_disaster.capacity_loss_recovery_step }
		if = {
			limit = { check_variable = { var = missiles_capacity_recovery_amount value = missiles_site_capacity_loss compare = greater_than } }
			set_temp_variable = { missiles_capacity_recovery_amount = missiles_site_capacity_loss }
		}
		set_temp_variable = { missiles_capacity_recovery_headroom = constant:missiles_site_capacity.strategic }
		subtract_from_temp_variable = { var = missiles_capacity_recovery_headroom value = missiles_site_capacity }
		if = {
			limit = { check_variable = { var = missiles_capacity_recovery_amount value = missiles_capacity_recovery_headroom compare = greater_than } }
			set_temp_variable = { missiles_capacity_recovery_amount = missiles_capacity_recovery_headroom }
		}
		if = {
			limit = { check_variable = { var = missiles_capacity_recovery_amount value = constant:missiles_program_value.zero compare = greater_than } }
			subtract_from_variable = { missiles_site_capacity_loss = missiles_capacity_recovery_amount }
			add_to_variable = { missiles_site_capacity = missiles_capacity_recovery_amount }
			clamp_variable = { var = missiles_site_capacity_loss min = constant:missiles_program_value.zero max = constant:missiles_site_capacity.strategic }
			clamp_variable = { var = missiles_site_capacity min = constant:missiles_site_capacity.minimum max = constant:missiles_site_capacity.strategic }
			set_temp_variable = { missiles_capacity_recovery_result = constant:missiles_result.completed }
			missiles_refresh_site_flags = yes
		}
	}
	set_variable = { missiles_site_capacity_recovery_result = missiles_capacity_recovery_result }
}

				limit = { check_variable = { var = missiles_captured_reserve_held value = PREV.missiles_site_captured_reserve compare = greater_than_or_equals } }
				subtract_from_variable = { missiles_captured_reserve_held = PREV.missiles_site_captured_reserve }
			}
			else = { set_variable = { missiles_captured_reserve_held = constant:missiles_program_value.zero } }
		}
		set_variable = { missiles_site_captured_reserve = constant:missiles_program_value.zero }
	}
	if = {
		limit = { has_variable = missiles_site_reserve }
		set_temp_variable = { missiles_scuttle_loss = missiles_site_reserve }
		if = { limit = { check_variable = { var = ROOT.missiles_operational_reserve value = missiles_scuttle_loss compare = greater_than_or_equals } } ROOT = { subtract_from_variable = { missiles_operational_reserve = missiles_scuttle_loss } } }
	}
	set_state_flag = missiles_site_scuttled
	clr_state_flag = missiles_site_active
	clr_state_flag = missiles_site_hardened
	clr_state_flag = missiles_site_compromised
	clr_state_flag = missiles_site_captured
	clr_state_flag = missiles_site_isolated

	set_variable = { missiles_site_operation_id = constant:missiles_event.zero }
	set_variable = { missiles_site_owner_id = constant:missiles_event.zero }
	set_variable = { missiles_site_controller_id = constant:missiles_event.zero }
	remove_from_array = { array = ROOT.missiles_launch_site_states value = THIS }
	ROOT = {
		set_variable = { missiles_achievement_removed_site_id = THIS.id }
		missiles_recalculate_network = yes
		missiles_rebalance_site_custody = yes
		if = {
			limit = { NOT = { any_controlled_state = { has_state_flag = missiles_site_captured } } }
			clr_country_flag = missiles_captured_site_unresolved
			if = { limit = { NOT = { has_country_flag = missiles_rogue_command_active } } clr_country_flag = missiles_incident_active }
		}
		missiles_refresh_program_status_idea = yes
		if = { limit = { has_country_flag = missiles_rogue_command_active } missiles_close_rogue_incident = yes }
	}
}

# COUNTRY scope is the candidate and ROOT is the operating country. The
# selector is called only when an operation needs a target; it is never a
# recurring world scan. Faction partners are filtered by the ordinary target
# trigger before they can contribute a score.
missiles_score_target_country_candidate = {
	set_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.baseline }
	if = {
		limit = { missiles_ordinary_target_is_valid = yes }
		add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.profile_value }
		if = { limit = { is_major = yes } add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.major } }
		set_temp_variable = { missiles_target_factory_score = num_of_factories }
		multiply_temp_variable = { var = missiles_target_factory_score value = constant:missiles_target_score.factory_step }
		add_to_temp_variable = { missiles_target_candidate_score = missiles_target_factory_score }
		set_temp_variable = { missiles_target_state_score = num_of_controlled_states }
		multiply_temp_variable = { var = missiles_target_state_score value = constant:missiles_target_score.controlled_state_step }
		add_to_temp_variable = { missiles_target_candidate_score = missiles_target_state_score }
		if = { limit = { has_war_with = ROOT } add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.active_war } }
		if = { limit = { has_country_flag = missiles_incident_history } add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.evidence_pressure } }
		if = { limit = { has_country_flag = missiles_verified_special_attack_history } add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.first_use_pressure } }
		if = { limit = { capital_scope = { is_island_state = yes } } add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.island } }
		if = {
			limit = {
				capital_scope = {
					OR = {
						resistance > constant:missiles_program_value.zero
						resistance_target > constant:missiles_program_value.zero
					}
				}
			}
			add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.resistance }
		}
		if = {
			limit = {
			ROOT = {
				OR = {
					check_variable = { var = missiles_ai_profile value = constant:missiles_ai_profile.saturation_offender compare = equals }
					check_variable = { var = missiles_ai_profile value = constant:missiles_ai_profile.counterforce_guardian compare = equals }
				}
			}
			any_controlled_state = { missiles_operation_target_has_strategic_value = yes }
		}
			add_to_temp_variable = { missiles_target_candidate_score = constant:missiles_target_score.strategic_depth }
		}
		if = {
			limit = { check_variable = { var = missiles_target_candidate_score value = ROOT.missiles_best_target_country_score compare = greater_than } }
			save_event_target_as = missiles_scored_target_candidate
			ROOT = {
				set_variable = { missiles_best_target_country_score = PREV.missiles_target_candidate_score }
				set_variable = { missiles_selected_target = event_target:missiles_scored_target_candidate }
				set_variable = { missiles_selected_target_country_id = event_target:missiles_scored_target_candidate.id }
				set_variable = { missiles_target_selection_result = constant:missiles_program_value.one }
				set_country_flag = missiles_target_selected
			}
		}
	}
	clear_temp_variable = missiles_target_candidate_score
}

		subtract_from_variable = { missiles_operation_preparation_days = constant:missiles_tuning.operation_preparation_wartime_shortening }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = missiles_site_compromise_active
				has_country_flag = missiles_code_compromise_active
				has_country_flag = missiles_civil_war_command_split
			}

missiles_prepare_operation_drift_pool = {
	clear_array = missiles_operation_drift_states
	clear_array = missiles_operation_neutral_states
	clear_array = missiles_operation_self_states
	set_variable = { missiles_operation_drift_state_count = constant:missiles_program_value.zero }
	set_variable = { missiles_operation_neutral_state_count = constant:missiles_program_value.zero }
	set_variable = { missiles_operation_self_state_count = constant:missiles_program_value.zero }
	if = {
		limit = { has_variable = missiles_operation_target_state }
		var:missiles_operation_target_state = {
			every_neighbor_state = {
				limit = {
					exists = yes
					impassable = no
					ROOT = {
						check_variable = { var = missiles_operation_drift_state_count value = constant:missiles_resolution.maximum_drift_states compare = less_than }
					}
				}
				add_to_array = { array = ROOT.missiles_operation_drift_states value = THIS }
				ROOT = { add_to_variable = { missiles_operation_drift_state_count = constant:missiles_program_value.one } }
				if = {
					limit = {
						is_controlled_by = ROOT
						ROOT = { check_variable = { var = missiles_operation_self_state_count value = constant:missiles_resolution.maximum_self_states compare = less_than } }
					}
					add_to_array = { array = ROOT.missiles_operation_self_states value = THIS }
					ROOT = { add_to_variable = { missiles_operation_self_state_count = constant:missiles_program_value.one } }
				}
				else_if = {
					limit = {
						CONTROLLER = {
							exists = yes
							NOT = { tag = ROOT }
							NOT = { has_war_with = ROOT }
							NOT = { is_in_faction_with = ROOT }
						}
						ROOT = { check_variable = { var = missiles_operation_neutral_state_count value = constant:missiles_resolution.maximum_neutral_states compare = less_than } }
					}
					add_to_array = { array = ROOT.missiles_operation_neutral_states value = THIS }
					ROOT = { add_to_variable = { missiles_operation_neutral_state_count = constant:missiles_program_value.one } }
				}
			}
		}
	}
	if = {
		limit = {
			has_variable = missiles_operation_launch_site
			NOT = { is_in_array = { array = missiles_operation_self_states value = var:missiles_operation_launch_site } }
			check_variable = { var = missiles_operation_self_state_count value = constant:missiles_resolution.maximum_self_states compare = less_than }
		}
		add_to_array = { array = missiles_operation_self_states value = var:missiles_operation_launch_site }
		add_to_variable = { missiles_operation_self_state_count = constant:missiles_program_value.one }
	}
}

missiles_calculate_operation_guidance = {
	set_variable = { missiles_operation_guidance_score = constant:missiles_guidance.base_score }
	set_temp_variable = { missiles_guidance_stage_component = missiles_program_stage }
	multiply_temp_variable = { var = missiles_guidance_stage_component value = constant:missiles_guidance.technology_step }
	add_to_variable = { missiles_operation_guidance_score = missiles_guidance_stage_component }
	set_temp_variable = { missiles_guidance_readiness_component = missiles_operation_readiness_snapshot }
	multiply_temp_variable = { var = missiles_guidance_readiness_component value = constant:missiles_guidance.readiness_step }
	add_to_variable = { missiles_operation_guidance_score = missiles_guidance_readiness_component }
	set_temp_variable = { missiles_guidance_control_component = missiles_operation_control_snapshot }
	multiply_temp_variable = { var = missiles_guidance_control_component value = constant:missiles_guidance.command_step }
	add_to_variable = { missiles_operation_guidance_score = missiles_guidance_control_component }
	if = {
		limit = { has_variable = missiles_guidance_training }
		set_temp_variable = { missiles_guidance_training_component = missiles_guidance_training }
		multiply_temp_variable = { var = missiles_guidance_training_component value = constant:missiles_guidance.training_step }
		add_to_variable = { missiles_operation_guidance_score = missiles_guidance_training_component }
	}
	if = {
		limit = { has_variable = missiles_operation_launch_site }
		var:missiles_operation_launch_site = {
			if = { limit = { has_state_flag = missiles_site_hardened } ROOT = { add_to_variable = { missiles_operation_guidance_score = constant:missiles_guidance.hardened_site_bonus } } }
			if = { limit = { has_variable = missiles_site_damage check_variable = { var = missiles_site_damage value = constant:missiles_program_value.zero compare = greater_than } } ROOT = { subtract_from_variable = { missiles_operation_guidance_score = constant:missiles_tuning.operation_guidance_damage_penalty } } }
		}
	}
	if = {
		limit = { missiles_operation_uses_saturation = yes }
		subtract_from_variable = { missiles_operation_guidance_score = constant:missiles_guidance.saturation_penalty }
	}
	if = {
		limit = { missiles_operation_has_special_payload = yes }
		subtract_from_variable = { missiles_operation_guidance_score = constant:missiles_guidance.special_payload_penalty }
	}
	if = {
		limit = {
			has_country_flag = missiles_evolution_unreliable_guidance
			NOT = { has_global_flag = event_evolution_disabled_32_2 }
		}
		subtract_from_variable = { missiles_operation_guidance_score = constant:missiles_guidance.unstable_program_penalty }
	}
	if = {
		limit = { has_country_flag = missiles_guidance_recalibrated_next_launch }
		add_to_variable = { missiles_operation_guidance_score = constant:missiles_guidance.emergency_recalibration_bonus }
	}
	clamp_variable = { var = missiles_operation_guidance_score min = constant:missiles_program_value.zero max = constant:missiles_program_value.maximum }
	set_variable = { missiles_operation_guidance_snapshot = missiles_operation_guidance_score }
	clear_temp_variable = missiles_guidance_stage_component
	clear_temp_variable = missiles_guidance_readiness_component
	clear_temp_variable = missiles_guidance_control_component
	clear_temp_variable = missiles_guidance_training_component
}

missiles_select_operation_impact_state = {
	clr_country_flag = missiles_operation_has_impact
	if = {
		limit = {
			OR = {
				check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.on_target compare = equals }
				check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.degraded compare = equals }
				check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.wrong_object compare = equals }
				check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.near_miss compare = equals }
			}
			has_variable = missiles_operation_target_state
		}
		set_variable = { missiles_operation_impact_state = missiles_operation_target_state }
		set_country_flag = missiles_operation_has_impact
	}
	else_if = {
		limit = {
			check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.wrong_state compare = equals }
			check_variable = { var = missiles_operation_drift_states^num value = constant:missiles_program_value.zero compare = greater_than }
		}
		random_scope_in_array = { array = missiles_operation_drift_states ROOT = { set_variable = { missiles_operation_impact_state = PREV } } }
		set_country_flag = missiles_operation_has_impact
	}
	else_if = {
		limit = {
			check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.neutral_accident compare = equals }
			check_variable = { var = missiles_operation_neutral_states^num value = constant:missiles_program_value.zero compare = greater_than }
		}
		random_scope_in_array = { array = missiles_operation_neutral_states ROOT = { set_variable = { missiles_operation_impact_state = PREV } } }
		set_country_flag = missiles_operation_has_impact
		set_country_flag = missiles_neutral_incident_active
	}
	else_if = {
		limit = {
			check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.self_strike compare = equals }
			check_variable = { var = missiles_operation_self_states^num value = constant:missiles_program_value.zero compare = greater_than }
		}
		random_scope_in_array = { array = missiles_operation_self_states ROOT = { set_variable = { missiles_operation_impact_state = PREV } } }
		set_country_flag = missiles_operation_has_impact
	}
	else_if = {
		limit = {
			check_variable = { var = missiles_operation_guidance_outcome value = constant:missiles_guidance_outcome.site_accident compare = equals }
			has_variable = missiles_operation_launch_site
		}
		set_variable = { missiles_operation_impact_state = missiles_operation_launch_site }
		set_country_flag = missiles_operation_has_impact
	}
	if = {
		limit = { has_country_flag = missiles_operation_has_impact has_variable = missiles_operation_impact_state }
		var:missiles_operation_impact_state = {
			ROOT = { set_variable = { missiles_operation_impact_state_id = PREV.id } }
			CONTROLLER = {
				if = {
					limit = { exists = yes }
					ROOT = {
						set_variable = { missiles_operation_impact_country = PREV }
						set_variable = { missiles_operation_impact_country_id = PREV.id }
					}
				}
			}
		}
	}
}

missiles_resolve_operation_interception = {
	# COUNTRY scope. The impact state is already frozen. Interception consumes a
	# bounded share of the frozen launch receipt and never erases a saturation
	# barrage completely.
	set_variable = { missiles_operation_launched = missiles_operation_missile_cost }
	set_variable = { missiles_operation_hits = missiles_operation_missile_cost }
	set_variable = { missiles_operation_intercepted = constant:missiles_program_value.zero }
	set_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_base }
	set_variable = { missiles_operation_interception_result = constant:missiles_interception_result.none }
	if = {
		limit = { has_country_flag = missiles_operation_has_impact has_variable = missiles_operation_impact_state }
		var:missiles_operation_impact_state = {
			if = { limit = { anti_air_building > constant:missiles_program_value.zero } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_antiair_step } } }
			if = { limit = { anti_air_building > constant:missiles_program_value.one } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_antiair_step } } }
			if = { limit = { anti_air_building > constant:missiles_program_value.two } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_antiair_step } } }
			if = { limit = { anti_air_building > constant:missiles_program_value.three } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_antiair_step } } }
			if = { limit = { anti_air_building > constant:missiles_program_value.four } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_antiair_step } } }
			if = { limit = { radar_station > constant:missiles_program_value.zero } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_radar_step } } }
			if = { limit = { radar_station > constant:missiles_program_value.one } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_radar_step } } }
			if = { limit = { radar_station > constant:missiles_program_value.two } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_radar_step } } }
			if = { limit = { radar_station > constant:missiles_program_value.three } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_radar_step } } }
			if = { limit = { radar_station > constant:missiles_program_value.four } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_radar_step } } }
			CONTROLLER = {
				if = { limit = { has_country_flag = missiles_retaliation_network_active } ROOT = { add_to_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_warning_step } } }
			}
		}
		if = {
			limit = { check_variable = { var = missiles_operation_strike_profile value = constant:missiles_strike_profile.saturation compare = equals } }
			subtract_from_variable = { missiles_operation_interception_score = constant:missiles_resolution.interception_saturation_step }
		}
		clamp_variable = { var = missiles_operation_interception_score min = constant:missiles_program_value.zero max = constant:missiles_program_value.maximum }
		if = {
			limit = { check_variable = { var = missiles_operation_interception_score value = constant:missiles_resolution.interception_complete_threshold compare = greater_than_or_equals } }
			if = {
				limit = { check_variable = { var = missiles_operation_strike_profile value = constant:missiles_strike_profile.saturation compare = equals } }
				set_variable = { missiles_operation_intercepted = missiles_operation_launched }
				multiply_variable = { var = missiles_operation_intercepted value = constant:missiles_resolution.interception_saturation_maximum_fraction }
				round_variable = missiles_operation_intercepted
				set_temp_variable = { missiles_operation_maximum_saturation_interception = missiles_operation_launched }
				subtract_from_temp_variable = { missiles_operation_maximum_saturation_interception = constant:missiles_resolution.interception_minimum_saturation_hits }
				if = { limit = { check_variable = { var = missiles_operation_intercepted value = missiles_operation_maximum_saturation_interception compare = greater_than } } set_variable = { missiles_operation_intercepted = missiles_operation_maximum_saturation_interception } }
				set_variable = { missiles_operation_interception_result = constant:missiles_interception_result.saturation_penetration }
			}
			else = {
				set_variable = { missiles_operation_intercepted = missiles_operation_launched }
				set_variable = { missiles_operation_interception_result = constant:missiles_interception_result.complete }
			}
		}
		else_if = {
			limit = { check_variable = { var = missiles_operation_interception_score value = constant:missiles_resolution.interception_partial_threshold compare = greater_than_or_equals } }
			set_variable = { missiles_operation_intercepted = constant:missiles_program_value.one }
			set_variable = { missiles_operation_interception_result = constant:missiles_interception_result.partial }
		}
		set_variable = { missiles_operation_hits = missiles_operation_launched }
		subtract_from_variable = { missiles_operation_hits = missiles_operation_intercepted }
		clamp_variable = { var = missiles_operation_hits min = constant:missiles_program_value.zero max = missiles_operation_launched }
		if = {
			limit = { check_variable = { var = missiles_operation_hits value = constant:missiles_program_value.zero compare = less_than_or_equals } }
			clr_country_flag = missiles_operation_has_impact
		}
		if = { limit = { check_variable = { var = missiles_operation_intercepted value = constant:missiles_program_value.zero compare = greater_than } } set_country_flag = missiles_operation_interception_recorded }
	}
}

	else_if = { limit = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.thermonuclear compare = equals } } set_variable = { missiles_current_incident_type = constant:missiles_incident_type.thermonuclear_delivery } }

			multiply_temp_variable = { var = missiles_civil_war_child_reserve value = missiles_civil_war_received_capacity }

		if = { limit = { event_target:missiles_civil_war_parent = { has_country_flag = missiles_evolution_automatic_retaliation } } set_country_flag = missiles_evolution_automatic_retaliation }

	}
	clear_array = missiles_launch_site_states
