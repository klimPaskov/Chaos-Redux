# ============================================================================
# CHAOS REDUX - EVENT 035 DECISION AND MISSION EFFECTS
# ============================================================================
# Overview:
# - Binds every Event 35 decision and mission identifier to a concrete,
#   receipt-backed outcome.
# - All national outcomes pay one declared cost family and pass Severity
#   through one clamped public path.
# - State and country targets retain their scope-specific consequences while
#   ROOT owns the national ledger and public lifecycle.
# - Missions use flags as compact objective receipts and never refund a cost.
# ============================================================================

great_depression_apply_public_action_delta = {
	if = {
		limit = { NOT = { has_variable = great_depression_depression_severity } }
		set_variable = { great_depression_depression_severity = constant:great_depression_api.zero }
	}
	add_to_variable = { great_depression_depression_severity = great_depression_action_delta }
	great_depression_clamp_public_severity = yes
	set_variable = { great_depression_severity_trend = great_depression_action_delta }
	if = {
		limit = { check_variable = { var = great_depression_depression_severity value = great_depression_episode_peak_severity compare = greater_than } }
		set_variable = { great_depression_episode_peak_severity = great_depression_depression_severity }
	}
	great_depression_set_qualitative_band = yes
	great_depression_update_phase_hysteresis = yes
	great_depression_apply_active_ideas = yes
	great_depression_refresh_national_modifier = yes
}

great_depression_record_action_receipt = {
	if = {
		limit = { has_variable = great_depression_successful_action_count }
		add_to_variable = { great_depression_successful_action_count = constant:great_depression_api.one }
	}
	else = {
		set_variable = { great_depression_successful_action_count = constant:great_depression_api.one }
	}
	set_variable = { great_depression_last_action_day = global.num_days }
	set_temp_variable = { great_depression_action_lock_duration = constant:great_depression_action.action_lock_days }
	set_country_flag = { flag = great_depression_action_lock days = great_depression_action_lock_duration }
}

# Rebuild the public mission-capacity ledger from the engine's active mission
# state. The activation triggers remain the authoritative hard gate; this
# ledger is a bounded receipt for UI, audits, and post-load synchronization.
great_depression_refresh_mission_capacity_ledger = {
	if = {
		limit = {
			has_active_mission = great_depression_reopen_depression_center
			has_variable = great_depression_selected_center_state
			NOT = { has_variable = great_depression_mission_reopen_center_state_id }
		}
		set_variable = { great_depression_mission_reopen_center_state_id = great_depression_selected_center_state }
		set_variable = { great_depression_mission_reopen_center_episode_id = great_depression_active_episode_id }
	}
	if = {
		limit = {
			has_active_mission = great_depression_national_employment_guarantee
			NOT = { has_variable = great_depression_employment_guarantee_due_day }
		}
		set_variable = { great_depression_employment_guarantee_due_day = global.num_days }
		add_to_variable = { great_depression_employment_guarantee_due_day = constant:great_depression_event.employment_guarantee_proof_days }
	}
	set_variable = { great_depression_active_main_mission_count = constant:great_depression_api.zero }
	set_variable = { great_depression_active_support_mission_count = constant:great_depression_api.zero }
	set_variable = { great_depression_active_mission_count = constant:great_depression_api.zero }
	if = { limit = { has_active_mission = great_depression_halt_the_panic } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_reopen_depression_center } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_prevent_a_relapse } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_maximum_severity_emergency } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_prevent_national_breakdown } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_contain_financial_contagion } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_prevent_general_strike } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_international_reconstruction } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_coordinate_international_rescue_mission } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_emergency_government_mandate } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_keep_essential_freight_moving } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_national_employment_guarantee } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_complete_national_recovery_plan } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_certify_cleared_economy } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_restore_essential_production } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	add_to_variable = { great_depression_active_mission_count = great_depression_active_main_mission_count }
	add_to_variable = { great_depression_active_mission_count = great_depression_active_support_mission_count }
	if = {
		limit = {
			check_variable = {
				var = great_depression_active_mission_count
				value = constant:great_depression_event.maximum_active_missions
				compare = greater_than_or_equals
			}
		}
		set_country_flag = great_depression_mission_capacity_committed
	}
	else = {
		clear_country_flag = great_depression_mission_capacity_committed
	}
}

# COUNTRY scope. Exposed and worldwide-pressure countries can pay for bounded
# response and reconstruction actions without silently receiving a national
# Depression Severity ledger or the full Event 35 crisis surface.

		limit = { great_depression_active_crisis_is_coherent = yes }
		great_depression_commit_country_action = yes

		great_depression_record_action_receipt = yes
		clear_variable = great_depression_cost_family
	}
}

great_depression_record_center_positive_resolution = {
	# A positive treatment resolves a centre once, even if later pressure changes
	# its qualitative status. The receipt prevents repeated decisions from
	# farming the recovery gate.
	if = {
		limit = { NOT = { has_state_flag = great_depression_center_positive_resolution_receipt } }
		set_state_flag = great_depression_center_positive_resolution_receipt
		if = {
			limit = { has_state_flag = great_depression_inherited_center }
			ROOT = { set_country_flag = great_depression_inherited_region_rescued }
		}
		ROOT = {
			set_country_flag = great_depression_center_mission_complete
			if = {
				limit = { has_variable = great_depression_centers_restructured_count }
				add_to_variable = { great_depression_centers_restructured_count = constant:great_depression_api.one }
			}
			else = {
				set_variable = { great_depression_centers_restructured_count = constant:great_depression_api.one }
			}
		}
	}
}

great_depression_remove_center_positive_resolution_credit = {
	if = {
		limit = { has_state_flag = great_depression_center_positive_resolution_receipt }
		ROOT = {
			if = {
				limit = {
					has_variable = great_depression_centers_restructured_count
					check_variable = { var = great_depression_centers_restructured_count value = constant:great_depression_api.zero compare = greater_than }
				}
				subtract_from_variable = { great_depression_centers_restructured_count = constant:great_depression_api.one }
			}
		}
		clr_state_flag = great_depression_center_positive_resolution_receipt
	}
}

great_depression_increment_global_contribution = {
	if = {
		limit = {
			has_global_flag = great_depression_worldwide_episode_active
			NOT = { has_global_flag = great_depression_worldwide_recovery_secured }
			NOT = { check_variable = { var = global.great_depression_global_stage value = constant:great_depression_global_stage.recovery compare = greater_than_or_equals } }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = great_depression_contribution_global_episode_id }
					NOT = { check_variable = { var = great_depression_contribution_global_episode_id value = global.great_depression_global_episode_id compare = equals } }
				}
			}
			set_variable = { great_depression_contribution_global_episode_id = global.great_depression_global_episode_id }
			set_variable = { great_depression_global_contribution_count = constant:great_depression_api.zero }
			clear_array = great_depression_global_contribution_receipt_ids
			clear_country_flag = great_depression_global_contribution_made
			clear_country_flag = great_depression_global_recipient_improved
			clear_country_flag = great_depression_global_major_recipient_improved
			clear_country_flag = great_depression_global_public_works_complete
			clear_country_flag = great_depression_international_debt_conference_complete
			clear_country_flag = great_depression_global_recovery_conference_complete
			clear_country_flag = great_depression_emergency_trade_clearing_complete
			clear_country_flag = great_depression_reconstruction_employment_fund
			clear_country_flag = great_depression_domestic_market_protected
			clear_country_flag = great_depression_industrial_demand_coordinated
			clear_country_flag = great_depression_regional_clearing_contribution
			clear_country_flag = great_depression_supplier_compact_contribution
			clear_country_flag = great_depression_international_reconstruction_complete
		}
		if = {
			limit = { NOT = { is_in_array = { array = global.great_depression_global_contribution_countries value = THIS } } }
			add_to_array = { array = global.great_depression_global_contribution_countries value = THIS }
			set_temp_variable = { great_depression_contribution_recorded_value = num_of_civilian_factories }
			multiply_temp_variable = { great_depression_contribution_recorded_value = constant:great_depression_api.two }
			set_temp_variable = { great_depression_contribution_military_weight = num_of_military_factories }
			set_temp_variable = { great_depression_contribution_naval_weight = num_of_naval_factories }
			add_to_temp_variable = { great_depression_contribution_recorded_value = great_depression_contribution_military_weight }
			add_to_temp_variable = { great_depression_contribution_recorded_value = great_depression_contribution_naval_weight }
			clamp_temp_variable = {
				var = great_depression_contribution_recorded_value
				min = constant:great_depression_api.one
				max = constant:great_depression_event.public_value_maximum
			}
			add_to_array = { array = global.great_depression_global_contribution_values value = great_depression_contribution_recorded_value }
			if = {
				limit = { has_variable = global.great_depression_global_contribution_industrial_weight }
				add_to_variable = { global.great_depression_global_contribution_industrial_weight = great_depression_contribution_recorded_value }
			}
			else = { set_variable = { global.great_depression_global_contribution_industrial_weight = great_depression_contribution_recorded_value } }
		}
		great_depression_allocate_receipt = yes
		set_variable = { great_depression_global_contribution_receipt_id = great_depression_last_receipt_id }
		add_to_array = { array = global.great_depression_global_contribution_receipts value = great_depression_global_contribution_receipt_id }
		add_to_array = { array = great_depression_global_contribution_receipt_ids value = great_depression_global_contribution_receipt_id }
		add_to_variable = { great_depression_global_contribution_count = constant:great_depression_api.one }
		set_country_flag = great_depression_global_contribution_made
		if = {
			limit = {
				is_ai = no
				NOT = { is_debug = yes }
				NOT = { has_country_flag = chaosx_test_country_initialized }
				NOT = { has_country_flag = force_trigger_mode_enabled }
				is_special_chaos_country = no
			}
			set_country_flag = great_depression_achievement_recovery_of_nations_candidate
			set_variable = { great_depression_achievement_recovery_of_nations_episode_id = global.great_depression_global_episode_id }
		}
	}
}

great_depression_contain_outgoing_exposures = {
	set_variable = { great_depression_contagion_containment_source_count = constant:great_depression_api.zero }
	set_temp_variable = { great_depression_containment_row_due_day = global.num_days }
	add_to_temp_variable = { great_depression_containment_row_due_day = constant:great_depression_contagion_row.pair_cooldown_days }
	if = {
		limit = {
			has_variable = great_depression_contagion_exposure_targets^num
			check_variable = { var = great_depression_contagion_exposure_targets^num value = constant:great_depression_api.zero compare = greater_than }
		}
		for_loop_effect = {
			start = 0
			end = great_depression_contagion_exposure_targets^num
			value = great_depression_containment_target_index
			set_variable = { great_depression_contagion_target_aid_state_entries^great_depression_containment_target_index = constant:great_depression_contagion_aid_state.contained }
			set_variable = { great_depression_contagion_target_material_source_entries^great_depression_containment_target_index = constant:great_depression_contagion_material_source.partner_aid }
			set_variable = { great_depression_contagion_target_cooldown_day_entries^great_depression_containment_target_index = great_depression_containment_row_due_day }
			var:great_depression_contagion_exposure_targets^great_depression_containment_target_index = {
				if = {
					limit = {
						NOT = { has_country_flag = great_depression_active }
						NOT = { has_country_flag = great_depression_contagion_converted }
					}
					set_country_flag = great_depression_contagion_contained
					set_country_flag = great_depression_contagion_containment_action_receipt
					clear_country_flag = great_depression_contagion_exposure_open
					remove_dynamic_modifier = { modifier = great_depression_financial_contagion }
					set_variable = { great_depression_contagion_response_stage = constant:great_depression_exposure_stage.resolved }
					if = {
						limit = { has_variable = great_depression_contagion_source_countries^num }
						for_loop_effect = {
							start = 0
							end = great_depression_contagion_source_countries^num
							value = great_depression_containment_source_row_index
							var:great_depression_contagion_source_countries^great_depression_containment_source_row_index = {
								if = {
									limit = { tag = ROOT }
									PREV = {
										set_variable = { great_depression_contagion_source_aid_state_entries^great_depression_containment_source_row_index = constant:great_depression_contagion_aid_state.contained }
										set_variable = { great_depression_contagion_source_material_source_entries^great_depression_containment_source_row_index = constant:great_depression_contagion_material_source.partner_aid }
										set_variable = { great_depression_contagion_source_cooldown_day_entries^great_depression_containment_source_row_index = great_depression_containment_row_due_day }
									}
								}
							}
						}
					}
					ROOT = { add_to_variable = { great_depression_contagion_containment_source_count = constant:great_depression_api.one } }
				}
			}
		}
	}
	if = {
		limit = {
			has_variable = great_depression_contagion_required_target_count
			check_variable = { var = great_depression_contagion_containment_source_count value = great_depression_contagion_required_target_count compare = greater_than_or_equals }
		}
		set_country_flag = great_depression_contagion_contained
	}
}

# -----------------------------------------------------------------------------
# Partner-targeted actions. The target records the relationship receipt and
# ROOT pays the source-country cost and owns Severity.
# -----------------------------------------------------------------------------

# COUNTRY target scope with ROOT as the actor. The timed flag is keyed by the
# actor tag and lives on this target, so unrelated partners remain available
# while the same source-target pair cannot farm several adjacent actions.
great_depression_register_partner_pair_cooldown = {
	set_temp_variable = { great_depression_partner_pair_cooldown_duration = constant:great_depression_contagion_row.pair_cooldown_days }
	meta_effect = {
		text = {
			set_country_flag = {
				flag = great_depression_partner_pair_cooldown_[GREAT_DEPRESSION_PARTNER_TAG]
				days = great_depression_partner_pair_cooldown_duration
			}
		}
		GREAT_DEPRESSION_PARTNER_TAG = "[ROOT.GetTag]"
	}
}

# COUNTRY actor scope. Mirrors a completed partner transaction into the exact
# sparse contagion row on both countries. Negative deltas are proved aid or
# diversification; positive deltas are deliberate abandonment or exploitation.
great_depression_record_partner_contagion_outcome = {
	if = {
		limit = {
			great_depression_active_crisis_is_coherent = yes
			has_country_flag = great_depression_evolution_financial_contagion
			has_event_target = great_depression_partner_action_target
			is_in_array = { array = great_depression_contagion_exposure_targets value = event_target:great_depression_partner_action_target }
		}
		set_temp_variable = { great_depression_partner_row_due_day = global.num_days }
		add_to_temp_variable = { great_depression_partner_row_due_day = constant:great_depression_contagion_row.pair_cooldown_days }
		set_temp_variable = { great_depression_partner_row_aid_state = constant:great_depression_contagion_aid_state.aid_received }
		set_temp_variable = { great_depression_partner_row_material_source = constant:great_depression_contagion_material_source.partner_aid }
		if = {
			limit = { check_variable = { var = great_depression_action_delta value = constant:great_depression_api.zero compare = greater_than } }
			set_temp_variable = { great_depression_partner_row_aid_state = constant:great_depression_contagion_aid_state.abandoned }
			set_temp_variable = { great_depression_partner_row_material_source = constant:great_depression_contagion_material_source.abandonment }
		}
		for_loop_effect = {
			start = 0
			end = great_depression_contagion_exposure_targets^num
			value = great_depression_partner_outgoing_row_index
			var:great_depression_contagion_exposure_targets^great_depression_partner_outgoing_row_index = {
				if = {
					limit = { tag = event_target:great_depression_partner_action_target }
					ROOT = {
						set_variable = { great_depression_contagion_target_aid_state_entries^great_depression_partner_outgoing_row_index = great_depression_partner_row_aid_state }
						set_variable = { great_depression_contagion_target_material_source_entries^great_depression_partner_outgoing_row_index = great_depression_partner_row_material_source }
						set_variable = { great_depression_contagion_target_cooldown_day_entries^great_depression_partner_outgoing_row_index = great_depression_partner_row_due_day }
					}
				}
			}
		}
		event_target:great_depression_partner_action_target = {
			if = {
				limit = { has_variable = great_depression_contagion_source_countries^num }
				for_loop_effect = {
					start = 0
					end = great_depression_contagion_source_countries^num
					value = great_depression_partner_incoming_row_index
					var:great_depression_contagion_source_countries^great_depression_partner_incoming_row_index = {
						if = {
							limit = { tag = ROOT }
							PREV = {
								set_variable = { great_depression_contagion_source_aid_state_entries^great_depression_partner_incoming_row_index = great_depression_partner_row_aid_state }
								set_variable = { great_depression_contagion_source_material_source_entries^great_depression_partner_incoming_row_index = great_depression_partner_row_material_source }
								set_variable = { great_depression_contagion_source_cooldown_day_entries^great_depression_partner_incoming_row_index = great_depression_partner_row_due_day }
							}
						}
					}
				}
			}
		}
	}
}

great_depression_commit_partner_action = {
	if = {
		limit = { has_event_target = great_depression_partner_action_target }
		event_target:great_depression_partner_action_target = {
			save_event_target_as = great_depression_last_partner
		}
		if = {
			limit = {
				great_depression_active_crisis_is_coherent = yes
				has_country_flag = great_depression_evolution_financial_contagion
				event_target:great_depression_partner_action_target = { great_depression_is_valid_contagion_network_target = yes }
				NOT = { is_in_array = { array = great_depression_contagion_exposure_targets value = event_target:great_depression_partner_action_target } }
			}
			event_target:great_depression_partner_action_target = { save_event_target_as = great_depression_exposure_target }
			set_variable = { great_depression_exposure_stage = constant:great_depression_exposure_stage.pressured }
			if = {
				limit = { check_variable = { var = great_depression_action_delta value = constant:great_depression_api.zero compare = greater_than } }
				set_variable = { great_depression_exposure_stage = constant:great_depression_exposure_stage.dependent }
			}
			set_variable = { great_depression_exposure_relationship_type = constant:great_depression_exposure_relationship.event_agreement }
			set_variable = { great_depression_exposure_relationship_proof = constant:great_depression_api.one }
			set_variable = { great_depression_exposure_aid_state = constant:great_depression_contagion_aid_state.aid_received }
			set_variable = { great_depression_exposure_material_source = constant:great_depression_contagion_material_source.partner_aid }
			if = {
				limit = { check_variable = { var = great_depression_action_delta value = constant:great_depression_api.zero compare = greater_than } }
				set_variable = { great_depression_exposure_aid_state = constant:great_depression_contagion_aid_state.abandoned }
				set_variable = { great_depression_exposure_material_source = constant:great_depression_contagion_material_source.abandonment }
			}
			set_variable = { great_depression_exposure_cooldown_day = global.num_days }
			add_to_variable = { great_depression_exposure_cooldown_day = constant:great_depression_contagion_row.pair_cooldown_days }
			great_depression_apply_exposure = yes
			clear_variable = great_depression_exposure_stage
			clear_variable = great_depression_exposure_relationship_type
			clear_variable = great_depression_exposure_relationship_proof
			clear_variable = great_depression_exposure_aid_state
			clear_variable = great_depression_exposure_material_source
			clear_variable = great_depression_exposure_cooldown_day
		}
		great_depression_commit_external_action = yes
		great_depression_record_partner_contagion_outcome = yes
		event_target:great_depression_partner_action_target = { great_depression_register_partner_pair_cooldown = yes }
		if = {
			limit = { has_global_flag = great_depression_worldwide_episode_active }
			set_country_flag = great_depression_global_recipient_action_made
			if = {
				limit = {
					check_variable = {
						var = great_depression_action_delta
						value = constant:great_depression_api.zero
						compare = less_than
					}
				}
				if = {
					limit = { great_depression_active_crisis_is_coherent = yes }
					set_country_flag = great_depression_global_recipient_improved
					if = {
						limit = { is_major = yes }
						set_country_flag = great_depression_global_major_recipient_improved
					}
				}
				else = {
					event_target:great_depression_partner_action_target = {
						set_country_flag = great_depression_global_recipient_improved
						ROOT = { set_country_flag = great_depression_global_recipient_improved }
						if = {
							limit = { is_major = yes }
							ROOT = { set_country_flag = great_depression_global_major_recipient_improved }
						}
					}
				}
			}
		}
	}
}

		great_depression_apply_public_action_delta = yes
		if = {
			# Active episode ids are globally unique. The provider therefore keeps a
			# bounded receipt set and can count once for each source episode without
			# blocking aid to another source or a later repeat crisis.
			limit = {
				ROOT = {
					NOT = {
						is_in_array = {
							array = great_depression_contagion_supported_episode_entries
							value = event_target:great_depression_partner_action_target.great_depression_active_episode_id
						}
					}
				}
			}
			ROOT = {
				if = {
					limit = {
						has_variable = great_depression_contagion_supported_episode_entries^num
						check_variable = {
							var = great_depression_contagion_supported_episode_entries^num
							value = constant:great_depression_event.contagion_source_cap
							compare = greater_than_or_equals
						}
					}
					remove_from_array = { array = great_depression_contagion_supported_episode_entries index = 0 }
				}
				add_to_array = {
					array = great_depression_contagion_supported_episode_entries
					value = event_target:great_depression_partner_action_target.great_depression_active_episode_id
				}
				set_country_flag = great_depression_contagion_support_counted
			}
			if = {
				limit = { has_variable = great_depression_contagion_containment_source_count }
				add_to_variable = { great_depression_contagion_containment_source_count = constant:great_depression_api.one }
			}
			else = { set_variable = { great_depression_contagion_containment_source_count = constant:great_depression_api.one } }
		}
		set_country_flag = great_depression_contagion_containment_action_receipt
		set_temp_variable = { great_depression_support_row_due_day = global.num_days }
		add_to_temp_variable = { great_depression_support_row_due_day = constant:great_depression_contagion_row.pair_cooldown_days }
		if = {
			limit = {
				has_variable = great_depression_contagion_exposure_targets^num
				is_in_array = { array = great_depression_contagion_exposure_targets value = ROOT }
			}
			for_loop_effect = {
				start = 0
				end = great_depression_contagion_exposure_targets^num
				value = great_depression_supported_target_row_index
				var:great_depression_contagion_exposure_targets^great_depression_supported_target_row_index = {
					if = {
						limit = { tag = ROOT }
						PREV = {
							set_variable = { great_depression_contagion_target_aid_state_entries^great_depression_supported_target_row_index = constant:great_depression_contagion_aid_state.aid_received }
							set_variable = { great_depression_contagion_target_material_source_entries^great_depression_supported_target_row_index = constant:great_depression_contagion_material_source.partner_aid }
							set_variable = { great_depression_contagion_target_cooldown_day_entries^great_depression_supported_target_row_index = great_depression_support_row_due_day }
						}
					}
				}
			}
		}
		if = {
			limit = {
				has_variable = great_depression_contagion_required_target_count
				check_variable = { var = great_depression_contagion_containment_source_count value = great_depression_contagion_required_target_count compare = greater_than_or_equals }
			}
			set_country_flag = great_depression_contagion_contained
		}
	}

				has_variable = great_depression_contagion_source_countries^num

				start = 0
				end = great_depression_contagion_source_countries^num
				value = great_depression_supporting_source_row_index
				var:great_depression_contagion_source_countries^great_depression_supporting_source_row_index = {
					if = {
						limit = { tag = event_target:great_depression_supported_source_country }
						PREV = {
							set_variable = { great_depression_contagion_source_aid_state_entries^great_depression_supporting_source_row_index = constant:great_depression_contagion_aid_state.aid_offered }
							set_variable = { great_depression_contagion_source_material_source_entries^great_depression_supporting_source_row_index = constant:great_depression_contagion_material_source.partner_aid }
							set_variable = { great_depression_contagion_source_cooldown_day_entries^great_depression_supporting_source_row_index = great_depression_support_row_due_day }
						}
					}
				}
			}
		}
		set_variable = { great_depression_cost_family = constant:great_depression_cost_family.contagion }
		set_temp_variable = { great_depression_action_delta = constant:great_depression_action.strong_relief }

		clear_country_flag = great_depression_contagion_exposure_open
		set_country_flag = great_depression_contagion_contained
		set_country_flag = great_depression_contagion_containment_action_receipt
		remove_dynamic_modifier = { modifier = great_depression_financial_contagion }
