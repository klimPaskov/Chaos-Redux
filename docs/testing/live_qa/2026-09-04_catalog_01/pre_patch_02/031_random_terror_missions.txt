# ============================================================================
# CHAOS REDUX - EVENT 031 RANDOM TERROR MISSIONS
# ============================================================================
# Missions remain ordinary decision missions so the response category stays
# compact.  Decisions reserve mission_slot_1, mission_slot_2, or mission_slot_3
# before activation.  Non-selectable missions remain visible until their real
# objective is satisfied, then complete automatically without a second click.
# The state subject flags are the exact map targets selected by the decisions.
# ============================================================================

@rt_mission_ai_low = 1

random_terror_contain_attack_wave = {
	icon = GFX_decision_mission_random_terror_contain_attack_wave
	days_mission_timeout = constant:random_terror_tuning.containment_days
	selectable_mission = no
	fire_only_once = no
	is_good = yes
	activation = {
		random_terror_can_government_response = yes
		has_country_flag = random_terror_mission_slot_1
		any_state = { has_state_flag = random_terror_containment_mission_subject }
	}
	available = {
		any_state = {
			has_state_flag = random_terror_containment_mission_subject
			has_state_flag = random_terror_containment_success
		}
	}
	cancel_trigger = {
		OR = {
			NOT = { random_terror_can_government_response = yes }
			has_global_flag = world_end
			NOT = { has_country_flag = random_terror_mission_slot_1 }
			NOT = { any_state = { has_state_flag = random_terror_containment_mission_subject } }
		}
	}
	cancel_effect = {
		random_terror_cancel_containment_mission = yes
		every_state = {
			limit = { has_state_flag = random_terror_containment_mission_subject }
			clr_state_flag = random_terror_containment_mission_subject
			clr_state_flag = random_terror_containment_success
		}
		clr_country_flag = random_terror_mission_slot_1
		clr_country_flag = random_terror_security_surge_active
	}
	complete_effect = {
		custom_effect_tooltip = random_terror_contain_attack_wave_complete_tt
		random_terror_resolve_containment_mission_success = yes
		every_state = {
			limit = { has_state_flag = random_terror_containment_mission_subject }
			clr_state_flag = random_terror_containment_mission_subject
			clr_state_flag = random_terror_containment_success
		}
		clr_country_flag = random_terror_mission_slot_1
		clr_country_flag = random_terror_security_surge_active
	}
	timeout_effect = {
		if = {
			limit = {
				any_state = {
					has_state_flag = random_terror_containment_mission_subject
					NOT = { check_variable = { random_terror_state_activity > constant:random_terror_tuning.state_activity_active_threshold } }
				}
			}
			custom_effect_tooltip = random_terror_contain_attack_wave_partial_tt
			random_terror_resolve_containment_mission_partial = yes
		}
		else = {
			custom_effect_tooltip = random_terror_contain_attack_wave_failure_tt
			random_terror_resolve_containment_mission_failure = yes
		}
		every_state = {
			limit = { has_state_flag = random_terror_containment_mission_subject }
			clr_state_flag = random_terror_containment_mission_subject
			clr_state_flag = random_terror_containment_success
		}
		clr_country_flag = random_terror_mission_slot_1
		clr_country_flag = random_terror_security_surge_active
	}
	ai_will_do = {
		base = @rt_mission_ai_high
		modifier = { factor = @rt_mission_ai_urgent_factor check_variable = { random_terror_pressure > constant:random_terror_value.high } }
		modifier = { factor = @rt_mission_ai_high_factor has_country_flag = random_terror_failed_raid_recorded }
		modifier = { factor = @rt_mission_ai_high_factor has_war = yes }
	}
}
