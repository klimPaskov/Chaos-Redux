# Event 021 evolution timing. These entries are sampled once per unlocked
# stage by the bounded global-host transition queue, then stored as absolute
# due dates. They never create a periodic world scan.

random_civil_war_evolution_i_days = {
	base = constant:random_civil_war_evolution_mtth.evolution_i_base_days

	modifier = {
		factor = constant:random_civil_war_evolution_mtth.authority_failure_factor
		random_civil_war_authority_is_failing = yes
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.large_country_factor
		num_of_controlled_states > constant:random_civil_war_evolution_mtth.large_country_state_threshold
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.extra_actor_factor
		OR = {
			has_country_flag = random_civil_war_secondary_front_active
			has_country_flag = random_civil_war_event6_route_ready
			has_country_flag = random_civil_war_multifront_active
		}
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.near_defeat_factor
		surrender_progress > constant:random_civil_war_evolution_mtth.near_defeat_threshold
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.settlement_progress_factor
		has_country_flag = random_civil_war_settlement_phase
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.no_extra_actor_factor
		NOT = {
			OR = {
				has_country_flag = random_civil_war_secondary_front_active
				has_country_flag = random_civil_war_event6_route_ready
				has_country_flag = random_civil_war_multifront_active
			}
		}
	}
}

random_civil_war_evolution_ii_days = {
	base = constant:random_civil_war_evolution_mtth.evolution_ii_base_days

	modifier = {
		factor = constant:random_civil_war_evolution_mtth.border_route_factor
		has_country_flag = random_civil_war_land_border_exposure_ready
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.sponsor_route_factor
		has_country_flag = random_civil_war_sponsor_commitment_active
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.independence_front_factor
		has_country_flag = random_civil_war_event6_front_active
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.multifront_factor
		has_country_flag = random_civil_war_multifront_active
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.settlement_progress_factor
		has_country_flag = random_civil_war_settlement_phase
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.strong_neighbor_factor
		any_neighbor_country = { stability > constant:random_civil_war_evolution_mtth.strong_neighbor_stability_threshold }
	}
	modifier = {
		factor = constant:random_civil_war_evolution_mtth.no_exposure_route_factor
		NOT = {
			OR = {
				has_country_flag = random_civil_war_land_border_exposure_ready
				has_country_flag = random_civil_war_sponsor_commitment_active
				has_country_flag = random_civil_war_event6_front_active
			}
		}
	}
}

random_civil_war_evolution_iii_days = {
	base = constant:random_civil_war_evolution_mtth.evolution_iii_base_days

	modifier = {
		factor = constant:random_civil_war_evolution_mtth.severe_war_factor
		OR = {
			check_variable = { var = random_civil_war_opening_severity value = constant:random_civil_war_severity.severe compare = equals }
			check_variable = { var = random_civil_war_opening_severity value = constant:random_civil_war_severity.critical compare = equals }
		}
	}
}
