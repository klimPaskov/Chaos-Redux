# Repression-owned deterministic quote and inclusive affordability predicates.
# Country scope; STATE is an explicit state scope (FROM or var:camp_selected_state_id).
# Temporary quote outputs are unscoped. Callers retaining outputs initialize them first.

camp_rework_prepare_site_cost_quote = {
	set_temp_variable = { camp_site_quote_levels = constant:camp_site_cost.minimum_levels }
	$STATE$ = {
		set_temp_variable = { camp_site_quote_levels = building_level@concentration_camp }
		add_to_temp_variable = { camp_site_quote_levels = building_level@extermination_camp }
		add_to_temp_variable = { camp_site_quote_levels = building_level@gulag_labor_camp_network }
	}
	clamp_temp_variable = { var = camp_site_quote_levels min = constant:camp_site_cost.minimum_levels }
	set_temp_variable = { camp_site_quote_labor_pp = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_labor_pp = constant:camp_site_cost.labor_pp_per_level }
	add_to_temp_variable = { camp_site_quote_labor_pp = constant:camp_site_cost.labor_pp_base }
	round_temp_variable = camp_site_quote_labor_pp
	set_temp_variable = { camp_site_quote_labor_motorized = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_labor_motorized = constant:camp_site_cost.labor_motorized_per_level }
	add_to_temp_variable = { camp_site_quote_labor_motorized = constant:camp_site_cost.labor_motorized_base }
	round_temp_variable = camp_site_quote_labor_motorized
	set_temp_variable = { camp_site_quote_labor_trains = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_labor_trains = constant:camp_site_cost.labor_trains_per_level }
	add_to_temp_variable = { camp_site_quote_labor_trains = constant:camp_site_cost.labor_trains_base }
	round_temp_variable = camp_site_quote_labor_trains
	set_temp_variable = { camp_site_quote_labor_support = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_labor_support = constant:camp_site_cost.labor_support_per_level }
	add_to_temp_variable = { camp_site_quote_labor_support = constant:camp_site_cost.labor_support_base }
	round_temp_variable = camp_site_quote_labor_support
	set_temp_variable = { camp_site_quote_inspect_pp = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_inspect_pp = constant:camp_site_cost.inspect_pp_per_level }
	add_to_temp_variable = { camp_site_quote_inspect_pp = constant:camp_site_cost.inspect_pp_base }
	round_temp_variable = camp_site_quote_inspect_pp
	set_temp_variable = { camp_site_quote_dismantle_pp = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_dismantle_pp = constant:camp_site_cost.dismantle_pp_per_level }
	add_to_temp_variable = { camp_site_quote_dismantle_pp = constant:camp_site_cost.dismantle_pp_base }
	round_temp_variable = camp_site_quote_dismantle_pp
	set_temp_variable = { camp_site_quote_dismantle_manpower = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_dismantle_manpower = constant:camp_site_cost.dismantle_manpower_per_level }
	add_to_temp_variable = { camp_site_quote_dismantle_manpower = constant:camp_site_cost.dismantle_manpower_base }
	round_temp_variable = camp_site_quote_dismantle_manpower
	set_temp_variable = { camp_site_quote_dismantle_support = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_dismantle_support = constant:camp_site_cost.dismantle_support_per_level }
	add_to_temp_variable = { camp_site_quote_dismantle_support = constant:camp_site_cost.dismantle_support_base }
	round_temp_variable = camp_site_quote_dismantle_support
	set_temp_variable = { camp_site_quote_evidence_pp = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_evidence_pp = constant:camp_site_cost.evidence_pp_per_level }
	add_to_temp_variable = { camp_site_quote_evidence_pp = constant:camp_site_cost.evidence_pp_base }
	round_temp_variable = camp_site_quote_evidence_pp
	set_temp_variable = { camp_site_quote_evidence_manpower = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_evidence_manpower = constant:camp_site_cost.evidence_manpower_per_level }
	add_to_temp_variable = { camp_site_quote_evidence_manpower = constant:camp_site_cost.evidence_manpower_base }
	round_temp_variable = camp_site_quote_evidence_manpower
	set_temp_variable = { camp_site_quote_evidence_command = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_evidence_command = constant:camp_site_cost.evidence_command_per_level }
	add_to_temp_variable = { camp_site_quote_evidence_command = constant:camp_site_cost.evidence_command_base }
	round_temp_variable = camp_site_quote_evidence_command
	set_temp_variable = { camp_site_quote_evidence_support = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_evidence_support = constant:camp_site_cost.evidence_support_per_level }
	add_to_temp_variable = { camp_site_quote_evidence_support = constant:camp_site_cost.evidence_support_base }
	round_temp_variable = camp_site_quote_evidence_support
	set_temp_variable = { camp_site_quote_restricted_pp = camp_site_quote_levels }
	multiply_temp_variable = { camp_site_quote_restricted_pp = constant:camp_site_cost.restricted_pp_per_level }
	add_to_temp_variable = { camp_site_quote_restricted_pp = constant:camp_site_cost.restricted_pp_base }
	round_temp_variable = camp_site_quote_restricted_pp
	set_temp_variable = { camp_site_quote_reserve_raw = camp_site_quote_labor_motorized }
	multiply_temp_variable = { camp_site_quote_reserve_raw = constant:camp_site_cost.reserve_ratio }
	set_temp_variable = { camp_site_quote_labor_motorized_reserve = camp_site_quote_reserve_raw }
	round_temp_variable = camp_site_quote_labor_motorized_reserve
	if = {
		limit = { check_variable = { camp_site_quote_labor_motorized_reserve < camp_site_quote_reserve_raw } }
		add_to_temp_variable = { camp_site_quote_labor_motorized_reserve = constant:camp_site_cost.integer_step }
	}
	clamp_temp_variable = { var = camp_site_quote_labor_motorized_reserve min = constant:camp_site_cost.minimum_reserve }
	set_temp_variable = { camp_site_quote_labor_motorized_start = camp_site_quote_labor_motorized }
	add_to_temp_variable = { camp_site_quote_labor_motorized_start = camp_site_quote_labor_motorized_reserve }
	set_temp_variable = { camp_site_quote_reserve_raw = camp_site_quote_labor_trains }
	multiply_temp_variable = { camp_site_quote_reserve_raw = constant:camp_site_cost.reserve_ratio }
	set_temp_variable = { camp_site_quote_labor_trains_reserve = camp_site_quote_reserve_raw }
	round_temp_variable = camp_site_quote_labor_trains_reserve
	if = {
		limit = { check_variable = { camp_site_quote_labor_trains_reserve < camp_site_quote_reserve_raw } }
		add_to_temp_variable = { camp_site_quote_labor_trains_reserve = constant:camp_site_cost.integer_step }
	}
	clamp_temp_variable = { var = camp_site_quote_labor_trains_reserve min = constant:camp_site_cost.minimum_reserve }
	set_temp_variable = { camp_site_quote_labor_trains_start = camp_site_quote_labor_trains }
	add_to_temp_variable = { camp_site_quote_labor_trains_start = camp_site_quote_labor_trains_reserve }
	set_temp_variable = { camp_site_quote_reserve_raw = camp_site_quote_labor_support }
	multiply_temp_variable = { camp_site_quote_reserve_raw = constant:camp_site_cost.reserve_ratio }
	set_temp_variable = { camp_site_quote_labor_support_reserve = camp_site_quote_reserve_raw }
	round_temp_variable = camp_site_quote_labor_support_reserve
	if = {
		limit = { check_variable = { camp_site_quote_labor_support_reserve < camp_site_quote_reserve_raw } }
		add_to_temp_variable = { camp_site_quote_labor_support_reserve = constant:camp_site_cost.integer_step }
	}
	clamp_temp_variable = { var = camp_site_quote_labor_support_reserve min = constant:camp_site_cost.minimum_reserve }
	set_temp_variable = { camp_site_quote_labor_support_start = camp_site_quote_labor_support }
	add_to_temp_variable = { camp_site_quote_labor_support_start = camp_site_quote_labor_support_reserve }
	set_temp_variable = { camp_site_quote_dismantle_manpower_k = camp_site_quote_dismantle_manpower }
	multiply_temp_variable = { camp_site_quote_dismantle_manpower_k = constant:camp_site_cost.manpower_to_thousands }
	set_temp_variable = { camp_site_quote_evidence_manpower_k = camp_site_quote_evidence_manpower }
	multiply_temp_variable = { camp_site_quote_evidence_manpower_k = constant:camp_site_cost.manpower_to_thousands }
	always = yes
}

camp_rework_can_pay_site_labor = {
	set_temp_variable = { camp_site_quote_labor_pp = 0 }
	set_temp_variable = { camp_site_quote_labor_motorized_start = 0 }
	set_temp_variable = { camp_site_quote_labor_trains_start = 0 }
	set_temp_variable = { camp_site_quote_labor_support_start = 0 }
	camp_rework_prepare_site_cost_quote = { STATE = $STATE$ }
	check_variable = { var = political_power value = camp_site_quote_labor_pp compare = greater_than_or_equals }
	check_variable = { var = num_equipment@motorized_equipment_1 value = camp_site_quote_labor_motorized_start compare = greater_than_or_equals }
	check_variable = { var = num_equipment@train_equipment_1 value = camp_site_quote_labor_trains_start compare = greater_than_or_equals }
	check_variable = { var = num_equipment@support_equipment_1 value = camp_site_quote_labor_support_start compare = greater_than_or_equals }
}
