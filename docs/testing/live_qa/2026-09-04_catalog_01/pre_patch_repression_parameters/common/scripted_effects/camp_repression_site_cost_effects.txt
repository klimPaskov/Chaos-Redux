# Repression-owned selected-location payments and labor reserve lifecycle.
# Country scope. Caller validates action/target and affordability before payment.
# STATE is the explicit state used by the matching quote and gate; never inferred.

camp_rework_pay_site_labor = {
	set_temp_variable = { camp_site_quote_labor_pp = 0 }
	set_temp_variable = { camp_site_quote_labor_motorized = 0 }
	set_temp_variable = { camp_site_quote_labor_trains = 0 }
	set_temp_variable = { camp_site_quote_labor_support = 0 }
	set_temp_variable = { camp_site_quote_labor_motorized_reserve = 0 }
	set_temp_variable = { camp_site_quote_labor_trains_reserve = 0 }
	set_temp_variable = { camp_site_quote_labor_support_reserve = 0 }
	if = {
		limit = { camp_rework_prepare_site_cost_quote = { STATE = $STATE$ } }
		set_temp_variable = { camp_site_payment_remove = camp_site_quote_labor_pp }
		multiply_temp_variable = { camp_site_payment_remove = -1 }
		add_political_power = camp_site_payment_remove
		set_temp_variable = { camp_site_payment_remove = camp_site_quote_labor_motorized }
		multiply_temp_variable = { camp_site_payment_remove = -1 }
		add_equipment_to_stockpile = { type = motorized_equipment_1 amount = camp_site_payment_remove }
		set_temp_variable = { camp_site_payment_remove = camp_site_quote_labor_trains }
		multiply_temp_variable = { camp_site_payment_remove = -1 }
		add_equipment_to_stockpile = { type = train_equipment_1 amount = camp_site_payment_remove }
		set_temp_variable = { camp_site_payment_remove = camp_site_quote_labor_support }
		multiply_temp_variable = { camp_site_payment_remove = -1 }
		add_equipment_to_stockpile = { type = support_equipment_1 amount = camp_site_payment_remove }
		set_variable = { generic_labor_project_motorized_reserve = camp_site_quote_labor_motorized_reserve }
		set_variable = { generic_labor_project_trains_reserve = camp_site_quote_labor_trains_reserve }
		set_variable = { generic_labor_project_support_reserve = camp_site_quote_labor_support_reserve }
	}
}
