# Selected-site cost display. Each GUI/decision getter recomputes its explicit target quote.
# Countrywide policy and activation cost getters retain their separate contracts.

defined_text = {
	name = GetCampUiSiteLaborPoliticalPower
	text = {
		trigger = {
			set_temp_variable = { camp_site_quote_labor_pp = 0 }
			camp_rework_prepare_site_cost_quote = { STATE = var:camp_selected_state_id }
			NOT = { check_variable = { var = political_power value = camp_site_quote_labor_pp compare = greater_than_or_equals } }
		}
		localization_key = camp_site_cost_labor_pp_red
	}
	text = {
		trigger = {
			set_temp_variable = { camp_site_quote_labor_pp = 0 }
			camp_rework_prepare_site_cost_quote = { STATE = var:camp_selected_state_id }
		}
		localization_key = camp_site_cost_labor_pp
	}
}
