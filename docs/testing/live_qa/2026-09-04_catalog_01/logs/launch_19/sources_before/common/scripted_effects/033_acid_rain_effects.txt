# One-time legacy read. It removes the prototype modifiers and missions but
# only carries forward a state's observable legacy exposure as coverage; it
# never converts a prototype modifier into deaths or an invented loss receipt.
acid_rain_migrate_legacy_runtime = {
	set_variable = { global.acid_rain_migration_was_active = constant:acid_rain_runtime.zero }
	set_variable = { global.acid_rain_migration_legacy_state_count = constant:acid_rain_runtime.zero }
	if = {
		limit = {
			OR = {
				has_global_flag = acid_rain
				has_variable = global.acid_rain_continent
				has_variable = global.acid_clouds_time
			}
			NOT = { has_global_flag = acid_rain_migration_receipt }
		}
		set_global_flag = acid_rain_migration_receipt

# ANY SCOPE. Converts only states that carried a legacy exposure marker into
# frozen coverage receipts. A partial legacy region resumes on its second
# visit, while a fully marked region is closed immediately. If the old
# runtime had no recoverable valid state, the new runtime fails closed instead
# of creating a fresh event with ambiguous migration state.
acid_rain_adopt_legacy_coverage = {
	set_variable = { global.acid_rain_migration_recovered_state_count = constant:acid_rain_runtime.zero }
	if = {
		limit = {
			check_variable = { var = global.acid_rain_migration_was_active value = constant:acid_rain_runtime.one compare = equals }
		}
		for_each_scope_loop = {
			array = global.acid_rain_frozen_states
			if = {
				limit = { has_state_flag = acid_rain_legacy_affected }
				if = {
					limit = { NOT = { has_state_flag = acid_rain_touched } }
					set_state_flag = acid_rain_touched
					add_to_variable = { global.acid_rain_total_touched_states = constant:acid_rain_runtime.loop_increment }
					add_to_variable = { global.acid_rain_region_touched_entries^acid_rain_region_id = constant:acid_rain_runtime.loop_increment }
					add_to_variable = { global.acid_rain_coverage_receipt_count = constant:acid_rain_runtime.loop_increment }
					set_variable = { acid_rain_state_coverage_receipt = global.acid_rain_coverage_receipt_count }
					add_to_array = { array = global.acid_rain_coverage_receipt_states value = THIS }
					add_to_array = { array = global.acid_rain_coverage_receipt_front_entries value = constant:acid_rain_runtime.zero }
					add_to_array = { array = global.acid_rain_coverage_receipt_generation_entries value = global.acid_rain_generation_id }
					add_to_variable = { global.acid_rain_total_exposure = constant:acid_rain_runtime.loop_increment }
				}
				set_variable = { acid_rain_target_generation = global.acid_rain_generation_id }
				set_variable = { acid_rain_state_aftermath_tier = constant:acid_rain_runtime.one }
				add_dynamic_modifier = { modifier = acid_rain_state_ordinary days = global.acid_rain_modifier_refresh_days }
				add_to_variable = { global.acid_rain_migration_recovered_state_count = constant:acid_rain_runtime.loop_increment }
				clr_state_flag = acid_rain_legacy_affected
			}
		}
		for_loop_effect = {
			start = constant:acid_rain_region.first
			end = constant:acid_rain_runtime.region_loop_end
			value = acid_rain_migration_region_id
			if = {
				limit = {
					check_variable = { var = global.acid_rain_region_valid_entries^acid_rain_migration_region_id value = constant:acid_rain_runtime.zero compare = greater_than }
					check_variable = { var = global.acid_rain_region_touched_entries^acid_rain_migration_region_id value = global.acid_rain_region_valid_entries^acid_rain_migration_region_id compare = equals }
					NOT = { check_variable = { var = global.acid_rain_region_state_entries^acid_rain_migration_region_id value = constant:acid_rain_region_state.complete compare = equals } }
				}
				set_variable = { global.acid_rain_region_state_entries^acid_rain_migration_region_id = constant:acid_rain_region_state.complete }
				add_to_variable = { global.acid_rain_completed_region_count = constant:acid_rain_runtime.loop_increment }
			}
			if = {
				limit = {
					check_variable = { var = global.acid_rain_region_touched_entries^acid_rain_migration_region_id value = constant:acid_rain_runtime.zero compare = greater_than }
					check_variable = { var = global.acid_rain_region_touched_entries^acid_rain_migration_region_id value = global.acid_rain_region_valid_entries^acid_rain_migration_region_id compare = less_than }
				}
				set_variable = { global.acid_rain_region_state_entries^acid_rain_migration_region_id = constant:acid_rain_region_state.second_visit }
			}
		}
		if = {
			limit = {
				check_variable = { var = global.acid_rain_migration_legacy_state_count value = constant:acid_rain_runtime.zero compare = greater_than }
				check_variable = { var = global.acid_rain_migration_recovered_state_count value = constant:acid_rain_runtime.zero compare = equals }
			}
			set_global_flag = acid_rain_migration_failed_closed
		}
	}
}

# STATE SCOPE. Called only for the changed frozen state. Active decisions are
# interrupted and refunded through their normal ledger, while historical death
# and damage totals remain with the country that received those exact receipts.
acid_rain_handle_state_control_change = {
	if = {
		limit = {
			acid_rain_state_is_frozen_valid = yes
			OR = {
				has_global_flag = acid_rain_runtime_active
				has_global_flag = acid_rain_aftermath_retained
			}
			has_event_target = acid_rain_transfer_new_controller
		}
		set_temp_variable = { acid_rain_transfer_state_id = THIS.id }
		if = {
			limit = {
				has_state_flag = acid_rain_action_lock
				has_event_target = acid_rain_transfer_old_controller
			}
			event_target:acid_rain_transfer_old_controller = {
				if = {
					limit = {
						has_country_flag = acid_rain_national_action_active
						check_variable = { var = acid_rain_action_target_state value = acid_rain_transfer_state_id compare = equals }
					}
					set_country_flag = acid_rain_action_interrupted_control_change
					acid_rain_release_current_action_commitments = yes
				}
			}
		}
		set_variable = { acid_rain_responsible_country_id = event_target:acid_rain_transfer_new_controller.id }
		if = {
			limit = { has_state_flag = acid_rain_evacuated }
			set_variable = { acid_rain_evacuation_country_id = event_target:acid_rain_transfer_new_controller.id }
		}
		event_target:acid_rain_transfer_new_controller = {
			acid_rain_register_country_on_control_transfer = yes
			acid_rain_rebuild_country_responsibility = yes
		}
		if = {
			limit = { has_event_target = acid_rain_transfer_old_controller }
			event_target:acid_rain_transfer_old_controller = {
				if = {
					limit = { has_country_flag = acid_rain_country_registered }
					acid_rain_rebuild_country_responsibility = yes
				}
			}
		}
	}
}

# ANY SCOPE. Retires 20 to 40 percent of the mature footprint, keeping at
# least two live states. The first pass preserves states adjacent to untouched
# growth; the second pass is the bounded disconnected-island release path.

	set_temp_variable = { acid_rain_front_retired_state_count = constant:acid_rain_runtime.zero }
	if = {
		limit = { check_variable = { var = global.acid_rain_front_active_state_count_entries^acid_rain_front_id value = constant:acid_rain_runtime.two compare = greater_than } }

			max = constant:acid_rain_front_timing.retire_percent_roll_max_exclusive

		}
		if = {
			limit = { check_variable = { var = global.acid_rain_front_visit_entries^acid_rain_front_id value = constant:acid_rain_runtime.one compare = equals } }
			set_temp_variable = { acid_rain_front_retire_percent = constant:acid_rain_coverage.first_visit_retire_percent }

		set_temp_variable = { acid_rain_front_retire_goal = acid_rain_front_retire_goal_unrounded }

		if = { limit = { check_variable = { var = acid_rain_front_retire_goal value = acid_rain_front_retire_goal_unrounded compare = greater_than } } subtract_from_temp_variable = { acid_rain_front_retire_goal = constant:acid_rain_runtime.loop_increment } }
		set_temp_variable = { acid_rain_front_retire_maximum = global.acid_rain_front_active_state_count_entries^acid_rain_front_id }
		subtract_from_temp_variable = { acid_rain_front_retire_maximum = constant:acid_rain_runtime.two }
		clamp_temp_variable = { var = acid_rain_front_retire_goal min = constant:acid_rain_runtime.one max = acid_rain_front_retire_maximum }
		for_each_scope_loop = {
			array = global.acid_rain_frozen_states

				limit = {
					has_state_flag = acid_rain_front_target

				add_to_temp_variable = { acid_rain_front_state_retire_day = constant:acid_rain_coverage.retire_min_days }
				if = {
					limit = { check_variable = { var = global.date value = acid_rain_front_state_retire_day compare = greater_than_or_equals } }
					acid_rain_release_front_state = yes
					add_to_temp_variable = { acid_rain_front_retired_state_count = constant:acid_rain_runtime.loop_increment }
				}
			}
		}

			for_each_scope_loop = {

						has_variable = acid_rain_front_entry_day
						NOT = { has_state_flag = acid_rain_severe_cell_target }
						check_variable = { var = acid_rain_front_retired_state_count value = acid_rain_front_retire_goal compare = less_than }
					}
					set_temp_variable = { acid_rain_front_state_retire_day = acid_rain_front_entry_day }
					add_to_temp_variable = { acid_rain_front_state_retire_day = constant:acid_rain_coverage.retire_min_days }
					if = {
						limit = { check_variable = { var = global.date value = acid_rain_front_state_retire_day compare = greater_than_or_equals } }

# STATE SCOPE. Loads the exact Event 33 mortality profile for the current
# receipt. The request is rounded only after the formula, vulnerability,
# protection, pulse, and episode caps have all been applied.
acid_rain_calculate_mortality_profile = {
	set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.ordinary_sustained_rate }
	set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.ordinary_sustained_cap }
	set_temp_variable = { acid_rain_pulse_floor = constant:acid_rain_mortality.ordinary_protection_floor }
	set_temp_variable = { acid_rain_episode_cap = constant:acid_rain_mortality.ordinary_episode_cap }
	if = {
		limit = {
			check_variable = {
				var = acid_rain_pulse_class
				value = constant:acid_rain_mortality_class.ordinary
				compare = equals
			}
			check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals }
		}
		set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.ordinary_opening_rate }
		set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.ordinary_opening_cap }
	}
	else_if = {
		limit = {
			check_variable = {
				var = acid_rain_pulse_class
				value = constant:acid_rain_mortality_class.severe
				compare = equals
			}
		}
		set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.severe_sustained_rate }
		set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.severe_sustained_cap }
		set_temp_variable = { acid_rain_pulse_floor = constant:acid_rain_mortality.severe_protection_floor }
		set_temp_variable = { acid_rain_episode_cap = constant:acid_rain_mortality.severe_episode_cap }
		if = {
			limit = { check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals } }
			set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.severe_opening_rate }
			set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.severe_opening_cap }
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = acid_rain_pulse_class
				value = constant:acid_rain_mortality_class.global
				compare = equals
			}
		}
		set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.global_sustained_rate }
		set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.global_sustained_cap }
		set_temp_variable = { acid_rain_pulse_floor = constant:acid_rain_mortality.ordinary_protection_floor }
		set_temp_variable = { acid_rain_episode_cap = constant:acid_rain_mortality.global_episode_cap }
		if = {
			limit = { check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals } }
			set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.global_opening_rate }
			set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.global_opening_cap }
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = acid_rain_pulse_class
				value = constant:acid_rain_mortality_class.superstorm
				compare = equals
			}
		}
		set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.superstorm_sustained_rate }
		set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.superstorm_sustained_cap }
		set_temp_variable = { acid_rain_pulse_floor = constant:acid_rain_mortality.superstorm_protection_floor }
		set_temp_variable = { acid_rain_episode_cap = constant:acid_rain_mortality.global_episode_cap }
		if = {
			limit = { check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals } }
			set_temp_variable = { acid_rain_pulse_rate = constant:acid_rain_mortality.superstorm_opening_rate }
			set_temp_variable = { acid_rain_pulse_cap = constant:acid_rain_mortality.superstorm_opening_cap }
		}
	}

	set_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.base_vulnerability }
	if = {
		limit = { natural_disaster_state_has_low_infrastructure = yes }
		add_to_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.low_infrastructure_vulnerability }
	}
	if = {
		limit = { rail_way < constant:acid_rain_runtime.two }
		add_to_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.damaged_transport_vulnerability }
	}
	if = {
		limit = { natural_disaster_state_has_war_logistics = yes }
		add_to_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.supply_isolation_vulnerability }
	}
	if = {
		limit = { natural_disaster_state_has_dense_population = yes }
		add_to_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.dense_population_vulnerability }
	}
	if = {
		limit = { NOT = { is_controlled_by = OWNER } }
		add_to_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.occupation_vulnerability }
	}
	if = {
		limit = {
			OR = {
				has_state_flag = acid_rain_severe_history
				has_state_flag = acid_rain_aftercare_unresolved
			}
		}
		add_to_temp_variable = { acid_rain_local_vulnerability = constant:acid_rain_mortality.severe_history_vulnerability }
	}
	clamp_temp_variable = {
		var = acid_rain_local_vulnerability
		min = constant:acid_rain_mortality.base_vulnerability
		max = constant:acid_rain_mortality.maximum_vulnerability
	}

	set_temp_variable = { acid_rain_wartime_exposure = constant:acid_rain_mortality.one }
	CONTROLLER = {
		if = {
			limit = { has_war = yes }
			set_temp_variable = { acid_rain_wartime_exposure = constant:acid_rain_mortality.wartime_exposure }
		}
	}
	if = {
		limit = { NOT = { is_controlled_by = OWNER } }
		set_temp_variable = { acid_rain_wartime_exposure = constant:acid_rain_mortality.occupied_exposure }
	}
	if = {
		limit = { natural_disaster_state_has_war_logistics = yes }
		set_temp_variable = { acid_rain_wartime_exposure = constant:acid_rain_mortality.combat_exposure }
	}
	clamp_temp_variable = {
		var = acid_rain_wartime_exposure
		min = constant:acid_rain_mortality.one
		max = constant:acid_rain_mortality.combat_exposure_cap
	}

	set_temp_variable = { acid_rain_protection = constant:acid_rain_mortality.one }
	set_temp_variable = { acid_rain_preparedness_for_pulse = constant:acid_rain_preparedness.minimum }
	set_temp_variable = { acid_rain_shelter_tier_for_pulse = constant:acid_rain_runtime.zero }
	set_temp_variable = { acid_rain_water_tier_for_pulse = constant:acid_rain_runtime.zero }
	set_temp_variable = { acid_rain_medical_tier_for_pulse = constant:acid_rain_runtime.zero }
	set_temp_variable = { acid_rain_transport_tier_for_pulse = constant:acid_rain_runtime.zero }
	CONTROLLER = {
		if = {
			limit = { has_variable = acid_rain_preparedness }
			set_temp_variable = { acid_rain_preparedness_for_pulse = acid_rain_preparedness }
		}
		if = {
			limit = { has_variable = acid_rain_shelter_tier }
			set_temp_variable = { acid_rain_shelter_tier_for_pulse = acid_rain_shelter_tier }
		}
		if = {
			limit = { has_variable = acid_rain_water_tier }
			set_temp_variable = { acid_rain_water_tier_for_pulse = acid_rain_water_tier }
		}
		if = {
			limit = { has_variable = acid_rain_medical_tier }
			set_temp_variable = { acid_rain_medical_tier_for_pulse = acid_rain_medical_tier }
		}
		if = {
			limit = { has_variable = acid_rain_transport_tier }
			set_temp_variable = { acid_rain_transport_tier_for_pulse = acid_rain_transport_tier }
		}
	}
	multiply_temp_variable = {
		acid_rain_protection = constant:acid_rain_mortality.protection_coefficient
	}
	multiply_temp_variable = { acid_rain_protection = acid_rain_preparedness_for_pulse }
	set_temp_variable = { acid_rain_protection_reduction = acid_rain_protection }
	set_temp_variable = { acid_rain_protection = constant:acid_rain_mortality.one }
	subtract_from_temp_variable = { acid_rain_protection = acid_rain_protection_reduction }

	if = {
		limit = { check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals } }
		set_temp_variable = { acid_rain_component_reduction = acid_rain_shelter_tier_for_pulse }
		multiply_temp_variable = { acid_rain_component_reduction = constant:acid_rain_mortality.shelter_opening_tier_reduction }
		subtract_from_temp_variable = { acid_rain_protection = acid_rain_component_reduction }
	}
	else = {
		set_temp_variable = { acid_rain_component_reduction = acid_rain_medical_tier_for_pulse }
		multiply_temp_variable = { acid_rain_component_reduction = constant:acid_rain_mortality.medical_sustained_tier_reduction }
		subtract_from_temp_variable = { acid_rain_protection = acid_rain_component_reduction }
		if = {
			limit = { check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.ordinary compare = equals } }
			set_temp_variable = { acid_rain_component_reduction = acid_rain_water_tier_for_pulse }
			multiply_temp_variable = { acid_rain_component_reduction = constant:acid_rain_mortality.water_sustained_tier_reduction }
			subtract_from_temp_variable = { acid_rain_protection = acid_rain_component_reduction }
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.severe compare = equals }
				check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.superstorm compare = equals }
			}
		}
		set_temp_variable = { acid_rain_component_reduction = acid_rain_transport_tier_for_pulse }
		multiply_temp_variable = { acid_rain_component_reduction = constant:acid_rain_mortality.transport_response_tier_reduction }
		subtract_from_temp_variable = { acid_rain_protection = acid_rain_component_reduction }
	}
	if = {
		limit = { has_state_flag = acid_rain_evacuated }
		subtract_from_temp_variable = { acid_rain_protection = constant:acid_rain_mortality.evacuation_protection }
			}
			CONTROLLER = {
		if = { limit = { has_country_flag = acid_rain_action_shelter_active } subtract_from_temp_variable = { acid_rain_protection = constant:acid_rain_response.shelter_protection } }
		if = { limit = { has_country_flag = acid_rain_action_water_active } subtract_from_temp_variable = { acid_rain_protection = constant:acid_rain_response.water_protection } }
		if = { limit = { has_country_flag = acid_rain_action_medical_active } subtract_from_temp_variable = { acid_rain_protection = constant:acid_rain_response.medical_protection } }
	}
	if = {
		limit = { has_variable = cbrn_civilian_mask_effective_coverage }
		set_temp_variable = { acid_rain_cbrn_reduction = cbrn_civilian_mask_effective_coverage }
		multiply_temp_variable = { acid_rain_cbrn_reduction = constant:acid_rain_mortality.cbrn_protection_coefficient }
		subtract_from_temp_variable = { acid_rain_protection = acid_rain_cbrn_reduction }
	}
	clamp_temp_variable = {
		var = acid_rain_protection
		min = acid_rain_pulse_floor
		max = constant:acid_rain_mortality.one
	}

	set_temp_variable = { acid_rain_pulse_requested_people = state_population_k }
	multiply_temp_variable = { acid_rain_pulse_requested_people = acid_rain_pulse_rate }
	multiply_temp_variable = { acid_rain_pulse_requested_people = acid_rain_pulse_intensity }
	multiply_temp_variable = { acid_rain_pulse_requested_people = acid_rain_local_vulnerability }
	multiply_temp_variable = { acid_rain_pulse_requested_people = acid_rain_wartime_exposure }
	multiply_temp_variable = { acid_rain_pulse_requested_people = acid_rain_protection }
	divide_temp_variable = { acid_rain_pulse_requested_people = constant:acid_rain_mortality.thousand_per_million }

	set_temp_variable = { acid_rain_pulse_state_cap = state_population_k }
	multiply_temp_variable = { acid_rain_pulse_state_cap = acid_rain_pulse_cap }
	clamp_temp_variable = {
		var = acid_rain_pulse_requested_people
		min = constant:acid_rain_runtime.zero
		max = acid_rain_pulse_state_cap
	}

	if = {
		limit = { has_variable = acid_rain_state_deaths }
		set_temp_variable = { acid_rain_episode_remaining = acid_rain_state_population_start_k }
		multiply_temp_variable = { acid_rain_episode_remaining = constant:chaos_meter_deaths.people_per_k }
		multiply_temp_variable = { acid_rain_episode_remaining = acid_rain_episode_cap }
		subtract_from_temp_variable = { acid_rain_episode_remaining = acid_rain_state_deaths }
		clamp_temp_variable = { var = acid_rain_episode_remaining min = constant:acid_rain_runtime.zero }
		clamp_temp_variable = {
			var = acid_rain_pulse_requested_people
			min = constant:acid_rain_runtime.zero
			max = acid_rain_episode_remaining
		}
	}
	# Counterfactual estimate for reports only. It follows the same current
	# population, intensity, vulnerability, pulse cap, and episode cap, but
	# deliberately omits protection and never enters the population gateway.
	set_temp_variable = { acid_rain_unprotected_requested_people = state_population_k }
	multiply_temp_variable = { acid_rain_unprotected_requested_people = acid_rain_pulse_rate }
	multiply_temp_variable = { acid_rain_unprotected_requested_people = acid_rain_pulse_intensity }
	multiply_temp_variable = { acid_rain_unprotected_requested_people = acid_rain_local_vulnerability }
	multiply_temp_variable = { acid_rain_unprotected_requested_people = acid_rain_wartime_exposure }
	divide_temp_variable = { acid_rain_unprotected_requested_people = constant:acid_rain_mortality.thousand_per_million }
	clamp_temp_variable = {
		var = acid_rain_unprotected_requested_people
		min = constant:acid_rain_runtime.zero
		max = acid_rain_pulse_state_cap
	}
	if = {
		limit = { has_variable = acid_rain_state_deaths }
		set_temp_variable = { acid_rain_unprotected_episode_remaining = acid_rain_state_population_start_k }
		multiply_temp_variable = { acid_rain_unprotected_episode_remaining = constant:chaos_meter_deaths.people_per_k }
		multiply_temp_variable = { acid_rain_unprotected_episode_remaining = acid_rain_episode_cap }
		subtract_from_temp_variable = { acid_rain_unprotected_episode_remaining = acid_rain_state_deaths }
		clamp_temp_variable = { var = acid_rain_unprotected_episode_remaining min = constant:acid_rain_runtime.zero }
		clamp_temp_variable = {
			var = acid_rain_unprotected_requested_people
			min = constant:acid_rain_runtime.zero
			max = acid_rain_unprotected_episode_remaining
		}
	}
	round_temp_variable = acid_rain_unprotected_requested_people
	round_temp_variable = acid_rain_pulse_requested_people
	set_temp_variable = { acid_rain_pulse_applied_request = acid_rain_pulse_requested_people }
}

# STATE SCOPE. This is the only Event 33 mortality gateway. A pulse key is
# written before the transaction, so a retry after save/load or a duplicate
# worker invocation cannot apply a second loss.
acid_rain_apply_state_pulse = {
	if = {
		limit = {
			acid_rain_state_is_frozen_valid = yes
			acid_rain_pulse_key_is_new = yes
		}
		add_to_variable = { global.acid_rain_pulse_receipt_count = constant:acid_rain_runtime.loop_increment }
		set_variable = { acid_rain_last_pulse_key = acid_rain_pulse_key }
		if = {
			limit = {
				check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals }
				NOT = { has_state_flag = acid_rain_opening_pulse_recorded }
			}
			set_state_flag = acid_rain_opening_pulse_recorded
			set_variable = { acid_rain_first_opening_shock_day = global.date }
		}
		set_temp_variable = { acid_rain_cbrn_snapshot_available = constant:acid_rain_runtime.one }
		cbrn_refresh_state_civilian_mask_snapshot = yes
		acid_rain_calculate_mortality_profile = yes
		set_variable = { acid_rain_last_population_before_people = state_population_k }
		multiply_variable = { acid_rain_last_population_before_people = constant:chaos_meter_deaths.people_per_k }
		set_temp_variable = { state_civilian_population_loss_requested = acid_rain_pulse_applied_request }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:acid_rain_mortality.minimum_remaining_people }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.acid_rain }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:chaos_meter_deaths.one }
		set_temp_variable = { state_civilian_population_loss_target_country = CONTROLLER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:chaos_meter_deaths.one }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.one }
		apply_exact_state_civilian_population_loss = yes
		set_variable = { acid_rain_state_last_applied_loss = state_civilian_population_loss_applied }
		set_variable = { acid_rain_last_population_after_people = state_population_k }
		multiply_variable = { acid_rain_last_population_after_people = constant:chaos_meter_deaths.people_per_k }
		set_variable = { acid_rain_state_observed_loss = acid_rain_last_population_before_people }
		subtract_from_variable = { acid_rain_state_observed_loss = acid_rain_last_population_after_people }
		clamp_variable = {
			var = acid_rain_state_observed_loss
			min = constant:chaos_meter_deaths.zero
			max = acid_rain_last_population_before_people
		}
		if = {
			limit = {
				check_variable = {
					var = acid_rain_state_observed_loss
					value = state_civilian_population_loss_applied
					compare = not_equals
				}
			}
			set_state_flag = acid_rain_population_contract_fault
		}
		if = {
			limit = { check_variable = { var = acid_rain_pulse_opening value = constant:acid_rain_runtime.one compare = equals } }
			CONTROLLER = {
				acid_rain_register_country_on_control_transfer = yes
				if = {
					limit = { NOT = { has_variable = acid_rain_first_opening_shock_day } }
					set_variable = { acid_rain_first_opening_shock_day = global.date }
					set_variable = { acid_rain_first_opening_shock_state_id = PREV.id }
				}
			}
		}
		if = {
			limit = {
				check_variable = {
					var = state_civilian_population_loss_applied
					value = constant:chaos_meter_deaths.zero
					compare = greater_than
				}
			}
			add_to_variable = { acid_rain_state_deaths = state_civilian_population_loss_applied }
			add_to_variable = { global.acid_rain_total_deaths = state_civilian_population_loss_applied }
			add_to_variable = { global.acid_rain_pulse_positive_receipt_count = constant:acid_rain_runtime.loop_increment }
			if = {
				limit = {
					check_variable = {
						var = global.acid_rain_positive_receipt_registry_count
						value = constant:acid_rain_event.max_positive_receipt_records
						compare = greater_than_or_equals
					}
				}
				remove_from_array = { array = global.acid_rain_pulse_receipt_states index = constant:acid_rain_runtime.first_array_index }
				remove_from_array = { array = global.acid_rain_pulse_receipt_applied_entries index = constant:acid_rain_runtime.first_array_index }
				remove_from_array = { array = global.acid_rain_pulse_receipt_front_entries index = constant:acid_rain_runtime.first_array_index }
				remove_from_array = { array = global.acid_rain_pulse_receipt_generation_entries index = constant:acid_rain_runtime.first_array_index }
			}
			else = {
				add_to_variable = { global.acid_rain_positive_receipt_registry_count = constant:acid_rain_runtime.loop_increment }
			}
			add_to_array = { array = global.acid_rain_pulse_receipt_states value = THIS }
			add_to_array = { array = global.acid_rain_pulse_receipt_applied_entries value = state_civilian_population_loss_applied }
			add_to_array = { array = global.acid_rain_pulse_receipt_front_entries value = acid_rain_pulse_class }
			add_to_array = { array = global.acid_rain_pulse_receipt_generation_entries value = global.acid_rain_generation_id }
			set_state_flag = acid_rain_positive_loss_receipt
			set_temp_variable = { acid_rain_prevented_loss = acid_rain_unprotected_requested_people }
			subtract_from_temp_variable = { acid_rain_prevented_loss = state_civilian_population_loss_applied }
			clamp_temp_variable = { var = acid_rain_prevented_loss min = constant:chaos_meter_deaths.zero }
			add_to_variable = { global.acid_rain_total_prevented_deaths = acid_rain_prevented_loss }
			if = {
				limit = {
					has_variable = acid_rain_severe_cell_id
					check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.severe compare = equals }
				}
				add_to_variable = { global.acid_rain_severe_cell_death_entries^acid_rain_severe_cell_id = state_civilian_population_loss_applied }
			}
			CONTROLLER = {
				if = {
					limit = { NOT = { has_variable = acid_rain_event_deaths } }
					set_variable = { acid_rain_event_deaths = constant:chaos_meter_deaths.zero }
				}
				add_to_variable = { acid_rain_event_deaths = state_civilian_population_loss_applied }
				add_to_variable = { acid_rain_prevented_deaths = acid_rain_prevented_loss }
				add_to_variable = { acid_rain_unprotected_estimate = acid_rain_unprotected_requested_people }
				acid_rain_update_casualty_report = yes
				acid_rain_try_preparation_report = yes
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.severe compare = equals }
					check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.superstorm compare = equals }
				}
				check_variable = {
					var = state_civilian_population_loss_applied
					value = constant:chaos_meter_deaths.zero
					compare = greater_than
				}
			}
			save_event_target_as = acid_rain_report_state
			CONTROLLER = {
				acid_rain_register_country_on_control_transfer = yes
				add_to_variable = { acid_rain_severe_pulse_count = constant:acid_rain_runtime.loop_increment }
				if = {
					limit = { NOT = { has_country_flag = acid_rain_severe_impact_reported } }
					set_country_flag = acid_rain_severe_impact_reported
					country_event = { id = chaosx.nr33.104 }
				}
			}
		}
	}
}

# STATE SCOPE. Building damage is a separate physical receipt and never a
# substitute for the exact population contract. Pressure is persistent on the
# frozen state, while each pulse spends at most its selected upper-band cap.
# Every spent point targets one real building level through a bounded cursor.
acid_rain_apply_state_damage = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = acid_rain_last_damage_pulse_key }
				check_variable = { var = acid_rain_last_damage_pulse_key value = acid_rain_pulse_key compare = not_equals }
			}
		}

# ANY SCOPE. Central Event 33 Air gateway. The request is clamped before the
# shared Air ledger is called, then the actual applied delta returned by that
# ledger is the only value counted in Event 33 totals.

	set_temp_variable = { acid_rain_air_applied_delta_bp = constant:acid_rain_runtime.zero }
	if = {
		limit = {
			acid_rain_air_can_receive_request = yes
			has_variable = acid_rain_air_requested_bp
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = acid_rain_air_request_is_formation
						value = constant:acid_rain_runtime.one
						compare = equals
					}
				}
			}
			clamp_temp_variable = {
				var = acid_rain_air_requested_bp
				min = constant:acid_rain_air.weekly_min_bp
				max = constant:acid_rain_air.weekly_max_bp
			}
		}
		set_temp_variable = { acid_rain_air_remaining_lifetime_bp = constant:acid_rain_air.lifetime_allowance_bp }
		subtract_from_temp_variable = {
			acid_rain_air_remaining_lifetime_bp = global.acid_rain_air_lifetime_added_bp
		}
		set_temp_variable = { acid_rain_air_remaining_ceiling_bp = constant:acid_rain_air.global_ceiling_bp }
		subtract_from_temp_variable = { acid_rain_air_remaining_ceiling_bp = global.air_contamination_bp }
		clamp_temp_variable = { var = acid_rain_air_remaining_lifetime_bp min = constant:acid_rain_runtime.zero }
		clamp_temp_variable = { var = acid_rain_air_remaining_ceiling_bp min = constant:acid_rain_runtime.zero }
		clamp_temp_variable = {
			var = acid_rain_air_requested_bp
			min = constant:acid_rain_runtime.zero
			max = acid_rain_air_remaining_lifetime_bp
		}
		clamp_temp_variable = {
			var = acid_rain_air_requested_bp
			min = constant:acid_rain_runtime.zero
			max = acid_rain_air_remaining_ceiling_bp
		}
		set_temp_variable = { acid_rain_air_applied_delta_bp = acid_rain_air_requested_bp }
		if = {
			limit = { check_variable = { var = acid_rain_air_applied_delta_bp value = constant:acid_rain_runtime.zero compare = greater_than } }
			if = {
				limit = { NOT = { has_variable = global.air_contamination_bp } }

# STATE SCOPE. Appends the current state once per integer target-weight point.
# The bounded repeated-scope pool gives `random_scope_in_array` an inspectable
# roulette distribution while keeping active-front geometry as the hard gate.
acid_rain_append_state_to_severe_weighted_pool = {
	set_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.base }
	if = {
		limit = { check_variable = { var = acid_rain_state_population_start_k value = constant:acid_rain_severe_target_weight.population_medium_k compare = greater_than_or_equals } }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.population_medium_bonus }

		limit = { check_variable = { var = acid_rain_state_population_start_k value = constant:acid_rain_severe_target_weight.population_large_k compare = greater_than_or_equals } }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.population_large_bonus }

		limit = { check_variable = { var = acid_rain_state_population_start_k value = constant:acid_rain_severe_target_weight.population_dense_k compare = greater_than_or_equals } }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.population_dense_bonus }

		limit = { natural_disaster_state_has_war_logistics = yes }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.logistics_bonus }
	}

		limit = { has_state_flag = acid_rain_opening_pulse_recorded }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.prior_exposure_bonus }
	}

		limit = { CONTROLLER = { check_variable = { var = acid_rain_preparedness value = constant:acid_rain_severe_target_weight.low_preparedness_threshold compare = less_than } } }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.low_preparedness_bonus }
	}
	else_if = {
		limit = { CONTROLLER = { check_variable = { var = acid_rain_preparedness value = constant:acid_rain_severe_target_weight.moderate_preparedness_threshold compare = less_than } } }
		add_to_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_severe_target_weight.moderate_preparedness_bonus }

		limit = { has_state_flag = acid_rain_severe_history }

		var = acid_rain_severe_state_weight

		max = constant:acid_rain_severe_target_weight.maximum

		limit = { check_variable = { acid_rain_severe_state_weight > constant:acid_rain_runtime.zero } }
		add_to_array = { array = global.acid_rain_severe_planner_weighted_candidates value = THIS }
		subtract_from_temp_variable = { acid_rain_severe_state_weight = constant:acid_rain_runtime.loop_increment }
	}
}

# ANY SCOPE. Selects and activates one bounded severe cell. Every pick uses a
# weighted active-footprint pool; adjacency is required while a connected
# candidate exists, then a second bounded pass admits disconnected islands.
acid_rain_create_severe_cell = {
	set_temp_variable = { acid_rain_active_front_state_count = constant:acid_rain_runtime.zero }
	for_each_scope_loop = {
		array = global.acid_rain_frozen_states
		if = {
			limit = {
				acid_rain_state_is_frozen_valid = yes
				check_variable = { var = acid_rain_target_front_id value = acid_rain_front_id compare = equals }
			}
			add_to_temp_variable = { acid_rain_active_front_state_count = constant:acid_rain_runtime.loop_increment }
		}
	}
	if = {
		limit = {
			check_variable = { var = acid_rain_active_front_state_count value = constant:acid_rain_coverage.first_visit_min_states compare = greater_than_or_equals }
			check_variable = { var = global.acid_rain_severe_cell_count value = constant:acid_rain_event.max_severe_cells compare = less_than }
		}
		set_variable_to_random = { var = acid_rain_severe_target_goal min = constant:acid_rain_evolution.severe_min_target_states max = constant:acid_rain_evolution.severe_max_target_states_exclusive integer = yes }
		clear_array = global.acid_rain_severe_planner_targets
		clear_array = global.acid_rain_severe_planner_frontier
		clear_array = global.acid_rain_severe_planner_next_frontier
		clear_array = global.acid_rain_severe_planner_weighted_candidates
		set_temp_variable = { acid_rain_severe_target_attempt = constant:acid_rain_runtime.zero }
		while_loop_effect = {
			limit = {
				check_variable = { var = global.acid_rain_severe_planner_targets^num value = acid_rain_severe_target_goal compare = less_than }
				check_variable = { var = acid_rain_severe_target_attempt value = constant:acid_rain_runtime.max_target_attempts compare = less_than }
			}
			clear_array = global.acid_rain_severe_planner_weighted_candidates
			for_each_scope_loop = {
				array = global.acid_rain_frozen_states
				if = {
					limit = {
						acid_rain_state_is_frozen_valid = yes
						check_variable = { var = acid_rain_target_front_id value = acid_rain_front_id compare = equals }
						NOT = { has_state_flag = acid_rain_severe_cell_target }
						NOT = { is_in_array = { array = global.acid_rain_severe_planner_targets value = THIS } }
						OR = {
							check_variable = { var = global.acid_rain_severe_planner_targets^num value = constant:acid_rain_runtime.zero compare = equals }
							any_neighbor_state = { is_in_array = { array = global.acid_rain_severe_planner_targets value = THIS } }
						}
					}
					acid_rain_append_state_to_severe_weighted_pool = yes
				}
			}
			if = {
				limit = { check_variable = { var = global.acid_rain_severe_planner_weighted_candidates^num value = constant:acid_rain_runtime.zero compare = equals } }
				# No connected candidate remains. Rebuild the same weighted pool
				# without adjacency so a disconnected island can be selected.
				for_each_scope_loop = {
					array = global.acid_rain_frozen_states
					if = {
						limit = {
							acid_rain_state_is_frozen_valid = yes
							check_variable = { var = acid_rain_target_front_id value = acid_rain_front_id compare = equals }
							NOT = { has_state_flag = acid_rain_severe_cell_target }
							NOT = { is_in_array = { array = global.acid_rain_severe_planner_targets value = THIS } }
						}
						acid_rain_append_state_to_severe_weighted_pool = yes
					}
				}
			}
			if = {
				limit = { check_variable = { var = global.acid_rain_severe_planner_weighted_candidates^num value = constant:acid_rain_runtime.zero compare = greater_than } }
				random_scope_in_array = {
					array = global.acid_rain_severe_planner_weighted_candidates
					limit = {
						acid_rain_state_is_frozen_valid = yes
						NOT = { is_in_array = { array = global.acid_rain_severe_planner_targets value = THIS } }
					}
					add_to_array = { array = global.acid_rain_severe_planner_targets value = THIS }
				}
			}
			else = {
				# Force the bounded selector to its terminal state when the
				# active footprint has no remaining eligible state.
				set_temp_variable = { acid_rain_severe_target_attempt = constant:acid_rain_runtime.max_target_attempts }
			}
			add_to_temp_variable = { acid_rain_severe_target_attempt = constant:acid_rain_runtime.loop_increment }
		}
		if = {
			limit = { check_variable = { var = global.acid_rain_severe_planner_targets^num value = acid_rain_severe_target_goal compare = less_than } }
			clear_array = global.acid_rain_severe_planner_weighted_candidates
			for_each_scope_loop = {
				array = global.acid_rain_frozen_states
				if = {
					limit = {
						acid_rain_state_is_frozen_valid = yes
						check_variable = { var = acid_rain_target_front_id value = acid_rain_front_id compare = equals }
						NOT = { has_state_flag = acid_rain_severe_cell_target }
						NOT = { is_in_array = { array = global.acid_rain_severe_planner_targets value = THIS } }
					}
					acid_rain_append_state_to_severe_weighted_pool = yes
				}
			}
			if = {
				limit = { check_variable = { var = global.acid_rain_severe_planner_weighted_candidates^num value = constant:acid_rain_runtime.zero compare = greater_than } }
				random_scope_in_array = {
					array = global.acid_rain_severe_planner_weighted_candidates
					limit = { NOT = { is_in_array = { array = global.acid_rain_severe_planner_targets value = THIS } } }
					add_to_array = { array = global.acid_rain_severe_planner_targets value = THIS }
				}
			}
		}
		if = {
			limit = { check_variable = { var = global.acid_rain_severe_planner_targets^num value = constant:acid_rain_evolution.severe_min_target_states compare = greater_than_or_equals } }
			add_to_variable = { global.acid_rain_severe_cell_count = constant:acid_rain_runtime.loop_increment }
			set_temp_variable = { acid_rain_new_severe_cell_id = global.acid_rain_severe_cell_count }
			add_to_array = { array = global.acid_rain_severe_cell_ids value = acid_rain_new_severe_cell_id }
			set_variable = { global.acid_rain_severe_cell_front_entries^acid_rain_new_severe_cell_id = acid_rain_front_id }
			set_variable = { global.acid_rain_severe_cell_start_entries^acid_rain_new_severe_cell_id = global.date }
			add_to_variable = { global.acid_rain_severe_cell_start_entries^acid_rain_new_severe_cell_id = constant:acid_rain_evolution.severe_forecast_lead_days }
			set_variable_to_random = { var = global.acid_rain_severe_cell_end_entries^acid_rain_new_severe_cell_id min = constant:acid_rain_evolution.severe_duration_min_days max = constant:acid_rain_evolution.severe_duration_max_days_exclusive integer = yes }
			add_to_variable = { global.acid_rain_severe_cell_end_entries^acid_rain_new_severe_cell_id = global.acid_rain_severe_cell_start_entries^acid_rain_new_severe_cell_id }
			set_variable = { global.acid_rain_severe_cell_target_count_entries^acid_rain_new_severe_cell_id = global.acid_rain_severe_planner_targets^num }
			set_variable = { global.acid_rain_severe_cell_active_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.zero }
			set_variable = { global.acid_rain_severe_cell_cooldown_day_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.zero }
			set_variable = { global.acid_rain_severe_cell_death_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.zero }
			set_variable = { global.acid_rain_severe_cell_damage_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.zero }
			set_variable = { global.acid_rain_severe_cell_low_preparedness_exposure_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.zero }
			set_variable = { global.acid_rain_severe_cell_supply_disruption_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.zero }
			for_each_scope_loop = {
				array = global.acid_rain_severe_planner_targets
				set_state_flag = acid_rain_severe_cell_target
				set_state_flag = acid_rain_severe_forecast_pending
				set_state_flag = acid_rain_severe_forecast_reported
				set_variable = { acid_rain_severe_cell_id = acid_rain_new_severe_cell_id }
				set_variable = { acid_rain_severe_forecast_generation = global.acid_rain_generation_id }
				set_variable = { acid_rain_severe_forecast_id = acid_rain_new_severe_cell_id }
				set_variable = { acid_rain_severe_forecast_day = global.date }
				set_variable = { acid_rain_severe_forecast_start_day = global.acid_rain_severe_cell_start_entries^acid_rain_new_severe_cell_id }
				set_variable = { acid_rain_severe_forecast_end_day = global.acid_rain_severe_cell_end_entries^acid_rain_new_severe_cell_id }
				set_variable = { acid_rain_last_severe_pulse_day = global.date }
				set_variable = { acid_rain_severe_cell_opening_recorded = constant:acid_rain_runtime.zero }
				save_event_target_as = acid_rain_report_state
				CONTROLLER = {
					acid_rain_register_country_on_control_transfer = yes
					set_country_flag = acid_rain_country_severe_forecast_active
					if = {
						limit = {
							OR = {
								NOT = { has_variable = acid_rain_last_severe_forecast_report_id }
								check_variable = { var = acid_rain_last_severe_forecast_report_id value = acid_rain_new_severe_cell_id compare = not_equals }
							}
						}
						set_variable = { acid_rain_last_severe_forecast_report_id = acid_rain_new_severe_cell_id }
						add_to_variable = { acid_rain_severe_forecast_count = constant:acid_rain_runtime.loop_increment }
						country_event = { id = chaosx.nr33.103 }
					}
				}
				if = {
					limit = { CONTROLLER = { check_variable = { var = acid_rain_preparedness value = constant:acid_rain_evolution.severe_low_preparedness_threshold compare = less_than } } }
					set_temp_variable = { acid_rain_low_preparedness_exposed_people = acid_rain_state_population_start_k }
					multiply_temp_variable = { acid_rain_low_preparedness_exposed_people = constant:chaos_meter_deaths.people_per_k }
					add_to_variable = { global.acid_rain_severe_cell_low_preparedness_exposure_entries^acid_rain_new_severe_cell_id = acid_rain_low_preparedness_exposed_people }
				}
				if = {
					limit = { natural_disaster_state_has_war_logistics = yes }
					set_variable = { global.acid_rain_severe_cell_supply_disruption_entries^acid_rain_new_severe_cell_id = constant:acid_rain_runtime.one }
				}
				add_dynamic_modifier = { modifier = acid_rain_state_severe days = global.acid_rain_modifier_refresh_days }
			}
		}
	}

# STATE SCOPE. Builds the bounded weighted target pool for a global
# superstorm. A state in a region without another current superstorm receives
# the largest bonus; population, logistics, and history refine the state pick.
acid_rain_append_state_to_superstorm_weighted_pool = {
	set_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_superstorm_target_weight.base }
	if = {
		limit = { check_variable = { var = global.acid_rain_superstorm_region_active_entries^acid_rain_region_id value = constant:acid_rain_runtime.zero compare = equals } }
		add_to_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_superstorm_target_weight.region_without_active_storm_bonus }
	}
	if = {
		limit = { check_variable = { var = acid_rain_state_population_start_k value = constant:acid_rain_superstorm_target_weight.population_medium_k compare = greater_than_or_equals } }
		add_to_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_superstorm_target_weight.population_medium_bonus }
	}
	if = {
		limit = { check_variable = { var = acid_rain_state_population_start_k value = constant:acid_rain_superstorm_target_weight.population_large_k compare = greater_than_or_equals } }
		add_to_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_superstorm_target_weight.population_large_bonus }
	}
	if = {
		limit = { natural_disaster_state_has_war_logistics = yes }
		add_to_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_superstorm_target_weight.logistics_bonus }
	}
	if = {
		limit = { has_state_flag = acid_rain_superstorm_history }
		subtract_from_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_superstorm_target_weight.previous_superstorm_penalty }
	}
	clamp_temp_variable = {
		var = acid_rain_superstorm_state_weight
		min = constant:acid_rain_superstorm_target_weight.minimum
		max = constant:acid_rain_superstorm_target_weight.maximum
	}
	while_loop_effect = {
		limit = { check_variable = { acid_rain_superstorm_state_weight > constant:acid_rain_runtime.zero } }
		add_to_array = { array = global.acid_rain_superstorm_planner_weighted_candidates value = THIS }
		subtract_from_temp_variable = { acid_rain_superstorm_state_weight = constant:acid_rain_runtime.loop_increment }
	}
}

				}
				set_temp_variable = { acid_rain_dissipation_superstorm_penalty = global.acid_rain_superstorm_count }
				multiply_temp_variable = { acid_rain_dissipation_superstorm_penalty = constant:acid_rain_dissipation.global_superstorm_penalty_each }
				add_to_temp_variable = { acid_rain_dissipation_pressure_penalty = acid_rain_dissipation_superstorm_penalty }
				clamp_temp_variable = { var = acid_rain_dissipation_pressure_penalty min = constant:acid_rain_runtime.zero max = constant:acid_rain_dissipation.global_pressure_penalty_cap }
				subtract_from_temp_variable = { acid_rain_dissipation_chance = acid_rain_dissipation_pressure_penalty }
				clamp_temp_variable = { var = acid_rain_dissipation_chance min = constant:acid_rain_dissipation.minimum_nonguaranteed_chance max = constant:acid_rain_dissipation.global_chance_eight }
			}
		}
		else = {
			if = { limit = { check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_dissipation.step_one compare = equals } } set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_one } }
			else_if = { limit = { check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_dissipation.step_two compare = equals } } set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_two } }
			else_if = { limit = { check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_dissipation.step_three compare = equals } } set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_three } }
			else_if = { limit = { check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_dissipation.step_four compare = equals } } set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_four } }
			else_if = { limit = { check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_dissipation.step_five compare = equals } } set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_five } }
			else_if = { limit = { check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_dissipation.step_six compare = equals } } set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_six } }
			else = {
				set_temp_variable = { acid_rain_dissipation_chance = constant:acid_rain_dissipation.ordinary_chance_seven }
				set_temp_variable = { acid_rain_dissipation_guaranteed = constant:acid_rain_runtime.one }
			}
		}
		set_variable_to_random = { var = acid_rain_dissipation_roll min = constant:acid_rain_runtime.one max = constant:acid_rain_event.probability_roll_max_exclusive integer = yes }
		set_variable = { global.acid_rain_last_dissipation_chance = acid_rain_dissipation_chance }
		set_variable = { global.acid_rain_last_dissipation_roll = acid_rain_dissipation_roll }
		if = {
			limit = {
				OR = {
					check_variable = { var = acid_rain_dissipation_guaranteed value = constant:acid_rain_runtime.one compare = equals }
					check_variable = { var = acid_rain_dissipation_roll value = acid_rain_dissipation_chance compare = less_than_or_equals }
				}
			}
			acid_rain_begin_dissipation = yes
		}
		else = {
			set_variable = { global.acid_rain_next_dissipation_check_day = global.date }
			if = {
				limit = { has_global_flag = acid_rain_global_layer_active }
				add_to_variable = { global.acid_rain_next_dissipation_check_day = constant:acid_rain_dissipation.check_interval_days }
			}
			else = {
				set_variable_to_random = { var = acid_rain_dissipation_retry_days min = constant:acid_rain_dissipation.failed_check_delay_min_days max = constant:acid_rain_dissipation.failed_check_delay_max_days_exclusive integer = yes }
				add_to_variable = { global.acid_rain_next_dissipation_check_day = acid_rain_dissipation_retry_days }
			}
		}
	}
}

# ANY SCOPE. A successful chance check enters one idempotent cleanup
# transaction and closes every creation path before cleanup begins.
acid_rain_begin_dissipation = {
	if = {
		limit = { NOT = { has_global_flag = acid_rain_dissipation_pending } }
		set_global_flag = acid_rain_dissipation_pending
		set_variable = { global.acid_rain_phase = constant:acid_rain_phase.dissipation }
		set_variable = { global.acid_rain_dissipation_step = constant:acid_rain_runtime.zero }
		set_variable = { global.acid_rain_next_dissipation_check_day = global.date }
	}
}

# ANY SCOPE. The chance ladder has already succeeded. This stored transaction
# step is safe to repeat after reload and leaves Air contamination untouched.
acid_rain_apply_dissipation_step = {
	if = {
		limit = { check_variable = { var = global.acid_rain_dissipation_step value = constant:acid_rain_runtime.zero compare = equals } }
		set_variable = { global.acid_rain_dissipation_step = constant:acid_rain_dissipation.step_one }
		set_variable = { global.acid_rain_dissipation_progress = constant:acid_rain_dissipation.global_coverage_ladder_100 }
		acid_rain_finish_runtime = yes
	}
}

# ANY SCOPE. One final bounded cleanup is idempotent. Durable coverage,
# generation, positive-loss receipts, Event 33 totals, Air additions, and
# aftermath flags remain available to reports and achievements.
acid_rain_finish_runtime = {
	if = {
		limit = { NOT = { has_global_flag = acid_rain_runtime_resolved } }
		for_each_scope_loop = {
			array = global.acid_rain_frozen_states
			remove_dynamic_modifier = { modifier = acid_rain_state_ordinary }
			remove_dynamic_modifier = { modifier = acid_rain_state_strong }
			remove_dynamic_modifier = { modifier = acid_rain_state_severe }
			remove_dynamic_modifier = { modifier = acid_rain_state_global }
			remove_dynamic_modifier = { modifier = acid_rain_state_superstorm }
			remove_dynamic_modifier = { modifier = acid_rain_aftermath_1 }
			remove_dynamic_modifier = { modifier = acid_rain_aftermath_2 }
			remove_dynamic_modifier = { modifier = acid_rain_aftermath_3 }
			if = {
				limit = { has_state_flag = acid_rain_touched }
				set_variable = { acid_rain_state_aftermath_tier = constant:acid_rain_runtime.one }
				if = {
					limit = { has_state_flag = acid_rain_severe_history }
					set_variable = { acid_rain_state_aftermath_tier = constant:acid_rain_response.recovery_tier_maximum }
				}
				else_if = {
					limit = { has_state_flag = acid_rain_superstorm_history }
					set_variable = { acid_rain_state_aftermath_tier = constant:acid_rain_response.recovery_tier_maximum }
				}
				else_if = {
					limit = { has_state_flag = acid_rain_global_pulse_receipt }
					set_variable = { acid_rain_state_aftermath_tier = constant:acid_rain_runtime.two }
				}
				set_variable = { acid_rain_aftercare_deadline = global.date }
				add_to_variable = { acid_rain_aftercare_deadline = constant:acid_rain_response.aftercare_recovery_window_days }
				set_state_flag = acid_rain_aftercare_unresolved
				if = {
					limit = { check_variable = { var = acid_rain_state_aftermath_tier value = constant:acid_rain_runtime.one compare = equals } }
					add_dynamic_modifier = { modifier = acid_rain_aftermath_1 days = constant:acid_rain_response.aftercare_recovery_window_days }
				}
				else_if = {
					limit = { check_variable = { var = acid_rain_state_aftermath_tier value = constant:acid_rain_runtime.two compare = equals } }
					add_dynamic_modifier = { modifier = acid_rain_aftermath_2 days = constant:acid_rain_response.aftercare_recovery_window_days }
				}
				else = {
					add_dynamic_modifier = { modifier = acid_rain_aftermath_3 days = constant:acid_rain_response.aftercare_recovery_window_days }
				}
			}
			else = {
				set_variable = { acid_rain_state_aftermath_tier = constant:acid_rain_runtime.zero }
				clr_state_flag = acid_rain_aftercare_unresolved
				clear_variable = acid_rain_aftercare_deadline
			}
			clr_state_flag = acid_rain_front_target
			clr_state_flag = acid_rain_severe_cell_target
			clr_state_flag = acid_rain_superstorm_target
			clr_state_flag = acid_rain_severe_forecast_pending
			clr_state_flag = acid_rain_superstorm_forecast_pending
			acid_rain_close_state_evacuation = yes
			clr_state_flag = acid_rain_action_lock
		}
		for_each_scope_loop = {
			array = global.acid_rain_participant_countries
			country_event = { id = chaosx.nr33.109 }
			acid_rain_enter_country_recovery_tail = yes
		}
		if = {
			limit = { NOT = { has_global_flag = acid_rain_final_dissipation_receipt } }
			set_global_flag = acid_rain_final_dissipation_receipt
			if = {
				limit = { has_global_flag = acid_rain_global_layer_receipt }
				set_temp_variable = { chaos_change = constant:acid_rain_evolution.global_final_chaos_loss }
			}
			else = { set_temp_variable = { chaos_change = constant:acid_rain_evolution.ordinary_final_chaos_loss } }
			add_chaos_meter_value = yes
		}
		if = { limit = { has_event_target = acid_rain_runtime_host } event_target:acid_rain_runtime_host = { news_event = { id = chaosx.nr33.205 } } }
		set_global_flag = acid_rain_runtime_resolved
		set_global_flag = acid_rain_endpoint_recorded
		set_global_flag = acid_rain_aftermath_retained
		clr_global_flag = acid_rain_runtime_active
		clr_global_flag = acid_rain_dissipation_pending
		clr_global_flag = acid_rain_global_transition_active
		clr_global_flag = acid_rain
		set_variable = { global.acid_rain_gui_window_visible = constant:acid_rain_runtime.zero }
		if = {
			limit = { has_event_target = acid_rain_runtime_host }
			event_target:acid_rain_runtime_host = {
				clr_country_flag = acid_rain_runtime_host
			}
			clear_global_event_target = acid_rain_runtime_host
		}
		clr_global_flag = acid_rain_host_worker_scheduled
		clr_global_flag = acid_rain_host_bound
		set_variable = { global.acid_rain_phase = constant:acid_rain_phase.closed }
		set_variable = { global.acid_rain_world_coverage = constant:acid_rain_dissipation.global_coverage_ladder_100 }
	}
}

# ANY SCOPE. This is the only scheduler entrypoint. It saves a global host
# target once and schedules a bounded country event; it never iterates the
# country list or state list to discover work.
acid_rain_bind_host_and_schedule_worker = {
	if = {
		limit = { NOT = { has_global_flag = acid_rain_host_bound } }
		save_global_event_target_as = acid_rain_runtime_host
		set_global_flag = acid_rain_host_bound
		set_country_flag = acid_rain_runtime_host
	}
	if = {
		limit = {
			has_global_flag = acid_rain_runtime_active
			NOT = { has_global_flag = acid_rain_host_worker_scheduled }
			has_event_target = acid_rain_runtime_host
		}
		set_global_flag = acid_rain_host_worker_scheduled
		event_target:acid_rain_runtime_host = {
			country_event = { id = chaosx.nr33.20 days = constant:acid_rain_runtime.worker_days }
		}
	}
}

# COUNTRY SCOPE. The event calls this effect on the stored host every three
# days. All work is bounded by frozen arrays and the three front/superstorm
# registries.
acid_rain_host_worker = {
	clr_global_flag = acid_rain_host_worker_scheduled
	if = {
		limit = { has_global_flag = acid_rain_runtime_active }
		for_each_scope_loop = {
			array = global.acid_rain_participant_countries
			if = {
				limit = { is_ai = yes }
				acid_rain_refresh_country_project_quotes = yes
				acid_rain_refresh_country_action_quotes = yes
			}
		}
		if = {
			limit = {
				check_variable = {
					var = global.acid_rain_phase
					value = constant:acid_rain_phase.regional
					compare = equals
				}
			}
			for_loop_effect = {
				start = constant:acid_rain_runtime.one
				end = constant:acid_rain_runtime.front_loop_end
				value = acid_rain_worker_front_id
				set_temp_variable = { acid_rain_front_id = acid_rain_worker_front_id }
				set_temp_variable = { acid_rain_selected_region_id = global.acid_rain_front_region_entries^acid_rain_front_id }
				acid_rain_process_front = yes
				acid_rain_pulse_front_states = yes
			}
			acid_rain_process_severe_cells = yes
			acid_rain_try_severe_evolution = yes
			acid_rain_try_multiple_fronts = yes
			if = {
			limit = {
				check_variable = {
					var = global.date
					value = global.acid_rain_air_week_start_day
					compare = greater_than_or_equals
				}
				check_variable = {
					var = global.acid_rain_total_valid_states
					value = constant:acid_rain_runtime.zero
					compare = greater_than
				}
			}
			set_temp_variable = { acid_rain_air_active_share = global.acid_rain_total_touched_states }
			multiply_temp_variable = { acid_rain_air_active_share = constant:acid_rain_runtime.preparedness_precision }
			divide_temp_variable = { acid_rain_air_active_share = global.acid_rain_total_valid_states }
			set_temp_variable = { acid_rain_air_requested_bp = constant:acid_rain_air.active_share_gain_low }
			if = {
				limit = { check_variable = { var = acid_rain_air_active_share value = constant:acid_rain_air.active_share_medium compare = greater_than_or_equals } }
				set_temp_variable = { acid_rain_air_requested_bp = constant:acid_rain_air.active_share_gain_medium }
			}
			if = {
				limit = { check_variable = { var = acid_rain_air_active_share value = constant:acid_rain_air.active_share_high compare = greater_than_or_equals } }
				set_temp_variable = { acid_rain_air_requested_bp = constant:acid_rain_air.active_share_gain_high }
			}
			if = {
				limit = { check_variable = { var = global.acid_rain_active_front_count value = constant:acid_rain_runtime.two compare = greater_than_or_equals } }
				add_to_temp_variable = { acid_rain_air_requested_bp = constant:acid_rain_air.split_gain }
			}
			if = {
				limit = { has_global_flag = acid_rain_global_layer_active }
				add_to_temp_variable = { acid_rain_air_requested_bp = constant:acid_rain_air.global_gain }
			}
			set_temp_variable = { acid_rain_active_severe_cell_count = constant:acid_rain_runtime.zero }
			set_temp_variable = { acid_rain_severe_cell_loop_end = constant:acid_rain_event.max_severe_cells }
			add_to_temp_variable = { acid_rain_severe_cell_loop_end = constant:acid_rain_runtime.loop_increment }
			for_loop_effect = {
				start = constant:acid_rain_runtime.one
				end = acid_rain_severe_cell_loop_end
				value = acid_rain_air_severe_cell_index
				if = {
					limit = { check_variable = { var = global.acid_rain_severe_cell_active_entries^acid_rain_air_severe_cell_index value = constant:acid_rain_runtime.one compare = equals } }
					add_to_temp_variable = { acid_rain_active_severe_cell_count = constant:acid_rain_runtime.loop_increment }
				}
			}
			if = {
				limit = { check_variable = { var = acid_rain_active_severe_cell_count value = constant:acid_rain_runtime.zero compare = greater_than } }
				set_temp_variable = { acid_rain_severe_air_gain = acid_rain_active_severe_cell_count }
				multiply_temp_variable = { acid_rain_severe_air_gain = constant:acid_rain_air.severe_gain }
				clamp_temp_variable = { var = acid_rain_severe_air_gain min = constant:acid_rain_runtime.zero max = constant:acid_rain_air.severe_gain_cap }
				add_to_temp_variable = { acid_rain_air_requested_bp = acid_rain_severe_air_gain }
			}
			set_temp_variable = { acid_rain_front_intensity_gain = constant:acid_rain_air.intensity_gain_weak }
			for_loop_effect = {
				start = constant:acid_rain_runtime.one
				end = constant:acid_rain_runtime.front_loop_end
				value = acid_rain_air_front_index
				if = {
					limit = { check_variable = { var = global.acid_rain_front_state_entries^acid_rain_air_front_index value = constant:acid_rain_front_state.active compare = equals } }
					if = {
						limit = { check_variable = { var = global.acid_rain_front_intensity_entries^acid_rain_air_front_index value = constant:acid_rain_intensity.extreme compare = greater_than_or_equals } }
						set_temp_variable = { acid_rain_front_intensity_gain = constant:acid_rain_air.intensity_gain_extreme }
					}
					else_if = {
						limit = { check_variable = { var = global.acid_rain_front_intensity_entries^acid_rain_air_front_index value = constant:acid_rain_intensity.strong compare = greater_than_or_equals } }
						if = { limit = { check_variable = { var = acid_rain_front_intensity_gain value = constant:acid_rain_air.intensity_gain_strong compare = less_than } } set_temp_variable = { acid_rain_front_intensity_gain = constant:acid_rain_air.intensity_gain_strong } }
					}
					else_if = {
						limit = { check_variable = { var = global.acid_rain_front_intensity_entries^acid_rain_air_front_index value = constant:acid_rain_intensity.standard compare = greater_than_or_equals } }
						if = { limit = { check_variable = { var = acid_rain_front_intensity_gain value = constant:acid_rain_air.intensity_gain_standard compare = less_than } } set_temp_variable = { acid_rain_front_intensity_gain = constant:acid_rain_air.intensity_gain_standard } }
					}
				}
			}
			add_to_temp_variable = { acid_rain_air_requested_bp = acid_rain_front_intensity_gain }
			clamp_temp_variable = { var = acid_rain_air_requested_bp min = constant:acid_rain_air.weekly_min_bp max = constant:acid_rain_air.weekly_max_bp }
			set_variable = { global.acid_rain_air_week_request_bp = acid_rain_air_requested_bp }
			set_temp_variable = { acid_rain_air_request_is_formation = constant:acid_rain_runtime.zero }
			acid_rain_apply_air_request = yes
			set_variable = { global.acid_rain_air_week_start_day = global.date }
			add_to_variable = { global.acid_rain_air_week_start_day = constant:acid_rain_runtime.contamination_days }
		}
		acid_rain_try_global_evolution = yes
	}
	if = {
		limit = {
			check_variable = {
				var = global.acid_rain_phase
				value = constant:acid_rain_phase.global_warning
				compare = equals
			}
		}
		acid_rain_try_global_evolution = yes
	}
	if = {
		limit = {
			has_global_flag = acid_rain_global_layer_active
			check_variable = {
				var = global.acid_rain_phase
				value = constant:acid_rain_phase.global
				compare = equals
			}
		}
		acid_rain_process_superstorms = yes
		acid_rain_pulse_global_states = yes
		acid_rain_try_extra_superstorm = yes
	}
	acid_rain_refresh_world_coverage = yes
	acid_rain_gui_refresh_read_only = yes
	if = {
		limit = {
			has_global_flag = acid_rain_runtime_active
			NOT = { has_global_flag = acid_rain_dissipation_pending }
			NOT = { has_global_flag = acid_rain_global_layer_active }
			acid_rain_all_regions_are_complete = yes
		}
		if = {
			limit = { NOT = { has_global_flag = acid_rain_coverage_complete } }
			set_global_flag = acid_rain_coverage_complete
			set_variable = { global.acid_rain_coverage_complete_day = global.date }
			set_variable = { global.acid_rain_dissipation_check_index = constant:acid_rain_runtime.zero }
			set_variable = { global.acid_rain_next_dissipation_check_day = global.date }
			add_to_variable = { global.acid_rain_next_dissipation_check_day = constant:acid_rain_dissipation.ordinary_completion_wait_days }
		}
		if = {
			limit = { check_variable = { global.date value = global.acid_rain_next_dissipation_check_day compare = greater_than_or_equals } }
			acid_rain_try_dissipation_check = yes
		}
	}
	if = {
		limit = {
			has_global_flag = acid_rain_global_layer_active
			NOT = { has_global_flag = acid_rain_dissipation_pending }
			check_variable = { var = global.date value = global.acid_rain_global_minimum_dissipation_day compare = greater_than_or_equals }
			check_variable = { var = global.date value = global.acid_rain_next_dissipation_check_day compare = greater_than_or_equals }
			check_variable = { var = global.acid_rain_total_touched_states value = global.acid_rain_total_valid_states compare = equals }
		}
		acid_rain_try_dissipation_check = yes
	}
	if = {
		limit = {
			has_global_flag = acid_rain_dissipation_pending
			check_variable = { var = global.date value = global.acid_rain_next_dissipation_check_day compare = greater_than_or_equals }
		}
		acid_rain_apply_dissipation_step = yes
	}
	if = {
		limit = { has_global_flag = acid_rain_runtime_active }
		acid_rain_bind_host_and_schedule_worker = yes
	}
}
