# CHAOS REDUX - EVENT 035 BOUNDED INCIDENT EFFECTS
# ============================================================================
# Event 35 owns a bounded, qualitative incident layer above the public
# Depression Severity ledger. The active-country registry calls one due review
# per weekly pulse; state targets come from the center registry or usable
# owned states, contagion targets come only from proved relationship rows, and
# every selected candidate receives one capped history receipt.
#
# These incidents are not a second meter, a routine factory-deletion loop, or a
# Chaos shop. Positive and negative candidates alter existing qualitative
# evidence, normal pressure is applied through the public Severity helper, and
# maximum political consequences continue through the existing guarded owners.
# A no-op branch and family cooldown keep the pool live without allowing one
# generic incident to dominate every band or evolution.
# ============================================================================

# COUNTRY scope. The bounded history survives recovery so relapse can see
# recent incident families, while the episode-local selection contract starts
# cleanly for every new crisis.
great_depression_incident_initialize_history = {
	if = {
		limit = { NOT = { has_variable = great_depression_committed_incident_receipt_ids^num } }
		clear_array = great_depression_committed_incident_receipt_ids
		clear_array = great_depression_committed_incident_kind_entries
		clear_array = great_depression_committed_incident_pool_entries
		clear_array = great_depression_committed_incident_episode_entries
		clear_array = great_depression_committed_incident_state_id_entries
		clear_array = great_depression_committed_incident_relationship_entries
	}
	if = {
		limit = { NOT = { has_variable = great_depression_incident_last_kind } }
		set_variable = { great_depression_incident_last_kind = constant:great_depression_incident_kind.none }
	}
	if = {
		limit = { NOT = { has_variable = great_depression_incident_last_pool } }
		set_variable = { great_depression_incident_last_pool = constant:great_depression_incident_pool.no_incident }
	}
	if = {
		limit = { NOT = { has_variable = great_depression_incident_last_day } }
		set_variable = { great_depression_incident_last_day = constant:great_depression_api.zero }
	}
}

# COUNTRY scope. Prefer one of the already registered centers, then fall back
# to a usable owned state. The selector is deliberately state-local and never
# scans or mutates the wider world.
great_depression_incident_select_state = {
	clear_variable = great_depression_incident_state
	clear_variable = great_depression_incident_state_id
	set_variable = { great_depression_incident_target_valid = constant:great_depression_api.zero }
	clear_event_target = great_depression_incident_state_target
	if = {
		limit = {
			has_variable = great_depression_center_state_ids^num
			check_variable = { var = great_depression_center_state_ids^num value = constant:great_depression_api.zero compare = greater_than }
		}
		random_scope_in_array = {
			array = great_depression_center_state_ids
			limit = { great_depression_incident_state_target_is_valid = yes }
			save_event_target_as = great_depression_incident_state_target
		}
	}
	if = {
		limit = {
			check_variable = { var = great_depression_incident_target_valid value = constant:great_depression_api.zero compare = equals }
		}
		random_owned_controlled_state = {
			limit = { great_depression_incident_state_target_is_valid = yes }
			save_event_target_as = great_depression_incident_state_target
		}
	}
	if = {
		limit = { has_event_target = great_depression_incident_state_target }
		set_variable = { great_depression_incident_state = event_target:great_depression_incident_state_target }
		event_target:great_depression_incident_state_target = {
			set_temp_variable = { great_depression_incident_state_id_request = id }
			ROOT = { set_variable = { great_depression_incident_state_id = great_depression_incident_state_id_request } }
		}
		set_variable = { great_depression_incident_target_valid = constant:great_depression_api.one }
	}
}

# COUNTRY scope. The partner is selected only from the target's existing
# contagion source registry. The aligned relationship row is recovered before
# an incident can enter the pool.
great_depression_incident_select_partner = {
	clear_variable = great_depression_incident_partner_country
	clear_variable = great_depression_incident_partner_relationship
	set_variable = { great_depression_incident_partner_valid = constant:great_depression_api.zero }
	clear_event_target = great_depression_incident_partner_target
	if = {
		limit = {
			has_variable = great_depression_contagion_source_countries^num
			check_variable = { var = great_depression_contagion_source_countries^num value = constant:great_depression_api.zero compare = greater_than }
		}
		random_scope_in_array = {
			array = great_depression_contagion_source_countries
			limit = { great_depression_incident_partner_target_is_valid = yes }
			save_event_target_as = great_depression_incident_partner_target
		}
	}
	if = {
		limit = { has_event_target = great_depression_incident_partner_target }
		set_variable = { great_depression_incident_partner_country = event_target:great_depression_incident_partner_target }
		set_temp_variable = { great_depression_incident_partner_row_found = constant:great_depression_api.zero }
		for_loop_effect = {
			start = 0
			end = great_depression_contagion_source_countries^num
			value = great_depression_incident_partner_row_index
			var:great_depression_contagion_source_countries^great_depression_incident_partner_row_index = {
				if = {
					limit = { tag = event_target:great_depression_incident_partner_target }
					PREV = {
						set_variable = { great_depression_incident_partner_relationship = great_depression_contagion_source_relationship_entries^great_depression_incident_partner_row_index }
						set_temp_variable = { great_depression_incident_partner_row_found = constant:great_depression_api.one }
					}
				}
			}
		}
		if = {
			limit = {
				check_variable = { var = great_depression_incident_partner_row_found value = constant:great_depression_api.one compare = equals }
				check_variable = { var = great_depression_incident_partner_relationship value = constant:great_depression_api.zero compare = greater_than }
			}
			set_variable = { great_depression_incident_partner_valid = constant:great_depression_api.one }
		}
	}
}

# COUNTRY scope. Partner-side qualitative evidence is written through the
# existing aligned row contract; no paid player action or Event 34 internals
# are called by a passive incident.
great_depression_incident_update_partner_row = {
	if = {
		limit = {
			has_event_target = great_depression_incident_partner_target
			has_variable = great_depression_contagion_source_countries^num
		}
		for_loop_effect = {
			start = 0
			end = great_depression_contagion_source_countries^num
			value = great_depression_incident_partner_update_index
			var:great_depression_contagion_source_countries^great_depression_incident_partner_update_index = {
				if = {
					limit = { tag = event_target:great_depression_incident_partner_target }
					PREV = {
						if = {
							limit = { has_variable = great_depression_incident_row_aid_state }
							set_variable = { great_depression_contagion_source_aid_state_entries^great_depression_incident_partner_update_index = great_depression_incident_row_aid_state }
						}
						if = {
							limit = { has_variable = great_depression_incident_row_material_source }
							set_variable = { great_depression_contagion_source_material_source_entries^great_depression_incident_partner_update_index = great_depression_incident_row_material_source }
						}
						if = {
							limit = { has_variable = great_depression_incident_row_cooldown_day }
							set_variable = { great_depression_contagion_source_cooldown_day_entries^great_depression_incident_partner_update_index = great_depression_incident_row_cooldown_day }
						}
					}
				}
			}
		}
	}
}

great_depression_incident_initialize_clock = {
	if = {
		limit = { NOT = { has_variable = great_depression_incident_next_day } }
		set_variable = { great_depression_incident_next_day = global.num_days }
		add_to_variable = { great_depression_incident_next_day = constant:great_depression_incident_timing.base_interval_days }
	}
	if = {
		limit = { NOT = { has_variable = great_depression_incident_family_cooldown_day } }
		set_variable = { great_depression_incident_family_cooldown_day = constant:great_depression_api.zero }
	}
}

great_depression_incident_defer = {
	set_variable = { great_depression_incident_next_day = global.num_days }
	add_to_variable = { great_depression_incident_next_day = constant:great_depression_incident_timing.no_op_retry_days }
	great_depression_incident_reset_selection = yes
}

great_depression_incident_expire_pressure = {
	if = {
		limit = {
			has_country_flag = great_depression_incident_open
			has_variable = great_depression_incident_pressure_until
			check_variable = { var = global.num_days value = great_depression_incident_pressure_until compare = greater_than_or_equals }
		}
		clr_country_flag = great_depression_incident_open
		clr_country_flag = great_depression_incident_pressure_active
		clear_variable = great_depression_incident_pressure_until
	}
}

great_depression_incident_schedule_next = {
	set_variable = { great_depression_incident_next_day = global.num_days }
	if = {
		limit = {
			OR = {
				has_country_flag = great_depression_evolution_financial_contagion
				has_country_flag = great_depression_evolution_social_collapse
			}
		}
		add_to_variable = { great_depression_incident_next_day = constant:great_depression_incident_timing.evolution_interval_days }
	}
	else = {
		add_to_variable = { great_depression_incident_next_day = constant:great_depression_incident_timing.base_interval_days }
	}
	set_variable = { great_depression_incident_family_cooldown_day = global.num_days }
	add_to_variable = { great_depression_incident_family_cooldown_day = constant:great_depression_incident_timing.family_cooldown_days }
}

great_depression_incident_record_receipt = {
	if = {
		limit = {
			check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.none compare = greater_than }
			great_depression_incident_receipt_capacity_is_available = yes
		}
		great_depression_allocate_receipt = yes
		set_variable = { great_depression_incident_receipt_id = great_depression_last_receipt_id }
		add_to_array = { array = great_depression_committed_incident_receipt_ids value = great_depression_incident_receipt_id }
		add_to_array = { array = great_depression_committed_incident_kind_entries value = great_depression_incident_kind }
		add_to_array = { array = great_depression_committed_incident_pool_entries value = great_depression_incident_pool }
		add_to_array = { array = great_depression_committed_incident_episode_entries value = great_depression_active_episode_id }
		add_to_array = { array = great_depression_committed_incident_state_id_entries value = great_depression_incident_state_id }
		add_to_array = { array = great_depression_committed_incident_relationship_entries value = great_depression_incident_partner_relationship }
		set_variable = { great_depression_incident_last_kind = great_depression_incident_kind }
		set_variable = { great_depression_incident_last_pool = great_depression_incident_pool }
		set_variable = { great_depression_incident_last_day = global.num_days }
	}
}

# COUNTRY scope. One selected candidate is committed through the public
# Severity path. The state/partner-specific branches below only add
# qualitative receipts after the ledger transaction succeeds.
great_depression_incident_apply_selected = {
	if = {
		limit = {
			great_depression_active_crisis_is_coherent = yes
			check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.none compare = greater_than }
			great_depression_incident_receipt_capacity_is_available = yes
		}
		if = {
			limit = { NOT = { has_variable = great_depression_incident_state_id } }
			set_variable = { great_depression_incident_state_id = constant:great_depression_api.zero }
		}
		if = {
			limit = { NOT = { has_variable = great_depression_incident_partner_relationship } }
			set_variable = { great_depression_incident_partner_relationship = constant:great_depression_api.zero }
		}
		great_depression_incident_record_receipt = yes
		set_country_flag = great_depression_incident_open
		set_variable = { great_depression_incident_pressure_until = global.num_days }
		add_to_variable = { great_depression_incident_pressure_until = constant:great_depression_incident_timing.active_window_days }
		if = {
			limit = { check_variable = { var = great_depression_incident_delta value = constant:great_depression_api.zero compare = greater_than } }
			set_country_flag = great_depression_incident_pressure_active
		}
		great_depression_incident_schedule_next = yes
		set_temp_variable = { great_depression_action_delta = great_depression_incident_delta }
		great_depression_apply_public_action_delta = yes

		if = {
			limit = {
				check_variable = { var = great_depression_incident_pool value = constant:great_depression_incident_pool.baseline compare = equals }
			}
			if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.factory_gate_closure compare = equals } }
				set_country_flag = great_depression_factory_gate_closure_observed
				event_target:great_depression_incident_state_target = { save_event_target_as = great_depression_center_status_target }
				set_variable = { great_depression_center_status = constant:great_depression_center_stage.idled }
				great_depression_set_center_status = yes
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.successful_reopening_shift compare = equals } }
				set_country_flag = great_depression_successful_reopening_observed
				event_target:great_depression_incident_state_target = { save_event_target_as = great_depression_center_status_target }
				set_variable = { great_depression_center_status = constant:great_depression_center_stage.reopened }
				great_depression_set_center_status = yes
				var:great_depression_incident_state = { great_depression_record_center_positive_resolution = yes }
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.unfinished_works_abandoned compare = equals } }
				set_country_flag = great_depression_unfinished_works_abandoned_observed
				var:great_depression_incident_state = { great_depression_fail_center_project = yes }
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.local_credit_cooperative compare = equals } }
				set_country_flag = great_depression_local_credit_cooperative_observed
				var:great_depression_incident_state = {
					set_state_flag = great_depression_center_finance_restored
					add_dynamic_modifier = { modifier = great_depression_center_protected }
				}
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.unpaid_wages compare = equals } }
				set_country_flag = great_depression_unpaid_wages_observed
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.freight_yards_idle compare = equals } }
				set_country_flag = great_depression_freight_yards_idle_observed
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.municipal_relief_failure compare = equals } }
				set_country_flag = great_depression_municipal_relief_failure_observed
				set_country_flag = great_depression_relief_offices_overwhelmed
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.skilled_workers_leave compare = equals } }
				set_country_flag = great_depression_skilled_workers_leave_observed
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.regional_aid_request compare = equals } }
				set_country_flag = great_depression_regional_aid_request_observed
				set_variable = { great_depression_regional_aid_request_state_id = great_depression_incident_state_id }
			}
			else_if = {
				limit = { check_variable = { var = great_depression_incident_kind value = constant:great_depression_incident_kind.war_orders_revival compare = equals } }
				set_country_flag = great_depression_war_orders_revival_observed
				set_country_flag = great_depression_freight_secured
			}
		}
		if = {
			limit = {
				check_variable = { var = great_depression_incident_pool value = constant:great_depression_incident_pool.financial_contagion compare = equals }
			}
			great_depression_incident_apply_contagion_outcome = yes
		}
		if = {
			limit = {
				check_variable = { var = great_depression_incident_pool value = constant:great_depression_incident_pool.social_collapse compare = equals }
			}
			great_depression_incident_apply_social_outcome = yes
		}
	}
}

great_depression_incident_run_baseline_pool = {
	set_variable = { great_depression_incident_pool = constant:great_depression_incident_pool.baseline }
	set_variable = { great_depression_incident_kind = constant:great_depression_incident_kind.none }
	set_variable = { great_depression_incident_delta = constant:great_depression_api.zero }
	great_depression_incident_select_state = yes
	random_list = {
		constant:great_depression_incident_weight.common = {
			modifier = { factor = constant:great_depression_incident_factor.disabled NOT = { great_depression_incident_current_distressed_center_target_is_valid = yes } }
			modifier = { factor = constant:great_depression_incident_factor.disabled check_variable = { var = great_depression_incident_last_kind value = constant:great_depression_incident_kind.factory_gate_closure compare = equals } }
			modifier = { factor = constant:great_depression_incident_factor.high check_variable = { var = great_depression_depression_severity value = constant:great_depression_event.deep_depression_threshold compare = greater_than_or_equals } }
			set_variable = { great_depression_incident_kind = constant:great_depression_incident_kind.factory_gate_closure }
			set_variable = { great_depression_incident_delta = constant:great_depression_incident_delta.major_pressure }

# COUNTRY scope. A passive due review is admitted only for an active country
# in the shared registry. The family cooldown is separate from the due clock;

		limit = {
			great_depression_active_crisis_is_coherent = yes

			constant:great_depression_incident_weight.baseline_pool = {
				modifier = { factor = constant:great_depression_incident_factor.disabled NOT = { great_depression_incident_has_valid_state = yes } }

			}

				modifier = { factor = constant:great_depression_incident_factor.disabled NOT = { has_country_flag = great_depression_evolution_financial_contagion } }
