# ============================================================================
# CHAOS REDUX - EVENT 028 ASTEROID INCOMING TRIGGERS
# ============================================================================
# Event-owned validation and ledger predicates for Event 028. These triggers
# deliberately validate a bounded target, state, or registry entry. They do
# not search the world and they do not choose a target. The parent event owns
# candidate weighting, evolution availability, decision definitions, and the
# final report surface.
#
# Target-country classification reuses the shared
# is_actual_nonhuman_country classifier. Special Chaos countries remain valid
# when they are still human-populated and use normal civilian systems.
# ============================================================================

asteroid_incoming_state_has_meaningful_settlement = {
	hidden_trigger = {
		scope_exists = yes
		impassable = no
		state_population_k > constant:asteroid_incoming_target.minimum_population_k
		OR = {
			has_state_category = city
			has_state_category = large_city
			has_state_category = metropolis
			has_state_category = megalopolis
			industrial_complex > constant:asteroid_incoming_target.minimum_building_level
			arms_factory > constant:asteroid_incoming_target.minimum_building_level
			dockyard > constant:asteroid_incoming_target.minimum_building_level
			infrastructure > constant:asteroid_incoming_target.minimum_building_level
			rail_way > constant:asteroid_incoming_target.minimum_building_level
			supply_node > constant:asteroid_incoming_target.minimum_building_level
			air_base > constant:asteroid_incoming_target.minimum_building_level
			naval_base > constant:asteroid_incoming_target.minimum_building_level
			state_strategic_value > constant:asteroid_incoming_target.minimum_strategic_value
		}
	}
}

asteroid_incoming_state_is_valid_target = {
	hidden_trigger = {
		scope_exists = yes
		impassable = no
		NOT = { has_state_category = wasteland }
		NOT = { has_state_flag = asteroid_incoming_main_crater }
		NOT = { has_state_flag = asteroid_incoming_fragment_site }
		asteroid_incoming_state_has_meaningful_settlement = yes
		OWNER = { exists = yes }
		CONTROLLER = { exists = yes }
	}
}

asteroid_incoming_country_is_valid_target = {
	hidden_trigger = {
		exists = yes
		is_actual_nonhuman_country = no
		has_event_target = asteroid_incoming_chooser
		NOT = { tag = event_target:asteroid_incoming_chooser }
		any_owned_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = PREV
		}
	}
}

asteroid_incoming_country_has_backup_state = {
	hidden_trigger = {
		exists = yes
		has_event_target = asteroid_incoming_candidate_state
		any_owned_state = {
			asteroid_incoming_state_is_valid_target = yes
			NOT = { state = event_target:asteroid_incoming_candidate_state }
		}
	}
}

# Called after the parent has saved candidate country and state targets. The
# state must still be owned and controlled by that same country at the point
# where the pair is accepted.
asteroid_incoming_target_pair_is_valid = {
	hidden_trigger = {
		has_event_target = asteroid_incoming_candidate_country
		has_event_target = asteroid_incoming_candidate_state
		event_target:asteroid_incoming_candidate_country = {
			asteroid_incoming_country_is_valid_target = yes
			asteroid_incoming_country_has_backup_state = yes
		}
		event_target:asteroid_incoming_candidate_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_candidate_country
		}
	}
}

# The locked pair preserves the accepted coordinates even if the original
# country is annexed or the state's owner, controller, category, or identity
# changes during the two-day trajectory. Only an unusable state scope or the
# absence of a live country to receive the physical transaction fails closed.
asteroid_incoming_locked_target_pair_is_valid = {
	hidden_trigger = {
		has_event_target = asteroid_incoming_locked_state
		event_target:asteroid_incoming_locked_state = {
			scope_exists = yes
			impassable = no
			NOT = { has_state_category = wasteland }
			OWNER = { exists = yes }
			CONTROLLER = { exists = yes }
		}
	}
}

asteroid_incoming_locked_transaction_is_valid = {
	hidden_trigger = {
		has_global_flag = asteroid_incoming_transaction_active
		has_global_flag = asteroid_incoming_target_locked
		NOT = { has_global_flag = asteroid_incoming_transaction_resolved }
		check_variable = {
			var = global.asteroid_incoming_transaction_id
			value = constant:asteroid_incoming_runtime.zero
			compare = greater_than
		}
		check_variable = {
			var = global.asteroid_incoming_lock_expires_day
			value = global.num_days
			compare = greater_than_or_equals
		}
		asteroid_incoming_locked_target_pair_is_valid = yes
	}
}

# COUNTRY scope, with the chooser in ROOT. Protected diplomatic relationships
# never receive a positive strike weight without a separately implemented and
# audited betrayal strategy; Event 028 does not invent one.
asteroid_incoming_ai_target_is_protected = {
	hidden_trigger = {
		OR = {
			tag = ROOT
			is_subject_of = ROOT
			ROOT = { is_subject_of = PREV }
			AND = { is_in_faction_with = ROOT NOT = { has_war_with = ROOT } }
			has_non_aggression_pact_with = ROOT
			has_guaranteed = ROOT
			is_guaranteed_by = ROOT
		}
	}
}

# Ring planning treats every adjacent state scope as a graph carrier and only
# excludes states already visited by this breadth-first search. Empty or
# impassable land states therefore carry the shortest path outward, while the
# later resolvable-transaction trigger prevents them from receiving losses or
# damage. A state in another profile remains traversable so the profile writer
# can deduplicate it by strongest applicable profile after discovery.
asteroid_incoming_state_is_valid_ring_candidate = {
	hidden_trigger = {
		scope_exists = yes
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_planner_seen_states
				value = THIS
			}
		}
	}
}

asteroid_incoming_state_is_valid_fragment_site = {
	hidden_trigger = {
		has_global_flag = asteroid_incoming_fragmentation_active
		check_variable = {
			var = global.asteroid_incoming_fragment_site_count
			value = constant:asteroid_incoming_runtime.maximum_fragment_sites
			compare = less_than
		}
		asteroid_incoming_state_is_valid_target = yes
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_main_footprint_states
				value = THIS
			}
		}
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_fragment_site_states
				value = THIS
			}
		}
		NOT = {
			is_in_array = {
				array = global.asteroid_incoming_fragment_reserved_first_ring_states
				value = THIS
			}
		}
	}
}

# A fragment center should have a usable land neighbor whenever the currently
# available global candidate pool contains one. The selector owns the
# fail-open pool check; this predicate only describes the preferred local
# geometry and deliberately does not require the neighbor to be unsettled.
asteroid_incoming_fragment_site_has_valid_land_neighbor = {
	hidden_trigger = {
		any_neighbor_state = {
			scope_exists = yes
			impassable = no
			NOT = { has_state_category = wasteland }
			OWNER = { exists = yes }
			CONTROLLER = { exists = yes }
			NOT = { has_state_flag = asteroid_incoming_main_crater }
			NOT = { has_state_flag = asteroid_incoming_fragment_site }
		}
	}
}

asteroid_incoming_state_is_valid_fragment_selection = {
	hidden_trigger = {
		asteroid_incoming_state_is_valid_fragment_site = yes
		OR = {
			NOT = { has_global_flag = asteroid_incoming_fragment_neighbor_pool_available }
			asteroid_incoming_fragment_site_has_valid_land_neighbor = yes
		}
	}
}

# COUNTRY scope. A national impact burden remains open while this country owns
# or controls at least one non-crater Event 028 state with a timed damage or
# rescue receipt, or while one of its exact-state recovery missions is active.
asteroid_incoming_country_has_unresolved_local_burden = {
	hidden_trigger = {
		OR = {
			NOT = { asteroid_incoming_no_active_recovery_mission = yes }
			any_owned_state = {
				has_state_flag = asteroid_incoming_affected_state
				NOT = { has_state_flag = asteroid_incoming_main_crater }
				OR = {
					has_state_flag = asteroid_incoming_rescue_burden_active
					has_dynamic_modifier = { modifier = asteroid_incoming_catastrophic_impact_zone }
					has_dynamic_modifier = { modifier = asteroid_incoming_major_impact_zone }
					has_dynamic_modifier = { modifier = asteroid_incoming_outer_shock_zone }
					has_dynamic_modifier = { modifier = asteroid_incoming_fragment_center_recovery }
					has_dynamic_modifier = { modifier = asteroid_incoming_heavy_fragment_damage }
					has_dynamic_modifier = { modifier = asteroid_incoming_light_fragment_damage }
				}
			}
			any_controlled_state = {
				has_state_flag = asteroid_incoming_affected_state
				NOT = { has_state_flag = asteroid_incoming_main_crater }
				OR = {
					has_state_flag = asteroid_incoming_rescue_burden_active
					has_dynamic_modifier = { modifier = asteroid_incoming_catastrophic_impact_zone }
					has_dynamic_modifier = { modifier = asteroid_incoming_major_impact_zone }
					has_dynamic_modifier = { modifier = asteroid_incoming_outer_shock_zone }
					has_dynamic_modifier = { modifier = asteroid_incoming_fragment_center_recovery }
					has_dynamic_modifier = { modifier = asteroid_incoming_heavy_fragment_damage }
					has_dynamic_modifier = { modifier = asteroid_incoming_light_fragment_damage }
				}
			}
		}
	}
}

# The dispatcher uses this as the fire-once gate. It is deliberately based on
# the six persisted targets rather than on a random-country count alone, so a
# partially populated pool can never expose a fourth or invalid target.
asteroid_incoming_has_three_valid_target_pairs = {
	hidden_trigger = {
		check_variable = {
			var = global.asteroid_incoming_target_pool_count
			value = constant:asteroid_incoming_runtime.minimum_target_pair_count
			compare = equals
		}
		has_event_target = asteroid_incoming_target_1_country
		has_event_target = asteroid_incoming_target_1_state
		has_event_target = asteroid_incoming_target_2_country
		has_event_target = asteroid_incoming_target_2_state
		has_event_target = asteroid_incoming_target_3_country
		has_event_target = asteroid_incoming_target_3_state
		event_target:asteroid_incoming_target_1_country = {
			exists = yes
			is_actual_nonhuman_country = no
			NOT = { tag = event_target:asteroid_incoming_chooser }
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				NOT = { state = event_target:asteroid_incoming_target_1_state }
			}
		}
		event_target:asteroid_incoming_target_1_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_target_1_country
		}
		event_target:asteroid_incoming_target_2_country = {
			exists = yes
			is_actual_nonhuman_country = no
			NOT = { tag = event_target:asteroid_incoming_chooser }
			NOT = { tag = event_target:asteroid_incoming_target_1_country }
			NOT = { tag = event_target:asteroid_incoming_target_3_country }
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				NOT = { state = event_target:asteroid_incoming_target_2_state }
			}
		}
		event_target:asteroid_incoming_target_2_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_target_2_country
			NOT = { state = event_target:asteroid_incoming_target_1_state }
			NOT = { state = event_target:asteroid_incoming_target_3_state }
		}
		event_target:asteroid_incoming_target_3_country = {
			exists = yes
			is_actual_nonhuman_country = no
			NOT = { tag = event_target:asteroid_incoming_chooser }
			NOT = { tag = event_target:asteroid_incoming_target_1_country }
			NOT = { tag = event_target:asteroid_incoming_target_2_country }
			any_owned_state = {
				asteroid_incoming_state_is_valid_target = yes
				is_owned_and_controlled_by = PREV
				NOT = { state = event_target:asteroid_incoming_target_3_state }
			}
		}
		event_target:asteroid_incoming_target_3_state = {
			asteroid_incoming_state_is_valid_target = yes
			is_owned_and_controlled_by = event_target:asteroid_incoming_target_3_country
			NOT = { state = event_target:asteroid_incoming_target_1_state }
			NOT = { state = event_target:asteroid_incoming_target_2_state }
		}
	}
}
