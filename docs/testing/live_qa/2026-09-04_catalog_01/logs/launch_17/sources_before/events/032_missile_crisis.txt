# Accepted retaliation completes preparation in a later event. This prevents
# direct recursive launch calls from the warning or damage chain.

	id = chaosx.nr32.83
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_country_flag = missiles_operation_active NOT = { has_country_flag = missiles_operation_committed } }
			missiles_complete_preparation = yes
			if = { limit = { missiles_operation_can_be_committed = yes } missiles_commit_operation = yes }
			else = { missiles_abort_operation = yes missiles_cleanup_operation = yes }
		}
	}
}
