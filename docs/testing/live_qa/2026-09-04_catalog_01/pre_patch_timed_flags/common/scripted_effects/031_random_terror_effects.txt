# ============================================================================
# CHAOS REDUX - EVENT 031 RANDOM TERROR EFFECTS
# ============================================================================
# Event-local runtime for the global Random Terror crisis.  The system keeps
# country pressure and response legitimacy separate from state activity, then
# promotes later network, territorial, and readiness values only through
# witnessed incidents, actor control, and accepted route actions.
#
# The automatic entry point is `random_terror_run_global_wave`.  It is called
# once by the Event 031 pacing dispatch and fans out to a bounded set of valid
# country/state targets.  Child report events never call event-system pacing.
# Actor carriers are ordinary predeclared country tags with existing HOI4 unit
# types; the dynamic package setup is transactional and can be cleaned up.
# ============================================================================

random_terror_update_state_stage = {
	random_terror_initialize_state_state = yes
	set_variable = { random_terror_state_stage = constant:random_terror_state_stage.clear }
	clr_state_flag = random_terror_state_dormant
	clr_state_flag = random_terror_state_active
	clr_state_flag = random_terror_state_entrenched
	clr_state_flag = random_terror_state_armed_insurgency
	clr_state_flag = random_terror_state_lost_local_control
	if = {
		limit = { check_variable = { random_terror_state_activity > constant:random_terror_value.zero } }
		set_variable = { random_terror_state_stage = constant:random_terror_state_stage.dormant }
		set_state_flag = random_terror_state_dormant
	}
	if = {
		limit = { check_variable = { random_terror_state_activity > constant:random_terror_tuning.state_activity_active_threshold } }
		set_variable = { random_terror_state_stage = constant:random_terror_state_stage.active }
		set_state_flag = random_terror_state_active
	}
	if = {
		limit = { check_variable = { random_terror_state_activity > constant:random_terror_tuning.state_activity_entrenched_threshold } }
		set_variable = { random_terror_state_stage = constant:random_terror_state_stage.entrenched }
		set_state_flag = random_terror_state_entrenched
	}
	if = {
		limit = { check_variable = { random_terror_state_activity > constant:random_terror_tuning.state_activity_armed_threshold } }
		set_variable = { random_terror_state_stage = constant:random_terror_state_stage.armed_insurgency }
		set_state_flag = random_terror_state_armed_insurgency
	}
	if = {
		limit = {
			check_variable = { random_terror_state_activity > constant:random_terror_value.minimum_pressure_for_territory }
			check_variable = { random_terror_state_pressure > constant:random_terror_value.minimum_pressure_for_territory }
		}
		set_variable = { random_terror_state_stage = constant:random_terror_state_stage.lost_local_control }
		set_state_flag = random_terror_state_lost_local_control
	}
	if = {
		limit = {
			OR = {
				has_state_flag = random_terror_state_active
				has_state_flag = random_terror_state_entrenched
				has_state_flag = random_terror_state_armed_insurgency
				has_state_flag = random_terror_state_lost_local_control
			}
			NOT = { has_state_flag = random_terror_crisis_counted }
		}
		set_state_flag = random_terror_crisis_counted
		add_to_variable = { global.random_terror_crisis_state_count = constant:random_terror_value.one }
	}
	random_terror_achievement_record_state_stage = yes
}

random_terror_resolve_targeted_raid_outcome = {
	random_terror_initialize_country_state = yes
	if = {
		limit = { has_event_target = random_terror_action_state }
		event_target:random_terror_action_state = {
			random_terror_initialize_state_state = yes
			set_temp_variable = { random_terror_raid_band = constant:random_terror_intensity.medium }
			if = {
				limit = {
					CONTROLLER = {
						check_variable = { random_terror_response_legitimacy > constant:random_terror_value.high }
					}
					check_variable = { random_terror_state_activity < constant:random_terror_tuning.state_activity_entrenched_threshold }
				}
				set_temp_variable = { random_terror_raid_band = constant:random_terror_intensity.high }
			}
			if = {
				limit = {
					OR = {
						check_variable = { random_terror_state_activity > constant:random_terror_tuning.state_activity_armed_threshold }
						CONTROLLER = {
							check_variable = { random_terror_response_legitimacy < constant:random_terror_value.low }
						}
					}
				}
				set_temp_variable = { random_terror_raid_band = constant:random_terror_intensity.low }
			}
			set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.partial_success }
			if = {
				limit = { check_variable = { random_terror_raid_band = constant:random_terror_intensity.high } }
				random_list = {
					@rt_raid_clean_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.clean_success } }
					@rt_raid_costly_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.costly_success } }
					@rt_raid_partial_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.partial_success } }
					@rt_raid_failure_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.failure } }
					@rt_raid_abusive_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.abusive_failure } }
				}
			}
			else_if = {
				limit = { check_variable = { random_terror_raid_band = constant:random_terror_intensity.low } }
				random_list = {
					@rt_raid_low_clean_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.clean_success } }
					@rt_raid_low_costly_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.costly_success } }
					@rt_raid_low_partial_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.partial_success } }
					@rt_raid_low_failure_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.failure } }
					@rt_raid_low_abusive_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.abusive_failure } }
				}
			}
			else = {
				random_list = {
					@rt_raid_mid_clean_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.clean_success } }
					@rt_raid_mid_costly_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.costly_success } }
					@rt_raid_mid_partial_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.partial_success } }
					@rt_raid_mid_failure_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.failure } }
					@rt_raid_mid_abusive_weight = { set_temp_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.abusive_failure } }
				}
			}
			if = {
				limit = { check_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.clean_success } }
				subtract_from_variable = { random_terror_state_activity = constant:random_terror_tuning.clean_operation_activity_reduction }
				subtract_from_variable = { random_terror_state_pressure = constant:random_terror_tuning.recovery_pressure_reduction }
				set_state_flag = random_terror_raid_clean_success
			}
			if = {
				limit = { check_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.costly_success } }
				subtract_from_variable = { random_terror_state_activity = constant:random_terror_tuning.costly_operation_activity_reduction }
				subtract_from_variable = { random_terror_state_pressure = constant:random_terror_tuning.protected_pressure_reduction }
				set_state_flag = random_terror_raid_costly_success
			}
			if = {
				limit = { check_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.partial_success } }
				subtract_from_variable = { random_terror_state_activity = constant:random_terror_tuning.partial_operation_activity_reduction }
				subtract_from_variable = { random_terror_state_pressure = constant:random_terror_tuning.state_activity_loss }
				set_state_flag = random_terror_raid_partial_success
			}
			if = {
				limit = {
					OR = {
						check_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.failure }
						check_variable = { random_terror_raid_outcome = constant:random_terror_operation_outcome.abusive_failure }
					}
				}
				add_to_variable = { random_terror_state_pressure = constant:random_terror_tuning.failed_operation_pressure_gain }
				add_to_variable = { random_terror_state_activity = constant:random_terror_tuning.state_activity_gain }
				set_state_flag = random_terror_failed_raid_recorded
			}
			if = {

				add_to_variable = { random_terror_state_pressure = constant:random_terror_tuning.abusive_operation_pressure_gain }
				set_state_flag = random_terror_raid_abusive_failure
				set_temp_variable = { random_terror_civilian_death_request = constant:random_terror_tuning.civilian_deaths_major }
				random_terror_apply_exact_state_loss = yes
				CONTROLLER = {
					set_temp_variable = { random_terror_pending_legitimacy_delta = constant:random_terror_tuning.abusive_operation_legitimacy_loss }
					multiply_temp_variable = { random_terror_pending_legitimacy_delta = -1 }
					random_terror_apply_legitimacy = yes

					random_terror_add_public_government_abuse_condemnation = yes
					random_terror_achievement_record_abusive_failure = yes
				}
			}
			clamp_variable = { var = random_terror_state_activity min = constant:random_terror_value.minimum max = constant:random_terror_value.maximum }
			clamp_variable = { var = random_terror_state_pressure min = constant:random_terror_value.minimum max = constant:random_terror_value.maximum }
			random_terror_update_state_stage = yes
			clr_state_flag = random_terror_raid_target
		}
	}
	set_country_flag = random_terror_raid_outcome_recorded
	random_terror_check_for_defeat = yes
}

random_terror_terminal_counteroffensive = {
	if = {
		limit = {
			has_global_flag = random_terror_terminal_active
			has_event_target = random_terror_final_actor
		}
		subtract_from_variable = { global.random_terror_apocalyptic_readiness = constant:random_terror_tuning.readiness_gain }
		subtract_from_variable = { global.random_terror_international_unity = constant:random_terror_tuning.unity_gain }
		clamp_variable = { var = global.random_terror_apocalyptic_readiness min = constant:random_terror_value.minimum max = constant:random_terror_value.maximum }
		clamp_variable = { var = global.random_terror_international_unity min = constant:random_terror_value.minimum max = constant:random_terror_value.maximum }
			event_target:random_terror_final_actor = {
			subtract_from_variable = { random_terror_territorial_control = constant:random_terror_tuning.terminal_denial_control_loss }
			subtract_from_variable = { random_terror_network_authority = constant:random_terror_tuning.terminal_denial_authority_loss }

			random_terror_clamp_actor_values = yes
			set_country_flag = random_terror_terminal_denial_pressure
		}
		if = {
			limit = { NOT = { has_variable = global.random_terror_terminal_counteroffensive_contributions } }
			set_variable = { global.random_terror_terminal_counteroffensive_contributions = constant:random_terror_value.zero }
		}
		add_to_variable = { global.random_terror_terminal_counteroffensive_contributions = constant:random_terror_value.one }
		set_country_flag = random_terror_terminal_counteroffensive_recorded
		random_terror_achievement_record_terminal_contribution = yes
		if = {
			limit = {
				check_variable = { var = global.random_terror_terminal_counteroffensive_contributions value = constant:random_terror_cost.terminal_counteroffensive_required_contributions compare = greater_than_or_equals }
				event_target:random_terror_final_actor = { random_terror_can_defeat_actor = yes }

			clr_global_flag = random_terror_false_revelation_pending
			clr_global_flag = random_terror_terminal_active
			event_target:random_terror_final_actor = {
				set_country_flag = random_terror_cleanup_requested
				random_terror_cleanup_actor = yes
			}
			random_terror_check_for_defeat = yes
		}
	}
}
