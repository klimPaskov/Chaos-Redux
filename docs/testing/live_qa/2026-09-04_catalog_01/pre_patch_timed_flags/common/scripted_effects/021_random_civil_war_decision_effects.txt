event021_add_same_tag_government_leverage = {
	if = {
		limit = {
			has_country_flag = random_civil_war_same_tag_contest_active
			has_variable = event021_same_tag_leverage_delta
		}
		add_to_variable = { random_civil_war_same_tag_government_leverage = event021_same_tag_leverage_delta }
		clamp_variable = { var = random_civil_war_same_tag_government_leverage min = constant:event021_same_tag_tuning.minimum max = constant:event021_same_tag_tuning.maximum }
	}
}

# COUNTRY scope on an exposed neighbor. Records the outbound relationship on
# the sponsor before the recipient-side helper stores its matching receipt.
event021_prepare_exposure_sponsor_commitment = {
	set_temp_variable = { event021_sponsor_id = THIS.id }
	set_temp_variable = { event021_support_amount = constant:event021_action_tuning.sponsor_support_amount }
	set_temp_variable = { event021_sponsor_duration = constant:event021_action_tuning.sponsor_commitment_days }
	if = {
		limit = { has_variable = random_civil_war_exposure_source_country_scope }
		var:random_civil_war_exposure_source_country_scope = {
			set_temp_variable = { ROOT.event021_recipient_id = THIS.id }
			if = {
				limit = { has_variable = random_civil_war_front_id }
				set_temp_variable = { ROOT.event021_front_id = random_civil_war_front_id }
			}
		}
	}
	event021_record_sponsor_commitment = yes
}

# COUNTRY scope on the actual receiving side. The sponsor remains ROOT while
# the selected government or claimant stores the paired commitment evidence.
event021_record_received_sponsor_commitment = {
	set_country_flag = random_civil_war_sponsor_commitment_received
	set_variable = { random_civil_war_sponsor_id = ROOT.id }
	set_variable = { random_civil_war_sponsor_recipient_id = THIS.id }
	set_variable = { ROOT.random_civil_war_sponsor_recipient_id = THIS.id }
	set_variable = { random_civil_war_sponsor_commitment_start_date = global.date }
	set_variable = { random_civil_war_sponsor_commitment_until = global.date }
	add_to_variable = { random_civil_war_sponsor_commitment_until = constant:event021_action_tuning.sponsor_commitment_days }
	if = {
		limit = { has_variable = random_civil_war_front_id }
		set_variable = { random_civil_war_sponsor_front_id = random_civil_war_front_id }
		set_variable = { ROOT.random_civil_war_sponsor_front_id = random_civil_war_front_id }
	}
	set_variable = { random_civil_war_sponsor_support_amount = constant:event021_action_tuning.sponsor_support_amount }
	set_temp_variable = { event021_evidence_source = constant:random_civil_war_evidence_source.sponsor }
	set_temp_variable = { event021_evidence_strength = constant:random_civil_war_evidence_strength.credible }
	set_temp_variable = { event021_evidence_duration = constant:event021_evidence_tuning.sponsor_receipt_days }
	event021_register_fracture_evidence = yes
}

event021_apply_government_support_to_exposure_target = {
	if = {
		limit = { has_variable = random_civil_war_exposure_source_country_scope }
		var:random_civil_war_exposure_source_country_scope = {
			if = {
				limit = { has_country_flag = random_civil_war_government_side }
				add_equipment_to_stockpile = { type = infantry_equipment amount = constant:event021_action_cost.government_support_infantry_transfer }
				set_country_flag = event021_government_support_received
				event021_record_received_sponsor_commitment = yes
			}
			else_if = {
				limit = {
					has_country_flag = random_civil_war_opposition_side
					has_variable = random_civil_war_host_country_scope
				}
				var:random_civil_war_host_country_scope = {
					add_equipment_to_stockpile = { type = infantry_equipment amount = constant:event021_action_cost.government_support_infantry_transfer }
					set_country_flag = event021_government_support_received
					event021_record_received_sponsor_commitment = yes
				}
			}
		}
	}
}

				is_owned_by = ROOT
				is_controlled_by = ROOT
				is_in_array = { array = ROOT.random_civil_war_opening_core_states value = THIS }
				OR = {
					supply_node > constant:random_civil_war_value.zero
					infrastructure > constant:event021_mission_tuning.rail_infrastructure_gate
				}
			}
		}
		event021_mission_secure_rail_complete = yes
	}
	else = { event021_mission_secure_rail_timeout = yes }
}

event021_mission_hold_settlement_terms_complete = {
	clr_country_flag = event021_settlement_terms_mission_active

		limit = { event021_parent_settlement_obligation_objective_valid = yes }
		set_country_flag = random_civil_war_settlement_obligation_objective_met
		clr_country_flag = random_civil_war_settlement_obligation_failed
	}
	else = { set_country_flag = random_civil_war_settlement_obligation_failed }
	event021_parent_review_settlement_obligation = yes
	event021_refresh_country_state = yes
}

		}

	}
	else = { event021_mission_hold_settlement_terms_timeout = yes }
}
