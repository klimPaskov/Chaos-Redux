event021_parent_country_is_opening_candidate = {
	# All normal and cross-event adapters share the release gate and the
	# individual-crisis capacity contract. Manual/debug force paths remain
	# explicitly disqualified by the dispatcher.
	has_global_flag = random_civil_war_rework_ready
	random_civil_war_is_normal_human_country = yes
	individual_crisis_load_is_below_cap = yes
	NOT = { random_civil_war_has_terminal_lock = yes }
	NOT = { random_civil_war_has_incompatible_bespoke_route = yes }
	NOT = { has_country_flag = random_civil_war_active }
	NOT = { has_country_flag = random_civil_war_target_reserved }
	NOT = { has_country_flag = random_civil_war_cluster_target_reserved }
	random_civil_war_generation_allowed = yes
	random_civil_war_has_valid_opposition_route = yes
}

event021_parent_event6_dormant_candidate = {
	has_global_flag = random_civil_war_rework_ready

# COUNTRY scope on the released carrier. This is a package-content proof
# rather than Event 006's full-origin completion trigger. The package-specific
# prepared predicate is selected dynamically, while Event 006 live arrays,
# origin history, evolution, and league membership remain explicitly absent.
event021_parent_event6_package_setup_proven = {
	has_country_flag = independence_wave_origin_prepared
	has_country_flag = independence_wave_package_setup_complete
	has_country_flag = independence_wave_command_roster_ready
	has_country_flag = independence_wave_force_mapping_loaded
	has_country_flag = independence_wave_force_package_applied
	has_country_flag = independence_wave_full_focus_framework
	check_variable = { independence_wave_focus_assignment = constant:independence_wave_focus_assignment.full_framework }
	has_independence_wave_generic_focus_contract = yes
	has_country_flag = independence_wave_generic_ai_profile
	has_event_target = independence_wave_setup_anchor_state
	has_event_target = independence_wave_setup_former_host
	has_variable = independence_wave_anchor_state
	has_variable = independence_wave_former_host
	var:independence_wave_anchor_state = {
		is_owned_and_controlled_by = PREV
		state_population_k > constant:event021_parent_tuning.minimum_state_population_k
		check_variable = { infrastructure_level > constant:random_civil_war_value.zero }
	}
	var:independence_wave_former_host = { exists = yes NOT = { tag = PREV } }
	has_independence_wave_force_package_for_current_generation = yes
	NOT = { has_country_flag = independence_wave_active_origin }
	NOT = { has_country_flag = independence_wave_network_member }
	NOT = { has_country_flag = independence_wave_league_member }
	if = {
		limit = {
			check_variable = {
				var = independence_wave_package_id
				value = constant:independence_wave_scenario_registry.two_digit_threshold
				compare = less_than
			}
		}
		meta_trigger = {
			text = { has_prepared_independence_wave_iw_00[PACKAGE_ID]_package_setup = yes }
			PACKAGE_ID = "[?independence_wave_package_id|.0]"
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = independence_wave_package_id
				value = constant:independence_wave_scenario_registry.three_digit_threshold
				compare = less_than
			}
		}
		meta_trigger = {
			text = { has_prepared_independence_wave_iw_0[PACKAGE_ID]_package_setup = yes }
			PACKAGE_ID = "[?independence_wave_package_id|.0]"
		}
	}
	else = {
		meta_trigger = {
			text = { has_prepared_independence_wave_iw_[PACKAGE_ID]_package_setup = yes }
			PACKAGE_ID = "[?independence_wave_package_id|.0]"
		}
	}
}

# COUNTRY scope on a source claimant. Evolution I may repeatedly detach one
# additional viable actor while the source, host-specific belligerent cap, and
# global front cap all remain safe. Every successful child receives its own
# aligned front row and the source is rechecked before another attempt.
event021_parent_secondary_front_candidate = {
	has_country_flag = random_civil_war_active
	has_country_flag = random_civil_war_opposition_actor
	num_of_controlled_states > constant:random_civil_war_value.one
	num_divisions > constant:random_civil_war_value.one
	NOT = { has_country_flag = random_civil_war_event6_front_active }
	NOT = { has_country_flag = random_civil_war_same_tag_route_active }
	random_civil_war_front_capacity_available = yes
	has_variable = random_civil_war_crisis_host_scope
	var:random_civil_war_crisis_host_scope = {
		exists = yes
		event021_parent_multifront_setup_available = yes
		has_variable = random_civil_war_total_belligerent_count
		has_variable = random_civil_war_belligerent_cap
		has_variable = random_civil_war_route_evidence_count
		check_variable = {
			var = random_civil_war_total_belligerent_count
			value = random_civil_war_belligerent_cap
			compare = less_than
		}
		check_variable = {
			var = random_civil_war_route_evidence_count
			value = random_civil_war_total_belligerent_count
			compare = greater_than_or_equals
		}
		event021_parent_has_unreserved_secondary_route = yes
	}
}

# COUNTRY scope of the shared scenario window. The selected type is copied into
# the global manual-scenario receipt only after the player confirms the launch;
# this gate therefore proves the shared, type-neutral pool before dispatch. The
# parent repeats the type-specific proof for every frozen row at commit time.
random_civil_war_triggerable_scenario_can_launch = {
	has_global_flag = random_civil_war_rework_ready
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = random_civil_war_manual_scenario_launch_in_progress }
	NOT = { has_global_flag = random_civil_war_launch_in_progress }
	NOT = { has_global_flag = random_civil_war_global_fracture_terminal }

		event021_parent_scenario_country_eligible = yes
	}
}

# COUNTRY scope. Manual Fracture Cascade selection is allowed to enter the
# parent transaction before a live route receipt exists. The parent then
# materialises the selected type's receipt and fails closed if actor, capital,
# state, or package proof cannot be completed. Automatic selection continues
# to use the stricter live route predicates in `021_random_civil_war_triggers`.
event021_parent_scenario_type_preflight_valid = {
	OR = {
		NOT = { has_variable = global.random_civil_war_manual_scenario_type }
		AND = {
			event021_parent_scenario_requires_same_tag = yes
			OR = {
				has_country_leader = { ruling_only = yes }
				num_divisions > constant:random_civil_war_value.zero
			}
		}
		AND = {
			NOT = { event021_parent_scenario_requires_same_tag = yes }
			check_variable = {
				var = global.random_civil_war_manual_scenario_type
				value = constant:random_civil_war_scenario_type.political_fracture
				compare = equals
			}
			OR = {
				has_country_leader = { ruling_only = yes }
				num_divisions > constant:random_civil_war_value.zero
			}
		}
		AND = {
			NOT = { event021_parent_scenario_requires_same_tag = yes }
			check_variable = {
				var = global.random_civil_war_manual_scenario_type
				value = constant:random_civil_war_scenario_type.independence_cascade
				compare = equals
			}
			event021_parent_event6_dormant_candidate = yes
		}
		AND = {
			NOT = { event021_parent_scenario_requires_same_tag = yes }
			check_variable = {
				var = global.random_civil_war_manual_scenario_type
				value = constant:random_civil_war_scenario_type.command_collapse
				compare = equals
			}
			num_divisions > constant:random_civil_war_value.one
		}
		AND = {
			NOT = { event021_parent_scenario_requires_same_tag = yes }
			check_variable = {
				var = global.random_civil_war_manual_scenario_type
				value = constant:random_civil_war_scenario_type.universal_fragmentation
				compare = equals
			}
			OR = {
				num_divisions > constant:random_civil_war_value.zero
				has_country_leader = { ruling_only = yes }
			}
		}
	}
}
