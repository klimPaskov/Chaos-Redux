# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND SCRIPTED TRIGGERS
# ============================================================================
# State and country predicates for the bounded mine registry. This file owns
# Event 029 target eligibility, public/hidden value bands, lifecycle gates,
# contract and mission validity, controller checks, and the three evolution
# gates. It deliberately has no cross-event resource, outbreak, actor, or
# world-transition dependencies.
# ============================================================================

# ---------------------------------------------------------------------------
# Target and country gates
# ---------------------------------------------------------------------------

riches_found_country_is_excluded_actor = {
	OR = {
		is_special_chaos_country = yes
		is_actual_nonhuman_country = yes
		has_country_flag = chaosx_test_country_initialized
	}
}

riches_found_state_is_excluded_target = {
	OR = {
		has_state_flag = riches_found_mine_registered
		has_state_flag = riches_found_target_excluded
		has_state_flag = riches_found_incompatible_terminal_site
		# Event 018 persistent sites are a separate system and can never be
		# converted into an Event 029 mine.
		has_state_flag = resources_found_field_active
		has_state_flag = resources_found_field_suspended
		has_state_flag = resources_found_field_closed_permanently
		has_state_flag = resources_found_field_converted_to_cave_origin
		has_state_flag = resources_found_field_converted_to_cave_breach
		has_state_flag = resources_found_cave_origin
		has_state_flag = resources_found_cave_anchor_activating
		has_state_flag = resources_found_cave_anchor_active
		has_state_flag = resources_found_cave_anchor_disrupted
		has_state_flag = resources_found_cave_mature_non_origin_anchor
		has_state_flag = resources_found_cave_cleanup_required
		has_state_flag = resources_found_cave_cleanup_complete
		has_state_flag = resources_found_cave_world_end_foothold
		has_state_flag = resources_found_field_transaction_lock
	}
}

riches_found_country_is_valid_recipient = {
	exists = yes
	NOT = { riches_found_country_is_excluded_actor = yes }
	has_capitulated = no
	capital_scope = {
		exists = yes
		is_controlled_by = PREV
	}
}

riches_found_state_is_valid_target = {
	impassable = no
	OWNER = {
		exists = yes
		NOT = { is_special_chaos_country = yes }
		NOT = { is_actual_nonhuman_country = yes }
	}
	CONTROLLER = {
		exists = yes
		NOT = { is_special_chaos_country = yes }
		NOT = { is_actual_nonhuman_country = yes }
	}
	is_owned_and_controlled_by = OWNER
	NOT = { has_global_flag = world_end }
	NOT = { riches_found_state_is_excluded_target = yes }
	OR = {
		check_variable = { state_population_k > constant:riches_found_threshold.minimum_state_population_thousands }
		check_variable = { infrastructure > constant:riches_found_threshold.minimum_infrastructure }
		check_variable = { rail_way > constant:riches_found_threshold.minimum_railway }
		supply_node > constant:riches_found_value.zero
		is_coastal = yes
	}
	OR = {
		NOT = { is_island_state = yes }
		AND = {
			is_island_state = yes
			OR = {
				naval_base > constant:riches_found_value.zero
				supply_node > constant:riches_found_value.zero
			}
		}
	}
}

riches_found_state_is_preferred_target = {
	riches_found_state_is_valid_target = yes
	OR = {
		check_variable = { state_population_k > constant:riches_found_threshold.preferred_state_population_thousands }
		check_variable = { infrastructure > constant:riches_found_threshold.preferred_infrastructure }
		check_variable = { rail_way > constant:riches_found_threshold.preferred_railway }
		supply_node > constant:riches_found_value.zero
	}
}

# first mine is selected automatically; a controller with multiple mines may
# replace the selection through the bounded selector decision.

	riches_found_mine_is_registered = yes
	CONTROLLER = {
		exists = yes
		riches_found_country_is_valid_recipient = yes
	}
	OR = {
		CONTROLLER = {
			NOT = { has_variable = riches_found_selected_mine }
		}
		CONTROLLER = {
			check_variable = {
				var = riches_found_selected_mine
				value = PREV
				compare = equals
			}
		}
		CONTROLLER = {
			NOT = { riches_found_country_has_multiple_mines = yes }
		}
	}
}

# COUNTRY scope. The nine mission definitions share one bounded three-slot
# budget. The count is rebuilt from controlled mine subjects whenever a
# decision starts, a mission resolves, or a control hook runs, so stale slots
# cannot be farmed through transfer or cancellation.
riches_found_country_has_free_mission_slot = {
	OR = {
		NOT = { has_variable = riches_found_active_mission_count }
		check_variable = {
			var = riches_found_active_mission_count
			value = constant:riches_found_threshold.maximum_active_missions
			compare = less_than
		}
	}
}

# STATE scope. A foreign actor is deliberately limited to an ordinary country
# controlling a neighbouring land state. This keeps concession selection local
# to the mine and avoids a country-wide or world-wide target scan.

	riches_found_mine_is_registered = yes
	CONTROLLER = { exists = yes }
	any_neighbor_state = {

			NOT = { is_special_chaos_country = yes }

			has_capitulated = no
		}
	}
}
