# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND INCIDENT EFFECTS
# ============================================================================
# State-scoped candidate pools for ordinary mine incidents, foreign pressure,
# concrete raids, collapse severity, evolved crises, and contract outcomes.
# The current controller's already-bounded pulse calls one registered mine at
# a time. Recurring foreign interference is limited to a stored contract actor
# or a directly neighbouring controller; no recurring whole-world scan occurs.
# Every candidate records one incident id on the mine, applies centralized
# value bands, and retains one-step memory so the same report cannot repeat
# immediately.
# ============================================================================

# STATE scope. Clears the temporary delta contract before one candidate writes
# its outcome. Temporary variables are deliberately unscoped.
riches_found_reset_incident_deltas = {
	set_temp_variable = { riches_found_incident_pressure_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_development_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_order_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_legitimacy_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_capture_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_foreign_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_gold_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_supernatural_delta = constant:riches_found_value.zero }
	set_temp_variable = { riches_found_incident_collapse_delta = constant:riches_found_value.zero }
}

# STATE scope. Commits one selected candidate and schedules the next bounded
# review. Good management spaces incidents farther apart; a crisis shortens
# the interval without making any single candidate certain.
riches_found_finish_incident = {
	add_to_variable = { riches_found_extraction_pressure = riches_found_incident_pressure_delta }
	add_to_variable = { riches_found_mine_development = riches_found_incident_development_delta }
	add_to_variable = { riches_found_local_order = riches_found_incident_order_delta }
	add_to_variable = { riches_found_revenue_legitimacy = riches_found_incident_legitimacy_delta }
	add_to_variable = { riches_found_illicit_capture = riches_found_incident_capture_delta }
	add_to_variable = { riches_found_foreign_claim_pressure = riches_found_incident_foreign_delta }
	add_to_variable = { riches_found_gold_disease_pressure = riches_found_incident_gold_delta }
	add_to_variable = { riches_found_supernatural_pressure = riches_found_incident_supernatural_delta }
	add_to_variable = { riches_found_collapse_pressure = riches_found_incident_collapse_delta }
	set_variable = { riches_found_previous_incident_kind = riches_found_last_incident_kind }
	set_variable = { riches_found_last_incident_kind = riches_found_incident_kind }
	if = { limit = { NOT = { has_variable = riches_found_incident_sequence } } set_variable = { riches_found_incident_sequence = constant:riches_found_value.zero } }
	add_to_variable = { riches_found_incident_sequence = constant:riches_found_value.one }
	set_variable = { riches_found_last_incident_date = global.date }
	set_variable = { riches_found_next_incident_date = global.date }
	set_temp_variable = { riches_found_incident_interval = constant:riches_found_duration.incident_interval_days }
	if = {
		limit = {
			check_variable = { var = riches_found_local_order value = constant:riches_found_threshold.order_high compare = greater_than_or_equals }
			check_variable = { var = riches_found_revenue_legitimacy value = constant:riches_found_threshold.legitimacy_high compare = greater_than_or_equals }
			check_variable = { var = riches_found_extraction_pressure value = constant:riches_found_threshold.pressure_warning compare = less_than }
		}
		set_temp_variable = { riches_found_incident_interval = constant:riches_found_duration.incident_interval_managed_days }
	}
	else_if = {
		limit = {
			OR = {
				riches_found_mine_is_severe_pressure = yes
				riches_found_mine_is_local_order_crisis = yes
				riches_found_mine_is_legitimacy_crisis = yes
			}
		}
		set_temp_variable = { riches_found_incident_interval = constant:riches_found_duration.incident_interval_crisis_days }
	}
	add_to_variable = { riches_found_next_incident_date = riches_found_incident_interval }
	riches_found_clamp_mine_values = yes
	riches_found_queue_report = yes
}

# STATE scope. Initializes the due date without immediately firing a passive
# checklist event on discovery.
riches_found_initialize_incident_clock = {
	if = {
		limit = { NOT = { has_variable = riches_found_next_incident_date } }
		set_variable = { riches_found_next_incident_date = global.date }
		add_to_variable = { riches_found_next_incident_date = constant:riches_found_duration.incident_interval_days }
	}
}

# COUNTRY scope. Inputs: regular event targets riches_found_interest_mine and
# riches_found_interest_controller. Output: unscoped candidate score used by
# the on-demand contract-target pass.
riches_found_calculate_foreign_interest_score = {
	set_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.base }
	if = { limit = { is_major = yes } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.major } }
	if = { limit = { num_of_factories > constant:riches_found_foreign_interest_threshold.medium_industry } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.medium_industry } }
	if = { limit = { num_of_factories > constant:riches_found_foreign_interest_threshold.large_industry } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.large_industry } }
	if = { limit = { is_neighbor_of = event_target:riches_found_interest_controller } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.neighbor } }
	if = { limit = { is_in_faction_with = event_target:riches_found_interest_controller } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.same_faction } }
	if = { limit = { has_opinion = { target = event_target:riches_found_interest_controller value > constant:riches_found_foreign_interest_threshold.positive_opinion } } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.positive_relations } }
	if = { limit = { has_opinion = { target = event_target:riches_found_interest_controller value < constant:riches_found_foreign_interest_threshold.hostile_opinion } } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.hostile_relations } }
	if = { limit = { event_target:riches_found_interest_controller = { has_war = yes } } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.wartime_need } }
	if = { limit = { event_target:riches_found_interest_controller = { num_of_factories < constant:riches_found_foreign_interest_threshold.weak_factory_count } } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.controller_weakness } }
	if = { limit = { has_country_flag = riches_found_foreign_contract_network } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.contract_network } }
	if = { limit = { has_equipment = { convoy_1 > constant:riches_found_foreign_interest_threshold.minimum_convoys } } add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.overseas_access } }
	if = {
		limit = {
			event_target:riches_found_interest_mine = {
				OR = {
					riches_found_mine_is_severe_pressure = yes
					has_state_flag = riches_found_severe_collapse
					has_state_flag = riches_found_supernatural_crisis
				}
			}
		}
		add_to_temp_variable = { riches_found_foreign_candidate_score = constant:riches_found_foreign_interest.severe_crisis }
	}
	clamp_temp_variable = { var = riches_found_foreign_candidate_score min = constant:riches_found_foreign_interest.minimum max = constant:riches_found_ai_weight.maximum }
}

# STATE scope. This is called by a deliberate contract action, not by the
# recurring pulse. It scores every plausible ordinary actor once and stores
# the highest-scoring valid partner. The persistent contract still records one
# partner only.
riches_found_prepare_scored_foreign_partner = {
	clear_variable = riches_found_foreign_target
	clr_state_flag = riches_found_foreign_target_available
	save_event_target_as = riches_found_interest_mine
	CONTROLLER = { save_event_target_as = riches_found_interest_controller }
	set_temp_variable = { riches_found_best_foreign_score = constant:riches_found_value.zero }
	every_country = {
		limit = {
			exists = yes
			NOT = { tag = event_target:riches_found_interest_controller }
			NOT = { is_special_chaos_country = yes }
			NOT = { is_actual_nonhuman_country = yes }
			has_capitulated = no
			NOT = { has_war_with = event_target:riches_found_interest_controller }
			OR = {
				is_neighbor_of = event_target:riches_found_interest_controller
				is_in_faction_with = event_target:riches_found_interest_controller
				has_opinion = { target = event_target:riches_found_interest_controller value > constant:riches_found_foreign_interest_threshold.positive_opinion }
				AND = {
					is_major = yes
					has_equipment = { convoy_1 > constant:riches_found_foreign_interest_threshold.minimum_convoys }
					event_target:riches_found_interest_mine = {
						OR = {
							has_state_flag = riches_found_port_route_valid
							has_state_flag = riches_found_port_route
							has_state_flag = riches_found_island_mine
						}
					}
				}
			}
		}
		riches_found_calculate_foreign_interest_score = yes
		if = {
			limit = { check_variable = { var = riches_found_foreign_candidate_score value = riches_found_best_foreign_score compare = greater_than } }
			set_temp_variable = { riches_found_best_foreign_score = riches_found_foreign_candidate_score }
			save_event_target_as = riches_found_contract_partner_target
		}
	}
	if = {
		limit = { has_event_target = riches_found_contract_partner_target }
		set_variable = { riches_found_foreign_target = event_target:riches_found_contract_partner_target }
		set_variable = { riches_found_foreign_interest_score = riches_found_best_foreign_score }
		set_state_flag = riches_found_foreign_target_available
	}
}

# STATE scope. Recurring interference never invokes the global scored pass. A
# durable contract partner is preferred; otherwise one directly neighbouring
# controller is selected through the state adjacency scope.
riches_found_prepare_narrow_interference_actor = {
	clear_variable = riches_found_interference_actor
	clr_state_flag = riches_found_interference_actor_friendly
	clr_state_flag = riches_found_interference_actor_hostile
	CONTROLLER = { save_event_target_as = riches_found_interference_controller }
	if = {
		limit = {
			has_variable = riches_found_contract_partner
			var:riches_found_contract_partner = { exists = yes }
		}
		set_variable = { riches_found_interference_actor = riches_found_contract_partner }
	}
	else_if = {
		limit = {
			any_neighbor_state = {
				impassable = no
				CONTROLLER = {
					exists = yes
					NOT = { tag = event_target:riches_found_interference_controller }
					NOT = { is_special_chaos_country = yes }
					NOT = { is_actual_nonhuman_country = yes }
					has_capitulated = no
				}
			}
		}
		random_neighbor_state = {
			limit = {
				impassable = no
				CONTROLLER = {
					exists = yes
					NOT = { tag = event_target:riches_found_interference_controller }
					NOT = { is_special_chaos_country = yes }
					NOT = { is_actual_nonhuman_country = yes }
					has_capitulated = no
				}
			}
			set_variable = { riches_found_interference_actor = CONTROLLER }
		}
	}
	if = {
		limit = { has_variable = riches_found_interference_actor }
		if = {
			limit = {
				var:riches_found_interference_actor = {
					OR = {
						is_in_faction_with = event_target:riches_found_interference_controller
						has_opinion = { target = event_target:riches_found_interference_controller value > constant:riches_found_foreign_interest_threshold.positive_opinion }
					}
				}
			}
			set_state_flag = riches_found_interference_actor_friendly
		}
		if = {
			limit = {
				var:riches_found_interference_actor = {
					OR = {
						has_war_with = event_target:riches_found_interference_controller
						has_opinion = { target = event_target:riches_found_interference_controller value < constant:riches_found_foreign_interest_threshold.hostile_opinion }
					}
				}
			}
			set_state_flag = riches_found_interference_actor_hostile
		}
	}
}

# STATE scope. Preserves the concrete actor and bounded evidence count. Audits,
# disclosure, civil policing, or a published contract reveal attribution;
# otherwise the operation remains publicly unattributed.

		set_variable = { riches_found_last_interference_actor = riches_found_interference_actor }
		add_to_variable = { riches_found_foreign_evidence_count = constant:riches_found_value.one }
		set_state_flag = riches_found_foreign_interference_evidence
		if = {
			limit = {
				OR = {
					has_state_flag = riches_found_audit_complete
					has_state_flag = riches_found_beneficial_owners_disclosed
					has_state_flag = riches_found_contract_published
					has_state_flag = riches_found_mine_police
				}
			}
			set_state_flag = riches_found_foreign_interference_attributed
			clr_state_flag = riches_found_foreign_interference_unattributed
			var:riches_found_interference_actor = { set_country_flag = riches_found_interference_exposed }
		}
		else = {
			set_state_flag = riches_found_foreign_interference_unattributed
		}
	}
}

# STATE scope. Called from the controller's registered-mine pulse. The outer
# list selects one complete candidate pool; each pool owns its own weights and
# invalid-target suppression. The no-incident branch preserves occasional
# quiet periods even under otherwise valid conditions.
riches_found_run_due_incident_review = {
	riches_found_initialize_incident_clock = yes
	if = {
		limit = {
			NOT = { has_state_flag = riches_found_report_pending }
			check_variable = { var = global.date value = riches_found_next_incident_date compare = greater_than_or_equals }
		}
		random_list = {
			constant:riches_found_incident_weight.ordinary_pool = {
				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { check_variable = { var = riches_found_phase value = constant:riches_found_phase.discovery_claims compare = equals } } }
				riches_found_run_claims_incident_pool = yes
			}
			constant:riches_found_incident_weight.ordinary_pool = {
				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { check_variable = { var = riches_found_phase value = constant:riches_found_phase.rush compare = equals } } }
				riches_found_run_rush_incident_pool = yes
			}
			constant:riches_found_incident_weight.ordinary_pool = {
				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { check_variable = { var = riches_found_phase value = constant:riches_found_phase.physical_development compare = equals } } }
				riches_found_run_development_incident_pool = yes
			}
			constant:riches_found_incident_weight.ordinary_pool = {
				modifier = { factor = constant:riches_found_incident_factor.high has_state_flag = riches_found_evolution_resource_curse }
				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { OR = { check_variable = { var = riches_found_phase value = constant:riches_found_phase.revenue_settlement compare = equals } has_state_flag = riches_found_evolution_resource_curse } } }
				riches_found_run_corruption_incident_pool = yes
			}
			constant:riches_found_incident_weight.foreign_pool = {
				modifier = { factor = constant:riches_found_incident_factor.high check_variable = { var = riches_found_foreign_claim_pressure value = constant:riches_found_threshold.concession_exposure_gate compare = greater_than_or_equals } }
				modifier = {
					factor = constant:riches_found_incident_factor.disabled
					NOT = {
						OR = {
							has_variable = riches_found_contract_partner
							any_neighbor_state = {
								impassable = no
								CONTROLLER = { exists = yes NOT = { tag = ROOT } NOT = { is_special_chaos_country = yes } NOT = { is_actual_nonhuman_country = yes } }
							}
						}
					}
				}
				riches_found_run_foreign_interference_pool = yes
			}
			constant:riches_found_incident_weight.raid_pool = {
				modifier = { factor = constant:riches_found_incident_factor.high riches_found_mine_is_local_order_crisis = yes }
				modifier = { factor = constant:riches_found_incident_factor.high has_state_flag = riches_found_raid_crisis }
				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { OR = { has_state_flag = riches_found_phase_security has_state_flag = riches_found_crime_crisis has_state_flag = riches_found_armed_claimants has_state_flag = riches_found_wage_dispute riches_found_mine_is_local_order_crisis = yes } } }

			}
			constant:riches_found_incident_weight.collapse_pool = {
				modifier = { factor = constant:riches_found_incident_factor.severe riches_found_mine_is_severe_pressure = yes }
				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { OR = { has_state_flag = riches_found_collapse_warning check_variable = { var = riches_found_collapse_pressure value = constant:riches_found_threshold.collapse_risk_gate compare = greater_than_or_equals } } } }

				riches_found_run_supernatural_incident_pool = yes

				modifier = { factor = constant:riches_found_incident_factor.disabled NOT = { OR = { has_state_flag = riches_found_contract_active has_state_flag = riches_found_contract_renewal_due has_state_flag = riches_found_contract_dispute } } }
