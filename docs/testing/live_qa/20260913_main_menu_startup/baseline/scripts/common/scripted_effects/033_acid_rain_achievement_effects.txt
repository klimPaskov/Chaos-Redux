# ============================================================================
# CHAOS REDUX - EVENT 033 ACID RAIN ACHIEVEMENT TRACKING EFFECTS
# ============================================================================
# This file is the Event 033 achievement ledger. It is deliberately separate
# from the weather runtime: the runtime owns weather state, while these
# effects retain the exact historical receipts used by the ten achievements.
#
# Contract rules
# - Every public recorder is generation-gated. The caller supplies a temporary
#   `acid_rain_033_input_generation` equal to global.acid_rain_generation_id.
# - Positive deaths are accepted only through the caller's exact population
#   gateway result in `acid_rain_033_input_applied_deaths`. No modifier or
#   current-state estimate is used here.
# - State and country arrays are append-only evidence ledgers for the current
#   generation. Duplicate receipt keys are no-ops, so save/load retries cannot
#   double-count an edge.
# - The transfer effect consumes the runtime's old/new controller event
#   targets in the same control-change chain and moves only current
#   responsibility. Historical deaths and completed receipts remain attached
#   to their original credited country.
# - No effect in this file runs a world or country scan. Parent callers invoke
#   recorders at formation, warning, arrival, pulse, action, closure, phase,
#   dissipation, recovery, transfer, and bounded deadline edges.
#
# Public hook order
# 1. Country formation: begin_generation, then register_state once per frozen
#    eligible state.
# 2. Warning/arrival: record_warning_visit and record_arrival.
# 3. Preparedness: record_preparedness_snapshot and
#    record_preparedness_component at paid tier completion.
# 4. Exposure: record_state_exposure once per exact pulse/coverage receipt;
#    use record_opening_shock for an opening pulse and pass the exact applied
#    loss. Severe forecast/cell helpers add the episode evidence.
# 5. Actions: record_transport_reroute, record_evacuation_start,
#    record_evacuation_complete, and record_urgent_action_completion.
# 6. Severe closure: record_severe_cell_closure.
# 7. Global layer: record_global_transition and
#    record_global_preparedness_check on each required continuity check.
# 8. Dissipation/recovery: record_dissipation_state_snapshot at the frozen
#    snapshot edge, record_recovery_completion at each accepted recovery, and
#    refresh_recovery_deadlines at the bounded deadline edge.
# 9. Aftermath: record_aftermath_tier whenever the stored tier changes and
#    refresh_no_severe_ruins_deadlines at each bounded ruin deadline edge.
# 10. State control change: transfer_state_responsibility while the runtime
#     owns acid_rain_transfer_old_controller and
#     acid_rain_transfer_new_controller.
#
# Inputs named `acid_rain_033_input_*` are caller-owned temporary variables.
# A missing or malformed required input fails closed by marking the relevant
# migration/baseline disqualifier. The parent must not substitute current
# modifiers for a missing receipt.
# ============================================================================

# COUNTRY SCOPE. Initializes a country only when the parent has established a
# global Event 033 generation. Re-entry in the same generation is idempotent;
# a newer generation receives a fresh ledger while old receipts remain.

	if = {
		limit = {
			has_variable = global.acid_rain_generation_id
			OR = {
				NOT = { has_country_flag = acid_rain_033_achievement_tracking_initialized }
				NOT = { check_variable = { var = acid_rain_033_achievement_generation_id value = global.acid_rain_generation_id compare = equals } }
			}
		}

# COUNTRY SCOPE. Starts a generation and freezes the immutable national facts.
# Required inputs: generation, formation_state_count, formation_population_people,
# start_civilian_factories, capital_id, and has_capital. Missing population,
# factory, or capital proof marks the baseline invalid; no current value is
# substituted later.
acid_rain_033_achievement_begin_generation = {
	if = {
		limit = { acid_rain_033_achievement_input_generation_is_current = yes }
		acid_rain_033_achievement_initialize_country = yes
		if = {
			limit = {
				has_variable = acid_rain_033_input_formation_state_count
				has_variable = acid_rain_033_input_formation_population_people

# STATE SCOPE. Moves current responsibility exactly once for one runtime
# control-change transaction. Required event targets are the runtime's
# `acid_rain_transfer_old_controller` and `acid_rain_transfer_new_controller`;
# the parent must call this effect before those chain-scoped targets disappear.
# Historical state deaths, action receipts, and recovery credit do not move.
acid_rain_033_achievement_transfer_state_responsibility = {
	if = {
		limit = {
			acid_rain_033_achievement_input_generation_is_current = yes
			has_state_flag = acid_rain_033_achievement_state_registered
			has_event_target = acid_rain_transfer_old_controller
			has_event_target = acid_rain_transfer_new_controller
			has_variable = acid_rain_033_achievement_state_responsible_country_id
		}
		save_event_target_as = acid_rain_033_achievement_transfer_state
		set_temp_variable = { acid_rain_033_achievement_old_controller_id = acid_rain_033_achievement_state_responsible_country_id }
		set_temp_variable = { acid_rain_033_achievement_new_controller_id = event_target:acid_rain_transfer_new_controller.id }
		if = {
			limit = {
				NOT = { check_variable = { var = acid_rain_033_achievement_old_controller_id value = acid_rain_033_achievement_new_controller_id compare = equals } }
			}
			if = {
				limit = { check_variable = { var = acid_rain_033_achievement_old_controller_id value = event_target:acid_rain_transfer_old_controller.id compare = equals } }
				event_target:acid_rain_transfer_old_controller = {
					remove_from_array = { array = acid_rain_033_achievement_responsible_states value = event_target:acid_rain_033_achievement_transfer_state }
					remove_from_array = { array = acid_rain_033_achievement_responsible_state_ids value = event_target:acid_rain_033_achievement_transfer_state.id }
				}
			}
			else = {
				set_state_flag = acid_rain_033_achievement_state_transfer_fault
				event_target:acid_rain_transfer_new_controller = { acid_rain_033_achievement_mark_migration_disqualified = yes }
			}
			event_target:acid_rain_transfer_new_controller = {
				acid_rain_033_achievement_initialize_country = yes
				if = {
					limit = { NOT = { is_in_array = { array = acid_rain_033_achievement_responsible_states value = event_target:acid_rain_033_achievement_transfer_state } } }
					add_to_array = { array = acid_rain_033_achievement_responsible_states value = event_target:acid_rain_033_achievement_transfer_state }
				}
				if = {
					limit = { NOT = { is_in_array = { array = acid_rain_033_achievement_responsible_state_ids value = event_target:acid_rain_033_achievement_transfer_state.id } } }
					add_to_array = { array = acid_rain_033_achievement_responsible_state_ids value = event_target:acid_rain_033_achievement_transfer_state.id }
				}
			}
			set_variable = { acid_rain_033_achievement_state_responsible_country_id = acid_rain_033_achievement_new_controller_id }
			set_variable = { acid_rain_033_achievement_state_responsible_country = event_target:acid_rain_transfer_new_controller }
			add_to_variable = { acid_rain_033_achievement_state_responsibility_transfer_count = @acid_rain_033_achievement_one }
			set_variable = { acid_rain_033_achievement_state_last_transfer_day = global.date }
			set_variable = { acid_rain_033_achievement_state_last_transfer_old_id = acid_rain_033_achievement_old_controller_id }
			set_variable = { acid_rain_033_achievement_state_last_transfer_new_id = acid_rain_033_achievement_new_controller_id }
			add_to_array = { array = acid_rain_033_achievement_state_controller_history_ids value = acid_rain_033_achievement_new_controller_id }
			add_to_array = { array = acid_rain_033_achievement_controller_history_ids value = acid_rain_033_achievement_state_id }
			add_to_array = { array = acid_rain_033_achievement_controller_history_days value = global.date }
			add_to_array = { array = acid_rain_033_achievement_controller_history_old_ids value = acid_rain_033_achievement_old_controller_id }
			add_to_array = { array = acid_rain_033_achievement_controller_history_new_ids value = acid_rain_033_achievement_new_controller_id }
			add_to_array = { array = acid_rain_033_achievement_controller_history_generation_entries value = acid_rain_033_input_generation }
		}
		clear_event_target = acid_rain_033_achievement_transfer_state
	}
	else = {
		if = { limit = { exists = yes } CONTROLLER = { acid_rain_033_achievement_mark_migration_disqualified = yes } }
		set_state_flag = acid_rain_033_achievement_state_transfer_fault
	}
}

# STATE SCOPE. Records one exact coverage or pulse receipt. Required inputs:
# generation, state_id, unique state/pulse receipt, exposure_class,
# episode_id (zero when not an episode), is_opening, and the exact applied
# population loss. Optional exposure_day_delta is deliberately not consumed;
# national exposure days have their own country receipt API.
acid_rain_033_achievement_record_state_exposure = {
	if = {
		limit = {
			acid_rain_033_achievement_input_generation_is_current = yes
			has_state_flag = acid_rain_033_achievement_state_registered
			has_variable = acid_rain_033_input_state_id
			has_variable = acid_rain_033_input_exposure_receipt
			has_variable = acid_rain_033_input_exposure_class
			has_variable = acid_rain_033_input_episode_id
			has_variable = acid_rain_033_input_is_opening
			has_variable = acid_rain_033_input_applied_deaths
			check_variable = { var = acid_rain_033_input_state_id value = acid_rain_033_achievement_state_id compare = equals }
			check_variable = { var = acid_rain_033_input_applied_deaths value = @acid_rain_033_achievement_zero compare = greater_than_or_equals }
		}
		if = {
			limit = { NOT = { is_in_array = { array = acid_rain_033_achievement_state_exposure_receipt_keys value = acid_rain_033_input_exposure_receipt } } }
			add_to_array = { array = acid_rain_033_achievement_state_exposure_receipt_keys value = acid_rain_033_input_exposure_receipt }
			add_to_variable = { acid_rain_033_achievement_state_exposure_count = @acid_rain_033_achievement_one }
			set_variable = { acid_rain_033_achievement_state_last_pulse_day = global.date }
			set_variable = { acid_rain_033_achievement_state_last_pulse_receipt = acid_rain_033_input_exposure_receipt }
			set_variable = { acid_rain_033_achievement_state_last_exposure_class = acid_rain_033_input_exposure_class }
			save_event_target_as = acid_rain_033_achievement_exposure_state
			if = {
				limit = { NOT = { has_state_flag = acid_rain_033_achievement_state_touched } }
				set_state_flag = acid_rain_033_achievement_state_touched
				set_variable = { acid_rain_033_achievement_state_first_exposure_day = global.date }
			}
			if = {
				limit = { check_variable = { var = acid_rain_033_input_is_opening value = @acid_rain_033_achievement_one compare = equals } }
				if = {
					limit = { NOT = { has_state_flag = acid_rain_033_achievement_state_opening_shock } }
					set_state_flag = acid_rain_033_achievement_state_opening_shock
					set_variable = { acid_rain_033_achievement_state_first_opening_shock_day = global.date }
				}
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = acid_rain_033_input_exposure_class value = @acid_rain_033_achievement_two compare = equals }
						check_variable = { var = acid_rain_033_input_exposure_class value = @acid_rain_033_achievement_four compare = equals }
					}
				}
				if = {
					limit = { NOT = { has_state_flag = acid_rain_033_achievement_state_severe_pulse } }
					set_state_flag = acid_rain_033_achievement_state_severe_pulse
					set_variable = { acid_rain_033_achievement_state_first_severe_pulse_day = global.date }
				}
			}
			if = {
				limit = { check_variable = { var = acid_rain_033_input_episode_id value = @acid_rain_033_achievement_zero compare = greater_than } }
				set_variable = { acid_rain_033_achievement_state_severe_episode_id = acid_rain_033_input_episode_id }
			}
			set_variable = { acid_rain_033_achievement_state_last_applied_deaths = acid_rain_033_input_applied_deaths }
			add_to_variable = { acid_rain_033_achievement_state_deaths = acid_rain_033_input_applied_deaths }
			if = {
				limit = { check_variable = { var = acid_rain_033_input_applied_deaths value = @acid_rain_033_achievement_zero compare = greater_than } }
				CONTROLLER = {
					acid_rain_033_achievement_initialize_country = yes
					add_to_variable = { acid_rain_033_achievement_event_deaths = acid_rain_033_input_applied_deaths }
				}
			}
			CONTROLLER = {
				acid_rain_033_achievement_initialize_country = yes
				if = {
					limit = { NOT = { is_in_array = { array = acid_rain_033_achievement_touched_states value = event_target:acid_rain_033_achievement_exposure_state } } }
					add_to_array = { array = acid_rain_033_achievement_touched_states value = event_target:acid_rain_033_achievement_exposure_state }
					add_to_array = { array = acid_rain_033_achievement_touched_state_ids value = event_target:acid_rain_033_achievement_exposure_state.id }
					add_to_variable = { acid_rain_033_achievement_touched_state_count = @acid_rain_033_achievement_one }
				}
				if = {
					limit = { NOT = { has_variable = acid_rain_033_achievement_first_exposure_day } }
					set_variable = { acid_rain_033_achievement_first_exposure_day = global.date }
				}
				if = {
					limit = { check_variable = { var = acid_rain_033_input_is_opening value = @acid_rain_033_achievement_one compare = equals } }
					if = {
						limit = { NOT = { has_variable = acid_rain_033_achievement_first_opening_shock_day } }
						set_variable = { acid_rain_033_achievement_first_opening_shock_day = global.date }
						set_variable = { acid_rain_033_achievement_first_opening_shock_state_id = event_target:acid_rain_033_achievement_exposure_state.id }
						if = {
							limit = {
								OR = {
									NOT = { check_variable = { var = acid_rain_033_achievement_component_tier_entries^1 value = @acid_rain_033_achievement_four compare = greater_than_or_equals } }
									NOT = { check_variable = { var = acid_rain_033_achievement_component_tier_entries^2 value = @acid_rain_033_achievement_four compare = greater_than_or_equals } }
									NOT = { check_variable = { var = acid_rain_033_achievement_component_tier_entries^3 value = @acid_rain_033_achievement_four compare = greater_than_or_equals } }
									NOT = { check_variable = { var = acid_rain_033_achievement_component_tier_entries^4 value = @acid_rain_033_achievement_four compare = greater_than_or_equals } }
								}
							}
							set_country_flag = acid_rain_033_achievement_before_the_first_drop_disqualified
						}
					}
				}
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = acid_rain_033_input_exposure_class value = @acid_rain_033_achievement_two compare = equals }
						check_variable = { var = acid_rain_033_input_exposure_class value = @acid_rain_033_achievement_four compare = equals }
					}
					check_variable = { var = acid_rain_033_input_episode_id value = @acid_rain_033_achievement_zero compare = greater_than }
				}
				set_temp_variable = { acid_rain_033_input_severe_pulse_day = global.date }
				acid_rain_033_achievement_record_severe_pulse = yes

# STATE SCOPE. Records one severe pulse against a previously frozen target
# row. Required inputs: generation, state_id, episode_id, exposure_receipt_key,
# severe_pulse_day, and exact applied deaths. The same exposure receipt used by
# record_state_exposure is the idempotence key; it is never counted twice.
acid_rain_033_achievement_record_severe_pulse = {
	if = {
		limit = {
			acid_rain_033_achievement_input_generation_is_current = yes
			has_state_flag = acid_rain_033_achievement_state_registered
			has_variable = acid_rain_033_input_state_id
			has_variable = acid_rain_033_input_severe_episode_id
			has_variable = acid_rain_033_input_exposure_receipt
			has_variable = acid_rain_033_input_severe_pulse_day
			has_variable = acid_rain_033_input_applied_deaths
			check_variable = { var = acid_rain_033_input_state_id value = acid_rain_033_achievement_state_id compare = equals }
			check_variable = { var = acid_rain_033_input_severe_episode_id value = acid_rain_033_achievement_state_severe_episode_id compare = equals }
			check_variable = { var = acid_rain_033_input_severe_pulse_day value = @acid_rain_033_achievement_zero compare = greater_than }
			check_variable = { var = acid_rain_033_input_applied_deaths value = @acid_rain_033_achievement_zero compare = greater_than_or_equals }
			NOT = { is_in_array = { array = acid_rain_033_achievement_state_severe_pulse_receipt_keys value = acid_rain_033_input_exposure_receipt } }
		}
		add_to_array = { array = acid_rain_033_achievement_state_severe_pulse_receipt_keys value = acid_rain_033_input_exposure_receipt }
		add_to_variable = { acid_rain_033_achievement_state_severe_pulse_count = @acid_rain_033_achievement_one }
		if = {
			limit = { check_variable = { var = acid_rain_033_input_applied_deaths value = @acid_rain_033_achievement_zero compare = greater_than } }
			add_to_variable = { acid_rain_033_achievement_state_severe_deaths = acid_rain_033_input_applied_deaths }
		}
		save_event_target_as = acid_rain_033_achievement_severe_pulse_state
		if = {
			limit = { NOT = { has_state_flag = acid_rain_033_achievement_state_severe_pulse } }
			set_state_flag = acid_rain_033_achievement_state_severe_pulse
			set_variable = { acid_rain_033_achievement_state_first_severe_pulse_day = acid_rain_033_input_severe_pulse_day }
		}
		CONTROLLER = {
			set_temp_variable = { acid_rain_033_achievement_severe_pulse_row = @acid_rain_033_achievement_zero }
			if = {
				limit = { has_variable = acid_rain_033_achievement_severe_target_row_count }
				for_loop_effect = {
					start = @acid_rain_033_achievement_one
					end = acid_rain_033_achievement_severe_target_row_count
					value = acid_rain_033_achievement_severe_index
					if = {
						limit = {
							check_variable = { var = acid_rain_033_achievement_severe_target_state_ids^acid_rain_033_achievement_severe_index value = acid_rain_033_input_state_id compare = equals }
							check_variable = { var = acid_rain_033_achievement_severe_target_episode_ids^acid_rain_033_achievement_severe_index value = acid_rain_033_input_severe_episode_id compare = equals }
							check_variable = { var = acid_rain_033_achievement_severe_target_generation_entries^acid_rain_033_achievement_severe_index value = acid_rain_033_input_generation compare = equals }
						}
						set_temp_variable = { acid_rain_033_achievement_severe_pulse_row = acid_rain_033_achievement_severe_index }
					}
				}
			}
			if = {
				limit = { check_variable = { var = acid_rain_033_achievement_severe_pulse_row value = @acid_rain_033_achievement_zero compare = greater_than } }
				set_temp_variable = { acid_rain_033_achievement_severe_previous_deaths = acid_rain_033_achievement_severe_target_death_entries^acid_rain_033_achievement_severe_pulse_row }
				add_to_temp_variable = { acid_rain_033_achievement_severe_previous_deaths = acid_rain_033_input_applied_deaths }
				set_variable = { acid_rain_033_achievement_severe_target_death_entries^acid_rain_033_achievement_severe_pulse_row = acid_rain_033_achievement_severe_previous_deaths }
				set_temp_variable = { acid_rain_033_achievement_severe_previous_pulses = acid_rain_033_achievement_severe_target_pulse_entries^acid_rain_033_achievement_severe_pulse_row }
				add_to_temp_variable = { acid_rain_033_achievement_severe_previous_pulses = @acid_rain_033_achievement_one }
				set_variable = { acid_rain_033_achievement_severe_target_pulse_entries^acid_rain_033_achievement_severe_pulse_row = acid_rain_033_achievement_severe_previous_pulses }
				set_variable = { acid_rain_033_achievement_severe_target_first_pulse_entries^acid_rain_033_achievement_severe_pulse_row = acid_rain_033_input_severe_pulse_day }
			}
			else = {
				acid_rain_033_achievement_mark_migration_disqualified = yes
			}
		}
		clear_event_target = acid_rain_033_achievement_severe_pulse_state
	}
	else = {
		CONTROLLER = { acid_rain_033_achievement_mark_migration_disqualified = yes }
	}
}

							}
						}
						add_to_temp_variable = { acid_rain_033_achievement_episode_protection = @acid_rain_033_achievement_one }
					}
				}
			}
		}
		set_temp_variable = { acid_rain_033_achievement_episode_row = @acid_rain_033_achievement_zero }
		if = {
			limit = { has_variable = acid_rain_033_achievement_severe_episode_row_count }
			for_loop_effect = {
				start = @acid_rain_033_achievement_one
				end = acid_rain_033_achievement_severe_episode_row_count
				value = acid_rain_033_achievement_episode_summary_index
				if = { limit = { check_variable = { var = acid_rain_033_achievement_severe_episode_ids^acid_rain_033_achievement_episode_summary_index value = acid_rain_033_input_severe_episode_id compare = equals } } set_temp_variable = { acid_rain_033_achievement_episode_row = acid_rain_033_achievement_episode_summary_index } }
			}
		}
		if = {
			limit = { check_variable = { var = acid_rain_033_achievement_episode_row value = @acid_rain_033_achievement_zero compare = equals } }
			if = {
				limit = {
					OR = {
						NOT = { has_variable = acid_rain_033_achievement_severe_episode_row_count }
						check_variable = { var = acid_rain_033_achievement_severe_episode_row_count value = @acid_rain_033_achievement_max_episode_rows compare = less_than }
					}
				}
				add_to_array = { array = acid_rain_033_achievement_severe_episode_ids value = acid_rain_033_input_severe_episode_id }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_state_counts value = acid_rain_033_achievement_episode_state_count }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_population_entries value = acid_rain_033_achievement_episode_population }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_death_entries value = acid_rain_033_achievement_episode_deaths }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_pulse_entries value = acid_rain_033_achievement_episode_pulses }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_protection_entries value = acid_rain_033_achievement_episode_protection }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_generation_entries value = acid_rain_033_input_generation }
				add_to_array = { array = acid_rain_033_achievement_severe_episode_complete_entries value = @acid_rain_033_achievement_zero }

				acid_rain_033_achievement_mark_migration_disqualified = yes

			set_variable = { acid_rain_033_achievement_severe_episode_state_counts^acid_rain_033_achievement_episode_row = acid_rain_033_achievement_episode_state_count }
			set_variable = { acid_rain_033_achievement_severe_episode_population_entries^acid_rain_033_achievement_episode_row = acid_rain_033_achievement_episode_population }
			set_variable = { acid_rain_033_achievement_severe_episode_death_entries^acid_rain_033_achievement_episode_row = acid_rain_033_achievement_episode_deaths }
			set_variable = { acid_rain_033_achievement_severe_episode_pulse_entries^acid_rain_033_achievement_episode_row = acid_rain_033_achievement_episode_pulses }
			set_variable = { acid_rain_033_achievement_severe_episode_protection_entries^acid_rain_033_achievement_episode_row = acid_rain_033_achievement_episode_protection }
		}
		if = {
			limit = {
				check_variable = { var = acid_rain_033_input_severe_episode_closed_valid value = @acid_rain_033_achievement_one compare = equals }
				check_variable = { var = acid_rain_033_achievement_episode_state_count value = @acid_rain_033_achievement_shelter_states compare = greater_than_or_equals }
				check_variable = { var = acid_rain_033_achievement_episode_pulses value = @acid_rain_033_achievement_shelter_pulses compare = greater_than_or_equals }
				check_variable = { var = acid_rain_033_achievement_episode_population value = @acid_rain_033_achievement_zero compare = greater_than }
			}
			if = {
				limit = { check_variable = { var = acid_rain_033_achievement_episode_protection value = @acid_rain_033_achievement_zero compare = equals } }
				set_country_flag = acid_rain_033_achievement_shelters_held_disqualified
			}
			else = {
				set_temp_variable = { acid_rain_033_achievement_episode_deaths_scaled = acid_rain_033_achievement_episode_deaths }
				multiply_temp_variable = { acid_rain_033_achievement_episode_deaths_scaled = @acid_rain_033_achievement_shelter_death_denominator }
				set_temp_variable = { acid_rain_033_achievement_episode_population_scaled = acid_rain_033_achievement_episode_population }
				multiply_temp_variable = { acid_rain_033_achievement_episode_population_scaled = @acid_rain_033_achievement_shelter_death_numerator }
				if = {
					limit = { check_variable = { var = acid_rain_033_achievement_episode_deaths_scaled value = acid_rain_033_achievement_episode_population_scaled compare = less_than } }
					if = {
						limit = { NOT = { check_variable = { var = acid_rain_033_achievement_severe_episode_complete_entries^acid_rain_033_achievement_episode_row value = @acid_rain_033_achievement_one compare = equals } } }
						set_variable = { acid_rain_033_achievement_severe_episode_complete_entries^acid_rain_033_achievement_episode_row = @acid_rain_033_achievement_one }
						add_to_variable = { acid_rain_033_achievement_shelter_qualifying_episode_count = @acid_rain_033_achievement_one }
					}
				}
			}
		}
	}
	else = {
		acid_rain_033_achievement_mark_migration_disqualified = yes
	}
}

# generation, state_id, forecast_id, episode_id, start_receipt_key,
