# ============================================================================
# CHAOS REDUX - EVENT 021 RANDOM CIVIL WAR PARENT EXECUTION
# ============================================================================
# The core Event 021 file owns reusable pressure, authority, reservations,
# capacity, recurrence, and cleanup primitives. This file owns the parent
# transaction: selecting a target, planning connected states, materialising a
# bounded claimant, adapting any complete registered Event 006 package, publishing
# fronts and logs, and resolving the resulting crisis.
#
# Scope contract:
# - country effects are called from the target country unless a comment says
#   otherwise;
# - state effects are called through `random_civil_war_current_state`;
# - global scheduler effects run only from the existing Chaos host pulse and
#   consume bounded registry/queue budgets;
# - no effect here creates a non-human country, a duplicate Event 006 identity,
#   or an unrestricted world iteration.
# ============================================================================

# COUNTRY scope on a candidate host. The Event 006 package registry remains
# authoritative: this bounded loop loads only the 32 source-attested package
# IDs, checks the loader's exact anchor and dormant-carrier preflight, and
# stores one admitted package for the current Event 021 transaction. A missing
# carrier tag is allowed only when the package registry can safely release it.
event021_parent_find_event6_package = {
	clr_country_flag = random_civil_war_event6_dormant_route_ready
	clear_variable = random_civil_war_event6_package_id
	clear_variable = random_civil_war_event6_anchor_state
	clear_variable = random_civil_war_event6_candidate_region
	clear_variable = random_civil_war_event6_candidate_depth
	clear_variable = random_civil_war_event6_candidate_archetype
	clear_event_target = random_civil_war_event6_selected_actor
	clear_event_target = random_civil_war_event6_selected_anchor
	set_temp_variable = { event021_parent_event6_package_found = constant:random_civil_war_value.zero }
	for_each_loop = {
		array = global.independence_wave_event021_complete_package_ids
		value = event021_parent_event6_candidate_package_id
		if = {
			limit = {
				check_variable = {
					var = event021_parent_event6_package_found
					value = constant:random_civil_war_value.zero
					compare = equals
				}
			}
			set_temp_variable = { independence_wave_execution_package_id = event021_parent_event6_candidate_package_id }
			set_temp_variable = { independence_wave_scenario_dispatch_package_id = event021_parent_event6_candidate_package_id }
			clear_event_target = liberation_candidate_country
			clear_event_target = liberation_candidate_anchor
			clear_event_target = liberation_candidate_primary_host
			independence_wave_scenario_load_dispatch_package = yes
			if = {
				limit = {
					has_event_target = liberation_candidate_anchor
					event_target:liberation_candidate_anchor = {
						is_owned_by = ROOT
						is_controlled_by = ROOT
						state_population_k > constant:event021_parent_tuning.minimum_state_population_k
						check_variable = { infrastructure_level > constant:random_civil_war_value.zero }
						OR = {
							NOT = { has_global_flag = random_civil_war_event6_secondary_search_active }
							AND = {
								NOT = { is_capital = yes }
								NOT = { is_in_array = { array = global.random_civil_war_reserved_states value = THIS } }
								NOT = { is_in_array = { array = global.random_civil_war_front_states value = THIS } }
							}
						}
					}
					OR = {
						NOT = { has_event_target = liberation_candidate_country }
						event_target:liberation_candidate_country = {
							is_independence_wave_runtime_package_preflight_ready = yes
							NOT = { tag = ROOT }
						}
					}
				}
				set_temp_variable = { event021_parent_event6_package_found = constant:random_civil_war_value.one }
				set_variable = { random_civil_war_event6_package_id = event021_parent_event6_candidate_package_id }
				set_variable = { random_civil_war_event6_candidate_region = independence_wave_candidate_region }
				set_variable = { random_civil_war_event6_candidate_depth = independence_wave_candidate_depth }
				set_variable = { random_civil_war_event6_candidate_archetype = independence_wave_candidate_archetype }
				event_target:liberation_candidate_anchor = {
					set_variable = { ROOT.random_civil_war_event6_anchor_state = THIS }
					save_event_target_as = random_civil_war_event6_selected_anchor
				}
				if = {
					limit = { has_event_target = liberation_candidate_country }
					event_target:liberation_candidate_country = { save_event_target_as = random_civil_war_event6_selected_actor }
				}
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = event021_parent_event6_package_found
				value = constant:random_civil_war_value.one
				compare = equals
			}
		}
		set_country_flag = random_civil_war_event6_dormant_route_ready
	}
	else = {
		clear_variable = random_civil_war_event6_package_id
		clear_variable = random_civil_war_event6_anchor_state
		clear_variable = random_civil_war_event6_candidate_region
		clear_variable = random_civil_war_event6_candidate_depth
		clear_variable = random_civil_war_event6_candidate_archetype
		clear_event_target = random_civil_war_event6_selected_actor
		clear_event_target = random_civil_war_event6_selected_anchor
	}
}

# ANY scope. Records one bounded Wars-cluster or scheduler skip. The caller
# supplies `event021_cluster_skip_reason` as a normal or temporary value from
# `random_civil_war_cluster_skip_reason`; missing input fails closed to the
# no-eligible-target reason. The receipt is intentionally compact so the
# shared Event Details view can explain why a member was not launched.
event021_parent_record_cluster_skip = {
	if = {
		limit = { NOT = { has_variable = event021_cluster_skip_reason } }
		set_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.no_eligible_target }
	}
	set_variable = { global.random_civil_war_cluster_skip_reason = event021_cluster_skip_reason }
	set_variable = { global.random_civil_war_cluster_skip_date = global.date }
	if = {
		limit = { NOT = { has_variable = global.random_civil_war_cluster_skip_count } }
		set_variable = { global.random_civil_war_cluster_skip_count = constant:random_civil_war_value.zero }
	}
	add_to_variable = { global.random_civil_war_cluster_skip_count = constant:random_civil_war_value.one }
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.no_eligible_actor compare = equals } }
		add_to_variable = { global.random_civil_war_skip_no_eligible_actor_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.no_eligible_target compare = equals } }
		add_to_variable = { global.random_civil_war_skip_no_eligible_target_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.role_collision compare = equals } }
		add_to_variable = { global.random_civil_war_skip_role_collision_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.tag_occupied compare = equals } }
		add_to_variable = { global.random_civil_war_skip_tag_occupied_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.state_reservation_failed compare = equals } }
		add_to_variable = { global.random_civil_war_skip_state_reservation_failed_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.cap_reached compare = equals } }
		add_to_variable = { global.random_civil_war_skip_cap_reached_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.generation_blocked compare = equals } }
		add_to_variable = { global.random_civil_war_skip_generation_blocked_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { check_variable = { var = event021_cluster_skip_reason value = constant:random_civil_war_cluster_skip_reason.stale_delayed_reservation compare = equals } }
		add_to_variable = { global.random_civil_war_skip_stale_delayed_reservation_count = constant:random_civil_war_value.one }
	}
	if = {
		limit = { has_country_flag = event_cluster_member_fire_context }
		set_variable = { random_civil_war_last_cluster_skip_reason = event021_cluster_skip_reason }
		set_variable = { random_civil_war_last_cluster_skip_date = global.date }
	}
}

# COUNTRY scope on a Wars-cluster reservation which could not be consumed.
# The first matching reason is retained so a collision is not flattened into
# a generic no-target receipt.
event021_parent_classify_cluster_reservation_skip = {
	set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.no_eligible_target }
	set_temp_variable = { event021_cluster_reservation_expiry = global.date }
	if = {
		limit = { has_variable = random_civil_war_cluster_reservation_date }
		set_temp_variable = { event021_cluster_reservation_expiry = random_civil_war_cluster_reservation_date }
		add_to_temp_variable = { event021_cluster_reservation_expiry = constant:event021_parent_tuning.cluster_reservation_valid_days }
	}
	if = {
		limit = {
			has_variable = random_civil_war_cluster_reservation_date
			check_variable = { var = global.date value = event021_cluster_reservation_expiry compare = greater_than }
		}
		set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.stale_delayed_reservation }
	}
	else_if = {
		limit = {
			OR = {
				has_country_flag = random_civil_war_active
				has_country_flag = random_civil_war_target_reserved
			}
		}
		set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.role_collision }
	}
	else_if = {
		limit = { NOT = { random_civil_war_is_normal_human_country = yes } }
		set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.no_eligible_actor }
	}
	else_if = {
		limit = { NOT = { random_civil_war_generation_allowed = yes } }
		set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.generation_blocked }
	}
	else_if = {
		limit = { NOT = { event021_parent_global_capacity_ready = yes } }
		set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.cap_reached }
	}
	else_if = {
		limit = {
			has_variable = random_civil_war_archetype
			check_variable = { var = random_civil_war_archetype value = constant:random_civil_war_archetype.event006 compare = equals }
			NOT = { event021_parent_event6_dormant_candidate = yes }
		}
		set_temp_variable = { event021_cluster_skip_reason = constant:random_civil_war_cluster_skip_reason.tag_occupied }
	}
	event021_parent_record_cluster_skip = yes
}

# COUNTRY scope. Records one manual Fracture Cascade attempt after the same
# immediate transaction used by automatic play. The scenario controller sets
# requested and eligible counts; this helper only records a committed actor so
# the final skipped count can be derived exactly as requested minus committed.
event021_parent_record_scenario_result = {
	if = {
		limit = { has_country_flag = random_civil_war_active }
		if = {
			limit = { NOT = { has_variable = global.random_civil_war_scenario_committed_count } }
			set_variable = { global.random_civil_war_scenario_committed_count = constant:random_civil_war_value.zero }
		}
		add_to_variable = { global.random_civil_war_scenario_committed_count = constant:random_civil_war_value.one }
	}
	else = {
		set_variable = { global.random_civil_war_scenario_last_skip_reason = constant:random_civil_war_cluster_skip_reason.no_eligible_target }
		if = {
			limit = { has_country_flag = random_civil_war_plan_rejected }
			set_variable = { global.random_civil_war_scenario_last_skip_reason = constant:random_civil_war_cluster_skip_reason.state_reservation_failed }
		}
	}
}

# COUNTRY scope. Runs from the existing Chaos global-host pulse. One pulse
# opens one review budget, visits only the registered array, and launches only
# the bounded critical queue. The nested loops are capped by constants and do
# not iterate every country in the world.
event021_parent_global_scheduler_pulse = {
	if = {
		limit = {
			has_country_flag = is_global_host
			NOT = { has_global_flag = world_end }
			OR = {
				has_global_flag = random_civil_war_rework_ready
				has_global_flag = random_civil_war_has_committed_opening
			}
			OR = {
				NOT = { has_variable = global.random_civil_war_last_scheduler_date }
				check_variable = { var = global.date value = global.random_civil_war_last_scheduler_date compare = greater_than }
			}
		}
		set_variable = { global.random_civil_war_last_scheduler_date = global.date }
		event021_parent_initialize_global_runtime = yes
		event021_sync_evolution_state = yes
		event021_refresh_global_capacity = yes
		event021_parent_process_evolution_schedule = yes
		# Evolution III owns broad Global Fracture registration. Earlier stages
		# retain only countries explicitly registered by their live crisis paths.
		if = {
			limit = { has_global_flag = random_civil_war_evolution_iii_active }
			# Add only a bounded sample each host pulse. Registered countries retain
			# their own review date, so this remains a persistent queue rather than a
			# hidden every-country daily loop.
			for_loop_effect = {
				start = constant:random_civil_war_value.zero
				end = constant:event021_parent_tuning.scheduler_registration_samples
				value = event021_parent_registration_sample_index
				random_country = {
					limit = {
						random_civil_war_is_normal_human_country = yes
						NOT = { has_country_flag = random_civil_war_global_registered }
					}
					event021_register_global_country = yes
				}
			}
		}
		if = {
			limit = { NOT = { has_global_flag = random_civil_war_review_batch_active } }
			event021_global_review_batch = yes
		}
		set_temp_variable = { event021_parent_review_scanned = constant:random_civil_war_value.zero }
		set_temp_variable = { event021_parent_review_skipped = global.random_civil_war_review_cursor }
		if = {
			limit = {
				check_variable = {
					var = global.random_civil_war_registered_countries^num
					value = constant:random_civil_war_value.zero
					compare = greater_than
				}
				check_variable = {
					var = global.random_civil_war_review_cursor
					value = global.random_civil_war_registered_countries^num
					compare = greater_than_or_equals
				}
			}
			set_variable = { global.random_civil_war_review_cursor = constant:event021_scheduler.first_array_index }
			set_temp_variable = { event021_parent_review_skipped = constant:event021_scheduler.first_array_index }
		}
		for_each_scope_loop = {
			array = global.random_civil_war_registered_countries
			if = {
				limit = {
					check_variable = {
						var = event021_parent_review_scanned
						value = constant:event021_scheduler.maximum_review_scan
						compare = less_than
					}
				}
				if = {
					limit = {
						check_variable = {
							var = event021_parent_review_skipped
							value = constant:random_civil_war_value.zero
							compare = greater_than
						}
					}
					subtract_from_temp_variable = { event021_parent_review_skipped = constant:random_civil_war_value.one }
				}
				else = {
					add_to_temp_variable = { event021_parent_review_scanned = constant:random_civil_war_value.one }
					country_event = { id = chaosx.nr21.14 }
				}
			}
		}
		add_to_variable = { global.random_civil_war_review_cursor = event021_parent_review_scanned }
		if = {
			limit = {
				check_variable = {
					var = global.random_civil_war_review_cursor
					value = global.random_civil_war_registered_countries^num
					compare = greater_than_or_equals
				}
			}
			set_variable = { global.random_civil_war_review_cursor = constant:event021_scheduler.first_array_index }
		}
		set_temp_variable = { event021_parent_queue_scanned = constant:random_civil_war_value.zero }
		set_temp_variable = { event021_parent_queue_skipped = global.random_civil_war_critical_cursor }
		if = {
			limit = {
				check_variable = {
					var = global.random_civil_war_critical_queue^num
					value = constant:random_civil_war_value.zero
					compare = greater_than
				}
				check_variable = {
					var = global.random_civil_war_critical_cursor
					value = global.random_civil_war_critical_queue^num
					compare = greater_than_or_equals
				}
			}
			set_variable = { global.random_civil_war_critical_cursor = constant:event021_scheduler.first_array_index }
			set_temp_variable = { event021_parent_queue_skipped = constant:event021_scheduler.first_array_index }
		}
		for_each_scope_loop = {
			array = global.random_civil_war_critical_queue
			if = {
				limit = {
					check_variable = {
						var = event021_parent_queue_scanned
						value = constant:event021_scheduler.maximum_critical_scan
						compare = less_than
					}
				}

# STATE scope. Scores an anchor using live population, factories, logistics,
# port access, coastal access, infrastructure, and core status. The anchor is
# a potential actor capital; the parent remnant's capital is protected by the
# opening-plan validator.
event021_parent_score_anchor_state = {
	set_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.base }
	if = { limit = { state_population_k > constant:event021_anchor_weight.population_dense_k } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.population_dense } }
	else_if = { limit = { state_population_k > constant:event021_anchor_weight.population_medium_k } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.population_medium } }
	if = { limit = { check_variable = { building_level@industrial_complex > constant:random_civil_war_value.zero } } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.civilian_factory } }
	if = { limit = { check_variable = { building_level@arms_factory > constant:random_civil_war_value.zero } } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.military_factory } }
	if = { limit = { infrastructure > constant:event021_anchor_weight.infrastructure_good } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.infrastructure } }
	if = { limit = { supply_node > constant:random_civil_war_value.zero } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.supply_node } }
	if = { limit = { naval_base > constant:random_civil_war_value.zero } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.naval_base } }
	if = { limit = { is_coastal = yes } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.coastal } }
	if = { limit = { is_core_of = event_target:random_civil_war_host_country } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.core } }
	if = { limit = { is_capital = yes } add_to_temp_variable = { event021_anchor_state_score = constant:event021_anchor_weight.capital } }
	clamp_temp_variable = { var = event021_anchor_state_score min = constant:random_civil_war_value.one max = constant:event021_anchor_weight.maximum }
	round_temp_variable = event021_anchor_state_score
}

# COUNTRY scope. Selects the highest-scoring viable anchor from aligned state
# and score arrays, then reserves it. This keeps route selection deterministic
# at the state layer while target and archetype selection remain weighted.
event021_parent_select_connected_anchor = {
	clear_temp_array = event021_parent_anchor_state_pool
	clear_temp_array = event021_parent_anchor_score_pool
	every_owned_state = {
		limit = { event021_parent_viable_anchor_state = yes }
		event021_parent_score_anchor_state = yes
		add_to_temp_array = { event021_parent_anchor_state_pool value = THIS }
		add_to_temp_array = { event021_parent_anchor_score_pool value = event021_anchor_state_score }
	}
	if = {
		limit = { check_variable = { var = event021_parent_anchor_state_pool^num value = constant:random_civil_war_value.zero compare = greater_than } }
		find_highest_in_array = {
			array = event021_parent_anchor_score_pool
			value = event021_parent_best_anchor_score
			index = event021_parent_best_anchor_index
		}
		set_temp_variable = { event021_parent_anchor_iteration = constant:random_civil_war_value.zero }
		set_temp_variable = { event021_parent_anchor_selected = constant:random_civil_war_value.zero }
		for_each_scope_loop = {
			array = event021_parent_anchor_state_pool
			if = {
				limit = {
					check_variable = { var = event021_parent_anchor_selected value = constant:random_civil_war_value.zero compare = equals }
					check_variable = { var = event021_parent_anchor_iteration value = event021_parent_best_anchor_index compare = equals }
				}
				set_variable = { ROOT.random_civil_war_anchor_state = THIS }
				set_variable = { ROOT.random_civil_war_anchor_state_id = id }
				set_variable = { ROOT.random_civil_war_current_state = THIS }
				set_variable = { ROOT.random_civil_war_rail_mission_state = THIS }
				set_variable = { ROOT.random_civil_war_rail_mission_state_id = id }
				save_event_target_as = random_civil_war_current_state
				set_temp_variable = { event021_parent_anchor_selected = constant:random_civil_war_value.one }
			}
			add_to_temp_variable = { event021_parent_anchor_iteration = constant:random_civil_war_value.one }
		}
	}
	clear_temp_array = event021_parent_anchor_state_pool
	clear_temp_array = event021_parent_anchor_score_pool
}

# COUNTRY scope. Expands the reserved seed through connected owned and
# controlled neighbors in bounded closure passes. The state cap is shared with
# the front cap, and every candidate is checked again by the reservation
# trigger, so no arbitrary half-country split or disconnected island appears.
event021_parent_expand_connected_region = {
	for_loop_effect = {
		start = constant:random_civil_war_value.zero
		end = constant:random_civil_war_capacity.maximum_multifront_states
		value = event021_parent_region_pass
		if = {
			limit = {
				check_variable = {
					var = global.random_civil_war_plan_state_count
					value = constant:random_civil_war_capacity.maximum_multifront_states
					compare = less_than
				}
			}
			every_owned_state = {
				limit = {
					event021_parent_viable_anchor_state = yes
					NOT = { is_in_array = { array = global.random_civil_war_plan_states value = THIS } }
					NOT = { is_in_array = { array = global.random_civil_war_reserved_states value = THIS } }
					any_neighbor_state = { is_in_array = { array = global.random_civil_war_plan_states value = THIS } }
			}
			set_variable = { ROOT.random_civil_war_current_state = THIS }
				save_event_target_as = random_civil_war_current_state
			ROOT = { event021_reserve_state = yes }
			}
		}
	}
}

# COUNTRY scope on the original host. Allocates two distinct front rows and a
# per-country generation value before the first ownership or war mutation.
# The IDs are monotonically allocated from one global counter so a failed
# transaction cannot collide with a later live front.
event021_parent_prepare_crisis_identity = {
	clr_country_flag = random_civil_war_event6_identity_at_opening
	clr_country_flag = random_civil_war_nested_crisis_candidate
	if = {
		limit = {
			random_civil_war_is_normal_human_country = yes
			OR = {
				is_independence_wave_active_country = yes
				has_country_flag = random_civil_war_event6_origin_recorded
			}
		}
		set_country_flag = random_civil_war_event6_identity_at_opening
	}
	# This is the single pre-commit capture point. Later route refreshes cannot
	# manufacture or erase the opening-time external-war achievement receipt.
	clr_country_flag = random_civil_war_external_war_at_opening
	if = {
		limit = {
			has_war = yes
			OR = { is_major = yes any_enemy_country = { is_major = yes } }
		}
		set_country_flag = random_civil_war_external_war_at_opening
	}
	if = {
		limit = { NOT = { has_variable = global.random_civil_war_next_front_id } }
		set_variable = { global.random_civil_war_next_front_id = constant:event021_parent_tuning.front_id_seed }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_generation } }
		set_variable = { random_civil_war_generation = constant:random_civil_war_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_generation_before_plan } }
		set_variable = { random_civil_war_generation_before_plan = random_civil_war_generation }
	}
	if = {
		limit = {
			NOT = { has_country_flag = random_civil_war_generation_seeded }
			OR = {
				has_country_flag = random_civil_war_event6_identity_at_opening
				has_variable = random_civil_war_last_cleanup_date
			}
			random_civil_war_generation_allowed = yes
		}
		add_to_variable = { random_civil_war_generation = constant:random_civil_war_value.one }
	}
	if = {
		limit = {
			has_country_flag = random_civil_war_event6_identity_at_opening
			check_variable = {
				var = random_civil_war_generation
				value = constant:event021_parent_tuning.nested_generation_minimum
				compare = greater_than_or_equals
			}
		}
		set_country_flag = random_civil_war_nested_crisis_candidate
	}
	set_country_flag = random_civil_war_generation_seeded
	set_variable = { random_civil_war_belligerent_cap = global.random_civil_war_plan_state_count }
	add_to_variable = { random_civil_war_belligerent_cap = constant:random_civil_war_value.one }
	if = {
		limit = {
			is_major = yes
			OR = {
				check_variable = { var = random_civil_war_opening_severity value = constant:random_civil_war_severity.severe compare = equals }
				check_variable = { var = random_civil_war_opening_severity value = constant:random_civil_war_severity.critical compare = equals }
			}
		}
		clamp_variable = {
			var = random_civil_war_belligerent_cap
			min = constant:event021_parent_tuning.minimum_belligerents
			max = constant:event021_parent_tuning.maximum_major_belligerents
		}
	}
	else = {
		clamp_variable = {
			var = random_civil_war_belligerent_cap
			min = constant:event021_parent_tuning.minimum_belligerents
			max = constant:event021_parent_tuning.maximum_minor_belligerents
		}
	}
	every_owned_state = { clr_state_flag = random_civil_war_opening_capital_state }
	capital_scope = {
		set_state_flag = random_civil_war_opening_capital_state
		save_event_target_as = random_civil_war_history_capital
	}
	set_variable = { random_civil_war_original_capital = event_target:random_civil_war_history_capital }
	clr_country_flag = random_civil_war_player_at_opening
	clr_country_flag = random_civil_war_original_player_government
	if = {
		limit = { is_ai = no }
		set_country_flag = random_civil_war_player_at_opening
		set_country_flag = random_civil_war_original_player_government
	}
	set_variable = { global.random_civil_war_pending_generation = random_civil_war_generation }
	set_variable = { global.random_civil_war_pending_host_front_id = global.random_civil_war_next_front_id }
	add_to_variable = { global.random_civil_war_next_front_id = constant:random_civil_war_value.one }
	set_variable = { global.random_civil_war_pending_opposition_front_id = global.random_civil_war_next_front_id }
	add_to_variable = { global.random_civil_war_next_front_id = constant:random_civil_war_value.one }
	set_variable = { global.random_civil_war_pending_crisis_id = global.random_civil_war_pending_host_front_id }
	set_variable = { global.random_civil_war_pending_archetype = random_civil_war_archetype }
	set_variable = { global.random_civil_war_pending_severity = random_civil_war_opening_severity }

# COUNTRY scope on a viable claimant source. Each pass first obtains one
# distinct evidence-backed route from the original host, then detaches one
# bounded state and rechecks the source and crisis caps after the child has
# initialized. Failure leaves every committed front intact and creates no
# placeholder actor or route reservation.
event021_parent_try_secondary_front = {
	if = {
		limit = { event021_parent_secondary_front_candidate = yes }
		set_country_flag = random_civil_war_secondary_front_attempted
		clear_global_event_target = random_civil_war_secondary_anchor_state
		clear_global_event_target = random_civil_war_secondary_source_country
		var:random_civil_war_crisis_host_scope = { event021_parent_prepare_secondary_route = yes }
		if = {
			limit = {
				has_global_flag = random_civil_war_secondary_route_prepared
				num_of_controlled_states > constant:random_civil_war_value.one
				NOT = { has_capitulated = yes }
			}
			random_owned_state = {
				limit = { event021_parent_secondary_state_candidate = yes }
				set_state_flag = random_civil_war_secondary_transfer_state
				save_global_event_target_as = random_civil_war_secondary_anchor_state
				set_variable = { global.random_civil_war_pending_secondary_anchor_state_id = id }
			}
		}
		if = {
			limit = {
				has_global_flag = random_civil_war_secondary_route_prepared
				has_event_target = random_civil_war_secondary_anchor_state
			}
			set_variable = { global.random_civil_war_pending_secondary_front_id = global.random_civil_war_next_front_id }
			add_to_variable = { global.random_civil_war_next_front_id = constant:random_civil_war_value.one }
			save_global_event_target_as = random_civil_war_secondary_source_country
			set_global_flag = random_civil_war_secondary_front_pending
			event021_parent_start_secondary_civil_war = yes
		}
		else = {
			set_country_flag = random_civil_war_secondary_front_rejected
			clr_global_flag = random_civil_war_secondary_route_prepared
			clear_variable = global.random_civil_war_pending_secondary_front_id
			clear_variable = global.random_civil_war_pending_secondary_front_goal
			clear_variable = global.random_civil_war_pending_secondary_archetype
		}
	}
}

				limit = {
					event_target:event021_settlement_manifest_actor = {
						exists = yes
						has_country_flag = random_civil_war_opposition_side
						NOT = { is_actual_nonhuman_country = yes }
						num_owned_states > constant:random_civil_war_value.zero
					}

						is_owned_by = event_target:event021_settlement_manifest_actor
						is_controlled_by = event_target:event021_settlement_manifest_actor
					}
				}
				var:global.random_civil_war_front_states^event021_settlement_manifest_index = { save_event_target_as = event021_settlement_manifest_anchor }

					limit = {
						event_target:event021_settlement_manifest_anchor = {

							check_variable = { var = global.random_civil_war_front_archetypes^event021_settlement_manifest_index value = constant:random_civil_war_archetype.legal compare = equals }

	else = { set_variable = { random_civil_war_settlement_type = constant:random_civil_war_settlement.government_victory } }

# COUNTRY scope. Ends the internal war with an ordinary generated actor. The
# actor is cleaned before annexation so its front row, ideas, mission state,
# and queue entry cannot survive as a stale dynamic country.
event021_parent_close_ordinary_opposition = {
	event021_parent_bind_current_crisis_targets = yes
	if = {
		limit = {
			NOT = { has_country_flag = random_civil_war_event6_host_side }
		}
		save_event_target_as = random_civil_war_ordinary_cleanup_host
		clear_temp_array = event021_ordinary_cleanup_actors
		for_each_scope_loop = {
			array = random_civil_war_opposition_actors
			if = {
				limit = {
					exists = yes
					has_country_flag = random_civil_war_opposition_actor
					NOT = { has_country_flag = random_civil_war_event6_front_active }
					NOT = { has_country_flag = random_civil_war_partition_successor }
					NOT = { has_country_flag = random_civil_war_successor_side }
				}
				add_to_temp_array = { array = event021_ordinary_cleanup_actors value = THIS }
			}
		}
		if = {
			limit = {
				check_variable = {
					var = event021_ordinary_cleanup_actors^num
					value = constant:random_civil_war_value.zero
					compare = greater_than
				}
			}
			for_each_scope_loop = {
				array = event021_ordinary_cleanup_actors
				save_event_target_as = random_civil_war_ordinary_cleanup_actor
				if = {
					limit = {
						has_variable = random_civil_war_immediate_opponent_scope
						var:random_civil_war_immediate_opponent_scope = { exists = yes }
						has_war_with = var:random_civil_war_immediate_opponent_scope
					}
					white_peace = var:random_civil_war_immediate_opponent_scope
				}
				if = {
					limit = {
						has_event_target = random_civil_war_ordinary_cleanup_host
						has_war_with = event_target:random_civil_war_ordinary_cleanup_host
					}
					white_peace = event_target:random_civil_war_ordinary_cleanup_host
				}
				if = { limit = { has_idea = event021_fractured_command } remove_ideas = event021_fractured_command }
				if = { limit = { has_idea = event021_war_torn_administration } remove_ideas = event021_war_torn_administration }
				if = { limit = { has_idea = event021_unsettled_settlement } remove_ideas = event021_unsettled_settlement }
				event021_treaty_mark_settlement_outcome_proof = yes
				clr_country_flag = random_civil_war_active
				set_country_flag = random_civil_war_cleanup_requested
				event021_cleanup_crisis = yes
				event_target:random_civil_war_ordinary_cleanup_host = {
					annex_country = {
						target = event_target:random_civil_war_ordinary_cleanup_actor
						transfer_troops = yes
					}
				}
			}
			clear_array = random_civil_war_opposition_actors
			clear_variable = random_civil_war_opposition_actor_scope
			clear_variable = random_civil_war_secondary_actor_scope
		}
		else_if = {
			limit = {
				has_event_target = random_civil_war_global_opposition_actor
				event_target:random_civil_war_global_opposition_actor = {
					NOT = { has_country_flag = random_civil_war_event6_front_active }
					NOT = { has_country_flag = random_civil_war_partition_successor }
					NOT = { has_country_flag = random_civil_war_successor_side }
				}
			}
			event_target:random_civil_war_global_opposition_actor = {
				if = {
					limit = { has_war_with = event_target:random_civil_war_global_host_country }
					white_peace = event_target:random_civil_war_global_host_country
				}
				if = { limit = { has_idea = event021_fractured_command } remove_ideas = event021_fractured_command }
				if = { limit = { has_idea = event021_war_torn_administration } remove_ideas = event021_war_torn_administration }
				if = { limit = { has_idea = event021_unsettled_settlement } remove_ideas = event021_unsettled_settlement }
				event021_treaty_mark_settlement_outcome_proof = yes
				clr_country_flag = random_civil_war_active
				set_country_flag = random_civil_war_cleanup_requested
				event021_cleanup_crisis = yes
			}
			annex_country = {
				target = event_target:random_civil_war_global_opposition_actor
				transfer_troops = yes
			}
		}
		clear_event_target = random_civil_war_ordinary_cleanup_actor
		clear_event_target = random_civil_war_ordinary_cleanup_host
	}
}

# COUNTRY scope on the original government. A topology-proven partition keeps
# every viable ordinary claimant as a real successor. The rows are copied to a

			limit = {
				check_variable = { var = global.random_civil_war_front_crisis_ids^event021_partition_manifest_index value = event021_partition_manifest_crisis_id compare = equals }

		set_variable = { random_civil_war_parent_log_payload = constant:random_civil_war_log.settlement }
		event021_parent_record_system_log = yes
		set_country_flag = random_civil_war_cleanup_requested
		event021_cleanup_crisis = yes
	}
}

# COUNTRY scope. A complete Event 006 country is recognized separately from an
# ordinary generated claimant. It remains a legitimate human country with its
# package, tree, leaders, flags, and formable routes; only the Event 021 war
# record is closed.
event021_parent_end_event6_war = {
	event021_parent_bind_current_crisis_targets = yes

	# remains only a transaction-time compatibility fallback.

		limit = {
			has_country_flag = random_civil_war_event6_front_active
			has_variable = random_civil_war_host_country_scope
			var:random_civil_war_host_country_scope = { exists = yes }
		}
		var:random_civil_war_host_country_scope = {
			if = {
				limit = { has_war_with = ROOT }
				white_peace = ROOT
			}
		}
		set_country_flag = random_civil_war_event6_recognized_independence
		set_country_flag = random_civil_war_event6_postwar_grace
		set_variable = { random_civil_war_event6_recognition_date = global.date }
		if = { limit = { has_idea = event021_fractured_command } remove_ideas = event021_fractured_command }
		event021_treaty_mark_settlement_outcome_proof = yes
	}
	else_if = {
		limit = {
			has_variable = random_civil_war_opposition_actor_scope
			var:random_civil_war_opposition_actor_scope = { exists = yes }
		}
		var:random_civil_war_opposition_actor_scope = {
			if = {
				limit = { has_war_with = ROOT }
				white_peace = ROOT
			}
			set_country_flag = random_civil_war_event6_recognized_independence
			set_country_flag = random_civil_war_event6_postwar_grace
			set_variable = { random_civil_war_event6_recognition_date = global.date }
			if = { limit = { has_idea = event021_fractured_command } remove_ideas = event021_fractured_command }
			event021_treaty_mark_settlement_outcome_proof = yes
		}
	}
	else_if = {
		limit = { has_event_target = random_civil_war_global_opposition_actor }
		event_target:random_civil_war_global_opposition_actor = {
			if = {
				limit = { has_war_with = event_target:random_civil_war_global_host_country }

			event021_treaty_mark_settlement_outcome_proof = yes
		}
	}

# COUNTRY scope on an opposition side that has genuinely won an ordinary
# territorial crisis. Preserve the predecessor and its front rows until the
# on_annex handover; Event 006 actors are excluded from this path.
event021_parent_promote_ordinary_successor = {
	event021_parent_bind_current_crisis_targets = yes
	# A global pointer is usable only when its crisis receipt identifies this
	# actor's own predecessor. Never annex a different theater's cached host.
	if = {
		limit = {
			NOT = { has_variable = random_civil_war_host_country_scope }
			has_variable = random_civil_war_crisis_id
			has_event_target = random_civil_war_global_host_country
			event_target:random_civil_war_global_host_country = {
				exists = yes
				NOT = { tag = PREV }
				has_variable = random_civil_war_crisis_id
				check_variable = { var = random_civil_war_crisis_id value = PREV.random_civil_war_crisis_id compare = equals }
			}
		}
		set_variable = { random_civil_war_host_country_scope = event_target:random_civil_war_global_host_country }
	}
	if = {
		limit = {
			has_country_flag = random_civil_war_opposition_actor
			NOT = { has_country_flag = random_civil_war_event6_front_active }
			has_variable = random_civil_war_crisis_id
			has_variable = random_civil_war_host_country_scope
			var:random_civil_war_host_country_scope = {
				exists = yes
				NOT = { tag = PREV }
				has_variable = random_civil_war_crisis_id
				check_variable = { var = random_civil_war_crisis_id value = PREV.random_civil_war_crisis_id compare = equals }
			}
		}
		var:random_civil_war_host_country_scope = {
			save_event_target_as = random_civil_war_history_predecessor
			save_event_target_as = event021_history_predecessor
		}
		event021_inherit_successor_achievement_history = yes
		save_event_target_as = random_civil_war_history_successor
		set_country_flag = random_civil_war_successor_side
		set_country_flag = random_civil_war_successor_claimed
		# Freeze the surviving-front roster while the predecessor still owns its
		# host rows. White peace and annexation follow this preparation boundary.
		event021_prepare_successor_roster_adoption = yes
		var:random_civil_war_host_country_scope = {
			if = {
				limit = { has_war_with = event_target:random_civil_war_history_successor }
				white_peace = event_target:random_civil_war_history_successor
			}
		}
		# on_annex owns predecessor teardown and surviving-front rebinding.
		# Core cleanup here would delete the records needed by that callback.
		annex_country = {
			target = event_target:random_civil_war_history_predecessor
			transfer_troops = yes
		}
	}
}

# COUNTRY scope of the triggerable-scenario controller. Low, medium, and high
# intensities consume a bounded number of immediate target attempts. Maximum
# intensity commits every eligible normal country through the same parent
# transaction. Lower intensities build their eligible pool in the same explicit
# one-shot manual request and then consume a bounded share. No recurring review
# or country-wide loop is left running after the launch request returns.
random_civil_war_trigger_manual_scenario = {
	event021_parent_initialize_global_runtime = yes
	if = {
		limit = {
			has_global_flag = random_civil_war_rework_ready
			NOT = { has_global_flag = world_end }
			NOT = { has_global_flag = random_civil_war_manual_scenario_launch_in_progress }
		}
		if = {
			limit = { NOT = { has_variable = triggerable_scenarios_random_civil_war_type } }
			set_variable = { triggerable_scenarios_random_civil_war_type = constant:random_civil_war_scenario_type.political_fracture }
		}
		if = {
			limit = { NOT = { has_variable = triggerable_scenarios_intensity } }
			set_variable = { triggerable_scenarios_intensity = constant:random_civil_war_scenario_intensity.low }
		}
		set_variable = { global.random_civil_war_manual_scenario_type = triggerable_scenarios_random_civil_war_type }
		set_variable = { global.random_civil_war_manual_scenario_intensity = triggerable_scenarios_intensity }
		set_variable = { global.random_civil_war_scenario_last_type = triggerable_scenarios_random_civil_war_type }
		set_variable = { global.random_civil_war_scenario_last_intensity = triggerable_scenarios_intensity }
		set_variable = { global.random_civil_war_scenario_eligible_count = constant:random_civil_war_value.zero }
		set_variable = { global.random_civil_war_scenario_requested_count = constant:random_civil_war_value.zero }
		set_variable = { global.random_civil_war_scenario_committed_count = constant:random_civil_war_value.zero }
		set_variable = { global.random_civil_war_scenario_skipped_count = constant:random_civil_war_value.zero }
		set_variable = { global.random_civil_war_scenario_last_skip_reason = constant:random_civil_war_cluster_skip_reason.none }
		set_global_flag = random_civil_war_manual_scenario_launch_in_progress
		if = {
			limit = { check_variable = { triggerable_scenarios_intensity = constant:random_civil_war_scenario_intensity.maximum } }
			set_global_flag = random_civil_war_scenario_maximum_mode
			clear_temp_array = event021_parent_scenario_target_pool
			every_country = {
				limit = { event021_parent_scenario_country_eligible = yes }
				add_to_temp_array = { event021_parent_scenario_target_pool = THIS }
			}
			set_variable = { global.random_civil_war_scenario_eligible_count = event021_parent_scenario_target_pool^num }
			set_variable = { global.random_civil_war_scenario_requested_count = event021_parent_scenario_target_pool^num }
			for_each_scope_loop = {
				array = event021_parent_scenario_target_pool
				set_variable = { random_civil_war_scenario_type = global.random_civil_war_manual_scenario_type }
				set_variable = { random_civil_war_scenario_intensity = global.random_civil_war_manual_scenario_intensity }
				set_country_flag = random_civil_war_scenario_target_confirmed
				country_event = { id = chaosx.nr21.15 }
				event021_parent_record_scenario_result = yes
				clr_country_flag = random_civil_war_scenario_target_confirmed
			}
			clear_temp_array = event021_parent_scenario_target_pool
		}
		else = {
			clear_temp_array = event021_parent_scenario_target_pool
			every_country = {
				limit = { event021_parent_scenario_country_eligible = yes }
				event021_parent_add_scenario_target_to_weighted_pool = yes
			}
			set_variable = { global.random_civil_war_manual_target_budget = global.random_civil_war_scenario_eligible_count }
			if = {
				limit = { check_variable = { triggerable_scenarios_intensity = constant:random_civil_war_scenario_intensity.low } }
				multiply_variable = { global.random_civil_war_manual_target_budget = constant:random_civil_war_scenario.low_share }
			}
			if = {
				limit = { check_variable = { triggerable_scenarios_intensity = constant:random_civil_war_scenario_intensity.medium } }
				multiply_variable = { global.random_civil_war_manual_target_budget = constant:random_civil_war_scenario.medium_share }
			}
			if = {
				limit = { check_variable = { triggerable_scenarios_intensity = constant:random_civil_war_scenario_intensity.high } }
				multiply_variable = { global.random_civil_war_manual_target_budget = constant:random_civil_war_scenario.high_share }
			}
			round_variable = global.random_civil_war_manual_target_budget
			if = {
				limit = {
					check_variable = {
						var = global.random_civil_war_scenario_eligible_count
						value = constant:random_civil_war_value.zero
						compare = greater_than
					}
					check_variable = {
						var = global.random_civil_war_manual_target_budget
						value = constant:random_civil_war_value.one
						compare = less_than
					}
				}
				set_variable = { global.random_civil_war_manual_target_budget = constant:random_civil_war_value.one }
			}
			if = {
				limit = {
					check_variable = {
						var = global.random_civil_war_manual_target_budget
						value = global.random_civil_war_scenario_eligible_count
						compare = greater_than
					}
				}
				set_variable = { global.random_civil_war_manual_target_budget = global.random_civil_war_scenario_eligible_count }
			}
			set_variable = { global.random_civil_war_scenario_requested_count = global.random_civil_war_manual_target_budget }
			set_temp_variable = { event021_parent_scenario_selected_count = constant:random_civil_war_value.zero }
			while_loop_effect = {
				limit = {
					check_variable = {
						var = event021_parent_scenario_selected_count
						value = global.random_civil_war_manual_target_budget
						compare = less_than
					}
					check_variable = {
						var = event021_parent_scenario_target_pool^num
						value = constant:random_civil_war_value.zero
						compare = greater_than
					}
				}
				clear_event_target = random_civil_war_scenario_selected_target
				random_scope_in_array = {
					array = event021_parent_scenario_target_pool
					limit = { event021_parent_scenario_country_eligible = yes }
					save_event_target_as = random_civil_war_scenario_selected_target
				}
				if = {

# COUNTRY scope. The scenario callback uses the same immediate transaction as
# normal play. It never schedules a delayed fallback and therefore cannot leave
# a scenario half-committed after its bounded confirmation window.
event021_parent_commit_scenario_country = {
	if = {
		limit = {
			OR = {
				has_global_flag = random_civil_war_scenario_bypass_active
				has_global_flag = random_civil_war_manual_scenario_launch_in_progress
				has_country_flag = random_civil_war_scenario_bypass_active
			}
			OR = {
				has_country_flag = random_civil_war_scenario_target_confirmed
				has_global_flag = random_civil_war_scenario_target_confirmed
			}
		}
		event021_parent_prepare_scenario_country = yes
		if = { limit = { has_country_flag = random_civil_war_target_ready } event021_parent_begin_target_transaction = yes }
		if = { limit = { has_global_flag = random_civil_war_plan_active } event021_parent_plan_connected_states = yes }
		if = { limit = { has_global_flag = random_civil_war_plan_active } event021_parent_validate_opening_plan = yes }
		if = {
			limit = {
				has_country_flag = random_civil_war_parent_plan_valid
				check_variable = {
					var = random_civil_war_archetype
					value = constant:random_civil_war_archetype.same_tag
					compare = equals
				}
			}
			country_event = { id = chaosx.nr21.16 }
		}
		else_if = {
			limit = {
				has_country_flag = random_civil_war_parent_plan_valid
				check_variable = {
					var = random_civil_war_archetype
					value = constant:random_civil_war_archetype.event006
					compare = equals
				}
				event021_parent_event6_dormant_candidate = yes
			}
			event021_parent_start_event6_front = yes
		}
		else_if = { limit = { has_country_flag = random_civil_war_parent_plan_valid } event021_parent_start_ordinary_civil_war = yes }
		else = { event021_parent_rollback_transaction = yes }
	}
}
