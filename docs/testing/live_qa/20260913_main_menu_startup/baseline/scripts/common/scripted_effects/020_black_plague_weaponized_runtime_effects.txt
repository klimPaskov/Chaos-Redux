# ============================================================================
# CHAOS REDUX - BLACK PLAGUE WEAPONIZED RUNTIME BRIDGE EFFECTS
# ============================================================================
# Bounded STATE-scope integration for authorized weaponized Black Plague
# exposure. The bridge preserves caller-supplied exposure inputs, invokes the
# existing shared exposure effect exactly once, and repairs one shared
# scheduler receipt only after the shared exposure accepts the state.
# ============================================================================

# STATE scope. ROOT is the country that owns the release transaction.
# Required input: temporary black_plague_exposure_provenance set to
# constant:black_plague_provenance.weaponized. Optional inputs retain the
# existing black_plague_apply_exposure defaults when absent:
# black_plague_exposure_amount and black_plague_exposure_route.
# Output: temporary black_plague_weaponized_runtime_bridge_result is exactly
# constant:black_plague_value.one when black_plague_apply_exposure accepts the
# state, otherwise constant:black_plague_value.zero.
#
# This effect owns no payload accounting, ordinary bio_lifecycle dispatch,
# natural-outbreak setup, origin/report state, or public attribution history.
black_plague_apply_weaponized_exposure_runtime = {
	set_temp_variable = { black_plague_weaponized_runtime_bridge_result = constant:black_plague_value.zero }
	set_temp_variable = { black_plague_exposure_result = constant:black_plague_value.zero }

	# The one-time initializer runs in ROOT and may rewrite runtime-wide
	# temporary context. Snapshot every caller input before entering it.
	if = {
		limit = { has_variable = black_plague_exposure_amount }
		set_temp_variable = { black_plague_weaponized_runtime_input_amount = black_plague_exposure_amount }
	}
	if = {
		limit = { has_variable = black_plague_exposure_route }
		set_temp_variable = { black_plague_weaponized_runtime_input_route = black_plague_exposure_route }
	}
	if = {
		limit = { has_variable = black_plague_exposure_provenance }
		set_temp_variable = { black_plague_weaponized_runtime_input_provenance = black_plague_exposure_provenance }
	}

	if = {
		limit = { black_plague_weaponized_runtime_bridge_target_is_valid = yes }
		ROOT = {
			if = {
				limit = {
					NOT = { has_global_flag = black_plague_system_started }
					NOT = { has_global_flag = black_plague_pulses_suppressed }
				}
				black_plague_initialize_runtime = yes
			}
		}
		# Restore the caller's state-scoped inputs before the post-bootstrap gate.
		# Initialization may reuse the shared temporary exposure selectors.
		if = {
			limit = { has_variable = black_plague_weaponized_runtime_input_amount }
			set_temp_variable = { black_plague_exposure_amount = black_plague_weaponized_runtime_input_amount }
		}
		if = {
			limit = { has_variable = black_plague_weaponized_runtime_input_route }
			set_temp_variable = { black_plague_exposure_route = black_plague_weaponized_runtime_input_route }
		}
		if = {
			limit = { has_variable = black_plague_weaponized_runtime_input_provenance }
			set_temp_variable = { black_plague_exposure_provenance = black_plague_weaponized_runtime_input_provenance }
		}
		# Re-check after bootstrap so an unexpected runtime transition cannot
		# turn a guarded initialization into an exposure or scheduler mutation.
		if = {
			limit = { black_plague_weaponized_runtime_bridge_target_is_valid = yes }
			black_plague_apply_exposure = yes
			set_temp_variable = { black_plague_weaponized_runtime_bridge_result = black_plague_exposure_result }
			if = {
				limit = {
					check_variable = {
						var = black_plague_weaponized_runtime_bridge_result
						value = constant:black_plague_value.one
						compare = equals
					}
				}
				black_plague_weaponized_runtime_bridge_ensure_scheduler = yes
			}
		}
	}

	# Temporary variables are scope-less, so the result is already available to
	# the country-scope caller after this state block returns. Remove only the
	# bridge's private input snapshots; keep both public result values intact.
	clear_temp_variable = black_plague_weaponized_runtime_input_amount
	clear_temp_variable = black_plague_weaponized_runtime_input_route
	clear_temp_variable = black_plague_weaponized_runtime_input_provenance
}

# STATE scope. Reconciles exactly one scheduler anchor after an accepted
# exposure. A valid receipt is untouched; an active anchor with an invalid or
# missing receipt is rescheduled in place. An absent or invalid anchor target
# is replaced only from the accepted current state, with no world scan.
black_plague_weaponized_runtime_bridge_ensure_scheduler = {
	if = {
		limit = {
			exists = yes
			has_global_flag = black_plague_system_active
			NOT = { has_global_flag = black_plague_pulses_suppressed }
			NOT = { has_global_flag = world_end }
			NOT = { has_global_flag = black_plague_terminal_takeover_active }
			NOT = { has_global_flag = black_plague_terminal_takeover_complete }
		}
		if = {
			limit = { black_plague_weaponized_runtime_scheduler_receipt_is_valid = yes }
		}
		else_if = {
			limit = { black_plague_weaponized_runtime_scheduler_anchor_is_active = yes }
			event_target:black_plague_scheduler_anchor_state = {
				black_plague_schedule_next_pulse = yes
			}
		}
		else = {
			# Only the reachable old target is cancelled. If the global pointer is
			# absent, an unreachable stale state flag is deliberately not scanned.
			if = {
				limit = {
					has_event_target = black_plague_scheduler_anchor_state
					event_target:black_plague_scheduler_anchor_state = { exists = yes }
				}
				event_target:black_plague_scheduler_anchor_state = {
					clr_state_flag = black_plague_scheduler_anchor
					clear_variable = black_plague_scheduler_generation
					clear_variable = black_plague_scheduler_ticket
					clear_variable = black_plague_scheduler_due_num_days
				}
			}
			clear_global_event_target = black_plague_scheduler_anchor_state
			save_global_event_target_as = black_plague_scheduler_anchor_state
			set_state_flag = black_plague_scheduler_anchor
			black_plague_schedule_next_pulse = yes
		}
	}
}
