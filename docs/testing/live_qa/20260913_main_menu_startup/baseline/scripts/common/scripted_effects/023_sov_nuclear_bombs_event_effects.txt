# ============================================================================
# Event-owned orchestration for the Soviet arsenal. Authorization, custody
# doctrine, evolution gates, selected targets, decision costs, posture, AI
# context, and player-facing follow-up events live here. Every physical bomb

	if = {
		limit = {
			sov_nuclear_bombs_event_has_public_threat_surface = yes
			NOT = { sov_nuclear_bombs_terminal_world_state_is_active = yes }
		}

# Select the strongest already-eligible opening package without granting or
# recording any disabled evolution. The caller runs this after the Event 23
# actor is active, so the same world gate is used by the later progression
# path.
sov_nuclear_bombs_prepare_opening_package = {
	set_variable = { sov_nuclear_bombs_opening_stage = constant:sov_nuclear_bombs_evolution.stage_baseline }
	if = {
		limit = {
			NOT = { has_global_flag = events_log_disabled_evolution_23_23_1 }
			check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.chaos_i compare = greater_than_or_equals }
		}
		set_variable = { sov_nuclear_bombs_opening_stage = constant:sov_nuclear_bombs_evolution.stage_i }
	}
	if = {
		limit = {
			NOT = { has_global_flag = events_log_disabled_evolution_23_23_2 }
			check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_i compare = greater_than_or_equals }
			check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.chaos_ii compare = greater_than_or_equals }
		}
		set_variable = { sov_nuclear_bombs_opening_stage = constant:sov_nuclear_bombs_evolution.stage_ii }
	}
	if = {
		limit = {
			NOT = { has_global_flag = events_log_disabled_evolution_23_23_3 }
			check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_ii compare = greater_than_or_equals }
			check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.chaos_iii compare = greater_than_or_equals }
			sov_nuclear_bombs_event_evolution_iii_world_gate = yes
		}
		set_variable = { sov_nuclear_bombs_opening_stage = constant:sov_nuclear_bombs_evolution.stage_iii }
	}
	if = {
		limit = {
			NOT = { has_global_flag = events_log_disabled_evolution_23_23_4 }
			check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_iii compare = greater_than_or_equals }
			check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.chaos_iv compare = greater_than_or_equals }
			sov_nuclear_bombs_event_evolution_iv_world_gate = yes
		}
		set_variable = { sov_nuclear_bombs_opening_stage = constant:sov_nuclear_bombs_evolution.stage_iv }
	}
	set_variable = { sov_nuclear_bombs_opening_amount = constant:sov_nuclear_bombs_evolution.baseline }
	if = {
		limit = { check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_i compare = greater_than_or_equals } }
		set_variable = { sov_nuclear_bombs_opening_amount = constant:sov_nuclear_bombs_evolution.total_i }
	}
	if = {
		limit = { check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_ii compare = greater_than_or_equals } }
		set_variable = { sov_nuclear_bombs_opening_amount = constant:sov_nuclear_bombs_evolution.total_ii }
	}
	if = {
		limit = { check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_iii compare = greater_than_or_equals } }
		set_variable = { sov_nuclear_bombs_opening_amount = constant:sov_nuclear_bombs_evolution.total_iii }
	}
	if = {
		limit = { check_variable = { var = sov_nuclear_bombs_opening_stage value = constant:sov_nuclear_bombs_evolution.stage_iv compare = greater_than_or_equals } }
		set_variable = { sov_nuclear_bombs_opening_amount = constant:sov_nuclear_bombs_evolution.total_iv }
	}
}

# Build the bounded central, storage, and test network used by the opening.
# The three storage allocations deliberately leave later-stage increments in
# the actor operational bucket until the player chooses where to place them.
sov_nuclear_bombs_initialize_opening_sites = {
	set_variable = { sov_nuclear_bombs_site_registration_amount = constant:sov_nuclear_bombs_tuning.baseline_site_primary }
	capital_scope = { sov_nuclear_bombs_register_storage_site = yes }
	set_variable = { sov_nuclear_bombs_site_registration_amount = constant:sov_nuclear_bombs_tuning.baseline_site_secondary }
	random_owned_controlled_state = {
		limit = {
			impassable = no
			NOT = { is_capital = yes }
			NOT = { has_state_flag = sov_nuclear_bombs_registered_storage_site }
		}
		sov_nuclear_bombs_register_storage_site = yes
	}
	set_variable = { sov_nuclear_bombs_site_registration_amount = constant:sov_nuclear_bombs_tuning.baseline_site_tertiary }
	random_owned_controlled_state = {
		limit = {
			impassable = no
			NOT = { is_capital = yes }
			NOT = { has_state_flag = sov_nuclear_bombs_registered_storage_site }
		}
		sov_nuclear_bombs_register_storage_site = yes
	}
	if = {
		limit = { NOT = { has_event_target = sov_nuclear_bombs_test_state } }
		random_owned_controlled_state = {
			limit = {
				impassable = no
				NOT = { is_capital = yes }
				has_state_flag = sov_nuclear_bombs_registered_storage_site
			}
			save_global_event_target_as = sov_nuclear_bombs_test_state
		}
	}
	if = {
		limit = { NOT = { has_event_target = sov_nuclear_bombs_test_state } }
		# A test target must remain a remote, non-capital site. If the map has no
		# qualifying registered state, leave the selector closed rather than
		# silently substituting the command capital.
		random_owned_controlled_state = {
			limit = { sov_nuclear_bombs_event_state_is_valid_test_site = yes has_state_flag = sov_nuclear_bombs_registered_storage_site }
			save_global_event_target_as = sov_nuclear_bombs_test_state
		}
	}
	clear_variable = sov_nuclear_bombs_site_registration_amount
}

# Reactor buildings are state-scoped and capped at one level in vanilla. Every
# Event 23 placement therefore receives its own state marker and is credited
# only after a later completion check observes the completed building.
sov_nuclear_bombs_queue_reactor_site = {
	if = {
		limit = { sov_nuclear_bombs_event_state_is_valid_reactor_site = yes }
		set_state_flag = sov_nuclear_bombs_reactor_site_pending
		add_building_construction = {
			type = nuclear_reactor
			level = constant:sov_nuclear_bombs_tuning.production_reactor_level
		}
		ROOT = {
			if = {
				limit = { has_variable = sov_nuclear_bombs_reactor_pending_count }
				add_to_variable = { sov_nuclear_bombs_reactor_pending_count = constant:sov_nuclear_bombs_ledger.one }
			}
			else = { set_variable = { sov_nuclear_bombs_reactor_pending_count = constant:sov_nuclear_bombs_ledger.one } }
		}
	}
}

sov_nuclear_bombs_queue_opening_reactors = {
	set_temp_variable = { sov_nuclear_bombs_opening_reactors_remaining = sov_nuclear_bombs_reactor_entitlement_count }
	subtract_from_temp_variable = { sov_nuclear_bombs_opening_reactors_remaining = sov_nuclear_bombs_reactor_count }
	if = {
		limit = { has_variable = sov_nuclear_bombs_reactor_pending_count }
		subtract_from_temp_variable = { sov_nuclear_bombs_opening_reactors_remaining = sov_nuclear_bombs_reactor_pending_count }
	}
	clamp_temp_variable = {
		var = sov_nuclear_bombs_opening_reactors_remaining
		min = constant:sov_nuclear_bombs_ledger.zero
		max = constant:sov_nuclear_bombs_tuning.reactor_building_limit
	}
	# Keep one persistent state target for each build. A single hidden check can
	# then reconcile the vanilla building completion before the next entitlement
	# is placed, which prevents multiple queue-time grants from sharing one target.
	if = {
		limit = { NOT = { has_event_target = sov_nuclear_bombs_pending_reactor_state } }
		random_owned_controlled_state = {
			limit = { sov_nuclear_bombs_event_pending_reactor_site_is_valid = yes }
			save_global_event_target_as = sov_nuclear_bombs_pending_reactor_state
			ROOT = { set_variable = { sov_nuclear_bombs_pending_reactor_state_id = THIS.id } }
		}
	}
	if = {
		limit = {
			check_variable = {
				var = sov_nuclear_bombs_opening_reactors_remaining
				value = constant:sov_nuclear_bombs_ledger.zero
				compare = greater_than
			}
			NOT = { has_event_target = sov_nuclear_bombs_pending_reactor_state }
			NOT = { any_owned_state = { has_state_flag = sov_nuclear_bombs_reactor_site_pending } }
		}
		random_owned_controlled_state = {
			limit = { sov_nuclear_bombs_event_state_is_valid_reactor_site = yes }
			sov_nuclear_bombs_queue_reactor_site = yes
			save_global_event_target_as = sov_nuclear_bombs_pending_reactor_state
			ROOT = { set_variable = { sov_nuclear_bombs_pending_reactor_state_id = THIS.id } }
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = { var = sov_nuclear_bombs_opening_reactors_remaining value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than }
				has_event_target = sov_nuclear_bombs_pending_reactor_state
				any_owned_state = { has_state_flag = sov_nuclear_bombs_reactor_site_pending }
			}
		}
		set_country_flag = sov_nuclear_bombs_reactor_opening_pending
		set_country_flag = sov_nuclear_bombs_reactor_construction_pending
	}
	else = { clr_country_flag = sov_nuclear_bombs_reactor_opening_pending }
	clear_temp_variable = sov_nuclear_bombs_opening_reactors_remaining
}

sov_nuclear_bombs_schedule_reactor_completion_check = {
	if = {
		limit = {
			sov_nuclear_bombs_event_actor_is_valid = yes
			sov_nuclear_bombs_event_has_pending_reactor_construction = yes
			NOT = { has_country_flag = sov_nuclear_bombs_reactor_completion_event_active }
		}
		set_country_flag = sov_nuclear_bombs_reactor_completion_event_active
		country_event = { id = chaosx.nr23.181 days = constant:sov_nuclear_bombs_tuning.reactor_completion_check_days }
	}
}

sov_nuclear_bombs_register_completed_reactor = {
	if = {
		limit = {
			has_state_flag = sov_nuclear_bombs_reactor_site_pending
			nuclear_reactor > 0
		}
		set_state_flag = sov_nuclear_bombs_reactor_event_owned
		clr_state_flag = sov_nuclear_bombs_reactor_site_pending
		ROOT = {
			if = {
				limit = { has_variable = sov_nuclear_bombs_reactor_pending_count }
				subtract_from_variable = { sov_nuclear_bombs_reactor_pending_count = constant:sov_nuclear_bombs_ledger.one }
				clamp_variable = {
					var = sov_nuclear_bombs_reactor_pending_count
					min = constant:sov_nuclear_bombs_ledger.zero
					max = constant:sov_nuclear_bombs_tuning.reactor_building_limit
				}
			}
			if = {
				limit = {
					has_variable = sov_nuclear_bombs_reactor_count
					check_variable = { var = sov_nuclear_bombs_reactor_count value = sov_nuclear_bombs_reactor_entitlement_count compare = less_than }
				}
				add_to_variable = { sov_nuclear_bombs_reactor_count = constant:sov_nuclear_bombs_ledger.one }
			}
		}
	}
}

sov_nuclear_bombs_verify_pending_reactor_construction = {
	clr_country_flag = sov_nuclear_bombs_reactor_completion_event_active
	if = {
		limit = { has_event_target = sov_nuclear_bombs_pending_reactor_state }
		if = {
			limit = { sov_nuclear_bombs_event_pending_reactor_completion_is_valid = yes }
			event_target:sov_nuclear_bombs_pending_reactor_state = { sov_nuclear_bombs_register_completed_reactor = yes }
			if = {
				limit = { has_country_flag = sov_nuclear_bombs_reactor_production_completion_pending }
				set_country_flag = sov_nuclear_bombs_reactor_entitlement_consumed
				set_country_flag = sov_nuclear_bombs_fissile_production_completed
				clr_country_flag = sov_nuclear_bombs_reactor_production_completion_pending
				set_variable = { sov_nuclear_bombs_phase = constant:sov_nuclear_bombs_phase.production }
				add_to_variable = { sov_nuclear_bombs_readiness = constant:sov_nuclear_bombs_tuning.production_complete_readiness }
				add_to_variable = { sov_nuclear_bombs_integrity = constant:sov_nuclear_bombs_tuning.production_complete_integrity }
				clamp_variable = { var = sov_nuclear_bombs_readiness min = 0 max = 100 }
				clamp_variable = { var = sov_nuclear_bombs_integrity min = 0 max = 100 }
			}
			clear_global_event_target = sov_nuclear_bombs_pending_reactor_state
			clear_variable = sov_nuclear_bombs_pending_reactor_state_id
		}
		else_if = {
			limit = { sov_nuclear_bombs_event_pending_reactor_site_is_valid = yes }
			# Construction is still in progress. Keep the entitlement, target, and
			# production reservation intact until a later bounded verification sees
			# the completed vanilla building.
			set_country_flag = sov_nuclear_bombs_reactor_construction_pending
		}
		else = {
			event_target:sov_nuclear_bombs_pending_reactor_state = {
				if = {
					limit = { has_state_flag = sov_nuclear_bombs_reactor_site_pending }
					clr_state_flag = sov_nuclear_bombs_reactor_site_pending
					ROOT = {
						if = {
							limit = { has_variable = sov_nuclear_bombs_reactor_pending_count }
							subtract_from_variable = { sov_nuclear_bombs_reactor_pending_count = constant:sov_nuclear_bombs_ledger.one }
							clamp_variable = {
								var = sov_nuclear_bombs_reactor_pending_count
								min = constant:sov_nuclear_bombs_ledger.zero
								max = constant:sov_nuclear_bombs_tuning.reactor_building_limit
							}
						}
					}
				}
			}
			clr_country_flag = sov_nuclear_bombs_reactor_production_completion_pending
			set_country_flag = sov_nuclear_bombs_reactor_construction_cancelled
			clear_global_event_target = sov_nuclear_bombs_pending_reactor_state
			clear_variable = sov_nuclear_bombs_pending_reactor_state_id
		}
	}
	if = {
		limit = {
			NOT = { has_event_target = sov_nuclear_bombs_pending_reactor_state }
			NOT = { has_country_flag = sov_nuclear_bombs_reactor_opening_pending }
			NOT = { any_owned_state = { has_state_flag = sov_nuclear_bombs_reactor_site_pending } }
		}
		clr_country_flag = sov_nuclear_bombs_reactor_construction_pending
	}
	if = {
		limit = { has_country_flag = sov_nuclear_bombs_reactor_opening_pending }
		sov_nuclear_bombs_queue_opening_reactors = yes
	}
}

sov_nuclear_bombs_reactor_control_changed = {
	if = {
		limit = { has_state_flag = sov_nuclear_bombs_reactor_site_pending }
		clr_state_flag = sov_nuclear_bombs_reactor_site_pending
		# Event 023 permits only one reactor construction target at a time, so
		# this exact pending state is the sole possible global reactor pointer.
		# Clear the pointer from the state scope even if SOV no longer exists.
		clear_global_event_target = sov_nuclear_bombs_pending_reactor_state
		SOV = {
			if = {
				limit = { has_variable = sov_nuclear_bombs_reactor_pending_count }
				subtract_from_variable = { sov_nuclear_bombs_reactor_pending_count = constant:sov_nuclear_bombs_ledger.one }
				clamp_variable = {
					var = sov_nuclear_bombs_reactor_pending_count
					min = constant:sov_nuclear_bombs_ledger.zero
					max = constant:sov_nuclear_bombs_tuning.reactor_building_limit
				}
			}
			clear_variable = sov_nuclear_bombs_pending_reactor_state_id
			clr_country_flag = sov_nuclear_bombs_reactor_construction_pending
			clr_country_flag = sov_nuclear_bombs_reactor_opening_pending
			clr_country_flag = sov_nuclear_bombs_reactor_production_completion_pending
		}
	}
}

# Stage the exact action context required by the shared native delivery
# contract. The selected state is persisted as a global event target by the
# targetable operational decision.
sov_nuclear_bombs_stage_action_context = {
	# Preserve the exact reserved request until execution or explicit release.
	if = {
		limit = {
			NOT = { has_country_flag = sov_nuclear_bombs_native_delivery_pending }
			NOT = { has_country_flag = sov_nuclear_bombs_action_reservation_active }
		}
		sov_nuclear_bombs_initialize_actor_ledger = yes
		clear_variable = sov_nuclear_bombs_action_state
		clear_variable = sov_nuclear_bombs_action_source_state
		clear_variable = sov_nuclear_bombs_action_target
		clear_variable = sov_nuclear_bombs_action_target_country_id
		clear_variable = sov_nuclear_bombs_action_reservation_holder
		if = {
			limit = {
				has_event_target = sov_nuclear_bombs_selected_test_state
				OR = {
					check_variable = { var = sov_nuclear_bombs_pending_action_type value = constant:sov_nuclear_bombs_action.test compare = equals }
					check_variable = { var = sov_nuclear_bombs_pending_action_type value = constant:sov_nuclear_bombs_action.demonstration compare = equals }
				}
			}
			set_variable = { sov_nuclear_bombs_action_state = event_target:sov_nuclear_bombs_selected_test_state }
		}
		else_if = {
			limit = { has_event_target = sov_nuclear_bombs_selected_state }
			set_variable = { sov_nuclear_bombs_action_state = event_target:sov_nuclear_bombs_selected_state }
		}
		if = {
			limit = { check_variable = { var = sov_nuclear_bombs_pending_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } }
			set_variable = { sov_nuclear_bombs_action_source_state = sov_nuclear_bombs_action_state }
		}
		else_if = {
			limit = {
				has_variable = sov_nuclear_bombs_action_state
				var:sov_nuclear_bombs_action_state = {
					sov_nuclear_bombs_state_is_registered = yes
					check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than }
				}
			}
			set_variable = { sov_nuclear_bombs_action_source_state = sov_nuclear_bombs_action_state }
		}
		else = {
			random_owned_controlled_state = {
				limit = {
					sov_nuclear_bombs_state_is_registered = yes
					check_variable = { var = sov_nuclear_bombs_site_assigned value = constant:sov_nuclear_bombs_ledger.one compare = greater_than }
				}
				set_variable = { ROOT.sov_nuclear_bombs_action_source_state = THIS }
			}
		}
		var:sov_nuclear_bombs_action_state = {
			set_variable = { ROOT.sov_nuclear_bombs_action_state_id = THIS.id }
			if = {
				limit = { CONTROLLER = { exists = yes } }
				set_variable = { ROOT.sov_nuclear_bombs_action_target = CONTROLLER }
			}
			else_if = {
				limit = { OWNER = { exists = yes } }
				set_variable = { ROOT.sov_nuclear_bombs_action_target = OWNER }
			}
		}
		var:sov_nuclear_bombs_action_target = {
			set_variable = { ROOT.sov_nuclear_bombs_action_target_country_id = THIS.id }
		}
		set_variable = { sov_nuclear_bombs_action_actor_proven = constant:sov_nuclear_bombs_ledger.one }
		set_variable = { sov_nuclear_bombs_action_authorized = constant:sov_nuclear_bombs_ledger.one }
		set_variable = { sov_nuclear_bombs_action_weapon = constant:sov_nuclear_bombs_action.nuclear_bomb }
		set_variable = { sov_nuclear_bombs_action_route = constant:sov_nuclear_bombs_action.native_launch_nuke }
		if = {
			limit = { check_variable = { var = sov_nuclear_bombs_pending_action_type value = constant:sov_nuclear_bombs_action.demolition compare = equals } }
			set_variable = { sov_nuclear_bombs_action_detonation = constant:sov_nuclear_bombs_action.non_detonating }
		}
		else_if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_pending_action_type value = constant:sov_nuclear_bombs_action.test compare = equals }
				has_variable = sov_nuclear_bombs_action_outcome
				check_variable = { var = sov_nuclear_bombs_action_outcome value = constant:sov_nuclear_bombs_tuning.test_result_failure compare = equals }
			}
			set_variable = { sov_nuclear_bombs_action_detonation = constant:sov_nuclear_bombs_action.non_detonating }
		}
		else = { set_variable = { sov_nuclear_bombs_action_detonation = constant:sov_nuclear_bombs_action.detonation_required } }
		set_variable = { sov_nuclear_bombs_action_type = sov_nuclear_bombs_pending_action_type }
		set_variable = { sov_nuclear_bombs_action_nonce = sov_nuclear_bombs_last_action_nonce }
		add_to_variable = { sov_nuclear_bombs_action_nonce = constant:sov_nuclear_bombs_ledger.one }
		sov_nuclear_bombs_reserve_device_for_action = yes
	}
}

# Classify a prepared test before the shared action contract is executed. A
# failure is a non-detonating instrument result and returns its reserved device
# intact; success and accident continue through the one native delivery route.
sov_nuclear_bombs_prepare_test_outcome = {
	if = {
		limit = {
			has_variable = sov_nuclear_bombs_action_type
			check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.test compare = equals }
			has_variable = sov_nuclear_bombs_action_nonce
		}
		clear_variable = sov_nuclear_bombs_action_outcome
		clear_variable = sov_nuclear_bombs_test_result
		clr_country_flag = sov_nuclear_bombs_test_success
		clr_country_flag = sov_nuclear_bombs_test_failure
		clr_country_flag = sov_nuclear_bombs_test_accident
		random_list = {
			constant:sov_nuclear_bombs_tuning.test_success_weight = { set_variable = { sov_nuclear_bombs_action_outcome = constant:sov_nuclear_bombs_tuning.test_result_success } }
			constant:sov_nuclear_bombs_tuning.test_failure_weight = { set_variable = { sov_nuclear_bombs_action_outcome = constant:sov_nuclear_bombs_tuning.test_result_failure } }
			constant:sov_nuclear_bombs_tuning.test_accident_weight = { set_variable = { sov_nuclear_bombs_action_outcome = constant:sov_nuclear_bombs_tuning.test_result_accident } }
		}
		if = {
			limit = { check_variable = { var = sov_nuclear_bombs_action_outcome value = constant:sov_nuclear_bombs_tuning.test_result_failure compare = equals } }
			set_variable = { sov_nuclear_bombs_action_detonation = constant:sov_nuclear_bombs_action.non_detonating }
		}
		else = { set_variable = { sov_nuclear_bombs_action_detonation = constant:sov_nuclear_bombs_action.detonation_required } }
	}
}

				if = { limit = { air_base > 0 } remove_building = { type = air_base level = 1 } }
				else_if = { limit = { arms_factory > 0 } remove_building = { type = arms_factory level = 1 } }
				else_if = { limit = { dockyard > 0 } remove_building = { type = dockyard level = 1 } }
				else_if = { limit = { naval_base > 0 } remove_building = { type = naval_base level = 1 } }
				else_if = { limit = { radar_station > 0 } remove_building = { type = radar_station level = 1 } }
				set_state_flag = sov_nuclear_bombs_demilitarized_zone
				set_state_flag = sov_nuclear_bombs_settlement_demilitarization_recorded
			}
			set_country_flag = sov_nuclear_bombs_settlement_terms_applied
			set_country_flag = sov_nuclear_bombs_settlement_demilitarization_recorded
		}
		if = {
			limit = { has_country_flag = sov_nuclear_bombs_partial_settlement_accepted }
			# A partial ceasefire is a bounded white peace, while a partial return
			# stays joint and a partial protection offer is only a guarantee.
			if = {
				limit = {
					OR = {
						check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.cease_hostilities compare = equals }
						check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals }

			}

				event_target:sov_nuclear_bombs_selected_state = {
					set_state_flag = sov_nuclear_bombs_site_joint_custody
					set_variable = { sov_nuclear_bombs_site_custody_state = constant:sov_nuclear_bombs_custody.joint }
					set_variable = { sov_nuclear_bombs_site_reconciliation_state = constant:sov_nuclear_bombs_custody.reconciled }
				}
				set_country_flag = sov_nuclear_bombs_settlement_terms_applied
				set_country_flag = sov_nuclear_bombs_settlement_return_recorded
			}
			else_if = {
				limit = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.soviet_protection compare = equals } }
				give_guarantee = event_target:sov_nuclear_bombs_coercion_target
				set_country_flag = sov_nuclear_bombs_settlement_terms_applied
				set_country_flag = sov_nuclear_bombs_settlement_protection_recorded
			}
			else_if = {
				limit = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.withdraw_territory compare = equals } }
				event_target:sov_nuclear_bombs_selected_state = { set_state_flag = sov_nuclear_bombs_demilitarized_zone set_state_flag = sov_nuclear_bombs_settlement_withdrawal_recorded }
				set_country_flag = sov_nuclear_bombs_settlement_terms_applied
				set_country_flag = sov_nuclear_bombs_settlement_withdrawal_recorded
			}
			else_if = {
				limit = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.demilitarize_zone compare = equals } }
				event_target:sov_nuclear_bombs_selected_state = {
					if = { limit = { air_base > 0 } remove_building = { type = air_base level = 1 } }
					else_if = { limit = { arms_factory > 0 } remove_building = { type = arms_factory level = 1 } }
					else_if = { limit = { dockyard > 0 } remove_building = { type = dockyard level = 1 } }
					else_if = { limit = { naval_base > 0 } remove_building = { type = naval_base level = 1 } }
					else_if = { limit = { radar_station > 0 } remove_building = { type = radar_station level = 1 } }
					set_state_flag = sov_nuclear_bombs_demilitarized_zone
					set_state_flag = sov_nuclear_bombs_settlement_demilitarization_recorded
				}

			}
		}

sov_nuclear_bombs_select_standdown_partner = {
	clear_global_event_target = sov_nuclear_bombs_standdown_partner
	random_country = {
		limit = {
			exists = yes
			is_major = yes
			NOT = { tag = SOV }
			NOT = { has_capitulated = yes }
			has_tech = nukes
			num_of_nukes > constant:sov_nuclear_bombs_ledger.zero
			NOT = { is_in_faction_with = SOV }
		}
		save_global_event_target_as = sov_nuclear_bombs_standdown_partner
	}
}

				sov_nuclear_bombs_event_selected_breakaway_site_is_valid_for_actor = yes
				event_target:sov_nuclear_bombs_selected_state = {
					check_variable = {
						var = sov_nuclear_bombs_site_custody_state
						value = constant:sov_nuclear_bombs_custody.breakaway_physical
						compare = equals
					}
				}
			}
			set_country_flag = sov_nuclear_bombs_breakaway_technical_access
			set_country_flag = sov_nuclear_bombs_breakaway_operationalization_succeeded
		}
		else_if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_breakaway_progress_stage value = constant:sov_nuclear_bombs_tuning.breakaway_command_stage compare = equals }
				sov_nuclear_bombs_event_has_breakaway_custody_crisis = yes
				has_country_flag = sov_nuclear_bombs_breakaway_technical_access
				sov_nuclear_bombs_event_selected_breakaway_site_is_valid_for_actor = yes
				event_target:sov_nuclear_bombs_selected_state = {
					check_variable = {
						var = sov_nuclear_bombs_site_custody_state
						value = constant:sov_nuclear_bombs_custody.breakaway_physical
						compare = equals
					}
				}
			}
			set_country_flag = sov_nuclear_bombs_breakaway_command_formed
			set_country_flag = sov_nuclear_bombs_breakaway_operationalization_succeeded
		}
		else_if = {
			limit = {
				check_variable = { var = sov_nuclear_bombs_breakaway_progress_stage value = constant:sov_nuclear_bombs_tuning.breakaway_delivery_stage compare = equals }
					sov_nuclear_bombs_event_has_breakaway_custody_crisis = yes
					has_country_flag = sov_nuclear_bombs_breakaway_command_formed
				sov_nuclear_bombs_event_selected_breakaway_site_has_delivery_capability = yes
				sov_nuclear_bombs_event_selected_breakaway_site_is_valid_for_actor = yes
				event_target:sov_nuclear_bombs_selected_state = {
					check_variable = {
						var = sov_nuclear_bombs_site_custody_state
						value = constant:sov_nuclear_bombs_custody.breakaway_physical
						compare = equals
					}
				}
			}
			event_target:sov_nuclear_bombs_selected_state = { sov_nuclear_bombs_operationalize_breakaway_site = yes }
			if = {
				limit = { event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_breakaway_site_operationalized } }
				set_country_flag = sov_nuclear_bombs_breakaway_delivery_integrated
				set_country_flag = sov_nuclear_bombs_breakaway_operationalization_complete
				set_country_flag = sov_nuclear_bombs_breakaway_operationalization_succeeded
			}
		}
		if = {
			limit = { NOT = { has_country_flag = sov_nuclear_bombs_breakaway_operationalization_succeeded } }
			set_country_flag = sov_nuclear_bombs_breakaway_operationalization_failed
		}
		clr_country_flag = sov_nuclear_bombs_breakaway_operationalization_mission_active
		clear_variable = sov_nuclear_bombs_breakaway_progress_stage
		sov_nuclear_bombs_clear_selected_state = yes
		sov_nuclear_bombs_achievement_refresh_receipts = yes
	}
}
