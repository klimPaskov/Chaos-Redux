# COUNTRY scope. Builds at most three anonymous rival cards from the stored
# participant registry. The card IDs are private viewer-local variables; the
# board reveals only an evidence band and uses the ID for a later target
# resolution. This helper never scans the world.
chaosx_nr25_prepare_rival_cards = {
	set_temp_variable = { chaosx_nr25_previous_rival_target_id = chaosx_nr25_rival_target_id }
	set_variable = { chaosx_nr25_rival_card_count = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_1_id = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_2_id = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_3_id = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_1_state = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_2_state = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_3_state = constant:chaosx_nr25_value.zero }
	clr_country_flag = chaosx_nr25_rival_slot_1_visible
	clr_country_flag = chaosx_nr25_rival_slot_2_visible
	clr_country_flag = chaosx_nr25_rival_slot_3_visible
	clr_country_flag = chaosx_nr25_rival_slot_1_selected
	clr_country_flag = chaosx_nr25_rival_slot_2_selected
	clr_country_flag = chaosx_nr25_rival_slot_3_selected
	clr_country_flag = chaosx_nr25_rival_target_selected
	for_each_scope_loop = {
		array = global.chaosx_nr25_participants
		if = {
			limit = {
				chaosx_nr25_target_participant_valid = yes
				check_variable = { var = ROOT.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.three compare = less_than }
			}
			ROOT = { add_to_variable = { chaosx_nr25_rival_card_count = constant:chaosx_nr25_value.one } }
			if = {
				limit = { check_variable = { var = ROOT.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.one compare = equals } }
				set_variable = { ROOT.chaosx_nr25_rival_slot_1_id = THIS.id }
				ROOT = { set_country_flag = chaosx_nr25_rival_slot_1_visible }
				if = {
					limit = { check_variable = { var = ROOT.chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
					ROOT = { set_country_flag = chaosx_nr25_rival_slot_1_selected set_country_flag = chaosx_nr25_rival_target_selected }
				}
			}
			else_if = {
				limit = { check_variable = { var = ROOT.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.two compare = equals } }
				set_variable = { ROOT.chaosx_nr25_rival_slot_2_id = THIS.id }
				ROOT = { set_country_flag = chaosx_nr25_rival_slot_2_visible }
				if = {
					limit = { check_variable = { var = ROOT.chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
					ROOT = { set_country_flag = chaosx_nr25_rival_slot_2_selected set_country_flag = chaosx_nr25_rival_target_selected }
				}
			}
			else_if = {
				limit = { check_variable = { var = ROOT.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.three compare = equals } }
				set_variable = { ROOT.chaosx_nr25_rival_slot_3_id = THIS.id }
				ROOT = { set_country_flag = chaosx_nr25_rival_slot_3_visible }
				if = {
					limit = { check_variable = { var = ROOT.chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
					ROOT = { set_country_flag = chaosx_nr25_rival_slot_3_selected set_country_flag = chaosx_nr25_rival_target_selected }
				}
			}
		}
	}
	if = {
		limit = { NOT = { has_country_flag = chaosx_nr25_rival_target_selected } }
		set_variable = { chaosx_nr25_rival_target_id = constant:chaosx_nr25_value.zero }
	}
}

# COUNTRY scope. Resolves a board-selected rival ID through the stored
# participant registry immediately before a hostile action. The board never
# stores a country scope in UI state, and a withdrawn target is cleared rather
# than silently retargeted.
chaosx_nr25_resolve_rival_target_id = {
	if = {
		limit = {
			has_variable = chaosx_nr25_rival_target_id
			check_variable = {
				var = chaosx_nr25_rival_target_id
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
			NOT = { has_event_target = chaosx_nr25_rival_target_scope }
		}
		for_each_scope_loop = {
			array = global.chaosx_nr25_participants
			if = {
				limit = {
					chaosx_nr25_target_participant_valid = yes
					check_variable = { var = ROOT.chaosx_nr25_rival_target_id value = THIS.id compare = equals }
				}
				save_event_target_as = chaosx_nr25_rival_target_scope
				ROOT = { set_variable = { chaosx_nr25_rival_target = THIS.id } }
			}
		}
	}
	if = {
		limit = {
			has_variable = chaosx_nr25_rival_target_id
			check_variable = {
				var = chaosx_nr25_rival_target_id
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
			NOT = { has_variable = chaosx_nr25_rival_target }
			NOT = { has_event_target = chaosx_nr25_rival_target_scope }
		}
		set_variable = { chaosx_nr25_rival_target_id = constant:chaosx_nr25_value.zero }
		clr_country_flag = chaosx_nr25_rival_target_selected
	}
}

# COUNTRY scope. Selects a bounded AI-major roster after all human invitations
# have been answered. The only non-major exception is the named KRG tag.
chaosx_nr25_select_bounded_ai_roster = {
	if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_value.zero compare = equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_zero_humans }
	}
	else_if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_value.one compare = equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_one_human }
	}
	else_if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_roster.human_count_three compare = less_than_or_equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_three_humans }
	}
	else_if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_roster.human_count_six compare = less_than_or_equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_six_humans }
	}
	else = { set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_many_humans } }

	while_loop_effect = {
		limit = {
			check_variable = { var = global.chaosx_nr25_ai_slots value = constant:chaosx_nr25_value.zero compare = greater_than }
			any_country = { chaosx_nr25_ai_candidate_available = yes }
		}
		random_country = {
			limit = { chaosx_nr25_ai_candidate_available = yes }
			chaosx_nr25_register_participant = yes
			set_temp_variable = { chaosx_nr25_action = constant:chaosx_nr25_action.authorize }
			chaosx_nr25_execute_action = yes
			subtract_from_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_value.one }
		}
	}
}

# COUNTRY scope. Adds the public Chaos meter pressure for one completed
# expedition action. This is deliberately routed through the shared meter
# helper so Event 025 does not create a second Chaos system.
chaosx_nr25_raise_chaos = {
	set_temp_variable = { chaos_change = constant:chaosx_nr25_tuning.action_chaos }
	set_temp_variable = { chaos_history_reason_custom = constant:chaos_meter_history_reason.generic.increase }
	add_chaos_meter_value = yes
	add_to_variable = { global.chaosx_nr25_chaos = constant:chaosx_nr25_tuning.action_chaos }
}

# COUNTRY scope. Registers one expedition casualty in the shared Chaos deaths
# history and leaves a durable rescue-memory marker for Evolution III. The
# amount is intentionally one per verified accident or rival seizure; callers
# cannot bypass the shared ledger with an unrelated Event 016 cause.
chaosx_nr25_record_casualties = {
	set_temp_variable = { chaosx_nr25_casualty_amount = constant:chaosx_nr25_value.one }
	add_to_variable = { chaosx_nr25_casualties = chaosx_nr25_casualty_amount }
	set_country_flag = chaosx_nr25_casualty_memory_ready
	set_temp_variable = { chaos_deaths_change = chaosx_nr25_casualty_amount }
	set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.military_casualties }
	set_temp_variable = { chaos_deaths_is_civilian = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaos_deaths_target_country = THIS }
	set_temp_variable = { chaos_deaths_has_target_country = constant:chaosx_nr25_value.one }
	chaos_meter_register_deaths = yes
}

# COUNTRY scope. Emergency evacuation failures record the configured field
# loss through the same shared deaths API without duplicating an inline call
# site. The batch is deliberately bounded to the two-casualty contract.
chaosx_nr25_record_evacuate_failure_casualties = {
	set_temp_variable = { chaosx_nr25_evacuate_failure_casualties = constant:chaosx_nr25_cost.mission_evacuate_failure_casualties }
	if = {
		limit = { check_variable = { var = chaosx_nr25_evacuate_failure_casualties value = constant:chaosx_nr25_value.two compare = greater_than_or_equals } }
		chaosx_nr25_record_casualties = yes
	}
	if = {
		limit = { check_variable = { var = chaosx_nr25_evacuate_failure_casualties value = constant:chaosx_nr25_value.one compare = greater_than_or_equals } }
		chaosx_nr25_record_casualties = yes
	}
}

# COUNTRY scope. Closes the race when every registered expedition has been
# withdrawn, invalidated, or failed its final mission. The current country is
# the sole recipient of the bounded closure notice; no world scan is needed.
chaosx_nr25_close_if_no_active_participants = {
	if = {
		limit = {
			has_global_flag = chaosx_nr25_active
			check_variable = { var = global.chaosx_nr25_active_participant_count value = constant:chaosx_nr25_value.zero compare = less_than_or_equals }
			check_variable = { var = global.chaosx_nr25_final_resolution_pending value = constant:chaosx_nr25_value.zero compare = equals }
		}
		clr_global_flag = chaosx_nr25_active
		clr_global_flag = chaosx_nr25_entry_open
		chaosx_nr25_clear_pending_evolutions = yes
		set_global_flag = chaosx_nr25_closed_after_expeditions
		set_variable = { global.chaosx_nr25_phase = constant:chaosx_nr25_phase.closed }
		country_event = { id = chaosx.nr25.298 days = 1 }
	}
}

# COUNTRY scope. Lexical comparator for candidates that become valid on the
# same day. The date window is handled by the caller; this helper only applies
# the accepted evidence ordering and ends with stable participant identifiers.
chaosx_nr25_candidate_is_better_same_day = {
	set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.zero }
	if = {
		limit = {
			check_variable = {
				var = chaosx_nr25_survey_evidence
				value = global.chaosx_nr25_final_candidate_survey
				compare = greater_than
			}
		}
		set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
	}
	else_if = {
		limit = {
			check_variable = {
				var = chaosx_nr25_survey_evidence
				value = global.chaosx_nr25_final_candidate_survey
				compare = equals
			}
		}
		if = {
			limit = { check_variable = { var = chaosx_nr25_logistics_readiness value = global.chaosx_nr25_final_candidate_readiness compare = greater_than } }
			set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_logistics_readiness value = global.chaosx_nr25_final_candidate_readiness compare = equals } }
			if = {
				limit = { check_variable = { var = chaosx_nr25_outpost_security value = global.chaosx_nr25_final_candidate_outpost compare = greater_than } }
				set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
			}
			else_if = {
				limit = { check_variable = { var = chaosx_nr25_outpost_security value = global.chaosx_nr25_final_candidate_outpost compare = equals } }
				if = {
					limit = { check_variable = { var = chaosx_nr25_exposure_risk value = global.chaosx_nr25_final_candidate_exposure compare = less_than } }
					set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
				}
				else_if = {
					limit = { check_variable = { var = chaosx_nr25_exposure_risk value = global.chaosx_nr25_final_candidate_exposure compare = equals } }
					if = {
						limit = { check_variable = { var = chaosx_nr25_final_ready_date value = global.chaosx_nr25_final_candidate_ready_date compare = less_than } }
						set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
					}
					else_if = {
						limit = { check_variable = { var = chaosx_nr25_final_ready_date value = global.chaosx_nr25_final_candidate_ready_date compare = equals } }
						if = {
							limit = { check_variable = { var = chaosx_nr25_entry_order value = global.chaosx_nr25_final_candidate_entry compare = less_than } }
							set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
						}
						# Equal registry positions are impossible for valid participants.
						# No random seed is used as a hidden fifth tie-breaker.
					}
				}
			}
		}
	}
}
