# GER/JAP/SOV institutions: one national slot, paid monthly work, finite single-use awards.
# Foundation owns population, payment ceilings and annual award count. No institution creates population or research slots.

camp_admin_country_initialize = {
	if = { limit = { camp_administration_civil_country_supported = yes } camp_administration_civil_initialize = yes camp_admin_country_refresh_civil_institution_display = yes }
	if = { limit = { camp_admin_country_is_major_kit = yes } camp_admin_country_cancel_routine_missions = yes }
	if = {
		limit = { camp_admin_country_is_major_kit = yes NOT = { has_country_flag = camp_admin_country_initialized } }
		set_country_flag = camp_admin_country_initialized
		set_variable = { camp_admin_country_sponsor = constant:camp_admin_country.sponsor_economic }
		set_variable = { camp_admin_country_authority_pressure = 0 }
		set_variable = { camp_admin_country_program = constant:camp_admin_country_program.none }
		set_variable = { camp_admin_institution_progress = 0 }
		set_variable = { camp_admin_institution_target = constant:camp_admin_country.program_months }
		set_variable = { camp_admin_research_progress = 0 }
		set_variable = { camp_admin_research_target = constant:camp_admin_country.program_months }
		if = { limit = { camp_admin_country_regime_valid = yes } set_variable = { camp_admin_mandate = constant:camp_admin_mandate.coercive } }
		if = { limit = { original_tag = GER } set_variable = { camp_admin_priority = constant:camp_admin_priority.industry } }
		else_if = { limit = { original_tag = JAP } set_variable = { camp_admin_priority = constant:camp_admin_priority.works } }
		else_if = { limit = { original_tag = SOV } set_variable = { camp_admin_priority = constant:camp_admin_priority.extraction } }
		if = {
			limit = { OR = { has_active_camp_network = yes has_country_flag = germany_mengele_program_authorized has_country_flag = japan_ishii_authority_route } }
			set_country_flag = camp_admin_country_legacy_active
		}
		if = {
			limit = { OR = { has_country_flag = germany_mengele_program_authorized has_country_flag = germany_mengele_program_restricted has_country_flag = japan_ishii_authority_route } }
			set_country_flag = camp_admin_country_programs_authorized
		}
	}
}

# Country, literal PROGRAM constant. Select first qualifying registered site; no random target weight.
camp_admin_country_start_program = {
	set_variable = { camp_admin_country_program = $PROGRAM$ }
	set_variable = { camp_admin_institution_progress = 0 }
	set_variable = { camp_admin_institution_target = constant:camp_admin_country.program_months }
	clear_variable = camp_admin_institution_state_id
	for_each_scope_loop = {
		array = camp_active_site_states
		if = {
			limit = {
				camp_admin_site_is_operational = yes
				check_variable = { camp_admin_detainees_k > 0 }
				OR = { PREV = { NOT = { original_tag = SOV } } gulag_labor_camp_network > 0 }
				PREV = { NOT = { has_variable = camp_admin_institution_state_id } }
			}
			PREV = { set_variable = { camp_admin_institution_state_id = PREV } }
		}
	}
	if = {
		limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.ger_auschwitz } }
		set_variable = { camp_admin_institution_state_id = 88 }
	}
	else_if = {
		limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_pingfang } }
		set_variable = { camp_admin_institution_state_id = 328 }
	}
	else_if = {
		limit = { original_tag = JAP }
		clear_variable = camp_admin_institution_state_id
		for_each_scope_loop = {
			array = camp_active_site_states
			if = {
				limit = {
					camp_admin_site_is_operational = yes
					camp_admin_country_china_geography = yes
					check_variable = { camp_admin_detainees_k > 0 }
					PREV = { NOT = { has_variable = camp_admin_institution_state_id } }
					OR = { PREV = { NOT = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_transport } } } camp_admin_country_census_jap_industry_host = yes }
				}
				PREV = { set_variable = { camp_admin_institution_state_id = PREV } }
			}
		}
	}
	if = {
		limit = { has_variable = camp_admin_institution_state_id }
		set_country_flag = camp_admin_institution_slot_active
		if = {
			limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.sov_engineer_release } }
			var:camp_admin_institution_state_id = {
				set_temp_variable = { camp_admin_engineer_cohort_k = camp_admin_detainees_k }
				clamp_temp_variable = { var = camp_admin_engineer_cohort_k min = 0 max = constant:camp_admin_country.engineer_cohort_k }
			}
			set_variable = { camp_admin_country_engineer_cohort_k = camp_admin_engineer_cohort_k }
		}
	}
	else = { set_variable = { camp_admin_country_program = constant:camp_admin_country_program.none } }
}

camp_admin_country_select_program = {
	if = {
		limit = { has_country_flag = camp_admin_country_programs_authorized NOT = { has_country_flag = camp_admin_development_paused } NOT = { has_country_flag = camp_admin_institution_slot_active } camp_admin_country_program_available = yes }
		if = { limit = { camp_admin_program_ger_inspectorate_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.ger_inspectorate } }
		else_if = { limit = { camp_admin_program_ger_wvha_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.ger_wvha } }
		else_if = { limit = { camp_admin_program_ger_auschwitz_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.ger_auschwitz } }
		else_if = { limit = { camp_admin_program_jap_transport_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.jap_transport } }
		else_if = { limit = { camp_admin_program_jap_army_review_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.jap_army_review } }
		else_if = { limit = { camp_admin_program_jap_pingfang_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.jap_pingfang } }
		else_if = { limit = { camp_admin_program_sov_design_bureaux_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.sov_design_bureaux } }
		else_if = { limit = { camp_admin_program_sov_wartime_industry_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.sov_wartime_industry } }
		else_if = { limit = { camp_admin_program_sov_engineer_release_open = yes } camp_admin_country_start_program = { PROGRAM = constant:camp_admin_country_program.sov_engineer_release } }
	}
}

# Fixed program name plus engine category anti-stacking. Research progress never reads deaths.
camp_admin_country_award_program = {
	if = {
		limit = {
			NOT = { has_country_flag = camp_admin_done_$PROGRAM$ }
			NOT = { has_tech_bonus = { category = $CATEGORY$ } }
			camp_admin_country_program_still_valid = yes
		}
		set_temp_variable = { camp_admin_research_requested_bonus = $BONUS$ }
		set_temp_variable = { camp_admin_research_contract = 1 }
		set_temp_variable = { camp_admin_research_program_valid = 1 }
		camp_admin_try_claim_research_award = yes
		if = {
			limit = { check_variable = { camp_admin_research_award_accepted = 1 } }
			meta_effect = {
				text = { add_tech_bonus = { name = camp_admin_bonus_$PROGRAM$ bonus = [BONUS_VALUE] uses = 1 category = $CATEGORY$ } }
				BONUS_VALUE = "[?camp_admin_research_award_bonus]"
			}
			set_country_flag = camp_admin_done_$PROGRAM$
			if = {
				limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.sov_engineer_release } }
				set_temp_variable = { camp_admin_release_requested_k = camp_admin_country_engineer_cohort_k }
				var:camp_admin_institution_state_id = { camp_admin_release_detainees = yes }
				set_variable = { camp_admin_country_last_engineers_released_k = camp_admin_release_actual_k }
				clear_variable = camp_admin_country_engineer_cohort_k
				set_country_flag = camp_admin_country_engineers_released
			}
			set_variable = { camp_admin_country_last_program = camp_admin_country_program }
			set_variable = { camp_admin_country_program = constant:camp_admin_country_program.none }
			clear_variable = camp_admin_institution_state_id
			clr_country_flag = camp_admin_country_institution_paused
		}
	}
}

	if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.ger_inspectorate } } camp_admin_country_award_program = { PROGRAM = ger_inspectorate CATEGORY = industry BONUS = constant:camp_admin_country.bonus_standard } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.ger_wvha } } camp_admin_country_award_program = { PROGRAM = ger_wvha CATEGORY = industry BONUS = constant:camp_admin_country.bonus_advanced } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.ger_auschwitz } } camp_admin_country_award_program = { PROGRAM = ger_auschwitz CATEGORY = support_tech BONUS = constant:camp_admin_country.bonus_standard } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_transport } } camp_admin_country_award_program = { PROGRAM = jap_transport CATEGORY = industry BONUS = constant:camp_admin_country.bonus_extended } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_army_review } } camp_admin_country_award_program = { PROGRAM = jap_army_review CATEGORY = support_tech BONUS = constant:camp_admin_country.bonus_standard } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.jap_pingfang } } camp_admin_country_award_program = { PROGRAM = jap_pingfang CATEGORY = support_tech BONUS = constant:camp_admin_country.bonus_standard } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.sov_design_bureaux } } camp_admin_country_award_program = { PROGRAM = sov_design_bureaux CATEGORY = engineers_tech BONUS = constant:camp_admin_country.bonus_extended } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.sov_wartime_industry } } camp_admin_country_award_program = { PROGRAM = sov_wartime_industry CATEGORY = industry BONUS = constant:camp_admin_country.bonus_advanced } }
	else_if = { limit = { check_variable = { camp_admin_country_program = constant:camp_admin_country_program.sov_engineer_release } } camp_admin_country_award_program = { PROGRAM = sov_engineer_release CATEGORY = engineers_tech BONUS = constant:camp_admin_country.bonus_standard } }
}

camp_admin_country_tick_institution = {
	set_country_flag = camp_admin_country_institution_paused
	if = {
		limit = { camp_admin_country_institution_supported = yes }
		if = {
			limit = { check_variable = { var = camp_admin_institution_progress value = camp_admin_institution_target compare = greater_than_or_equals } }
			camp_admin_country_finish_program = yes
		}
		else = {
			set_temp_variable = { camp_admin_quote_support = constant:camp_admin_country.monthly_support }
			set_temp_variable = { camp_admin_quote_pp = constant:camp_admin_country.monthly_pp }
			camp_admin_try_pay_project = yes
			if = {
				limit = { check_variable = { camp_admin_payment_accepted = 1 } }
				clr_country_flag = camp_admin_country_institution_paused
				if = { limit = { check_variable = { var = camp_admin_country_authority_pressure value = constant:camp_admin_country.authority_contested compare = greater_than_or_equals } } add_to_variable = { camp_admin_institution_progress = constant:camp_admin_country.contested_progress } }
				else = { add_to_variable = { camp_admin_institution_progress = constant:camp_admin_country.monthly_progress } }
				clamp_variable = { var = camp_admin_institution_progress min = 0 max = constant:camp_admin_country.program_months }
				if = {
					limit = { check_variable = { var = camp_admin_institution_progress value = camp_admin_institution_target compare = greater_than_or_equals } }
					camp_admin_country_finish_program = yes
				}
			}
		}
	}
}

# Economic ministries, military commands and security authorities compete for the same commitments.
camp_admin_country_tick_authority = {
	if = {
		limit = { has_country_flag = camp_admin_country_programs_authorized }
		if = {
			limit = { check_variable = { camp_admin_country_sponsor = constant:camp_admin_country.sponsor_security } }
			add_to_variable = { camp_admin_country_authority_pressure = constant:camp_admin_country.authority_monthly }
			if = { limit = { original_tag = GER } add_to_variable = { ss_archive_control = constant:camp_admin_country.authority_monthly } }
			else_if = { limit = { original_tag = JAP } add_to_variable = { kwantung_autonomy = constant:camp_admin_country.authority_monthly } }
			else_if = { limit = { original_tag = SOV } add_to_variable = { nkvd_authority = constant:camp_admin_country.authority_monthly } }
		}
		else_if = {
			limit = { check_variable = { camp_admin_country_sponsor = constant:camp_admin_country.sponsor_economic } has_country_flag = camp_admin_institution_slot_active }
			add_to_variable = { camp_admin_country_authority_pressure = constant:camp_admin_country.monthly_progress }
		}
	}
	if = {
		limit = { has_country_flag = camp_admin_country_oversight_active }
		set_temp_variable = { camp_admin_quote_support = constant:camp_admin_country.monthly_support }
		set_temp_variable = { camp_admin_quote_pp = constant:camp_admin_country.monthly_pp }
		camp_admin_try_pay_project = yes
		if = {
			limit = { check_variable = { camp_admin_payment_accepted = 1 } }
			add_to_variable = { camp_admin_country_review_progress = constant:camp_admin_country.monthly_progress }
			if = {
				limit = { check_variable = { var = camp_admin_country_review_progress value = constant:camp_admin_country.oversight_months compare = greater_than_or_equals } }
				add_to_variable = { camp_admin_country_authority_pressure = constant:camp_admin_country.authority_review_relief }
				if = { limit = { original_tag = GER } add_to_variable = { ss_archive_control = constant:camp_admin_country.authority_review_relief } clamp_variable = { var = ss_archive_control min = 0 max = constant:camp_admin_country.authority_max } }
				else_if = { limit = { original_tag = JAP } add_to_variable = { kwantung_autonomy = constant:camp_admin_country.authority_review_relief } clamp_variable = { var = kwantung_autonomy min = 0 max = constant:camp_admin_country.authority_max } }
				else_if = { limit = { original_tag = SOV } add_to_variable = { nkvd_authority = constant:camp_admin_country.authority_review_relief } clamp_variable = { var = nkvd_authority min = 0 max = constant:camp_admin_country.authority_max } }
				clr_country_flag = camp_admin_country_oversight_active
				clr_country_flag = camp_admin_institution_slot_active
				set_country_flag = camp_admin_country_oversight_completed
			}
		}
	}
	clamp_variable = { var = camp_admin_country_authority_pressure min = 0 max = constant:camp_admin_country.authority_max }
}

camp_admin_country_monthly = {
	if = { limit = { camp_administration_civil_country_supported = yes } camp_administration_civil_monthly = yes camp_admin_country_refresh_civil_institution_display = yes }
	camp_admin_prepare_month_key = yes
	if = {
		limit = {
			camp_admin_country_is_major_kit = yes
			OR = { NOT = { has_variable = camp_admin_country_last_month_key } NOT = { check_variable = { camp_admin_country_last_month_key = camp_admin_month_key } } }
		}
		camp_admin_country_initialize = yes
		set_variable = { camp_admin_country_last_month_key = camp_admin_month_key }
		if = {
			limit = { NOT = { camp_admin_country_regime_valid = yes } NOT = { has_country_flag = camp_admin_country_reformed } has_country_flag = camp_admin_country_legacy_active }
			camp_admin_country_reform = yes
		}
		if = {
			limit = { is_ai = yes camp_admin_country_action_1_available = yes }
			camp_admin_country_action_1 = yes
		}
		if = {
			limit = { camp_admin_country_regime_valid = yes }
			if = {
				limit = { is_ai = yes NOT = { has_country_flag = camp_admin_institution_slot_active } camp_admin_country_can_commit = yes check_variable = { var = camp_admin_country_authority_pressure value = constant:camp_admin_country.authority_contested compare = greater_than_or_equals } }
				camp_admin_country_start_oversight = yes
			}
			camp_admin_country_select_program = yes
			camp_admin_country_tick_authority = yes
			if = { limit = { has_country_flag = camp_admin_institution_slot_active NOT = { has_country_flag = camp_admin_country_oversight_active } } camp_admin_country_tick_institution = yes }
		}
		else_if = { limit = { OR = { has_country_flag = camp_admin_country_programs_authorized has_country_flag = camp_admin_country_oversight_active } } camp_admin_country_halt = yes }
		if = {
			limit = { is_ai = yes camp_admin_country_archives_available = yes }
			camp_admin_country_start_archives = yes
		}
		camp_admin_country_tick_aftermath = yes
		set_variable = { camp_admin_research_progress = camp_admin_institution_progress }
		set_variable = { camp_admin_research_target = camp_admin_institution_target }
	}
}

		else_if = { limit = { OR = { has_country_flag = camp_admin_country_programs_authorized has_country_flag = camp_admin_country_oversight_active } } camp_admin_country_halt = yes }
		else = { camp_admin_country_reform = yes }
	}
}

# Display adapters copy actual civil progress; they never advance a civil project.

		set_variable = { camp_admin_institution_progress = camp_admin_civil_review_progress }
		set_variable = { camp_admin_institution_target = constant:camp_admin_civil.review_months }

	set_variable = { camp_admin_research_target = camp_admin_institution_target }
}

# Deterministic AI policy uses the same quoted, paid budget command as player controls.
# A three-month hold and distinct raise/lower thresholds prevent monthly policy oscillation.
camp_admin_country_prepare_ai_budget = {
	camp_admin_prepare_month_key = yes
	if = {
		limit = {
			is_ai = yes
			camp_admin_country_is_major_kit = yes
			NOT = { has_country_flag = camp_admin_development_paused }
			OR = { NOT = { has_variable = camp_admin_country_ai_policy_next_month } check_variable = { var = camp_admin_month_key value = camp_admin_country_ai_policy_next_month compare = greater_than_or_equals } }
		}
		set_temp_variable = { camp_admin_ai_maintenance_k = 0 }
		for_each_scope_loop = {
			array = camp_active_site_states
			if = { limit = { camp_admin_site_is_operational = yes } add_to_temp_variable = { camp_admin_ai_maintenance_k = camp_admin_detainees_k } }
		}
		for_each_scope_loop = { array = camp_admin_civilian_states add_to_temp_variable = { camp_admin_ai_maintenance_k = camp_admin_civilian_workforce_k } }
		multiply_temp_variable = { camp_admin_ai_maintenance_k = constant:camp_admin_value.maintenance_support_per_k }
		set_temp_variable = { camp_admin_ai_low_stock = camp_admin_ai_maintenance_k }
		add_to_temp_variable = { camp_admin_ai_low_stock = constant:camp_admin_value.support_reserve }
		set_temp_variable = { camp_admin_ai_high_stock = camp_admin_ai_maintenance_k }
		add_to_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_value.project_support }
		add_to_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_country.monthly_support }
		multiply_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_country.ai_commitment_buffer_months }
		add_to_temp_variable = { camp_admin_ai_high_stock = constant:camp_admin_value.support_reserve }
		set_temp_variable = { camp_admin_ai_budget_before = camp_admin_budget }
		if = {
			limit = { check_variable = { camp_admin_budget > constant:camp_admin_budget.maintenance } check_variable = { num_equipment@support_equipment < camp_admin_ai_low_stock } }
			set_temp_variable = { camp_admin_control_requested = constant:camp_admin_budget.maintenance }
			camp_admin_set_budget = yes
		}
		else_if = {
			limit = {
				check_variable = { camp_admin_budget = constant:camp_admin_budget.maintenance }
				check_variable = { var = num_equipment@support_equipment value = camp_admin_ai_high_stock compare = greater_than_or_equals }
				OR = {
					camp_admin_country_program_available = yes
					camp_admin_country_archives_available = yes
					any_of_scopes = { array = camp_active_site_states camp_admin_has_supported_workforce = yes check_variable = { camp_admin_development_level < constant:camp_admin_value.max_development } }
				}
			}
			set_temp_variable = { camp_admin_control_requested = constant:camp_admin_budget.development }
			camp_admin_set_budget = yes
		}
		if = {
			limit = { NOT = { check_variable = { camp_admin_budget = camp_admin_ai_budget_before } } }
			set_variable = { camp_admin_country_ai_policy_next_month = camp_admin_month_key }
			add_to_variable = { camp_admin_country_ai_policy_next_month = constant:camp_admin_country.ai_policy_hold_months }
		}
	}
}
