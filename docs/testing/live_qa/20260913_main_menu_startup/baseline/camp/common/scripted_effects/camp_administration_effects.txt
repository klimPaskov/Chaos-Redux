# Camp administration foundation.
# Bounded country/site callbacks; explicit finite admission; measured death receipts;
# maintenance before development; distinct productive allocations and reform replacement.
# Public contracts, units, defaults and parent hooks: camp_administration_effects.md.

camp_admin_initialize_records = {
	if = { limit = { NOT = { has_variable = camp_admin_total_deaths_k } } set_variable = { camp_admin_total_deaths_k = 0 } }
	if = { limit = { NOT = { has_variable = camp_admin_month_deaths_k } } set_variable = { camp_admin_month_deaths_k = 0 } }
	if = { limit = { NOT = { has_variable = camp_admin_total_released_k } } set_variable = { camp_admin_total_released_k = 0 } }
	camp_admin_initialize_cause_records = { CAUSE = custody }
	camp_admin_initialize_cause_records = { CAUSE = labor }
	camp_admin_initialize_cause_records = { CAUSE = violence }
	camp_admin_initialize_cause_records = { CAUSE = institutional }
}
camp_admin_initialize_cause_records = {

	if = { limit = { NOT = { has_variable = camp_admin_total_$CAUSE$_deaths_k } } set_variable = { camp_admin_total_$CAUSE$_deaths_k = 0 } }
}

		set_variable = { camp_admin_budget_pp_remaining = 0 }
		set_variable = { camp_admin_economy_progress = 0 }
		set_variable = { camp_admin_economy_target = 0 }

camp_admin_initialize_site = {
	if = {
		limit = { NOT = { has_state_flag = camp_admin_initialized } has_variable = genocide_responsible_country }
		set_state_flag = camp_admin_initialized
		set_variable = { camp_admin_responsible_country = genocide_responsible_country }
		set_variable = { camp_admin_detainees_k = 0 }
		set_variable = { camp_admin_total_admitted_k = 0 }
		set_variable = { camp_admin_external_attrition_k = 0 }
		set_variable = { camp_admin_development_level = 0 }
		set_variable = { camp_admin_support_ratio = 0 }
		set_variable = { camp_admin_site_output_scale = 1 }
		set_variable = { camp_admin_civilian_workforce_k = 0 }
		set_variable = { camp_admin_civilian_target_k = 0 }
		set_variable = { camp_admin_civilian_replacement = 0 }
		set_variable = { camp_admin_role = constant:camp_admin_role.custody }
		if = { limit = { has_state_flag = camp_rework_expanded_labor_site } set_variable = { camp_admin_role = constant:camp_admin_role.labor } }
		if = { limit = { gulag_labor_camp_network > 0 } set_variable = { camp_admin_role = constant:camp_admin_role.penal } }
		if = { limit = { extermination_camp > 0 } set_variable = { camp_admin_role = constant:camp_admin_role.killing } }
	}
	camp_admin_initialize_records = yes
	camp_admin_refresh_capacity = yes
}
camp_admin_refresh_capacity = {
	clamp_variable = { var = camp_admin_development_level min = 0 max = constant:camp_admin_value.max_development }
	set_variable = { camp_admin_capacity_k = building_level@concentration_camp }
	add_to_variable = { camp_admin_capacity_k = building_level@gulag_labor_camp_network }
	add_to_variable = { camp_admin_capacity_k = building_level@extermination_camp }
	multiply_variable = { camp_admin_capacity_k = constant:camp_admin_value.capacity_per_building_k }
	if = {
		limit = { camp_admin_site_is_operational = yes check_variable = { camp_admin_capacity_k < constant:camp_admin_value.marker_capacity_k } }
		set_variable = { camp_admin_capacity_k = constant:camp_admin_value.marker_capacity_k }
	}
	set_temp_variable = { camp_admin_developed_capacity_k = camp_admin_development_level }
	multiply_temp_variable = { camp_admin_developed_capacity_k = constant:camp_admin_value.capacity_per_building_k }
	add_to_variable = { camp_admin_capacity_k = camp_admin_developed_capacity_k }
	set_temp_variable = { camp_admin_population_headroom_k = state_population_k }
	subtract_from_temp_variable = { camp_admin_population_headroom_k = constant:camp_admin_value.minimum_population_k }
	clamp_temp_variable = { var = camp_admin_population_headroom_k min = 0 }
	# External population loss cannot leave a cohort larger than its host population.
	# This is reconciliation only: no second death debit or invented camp cause.
	if = {
		limit = { check_variable = { camp_admin_detainees_k > camp_admin_population_headroom_k } }
		set_temp_variable = { camp_admin_missing_k = camp_admin_detainees_k }
		subtract_from_temp_variable = { camp_admin_missing_k = camp_admin_population_headroom_k }
		add_to_variable = { camp_admin_external_attrition_k = camp_admin_missing_k }
		set_variable = { camp_admin_detainees_k = camp_admin_population_headroom_k }
	}
	clamp_variable = { var = camp_admin_capacity_k min = 0 max = camp_admin_population_headroom_k }
}
camp_admin_admit_detainees = {
	set_temp_variable = { camp_admin_admission_accepted_k = 0 }
	camp_admin_refresh_capacity = yes
	if = {
		limit = {
			camp_admin_site_is_operational = yes
			check_variable = { camp_admin_admission_proven = 1 }
			check_variable = { camp_admin_admission_requested_k > 0 }
			check_variable = { camp_admin_admission_source_k > 0 }
		}
		set_temp_variable = { camp_admin_admission_headroom_k = state_population_k }
		multiply_temp_variable = { camp_admin_admission_headroom_k = constant:camp_admin_value.maximum_population_share }
		subtract_from_temp_variable = { camp_admin_admission_headroom_k = camp_admin_detainees_k }
		set_temp_variable = { camp_admin_uncommitted_k = camp_admin_population_headroom_k }
		subtract_from_temp_variable = { camp_admin_uncommitted_k = camp_admin_detainees_k }
		if = { limit = { has_variable = camp_occ_remaining_k } subtract_from_temp_variable = { camp_admin_uncommitted_k = camp_occ_remaining_k } }
		clamp_temp_variable = { var = camp_admin_uncommitted_k min = 0 }
		set_temp_variable = { camp_admin_free_capacity_k = camp_admin_capacity_k }
		subtract_from_temp_variable = { camp_admin_free_capacity_k = camp_admin_detainees_k }
		clamp_temp_variable = { var = camp_admin_free_capacity_k min = 0 }
		clamp_temp_variable = { var = camp_admin_admission_headroom_k min = 0 max = camp_admin_uncommitted_k }
		clamp_temp_variable = { var = camp_admin_admission_headroom_k min = 0 max = camp_admin_free_capacity_k }
		set_temp_variable = { camp_admin_admission_accepted_k = camp_admin_admission_requested_k }
		clamp_temp_variable = { var = camp_admin_admission_accepted_k min = 0 max = camp_admin_admission_headroom_k }
		clamp_temp_variable = { var = camp_admin_admission_accepted_k min = 0 max = camp_admin_admission_source_k }
		add_to_variable = { camp_admin_detainees_k = camp_admin_admission_accepted_k }
		add_to_variable = { camp_admin_total_admitted_k = camp_admin_admission_accepted_k }
	}
	set_temp_variable = { camp_admin_admission_requested_k = 0 }
	set_temp_variable = { camp_admin_admission_source_k = 0 }
	set_temp_variable = { camp_admin_admission_proven = 0 }
}

# Calendar guard: meta dates use current year, avoiding an approximate 30-day month.
camp_admin_prepare_month_key = {
	set_temp_variable = { camp_admin_month_key = global.year }
	multiply_temp_variable = { camp_admin_month_key = constant:camp_admin_value.months_per_year }
	set_temp_variable = { camp_admin_month_number = 12 }
	meta_effect = {
		text = {
			if = { limit = { date < "[YEAR].2.1" } set_temp_variable = { camp_admin_month_number = 1 } }
			else_if = { limit = { date < "[YEAR].3.1" } set_temp_variable = { camp_admin_month_number = 2 } }
			else_if = { limit = { date < "[YEAR].4.1" } set_temp_variable = { camp_admin_month_number = 3 } }
			else_if = { limit = { date < "[YEAR].5.1" } set_temp_variable = { camp_admin_month_number = 4 } }
			else_if = { limit = { date < "[YEAR].6.1" } set_temp_variable = { camp_admin_month_number = 5 } }
			else_if = { limit = { date < "[YEAR].7.1" } set_temp_variable = { camp_admin_month_number = 6 } }
			else_if = { limit = { date < "[YEAR].8.1" } set_temp_variable = { camp_admin_month_number = 7 } }
			else_if = { limit = { date < "[YEAR].9.1" } set_temp_variable = { camp_admin_month_number = 8 } }
			else_if = { limit = { date < "[YEAR].10.1" } set_temp_variable = { camp_admin_month_number = 9 } }
			else_if = { limit = { date < "[YEAR].11.1" } set_temp_variable = { camp_admin_month_number = 10 } }
			else_if = { limit = { date < "[YEAR].12.1" } set_temp_variable = { camp_admin_month_number = 11 } }
		}
		YEAR = "[?global.year|.0]"
	}
	add_to_temp_variable = { camp_admin_month_key = camp_admin_month_number }
}

camp_admin_record_deaths = {
	set_temp_variable = { camp_admin_death_actual_k = 0 }
	if = {
		limit = {
			camp_admin_site_is_operational = yes
			NOT = { has_global_flag = settings_chaos_deaths_disabled }
			check_variable = { camp_admin_death_contract = 1 }
			check_variable = { camp_admin_death_requested_k > 0 }
			check_variable = { camp_admin_death_cause > 0 }
			check_variable = { camp_admin_death_cause < 5 }
		}
		camp_admin_refresh_capacity = yes
		set_temp_variable = { camp_admin_death_before_k = state_population_k }
		set_temp_variable = { camp_admin_death_request_k = camp_admin_death_requested_k }
		clamp_temp_variable = { var = camp_admin_death_request_k min = 0 max = camp_admin_detainees_k }
		clamp_temp_variable = { var = camp_admin_death_request_k min = 0 max = constant:camp_admin_value.maximum_receipt_k }
		set_temp_variable = { state_civilian_population_loss_requested = camp_admin_death_request_k }
		multiply_temp_variable = { state_civilian_population_loss_requested = constant:camp_admin_value.people_per_k }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:camp_admin_value.minimum_population_k }
		multiply_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:camp_admin_value.people_per_k }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.camp_atrocity }
		# Physical debit once; measure before registering its actual receipt.
		set_temp_variable = { state_civilian_population_loss_log_deaths = 0 }
		set_temp_variable = { state_civilian_population_loss_target_country = camp_admin_responsible_country }
		set_temp_variable = { state_civilian_population_loss_has_target_country = 1 }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = 1 }
		apply_exact_state_civilian_population_loss = yes
		set_temp_variable = { camp_admin_death_actual_k = camp_admin_death_before_k }
		subtract_from_temp_variable = { camp_admin_death_actual_k = state_population_k }
		clamp_temp_variable = { var = camp_admin_death_actual_k min = 0 max = camp_admin_detainees_k }
		subtract_from_variable = { camp_admin_detainees_k = camp_admin_death_actual_k }
		set_temp_variable = { chaos_deaths_change = camp_admin_death_actual_k }
		multiply_temp_variable = { chaos_deaths_change = constant:camp_admin_value.people_per_k }
		set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.camp_atrocity }
		set_temp_variable = { chaos_deaths_is_civilian = 1 }
		set_temp_variable = { chaos_deaths_apply_state_pop = 0 }
		set_temp_variable = { chaos_deaths_record_state_ledger = 1 }
		set_temp_variable = { chaos_deaths_target_country = camp_admin_responsible_country }
		set_temp_variable = { chaos_deaths_has_target_country = 1 }
		chaos_meter_register_deaths = yes
		camp_rework_record_latest_state_deaths = yes
		camp_admin_record_death_projection = yes
		var:camp_admin_responsible_country = { camp_admin_record_death_projection = yes }
	}
	set_temp_variable = { camp_admin_death_requested_k = 0 }
	set_temp_variable = { camp_admin_death_contract = 0 }
	set_temp_variable = { camp_admin_death_cause = 0 }
}

camp_admin_refresh_state_economy = {
	camp_admin_clear_state_output = yes
	camp_admin_refresh_capacity = yes
	if = {
		limit = {
			camp_admin_has_supported_workforce = yes
			NOT = { has_state_flag = camp_admin_site_suspended }
			OR = {
				check_variable = { camp_admin_role = constant:camp_admin_role.labor }
				check_variable = { camp_admin_role = constant:camp_admin_role.penal }
			}
			var:camp_admin_responsible_country = { check_variable = { camp_admin_mandate > constant:camp_admin_mandate.custody } }
		}
		set_variable = { camp_admin_workforce_k = camp_admin_detainees_k }
		clamp_variable = { var = camp_admin_workforce_k min = 0 max = camp_admin_capacity_k }
		multiply_variable = { camp_admin_workforce_k = constant:camp_admin_value.workforce_share }
		multiply_variable = { camp_admin_workforce_k = camp_admin_support_ratio }
	}
	else_if = {
		limit = {
			has_state_flag = camp_admin_civilian_converted
			camp_admin_site_authority_is_valid = yes
		}
		set_variable = { camp_admin_workforce_k = camp_admin_civilian_workforce_k }
		clamp_variable = { var = camp_admin_workforce_k min = 0 max = camp_admin_population_headroom_k }
		multiply_variable = { camp_admin_workforce_k = camp_admin_support_ratio }
	}
	if = {
		limit = { check_variable = { camp_admin_workforce_k > 0 } }
		set_temp_variable = { camp_admin_chosen_priority = 0 }
		var:camp_admin_responsible_country = { set_temp_variable = { camp_admin_chosen_priority = camp_admin_priority } }
		set_temp_variable = { camp_admin_development_factor = camp_admin_development_level }
		multiply_temp_variable = { camp_admin_development_factor = constant:camp_admin_value.development_step }
		add_to_temp_variable = { camp_admin_development_factor = constant:camp_admin_value.base_development }
		clamp_temp_variable = { var = camp_admin_development_factor min = 0 max = 1 }
		clamp_variable = { var = camp_admin_site_output_scale min = 0 max = 1 }
		multiply_temp_variable = { camp_admin_development_factor = camp_admin_site_output_scale }
		if = {
			limit = { check_variable = { camp_admin_chosen_priority = constant:camp_admin_priority.industry } }
			set_temp_variable = { camp_admin_opportunity = non_damaged_building_level@arms_factory }
			if = {
				limit = { check_variable = { camp_admin_opportunity > 0 } }
				set_variable = { camp_admin_industry_k = camp_admin_workforce_k }
				set_temp_variable = { camp_admin_relevant_workforce_k = camp_admin_opportunity }
				multiply_temp_variable = { camp_admin_relevant_workforce_k = constant:camp_admin_value.workforce_per_factory_k }
				clamp_variable = { var = camp_admin_industry_k min = 0 max = camp_admin_relevant_workforce_k }
				set_variable = { camp_admin_industrial_equivalent = camp_admin_industry_k }
				divide_variable = { camp_admin_industrial_equivalent = constant:camp_admin_value.workforce_per_factory_k }
				multiply_variable = { camp_admin_industrial_equivalent = constant:camp_admin_value.industrial_cap }
				multiply_variable = { camp_admin_industrial_equivalent = camp_admin_development_factor }
			}
		}
		else_if = {
			limit = { check_variable = { camp_admin_chosen_priority = constant:camp_admin_priority.works } }
			# Only the currently funded works project grants construction output.
			if = {
				limit = {
					var:camp_admin_responsible_country = {
						has_country_flag = camp_admin_economy_slot_active
						check_variable = { camp_admin_economy_state_id = PREV.id }
						check_variable = { camp_admin_economy_project_type = constant:camp_admin_priority.works }
					}
				}
				set_variable = { camp_admin_works_k = camp_admin_workforce_k }
				clamp_variable = { var = camp_admin_works_k min = 0 max = constant:camp_admin_value.workforce_per_works_k }
				set_variable = { camp_admin_construction_factor = camp_admin_works_k }
				divide_variable = { camp_admin_construction_factor = constant:camp_admin_value.workforce_per_works_k }
				multiply_variable = { camp_admin_construction_factor = constant:camp_admin_value.construction_cap }
				multiply_variable = { camp_admin_construction_factor = camp_admin_development_factor }
			}
		}
		else_if = {
			limit = { check_variable = { camp_admin_chosen_priority = constant:camp_admin_priority.extraction } }
			set_temp_variable = { camp_admin_resource_opportunity = resource@steel }
			add_to_temp_variable = { camp_admin_resource_opportunity = resource@aluminium }
			add_to_temp_variable = { camp_admin_resource_opportunity = resource@tungsten }
			add_to_temp_variable = { camp_admin_resource_opportunity = resource@chromium }
			add_to_temp_variable = { camp_admin_resource_opportunity = resource@rubber }
			add_to_temp_variable = { camp_admin_resource_opportunity = resource@oil }
			if = {
				limit = { check_variable = { camp_admin_resource_opportunity > 0 } }
				multiply_temp_variable = { camp_admin_resource_opportunity = constant:camp_admin_value.workforce_per_resource_k }
				set_variable = { camp_admin_extraction_k = camp_admin_workforce_k }
				clamp_variable = { var = camp_admin_extraction_k min = 0 max = camp_admin_resource_opportunity }
				set_variable = { camp_admin_extraction_factor = camp_admin_extraction_k }
				divide_variable = { camp_admin_extraction_factor = camp_admin_resource_opportunity }
				multiply_variable = { camp_admin_extraction_factor = constant:camp_admin_value.extraction_cap }
				multiply_variable = { camp_admin_extraction_factor = camp_admin_development_factor }
			}
		}
		if = {
			limit = { OR = { check_variable = { camp_admin_construction_factor > 0 } check_variable = { camp_admin_extraction_factor > 0 } } }
			add_dynamic_modifier = { modifier = camp_admin_state_economy }
		}
	}
}

camp_admin_refresh_country_economy = {
	save_event_target_as = camp_admin_actor
	set_variable = { camp_admin_industrial_factor = 0 }
	set_variable = { camp_admin_total_detainees_k = 0 }
	set_variable = { camp_admin_total_workforce_k = 0 }
	for_each_scope_loop = {
		array = camp_active_site_states
		if = { limit = { NOT = { has_state_flag = camp_admin_closed } }
			camp_admin_accumulate_state_economy = yes

	}}
	for_each_scope_loop = { array = camp_admin_civilian_states camp_admin_accumulate_state_economy = yes }
	set_temp_variable = { camp_admin_national_industry = num_of_military_factories }
	if = {
		limit = { check_variable = { camp_admin_national_industry > 0 } }
		divide_variable = { camp_admin_industrial_factor = camp_admin_national_industry }
		clamp_variable = { var = camp_admin_industrial_factor min = 0 max = constant:camp_admin_value.national_cap }
	}
	else = { set_variable = { camp_admin_industrial_factor = 0 } }
	if = { limit = { check_variable = { camp_admin_industrial_factor > 0 } } add_dynamic_modifier = { modifier = camp_admin_country_economy } }
	else = { remove_dynamic_modifier = { modifier = camp_admin_country_economy } }

		var:camp_admin_economy_state_id = {
			camp_admin_refresh_state_economy = yes
			if = {
				limit = {
					camp_admin_site_authority_is_valid = yes
					NOT = { has_state_flag = camp_admin_site_suspended }
					OR = {
						AND = {
							camp_admin_has_supported_workforce = yes
							check_variable = { camp_admin_workforce_k > 0 }
							NOT = { check_variable = { camp_admin_project_type = constant:camp_admin_priority.civilian } }
							event_target:camp_admin_actor = { check_variable = { camp_admin_priority = camp_admin_economy_project_type } }
						}
						AND = { has_state_flag = camp_admin_civilian_conversion_pending check_variable = { camp_admin_project_type = constant:camp_admin_priority.civilian } }
					}
				}
				set_temp_variable = { camp_admin_project_valid = 1 }
				if = { limit = { check_variable = { camp_admin_project_type = constant:camp_admin_priority.industry } } set_temp_variable = { camp_admin_project_available_workers_k = camp_admin_industry_k } }
				else_if = { limit = { check_variable = { camp_admin_project_type = constant:camp_admin_priority.works } } set_temp_variable = { camp_admin_project_available_workers_k = camp_admin_works_k } }

				else_if = {
					limit = { check_variable = { camp_admin_project_type = constant:camp_admin_priority.civilian } }
					# The next finite group of released civilians performs its conversion work.
					set_temp_variable = { camp_admin_project_available_workers_k = camp_admin_civilian_target_k }
					multiply_temp_variable = { camp_admin_project_available_workers_k = constant:camp_admin_value.civilian_replacement_step }

					clamp_temp_variable = { var = camp_admin_unconverted_workers_k min = 0 max = camp_admin_population_headroom_k }
					clamp_temp_variable = { var = camp_admin_project_available_workers_k min = 0 max = camp_admin_unconverted_workers_k }
					multiply_temp_variable = { camp_admin_project_available_workers_k = camp_admin_support_ratio }
				}
			}
		}
		set_variable = { camp_admin_economy_work_rate = camp_admin_project_available_workers_k }
		divide_variable = { camp_admin_economy_work_rate = constant:camp_admin_value.project_workers_per_work_unit_k }
		clamp_variable = { var = camp_admin_economy_work_rate min = 0 max = constant:camp_admin_value.project_max_monthly_work }

			limit = { check_variable = { camp_admin_project_valid = 1 } check_variable = { camp_admin_economy_work_rate > 0 } NOT = { has_country_flag = camp_admin_maintenance_shortage } }
			camp_admin_prepare_project_quote = yes
			camp_admin_try_pay_project = yes
			if = {
				limit = { check_variable = { camp_admin_payment_accepted = 1 } }
				add_to_variable = { camp_admin_economy_progress = camp_admin_economy_work_rate }
				if = {
					limit = { check_variable = { var = camp_admin_economy_progress value = camp_admin_economy_target compare = greater_than_or_equals } }
					set_variable = { camp_admin_economy_completed_state_id = camp_admin_economy_state_id }
					set_variable = { camp_admin_economy_completed_type = camp_admin_economy_project_type }
					set_country_flag = camp_admin_economy_completed_this_month
					var:camp_admin_economy_state_id = {

							add_to_variable = { camp_admin_civilian_replacement = constant:camp_admin_value.civilian_replacement_step }

							multiply_variable = { camp_admin_civilian_workforce_k = camp_admin_civilian_replacement }
							set_state_flag = camp_admin_civilian_converted
							if = { limit = { check_variable = { camp_admin_civilian_replacement = 1 } } clr_state_flag = camp_admin_civilian_conversion_pending }
						}
						else = {
							add_to_variable = { camp_admin_development_level = 1 }
							clamp_variable = { var = camp_admin_development_level min = 0 max = constant:camp_admin_value.max_development }
							if = { limit = { check_variable = { camp_admin_project_type = constant:camp_admin_priority.works } infrastructure < 5 } add_building_construction = { type = infrastructure level = 1 instant_build = yes } }
						}
					}

	}

	camp_admin_initialize_country = yes

	set_temp_variable = { camp_admin_research_award_bonus = 0 }

		set_variable = { camp_admin_research_awards_this_year = 0 }

			has_country_flag = camp_admin_institution_slot_active

			check_variable = { camp_admin_research_program_valid = 1 }

	}
