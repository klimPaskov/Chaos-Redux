# Event 016 private conventional incident predicates, country scope.
# Shared provider/stage predicates reuse the existing private Computation contract.
# Independent family receipts allow simultaneous incidents without cross-settlement.

brilliant_scientist_mengele_conventional_incident_provider_is_valid = {
	brilliant_scientist_mengele_project_stage_provider_is_valid = yes
}

brilliant_scientist_mengele_conventional_incident_stage_is_valid = {
	brilliant_scientist_mengele_computation_incident_stage_is_valid = yes
}

# Only the requested family's transient state blocks its roll.
# A different family's incident does not suppress this project's consequences.
brilliant_scientist_mengele_can_dispatch_conventional_incident = {
	brilliant_scientist_mengele_conventional_incident_provider_is_valid = yes
	brilliant_scientist_mengele_conventional_incident_stage_is_valid = yes
	OR = {
		AND = {
			check_variable = { var = mengele_event016_project_family value = constant:brilliant_scientist_project_family.electronics compare = equals }
			NOT = { has_country_flag = mengele_event016_electronics_incident_active }
			NOT = { has_country_flag = mengele_event016_electronics_incident_recovery_active }
		}
		AND = {
			check_variable = { var = mengele_event016_project_family value = constant:brilliant_scientist_project_family.materials compare = equals }
			NOT = { has_country_flag = mengele_event016_materials_incident_active }
			NOT = { has_country_flag = mengele_event016_materials_incident_recovery_active }
		}
		AND = {
			check_variable = { var = mengele_event016_project_family value = constant:brilliant_scientist_project_family.rocketry compare = equals }
			NOT = { has_country_flag = mengele_event016_rocketry_incident_active }
			NOT = { has_country_flag = mengele_event016_rocketry_incident_recovery_active }
		}
		AND = {
			check_variable = { var = mengele_event016_project_family value = constant:brilliant_scientist_project_family.high_energy compare = equals }
			NOT = { has_country_flag = mengele_event016_high_energy_incident_active }
			NOT = { has_country_flag = mengele_event016_high_energy_incident_recovery_active }
		}
		AND = {
			check_variable = { var = mengele_event016_project_family value = constant:brilliant_scientist_project_family.biomedical compare = equals }
			NOT = { has_country_flag = mengele_event016_biomedical_incident_active }
			NOT = { has_country_flag = mengele_event016_biomedical_incident_recovery_active }
		}
	}
}
