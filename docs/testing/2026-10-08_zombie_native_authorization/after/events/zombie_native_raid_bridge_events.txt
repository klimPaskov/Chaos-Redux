# Chaos Redux weaponized zombie native raid country-root bridges.
# Each fixed ID identifies one prepared tier/friendly family and result strength.
# Exact regular targets come from the native callback; FROM is not the victim.

add_namespace = zombie_native_raid_bridge

country_event = {
	id = zombie_native_raid_bridge.1
	is_triggered_only = yes
	hidden = yes
	immediate = {
		if = {
			limit = {
				has_country_flag = weaponized_zombie_raid_success_low
				NOT = { has_country_flag = weaponized_zombie_friendly_to_humans }
			}
			resolve_minor_weaponized_zombie_native_raid = yes
		}
	}
}
