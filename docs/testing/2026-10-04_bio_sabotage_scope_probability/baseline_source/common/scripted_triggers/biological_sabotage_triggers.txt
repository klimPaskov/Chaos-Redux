bio_sabotage_actor_cooldown_is_ready = {
	OR = {
		NOT = { has_variable = bio_sabotage_next_allowed_day }
		check_variable = {
			var = bio_sabotage_next_allowed_day
			value = global.num_days
			compare = less_than_or_equals
		}
	}
}

bio_sabotage_state_relationship_is_valid = {
	CONTROLLER = {
		exists = yes
		NOT = { tag = ROOT }
		NOT = { is_subject_of = ROOT }
		ROOT = {
			NOT = { is_in_faction_with = PREV }
			NOT = { is_subject_of = PREV }
			PREV = { set_temp_variable = { cbrn_retaliation_target_id = id } }
			OR = {
				AND = {
					has_war_with = PREV
					cbrn_policy_allows_strategic_use_against_target_id = yes
				}
				AND = {
					NOT = { has_war_with = PREV }
					cbrn_policy_allows_extreme_use = yes
				}
			}
		}
	}
}

bio_sabotage_common_target_state = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	impassable = no
	state_population_k > constant:bio_sabotage_cost.minimum_target_population_k
	NOT = { has_state_flag = bio_sabotage_operation_active }
	bio_sabotage_state_relationship_is_valid = yes
}

bio_sabotage_anthrax_target_state = {
	bio_sabotage_common_target_state = yes
	NOT = { bio_lifecycle_state_has_anthrax_episode = yes }
	OR = {
		naval_base > constant:bio_sabotage_target_profile.building_floor
		supply_node > constant:bio_sabotage_target_profile.building_floor
		industrial_complex > constant:bio_sabotage_target_profile.anthrax_industry_threshold
		arms_factory > constant:bio_sabotage_target_profile.anthrax_industry_threshold
		biowarfare_facility > constant:bio_sabotage_target_profile.building_floor
	}
}

bio_sabotage_plague_target_state = {
	bio_sabotage_common_target_state = yes
