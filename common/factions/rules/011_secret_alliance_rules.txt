# Overview:
# - Faction rules for the public Anti-[target] Pact created by Event 011.
# - They keep joining restricted to live signatories and represent the pact's
#   liaison offices and synchronized war calendar after reveal.

@secret_alliance_rule_liaison_planning_speed = 0.02
@secret_alliance_rule_liaison_org_regain = 0.02
@secret_alliance_rule_calendar_army_attack = 0.02
@secret_alliance_rule_calendar_army_defence = 0.03

secret_alliance_joining_rule_public_signatories = {
	type = joining_rules

	visible = {
		has_faction_template = faction_template_secret_alliance_anti_target_pact
	}

	available = {
		has_global_flag = secret_alliance_active
	}

	can_remove = {
		always = no
	}

	trigger = {
		has_country_flag = secret_alliance_member
		has_country_flag = secret_alliance_member_confirmed
		NOT = { has_country_flag = secret_alliance_member_left_pact }
	}

	ai_will_do = {
		base = 0
	}
}

secret_alliance_member_rule_protocol_liaison = {
	type = member_rules

	visible = {
		has_faction_template = faction_template_secret_alliance_anti_target_pact
	}

	available = {
		has_global_flag = secret_alliance_active
	}

	can_remove = {
		always = no
	}

	trigger = {
		always = yes
	}

	modifier = {
		planning_speed = @secret_alliance_rule_liaison_planning_speed
		army_org_regain = @secret_alliance_rule_liaison_org_regain
	}

	ai_will_do = {
		base = 0
	}
}

secret_alliance_member_rule_war_calendar = {
	type = member_rules

	visible = {
		has_faction_template = faction_template_secret_alliance_anti_target_pact
	}

	available = {
		has_global_flag = secret_alliance_active
	}

	can_remove = {
		always = no
	}

	trigger = {
		always = yes
	}

	modifier = {
		army_attack_factor = @secret_alliance_rule_calendar_army_attack
		army_defence_factor = @secret_alliance_rule_calendar_army_defence
	}

	ai_will_do = {
		base = 0
	}
}
