# ============================================================================
# CHAOS REDUX - EVENT 011 FACTION RULES
# ============================================================================

secret_alliance_joining_rule = {
	type = joining_rules
	visible = { has_country_flag = secret_alliance_public_faction_leader }
	trigger = {
		has_country_flag = secret_alliance_reveal_member_snapshot
		has_country_flag = secret_alliance_active_member
	}
	ai_will_do = { base = 0 }
}

secret_alliance_war_declaration_rule = {
	type = war_declaration_rules
	visible = { has_country_flag = secret_alliance_public_faction_leader }
	trigger = {
		is_faction_leader = yes
		FROM = { tag = event_target:secret_alliance_target }
	}
	ai_will_do = { base = 0 }
}

secret_alliance_call_to_war_rule = {
	type = call_to_war_rules
	visible = { has_country_flag = secret_alliance_public_faction_leader }
	trigger = {
		has_country_flag = secret_alliance_reveal_member_snapshot
		NOT = { has_country_flag = secret_alliance_member_removed }
	}
	ai_will_do = { base = 0 }
}

secret_alliance_change_leader_rule = {
	type = change_leader_rules
	visible = { has_country_flag = secret_alliance_public_faction_leader }
	trigger = {
		OR = {
			has_country_flag = secret_alliance_major_sponsor
			has_country_flag = secret_alliance_founder
		}
		has_country_flag = secret_alliance_reveal_member_snapshot
	}
	ai_will_do = { base = 0 }
}
