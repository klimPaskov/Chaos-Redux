# Overview:
# - Manifest and goals for the Event 011 public Anti-[target] Pact.
# - Goals stay narrow and diplomatic so the public pact crisis is readable
#   without turning Secret Alliance into a full custom faction route.

@secret_alliance_manifest_public_evidence_scale = 0.02
@secret_alliance_manifest_readiness_scale = 0.02
@secret_alliance_goal_projection_short = 60
@secret_alliance_goal_projection_medium = 120
@secret_alliance_goal_projection_long = 180
@secret_alliance_goal_initiative_reward = 1

faction_manifest_secret_alliance_public_protocol = {
	name = faction_manifest_secret_alliance_public_protocol_name
	description = faction_manifest_secret_alliance_public_protocol_desc
	is_manifest = yes

	ratio_progress = {
		total_amount = 100
		completed_amount = global.secret_alliance_readiness

		scale = {
			planning_speed = @secret_alliance_manifest_public_evidence_scale
			army_org_regain = @secret_alliance_manifest_readiness_scale
		}
	}
}

faction_goal_secret_alliance_publish_lists = {
	name = faction_goal_secret_alliance_publish_lists_name
	description = faction_goal_secret_alliance_publish_lists_desc
	category = short_term
	group = FOCUS_FILTER_POLITICAL

	visible = {
		ROOT = {
			has_faction_template = faction_template_secret_alliance_anti_target_pact
		}
	}

	available = {
		has_global_flag = secret_alliance_active
	}

	completed = {
		NOT = {
			any_country = {
				has_country_flag = secret_alliance_member
				NOT = { has_country_flag = secret_alliance_member_confirmed }
			}
		}
	}

	complete_effect = {
		add_faction_power_projection = @secret_alliance_goal_projection_short
		add_faction_initiative = @secret_alliance_goal_initiative_reward
	}

	ai_will_do = {
		base = 6
	}
}

faction_goal_secret_alliance_joint_pressure = {
	name = faction_goal_secret_alliance_joint_pressure_name
	description = faction_goal_secret_alliance_joint_pressure_desc
	category = medium_term
	group = FOCUS_FILTER_POLITICAL

	visible = {
		ROOT = {
			has_faction_template = faction_template_secret_alliance_anti_target_pact
		}
	}

	available = {
		has_global_flag = secret_alliance_active
	}

	completed = {
		check_variable = {
			var = global.secret_alliance_cohesion
			value = constant:secret_alliance_meter.high
			compare = greater_than_or_equals
		}
	}

	complete_effect = {
		add_faction_power_projection = @secret_alliance_goal_projection_medium
		add_faction_initiative = @secret_alliance_goal_initiative_reward
	}

	ai_will_do = {
		base = 4
	}
}

faction_goal_secret_alliance_war_calendar = {
	name = faction_goal_secret_alliance_war_calendar_name
	description = faction_goal_secret_alliance_war_calendar_desc
	category = long_term
	group = FOCUS_FILTER_ARMY

	}

	}

		base = 3
	}
