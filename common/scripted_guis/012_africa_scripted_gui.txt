@africa_gui_congress_session_cooldown_days = 35
@africa_gui_refresh_register_cooldown_days = 35
@africa_gui_dossier_action_cooldown_days = 45
@africa_gui_authority_seats_cooldown_days = 60
@africa_gui_bestiary_terms_cooldown_days = 60
@africa_gui_sponsor_readiness_cooldown_days = 70

scripted_gui = {
	africa_continental_congress_scripted_gui = {
		context_type = decision_category
		window_name = "africa_continental_congress_container"

		ai_enabled = { always = no }
		visible = {
			is_ai = no
			has_global_flag = africa_runtime_context_ready
			has_country_flag = africa_unifier_active
		}

			effects = {
				africa_gui_congress_session_button_click = {
					if = {
						limit = { can_africa_use_gui_congress_session = yes }
						add_political_power = constant:africa_decision.congress_cost_spend
						add_to_variable = { africa_legitimacy = constant:africa_value_delta.charter_opened_legitimacy }
						add_to_variable = { africa_league_cohesion = constant:africa_value_delta.charter_join_cohesion }
						africa_clamp_core_values = yes
						set_country_flag = africa_congress_session_complete
						set_country_flag = { flag = africa_gui_congress_session_recent days = @africa_gui_congress_session_cooldown_days }
					}
				}

				africa_gui_refresh_register_button_click = {
					if = {
						limit = { can_africa_use_gui_refresh_register = yes }
						add_political_power = constant:africa_decision.observer_cost_spend
						africa_refresh_charter_candidate_pool = yes
						africa_refresh_integration_target_states = yes
						set_country_flag = { flag = africa_gui_refresh_register_recent days = @africa_gui_refresh_register_cooldown_days }
					}
				}

				africa_gui_start_dossier_button_click = {
					if = {
						limit = { can_africa_use_gui_start_dossier = yes }
						add_political_power = constant:africa_decision.dossier_cost_spend
						africa_pay_selected_dossier_survey_cost = yes
						africa_open_next_historical_dossier = yes
						africa_clamp_core_values = yes
						set_country_flag = { flag = africa_gui_dossier_action_recent days = @africa_gui_dossier_action_cooldown_days }
					}
				}

				africa_gui_authority_seats_button_click = {
					if = {
						limit = { can_africa_use_gui_authority_seats = yes }
						add_political_power = constant:africa_decision.observer_cost_spend
						africa_pay_authority_seat_cost = yes
						africa_spawn_regional_authorities_from_controlled_seats = yes
						africa_refresh_integration_target_states = yes
						add_to_variable = { africa_authority = constant:africa_value_delta.integration_authority }
						add_to_variable = { africa_regional_trust = constant:africa_value_delta.regional_authority_trust }
						africa_clamp_core_values = yes
						set_country_flag = { flag = africa_gui_authority_seats_recent days = @africa_gui_authority_seats_cooldown_days }
					}
				}

				africa_gui_bestiary_terms_button_click = {
					if = {
						limit = { can_africa_use_gui_bestiary_terms = yes }
						add_political_power = constant:africa_decision.high_chaos_package_cost_spend
						africa_pay_bestiary_habitat_terms_cost = yes
						add_to_variable = { africa_habitat_trust = constant:africa_value_delta.high_chaos_nonhuman_sovereignty }
						add_to_variable = { africa_bestiary_alarm = constant:africa_value_delta.integration_burden_relief }
						add_to_variable = { africa_covenant_pressure = constant:africa_value_delta.covenant_pressure_relief }
						africa_clamp_core_values = yes
						set_country_flag = africa_bestiary_habitat_terms_negotiated
						set_country_flag = { flag = africa_gui_bestiary_terms_recent days = @africa_gui_bestiary_terms_cooldown_days }
					}
				}

				africa_gui_sponsor_readiness_button_click = {
					if = {
						limit = { can_africa_use_gui_sponsor_readiness = yes }
						africa_start_continent_sponsor_readiness_mission = yes
						set_country_flag = { flag = africa_gui_sponsor_readiness_recent days = @africa_gui_sponsor_readiness_cooldown_days }
					}
				}
			}

		triggers = {
			africa_continental_congress_charter_banner_static_visible = {
				OR = {
					has_global_flag = africa_charter_league_created
					has_country_flag = africa_charter_league_leader
					has_country_flag = africa_focus_charter_mandate_complete
				}
			}

			africa_continental_congress_charter_banner_anim_visible = {
				OR = {
					has_global_flag = africa_charter_league_created
					has_country_flag = africa_charter_league_leader
					has_country_flag = africa_focus_charter_mandate_complete
				}
			}

			africa_continental_congress_charter_seal_static_visible = {
				OR = {
					has_global_flag = africa_charter_league_created
					has_country_flag = africa_charter_league_leader
					has_country_flag = africa_focus_charter_mandate_complete
				}
			}

			africa_continental_congress_charter_seal_anim_visible = {
				OR = {
					has_global_flag = africa_charter_league_created
					has_country_flag = africa_charter_league_leader
					has_country_flag = africa_focus_charter_mandate_complete
				}
			}

			africa_continental_congress_authority_seal_static_visible = {
				OR = {
					has_country_flag = africa_authority_atlas_open
					has_country_flag = africa_archive_old_seats_open
					has_country_flag = africa_dossier_selection_office_open
				}
			}

			africa_continental_congress_authority_seal_anim_visible = {
				OR = {
					has_country_flag = africa_archive_old_seats_open
					has_country_flag = africa_archive_guard_mission_active
					has_country_flag = africa_selected_dossier_survey_mission_active
				}
			}

			africa_continental_congress_bestiary_warning_static_visible = {
				OR = {
					has_country_flag = africa_habitat_trust_board_open
					has_country_flag = africa_minimum_high_chaos_packages_ready
					has_country_flag = africa_all_high_chaos_packages_unlocked
				}
			}

			africa_continental_congress_bestiary_warning_anim_visible = {
				OR = {
					has_country_flag = africa_habitat_trust_board_open
					has_country_flag = africa_bestiary_warning_active
					has_country_flag = africa_bestiary_containment_mission_active
				}
			}

			africa_continental_congress_cohesion_warning_border_static_visible = {
				OR = {
					has_country_flag = africa_bestiary_warning_active
					has_country_flag = africa_bestiary_warning_overused
					has_country_flag = africa_dossier_resistance_watch_active
					has_country_flag = africa_old_seat_arbitration_active
					check_variable = { africa_bestiary_warning_count > 0 }
					check_variable = { africa_bestiary_warning_defiance_count > 0 }
					check_variable = { africa_league_cohesion < constant:africa_decision.cohesion_gate }
				}
			}

			africa_continental_congress_cohesion_warning_border_anim_visible = {
				OR = {
					has_country_flag = africa_bestiary_warning_active
					has_country_flag = africa_bestiary_warning_overused
					has_country_flag = africa_dossier_resistance_watch_active
					has_country_flag = africa_old_seat_arbitration_active
					check_variable = { africa_bestiary_warning_defiance_count > 0 }
					check_variable = { africa_league_cohesion < constant:africa_decision.cohesion_gate }
					check_variable = { africa_bestiary_alarm > constant:africa_decision.cohesion_gate }
				}
			}

			africa_continental_congress_formable_ready_static_visible = {
				OR = {
					has_global_flag = africa_continental_pole_ready
					has_global_flag = africa_world_is_one_gate_ready
					has_country_flag = africa_continental_formable_complete
					has_country_flag = africa_continent_unifiers_certified_for_world_is_one
					has_country_flag = africa_world_is_one_gate_prepared
				}
			}

			africa_continental_congress_formable_ready_anim_visible = {
				OR = {
					can_africa_prepare_world_is_one_gate = yes
					has_global_flag = africa_world_is_one_gate_ready
					has_country_flag = africa_world_is_one_gate_prepared
				}
			}

			africa_continental_congress_green_covenant_static_visible = {
				OR = {
					has_country_flag = africa_habitat_trust_board_open
					has_country_flag = africa_treaty_teeth_roots_open
					has_country_flag = africa_bestiary_forest_covenant
					has_country_flag = africa_bestiary_nature_courts_covenant
					has_country_flag = africa_world_root_mandate_open
				}
			}

			africa_continental_congress_green_covenant_anim_visible = {
				OR = {
					has_country_flag = africa_bestiary_forest_covenant
					has_country_flag = africa_bestiary_nature_courts_covenant
					has_country_flag = africa_world_root_mandate_open
					check_variable = { africa_covenant_pressure > constant:africa_decision.cohesion_gate }
				}
			}

			africa_gui_congress_session_button_click_enabled = {
				can_africa_use_gui_congress_session = yes
			}

			africa_gui_refresh_register_button_click_enabled = {
				can_africa_use_gui_refresh_register = yes
			}

			africa_gui_start_dossier_button_click_enabled = {
				can_africa_use_gui_start_dossier = yes
			}

			africa_gui_authority_seats_button_click_enabled = {
				can_africa_use_gui_authority_seats = yes
			}

			africa_gui_bestiary_terms_button_click_enabled = {
				can_africa_use_gui_bestiary_terms = yes
			}

			africa_gui_sponsor_readiness_button_click_enabled = {
				can_africa_use_gui_sponsor_readiness = yes
			}
		}
	}
}
