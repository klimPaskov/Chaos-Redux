# ============================================================================
# CHAOS REDUX - REPRESSION LEDGER SCRIPTED GUI
# ============================================================================
# Overview:
# - Provides an optional, player-opened decision-category ledger for the camp
#   repression system without adding any AI dependency or recurring popup.
# - Reads bounded country arrays rebuilt by camp_rework_rebuild_display_values.
# - Every operational button calls the same shared effects used by decisions;
#   AI continues to act through decisions and country-kit monthly bridges.
# ============================================================================

scripted_gui = {
	camp_repression_ledger_scripted_gui = {
		context_type = decision_category
		window_name = "repression_ledger_window"

		ai_enabled = { always = no }
		visible = {
			ROOT = {
				is_ai = no
				has_country_flag = camp_repression_ledger_open
				OR = {
					has_active_camp_network = yes
					has_visible_camp_reform_work = yes
					has_camp_country_specific_route = yes
					has_country_flag = camp_rework_managed_network_visible
				}
			}
		}

		effects = {
			camp_gui_close_window_click = { ROOT = { camp_rework_close_ledger = yes } }
			camp_gui_refresh_click = { ROOT = { camp_rework_rebuild_display_values = yes } }
			camp_gui_show_directives_click = { ROOT = { camp_rework_gui_reveal_country_directives = yes } }

			camp_gui_tab_overview_click = {
				ROOT = {
					clr_country_flag = camp_ledger_tab_state_pools
					clr_country_flag = camp_ledger_tab_sites
					clr_country_flag = camp_ledger_tab_country
					clr_country_flag = camp_ledger_tab_discovery
					set_country_flag = camp_ledger_tab_overview
					camp_rework_rebuild_display_values = yes
				}
			}
			camp_gui_tab_state_pools_click = {
				ROOT = {
					clr_country_flag = camp_ledger_tab_overview
					clr_country_flag = camp_ledger_tab_sites
					clr_country_flag = camp_ledger_tab_country
					clr_country_flag = camp_ledger_tab_discovery
					set_country_flag = camp_ledger_tab_state_pools
					camp_rework_rebuild_display_values = yes
				}
			}
			camp_gui_tab_sites_click = {
				ROOT = {
					clr_country_flag = camp_ledger_tab_overview
					clr_country_flag = camp_ledger_tab_state_pools
					clr_country_flag = camp_ledger_tab_country
					clr_country_flag = camp_ledger_tab_discovery
					set_country_flag = camp_ledger_tab_sites
					camp_rework_rebuild_display_values = yes
				}
			}
			camp_gui_tab_country_click = {
				ROOT = {
					clr_country_flag = camp_ledger_tab_overview
					clr_country_flag = camp_ledger_tab_state_pools
					clr_country_flag = camp_ledger_tab_sites
					clr_country_flag = camp_ledger_tab_discovery
					set_country_flag = camp_ledger_tab_country
					camp_rework_rebuild_display_values = yes
				}
			}
			camp_gui_tab_discovery_click = {
				ROOT = {
					clr_country_flag = camp_ledger_tab_overview
					clr_country_flag = camp_ledger_tab_state_pools
					clr_country_flag = camp_ledger_tab_sites
					clr_country_flag = camp_ledger_tab_country
					set_country_flag = camp_ledger_tab_discovery
					camp_rework_rebuild_display_values = yes
				}
			}

			camp_gui_select_pool_occupied_click = {
				ROOT = {
					set_variable = { camp_selected_state_id = camp_pool_occupied_states^0 }
					camp_rework_validate_selected_state = yes
				}
			}
			camp_gui_select_pool_colonial_click = {
				ROOT = {
					set_variable = { camp_selected_state_id = camp_pool_colonial_states^0 }
					camp_rework_validate_selected_state = yes
				}
			}
			camp_gui_select_pool_noncore_click = {
				ROOT = {
					set_variable = { camp_selected_state_id = camp_pool_noncore_states^0 }
					camp_rework_validate_selected_state = yes
				}
			}
			camp_gui_select_pool_periphery_click = {
				ROOT = {
					set_variable = { camp_selected_state_id = camp_pool_periphery_states^0 }
					camp_rework_validate_selected_state = yes
				}
			}
			camp_gui_select_pool_core_click = {
				ROOT = {
					set_variable = { camp_selected_state_id = camp_pool_core_fallback_states^0 }
					camp_rework_validate_selected_state = yes
				}
			}

			camp_gui_select_site_1_click = { ROOT = { set_variable = { camp_selected_state_id = camp_ledger_site_states^0 } camp_rework_validate_selected_state = yes } }
			camp_gui_select_site_2_click = { ROOT = { set_variable = { camp_selected_state_id = camp_ledger_site_states^1 } camp_rework_validate_selected_state = yes } }
			camp_gui_select_site_3_click = { ROOT = { set_variable = { camp_selected_state_id = camp_ledger_site_states^2 } camp_rework_validate_selected_state = yes } }
			camp_gui_select_site_4_click = { ROOT = { set_variable = { camp_selected_state_id = camp_ledger_site_states^3 } camp_rework_validate_selected_state = yes } }
			camp_gui_select_site_5_click = { ROOT = { set_variable = { camp_selected_state_id = camp_ledger_site_states^4 } camp_rework_validate_selected_state = yes } }
			camp_gui_select_site_6_click = { ROOT = { set_variable = { camp_selected_state_id = camp_ledger_site_states^5 } camp_rework_validate_selected_state = yes } }

			camp_gui_expand_selected_pool_click = { ROOT = { camp_rework_gui_expand_selected_pool = yes } }
			camp_gui_start_labor_project_click = { ROOT = { camp_rework_gui_start_labor_project = yes } }
			camp_gui_allocate_guards_click = { ROOT = { camp_rework_gui_allocate_guards = yes } }
			camp_gui_reduce_quotas_click = { ROOT = { camp_rework_gui_reduce_quotas = yes } }
			camp_gui_inspect_selected_site_click = { ROOT = { camp_rework_gui_inspect_selected_site = yes } }
			camp_gui_dismantle_selected_site_click = { ROOT = { camp_rework_gui_begin_selected_dismantlement = yes } }
			camp_gui_destroy_evidence_click = { ROOT = { camp_rework_gui_destroy_selected_evidence = yes } }
			camp_gui_chemical_method_click = { ROOT = { camp_rework_gui_apply_selected_chemical_method = yes } }
			camp_gui_biological_method_click = { ROOT = { camp_rework_gui_apply_selected_biological_method = yes } }
		}

		triggers = {
			repression_ledger_overview_panel_visible = { ROOT = { has_country_flag = camp_ledger_tab_overview } }
			repression_ledger_state_pools_panel_visible = { ROOT = { has_country_flag = camp_ledger_tab_state_pools } }
			repression_ledger_sites_panel_visible = { ROOT = { has_country_flag = camp_ledger_tab_sites } }
			repression_ledger_country_panel_visible = { ROOT = { has_country_flag = camp_ledger_tab_country } }
			repression_ledger_discovery_panel_visible = { ROOT = { has_country_flag = camp_ledger_tab_discovery } }

			camp_gui_select_pool_occupied_visible = { ROOT = { check_variable = { camp_pool_occupied_states^num > 0 } } }
			camp_gui_select_pool_colonial_visible = { ROOT = { check_variable = { camp_pool_colonial_states^num > 0 } } }
			camp_gui_select_pool_noncore_visible = { ROOT = { check_variable = { camp_pool_noncore_states^num > 0 } } }
			camp_gui_select_pool_periphery_visible = { ROOT = { check_variable = { camp_pool_periphery_states^num > 0 } } }
			camp_gui_select_pool_core_visible = { ROOT = { check_variable = { camp_pool_core_fallback_states^num > 0 } } }
			camp_gui_select_site_1_visible = { ROOT = { check_variable = { camp_ledger_site_states^num > 0 } } }
			camp_gui_select_site_2_visible = { ROOT = { check_variable = { camp_ledger_site_states^num > 1 } } }
			camp_gui_select_site_3_visible = { ROOT = { check_variable = { camp_ledger_site_states^num > 2 } } }
			camp_gui_select_site_4_visible = { ROOT = { check_variable = { camp_ledger_site_states^num > 3 } } }
			camp_gui_select_site_5_visible = { ROOT = { check_variable = { camp_ledger_site_states^num > 4 } } }
			camp_gui_select_site_6_visible = { ROOT = { check_variable = { camp_ledger_site_states^num > 5 } } }

			camp_gui_expand_selected_pool_click_enabled = {
				ROOT = {
					camp_rework_country_uses_generic_kit = yes
					has_variable = camp_selected_state_id
					NOT = { has_country_flag = camp_rework_expansion_frozen }
					var:camp_selected_state_id = {
						is_controlled_by = ROOT
						camp_rework_state_can_accept_responsibility_from_root = yes
					}
					OR = {
						AND = {
							var:camp_selected_state_id = { is_valid_camp_active_site_state = yes }
							has_political_power > constant:camp_rework_cost.expansion_pp
							manpower > constant:camp_rework_cost.expansion_manpower
							command_power > constant:camp_rework_cost.expansion_command
							has_equipment = { infantry_equipment > constant:camp_rework_equipment_cost.infantry_equipment }
							has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
							has_equipment = { train_equipment_1 > constant:camp_rework_equipment_cost.trains }
						}
						AND = {
							var:camp_selected_state_id = { NOT = { is_valid_camp_active_site_state = yes } camp_rework_state_matches_country_kit_pool = yes }
							has_political_power > constant:camp_rework_cost.activation_pp
							manpower > constant:camp_rework_cost.activation_manpower
							command_power > constant:camp_rework_cost.activation_command
							has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
						}
					}
				}
			}
			camp_gui_start_labor_project_click_enabled = {
				ROOT = {
					camp_rework_country_uses_generic_kit = yes
					has_variable = camp_selected_state_id
					NOT = { has_country_flag = generic_labor_project_active }
					has_political_power > constant:camp_rework_cost.labor_project_pp
					has_equipment = { motorized_equipment_1 > constant:camp_rework_equipment_cost.motorized_equipment }
					has_equipment = { train_equipment_1 > constant:camp_rework_equipment_cost.trains }
					has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
					var:camp_selected_state_id = { is_valid_camp_active_site_state = yes is_controlled_by = ROOT }
				}
			}
			camp_gui_allocate_guards_click_enabled = {
				ROOT = {
					has_active_camp_network = yes
					NOT = { has_country_flag = camp_rework_guards_allocated }
					has_political_power > constant:camp_rework_cost.guard_pp
					manpower > constant:camp_rework_cost.guard_manpower
					command_power > constant:camp_rework_cost.guard_command
					has_equipment = { infantry_equipment > constant:camp_rework_equipment_cost.infantry_equipment }
				}
			}
			camp_gui_reduce_quotas_click_enabled = {
				ROOT = {
					has_active_camp_network = yes
					check_variable = { camp_labor_output > constant:camp_rework_threshold.band_low }
					NOT = { has_country_flag = camp_rework_quotas_reduced }
					has_political_power > constant:camp_rework_cost.guard_pp
				}
			}
			camp_gui_inspect_selected_site_click_enabled = {
				ROOT = {
					has_variable = camp_selected_state_id
					NOT = { has_country_flag = camp_rework_inspection_active }
					has_political_power > constant:camp_rework_cost.review_pp
					var:camp_selected_state_id = {
						has_variable = genocide_responsible_country
						var:genocide_responsible_country = { tag = ROOT }
					}
				}
			}
			camp_gui_dismantle_selected_site_click_enabled = {
				ROOT = {
					has_variable = camp_selected_state_id
					camp_rework_country_can_reform = yes
					NOT = { has_country_flag = camp_rework_gui_dismantlement_active }
					has_political_power > constant:camp_rework_cost.dismantle_pp
					manpower > constant:camp_rework_cost.dismantle_manpower
					has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
					var:camp_selected_state_id = { is_valid_camp_dismantlement_state = yes }
				}
			}
			camp_gui_destroy_evidence_click_enabled = {
				ROOT = {
					has_variable = camp_selected_state_id
					has_political_power > constant:camp_rework_cost.evidence_pp
					manpower > constant:camp_rework_cost.activation_manpower
					command_power > constant:camp_rework_cost.evidence_command
					has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
					var:camp_selected_state_id = { is_valid_camp_enemy_proximity_state = yes }
				}
			}
			camp_gui_chemical_method_click_enabled = {
				ROOT = {
					has_variable = camp_selected_state_id
					has_political_power > constant:camp_rework_cost.chemical_pp
					camp_rework_country_has_chemical_escalation_capacity = yes
					var:camp_selected_state_id = { camp_rework_state_is_chemical_escalation_target = yes }
				}
			}
			camp_gui_biological_method_click_enabled = {
				ROOT = {
					has_variable = camp_selected_state_id
					has_political_power > constant:camp_rework_cost.biological_pp
					camp_rework_country_has_biological_escalation_capacity = yes
					var:camp_selected_state_id = { camp_rework_state_is_biological_escalation_target = yes }
				}
			}
		}
	}
}
