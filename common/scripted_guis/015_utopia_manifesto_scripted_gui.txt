# ============================================================================
# CHAOS REDUX - EVENT 015 COMMONWEALTH LEDGER SCRIPTED GUI
# ============================================================================
# Provides four player-facing pages for the Ledger overview, national callings,
# stores and settlements, and Necessary Ground with associates. A single tab
# state variable drives all page selection so stale mutually-exclusive flags
# cannot leave the panel in an ambiguous state. Gameplay actions
# remain in the mapped decision families so costs, duration, AI, and cleanup stay
# identical for visible and AI-controlled countries.
# ============================================================================

scripted_gui = {
	utopia_manifesto_ledger_scripted_gui = {
		context_type = decision_category
		window_name = "utopia_manifesto_ledger_container"

		ai_enabled = { always = no }
		visible = {
			is_ai = no
			has_country_flag = utopia_manifesto_ledger_visible
		}

		effects = {
			utopia_ledger_refresh_button_click = {
				utopia_manifesto_refresh_ledger = yes
				utopia_manifesto_refresh_stewardship_proof = yes
				utopia_manifesto_refresh_league_state = yes
				utopia_manifesto_refresh_formation_proof = yes
			}
			utopia_ledger_tab_overview_button_click = {
				set_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.overview }
			}
			utopia_ledger_tab_callings_button_click = {
				set_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.callings }
			}
			utopia_ledger_tab_stores_button_click = {
				set_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.stores }
			}
			utopia_ledger_tab_ground_button_click = {
				set_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.ground }
			}
		}

		triggers = {
			utopia_ledger_refresh_button_click_enabled = {
				has_country_flag = utopia_manifesto_ledger_visible
			}
			utopia_ledger_tab_overview_button_click_enabled = { has_country_flag = utopia_manifesto_ledger_visible }
			utopia_ledger_tab_callings_button_click_enabled = { has_country_flag = utopia_manifesto_ledger_visible }
			utopia_ledger_tab_stores_button_click_enabled = { has_country_flag = utopia_manifesto_ledger_visible }
			utopia_ledger_tab_ground_button_click_enabled = { has_country_flag = utopia_manifesto_ledger_visible }

			utopia_ledger_overview_panel_visible = {

					check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.overview }

				}
			}
			utopia_ledger_callings_panel_visible = { check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.callings } }
			utopia_ledger_stores_panel_visible = { check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.stores } }
			utopia_ledger_ground_panel_visible = { check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.ground } }
			utopia_ledger_tab_overview_selected_visible = {
				OR = {
					check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.overview }
					NOT = { has_variable = utopia_manifesto_ledger_tab }
				}
			}
			utopia_ledger_tab_callings_selected_visible = { check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.callings } }
			utopia_ledger_tab_stores_selected_visible = { check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.stores } }
			utopia_ledger_tab_ground_selected_visible = { check_variable = { utopia_manifesto_ledger_tab = constant:utopia_manifesto_ledger_tab.ground } }

			utopia_ledger_need_warning_visible = {

					utopia_manifesto_need_is_high = yes

				has_country_flag = utopia_manifesto_balance_shift_to_choice_recent
			}
			utopia_ledger_balance_to_assignment_visible = {
				has_country_flag = utopia_manifesto_balance_shift_to_assignment_recent
			}

			# Necessary Ground case cards use mutually exclusive live-state priority.

				has_country_flag = utopia_manifesto_stewardship_active
			}
			utopia_ledger_case_refusal_visible = {
				has_country_flag = utopia_manifesto_need_case_active
				NOT = { has_country_flag = utopia_manifesto_stewardship_active }

					has_country_flag = utopia_manifesto_case_settlement_refused
					has_country_flag = utopia_manifesto_case_ultimatum_refused
					has_country_flag = utopia_manifesto_case_target_refused

			}
			utopia_ledger_case_counteroffer_visible = {
				has_country_flag = utopia_manifesto_need_case_active
				has_country_flag = utopia_manifesto_case_counteroffer_recorded
				NOT = { has_country_flag = utopia_manifesto_stewardship_active }
				NOR = {
					has_country_flag = utopia_manifesto_case_settlement_refused
					has_country_flag = utopia_manifesto_case_ultimatum_refused
					has_country_flag = utopia_manifesto_case_target_refused

			utopia_ledger_case_ultimatum_available_visible = {

				NOT = { has_country_flag = utopia_manifesto_stewardship_active }
				NOR = {
					has_country_flag = utopia_manifesto_case_response_active
					has_country_flag = utopia_manifesto_case_counteroffer_recorded
					has_country_flag = utopia_manifesto_case_settlement_refused
					has_country_flag = utopia_manifesto_case_ultimatum_refused
					has_country_flag = utopia_manifesto_case_target_refused
				}

			utopia_ledger_case_offer_pending_visible = {
				has_country_flag = utopia_manifesto_need_case_active
				has_country_flag = utopia_manifesto_case_response_active
				NOT = { has_country_flag = utopia_manifesto_stewardship_active }
				NOR = {
					has_country_flag = utopia_manifesto_case_counteroffer_recorded
					has_country_flag = utopia_manifesto_case_settlement_refused
					has_country_flag = utopia_manifesto_case_ultimatum_refused
					has_country_flag = utopia_manifesto_case_target_refused

				NOT = { has_country_flag = utopia_manifesto_stewardship_active }

			utopia_ledger_case_target_eligible_visible = {
				NOT = { has_country_flag = utopia_manifesto_need_case_active }

					var = utopia_manifesto_case_candidate_targets^num

					compare = greater_than
				}
			}
			utopia_ledger_case_associate_established_visible = {
				NOT = { has_country_flag = utopia_manifesto_need_case_active }

				NOT = { has_country_flag = utopia_manifesto_stewardship_active }

					var = utopia_manifesto_case_candidate_targets^num
					value = constant:utopia_manifesto_math.zero
					compare = equals
				}
				OR = {
					has_country_flag = utopia_manifesto_first_associate_recognized
					check_variable = {
						var = utopia_manifesto_recognized_associates^num
						value = constant:utopia_manifesto_math.zero
						compare = greater_than
					}
				}
			}
			utopia_ledger_case_expired_visible = {
				has_country_flag = utopia_manifesto_case_expired

				NOT = { has_variable = utopia_manifesto_selected_country_id }
				NOT = { has_country_flag = utopia_manifesto_stewardship_active }
				NOT = { has_country_flag = utopia_manifesto_first_associate_recognized }
				check_variable = {
					var = utopia_manifesto_recognized_associates^num
					value = constant:utopia_manifesto_math.zero
					compare = equals
				}
				check_variable = {
					var = utopia_manifesto_case_candidate_targets^num
					value = constant:utopia_manifesto_math.zero
					compare = equals
				}
			}
			utopia_ledger_case_no_target_visible = {
				NOT = { has_country_flag = utopia_manifesto_need_case_active }
				NOT = { has_variable = utopia_manifesto_selected_country_id }
				NOT = { has_country_flag = utopia_manifesto_stewardship_active }
				NOT = { has_country_flag = utopia_manifesto_case_expired }
				NOT = { has_country_flag = utopia_manifesto_first_associate_recognized }
				check_variable = {
					var = utopia_manifesto_recognized_associates^num
					value = constant:utopia_manifesto_math.zero

				}

					var = utopia_manifesto_case_candidate_targets^num
					value = constant:utopia_manifesto_math.zero

				check_variable = { utopia_manifesto_district_visual_role = constant:utopia_manifesto_decision_district_role.rail_junction }

				check_variable = { utopia_manifesto_district_visual_role = constant:utopia_manifesto_decision_district_role.port_town }
			}
			utopia_ledger_district_role_research_town_visible = {
				check_variable = { utopia_manifesto_district_visual_role = constant:utopia_manifesto_decision_district_role.research_town }

								has_state_flag = utopia_manifesto_district_obligations_breached

						}

			utopia_ledger_district_state_blocked_visible = {
				NOR = {
					has_country_flag = utopia_manifesto_district_obligation_dispute_unresolved

							OR = {
								has_state_flag = utopia_manifesto_district_obligations_breached
								has_state_flag = utopia_manifesto_district_project_refused
								has_state_flag = utopia_manifesto_district_project_lost
							}
						}
					}
				}
				OR = {
					has_country_flag = utopia_manifesto_district_survey_found_no_role
					has_country_flag = utopia_manifesto_district_project_debt
					check_variable = { utopia_manifesto_district_project_phase = constant:utopia_manifesto_district_phase.delayed }

						has_variable = utopia_manifesto_district_project_state

				}

			utopia_ledger_district_state_complete_visible = {
				NOR = {

					check_variable = { utopia_manifesto_district_project_phase = constant:utopia_manifesto_district_phase.delayed }
					AND = {
						has_variable = utopia_manifesto_district_project_state
						var:utopia_manifesto_district_project_state = {

								has_state_flag = utopia_manifesto_district_obligations_breached
								has_state_flag = utopia_manifesto_district_project_refused
								has_state_flag = utopia_manifesto_district_project_lost
								has_state_flag = utopia_manifesto_district_project_delayed
							}
						}
					}
				}
				OR = {
					check_variable = { utopia_manifesto_district_project_phase = constant:utopia_manifesto_district_phase.built }

				NOR = {
					has_country_flag = utopia_manifesto_district_obligation_dispute_unresolved
					has_country_flag = utopia_manifesto_district_project_debt
					has_country_flag = utopia_manifesto_district_survey_found_no_role
					var:utopia_manifesto_district_project_state = {
