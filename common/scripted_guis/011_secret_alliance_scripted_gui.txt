scripted_gui = {
	secret_alliance_dossier_scripted_gui = {
		context_type = decision_category
		window_name = "secret_alliance_dossier_container"

		ai_enabled = { always = no }
		visible = {
			is_ai = no
			secret_alliance_target_has_active_pact = yes
		}

		triggers = {
			secret_alliance_dossier_evidence_pulse_visible = {
				OR = {
					has_country_flag = secret_alliance_dossier_open
					has_country_flag = secret_alliance_couriers_traced
					has_country_flag = secret_alliance_payments_audited
					check_variable = {
						var = secret_alliance_evidence
						value = constant:secret_alliance_meter.low
						compare = greater_than_or_equals
					}
				}
			}
			secret_alliance_dossier_readiness_warning_visible = {
				OR = {
					has_country_flag = secret_alliance_protocol_deadline_active
					has_global_flag = secret_alliance_evolution_ii_unlocked
					check_variable = {
						var = global.secret_alliance_readiness
						value = constant:secret_alliance_meter.mid
						compare = greater_than_or_equals
					}
				}
			}
			secret_alliance_dossier_exposed_card_visible = {
				OR = {
					has_country_flag = secret_alliance_publicly_revealed
					has_country_flag = secret_alliance_protocol_exposed_by_target
					has_country_flag = secret_alliance_narrow_dossier_published
				}
			}
			secret_alliance_dossier_war_countdown_visible = {
				OR = {
					has_country_flag = secret_alliance_war_revealed
					has_country_flag = secret_alliance_strike_first_authorized
					has_global_flag = secret_alliance_evolution_iii_unlocked
				}
			}
			secret_alliance_dossier_hidden_protocol_visible = {
				OR = {
					NOT = { has_country_flag = secret_alliance_publicly_revealed }
					has_global_flag = secret_alliance_evolution_i_unlocked
					has_global_flag = secret_alliance_evolution_ii_unlocked
				}
			}
			secret_alliance_dossier_unknown_card_visible = {
				NOT = { has_country_flag = secret_alliance_publicly_revealed }
			}
			secret_alliance_dossier_convener_card_visible = {
				has_event_target = secret_alliance_founder_convener
				event_target:secret_alliance_founder_convener = { secret_alliance_known_member_target = yes }
			}
			secret_alliance_dossier_financier_card_visible = {
				has_event_target = secret_alliance_founder_financier
				event_target:secret_alliance_founder_financier = { secret_alliance_known_member_target = yes }
			}
			secret_alliance_dossier_provocateur_card_visible = {
				has_event_target = secret_alliance_founder_provocateur
				event_target:secret_alliance_founder_provocateur = { secret_alliance_known_member_target = yes }
			}
			secret_alliance_dossier_patron_card_visible = {
				has_event_target = secret_alliance_major_patron
				event_target:secret_alliance_major_patron = { secret_alliance_known_member_target = yes }
			}
			secret_alliance_dossier_leader_card_visible = {
				has_event_target = secret_alliance_public_leader
				has_country_flag = secret_alliance_publicly_revealed
			}
		}
	}
}
