# ============================================================================

scripted_gui = {
	secret_alliance_counter_network_scripted_gui = {
		context_type = decision_category
		window_name = "secret_alliance_counter_network_container"
		ai_enabled = { always = no }
		visible = {
			ROOT = {
				is_ai = no
				has_country_flag = secret_alliance_target_country
				has_global_flag = secret_alliance_active
				OR = {
					has_global_flag = secret_alliance_response_category_open
					has_global_flag = secret_alliance_evolution_iii
					has_global_flag = secret_alliance_revealed
				}
			}
		}
		effects = {
			secret_alliance_suspect_card_1_select_click = {
				ROOT = {
					if = {
						limit = {
							NOT = { has_global_flag = secret_alliance_border_conflict_unresolved }
							check_variable = { global.secret_alliance_visible_suspects^num > 0 }
						}
						set_variable = { secret_alliance_selected_suspect_scope = global.secret_alliance_visible_suspects^0 }
						secret_alliance_refresh_prepared_border_pair = yes
					}
				}
			}
			secret_alliance_suspect_card_2_select_click = {
				ROOT = {
					if = {
						limit = {
							NOT = { has_global_flag = secret_alliance_border_conflict_unresolved }
							check_variable = { global.secret_alliance_visible_suspects^num > 1 }
						}
						set_variable = { secret_alliance_selected_suspect_scope = global.secret_alliance_visible_suspects^1 }
						secret_alliance_refresh_prepared_border_pair = yes
					}
				}
			}
			secret_alliance_suspect_card_3_select_click = {
				ROOT = {
					if = {
						limit = {
							NOT = { has_global_flag = secret_alliance_border_conflict_unresolved }
							check_variable = { global.secret_alliance_visible_suspects^num > 2 }
						}
						set_variable = { secret_alliance_selected_suspect_scope = global.secret_alliance_visible_suspects^2 }
						secret_alliance_refresh_prepared_border_pair = yes
					}
				}
			}
			secret_alliance_gui_clear_selection_click = {
				ROOT = {
					if = {
						limit = { NOT = { has_global_flag = secret_alliance_border_conflict_unresolved } }
						clear_variable = secret_alliance_selected_suspect_scope
						secret_alliance_refresh_prepared_border_pair = yes
					}
				}
			}
			secret_alliance_toggle_animation_click = {
				ROOT = {
					if = { limit = { has_country_flag = secret_alliance_animations_disabled } clr_country_flag = secret_alliance_animations_disabled }
					else = { set_country_flag = secret_alliance_animations_disabled }
				}
			}
		}
		triggers = {
			secret_alliance_suspect_card_1_select_visible = { ROOT = { check_variable = { global.secret_alliance_visible_suspects^num > 0 } } }
			secret_alliance_suspect_card_2_select_visible = { ROOT = { check_variable = { global.secret_alliance_visible_suspects^num > 1 } } }
			secret_alliance_suspect_card_3_select_visible = { ROOT = { check_variable = { global.secret_alliance_visible_suspects^num > 2 } } }
			secret_alliance_suspect_card_1_select_click_enabled = { ROOT = { NOT = { has_global_flag = secret_alliance_border_conflict_unresolved } check_variable = { global.secret_alliance_visible_suspects^num > 0 } } }
			secret_alliance_suspect_card_2_select_click_enabled = { ROOT = { NOT = { has_global_flag = secret_alliance_border_conflict_unresolved } check_variable = { global.secret_alliance_visible_suspects^num > 1 } } }
			secret_alliance_suspect_card_3_select_click_enabled = { ROOT = { NOT = { has_global_flag = secret_alliance_border_conflict_unresolved } check_variable = { global.secret_alliance_visible_suspects^num > 2 } } }
			secret_alliance_gui_clear_selection_visible = { ROOT = { has_variable = secret_alliance_selected_suspect_scope } }
			secret_alliance_gui_clear_selection_click_enabled = { ROOT = { NOT = { has_global_flag = secret_alliance_border_conflict_unresolved } has_variable = secret_alliance_selected_suspect_scope } }
			secret_alliance_hidden_state_text_visible = { ROOT = { has_global_flag = secret_alliance_evolution_ii NOT = { has_global_flag = secret_alliance_revealed } } }
			secret_alliance_warning_animation_visible = { ROOT = { has_global_flag = secret_alliance_evolution_iii NOT = { has_global_flag = secret_alliance_revealed } NOT = { has_country_flag = secret_alliance_animations_disabled } } }
			secret_alliance_warning_static_fallback_visible = { ROOT = { has_global_flag = secret_alliance_evolution_iii NOT = { has_global_flag = secret_alliance_revealed } has_country_flag = secret_alliance_animations_disabled } }
			secret_alliance_war_pressure_text_visible = { ROOT = { has_global_flag = secret_alliance_evolution_iii } }
			secret_alliance_toggle_animation_visible = { ROOT = { has_global_flag = secret_alliance_evolution_iii NOT = { has_global_flag = secret_alliance_revealed } } }
			secret_alliance_toggle_animation_click_enabled = { ROOT = { has_global_flag = secret_alliance_evolution_iii NOT = { has_global_flag = secret_alliance_revealed } } }
		}
		properties = {
			secret_alliance_evidence_meter_fill = { x = ROOT.secret_alliance_gui_evidence_fill_x }
			secret_alliance_preparedness_meter_fill = { x = ROOT.secret_alliance_gui_preparedness_fill_x }
			secret_alliance_suspect_card_1 = { frame = ROOT.secret_alliance_gui_card_1_frame }
			secret_alliance_suspect_card_2 = { frame = ROOT.secret_alliance_gui_card_2_frame }
			secret_alliance_suspect_card_3 = { frame = ROOT.secret_alliance_gui_card_3_frame }
		}
	}
}
