# ============================================================================
# CHAOS REDUX - SHARED DISEASE CONTAINMENT SCRIPTED GUI
# ============================================================================
# The category header and popup are human presentation surfaces. Disease
# actions remain decisions, and AI evaluates disease-owned targets directly.
# Event targets are deliberately absent from this interface contract.
# ============================================================================

scripted_gui = {
	disease_containment_header_scripted_gui = {
		context_type = decision_category
		window_name = "disease_containment_header_window"

		visible = {
			is_ai = no
			disease_containment_board_category_is_visible = yes
			NOT = { has_global_flag = world_end }
		}

		effects = {
			disease_containment_board_open_click = { disease_containment_board_open = yes }
		}

		triggers = {
			disease_containment_board_open_click_enabled = { disease_containment_board_category_is_visible = yes }
			disease_containment_header_black_plague_visible = { disease_containment_board_is_black_plague_selected = yes }
			disease_containment_header_black_plague_infestation_visible = { disease_containment_board_is_black_plague_selected = yes }
			disease_containment_header_black_plague_infestation_animated_visible = { disease_containment_board_is_black_plague_selected = yes }
		}

		ai_enabled = { always = no }
	}

	disease_containment_board_scripted_gui = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "disease_containment_board_window"
		dirty = disease_containment_board_dirty

		visible = {
			is_ai = no
			has_country_flag = disease_containment_board_open
			disease_containment_board_category_is_visible = yes
			NOT = { has_global_flag = world_end }
		}

		effects = {
			disease_containment_board_close_click = { disease_containment_board_close = yes }
			disease_containment_board_refresh_click = { disease_containment_board_refresh = yes }
			disease_containment_board_goto_click = { disease_containment_board_goto_view_state = yes }
			disease_containment_board_tab_anthrax_click = { disease_containment_board_select_anthrax_tab = yes }
			disease_containment_board_tab_plague_click = { disease_containment_board_select_plague_tab = yes }
			disease_containment_board_tab_tularemia_click = { disease_containment_board_select_tularemia_tab = yes }
			disease_containment_board_tab_smallpox_click = { disease_containment_board_select_smallpox_tab = yes }
			disease_containment_board_tab_black_plague_click = { disease_containment_board_select_black_plague_tab = yes }
			disease_containment_board_status_filter_click = { disease_containment_board_cycle_status_filter = yes }
			disease_containment_board_ownership_filter_click = { disease_containment_board_cycle_ownership_filter = yes }
			disease_containment_board_region_filter_click = { disease_containment_board_toggle_region_filter = yes }
			disease_containment_board_urgency_filter_click = { disease_containment_board_cycle_urgency_filter = yes }
			disease_containment_board_state_entry_select_click = { ROOT = { disease_containment_board_select_state_entry = yes } }
		}

		triggers = {
			disease_containment_board_selected_card_visible = { disease_containment_board_has_valid_view_state = yes }
			disease_containment_board_selected_card_private_values_visible = {
				AND = {
					disease_containment_board_view_state_has_private_access = yes
					NOT = { disease_containment_board_is_black_plague_selected = yes }
				}
			}
			disease_containment_board_selected_card_public_values_visible = {
				AND = {
					disease_containment_board_view_state_is_public_foreign = yes
					NOT = { disease_containment_board_is_black_plague_selected = yes }
				}
			}
			disease_containment_board_response_target_visible = { black_plague_country_has_valid_response_state = yes }
			disease_containment_board_black_plague_metrics_visible = { disease_containment_board_is_black_plague_selected = yes }
			disease_containment_board_goto_click_enabled = { disease_containment_board_has_valid_view_state = yes }
			disease_containment_board_region_filter_click_enabled = {
				OR = {
					check_variable = { disease_containment_board_region_filter = constant:disease_containment_region_filter.selected_region }
					disease_containment_board_has_valid_view_state = yes
				}
			}
			disease_containment_board_state_entry_select_click_enabled = { always = yes }
			disease_containment_board_state_entry_selected_visible = { disease_containment_board_entry_is_selected = yes }
			disease_containment_board_state_entry_response_visible = { disease_containment_board_entry_is_response_target = yes }
			disease_containment_board_no_states_visible = {
				check_variable = { disease_containment_board_state_count = constant:disease_containment_board_value.zero }
			}
		}

		dynamic_lists = {
			disease_containment_board_state_dynamic_list = {
				array = disease_containment_board_state_entries
				value = disease_containment_board_state_entry
				index = disease_containment_board_state_entry_index
				change_scope = yes
				entry_container = "disease_containment_board_state_entry"
			}
		}

		ai_enabled = { always = no }
	}
}
