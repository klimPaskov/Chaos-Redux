		}
		modifier = {
			factor = @CBRN_CHAOS_WARFARE_AI_VIABLE_PROGRAM
			cbrn_chaos_warfare_ai_has_viable_program = yes
		}
		modifier = {
			factor = @CBRN_CHAOS_WARFARE_AI_PREPARED_PROFILE

			factor = @CBRN_CHAOS_WARFARE_AI_WAR
