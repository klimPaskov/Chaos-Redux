# ============================================================================
# CHAOS REDUX - EVENT 038 MALTA CRUSADERS FOCUS STRATEGY PLANS
# ============================================================================
# Twelve plans steer the AI through the Malta focus trees. The opening and the
# Eleventh Crusade plans carry a fixed focus order, because those chains are
# linear and must be followed. Every other plan only multiplies the focus
# weights of one lane, so an invalid route simply never matches: its enable
# reads the route flag the focus tree itself sets, and its abort fires only on
# the opposing terminal state (another settlement route, another dominant
# order, a Kingdom, an independence claim), never on a route that is merely not
# chosen yet. Nothing here iterates the world: the plans are
# evaluated by the engine for the single country that loaded the tree.
#
# Plan                      Tree      Valid while
#   survival first            Malta     the council is not yet convened and held
#   Grand Magistry            Malta     malta_route_magistry
#   Langues confederation     Malta     malta_route_langues
#   civilian constitutional   Malta     malta_route_civil
#   Hospitaller defensive     Malta     the Hospitallers are the dominant order
#   Templar financier         Malta     the Templars are the dominant order
#   Teutonic heavy armour     Malta     the Teutonic Order is the dominant order
#   naval island campaign     Malta     the Naval Order is the dominant order
#   siege conquest            Malta     the Siege Order is the dominant order
#   Papal Holy See            Malta     the western question is open, no Kingdom
#   Eleventh recovery         Malta     the Eleventh Crusade is running
#   principality federation   Principality  the federation exists, loyalty holds
#
# The hidden Teutonic alliance and the Holy World preparation plans of the
# specification belong to the deferred hidden routes and the terminal; they are
# written with those routes. File-scoped constants are used because plan
# factors do not parse script constants.
# ============================================================================

@mlt_ai_plan_weight = 1.0
@mlt_ai_plan_lane = 2.0
@mlt_ai_plan_support = 1.5
@mlt_ai_plan_damped = 0.5

# ---------------------------------------------------------------------------
# Opening
# ---------------------------------------------------------------------------

MLT_focus_survival_first_plan = {
	name = "Malta Survival First Plan"
	desc = "Hold the anchors, convene the council and secure the sea road and the Holy Land before any region is opened."
	allowed = { original_tag = MLT }
	enable = {
		has_country_flag = malta_origin
		NOT = { has_country_flag = malta_eleventh_active }
	}
	abort = {
		has_completed_focus = mcf_convene_the_council
		malta_standing_is_functional_or_better = yes
	}

	ai_national_focuses = {
		mcf_crusade_is_underway
		mcf_hold_the_four_anchors
		mcf_convene_the_council
		mcf_secure_the_sea_road
		mcf_fortify_the_holy_land
		mcf_secure_jerusalem
	}

	focus_factors = {
		mcf_fortress_malta = @mlt_ai_plan_support
		mcf_chain_of_fortified_ports = @mlt_ai_plan_support
		mcf_the_levantine_frontier = @mlt_ai_plan_damped
		mcf_islands_of_the_cross = @mlt_ai_plan_damped
		mcf_gates_of_anatolia = @mlt_ai_plan_damped
		mcf_fortresses_of_the_southern_sea = @mlt_ai_plan_damped
	}

	weight = { factor = @mlt_ai_plan_weight }
}
