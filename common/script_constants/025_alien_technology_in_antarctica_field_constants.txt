# ============================================================================
# CHAOS REDUX - EVENT 025 FIELD HAZARDS AND RARE INCIDENTS
# ============================================================================
# Overview:
# - Tuning for the baseline field hazards rolled by an expedition's own pulse
#   once it has a camp on the ice, the rare once-per-expedition incidents, and
#   the stricter pace of a solo expedition.
# - Chances are percentages per pulse. Consequences are positive amounts; the
#   field effects subtract or add them.
# - Nothing here depends on an evolution, so the baseline race keeps its
#   hazards with every evolution disabled.
# ============================================================================

# Hazard chance and consequences.
chaosx_nr25_hazard = {
	schema = {
		any_key = yes
		data = fixed_point
	}

	# Chance per pulse, in percent, before preparation.
	base_chance = 12
	# Exposure Risk raises the chance.
	exposure_moderate = 50
	exposure_moderate_add = 6
	exposure_severe = 80
	exposure_severe_add = 8
	# Each general preparation (forward depot, radio net, polar crew) lowers
	# the chance.
	prepared_factor = 0.85
	# A well-supplied expedition (Logistics Readiness at the final-recovery
	# safety level) meets fewer hazards; a strained one meets more.
	ready_factor = 0.75
	strained_factor = 1.3
	# A logistics depot or fortified camp emphasis lowers the chance.
	emphasis_factor = 0.8
	# A solo expedition races the weather alone.
	solo_factor = 1.25
	max_chance = 40
	# Draw weights: a hazard whose own preparation was made is rarer.
	weight = 10
	prepared_weight = 4
	# A hazard whose own preparation was made does half the damage and kills
	# no one.
	mitigated_factor = 0.5
	# Consequences.
	pack_ice_progress = 6
	pack_ice_readiness = 4
	whiteout_exposure = 8
	whiteout_readiness = 3
	crevasse_crew = 1
	crevasse_readiness = 5
	radio_loss_intelligence = 6
	radio_loss_exposure = 4
	engines_fuel = 100
	engines_readiness = 6
	sledges_progress = 5
	sledges_readiness = 4
	medical_crew = 1
	medical_readiness = 4
	# Solo expedition pace.
	solo_exposure_pulse = 1
	solo_progress_drain = 1
}
