# ============================================================================
# CHAOS REDUX - EVENT 025 DIPLOMATIC INCIDENTS
# ============================================================================
# Overview:
# - Tuning for the diplomatic follow-through of an identified rival
#   operation (public accusation or silence, then denial, compensation or an
#   admitted limited operation) and of an armed seizure (a demand to return
#   people and material, accepted or refused).
# - Every exchange is bilateral; only a public step adds world tension, and
#   nothing here declares a war.
# ============================================================================

chaosx_nr25_incident = {
	schema = {
		any_key = yes
		data = fixed_point
	}

	# World tension of a public accusation or a refused demand.
	public_threat = 1
	# Keeping an identified operation quiet keeps an intelligence edge.
	quiet_intelligence = 6
	# Compensation paid by an accused government: the political power it
	# needs and the signed amount it pays.
	compensation_pp = 30
	compensation_pp_delta = -30
	compensation_readiness = 8
	compensation_evidence = 5
	# An admitted limited operation.
	admit_pp = 10
	admit_pp_delta = -10
	admit_exposure = 6
	# A settled seizure returns personnel and lowers militarisation.
	exchange_crew = 1
	exchange_militarisation_drop = 8
	refused_militarisation_rise = 3

	# Answers stored for the accuser's report.
	answer_denied = 1
	answer_compensated = 2
	answer_admitted = 3
}
