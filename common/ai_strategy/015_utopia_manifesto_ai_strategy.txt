utopia_manifesto_foundation_restraint = {
	allowed = { always = yes }
	enable = {
		has_country_flag = utopia_manifesto_accepted
		NOT = {
			OR = {
				has_country_flag = utopia_manifesto_route_consent_of_households
				has_country_flag = utopia_manifesto_route_common_table
				has_country_flag = utopia_manifesto_route_guardians_of_measure
				has_country_flag = utopia_manifesto_route_closed_island
				has_country_flag = utopia_manifesto_route_joke_understood
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = avoid_starting_wars
		value = @utopia_ai_strategy_war_restraint
	}
	ai_strategy = {
		type = build_building
		id = infrastructure
		value = @utopia_ai_strategy_moderate
	}
}
