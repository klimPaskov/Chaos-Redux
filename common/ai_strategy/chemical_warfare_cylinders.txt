# ============================================================================
# CHAOS REDUX - MODERN CHEMICAL PAYLOAD PRODUCTION AI
# ============================================================================
# Overview:
# - Retains the compatibility filename while retiring AI pressure for legacy
#   cylinder equipment.
# - Produces the modern agent-lot and shell-lot families only after protection,
#   conventional supply, policy, route, and industry gates pass.
# - Uses exact model surplus weights to preserve country agent identities.
# - Produces no chemical air payload while the exact native raid adapter remains
#   unavailable; idle or merely capable aircraft never create a production or
#   contamination signal.
# - `ai_strategy.value` does not parse script constants in this context, so all
#   production values and stock targets are file-local constants.
# ============================================================================

cbrn_payload_pressure_retaliatory = {
	allowed = { NOT = { is_actual_nonhuman_country = yes } }
	enable = {
		cbrn_ai_can_expand_offensive_cbrn_production = yes
		cbrn_ai_has_retaliatory_arsenal_posture = yes
		NOT = { cbrn_ai_has_battlefield_program_posture = yes }
		NOT = { cbrn_ai_has_strategic_program_posture = yes }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = equipment_production_factor
		id = chemical_agent_payload
		value = @CBRN_PAYLOAD_PRESSURE_RETALIATORY
	}
}
