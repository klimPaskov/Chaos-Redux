	ai_strategy = { type = force_concentration_factor value = 100 }

# Hierarchy route strategies consume the completed RTA hierarchy variable. They
# only bias existing rat templates and front requests. Focus AI remains the
# owner of the one-time Distributed, Dominant, or Emergent route choice.

	allowed = { original_tag = RTA }
	enable = {
		has_country_flag = black_plague_rat_country
		black_plague_rat_hierarchy_is_distributed = yes
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = template_prio id = rat_burrowers value = constant:black_plague_rat_hierarchy_route.distributed_burrow_template_priority }
	ai_strategy = { type = template_prio id = rat_swarm value = constant:black_plague_rat_hierarchy_route.distributed_swarm_template_priority }
	ai_strategy = { type = role_ratio id = rat_burrowers value = 2 }
	ai_strategy = { type = spare_unit_factor value = constant:black_plague_rat_hierarchy_route.distributed_spare_unit_factor }
	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			NOT = { is_controlled_by = ROOT }
			any_neighbor_state = { is_controlled_by = ROOT }
		}
		value = constant:black_plague_rat_hierarchy_route.distributed_front_request_priority
	}
}
