# ============================================================================
# CHAOS REDUX - EVENT 016 GENERIC PROJECT-FORCE AI PRODUCTION
# ============================================================================
# Overview:
# - Assigns a bounded formation share and a real production floor to every
#   trainable reusable project-force family after its hidden grant technology.
# - Strategies abort immediately when the technology is unavailable.
# - Alien Infantry is excluded because its API reserves laser stock and creates
#   cohorts only through scripted landings.
# ============================================================================

event016_clone_infantry_production = {
	allowed = { always = yes }
	enable = {
		OR = {
			has_tech = clone_infantry_access_tech
			has_tech = brilliant_scientist_clone_formations_tech
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = role_ratio id = clone_infantry value = @EVENT016_AI_ROLE_CLONE }
	ai_strategy = { type = equipment_production_min_factories_archetype id = clone_equipment value = @EVENT016_AI_FACTORIES_CLONE }
}
