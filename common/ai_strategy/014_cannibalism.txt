cannibalism_outbreak_transport_surge = {
	allowed = { always = yes }
	enable = {
		cannibalism_country_has_outbreak = yes
		NOT = { is_cannibal_commune_country = yes }
		OR = {
			has_country_flag = cannibalism_guard_ration_rail_active
			has_country_flag = cannibalism_prison_kitchen_active
			has_country_flag = cannibalism_island_inspection_active
			has_country_flag = cannibalism_evacuation_active
			cannibalism_country_is_in_island_stage = yes
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_factor
		id = train
		value = @cannibalism_ai_outbreak_transport_surge
	}
	ai_strategy = {
		type = equipment_production_factor
		id = convoy
		value = @cannibalism_ai_outbreak_transport_surge
	}
	ai_strategy = {
		type = build_building
		id = infrastructure
		value = @cannibalism_ai_outbreak_supply_building
	}
	ai_strategy = {
		type = build_building
		id = naval_base
		value = @cannibalism_ai_outbreak_supply_building
	}
}

cannibalism_outbreak_crisis_containment = {
	allowed = { always = yes }
	enable = {
		cannibalism_country_has_outbreak = yes
		NOT = { is_cannibal_commune_country = yes }
		OR = {
			check_variable = {
				var = cannibalism_hunger
				value = constant:cannibalism_meter.high
				compare = greater_than_or_equals
			}
			check_variable = {
				var = cannibalism_cult
				value = constant:cannibalism_meter.high
				compare = greater_than_or_equals
			}
			check_variable = {
				var = cannibalism_spread
				value = constant:cannibalism_meter.high
				compare = greater_than_or_equals
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_army
		value = @cannibalism_ai_outbreak_crisis_containment
	}
	ai_strategy = {
		type = avoid_starting_wars
		value = @cannibalism_ai_outbreak_crisis_containment
	}
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = @cannibalism_ai_outbreak_infantry_production
	}
	ai_strategy = {
		type = equipment_production_factor
		id = support
		value = @cannibalism_ai_outbreak_support_production
	}
}
