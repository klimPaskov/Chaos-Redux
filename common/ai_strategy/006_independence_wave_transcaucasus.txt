# ============================================================================
# These strategies are additive to the vanilla ARM/GEO/AZR profiles. They
# keep each carrier alive, prioritize the researched regional force profile,
# and delay opportunistic wars until the founding and arbitration receipts are
# complete. FORM-16 receives a separate federal consolidation posture.

independence_wave_transcaucasus_frontier_survival = {
	allowed = {
		OR = { original_tag = ARM original_tag = GEO original_tag = AZR }
	}
	enable = {
		OR = {
			is_independence_wave_iw070_country = yes
			is_independence_wave_iw071_country = yes
			is_independence_wave_iw072_country = yes
		}
		has_country_flag = independence_wave_generic_ai_profile
		NOT = { has_country_flag = independence_wave_formable_active }
		OR = {
			has_independence_wave_severe_instability = yes
			NOT = { has_independence_wave_functioning_capacity = yes }
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = build_army value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_SURVIVAL_ARMY }
	ai_strategy = { type = equipment_production_factor id = infantry value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_SURVIVAL_INFANTRY }
	ai_strategy = { type = equipment_production_factor id = support value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_SURVIVAL_SUPPORT }
	ai_strategy = { type = equipment_production_factor id = artillery value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_SURVIVAL_ARTILLERY }
	ai_strategy = { type = build_building id = infrastructure value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_SURVIVAL_INFRASTRUCTURE }
	ai_strategy = { type = build_building id = bunker value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_SURVIVAL_DEFENSE }
	ai_strategy = { type = avoid_starting_wars value = @CR_SC_INDEPENDENCE_WAVE_TRANSCAUCASUS_RESTRAINT }
}
