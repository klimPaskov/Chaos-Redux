# Camps and repression-network AI strategy package.
# All construction and escalation is decision-owned so the shared state pools,
# responsibility pointer, Deaths registration, and country caps cannot be
# bypassed. These strategies only adjust broad behavior around that decision
# layer. Exact caps live in camp_rework_country_under_ai_* scripted triggers.

@GENOCIDE_AI_ACTIVE_BUILD_ARMY = 35
@GENOCIDE_AI_EXPOSED_AVOID_WARS = 80
@GENOCIDE_AI_SANCTIONED_BUILD_ARMY = 60
@GENOCIDE_AI_JAP_INFANTRY_MIN_FACTORIES = 3
@GENOCIDE_AI_SOV_INFANTRY_MIN_FACTORIES = 3
@GENOCIDE_AI_CAP_REACHED_AVOID_WARS = 55
@GENOCIDE_AI_REFORM_AVOID_WARS = 100

genocide_crisis_active_regime_runtime = {
	allowed = { always = yes }
	enable = {
		genocide_country_has_crisis = yes
		OR = {
			original_tag = GER
			original_tag = JAP
			original_tag = SOV
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_army
		value = @GENOCIDE_AI_ACTIVE_BUILD_ARMY
	}
}

genocide_crisis_exposed_regime_runtime = {
	allowed = { always = yes }
	enable = {
		is_ai = yes
		OR = {
			has_country_flag = genocide_evidence_reached_foreign_observers
			has_country_flag = camp_rework_crisis_exposed
			AND = {
				genocide_country_has_discovered_sites = yes
				genocide_country_condemnation_75 = yes
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = avoid_starting_wars
		value = @GENOCIDE_AI_EXPOSED_AVOID_WARS
	}
	ai_strategy = {
		type = build_army
		value = @GENOCIDE_AI_SANCTIONED_BUILD_ARMY
	}
}

camp_repression_historical_cap_guard_runtime = {
	allowed = { always = yes }
	enable = {
		is_ai = yes
		has_active_camp_network = yes
		NOT = { camp_rework_country_under_ai_site_cap = yes }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = avoid_starting_wars
		value = @GENOCIDE_AI_CAP_REACHED_AVOID_WARS
	}
}

camp_repression_reform_runtime = {
	allowed = { always = yes }
	enable = {
		is_ai = yes
		OR = {
			has_country_flag = camp_rework_reform_route_open
			has_country_flag = camp_rework_expansion_frozen
			has_country_flag = camp_rework_dismantlement_in_progress
			has_country_flag = camp_rework_reformed_legacy
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = avoid_starting_wars
		value = @GENOCIDE_AI_REFORM_AVOID_WARS
	}
}
