# Additive strategy layers for vanilla CAT. The carrier keeps vanilla history

# host settlement, and chosen route government.

independence_wave_cat_industrial_survival = {
	allowed = { original_tag = CAT }
	enable = {
		is_independence_wave_cat_package = yes
		has_country_flag = independence_wave_iw_014_setup_complete
		has_country_flag = independence_wave_cat_ai_profile
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = build_army value = constant:independence_wave_catalonia_ai.army_priority }
	ai_strategy = { type = equipment_production_factor id = infantry value = constant:independence_wave_catalonia_ai.production_priority }
	ai_strategy = { type = equipment_production_factor id = artillery value = constant:independence_wave_catalonia_ai.artillery_priority }
	ai_strategy = { type = equipment_production_factor id = support value = constant:independence_wave_catalonia_ai.support_priority }
	ai_strategy = { type = build_building id = infrastructure value = constant:independence_wave_catalonia_ai.infrastructure_priority }
	ai_strategy = { type = build_building id = bunker value = constant:independence_wave_catalonia_ai.defense_priority }
}
