# ============================================================================
# CHAOS REDUX - CBRN PROTECTIVE-EQUIPMENT PRODUCTION AI
# ============================================================================
# Overview:
# - Starts respirator production only after a real program, shortage, chemical
#   threat/use signal, allied request, or Chaos Warfare posture exists.
# - Stops minimum-factory pressure after the profile-specific military,
#   reserve, replacement, and civilian-distribution targets are satisfied.
# - Maps the accepted 4-10 percent mask-production band to non-overlapping
#   military-factory ranges because ai_strategy values cannot be dynamic.
# - Uses archetype production factors for route/profile differentiation.
#
# NOTE: ai_strategy.value does not parse script constants in this context, so
# all values are centralized as file-local constants.
# ============================================================================

cbrn_mask_production_base = {
	allowed = { always = yes }
	enable = { cbrn_country_should_produce_masks = yes }
	abort_when_not_enabled = yes

	ai_strategy = {
		type = equipment_variant_production_factor
		id = gas_mask_equipment
		value = @CBRN_MASK_PRODUCTION_FACTOR_BASE
	}
}
