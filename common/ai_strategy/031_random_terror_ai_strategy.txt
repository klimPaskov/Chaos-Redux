# ============================================================================
# CHAOS REDUX - EVENT 031 RANDOM TERROR AI STRATEGY
# ============================================================================
# Strategies for live Event 31 actors only (the eight carrier tags while
# random_terror_is_actor holds).  Part 9: an actor secures its own war with
# its parent before distant fronts, and compatible actors support one another.
# The parent carries random_terror_parent_of_<actor tag>, set when the actor
# forms and cleared with the actor's runtime.  Front strategies use
# country_trigger, where the scope is the enemy and FROM is the actor.
# ============================================================================

random_terror_actor_parent_war = {
	allowed = {
		OR = {
			original_tag = JHX
			original_tag = JIX
			original_tag = JKX
			original_tag = JLX
			original_tag = JMX
			original_tag = JNX
			original_tag = JOX
			original_tag = JPX
		}
	}
	enable = {
		random_terror_is_actor = yes
		has_war = yes
	}
	abort_when_not_enabled = yes
	# Concentrate on the parent government's front.
	ai_strategy = {
		type = front_unit_request
		country_trigger = { has_country_flag = random_terror_parent_of_@FROM }
		value = 100
	}
	ai_strategy = {
		type = force_concentration_target_weight
		country_trigger = { has_country_flag = random_terror_parent_of_@FROM }
		value = 40
	}
	# Avoid committing the small army to fronts against distant enemies.
	ai_strategy = {
		type = front_unit_request
		country_trigger = { NOT = { is_neighbor_of = FROM } }
		value = -60
	}
}
