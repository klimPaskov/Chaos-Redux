# ============================================================================
# CHAOS REDUX - EVENT 031 RANDOM TERROR AI STRATEGY
# ============================================================================
# Strategies for live Event 31 actors (the eight carrier tags while
# random_terror_is_actor holds), plus the Event 14 rivalry strategies at the
# end of the file, which also run on cannibal tags facing a neighbouring
# actor.  Part 9: an actor secures its own war with its parent before distant
# fronts, and compatible actors support one another.
# The parent carries random_terror_parent_of_<actor tag>, set when the actor
# forms and cleared with the actor's runtime.  Front strategies use
# country_trigger, where the scope is the enemy and FROM is the actor.
# ============================================================================

random_terror_actor_parent_war = {
	allowed = {
		OR = {
			original_tag = JHX
			original_tag = JIX
			original_tag = JKX
			original_tag = JLX
			original_tag = JMX
			original_tag = JNX
			original_tag = JOX
			original_tag = JPX
		}
	}
	enable = {
		random_terror_is_actor = yes
		has_war = yes
	}
	abort_when_not_enabled = yes
	# Concentrate on the parent government's front.
	ai_strategy = {
		type = front_unit_request
		country_trigger = { has_country_flag = random_terror_parent_of_@FROM }
		value = 100
	}
	ai_strategy = {
		type = force_concentration_target_weight
		country_trigger = { has_country_flag = random_terror_parent_of_@FROM }
		value = 40
	}
	# Avoid committing the small army to fronts against distant enemies.
	ai_strategy = {
		type = front_unit_request
		country_trigger = { NOT = { is_neighbor_of = FROM } }
		value = -60
	}
}

# Cooperation between Event 31 commands.  One strategy per target tag: every
# other cooperating command befriends and supports the target.  A command kept
# independent by Global Jihad setup or locked in a leadership contest neither
# cooperates nor receives cooperation, so other carriers are neutral toward it
# and it stays outside the network until the setup flag or the contest clears.
# The shared trigger random_terror_is_cooperating_actor holds for both sides.

random_terror_actor_cooperation_jhx = {
	allowed = {
		OR = {
			original_tag = JIX
			original_tag = JKX
			original_tag = JLX
			original_tag = JMX
			original_tag = JNX
			original_tag = JOX
			original_tag = JPX
		}
	}
	enable = {
		random_terror_is_cooperating_actor = yes
		JHX = { random_terror_is_cooperating_actor = yes }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = befriend id = JHX value = 50 }
	ai_strategy = { type = support id = JHX value = 50 }
}

# Event 14 rivalry (Part 10).  A live Event 31 command and a neighbouring
# cannibal warband or unified cannibal state (is_cannibalism_country) treat
# each other as enemies: each antagonizes and seeks to conquer the other, and
# no Event 31 faction, merger, sponsor or join path accepts a cannibal
# country.  One strategy per target tag keeps the hostility to neighbours.
# ----------------------------------------------------------------------------

	allowed = {
		OR = {
			original_tag = JHX

			original_tag = JKX
			original_tag = JLX
			original_tag = JMX
			original_tag = JNX
			original_tag = JOX
			original_tag = JPX
		}
	}
	enable = {
		random_terror_is_actor = yes
		CBA = {
			exists = yes
			is_cannibalism_country = yes
			is_neighbor_of = ROOT
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = antagonize id = CBA value = 100 }
	ai_strategy = { type = conquer id = CBA value = 80 }
}
