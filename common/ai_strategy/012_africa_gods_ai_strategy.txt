africa_gods_host_reciprocal_ai = {
	allowed = { has_country_flag = gods_of_africa_active }
	enable = {
		has_country_flag = gods_of_africa_active
		check_variable = { gods_of_africa_doctrine = constant:gods_of_africa_doctrine.reciprocal_covenant }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = pp_spend_priority value = 20 }
	ai_strategy = { type = force_build_armies value = 150 }
}
