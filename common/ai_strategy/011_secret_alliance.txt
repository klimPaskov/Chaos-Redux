secret_alliance_hidden_member_posture = {
	allowed = { always = yes }
	enable = {
		has_country_flag = secret_alliance_active_member
		NOT = { has_country_flag = secret_alliance_reveal_member_snapshot }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_army
		value = constant:secret_alliance_ai.hidden_build_army
	}
	ai_strategy = {
		type = avoid_starting_wars
		value = constant:secret_alliance_ai.hidden_avoid_war
	}
}
