# DHO alternates between anchor defense, measured resource-corridor attacks,
# origin recovery, and terminal multi-front pressure. Ordinary countries at war

resources_found_cave_under_capacity_fronts = {
	allowed = { original_tag = DHO }
	enable = {
		has_country_flag = resources_found_cave_country
		has_war = yes
		resources_found_cave_has_capacity_deficit = yes
		NOT = { has_global_flag = resources_found_cave_world_end }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		state_trigger = {
			is_controlled_by = DHO
			OR = {
				has_state_flag = resources_found_cave_origin
				has_state_flag = resources_found_cave_anchor_active
			}
		}
		ratio = @CAVE_FRONT_RATIO
		priority = @CAVE_UNDER_CAPACITY_PRIORITY
		ordertype = front
		execution_type = careful
		execute_order = no
		manual_attack = no
	}

	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			is_controlled_by = DHO
			OR = {
				has_state_flag = resources_found_cave_origin
				has_state_flag = resources_found_cave_anchor_active
			}
		}
		value = @CAVE_ANCHOR_FRONT_REQUEST
	}
}

# Hierarchy strategies consume the persistent resource objective and anchor
# flags. One Maw concentrates on one weighted route. Many Chambers distributes
# more defenders among active anchors. Hoard the Veins makes feeding states the
# strongest unit-request target while still advancing toward the marked vein.

	allowed = { original_tag = DHO }
	enable = {
		has_country_flag = resources_found_cave_country

	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			has_state_flag = resources_found_cave_marked_objective

		}
		value = @CAVE_ONE_MAW_FRONT_REQUEST
