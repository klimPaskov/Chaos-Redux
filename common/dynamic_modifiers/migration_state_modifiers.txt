# Chaos Redux migration state modifiers.


# Flow modifiers are enabled only by the presentation adapter. The core flags
# still own lifecycle truth; the adapter adds dynamic values derived from exact
# receipts, loads, and state denominators before each flow modifier is added.
migration_state_preparing_to_leave = {
	icon = GFX_migration_state_exodus
	attacker_modifier = yes
	enable = { has_state_flag = migration_presentation_preparing_to_leave }
	remove_trigger = { NOT = { has_state_flag = migration_presentation_preparing_to_leave } }
	state_production_speed_buildings_factor = var:migration_presentation_preparing_production
	army_speed_factor_for_controller = var:migration_presentation_preparing_speed
}
