# The perimeter belongs to the country that secured it and lapses as soon as
# another country controls the crater.
asteroid_incoming_crater_perimeter_security = {
	icon = GFX_idea_asteroid_incoming_fragment_crater
	enable = { asteroid_incoming_crater_perimeter_is_held = yes }
	remove_trigger = { NOT = { asteroid_incoming_crater_perimeter_is_held = yes } }
	army_defence_factor = constant:asteroid_incoming_crater.security_defence
	attrition_for_controller = constant:asteroid_incoming_crater.security_attrition
}

# The state flag and this visible modifier preserve the surveyed material
# profile after its one-use research bonus has been granted.
asteroid_incoming_crater_surveyed_material = {
	icon = GFX_idea_asteroid_incoming_fragment_crater
	enable = { has_state_flag = asteroid_incoming_material_surveyed }
	remove_trigger = { NOT = { has_state_flag = asteroid_incoming_material_surveyed } }
}
