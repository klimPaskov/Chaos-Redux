# ============================================================================
# CHAOS REDUX - EVENT 028 CRATER ACTION MODIFIERS
# ============================================================================
# Local, state-scoped outcomes for crater security, survey records, hostile
# access disruption, and a prepared attack. These modifiers do not alter
# Event 028's permanent controller-based armour benefits. Magnitudes come
# from the asteroid_incoming_crater script constants.
# ============================================================================

# The perimeter belongs to the country that secured it and lapses as soon as
# another country controls the crater. The modifier is added with scope set to
# that country, so its triggers name the crater state through ROOT, as vanilla
# scoped state modifiers do (CHI_bog_defensive_modifier,
# CZE_sudetenland_paramilitaries_modifier).
asteroid_incoming_crater_perimeter_security = {
	icon = GFX_idea_asteroid_incoming_fragment_crater
	enable = { ROOT = { asteroid_incoming_crater_perimeter_is_held = yes } }
	remove_trigger = { ROOT = { NOT = { asteroid_incoming_crater_perimeter_is_held = yes } } }
	army_defence_factor = constant:asteroid_incoming_crater.security_defence
	attrition_for_controller = constant:asteroid_incoming_crater.security_attrition
}

# The state flag and this visible modifier preserve the surveyed material
# profile after its one-use research bonus has been granted.
asteroid_incoming_crater_surveyed_material = {
	icon = GFX_idea_asteroid_incoming_fragment_crater
	enable = { has_state_flag = asteroid_incoming_material_surveyed }
	remove_trigger = { NOT = { has_state_flag = asteroid_incoming_material_surveyed } }
}

# Added with scope set to the hostile controller, so only that country loses
# local supplies in the access state. A negative local_supplies_for_controller
# value is the vanilla supply penalty (ITA_mafia_on_the_rise); a negative
# local_supply_impact_factor would instead ease the controller's supply.
asteroid_incoming_crater_access_supply_disruption = {
	icon = GFX_idea_asteroid_incoming_fragment_crater
	local_supplies_for_controller = constant:asteroid_incoming_crater.access_supply_disruption
}

asteroid_incoming_crater_offensive_plan = {
	icon = GFX_idea_asteroid_incoming_fragment_crater
	attacker_modifier = yes
	army_attack_factor = constant:asteroid_incoming_crater.offensive_attack
}
