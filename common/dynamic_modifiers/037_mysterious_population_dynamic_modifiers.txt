# ============================================================================
# CHAOS REDUX - EVENT 037 MYSTERIOUS PEOPLE DYNAMIC MODIFIERS
# ============================================================================
# mysterious_people_newcomers is the one national modifier of Event 037. Its
# values are country variables written by mysterious_people_refresh_modifier,
# so it never stacks per firing or stage:
# - construction and factory output from settled newcomers (workforce bonus)
# - stability loss above Overcrowded, construction loss above Emergency
# - conscription_factor that keeps unsettled newcomers out of the recruitable
#   population until they settle
# The other modifiers are short-lived consequences of actions and flavour
# choices, each driven by one variable and a native duration.
# ============================================================================

mysterious_people_newcomers = {
	icon = GFX_mysterious_people_modifier_newcomers
	production_speed_buildings_factor = mysterious_people_mod_construction
	industrial_capacity_factory = mysterious_people_mod_output
	stability_factor = mysterious_people_mod_stability
	conscription_factor = mysterious_people_mod_recruit
}

# Civilian factories committed to running Event 037 state projects.
mysterious_people_projects = {
	icon = GFX_mysterious_people_emergency_housing
	civilian_factory_use = mysterious_people_project_factories
}

# Temporary unrest from requisitions, a failed mission, or flavour choices.
mysterious_people_unrest = {
	icon = GFX_mysterious_people_stance_restricted
	stability_factor = mysterious_people_unrest_value
}

# Temporary calm from flavour choices.
mysterious_people_calm = {
	icon = GFX_mysterious_people_stance_open
	stability_factor = mysterious_people_calm_value
}

# Temporary change in factory output from flavour choices.
mysterious_people_shift = {
	icon = GFX_mysterious_people_modifier_newcomers
	industrial_capacity_factory = mysterious_people_shift_value
}
