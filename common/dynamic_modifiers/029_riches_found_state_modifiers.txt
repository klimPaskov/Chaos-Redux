# after containment: gold pressure builds faster in this district and part of
# its labour keeps drifting to private searching. Modifier fields do not read
# script constants, so the value is a file-scoped constant.

	icon = GFX_riches_found_state_gold_sick
	enable = { always = yes }
	local_manpower = @RICHES_FOUND_GOLD_SUSCEPTIBILITY_MANPOWER
}
