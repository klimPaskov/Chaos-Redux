# ============================================================================
# CHAOS REDUX - EVENT 013 NATURAL DISASTERS DYNAMIC MODIFIERS
# ============================================================================
# State modifier shells for Natural Disasters. Event 13 helper effects write the
# state variables and flags that make these modifiers family-specific at runtime.
# ============================================================================

natural_disaster_state_disruption = {
	attacker_modifier = yes
	enable = {
		has_state_flag = natural_disaster_aftermath_active
	}
	remove_trigger = {
		NOT = { has_state_flag = natural_disaster_aftermath_active }
	}

	local_supplies = var:natural_disaster_state_local_supplies
	state_production_speed_buildings_factor = var:natural_disaster_state_production_speed_buildings_factor
	army_speed_factor_for_controller = var:natural_disaster_state_army_speed_factor_for_controller
	attrition_for_controller = var:natural_disaster_state_attrition_for_controller
}

natural_disaster_warning_disruption = {
	enable = {
		has_state_flag = natural_disaster_warning_active
	}
	remove_trigger = {
		NOT = { has_state_flag = natural_disaster_warning_active }
	}

	local_supplies = var:natural_disaster_warning_local_supplies
	state_production_speed_buildings_factor = var:natural_disaster_warning_production_speed_buildings_factor
}

natural_disaster_aftershock_watch = {
	attacker_modifier = yes
	enable = {
		has_state_flag = natural_disaster_aftershock_watch
	}
	remove_trigger = {
		NOT = { has_state_flag = natural_disaster_aftershock_watch }
	}

	local_supplies = -0.15
	state_production_speed_buildings_factor = -0.10
	army_speed_factor_for_controller = -0.12
	attrition_for_controller = 0.03
}
