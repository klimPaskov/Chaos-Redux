# ============================================================================
# JAPAN CHEMICAL CAMPAIGN DECISIONS
# ============================================================================
# State-targeted chemical cylinder operations for Japan during the China war.
# One free selector decision rotates the prepared agent. The attack decision
# supplies an exact selected-state receipt, consumes the matching legacy agent
# cylinders, resolves protection, and dispatches the shared exposure pipeline.
# ============================================================================

japan_chemical_campaign_category = {
	japan_chemical_campaign_cycle_agent = {
		icon = GFX_decision_japan_chemical_campaign_cycle_agent

		available = {
			is_japan_china_chemical_campaign_war = yes
		}

		visible = {
			is_japan_china_chemical_campaign_war = yes
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = constant:chem_japan_campaign_ai.defaults.selector_recovery_add
				NOT = { has_japan_chemical_campaign_selected_stockpile_attack = yes }
				has_japan_chemical_campaign_any_stockpile_attack = yes
			}
		}

		complete_effect = {
			japan_cycle_targeted_chemical_campaign_agent = yes
		}
	}

	japan_chemical_campaign_attack = {
		icon = GFX_decision_japan_chemical_campaign_attack
		state_target = asia
		on_map_mode = map_and_decisions_view

		target_trigger = {
			FROM = { is_japan_chemical_campaign_target_state = yes }
		}

		available = {
			is_japan_china_chemical_campaign_war = yes
			NOT = { has_country_flag = japan_chemical_campaign_attack_cooldown }
			custom_trigger_tooltip = {
				tooltip = japan_chemical_campaign_readiness_policy_tt
				cbrn_country_has_operational_readiness = yes
				cbrn_policy_allows_battlefield_use = yes
			}
			custom_trigger_tooltip = {
				tooltip = japan_chemical_campaign_attack_stockpile_tt
				has_japan_chemical_campaign_selected_stockpile_attack = yes
			}
		}

		custom_cost_trigger = {
			has_japan_chemical_campaign_attack_command_power = yes
			has_japan_chemical_campaign_selected_stockpile_attack = yes
		}
		custom_cost_text = japan_chemical_campaign_attack_cost

		ai_will_do = {
			base = constant:chem_japan_campaign_ai.defaults.attack_base
			modifier = {
				add = constant:chem_japan_campaign_ai.defaults.attack_add_core_state
				FROM = { is_core_of = CHI }
			}
			modifier = {
				factor = constant:chem_japan_campaign_ai.defaults.treaty_member_factor
				has_country_flag = air_cleanliness_treaty_member
			}
		}

		complete_effect = {
			FROM = { save_event_target_as = japan_chemical_campaign_target_state }
			japan_apply_targeted_chemical_campaign_attack = yes
		}
	}
}
