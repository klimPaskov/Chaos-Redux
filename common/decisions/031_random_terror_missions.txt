# ============================================================================
# CHAOS REDUX - EVENT 031 RANDOM TERROR MISSIONS
# ============================================================================
# Missions remain ordinary decision missions so the response category stays
# compact.  Each mission is started by one begin helper in
# 031_random_terror_effects.txt, which reserves mission_slot_1, _2 or _3,
# keys its state receipt to the receiving country (@ROOT here) and stores the
# values the objective is measured against.
#
# Achieve-type missions (Contain the Attack Wave, Break the Cross-Border
# Corridor, Retake the Lost State, Restore Civil Authority, The Hostage
# Deadline, Stop the Coordinated Uprising) complete as soon as measured
# progress reaches the stored goal, and resolve their deadline as partial or
# failure.  Hold-type missions (Protect the Transport Network, Prevent the
# Capital's Seizure) use is_good = yes: their available block is the failure
# condition, and the deadline resolves success or partial according to
# whether a disruption occurred.  Every end path runs the family finish helper,
# which clears only this receiver's receipt.
#
# The cancel triggers deliberately use random_terror_can_hold_response_mission
# rather than the response-decision gate, so the short action lock that every
# response decision applies never cancels the mission it just started.
# ============================================================================

		icon = GFX_decision_mission_random_terror_contain_attack_wave
		days_mission_timeout = constant:random_terror_tuning.containment_days
		selectable_mission = no
		fire_only_once = no
		is_good = no
		activation = {
			has_country_flag = random_terror_mission_active_containment
			random_terror_can_hold_response_mission = yes
			has_country_flag = random_terror_mission_slot_1
			any_state = { has_state_flag = random_terror_containment_mission_subject_@ROOT }
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = random_terror_contain_attack_wave_objective_tt
				has_country_flag = random_terror_mission_active_containment
				check_variable = { var = random_terror_hotspot_count value = random_terror_containment_start_hotspots compare = less_than_or_equals }
				any_state = {
					has_state_flag = random_terror_containment_mission_subject_@ROOT
					OR = {
						NOT = { random_terror_state_is_active_crisis = yes }
						check_variable = { var = random_terror_containment_goal_@ROOT value = random_terror_state_activity compare = greater_than_or_equals }
					}
				}
			}
		}
		cancel_trigger = {
			OR = {
				NOT = { has_country_flag = random_terror_mission_active_containment }
				NOT = { random_terror_can_hold_response_mission = yes }
				has_global_flag = world_end
				NOT = { has_country_flag = random_terror_mission_slot_1 }
				NOT = { any_state = { has_state_flag = random_terror_containment_mission_subject_@ROOT } }
			}
		}
		cancel_effect = {
			if = {
				limit = { has_country_flag = random_terror_mission_active_containment }
				random_terror_cancel_containment_mission = yes
				random_terror_finish_containment_mission = yes
				random_terror_close_country_recovery = yes
			}
		}
		complete_effect = {
			if = {
				limit = { has_country_flag = random_terror_mission_active_containment }
				custom_effect_tooltip = random_terror_contain_attack_wave_complete_tt
				random_terror_resolve_containment_mission_success = yes
				random_terror_finish_containment_mission = yes
				random_terror_close_country_recovery = yes
			}
		}
		timeout_effect = {
			if = {
				limit = { has_country_flag = random_terror_mission_active_containment }
				if = {
					limit = {
						OR = {
							check_variable = { var = random_terror_hotspot_count value = random_terror_containment_start_hotspots compare = greater_than }
							any_state = {
								has_state_flag = random_terror_containment_mission_subject_@ROOT
								check_variable = { var = random_terror_containment_stage_@ROOT value = random_terror_state_stage compare = less_than }

					random_terror_resolve_containment_mission_failure = yes
				}

					random_terror_resolve_containment_mission_partial = yes
				}
				random_terror_finish_containment_mission = yes
				random_terror_close_country_recovery = yes
			}
		}
		ai_will_do = {
			base = @rt_mission_ai_high
			modifier = { factor = @rt_mission_ai_urgent_factor check_variable = { random_terror_pressure > constant:random_terror_value.high } }
