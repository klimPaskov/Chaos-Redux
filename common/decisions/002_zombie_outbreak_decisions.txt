			}
		}
	}
	# Target a live profile's capital; keep only the active project's marker while its timer runs.
	develop_weaponized_zombie_counterstrain = {
		icon = GFX_decision_generic_research

			OR = {
				has_idea = searching_for_cure

			}
		}
		state_target = yes
		target_trigger = {
			FROM = {
				OR = {
					has_state_flag = weaponized_counterstrain_target_by_@ROOT
					AND = {
						ROOT = { NOT = { has_idea = searching_for_cure } }
						is_capital = yes
						owner = { is_live_weaponized_zombie_profile_country = yes }
						NOT = {
							owner = {
								has_country_flag = weaponized_zombie_type_wendigo
								has_country_flag = cannibalism_wendigo_transformation_locked
							}
						}
						OR = {
							NOT = { owner = { is_single_weaponized_zombie_profile_type = yes } }
							ROOT = { weaponized_zombie_counterstrain_progress_open_for_state_target = yes }
						}
					}
				}
			}
		}
		on_map_mode = map_only
		available = {
			is_zombie_counterstrain_system_active = yes
			NOT = { has_idea = searching_for_cure }
			custom_trigger_tooltip = {
				tooltip = zombie_counterstrain_profile_type_required_tt
				trigger = { FROM = { owner = { is_single_weaponized_zombie_profile_type = yes } } }
			}
			custom_trigger_tooltip = {
				tooltip = zombie_counterstrain_research_available_tt
				trigger = {
					OR = {
						FROM = { owner = { NOT = { is_single_weaponized_zombie_profile_type = yes } } }
						weaponized_zombie_counterstrain_progress_open_for_state_target = yes
					}
				}
			}
		}
		cost = global.black_friday_price_triggerable_scenario_cure_counterstrain_cost
		days_remove = constant:triggerable_scenario_cure.counterstrain_days
		fire_only_once = no
		ai_will_do = { factor = 1 }
		complete_effect = {
			hidden_effect = {
				clr_country_flag = develop_weaponized_zombie_counterstrain_sale_credit
				clear_variable = develop_weaponized_zombie_counterstrain_sale_sequence
				set_temp_variable = { black_friday_political_power_receipt_credited = constant:universal_cost_framework.zero }
				set_temp_variable = { black_friday_political_power_receipt_sale_sequence = constant:universal_cost_framework.zero }
				black_friday_credit_political_power_purchase_with_receipt = yes
				if = {
					limit = { check_variable = { black_friday_political_power_receipt_credited = constant:universal_cost_framework.one } }
					set_country_flag = develop_weaponized_zombie_counterstrain_sale_credit
					set_variable = { develop_weaponized_zombie_counterstrain_sale_sequence = black_friday_political_power_receipt_sale_sequence }
				}
				set_variable = { weaponized_zombie_counterstrain_paid_pp = global.black_friday_price_triggerable_scenario_cure_counterstrain_cost }
				set_weaponized_zombie_counterstrain_type_from_selected_profile = yes
				FROM = {
					set_state_flag = weaponized_counterstrain_target_by_@ROOT
					owner = { set_country_flag = weaponized_counterstrain_owner_by_@ROOT }
				}
			}
			add_timed_idea = { idea = searching_for_cure days = constant:triggerable_scenario_cure.counterstrain_days }
		}
		cancel_trigger = {
			OR = {
				NOT = { is_zombie_counterstrain_system_active = yes }
				NOT = { weaponized_zombie_counterstrain_target_matches_selected_type = yes }
				NOT = { weaponized_zombie_counterstrain_progress_open_for_state_target = yes }
			}
		}
		cancel_effect = { refund_weaponized_zombie_counterstrain_research = yes }
		remove_effect = {
			if = {
				limit = {
					AND = {
						weaponized_zombie_counterstrain_target_matches_selected_type = yes
						weaponized_zombie_counterstrain_progress_open_for_state_target = yes
					}
				}
				apply_weaponized_zombie_counterstrain_research = yes
				clear_variable = weaponized_zombie_counterstrain_type
				clear_variable = weaponized_zombie_counterstrain_paid_pp
				FROM = {
					clr_state_flag = weaponized_counterstrain_target_by_@ROOT
					owner = { clr_country_flag = weaponized_counterstrain_owner_by_@ROOT }
				}
				clr_country_flag = develop_weaponized_zombie_counterstrain_sale_credit
				clear_variable = develop_weaponized_zombie_counterstrain_sale_sequence
			}
			else = { refund_weaponized_zombie_counterstrain_research = yes }
		}
	}
