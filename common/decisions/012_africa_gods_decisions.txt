# ============================================================================
# CHAOS REDUX - EVENT 012 GODS OF AFRICA DECISIONS AND MISSIONS
# ============================================================================
# Event 012 owns the persistent Gods of Africa relationship layer. Participant
# responses are phased by the active-demand, negotiation, defiance, and
# reconciliation flags. The host actions use the bounded participant target
# array maintained by the owner registry; no human selector is required for
# AI evaluation.
#
# Owner helper contracts used below:
# - gods_of_africa_current_demand_can_be_paid (participant cost gate)
# - gods_of_africa_current_demand_cost (icon-first dynamic cost text)
# - gods_of_africa_reconciliation_can_be_paid (reconciliation cost gate)
# - gods_of_africa_reconciliation_cost (icon-first dynamic cost text)
# - gods_of_africa_participants (host-owned bounded participant registry)
#
# The parent-owned scripted effects implement all state mutation, debit/credit,
# generation checks, cooldowns, and cleanup. These decisions only expose the
# validated entry points and keep their presentation phases mutually exclusive.
# ============================================================================

gods_of_africa_participant_category = {
	gods_of_africa_demand_mission = {
		name = gods_of_africa_demand_mission
		desc = gods_of_africa_demand_mission_desc
		icon = GFX_decision_012_africa_gods_fulfill_demand
		activation = { always = no }
		days_mission_timeout = constant:gods_of_africa_demand.maximum_mission_timeout
		is_good = no
		visible = {
			gods_of_africa_participant_host_is_active = yes
			gods_of_africa_participant_generation_is_current = yes
			has_country_flag = gods_of_africa_demand_active
			NOT = { has_country_flag = gods_of_africa_defiant }
		}
		# Response decisions resolve the mission; it is intentionally not an
		# auto-completing objective. The owner timeout path handles failure.
		available = { hidden_trigger = { always = no } }
		cancel_trigger = {
			OR = {
				NOT = { gods_of_africa_participant_generation_is_current = yes }
				NOT = { has_country_flag = gods_of_africa_demand_active }
				has_country_flag = gods_of_africa_defiant
				has_country_flag = gods_of_africa_final_settlement_committed
			}
		}
		cancel_effect = {
			hidden_effect = { remove_mission = gods_of_africa_demand_mission }
		}
		complete_effect = { gods_of_africa_comply_demand = yes }
		timeout_effect = { gods_of_africa_refuse_demand = yes }
		ai_will_do = { base = 1 }
	}
