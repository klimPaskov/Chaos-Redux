# ============================================================================
# Evolution I (the Arab Revolt Network): breaking the cells (target), funding
# the network and backing a government replacement (Britain).
# Evolution II (the British Arabian System): six institutions in a chain

# (the regional category, visible to nodes and to independent candidates).

	lawrence_break_the_revolt_cells = {
		icon = GFX_decision_generic_wreckers
		priority = 101
		visible = { lawrence_shows_break_cells = yes }
		available = { hidden_trigger = { lawrence_can_pay_break_cells = yes } }
		custom_cost_trigger = { lawrence_can_pay_break_cells = yes }
		custom_cost_text = lawrence_cost_break
		complete_effect = {
			custom_effect_tooltip = lawrence_break_cells_tt
			hidden_effect = { lawrence_start_break_cells_mission = yes }
		}
		ai_will_do = {
			factor = 0
			modifier = {
				set_temp_variable = { lawrence_ai_weight = mtth:lawrence_ai_break_cells }
				add = lawrence_ai_weight
			}
		}
	}

	lawrence_break_cells_mission = {
		icon = GFX_decision_generic_wreckers
		priority = 116
		allowed = { always = yes }
		activation = { always = no }
		available = {
			custom_trigger_tooltip = {
				tooltip = lawrence_break_cells_hold_tt
				hidden_trigger = {
					check_variable = { var = lawrence_cells value = constant:lawrence_cells.break_target compare = less_than_or_equals }
				}
			}
		}
		days_mission_timeout = constant:lawrence_cells.break_days
		cancel_trigger = { NOT = { lawrence_is_live_target = yes } }
		complete_effect = {
			custom_effect_tooltip = lawrence_break_cells_success_tt
			hidden_effect = { lawrence_break_cells_success = yes }
		}
		timeout_effect = {
			if = {
				limit = { check_variable = { var = lawrence_cells value = constant:lawrence_cells.partial_floor compare = less_than_or_equals } }
				custom_effect_tooltip = lawrence_break_cells_partial_tt
				hidden_effect = { lawrence_break_cells_partial = yes }
			}
			else = {
				custom_effect_tooltip = lawrence_break_cells_failure_tt
				hidden_effect = { lawrence_break_cells_failure = yes }
			}
		}
	}
}
