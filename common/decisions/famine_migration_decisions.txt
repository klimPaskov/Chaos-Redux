# Chaos Redux shared famine and migration decisions.

# This file owns only the decision surface. Food, movement, reception, return,
# and conservation state remain owned by the reusable contracts in
# chaosx_famine_migration_effects.txt. Every movement block below submits one
# exact civilian-transfer request and then consumes its actual debit, route
# deaths, and survivor credit.

						has_state_flag = famine_migration_displacement_active

							check_variable = { var = famine_migration_flight_pressure value = constant:famine_migration_decision_threshold.category_flight_people compare = greater_than_or_equals }
							check_variable = { var = famine_migration_trapped_population value = constant:famine_migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }

			modifier = { factor = constant:famine_migration_decision_ai.prepare_factor_1 FROM = { has_state_flag = famine_migration_displacement_active } }
