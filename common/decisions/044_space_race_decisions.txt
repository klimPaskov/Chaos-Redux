# Stealing from a leading rival: the success chance and the research bonus
# are file-scoped because `random` and `steal_random_tech_bonus` read literals.
# Every space-race technology sits in the electronics folder.
@space_race_steal_chance = 50
@space_race_steal_bonus = 1.0
@space_race_steal_ahead_reduction = 1.0
@space_race_steal_base_bonus = 0.25

chaosx_space_race_decision_category = {
	space_race_launch_decision = {

		# Many things are done in on actions on daily

		available = { # Checks whether it should be greyed out
			has_tech = advanced_rocket_engines
			# has_tech = advanced_centimetric_radar
			has_tech = advanced_computing_machine
			if = { limit = { has_dlc = "Man the Guns" has_dlc = "La Resistance" }
				has_tech = advanced_fire_control_system
			}
		}
		icon = GFX_decision_space_race
		cost = global.black_friday_price_pp_50

		ai_will_do = {
			factor = 100 # If ai is able then it must
		}

		complete_effect = {
			hidden_effect = { black_friday_credit_political_power_purchase = yes }
			clr_global_flag = space_race
			hidden_effect = { clear_array = global.space_race_contenders }
			country_event = { id = chaosx.nr44.3 days = 2 }
			hidden_effect = { remove_ideas = leading_space_race }
		}
	}
	space_race_tech_bonus_decision = {
		icon = GFX_decision_generic_electricity
		visible = {
		}

		days_re_enable = 10
		cost = global.black_friday_price_pp_100

		ai_will_do = {
			factor = 2
			modifier = {
				add = 2
				has_war = no
			}
			modifier = {
				add = 2
				check_variable = { space_race_points = global.most_space_race_points }
			}
			modifier = {
				add = 5
				amount_research_slots > 4
			}
		}
		complete_effect = {
			hidden_effect = { black_friday_credit_political_power_purchase = yes }
			add_tech_bonus = {
				bonus = 0.5
				ahead_reduction = 0.5
				uses = 2
				category = electronics
				category = rocketry
			}
		}
	}
	space_race_milestone_decision = {
		icon = GFX_decision_generic_electricity
		visible = { space_race_milestone_available = yes }

		days_re_enable = 5
		cost = global.black_friday_price_pp_20

		ai_will_do = {
			# 2 base, plus 2 because the ladder never funds a technology that already has a bonus.
			factor = 4
			modifier = {
				add = 2
				has_war = no
			}
			modifier = {
				add = 2
				check_variable = { space_race_points = global.most_space_race_points }
			}
			modifier = {
				add = 5
				amount_research_slots > 4
			}
		}
		complete_effect = {
			custom_effect_tooltip = space_race_milestone_decision_effect_tt
			hidden_effect = {
				black_friday_credit_political_power_purchase = yes
				space_race_fund_next_milestone = yes
			}
		}
	}

	space_race_steal_tech_targeted_decision = {
		icon = GFX_decision_generic_electricity
		available = {
		}

		days_re_enable = 20
		cost = global.black_friday_price_pp_100
		fixed_random_seed = no

		target_array = global.space_race_contenders
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = ROOT }
				# Only a rival ahead of us has research worth taking.
				check_variable = { var = space_race_points value = ROOT.space_race_points compare = greater_than }
			}
		}

		ai_will_do = {
			factor = 0
			modifier = {
				add = 1
				has_war = no
			}
			modifier = {
				add = 1
				NOT = { check_variable = { space_race_points = global.most_space_race_points } }
			}
			modifier = {
				add = 2
				check_variable = { FROM.space_race_points > space_race_points }
			}
			modifier = {
				add = 1
				FROM = { is_ai = no }
			}
		}

		complete_effect = {
			hidden_effect = { black_friday_credit_political_power_purchase = yes }
			custom_effect_tooltip = space_race_steal_tech_targeted_decision_effect_tt
			random = {
				chance = @space_race_steal_chance
				steal_random_tech_bonus = {
					folder = electronics_folder
					target = FROM
					bonus = @space_race_steal_bonus
					ahead_reduction = @space_race_steal_ahead_reduction
					base_bonus = @space_race_steal_base_bonus
					name = space_race_stolen_research
					uses = 1
				}
			}
		}
	}
}
