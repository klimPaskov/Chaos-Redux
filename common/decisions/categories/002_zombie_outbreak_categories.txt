chaosx_anti_zombie_league_category = {
	visible = {
		has_global_flag = has_zombie_outbreak
		NOT = { has_country_flag = azl_refused_to_join }
		NOT = { original_tag = ZZZ }
		NOT = { has_country_flag = zombie_outbreak_dynamic_country }
		NOT = { has_global_flag = azl_dissolved }
		OR = {
			# Show for formation candidates
			AND = {
				NOT = { has_global_flag = anti_zombie_league_formed }
				can_lead_anti_zombie_league = yes
				date > 1936.7.1
			}
			# Show for joining candidates
			AND = {
				has_global_flag = anti_zombie_league_formed
				has_country_flag = azl_can_join_later
				should_actively_join_anti_zombie_league = yes
			}
			# Show for current members
			hidden_trigger = { is_anti_zombie_league_member = yes }
		}
	}
	icon = GFX_decision_category_anti_zombie_league
	priority = 100
}
