random_terror_government_response_category = {
	visible = {
		NOT = { has_country_flag = random_terror_actor }
		NOT = { has_global_flag = world_end }
		OR = {
			random_terror_can_government_response = yes
			has_country_flag = random_terror_reconstruction_available
		}
	}
	visible_when_empty = yes
	picture = GFX_decision_cat_picture_031_random_terror_government_response
	icon = GFX_decision_category_031_random_terror_government_response
	priority = 83
	highlight_states = {
		highlight_states_trigger = {
			OR = {
				is_owned_by = ROOT
				is_claimed_by = ROOT
				is_core_of = ROOT
			}
			OR = {
				has_state_flag = random_terror_state_dormant
				has_state_flag = random_terror_state_active
				has_state_flag = random_terror_state_entrenched
				has_state_flag = random_terror_state_armed_insurgency
				has_state_flag = random_terror_state_lost_local_control
				has_state_flag = random_terror_state_recovery
			}
		}
		highlight_color_before_active = 2
		highlight_color_while_active = 3
	}
}
