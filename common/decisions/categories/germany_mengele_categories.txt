germany_final_solution_category = {
	visible = {
		original_tag = GER
		has_government = fascism
		NOT = { has_country_flag = germany_mengele_faction }
		OR = {
			is_ai = yes
			has_country_flag = genocide_decisions_visible
		}
	}
	icon = GFX_decision_category_chaos_doom
	priority = 36
}

germany_tibet_expedition_category = {
	visible = {
		original_tag = GER
		has_government = fascism
		NOT = { has_country_flag = germany_mengele_faction }
		OR = {
			check_variable = { aryan_origin_preparation > 0 }
			has_country_flag = germany_tibet_expedition_active
			has_country_flag = germany_tibet_expedition_completed
		}
	}
	icon = GFX_decision_category_chaos_doom
	priority = 37
}
