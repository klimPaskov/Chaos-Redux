	scripted_gui = camp_repression_ledger_category_scripted_gui

genocide_crisis_category = {
	visible = {
		OR = {
			AND = {
				NOT = { original_tag = SOV }
				OR = {
					genocide_country_has_german_holocaust_target = yes
					genocide_country_has_extermination_upgrade_target = yes
					genocide_country_has_restricted_chemical_site_target = yes
					check_variable = { genocide_concentration_sites > 0 }
					check_variable = { genocide_extermination_sites > 0 }
					check_variable = { genocide_escalation > 0 }
					check_variable = { genocide_decisions_taken > 0 }
				}
			}
			AND = {
				original_tag = SOV
				genocide_country_has_restricted_chemical_site_target = yes
			}
		}
	}
	icon = GFX_decision_category_chaos_doom
	highlight_states = {
		highlight_states_trigger = {
			is_controlled_by = ROOT
			concentration_camp > 0
		}
		highlight_color_before_active = 2
		highlight_color_while_active = 2
	}
	priority = -10000
}

	visible_when_empty = yes
	visible = {
		original_tag = SOV
		OR = {
			AND = {
				has_government = communism
				any_controlled_state = {
					NOT = { gulag_labor_camp_network > 0 }
					OR = {
						is_soviet_gulag_periphery_pool_state = yes
						NOT = { is_core_of = ROOT }
						has_resistance = yes
					}
				}
			}
			has_active_camp_network = yes
			check_variable = { camp_gulag_site_count > 0 }
			check_variable = { famine_pressure > 0 }
			has_country_flag = genocide_gulag_decisions_visible
			has_country_flag = sov_famine_crisis_active
			has_country_flag = sov_gulag_dismantlement_active
			has_country_flag = sov_post_collapse_records_exposed
			sov_camp_is_reform_route = yes
			any_controlled_state = {
				OR = {
					has_state_flag = genocide_soviet_mass_repression_site
					has_state_flag = genocide_soviet_famine_pressure_site
					has_state_flag = camp_rework_soviet_records_concealed
				}
			}
		}
	}
	icon = GFX_decision_category_chaos_doom
	priority = 1
}
