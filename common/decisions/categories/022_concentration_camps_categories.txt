# A simple category: a description built by scripted localisation that states
# the reach of the network, Exposure and Resistance Pressure, the ordinary
# decision list and the mission rows. Routine management of the network lives
# in the shared Camp Administration window.

concentration_camps_category = {
	visible = {
		concentration_camps_category_is_visible = yes
	}
	visible_when_empty = yes
	icon = GFX_decision_category_repression_ledger
	highlight_states = {
		highlight_states_trigger = {
			is_controlled_by = ROOT
			has_state_flag = concentration_camps_site
			is_valid_camp_active_site_state = yes
		}
		highlight_color_before_active = 2
		highlight_color_while_active = 2
	}
	priority = 91
}
