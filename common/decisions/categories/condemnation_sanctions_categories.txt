# ============================================================================
# CHAOS REDUX - CONDEMNATION SANCTIONS DECISION CATEGORIES
# ============================================================================
# Separate target responses from foreign participant actions so condemned
# countries and outside powers see only the decisions relevant to their role.
# ============================================================================

condemnation_target_response_category = {
	visible_when_empty = yes
	visible = {
		OR = {
			condemnation_has_public_record = yes
			condemnation_has_active_sanctions = yes
		}
	}
	icon = GFX_decision_category_generic_crisis
	priority = 4
}

condemnation_participant_action_category = {
	visible_when_empty = yes
	visible = {
		any_country = {
			exists = yes
			condemnation_is_formally_censured = yes
		}
	}
	icon = GFX_decision_category_generic_economy
	priority = 3
}
