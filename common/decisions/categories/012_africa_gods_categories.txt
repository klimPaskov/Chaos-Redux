# ============================================================================
# CHAOS REDUX - EVENT 012 GODS OF AFRICA DECISION CATEGORIES
# ============================================================================
# Both categories attach one read-only public-values inlay above their ordinary
# decisions. The participant and host keep their existing category identities
# and decision actions; the inlay adds no action or metric.
# ============================================================================

gods_of_africa_participant_category = {
	icon = GFX_decision_category_012_africa_charter_ledger
	scripted_gui = gods_of_africa_public_values
	priority = 123
	visible = {
		OR = {
			gods_of_africa_recovery_member_request_is_current = yes
			gods_of_africa_victor_renewal_can_withdraw = yes
			gods_of_africa_victor_relationship_is_available = yes
			AND = {
				gods_of_africa_participant_is_registered = yes
				event_target:africa_host = {
					has_country_flag = gods_of_africa_active
					NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
				}
			}
		}
	}
}

gods_of_africa_host_category = {
	icon = GFX_decision_category_012_africa_charter_ledger
	scripted_gui = gods_of_africa_public_values
	priority = 122
	visible = {
		OR = {
			gods_of_africa_active_host_is_valid = yes
			# The settling government keeps the category while it still owes
			# settlement support.
			gods_of_africa_settlement_source_has_pending_delivery = yes
			gods_of_africa_recovery_can_resume = yes
			gods_of_africa_original_host_recovery_is_open = yes
			AND = {
				gods_of_africa_recovery_committed_identity_is_current = yes
				has_country_flag = gods_of_africa_consolidation_pending
			}
		}
	}
}
