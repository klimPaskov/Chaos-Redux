# CHAOS REDUX - EVENT 006 SCENARIO LEDGER DECISIONS
# ============================================================================
# Zero-reward navigation controls for the frozen SCN-008 rejection ledger.
# They change no gameplay value and are never selected by AI.
# ============================================================================

independence_wave_scenario_ledger_category = {
	independence_wave_scenario_ledger_previous = {
		icon = GFX_decision_independence_wave_border_arbitration
		visible = { has_country_flag = independence_wave_scenario_ledger_visible }
		available = {
			check_variable = {
				var = global.independence_wave_scenario_blocked_package_ids^num
				value = constant:independence_wave_scenario_registry.one
				compare = greater_than
			}
		}
		cost = 0
		days_re_enable = 0
		fire_only_once = no
		complete_effect = {
			subtract_from_variable = { independence_wave_scenario_ledger_index = constant:independence_wave_scenario_registry.one }
			if = {
				limit = {
					check_variable = {
						var = independence_wave_scenario_ledger_index
						value = constant:independence_wave_scenario_registry.first_array_index
						compare = less_than
					}
				}
				set_variable = { independence_wave_scenario_ledger_index = global.independence_wave_scenario_blocked_package_ids^num }
				subtract_from_variable = { independence_wave_scenario_ledger_index = constant:independence_wave_scenario_registry.one }
			}
			set_variable = { independence_wave_scenario_ledger_display_index = independence_wave_scenario_ledger_index }
			add_to_variable = { independence_wave_scenario_ledger_display_index = constant:independence_wave_scenario_registry.one }
		}
		ai_will_do = { base = 0 }
	}

	independence_wave_scenario_ledger_next = {
		icon = GFX_decision_independence_wave_border_arbitration
		visible = { has_country_flag = independence_wave_scenario_ledger_visible }
		available = {
			check_variable = {
				var = global.independence_wave_scenario_blocked_package_ids^num
				value = constant:independence_wave_scenario_registry.one
				compare = greater_than
			}
		}
		cost = 0
		days_re_enable = 0
		fire_only_once = no
		complete_effect = {
			add_to_variable = { independence_wave_scenario_ledger_index = constant:independence_wave_scenario_registry.one }
			if = {
				limit = {
					check_variable = {
						var = independence_wave_scenario_ledger_index
						value = global.independence_wave_scenario_blocked_package_ids^num
						compare = greater_than_or_equals
					}
				}
				set_variable = { independence_wave_scenario_ledger_index = constant:independence_wave_scenario_registry.first_array_index }
			}
			set_variable = { independence_wave_scenario_ledger_display_index = independence_wave_scenario_ledger_index }
			add_to_variable = { independence_wave_scenario_ledger_display_index = constant:independence_wave_scenario_registry.one }
		}
		ai_will_do = { base = 0 }
	}

	independence_wave_scenario_ledger_close = {
		icon = GFX_decision_independence_wave_border_arbitration
		visible = { has_country_flag = independence_wave_scenario_ledger_visible }
		available = { always = yes }
		cost = 0
		days_re_enable = 0
		fire_only_once = no
		complete_effect = { clr_country_flag = independence_wave_scenario_ledger_visible }
		ai_will_do = { base = 0 }
	}
}
