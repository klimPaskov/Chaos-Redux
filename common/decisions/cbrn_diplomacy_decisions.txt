# ============================================================================
# CHAOS REDUX - CBRN INTERNATIONAL RESPONSE DECISIONS
# ============================================================================
# Overview:
# - Demand inspections targets one exact condemned country and opens a timed
#   answer mission for that country.
# - Forensic publication targets one controlled state with one exact unshared
#   chemical action or detected ordinary-pathogen episode.
# - International decontamination targets one consenting foreign-controlled
#   contaminated state and consumes a real relief package.
# ============================================================================

		icon = GFX_decision_cbrn_demand_inspections
		target_array = global.enabled_countries_list
		target_root_trigger = {
			NOT = { is_actual_nonhuman_country = yes }
			black_friday_can_pay_cbrn_demand_inspections = yes
		}
		target_trigger = {
			FROM = { cbrn_target_can_receive_inspection_demand_from_root = yes }
		}
		available = {

		}
		custom_cost_text = cbrn_demand_inspections_cost
		complete_effect = {
			custom_effect_tooltip = cbrn_demand_inspections_tt
			hidden_effect = {
				FROM = { save_event_target_as = cbrn_diplomacy_target_country }

			}
		}
		ai_will_do = {
			base = constant:cbrn_diplomacy_ai.base

			}
			modifier = {
				factor = constant:cbrn_diplomacy_ai.high
				has_government = democratic
			}
			modifier = {
				factor = constant:cbrn_diplomacy_ai.low

			}
		}
	}

			hidden_trigger = { black_friday_can_pay_cbrn_sponsor_decontamination_ordinary = yes }

		complete_effect = {
			custom_effect_tooltip = cbrn_sponsor_decontamination_mission_begin_tt

				set_temp_variable = { black_friday_cbrn_decon_requested_sale_variant = constant:universal_cost_framework.one }

					limit = { check_variable = { var = black_friday_cbrn_decon_payment_success value = constant:universal_cost_framework.one compare = equals } }

				}
			}
		}
		remove_effect = {
			custom_effect_tooltip = cbrn_sponsor_decontamination_mission_complete_tt

				FROM = { cbrn_diplomacy_complete_international_decon_mission = yes }
			}
		}
		cancel_trigger = {
			FROM = {
				OR = {
					NOT = { has_state_flag = cbrn_international_decon_mission_active }
					NOT = { has_variable = cbrn_international_decon_recipient }

			}
			modifier = {
				factor = constant:cbrn_diplomacy_ai.low

			}
