		state_target = any_controlled_state

		ai_will_do = { factor = 0 }

	# A successful API call exposes its exact scheduled state immediately. The
	# actionable family warning remains conditional and delayed, but the player
	# never has to infer from a hidden root event that a disaster was accepted.
	natural_disaster_forecast_card = {
		state_target = any_controlled_state
		target_root_trigger = { natural_disaster_country_has_warning = yes }
		target_trigger = {
			FROM = {
				has_state_flag = natural_disaster_impact_scheduled
				NOT = { has_state_flag = natural_disaster_warning_active }
				NOT = { has_state_flag = natural_disaster_recovery_unresolved_territory }
			}

		priority = @ND_PRIORITY_WARNING
		available = {
			FROM = { NOT = { has_state_flag = natural_disaster_forecast_acknowledged } }

			custom_effect_tooltip = natural_disaster_forecast_card_acknowledge_tt
			FROM = {
				set_state_flag = natural_disaster_forecast_acknowledged

		}

	}
