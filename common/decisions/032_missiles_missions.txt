# Event 032 Missiles timed missions.
#
# Missions are activated by the ordinary decisions in 032_missiles_decisions.txt. The Event 032
# core owns the state, site, operation, warning, and restoration flags consumed here. Every
# preparation cancellation/timeout settles its launch receipt. Other native projects clear only
# their own pending work; they cannot abort a newer or unrelated operation.
#
# Project missions (survey, secondary site, repair, recovery, verification, restoration) are
# goal-style: each completes on the first day its stored due date has passed and its objective
# still holds. Every due date in missiles_evolution_timing is shorter than the native timeout
# below, so the work always finishes inside the mission window.

@m32_mission_survey_days = 120
@m32_mission_secondary_days = 180
@m32_mission_repair_days = 120
# Aligned with missiles_rogue_tuning.crisis_window_days: the rogue crisis resolves itself then.
@m32_mission_recovery_days = 30
# days_mission_timeout uses a file-local literal; activation applies the dynamic
# duration. Keep this anchor aligned with missiles_tuning.operation_mission_anchor_days.
@m32_mission_operation_anchor_days = 1
@m32_mission_warning_days = 14
@m32_mission_network_days = 150
@m32_mission_ai_low = 1
@m32_mission_ai_medium = 4
