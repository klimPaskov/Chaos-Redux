# Event 032 Missiles timed missions.
#
# Missions are activated by the ordinary decisions in 032_missiles_decisions.txt. The Event 032
# core owns the state, site, operation, warning, and restoration flags consumed here. Every
# cancellation and timeout aborts and cleans the active operation receipt so no reserved missile,
# payload, site, or target remains stranded.

		is_good = no
		fire_only_once = no
		activation = { always = no }
		available = { has_country_flag = missiles_survey_complete }
		cancel_trigger = {
			OR = {
				NOT = { missiles_country_can_manage_program = yes }
				NOT = { has_country_flag = missiles_program_phase_establishment }
			}
		}
		complete_effect = {
			custom_effect_tooltip = missiles_survey_mission_success_tt

		}
		cancel_effect = {
			custom_effect_tooltip = missiles_survey_mission_cancelled_tt

				missiles_cleanup_operation = yes
			}
		}
