# ============================================================================
# CHAOS REDUX - FALLOUT FOOD SECURITY DECISIONS
# ============================================================================
# These two dormant missions are opened by The Last Inventory callback. They
# spend survival-ledger resources rather than political power and close after
# one authored review. They never activate the Fallout scheduler.
# ============================================================================

fallout_food_security_category = {
	fallout_event_100_ration_law = {
		icon = GFX_idea_fallout_state_grade
		selectable_mission = yes
		visible = {
			has_country_flag = fallout_event_100_ration_law_chain_open
			NOT = { has_country_flag = fallout_event_100_ration_law_active }
			NOT = { has_country_flag = fallout_event_100_ration_law_complete }
		}
		activation = {
			fallout_survival_country_resource_row_is_durable = yes
			fallout_event_orientation_is_complete = yes
		}
		available = {
			fallout_survival_country_resource_row_is_durable = yes
			check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_decision.ration_law_food_cost compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_100_decision.ration_law_filter_cost compare = greater_than_or_equals }
		}
		custom_cost_trigger = {
			fallout_survival_country_resource_row_is_durable = yes
			check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_decision.ration_law_food_cost compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_100_decision.ration_law_filter_cost compare = greater_than_or_equals }
		}
		custom_cost_text = fallout.event_log.last_inventory.ration_law_cost
		days_mission_timeout = constant:fallout_event_100_decision.ration_law_days
		complete_effect = {
			set_country_flag = fallout_event_100_ration_law_active
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_100_decision.ration_law_food_cost }
			subtract_from_variable = { fallout_survival_current_entries^7 = constant:fallout_event_100_decision.ration_law_filter_cost }
		}
		cancel_trigger = {
			OR = {
				NOT = { has_country_flag = fallout_event_100_ration_law_chain_open }
				NOT = { fallout_survival_country_resource_row_is_durable = yes }
			}
		}
		cancel_effect = {
			clr_country_flag = fallout_event_100_ration_law_active
			set_country_flag = fallout_event_100_ration_law_failed
			set_temp_variable = { fallout_event_100_modifier_days = constant:fallout_event_100_timing.grievance_modifier_days }
			add_dynamic_modifier = { modifier = fallout_food_grievance_aftershock days = fallout_event_100_modifier_days }
			clear_temp_variable = fallout_event_100_modifier_days
		}
		timeout_effect = {
			clr_country_flag = fallout_event_100_ration_law_active
			set_country_flag = fallout_event_100_ration_law_complete
			add_to_variable = { fallout_survival_current_entries^1 = constant:fallout_event_100_decision.ration_law_food_gain }
			add_to_variable = { fallout_survival_current_entries^9 = constant:fallout_event_100_decision.ration_law_recognition_gain }
			set_temp_variable = { fallout_event_100_cohesion_delta = constant:fallout_event_100_decision.ration_law_cohesion_gain }
			fallout_survival_apply_cohesion_delta = yes
			set_temp_variable = { fallout_event_100_modifier_days = constant:fallout_event_100_timing.ration_modifier_days }
			add_dynamic_modifier = { modifier = fallout_food_public_ration_law days = fallout_event_100_modifier_days }
			fallout_orientation_clamp_resources = yes
			clear_temp_variable = fallout_event_100_cohesion_delta
			clear_temp_variable = fallout_event_100_modifier_days
		}
		fire_only_once = yes
		ai_will_do = { base = 4 modifier = { factor = 2 check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_partial compare = less_than } } }
	}

	fallout_event_100_first_hunger_mission = {
		icon = GFX_idea_fallout_state_grade
		selectable_mission = yes
		visible = {
			has_country_flag = fallout_event_100_first_hunger_mission_open
			NOT = { has_country_flag = fallout_event_100_first_hunger_mission_active }
			NOT = { has_country_flag = fallout_event_100_first_hunger_mission_complete }
		}
		activation = {
			fallout_survival_country_resource_row_is_durable = yes
			fallout_event_orientation_is_complete = yes
		}
		available = {
			fallout_survival_country_resource_row_is_durable = yes
			check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_decision.hunger_mission_food_cost compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_100_decision.hunger_mission_medicine_cost compare = greater_than_or_equals }
		}
		custom_cost_trigger = {
			fallout_survival_country_resource_row_is_durable = yes
			check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_decision.hunger_mission_food_cost compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_100_decision.hunger_mission_medicine_cost compare = greater_than_or_equals }
		}
		custom_cost_text = fallout.event_log.last_inventory.hunger_mission_cost
		days_mission_timeout = constant:fallout_event_100_decision.hunger_mission_days
		complete_effect = {
			set_country_flag = fallout_event_100_first_hunger_mission_active
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_100_decision.hunger_mission_food_cost }
			subtract_from_variable = { fallout_survival_current_entries^3 = constant:fallout_event_100_decision.hunger_mission_medicine_cost }
		}
		cancel_trigger = {
			OR = {
				NOT = { has_country_flag = fallout_event_100_first_hunger_mission_open }
				NOT = { fallout_survival_country_resource_row_is_durable = yes }
			}
		}
		cancel_effect = {
			clr_country_flag = fallout_event_100_first_hunger_mission_active
			set_country_flag = fallout_event_100_first_hunger_mission_failed
			set_temp_variable = { fallout_event_100_modifier_days = constant:fallout_event_100_timing.grievance_modifier_days }
			add_dynamic_modifier = { modifier = fallout_food_grievance_aftershock days = fallout_event_100_modifier_days }
			clear_temp_variable = fallout_event_100_modifier_days
		}
		timeout_effect = {
			clr_country_flag = fallout_event_100_first_hunger_mission_active
			set_country_flag = fallout_event_100_first_hunger_mission_complete
			add_to_variable = { fallout_survival_current_entries^1 = constant:fallout_event_100_decision.hunger_mission_food_gain }
			add_to_variable = { fallout_survival_current_entries^9 = constant:fallout_event_100_decision.hunger_mission_recognition_gain }
			set_temp_variable = { fallout_event_100_cohesion_delta = constant:fallout_event_100_decision.hunger_mission_cohesion_gain }
			fallout_survival_apply_cohesion_delta = yes
			fallout_orientation_clamp_resources = yes
			clear_temp_variable = fallout_event_100_cohesion_delta
		}
		fire_only_once = yes
		ai_will_do = { base = 5 modifier = { factor = 2 check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_100_decision.hunger_mission_medicine_cost compare = greater_than_or_equals } } }
	}
}
