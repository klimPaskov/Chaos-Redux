# ============================================================================
# CHAOS REDUX - EVENT 012 HOST TRANSFER EFFECTS
# ============================================================================
# Overview:
# - Stages the committed host ledger before an ordinary annexation removes the
#   country scope.
# - Cancels active action records and removes their shared missions exactly once.
# - Offers the recorded Charter pool to an explicit congress/crisis resolver.
# - Commits at most one successor, increments the host generation, relinks every
#   surviving member, reloads the focus tree, and retargets the League globally.
# - A missing successor leaves a visible suspended-host crisis instead of making
#   a generic country the new owner.
# ============================================================================

# COUNTRY scope on the old host. Active actions are a host-owned transaction and
# cannot be allowed to outlive the host generation. The action kernel owns the
# actual refund/mission cleanup; this helper only snapshots the bounded target
# list so the array can be safely mutated while it is being consumed.
africa_host_transfer_cancel_active_actions = {
	set_variable = { global.africa_host_transfer_cancelled_action_count = africa_active_action_targets^num }
	clear_array = africa_host_transfer_action_cleanup
	for_each_scope_loop = {
		array = africa_active_action_targets
		event_target:africa_previous_host = {
			add_to_array = { array = africa_host_transfer_action_cleanup value = PREV }
		}
	}
	for_each_scope_loop = {
		array = africa_host_transfer_action_cleanup
		if = {
			limit = {
				exists = yes
				has_country_flag = africa_action_record_active
			}
			africa_cleanup_action = yes
		}
	}
	clear_array = africa_host_transfer_action_cleanup

	# Peace-exemption missions are not action records, so close their exact target
	# mission and state flags before the old host pointer is removed.
	for_each_scope_loop = {
		array = africa_peace_exemption_countries
		if = {
			limit = { exists = yes }
			event_target:africa_previous_host = {
				remove_targeted_decision = { target = PREV decision = mission_africa_complete_continental_peace_exemption }
			}
			for_each_scope_loop = {
				array = africa_continental_peace_exemption_states
				clr_state_flag = africa_continental_peace_exemption_active
			}
			clear_array = africa_continental_peace_exemption_states
			clr_country_flag = africa_continental_peace_completion_mission_active
			clr_country_flag = africa_continental_peace_terms_accepted
		}
	}
	# These exemption registries are transaction-local. The mission and state
	# flags above are already closed; keeping the host indexes would resurrect a
	# phantom peace docket on the successor.
	clear_array = africa_peace_exemption_countries
	clear_array = africa_peace_exemption_states
}

# COUNTRY scope on the old host. The on-annex bridge calls this while the host
# still exists. A previously recorded constitutional designation can therefore
# complete atomically and preserve completed focuses; otherwise the old owner is
# cleared and the resolver receives a visible staged crisis.
africa_begin_host_transfer = {
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { has_global_flag = africa_host_successor_consumed }
			NOT = { has_global_flag = africa_host_transfer_pending }
		}
		ROOT = { save_global_event_target_as = africa_host_transfer_resolver }
		save_global_event_target_as = africa_previous_host
		set_global_flag = africa_host_transfer_pending
		set_global_flag = africa_host_transfer_resolution_open
		africa_host_transfer_cancel_active_actions = yes
		for_each_scope_loop = {
			array = africa_relationship_countries
			if = {
				limit = { exists = yes africa_is_charter_subject_of_current_host = yes }
				africa_release_member_from_continental_subject_status = yes
			}
		}
		africa_host_transfer_stage_scalar_ledger = yes
		africa_host_transfer_stage_first_proof_state = yes
		africa_host_transfer_stage_arrays = yes
		if = {
			limit = { africa_host_transfer_designated_successor_is_valid = yes }
			event_target:africa_host_transfer_designated_successor = { africa_host_transfer_complete_successor = yes }
		}
		else = {
			clr_country_flag = africa_unifier_host
			clr_country_flag = africa_host_commit_completed
			if = {
				limit = { has_event_target = africa_host }
				clear_global_event_target = africa_host
			}
			event_target:africa_host_transfer_resolver = {
				set_country_flag = africa_host_transfer_resolver
				country_event = { id = chaosx.nr12.230 }
			}
		}
	}
}

# COUNTRY scope on the successor host while ROOT is the second annexer. The
# one-use continuity mandate is terminal after this loss: active ledgers are
# closed, the global host pointer is cleared, and the project records an
# explicit dissolution notice instead of silently retaining a dead scope.
africa_host_transfer_record_terminal_loss = {
	if = {
		limit = {
			africa_is_current_host = yes
			has_global_flag = africa_host_successor_consumed
		}
		save_event_target_as = africa_previous_host
		africa_host_transfer_cancel_active_actions = yes
		for_each_scope_loop = {
			array = africa_relationship_countries
			if = {
				limit = { exists = yes africa_is_charter_subject_of_current_host = yes }
				africa_release_member_from_continental_subject_status = yes
			}
		}
		clr_country_flag = africa_unifier_host
		clr_country_flag = africa_host_commit_completed
		clr_country_flag = africa_host_successor_received_mandate
		clr_global_flag = africa_host_transfer_pending
		clr_global_flag = africa_host_transfer_resolution_open
		clr_global_flag = africa_host_transfer_suspended
		if = {
			limit = { has_event_target = africa_host }
			clear_global_event_target = africa_host
		}
		clr_global_flag = africa_event_active
		set_global_flag = africa_event_resolved
		set_global_flag = africa_host_transfer_terminal_after_successor_loss
		set_global_flag = africa_achievement_league_destroyed
		set_temp_variable = { africa_event_log_payload = constant:africa_event_log.host_succession_terminal }
		set_temp_variable = { africa_event_log_actor = THIS.id }
		set_temp_variable = { africa_event_log_has_actor = constant:africa_value.one }
		set_temp_variable = { africa_event_log_secondary_actor = ROOT.id }
		set_temp_variable = { africa_event_log_has_secondary_actor = constant:africa_value.one }
		africa_record_event_log_world_order_entry = yes
		ROOT = {
			set_country_flag = africa_host_transfer_terminal_notice
			country_event = { id = chaosx.nr12.233 }
		}
	}
}

# COUNTRY scope on the resolver. The single-candidate event option uses this
