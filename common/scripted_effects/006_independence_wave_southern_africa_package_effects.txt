# ---------------------------------------------------------------------------
# IW-126 Barotseland (BAR)
# ---------------------------------------------------------------------------

independence_wave_bar_apply_package_layer = {
	add_ideas = independence_wave_bar_divided_kuta
	set_country_flag = independence_wave_framework_package_layer_applied
}

independence_wave_bar_complete_national_kuta = {
	independence_wave_reset_decision_deltas = yes
	remove_ideas = independence_wave_bar_divided_kuta
	add_ideas = independence_wave_bar_national_kuta
	set_temp_variable = { independence_wave_decision_legitimacy_delta = constant:independence_wave_decision_effect.value_standard }
	set_temp_variable = { independence_wave_decision_instability_delta = constant:independence_wave_decision_effect.value_minor_loss }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_bar_kuta_convened
}

independence_wave_bar_complete_canals = {
	independence_wave_reset_decision_deltas = yes
	add_ideas = independence_wave_bar_floodplain_canals
	set_temp_variable = { independence_wave_decision_capacity_delta = constant:independence_wave_decision_effect.value_standard }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_bar_canals_restored
}

independence_wave_bar_complete_mine_contracts = {
	independence_wave_reset_decision_deltas = yes
	set_temp_variable = { independence_wave_decision_recognition_delta = constant:independence_wave_decision_effect.value_standard }
	set_temp_variable = { independence_wave_decision_legitimacy_delta = constant:independence_wave_decision_effect.value_minor }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_bar_contracts_regulated
}

independence_wave_bar_cleanup_package_layer = {
	if = {
		limit = { original_tag = BAR check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_126 } }
		remove_decision = independence_wave_bar_convene_national_kuta
		remove_decision = independence_wave_bar_restore_floodplain_canals
		remove_decision = independence_wave_bar_regulate_mine_contracts
		remove_ideas = { independence_wave_bar_divided_kuta independence_wave_bar_national_kuta independence_wave_bar_floodplain_canals }
		clr_country_flag = independence_wave_bar_kuta_convened
		clr_country_flag = independence_wave_bar_canals_restored
		clr_country_flag = independence_wave_bar_contracts_regulated
	}
}
