fallout_event_705_record_history = {
	set_temp_variable = { fallout_event_705_history_should_record = constant:fallout_event_runtime.zero }
	if = { limit = { AND = { check_variable = { fallout_event_705_history_stage = constant:fallout_event_runtime.zero } NOT = { has_country_flag = fallout_event_705_choice_history_recorded } } } set_temp_variable = { fallout_event_705_history_should_record = constant:fallout_event_runtime.one } }

	if = {
		limit = { check_variable = { fallout_event_705_history_should_record = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_705_history_payload = constant:fallout_event_705_log.cancelled }
		if = {
			limit = { check_variable = { fallout_event_705_history_stage = constant:fallout_event_runtime.zero } }
			if = { limit = { check_variable = { fallout_event_705_selected_branch value = constant:fallout_event_705_branch.accept_technocratic_rule compare = equals } } set_temp_variable = { fallout_event_705_history_payload = constant:fallout_event_705_log.choice_accept_technocratic_rule } }

			else = { set_temp_variable = { fallout_event_705_history_payload = constant:fallout_event_705_log.choice_public_vote } }
		}
		else_if = {
			limit = { check_variable = { fallout_event_705_history_stage = constant:fallout_event_runtime.one } }
			if = {
				limit = { check_variable = { fallout_event_705_selected_branch value = constant:fallout_event_705_branch.accept_technocratic_rule compare = equals } }

				else = { set_temp_variable = { fallout_event_705_history_payload = constant:fallout_event_705_log.accept_technocratic_rule_failure } }
			}
