# Evolution III: the entry and the evolved opening, the three congress calls,
# ratification, core selection, the validated formation transaction, the
# failure paths, the federation identity, Federal Authority, its decisions and
# the monthly tick.
#
# Carrier decision: the federation transforms its strongest member. No new
# country tag exists, so there is no tag collision. Common/countries holds
# three cosmetic tags (LAWF_BRITISH, LAWF_ARAB, LAWF_KINGDOM) and the focus
# tree lawrence_federation_focus_tree is loaded onto the core. Founding AI
# members are annexed with their troops, so no unit or stockpile is
# duplicated. A human member that is not the core stays a country as an
# autonomous dominion of the core, so nobody is stranded.
#
# The formation validates everything before the first mutation, so a failed
# validation changes nothing and needs no rollback: members keep their
# states, units and leaders and the reason is stored in
# global.lawrence_fed_failure.
# Scope: COUNTRY unless noted.
# ============================================================================

# COUNTRY, the caller. Input: temporary lawrence_fed_kind. Builds the invitation
# list from real arrays and sends the invitations. The answers arrive over the
# next weeks and chaosx.nr40.121 judges them.
lawrence_open_congress = {
	set_global_flag = lawrence_congress_open
	set_variable = { global.lawrence_fed_outcome = lawrence_fed_kind }
	set_variable = { global.lawrence_fed_caller = THIS }
	clear_array = global.lawrence_fed_invited
	clear_array = global.lawrence_fed_members
	if = {
		limit = { check_variable = { var = lawrence_fed_kind value = constant:lawrence_fed.outcome_british compare = equals } }
		ENG = { lawrence_collect_nodes = yes }
		for_each_scope_loop = {
			array = global.lawrence_nodes
			if = {
				limit = {
					lawrence_fed_participant_is_valid = yes
					NOT = { is_in_array = { array = global.lawrence_fed_invited value = THIS } }
				}
				add_to_array = { global.lawrence_fed_invited = THIS }
			}
		}
	}
	else = {
		for_each_scope_loop = {
			array = global.lawrence_bloc_members
			if = {
				limit = {
					lawrence_fed_participant_is_valid = yes
					NOT = { is_in_array = { array = global.lawrence_fed_invited value = THIS } }
				}
				add_to_array = { global.lawrence_fed_invited = THIS }
			}
		}
		if = {
			limit = { check_variable = { var = lawrence_fed_kind value = constant:lawrence_fed.outcome_kingdom compare = equals } }
			for_each_scope_loop = {
				array = global.lawrence_independent_partners
				if = {
					limit = {
						lawrence_fed_participant_is_valid = yes
						NOT = { is_in_array = { array = global.lawrence_fed_invited value = THIS } }
					}
					add_to_array = { global.lawrence_fed_invited = THIS }
				}
			}
		}
		# The caller joins its own congress.
		if = {
			limit = { NOT = { is_in_array = { array = global.lawrence_fed_members value = THIS } } }
			add_to_array = { global.lawrence_fed_members = THIS }
			set_country_flag = lawrence_fed_ratified
		}
	}
	for_each_scope_loop = {
		array = global.lawrence_fed_invited
		if = {
			limit = { NOT = { has_country_flag = lawrence_fed_ratified } }
			country_event = { id = chaosx.nr40.120 }
		}
	}
	set_temp_variable = { lawrence_congress_delay = constant:lawrence_fed.congress_days }
	country_event = { id = chaosx.nr40.121 days = lawrence_congress_delay }
	ENG = { country_event = { id = chaosx.nr40.132 days = constant:lawrence_event.report_delay_days } }
}

# COUNTRY, a participant. Sets variable lawrence_fed_score: industry, army,
# stability, and the standing the outcome asks for.
lawrence_fed_score_core = {
	set_variable = { lawrence_fed_score = num_of_factories }
	multiply_variable = { lawrence_fed_score = constant:lawrence_fed.score_factory }
	set_temp_variable = { lawrence_fed_part = num_divisions }
	multiply_temp_variable = { lawrence_fed_part = constant:lawrence_fed.score_division }
	add_to_variable = { lawrence_fed_score = lawrence_fed_part }
	set_temp_variable = { lawrence_fed_part = stability }
	multiply_temp_variable = { lawrence_fed_part = constant:lawrence_fed.score_stability }
	add_to_variable = { lawrence_fed_score = lawrence_fed_part }
	# A player is never absorbed into an AI core without consent.
	if = {
		limit = { is_ai = no }
		add_to_variable = { lawrence_fed_score = constant:lawrence_fed.score_human }
	}
	if = {
		limit = {
			check_variable = { var = global.lawrence_fed_outcome value = constant:lawrence_fed.outcome_british compare = equals }
			is_subject_of = ENG
		}
		add_to_variable = { lawrence_fed_score = constant:lawrence_fed.score_british_backing }
	}
	if = {
		limit = {
			NOT = { check_variable = { var = global.lawrence_fed_outcome value = constant:lawrence_fed.outcome_british compare = equals } }
			is_subject = no
		}
		add_to_variable = { lawrence_fed_score = constant:lawrence_fed.score_sovereign_standing }
	}
}

# COUNTRY. Picks the ratifier with the highest score as the core.
lawrence_fed_select_core = {
	clear_variable = global.lawrence_fed_core
	if = {
		limit = { check_variable = { var = global.lawrence_fed_outcome value = constant:lawrence_fed.outcome_kingdom compare = equals } }
		# The personal congress is held in the country that hosts Lawrence.
		set_variable = { global.lawrence_fed_core = THIS }
	}
	else = {
		set_temp_variable = { lawrence_fed_best = constant:lawrence_event.zero }
		for_each_scope_loop = {
			array = global.lawrence_fed_members
			if = {
				limit = { lawrence_fed_participant_is_valid = yes }
				lawrence_fed_score_core = yes
				if = {
					limit = { check_variable = { var = lawrence_fed_score value = lawrence_fed_best compare = greater_than } }
					set_temp_variable = { lawrence_fed_best = lawrence_fed_score }
					set_variable = { global.lawrence_fed_core = THIS }
				}
			}
		}
	}
}

# COUNTRY. Validates before any mutation. Output: global.lawrence_fed_failure
# (0 when valid). The kingdom needs Lawrence free in the core, the sovereign
# route needs an independent core and the British route needs a Britain that
# can still sustain the federation.
lawrence_fed_validate = {
	if = {
		limit = { NOT = { has_variable = global.lawrence_fed_core } }
		set_variable = { global.lawrence_fed_failure = constant:lawrence_fed.failure_no_core }
	}
	else_if = {
		limit = { check_variable = { var = global.lawrence_fed_members^num value = constant:lawrence_fed.minimum_members compare = less_than } }
		set_variable = { global.lawrence_fed_failure = constant:lawrence_fed.failure_too_few }
	}
	else_if = {
		limit = {
			check_variable = { var = global.lawrence_fed_outcome value = constant:lawrence_fed.outcome_british compare = equals }
			ENG = { NOT = { lawrence_sponsor_can_act = yes } }
		}
		set_variable = { global.lawrence_fed_failure = constant:lawrence_fed.failure_britain }
	}
	else_if = {
		limit = {
			check_variable = { var = global.lawrence_fed_outcome value = constant:lawrence_fed.outcome_sovereign compare = equals }
			var:global.lawrence_fed_core = { is_subject = yes }
		}
		set_variable = { global.lawrence_fed_failure = constant:lawrence_fed.failure_subject_core }
	}
	else_if = {
		limit = {
			check_variable = { var = global.lawrence_fed_outcome value = constant:lawrence_fed.outcome_kingdom compare = equals }
			OR = {
				NOT = { lawrence_status_is_defected = yes }
				var:global.lawrence_fed_core = { NOT = { is_in_array = { array = global.lawrence_independent_partners value = THIS } } }
			}
		}
		set_variable = { global.lawrence_fed_failure = constant:lawrence_fed.failure_lawrence }
	}
	else = {
		# No participant may be at war with the core.
		for_each_scope_loop = {
			array = global.lawrence_fed_members
			if = {
				limit = { has_war_with = var:global.lawrence_fed_core }
				set_variable = { global.lawrence_fed_failure = constant:lawrence_fed.failure_war }
			}
		}
	}
}

# COUNTRY, the caller. One transaction after a successful validation.
lawrence_fed_form = {
	set_global_flag = lawrence_federation_formed
	clr_global_flag = lawrence_congress_open
	lawrence_fed_clear_congress_flags = yes
	var:global.lawrence_fed_core = {
		set_country_flag = lawrence_fed_is_core
		set_variable = { lawrence_fed_outcome = global.lawrence_fed_outcome }
		set_variable = { lawrence_fed_member_count = global.lawrence_fed_members^num }
		every_owned_state = { set_state_flag = lawrence_fed_founding_state }
		# Ratifying governments join. An AI member is annexed with its troops
		# and its states become core territory of the federation. A player
		# member stays a country as an autonomous dominion of the core.
		for_each_scope_loop = {
			array = global.lawrence_fed_members
			if = {
				limit = { NOT = { has_country_flag = lawrence_fed_is_core } }
				set_country_flag = lawrence_fed_member
				if = {
					limit = { is_ai = no }
					var:global.lawrence_fed_core = {
						set_autonomy = {
							target = PREV
							autonomy_state = autonomy_dominion
							end_wars = no
						}
					}
				}
				else = {
					every_owned_state = { set_state_flag = lawrence_fed_founding_state }
					var:global.lawrence_fed_core = {
						annex_country = {
							target = PREV
							transfer_troops = yes
						}
					}
				}
			}
		}
		every_owned_state = {
			limit = { has_state_flag = lawrence_fed_founding_state }
			add_core_of = PREV
		}
		clr_country_flag = lawrence_fed_is_core
		lawrence_fed_apply_identity = yes
	}
	lawrence_fed_award_achievements = yes
	lawrence_fed_close_campaign = yes
	lawrence_fed_present_super_event = yes
	lawrence_fed_notify_formation = yes
	clear_array = global.lawrence_fed_invited
}

# COUNTRY, the core. Authority from 0 to 100 and its band.
lawrence_fed_update_authority_band = {
	clamp_variable = { var = lawrence_federal_authority min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
	set_variable = { lawrence_fed_band = constant:lawrence_fed.band_fragile }
	if = {
		limit = { lawrence_fed_band_is_weak = yes }
		set_variable = { lawrence_fed_band = constant:lawrence_fed.band_weak }
	}
	else_if = {
		limit = { lawrence_fed_band_is_working = yes }
		set_variable = { lawrence_fed_band = constant:lawrence_fed.band_working }
	}
	else_if = {
		limit = { lawrence_fed_band_is_strong = yes }
		set_variable = { lawrence_fed_band = constant:lawrence_fed.band_strong }
	}
	remove_ideas = lawrence_authority_fragile
	remove_ideas = lawrence_authority_weak
	remove_ideas = lawrence_authority_working
	remove_ideas = lawrence_authority_strong
	if = {
		limit = { lawrence_fed_band_is_fragile = yes }
		add_ideas = lawrence_authority_fragile
	}
	else_if = {
		limit = { lawrence_fed_band_is_weak = yes }
		add_ideas = lawrence_authority_weak
	}
	else_if = {
		limit = { lawrence_fed_band_is_working = yes }
		add_ideas = lawrence_authority_working
	}
	else = {
		add_ideas = lawrence_authority_strong
	}

	if = {
		limit = { has_country_flag = lawrence_federation_core }
		set_temp_variable = { lawrence_authority_change = lawrence_fed_baseline }
		if = {
			limit = { has_idea = lawrence_uneven_administration }
			add_to_temp_variable = { lawrence_authority_change = constant:lawrence_fed.drift_administration }
		}
		if = {
			limit = { has_idea = lawrence_administrative_secession }
			add_to_temp_variable = { lawrence_authority_change = constant:lawrence_fed.drift_secession }
		}
		if = {
			limit = { has_idea = lawrence_rival_commands }
			add_to_temp_variable = { lawrence_authority_change = constant:lawrence_fed.drift_commands }
		}
		if = {
			limit = {
				OR = {
					has_idea = lawrence_promise_debt
					has_idea = lawrence_federal_mistrust
					has_idea = lawrence_personal_settlement
				}
			}
			add_to_temp_variable = { lawrence_authority_change = constant:lawrence_fed.drift_political }
		}
		set_temp_variable = { lawrence_fed_stability = stability }
		if = {
			limit = { check_variable = { var = lawrence_fed_stability value = constant:lawrence_fed.stable_threshold compare = greater_than } }
			add_to_temp_variable = { lawrence_authority_change = constant:lawrence_fed.drift_stable }
		}
		if = {
			limit = { check_variable = { var = lawrence_fed_stability value = constant:lawrence_fed.unstable_threshold compare = less_than } }
			add_to_temp_variable = { lawrence_authority_change = constant:lawrence_fed.drift_unstable }
		}
		lawrence_fed_add_authority = yes
		lawrence_fed_refresh_costs = yes
		lawrence_fed_track_secession = yes
		lawrence_fed_update_trend = yes
		set_temp_variable = { lawrence_tick_delay = constant:lawrence_fed.tick_days }
		country_event = { id = chaosx.nr40.140 days = lawrence_tick_delay }
	}
}

# COUNTRY, the core. Months in the fragile band worsen the administration
# into Administrative Secession and months of recovery restore it.

		limit = { lawrence_fed_band_is_fragile = yes }
		add_to_variable = { lawrence_fed_low_months = constant:lawrence_event.one }
		set_variable = { lawrence_fed_recovery_months = constant:lawrence_event.zero }
	}
	else = {
		set_variable = { lawrence_fed_low_months = constant:lawrence_event.zero }
	}
	if = {
		limit = {
			has_idea = lawrence_uneven_administration
			check_variable = { var = lawrence_fed_low_months value = constant:lawrence_fed.secession_months compare = greater_than_or_equals }
		}
		swap_ideas = { remove_idea = lawrence_uneven_administration add_idea = lawrence_administrative_secession }
		set_variable = { lawrence_fed_low_months = constant:lawrence_event.zero }
		set_temp_variable = { lawrence_report_delay = constant:lawrence_event.report_delay_days }
		country_event = { id = chaosx.nr40.141 days = lawrence_report_delay }
	}
	else_if = {
		limit = { has_idea = lawrence_administrative_secession }
		if = {
			limit = {
				OR = {
					lawrence_fed_band_is_working = yes
					lawrence_fed_band_is_strong = yes
				}
			}
			add_to_variable = { lawrence_fed_recovery_months = constant:lawrence_event.one }
			if = {
				limit = { check_variable = { var = lawrence_fed_recovery_months value = constant:lawrence_fed.secession_recovery_months compare = greater_than_or_equals } }
				swap_ideas = { remove_idea = lawrence_administrative_secession add_idea = lawrence_uneven_administration }
				set_variable = { lawrence_fed_recovery_months = constant:lawrence_event.zero }
			}
		}
	}
}

# COUNTRY, the core. One step toward replacing Uneven Administration. After
# the required number of steps the spirit becomes the final form of the route.

		limit = {

			limit = { check_variable = { var = lawrence_fed_admin_steps value = constant:lawrence_focus.lifecycle_steps compare = greater_than_or_equals } }
			remove_ideas = lawrence_uneven_administration
			remove_ideas = lawrence_administrative_secession
			if = {
				limit = { lawrence_fed_route_is_british = yes }
				add_ideas = lawrence_imperial_liaison_administration
			}
			else_if = {
				limit = { has_country_flag = lawrence_fed_central_state }
				add_ideas = lawrence_central_administration
			}
			else = {
				add_ideas = lawrence_charter_administration
			}
			set_country_flag = lawrence_fed_administration_settled
		}
	}
}
