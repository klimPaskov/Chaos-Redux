independence_wave_refresh_axx_lifecycle = {
	if = {
		limit = { is_independence_wave_axx_package = yes }
		remove_ideas = { axx_divided_banat_authority axx_banat_compact }
		if = {
			limit = { has_stable_independence_wave_axx_ledgers = yes }
