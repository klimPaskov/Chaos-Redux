humanitarian_initialize_runtime = {
	if = {
		limit = { NOT = { has_global_flag = humanitarian_runtime_initialized } }
		famine_initialize_runtime = yes
		migration_initialize_runtime = yes
		civilian_transfer_initialize_runtime = yes
		set_global_flag = humanitarian_runtime_initialized
	}
}

humanitarian_process_registered_runtime = {
	if = {
		limit = { has_country_flag = is_global_host }
		humanitarian_initialize_runtime = yes
		famine_bootstrap_historical_profile_anchors = yes
		for_each_scope_loop = {
			array = global.famine_historical_profile_anchor_states
			famine_process_registered_historical_profile_anchor = yes
		}
		for_each_scope_loop = {
			array = global.famine_historical_profile_candidate_states
			famine_process_registered_historical_profile_candidate = yes
		}
		for_each_scope_loop = {
			array = global.famine_active_states
			famine_process_registered_state = yes
		}
		for_each_scope_loop = {
			array = global.migration_active_states
			migration_process_registered_state = yes
		}
		for_each_scope_loop = {
			array = global.migration_reception_candidate_states
			migration_clear_stale_normal_destination_proofs = yes
			migration_refresh_native_state_safety_projections = yes
			migration_consume_registered_state_reassessment = yes
			if = { limit = { humanitarian_state_is_valid = yes } migration_reconcile_state_cohort_selection = yes }
		}
		for_each_scope_loop = {
			array = global.migration_active_countries
			migration_process_registered_country = yes
		}
		for_each_scope_loop = {
			array = global.famine_relief_registered_donor_states
			famine_relief_process_registered_donor_state = yes
		}
	}
}
