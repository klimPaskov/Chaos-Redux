fallout_nzl_issue_sea_road_patrol_window = {
	if = {
		limit = { fallout_nzl_sea_road_licensing_is_current = yes }

		set_country_flag = {
			flag = fallout_nzl_sea_road_patrol_window_current
			days = fallout_nzl_patrol_window_days
		}
		set_variable = { fallout_nzl_sea_road_patrol_window_generation = global.fallout_transition_generation }
		if = {
			limit = { NOT = { has_variable = fallout_nzl_sea_road_license_serial } }
			set_variable = { fallout_nzl_sea_road_license_serial = constant:fallout_nzl_schema.zero }
		}
		add_to_variable = { fallout_nzl_sea_road_license_serial = constant:fallout_nzl_schema.one }

			limit = { NOT = { has_dynamic_modifier = fallout_nzl_quiet_seas_patrol_window } }

		}
		force_update_dynamic_modifier = yes

}
