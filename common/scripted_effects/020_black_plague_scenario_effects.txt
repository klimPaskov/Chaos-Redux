# ============================================================================
# CHAOS REDUX - EVENT 020 SCN-012 BLACK PLAGUE UNBOUND
# ============================================================================
# The triggerable scenario is a single, idempotent bootstrap transaction. It
# seeds the ordinary Event 020 state machine, activates Evolutions I-IV in
# order, creates the reusable Rat Nation, and establishes a
# separate Royal Basin for RTX. The scoped bootstrap flag is the only bypass;
# it is cleared before ordinary pulses resume. Evolution V and world_end are
# deliberately outside this entry point.
# ============================================================================

black_plague_scenario_register_commit_provider = {
	set_global_flag = black_plague_scenario_commit_provider_ready
}

black_plague_scenario_suppress_fire_once = {
	set_global_flag = black_plague_triggerable_scenario_suppressed_fire_once
	set_temp_variable = { black_plague_scenario_event_id = constant:black_plague_identity.event_id }
	if = {
		limit = { NOT = { is_in_array = { global.disabled_events = black_plague_scenario_event_id } } }
		add_to_array = { global.disabled_events = black_plague_scenario_event_id }
	}
	if = {
		limit = { NOT = { is_in_array = { global.fired_events = black_plague_scenario_event_id } } }
		add_to_array = { global.fired_events = black_plague_scenario_event_id }
	}
	if = {
		limit = { is_in_array = { global.fire_once_events = black_plague_scenario_event_id } }
		remove_from_array = { global.fire_once_events = black_plague_scenario_event_id }
	}
	set_temp_variable = { new_weight = 0 }
	set_event_weight = yes
	update_unfired_events_counts = yes
	calculate_dynamic_major_weight_gain = yes
}

		add_to_variable = { global.black_plague_scenario_division_bonus = constant:black_plague_value.one }
		set_variable = { global.black_plague_scenario_internal_brood_target = constant:black_plague_scenario.high_internal_brood_target }

black_plague_scenario_collect_candidates = {
	clear_array = global.black_plague_scenario_candidate_states
	clear_array = global.black_plague_scenario_anchor_states
	clear_array = global.black_plague_scenario_rat_candidate_states
	set_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.zero }
	set_variable = { global.black_plague_scenario_existing_established_count = constant:black_plague_value.zero }
	set_variable = { global.black_plague_scenario_unestablished_human_capacity = constant:black_plague_value.zero }
	every_state = {
		limit = { black_plague_scenario_valid_human_state = yes }
		add_to_array = { array = global.black_plague_scenario_candidate_states value = THIS }
		if = {
			limit = { NOT = { black_plague_state_is_established = yes } }
			add_to_variable = { global.black_plague_scenario_unestablished_human_capacity = constant:black_plague_value.one }
		}
	}
	every_state = {
		limit = { black_plague_state_is_established = yes }
		add_to_variable = { global.black_plague_scenario_existing_established_count = constant:black_plague_value.one }
	}
	set_variable = { global.black_plague_scenario_established_capacity = global.black_plague_scenario_existing_established_count }
	add_to_variable = { global.black_plague_scenario_established_capacity = global.black_plague_scenario_unestablished_human_capacity }
	every_state = {
		limit = {
			black_plague_scenario_valid_human_state = yes
			NOT = { black_plague_state_is_rat_controlled = yes }
			black_plague_state_is_infected_or_worse = yes
		}
		add_to_array = { array = global.black_plague_scenario_rat_candidate_states value = THIS }
	}
	set_variable = { global.black_plague_scenario_seeded_states = global.black_plague_scenario_existing_established_count }
	if = { limit = { black_plague_scenario_has_europe = yes } add_to_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.one } }
	if = { limit = { black_plague_scenario_has_asia = yes } add_to_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.one } }
	if = { limit = { black_plague_scenario_has_africa = yes } add_to_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.one } }
	if = { limit = { black_plague_scenario_has_north_america = yes } add_to_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.one } }
	if = { limit = { black_plague_scenario_has_south_america = yes } add_to_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.one } }
	if = { limit = { black_plague_scenario_has_australia = yes } add_to_variable = { global.black_plague_scenario_eligible_continents = constant:black_plague_value.one } }
}

# The six continent helpers deliberately remain explicit so the launch can
# guarantee geographic distribution without relying on dynamic token effects.
black_plague_scenario_select_europe_anchor = {
	if = {
		limit = { check_variable = { global.black_plague_scenario_selected_continents < global.black_plague_scenario_continent_target } }
		clear_temp_array = black_plague_scenario_continent_pool
		every_state = { limit = { black_plague_scenario_valid_human_state = yes NOT = { black_plague_state_is_established = yes } is_on_continent = europe NOT = { has_state_flag = black_plague_scenario_reserved } } add_to_temp_array = { black_plague_scenario_continent_pool = THIS } }
		if = { limit = { check_variable = { black_plague_scenario_continent_pool^num > constant:black_plague_value.zero } } random_scope_in_array = { array = black_plague_scenario_continent_pool save_event_target_as = black_plague_scenario_anchor_state set_state_flag = black_plague_scenario_reserved add_to_array = { array = global.black_plague_scenario_anchor_states value = THIS } add_to_variable = { global.black_plague_scenario_selected_continents = constant:black_plague_value.one } } }
		else = {
			clear_temp_array = black_plague_scenario_continent_pool
			every_state = { limit = { black_plague_scenario_valid_human_state = yes is_on_continent = europe NOT = { has_state_flag = black_plague_scenario_reserved } } add_to_temp_array = { black_plague_scenario_continent_pool = THIS } }
			if = { limit = { check_variable = { black_plague_scenario_continent_pool^num > constant:black_plague_value.zero } } random_scope_in_array = { array = black_plague_scenario_continent_pool save_event_target_as = black_plague_scenario_anchor_state set_state_flag = black_plague_scenario_reserved add_to_array = { array = global.black_plague_scenario_anchor_states value = THIS } add_to_variable = { global.black_plague_scenario_selected_continents = constant:black_plague_value.one } } }

	}

black_plague_scenario_select_anchor_states = {
	set_variable = { global.black_plague_scenario_continent_target = constant:black_plague_scenario.low_continent_target }
	if = { limit = { check_variable = { global.black_plague_scenario_intensity = constant:triggerable_scenario_intensity.medium } } set_variable = { global.black_plague_scenario_continent_target = constant:black_plague_scenario.medium_continent_target } }
	else_if = { limit = { check_variable = { global.black_plague_scenario_intensity = constant:triggerable_scenario_intensity.high } } set_variable = { global.black_plague_scenario_continent_target = constant:black_plague_scenario.high_continent_target } }
	else_if = { limit = { check_variable = { global.black_plague_scenario_intensity = constant:triggerable_scenario_intensity.maximum } } set_variable = { global.black_plague_scenario_continent_target = constant:black_plague_scenario.maximum_continent_target } }
	# A valid launch may occur in a reduced-world or heavily conquered save.
	# Never demand more continents than the candidate scan proved usable.
	if = {
		limit = {
			check_variable = {
				var = global.black_plague_scenario_continent_target
				value = global.black_plague_scenario_eligible_continents
				compare = greater_than
			}
		}
		set_variable = { global.black_plague_scenario_continent_target = global.black_plague_scenario_eligible_continents }
	}
	black_plague_scenario_select_europe_anchor = yes
	black_plague_scenario_select_asia_anchor = yes
	black_plague_scenario_select_africa_anchor = yes
	black_plague_scenario_select_north_america_anchor = yes
	black_plague_scenario_select_south_america_anchor = yes
	black_plague_scenario_select_australia_anchor = yes
}

black_plague_scenario_seed_state = {
	black_plague_ensure_current_state_values = yes
	if = {
		limit = { has_state_flag = black_plague_scenario_seeded }
		set_temp_variable = { black_plague_scenario_state_was_seeded = constant:black_plague_value.one }
	}
	else = {
		set_temp_variable = { black_plague_scenario_state_was_seeded = constant:black_plague_value.zero }

# Convert additional infected-or-worse scenario candidates into internal RTA
# warrens. The two-tag correction forbids new country tags, so these broods
# remain distinct through their state markers, infestation, and brood strength
# while sharing the carrier's capped pulse system.
black_plague_scenario_seed_internal_broods = {
	if = {
		limit = {
			RTA = { black_plague_rat_country_is_active = yes }
		}
		# Count the carrier basin and every existing internal marker before the
		# loop. The configured target is a total, making repeated launches
		# idempotent while still filling missing warrens.
		set_variable = { global.black_plague_scenario_internal_broods_created = constant:black_plague_value.zero }
		every_state = {
			limit = {
				has_state_flag = black_plague_scenario_internal_brood
				NOT = { CONTROLLER = { has_country_flag = black_plague_rat_country } }
			}
			clr_state_flag = black_plague_scenario_internal_brood
		}
		RTA = { add_to_variable = { global.black_plague_scenario_internal_broods_created = constant:black_plague_value.one } }
		every_state = {
			limit = { has_state_flag = black_plague_scenario_internal_brood CONTROLLER = { has_country_flag = black_plague_rat_country } }
			add_to_variable = { global.black_plague_scenario_internal_broods_created = constant:black_plague_value.one }
		}
		if = {
			limit = {
				check_variable = { global.black_plague_scenario_internal_brood_target > constant:black_plague_value.one }
				check_variable = { global.black_plague_scenario_rat_candidate_states^num > constant:black_plague_value.zero }
			}
			for_loop_effect = {
				start = 0
				end = global.black_plague_scenario_internal_brood_target
				value = black_plague_scenario_internal_brood_index
				if = {
					limit = {
						check_variable = { global.black_plague_scenario_internal_broods_created < global.black_plague_scenario_internal_brood_target }
						check_variable = { global.black_plague_scenario_rat_candidate_states^num > constant:black_plague_value.zero }
					}
					random_scope_in_array = {
						array = global.black_plague_scenario_rat_candidate_states
						save_event_target_as = black_plague_scenario_internal_brood_state
						remove_from_array = { array = global.black_plague_scenario_rat_candidate_states value = THIS }
					}
					if = {
						limit = { has_event_target = black_plague_scenario_internal_brood_state }
						event_target:black_plague_scenario_internal_brood_state = {
							set_state_flag = black_plague_rat_nation_state
							set_state_flag = black_plague_scenario_internal_brood
							set_variable = { black_plague_provenance = constant:black_plague_provenance.rat_occupation }
							set_variable = { black_plague_rat_infestation = constant:black_plague_evolution_runtime.scenario_rat_infestation }
							set_temp_variable = { black_plague_next_phase = constant:black_plague_phase.rat_controlled }
							black_plague_set_current_state_phase = yes
							black_plague_rat_initialize_state_brood_marker = yes
							RTA = { transfer_state = PREV add_state_core = PREV }
						}
						add_to_variable = { global.black_plague_scenario_internal_broods_created = constant:black_plague_value.one }
					}
				}
			}
		}
	}
}

		clamp_variable = { var = global.air_contamination_natural_source_reservoir_bp min = constant:air_contamination_natural_source.zero max = constant:air_contamination_natural_source.monthly_cap_bp }
	}
}

black_plague_scenario_refresh_shared_threat = {
	if = {
		limit = {
			OR = {
				has_global_flag = black_plague_evolution_iii_active
				has_global_flag = black_plague_rat_king_active
				check_variable = { global.black_plague_established_state_count > constant:black_plague_scenario.minimum_established_states_for_shared_threat }
			}
		}
		set_global_flag = world_threat_source_black_plague
	}
	else = { clr_global_flag = world_threat_source_black_plague }
	refresh_world_threat_state = yes
}

# SCN-012 reuses the ordinary state-owned scheduler. The scenario bootstrap
# must nominate one of its established anchors before the first .900 callback;
# otherwise the launch would seed static values without ever advancing them.
black_plague_scenario_start_scheduler = {
	if = {
		limit = { check_variable = { global.black_plague_scenario_anchor_states^num > constant:black_plague_value.zero } }
		every_state = {
			limit = { has_state_flag = black_plague_scheduler_anchor }
			clr_state_flag = black_plague_scheduler_anchor
		}
		clear_global_event_target = black_plague_scheduler_anchor_state
		random_scope_in_array = {
			array = global.black_plague_scenario_anchor_states
			save_global_event_target_as = black_plague_scheduler_anchor_state
			set_state_flag = black_plague_scheduler_anchor
			black_plague_schedule_next_pulse = yes
		}
	}
	else = {
		# An active-crisis launch may have no unestablished anchor left. Keep the
		# existing disease scheduler alive by selecting one established state.
		clear_temp_array = black_plague_scenario_scheduler_fallback_pool
		for_each_scope_loop = {
			array = global.black_plague_active_states
			if = {
				limit = { black_plague_state_is_established = yes }
				add_to_temp_array = { black_plague_scenario_scheduler_fallback_pool = THIS }
			}
		}
		if = {
			limit = { check_variable = { black_plague_scenario_scheduler_fallback_pool^num > constant:black_plague_value.zero } }
			every_state = { limit = { has_state_flag = black_plague_scheduler_anchor } clr_state_flag = black_plague_scheduler_anchor }
			clear_global_event_target = black_plague_scheduler_anchor_state
			random_scope_in_array = {
				array = black_plague_scenario_scheduler_fallback_pool
				save_global_event_target_as = black_plague_scheduler_anchor_state
				set_state_flag = black_plague_scheduler_anchor
				black_plague_schedule_next_pulse = yes
			}
		}
		clear_temp_array = black_plague_scenario_scheduler_fallback_pool
	}
}

black_plague_triggerable_scenario_launch = {
	if = {
		limit = { black_plague_scenario_can_reconcile_from_triggerable_scenarios = yes }
		set_global_flag = black_plague_triggerable_scenario_bootstrap
		clr_global_flag = black_plague_triggerable_scenario_setup_failed
		# A repeat signal is reconciliation-only, but it still has to restore the
		# configured intensity floor after a save/reload, lost divisions, or a
		# surviving carrier losing one of its internal basins. Candidate collection
		# counts the live established states and infected human basins without
		# reseeding ordinary disease or replaying any evolution.
		black_plague_scenario_collect_candidates = yes
		black_plague_scenario_seed_internal_broods = yes
		black_plague_scenario_expand_king_basin = yes
		black_plague_scenario_raise_rat_army_to_floor = yes
		black_plague_scenario_assign_king_army = yes
		black_plague_rebuild_runtime_counts = yes
		black_plague_refresh_air_cleanliness_source = yes
		black_plague_scenario_refresh_shared_threat = yes
		black_plague_refresh_mapmode_once = yes
		disease_containment_board_refresh_open_black_plague_boards = yes
		if = {
			limit = { NOT = { black_plague_scenario_reconciliation_postcondition_is_ready = yes } }
			set_global_flag = black_plague_triggerable_scenario_setup_failed
		}
		clear_array = global.black_plague_scenario_candidate_states
		clear_array = global.black_plague_scenario_anchor_states
		clear_array = global.black_plague_scenario_rat_candidate_states
		every_state = {
			limit = {
				OR = {
					has_state_flag = black_plague_scenario_reserved
					has_state_flag = black_plague_scenario_anchor
					has_state_flag = black_plague_scenario_king_reserved
				}
			}
			clr_state_flag = black_plague_scenario_reserved
			clr_state_flag = black_plague_scenario_anchor
			clr_state_flag = black_plague_scenario_king_reserved
		}
		clear_variable = global.black_plague_scenario_existing_established_count
		clear_variable = global.black_plague_scenario_unestablished_human_capacity
		clear_variable = global.black_plague_scenario_established_capacity
		clr_global_flag = black_plague_triggerable_scenario_bootstrap
	}
	else_if = {
		limit = { black_plague_scenario_can_start_from_triggerable_scenarios = yes }
		set_global_flag = black_plague_triggerable_scenario_bootstrap
		clr_global_flag = black_plague_triggerable_scenario_setup_failed
		black_plague_scenario_set_intensity_tuning = yes
		black_plague_scenario_collect_candidates = yes
		black_plague_scenario_select_anchor_states = yes
		if = {
			limit = { black_plague_scenario_setup_preflight_is_ready = yes }
			# Do not initialize either runtime until the geography, candidate pool,
			# and reusable RTA/RTX package have passed the fail-closed preflight.
			# A rejected launch therefore cannot turn an untouched campaign into a
			# partially initialized disease run.
			set_global_flag = black_plague_natural_outbreak_started
			black_plague_initialize_runtime = yes
			black_plague_rat_initialize_runtime = yes
			clear_global_event_target = black_plague_rat_evolution_actor
			black_plague_scenario_seed_established_states = yes
			black_plague_scenario_ensure_evolution_ii_port = yes
		black_plague_scenario_create_rat_nations = yes
		black_plague_activate_evolution_i = yes
		black_plague_activate_evolution_ii = yes
		black_plague_scenario_seed_internal_broods = yes
		black_plague_activate_evolution_iii = yes
		black_plague_activate_evolution_iv = yes
		black_plague_scenario_expand_king_basin = yes
		black_plague_scenario_raise_rat_army_to_floor = yes
		black_plague_scenario_assign_king_army = yes
		black_plague_scenario_start_scheduler = yes
		if = {
			limit = { check_variable = { global.chaos_meter_value < global.black_plague_scenario_chaos_floor } }
			set_variable = { global.chaos_meter_value = global.black_plague_scenario_chaos_floor }
		}
		black_plague_rebuild_runtime_counts = yes
		black_plague_refresh_air_cleanliness_source = yes
		black_plague_scenario_refresh_shared_threat = yes
		black_plague_refresh_mapmode_once = yes
		disease_containment_board_refresh_open_black_plague_boards = yes
		clear_array = global.black_plague_scenario_candidate_states
		clear_array = global.black_plague_scenario_anchor_states
		clear_array = global.black_plague_scenario_rat_candidate_states
		every_state = { limit = { has_state_flag = black_plague_scenario_reserved } clr_state_flag = black_plague_scenario_reserved }
		every_state = { limit = { has_state_flag = black_plague_scenario_anchor } clr_state_flag = black_plague_scenario_anchor }
		every_state = { limit = { has_state_flag = black_plague_scenario_king_reserved } clr_state_flag = black_plague_scenario_king_reserved }
		if = {
			limit = {
				RTA = { black_plague_rat_country_is_active = yes }
				RTX = { exists = yes has_country_flag = black_plague_rat_king_country }
				has_global_flag = black_plague_evolution_i_active
				has_global_flag = black_plague_evolution_ii_active
				has_global_flag = black_plague_evolution_iii_active
				has_global_flag = black_plague_evolution_iv_active
				has_global_flag = black_plague_rat_king_active
				check_variable = { global.black_plague_scheduler_ticket > constant:black_plague_value.zero }
				any_state = { has_state_flag = black_plague_scheduler_anchor }
				check_variable = { global.black_plague_scenario_selected_continents > constant:black_plague_value.one }
				check_variable = {
					var = global.black_plague_scenario_rat_created
					value = global.black_plague_scenario_rat_target
					compare = greater_than_or_equals
				}
				check_variable = {
					var = global.black_plague_scenario_seeded_states
					value = global.black_plague_scenario_established_target
					compare = greater_than_or_equals
				}
				check_variable = {
					var = global.black_plague_scenario_internal_broods_created
					value = global.black_plague_scenario_internal_brood_target
					compare = greater_than_or_equals
				}
				check_variable = {
					var = global.black_plague_scenario_king_states_transferred
					value = global.black_plague_scenario_king_state_target
					compare = greater_than_or_equals
				}
				RTA = {
					check_variable = {
						var = black_plague_rat_division_count
						value = global.black_plague_scenario_rat_division_floor
						compare = greater_than_or_equals
					}
				}
				RTX = {
					check_variable = {
						var = black_plague_rat_division_count
						value = global.black_plague_scenario_king_division_floor
						compare = greater_than_or_equals
					}
				}
			}
			black_plague_scenario_suppress_fire_once = yes
			black_plague_scenario_record_history = yes
			set_global_flag = black_plague_scenario_shortcut_disqualifier
			set_global_flag = black_plague_triggerable_scenario_launched
			if = {
				limit = { NOT = { has_global_flag = black_plague_scenario_report_emitted } }
				news_event = { id = chaosx.nr20.90 }
				set_global_flag = black_plague_scenario_report_emitted
			}
			if = {
				limit = {
					RTX = { exists = yes has_country_flag = black_plague_rat_king_country }
					NOT = { has_global_flag = black_plague_rat_king_coronation_emitted }
				}
				set_temp_variable = { black_plague_coronation_super_event_id = constant:black_plague_identity.rat_king_coronation_super_event_id }
				set_temp_variable = { black_plague_coronation_super_event_days = constant:black_plague_evolution_runtime.coronation_super_event_visible_days }
				set_global_flag = { flag = super_event_visible days = black_plague_coronation_super_event_days value = black_plague_coronation_super_event_id }
				set_variable = { global.current_super_event_audio_id = constant:black_plague_identity.rat_king_coronation_audio_id }
				every_country = { limit = { is_ai = no } play_current_super_event_audio = yes }
				set_global_flag = black_plague_rat_king_coronation_emitted
			}
		}
		else = {
			# Never leave the scoped bypass or reservation markers behind when a
			# downstream package/postcondition fails. The launch flag is deliberately
			# withheld so the triggerable scenario remains retryable and idempotent.
			set_global_flag = black_plague_triggerable_scenario_setup_failed
			clear_global_event_target = black_plague_scheduler_anchor_state
			clear_global_event_target = black_plague_scenario_evolution_ii_port_state
			clear_global_event_target = black_plague_scenario_selected_rat_state
			clear_global_event_target = black_plague_scenario_internal_brood_state
			clear_global_event_target = black_plague_scenario_king_next_state
			every_state = {
				limit = {
					OR = {
						has_state_flag = black_plague_scenario_reserved
						has_state_flag = black_plague_scenario_anchor
						has_state_flag = black_plague_scenario_king_reserved
					}
				}
				clr_state_flag = black_plague_scenario_reserved
				clr_state_flag = black_plague_scenario_anchor
				clr_state_flag = black_plague_scenario_king_reserved
			}
		}
		}
		else = {
			# Fail before any state, country, evolution, or log mutation when the
			# reserved geography cannot satisfy the selected intensity.
			set_global_flag = black_plague_triggerable_scenario_setup_failed
			clear_global_event_target = black_plague_scheduler_anchor_state
			clear_global_event_target = black_plague_scenario_evolution_ii_port_state
			clear_global_event_target = black_plague_scenario_selected_rat_state
			clear_global_event_target = black_plague_scenario_internal_brood_state
			clear_global_event_target = black_plague_scenario_king_next_state
			every_state = {
				limit = {
					OR = {
						has_state_flag = black_plague_scenario_reserved
						has_state_flag = black_plague_scenario_anchor
						has_state_flag = black_plague_scenario_king_reserved
					}
				}
				clr_state_flag = black_plague_scenario_reserved
				clr_state_flag = black_plague_scenario_anchor
				clr_state_flag = black_plague_scenario_king_reserved
			}
		}
		clear_variable = global.black_plague_scenario_existing_established_count
		clear_variable = global.black_plague_scenario_unestablished_human_capacity
		clear_variable = global.black_plague_scenario_established_capacity
		clr_global_flag = black_plague_triggerable_scenario_bootstrap
	}
}
