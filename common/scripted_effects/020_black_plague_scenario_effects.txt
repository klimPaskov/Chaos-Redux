# ============================================================================
# CHAOS REDUX - EVENT 020 SCN-012 BLACK PLAGUE UNBOUND
# ============================================================================
# The triggerable scenario is a single, idempotent bootstrap transaction. It
# seeds the ordinary Event 020 state machine, activates Evolutions I-IV in
# order, creates the reusable Rat Nation, and establishes a
# separate Royal Basin for RTX. The scoped bootstrap flag is the only bypass;
# it is cleared before ordinary pulses resume. Evolution V and world_end are
# deliberately outside this entry point.

	set_global_flag = black_plague_scenario_commit_provider_ready
}

black_plague_scenario_suppress_fire_once = {
	set_global_flag = black_plague_triggerable_scenario_suppressed_fire_once
	set_temp_variable = { black_plague_scenario_event_id = constant:black_plague_identity.event_id }
	if = {
		limit = { NOT = { is_in_array = { global.disabled_events = black_plague_scenario_event_id } } }
		add_to_array = { global.disabled_events = black_plague_scenario_event_id }
	}
	if = {
		limit = { NOT = { is_in_array = { global.fired_events = black_plague_scenario_event_id } } }
		add_to_array = { global.fired_events = black_plague_scenario_event_id }
	}
	if = {
		limit = { is_in_array = { global.fire_once_events = black_plague_scenario_event_id } }
		remove_from_array = { global.fire_once_events = black_plague_scenario_event_id }
	}
	set_temp_variable = { new_weight = 0 }
	set_event_weight = yes
	update_unfired_events_counts = yes
	calculate_dynamic_major_weight_gain = yes
}

black_plague_scenario_seed_state = {
	black_plague_ensure_current_state_values = yes
	set_variable = { black_plague_provenance = constant:black_plague_provenance.natural }
	set_variable = { black_plague_load = constant:black_plague_evolution_runtime.scenario_seed_load }
	set_variable = { black_plague_mortality_pressure = constant:black_plague_evolution_runtime.scenario_seed_mortality }
	set_variable = { black_plague_spread_pressure = constant:black_plague_evolution_runtime.scenario_seed_spread }
	set_variable = { black_plague_containment = constant:black_plague_evolution_runtime.scenario_seed_containment }
	set_variable = { black_plague_treatment_coverage = constant:black_plague_evolution_runtime.scenario_seed_treatment }
	set_variable = { black_plague_relapse_risk = constant:black_plague_evolution_runtime.scenario_seed_relapse }
	set_variable = { black_plague_rat_infestation = constant:black_plague_evolution_runtime.scenario_seed_infestation }
	set_variable = { black_plague_incoming_exposure = constant:black_plague_value.zero }
	set_temp_variable = { black_plague_scenario_seed_phase = constant:black_plague_phase.infected }
	if = {
		limit = { check_variable = { global.black_plague_scenario_intensity = constant:triggerable_scenario_intensity.medium } }
		set_temp_variable = { black_plague_scenario_seed_phase = constant:black_plague_phase.severe_crisis }
	}
	else_if = {
		limit = { check_variable = { global.black_plague_scenario_intensity > constant:triggerable_scenario_intensity.medium } }
		set_temp_variable = { black_plague_scenario_seed_phase = constant:black_plague_phase.severe_crisis }
	}
	if = {
		limit = {
			has_state_flag = black_plague_scenario_anchor
			check_variable = { global.black_plague_scenario_intensity = constant:triggerable_scenario_intensity.maximum }
		}
		set_temp_variable = { black_plague_scenario_seed_phase = constant:black_plague_phase.collapsed }
	}
	set_temp_variable = { black_plague_next_phase = black_plague_scenario_seed_phase }
	black_plague_set_current_state_phase = yes
	set_state_flag = black_plague_scenario_seeded
	if = { limit = { black_plague_state_is_severe = yes } set_state_flag = black_plague_scenario_rat_candidate add_to_array = { array = global.black_plague_scenario_rat_candidate_states value = THIS } }
	if = { limit = { black_plague_state_is_collapsed = yes } set_state_flag = black_plague_scenario_rat_candidate add_to_array = { array = global.black_plague_scenario_rat_candidate_states value = THIS } }
	if = {
		limit = { NOT = { has_state_flag = black_plague_scenario_opening_deaths_applied } }
		set_temp_variable = { black_plague_scenario_opening_state_deaths = global.black_plague_scenario_opening_deaths_remaining }
		divide_temp_variable = { black_plague_scenario_opening_state_deaths = global.black_plague_scenario_established_target }
		clamp_temp_variable = { var = black_plague_scenario_opening_state_deaths min = constant:black_plague_scenario.opening_deaths_minimum_remaining }
		set_temp_variable = { state_civilian_population_loss_requested = black_plague_scenario_opening_state_deaths }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:black_plague_scenario.opening_population_floor }
		multiply_temp_variable = { state_civilian_population_loss_minimum_remaining = state_population_k }
		multiply_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:black_plague_value.people_per_thousand }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.black_plague }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:black_plague_value.one }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:black_plague_value.one }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:black_plague_value.one }
		apply_exact_state_civilian_population_loss = yes
		if = {
			limit = { check_variable = { state_civilian_population_loss_applied > constant:black_plague_value.zero } }
			add_to_variable = { global.black_plague_deaths_total = state_civilian_population_loss_applied }
			subtract_from_variable = { global.black_plague_scenario_opening_deaths_remaining = state_civilian_population_loss_applied }
		}
		set_state_flag = black_plague_scenario_opening_deaths_applied
	}
	black_plague_register_current_state = yes
	add_to_variable = { global.black_plague_scenario_seeded_states = constant:black_plague_value.one }
	every_neighbor_state = {
		limit = {
			black_plague_scenario_valid_human_state = yes
			NOT = { has_state_flag = black_plague_scenario_seeded }
			NOT = { has_state_flag = black_plague_established_infection }
		}
		black_plague_ensure_current_state_values = yes
		set_variable = { black_plague_incoming_exposure = constant:black_plague_scenario.outer_ring_exposure }
		set_variable = { black_plague_provenance = constant:black_plague_provenance.natural }
		set_temp_variable = { black_plague_next_phase = constant:black_plague_phase.threatened }
		black_plague_set_current_state_phase = yes
		black_plague_register_current_state = yes
	}
}

black_plague_scenario_seed_established_states = {
	for_each_scope_loop = { array = global.black_plague_scenario_anchor_states set_state_flag = black_plague_scenario_anchor black_plague_scenario_seed_state = yes }
	clear_temp_array = black_plague_scenario_seed_pool
	every_state = { limit = { black_plague_scenario_valid_human_state = yes NOT = { has_state_flag = black_plague_scenario_seeded } } add_to_temp_array = { black_plague_scenario_seed_pool = THIS } }
	set_temp_variable = { black_plague_scenario_seed_remaining = global.black_plague_scenario_established_target }
	subtract_from_temp_variable = { black_plague_scenario_seed_remaining = global.black_plague_scenario_seeded_states }
	clamp_temp_variable = { var = black_plague_scenario_seed_remaining min = constant:black_plague_value.zero }
	for_loop_effect = {
		start = 0
		end = black_plague_scenario_seed_remaining
		value = black_plague_scenario_seed_index
		if = {
			limit = { check_variable = { black_plague_scenario_seed_pool^num > constant:black_plague_value.zero } }
			random_scope_in_array = { array = black_plague_scenario_seed_pool set_state_flag = black_plague_scenario_anchor black_plague_scenario_seed_state = yes remove_from_temp_array = { array = black_plague_scenario_seed_pool value = THIS } }
		}
	}
	clear_temp_array = black_plague_scenario_seed_pool
}

black_plague_scenario_create_rat_nations = {
	if = {
		limit = {
			check_variable = { global.black_plague_rat_countries_count = constant:black_plague_value.zero }
			check_variable = { global.black_plague_scenario_rat_candidate_states^num > constant:black_plague_value.zero }
		}
		random_scope_in_array = {
			array = global.black_plague_scenario_rat_candidate_states
			save_event_target_as = black_plague_scenario_selected_rat_state
			remove_from_array = { array = global.black_plague_scenario_rat_candidate_states value = THIS }
		}
		black_plague_rat_select_free_slot = yes
		if = {
			limit = { check_variable = { black_plague_rat_slot_result = constant:black_plague_value.one } }
			event_target:black_plague_scenario_selected_rat_state = {
				save_event_target_as = black_plague_rat_spawn_state
				black_plague_rat_create_from_state = yes
			}
			event_target:black_plague_rat_selected_country = { save_event_target_as = black_plague_rat_evolution_actor }
			set_variable = { global.black_plague_scenario_rat_created = constant:black_plague_value.one }
		}
	}
}

black_plague_scenario_expand_king_basin = {
	if = {
		limit = { RTX = { exists = yes has_country_flag = black_plague_rat_king_country } }
		set_variable = { global.black_plague_scenario_king_states_transferred = constant:black_plague_value.zero }
		RTX = { set_variable = { global.black_plague_scenario_king_states_transferred = num_of_controlled_states } }
		for_loop_effect = {
			start = 0
			end = global.black_plague_scenario_king_state_target
			value = black_plague_scenario_king_expansion_index
			if = {
				limit = { check_variable = { global.black_plague_scenario_king_states_transferred < global.black_plague_scenario_king_state_target } }
				clear_temp_array = black_plague_scenario_king_state_pool

black_plague_refresh_air_cleanliness_source = {
	if = {
		limit = {
			OR = {
				has_global_flag = fallout_air_cleanliness_disabled
				has_global_flag = fallout_air_contamination_permanent_99
			}
		}
		set_variable = { global.air_contamination_natural_source_reservoir_bp = constant:air_contamination_natural_source.zero }
		set_variable = { global.air_contamination_natural_source_bp = constant:air_contamination_natural_source.zero }
		set_variable = { global.black_plague_air_source_bp = constant:air_contamination_natural_source.zero }
		set_variable = { global.black_plague_air_source_previous_bp = constant:air_contamination_natural_source.zero }
	}
	else = {
		if = {
			limit = {
				has_variable = global.air_contamination_natural_source_reservoir_bp
				has_variable = global.black_plague_air_source_previous_bp
			}
			set_temp_variable = { black_plague_air_source_remove = global.black_plague_air_source_previous_bp }
			multiply_temp_variable = { black_plague_air_source_remove = constant:air_contamination_delta_bp.negative_one }
			add_to_variable = { global.air_contamination_natural_source_reservoir_bp = black_plague_air_source_remove }
		}
		if = {
			limit = { NOT = { has_variable = global.air_contamination_natural_source_reservoir_bp } }
			set_variable = { global.air_contamination_natural_source_reservoir_bp = constant:air_contamination_natural_source.zero }
		}
		set_variable = { global.black_plague_air_source_bp = constant:air_contamination_natural_source.zero }
		for_each_scope_loop = {
			array = global.black_plague_active_states
			if = {
				limit = { black_plague_state_is_established = yes }
				if = { limit = { black_plague_state_is_infected = yes } add_to_variable = { global.black_plague_air_source_bp = constant:black_plague_scenario.air_source_infected } }
				if = { limit = { black_plague_state_is_severe = yes } add_to_variable = { global.black_plague_air_source_bp = constant:black_plague_scenario.air_source_severe } }
				if = { limit = { black_plague_state_is_collapsed = yes } add_to_variable = { global.black_plague_air_source_bp = constant:black_plague_scenario.air_source_collapsed } }
				if = { limit = { black_plague_state_is_contained = yes } add_to_variable = { global.black_plague_air_source_bp = constant:black_plague_scenario.air_source_contained } }
				if = { limit = { black_plague_state_is_recovering = yes } add_to_variable = { global.black_plague_air_source_bp = constant:black_plague_scenario.air_source_recovery } }
				if = { limit = { black_plague_state_is_rat_controlled = yes } add_to_variable = { global.black_plague_air_source_bp = constant:black_plague_scenario.air_source_rat_controlled } }
			}
		}
		clamp_variable = { var = global.black_plague_air_source_bp min = constant:air_contamination_natural_source.zero max = constant:black_plague_scenario.air_source_cap }
		set_variable = { global.black_plague_air_source_previous_bp = global.black_plague_air_source_bp }
		add_to_variable = { global.air_contamination_natural_source_reservoir_bp = global.black_plague_air_source_bp }
		clamp_variable = { var = global.air_contamination_natural_source_reservoir_bp min = constant:air_contamination_natural_source.zero max = constant:air_contamination_natural_source.monthly_cap_bp }
	}
}
