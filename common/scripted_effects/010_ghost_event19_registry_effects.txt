chaos_unit_family_provider_502_event19_evaluate_management = {
	set_temp_variable = { infantry_spawn_family_provider_can_train = 0 }
	set_temp_variable = { infantry_spawn_family_provider_can_spawn = 0 }
	set_temp_variable = { infantry_spawn_family_provider_can_sustain = 0 }
	set_temp_variable = { infantry_spawn_family_provider_uses_training = 0 }
	if = {
		limit = {
			NOT = { has_political_power < constant:event19_ghost_family.manifest_political_power_cost }
			NOT = { command_power < constant:event19_ghost_family.manifest_command_power_cost }
			infantry_spawn_country_has_eligible_state = yes
		}
		set_temp_variable = { infantry_spawn_family_provider_can_spawn = 1 }
	}
	if = {
		limit = {
			check_variable = { infantry_spawn_family_live_division_count > 0 }
			NOT = { has_political_power < constant:event19_ghost_family.sustainment_political_power_cost }
			NOT = { command_power < constant:event19_ghost_family.sustainment_command_power_cost }
		}
		set_temp_variable = { infantry_spawn_family_provider_can_sustain = 1 }
	}
}

chaos_unit_family_provider_502_event19_pay_management_action = {
	set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
			check_variable = { infantry_spawn_family_provider_can_spawn > 0 }
		}
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_ghost_family.manifest_political_power_cost }
		add_political_power = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_ghost_family.manifest_command_power_cost }
		add_command_power = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
	else_if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.sustainment }
			check_variable = { infantry_spawn_family_provider_can_sustain > 0 }
		}
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_ghost_family.sustainment_political_power_cost }
		add_political_power = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_ghost_family.sustainment_command_power_cost }
		add_command_power = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
}

chaos_unit_family_provider_502_event19_refund_management_action = {
	if = {
		limit = {
			check_variable = { infantry_spawn_family_provider_payment_succeeded > 0 }
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
		}
		add_political_power = constant:event19_ghost_family.manifest_political_power_cost
		add_command_power = constant:event19_ghost_family.manifest_command_power_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	}
	else_if = {
		limit = {
			check_variable = { infantry_spawn_family_provider_payment_succeeded > 0 }
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.sustainment }
		}
		add_political_power = constant:event19_ghost_family.sustainment_political_power_cost
		add_command_power = constant:event19_ghost_family.sustainment_command_power_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	}
}
