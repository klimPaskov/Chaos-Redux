# ============================================================================
# CHAOS REDUX - EVENT 030 TIME TRAVELER INTELLIGENCE WAR EFFECTS
# ============================================================================
# The interest registry, the four foreign operations, the four host
# countermeasures, the double, custody transfer and the loss of the traveler.
#
# Registry: global.time_traveler_interest holds at most two, four or six
# countries (Secret, Suspected, Public Exposure). It is rebuilt only after a
# public revelation, a custody transfer, a new evolution or when a member stops
# being valid, never by a recurring world scan. A member carries the flag
# time_traveler_interested and a hidden network level time_traveler_network
# from 0 to 3 that no screen prints as a number.
# AI members act through a self-rescheduling hidden event, chaosx.nr30.30, one
# per member, so no on_action is used. Human members act through decisions.
# The operations are always scoped to the attacker. The custodian is read from
# global.time_traveler_custodian.
# ============================================================================

# add_tech_bonus reads plain numbers, so its strength is a file constant.
@time_traveler_stolen_bonus = 0.5

# ---------------------------------------------------------------------------
# Dice
# ---------------------------------------------------------------------------

# ANY scope. Output: temp time_traveler_roll in [0, 100).
time_traveler_roll_percent = {
	set_temp_variable_to_random = { var = time_traveler_roll min = constant:time_traveler_value.zero max = constant:time_traveler_value.percent_max }
}

# ANY scope. Input: temp time_traveler_chance. Clamps it and rolls. Output: temp
# time_traveler_success (1 when the roll is below the chance).
time_traveler_resolve_chance = {
	clamp_temp_variable = { var = time_traveler_chance min = constant:time_traveler_chance.minimum max = constant:time_traveler_chance.maximum }
	time_traveler_roll_percent = yes
	set_temp_variable = { time_traveler_success = constant:time_traveler_value.zero }
	if = {
		limit = { check_variable = { time_traveler_roll < time_traveler_chance } }
		set_temp_variable = { time_traveler_success = constant:time_traveler_value.one }
	}
}

# COUNTRY (candidate). Scores how likely this country is to chase the traveler.
# Reads the custodian saved as time_traveler_scope_country.
time_traveler_score_interest_candidate = {
	set_variable = { time_traveler_interest_score = constant:time_traveler_value.zero }
	if = {
		limit = { is_major = yes }
		add_to_variable = { time_traveler_interest_score = constant:time_traveler_interest.major }
	}
	if = {
		limit = { is_ai = no }
		add_to_variable = { time_traveler_interest_score = constant:time_traveler_interest.player }
	}
	if = {
		limit = { has_intelligence_agency = yes }
		add_to_variable = { time_traveler_interest_score = constant:time_traveler_interest.agency }
	}
	if = {
		limit = { has_war_with = event_target:time_traveler_scope_country }
		add_to_variable = { time_traveler_interest_score = constant:time_traveler_interest.war_with_custodian }
	}
	if = {
		limit = { is_neighbor_of = event_target:time_traveler_scope_country }
		add_to_variable = { time_traveler_interest_score = constant:time_traveler_interest.neighbor }
	}
	if = {
		limit = { is_in_faction_with = event_target:time_traveler_scope_country }
		add_to_variable = { time_traveler_interest_score = constant:time_traveler_interest.ally }
	}
}

# COUNTRY (candidate). Adds this country to the registry.
time_traveler_register_member = {
	add_to_array = { global.time_traveler_interest = THIS }
	set_country_flag = time_traveler_interested
	if = {
		limit = { NOT = { has_variable = time_traveler_network } }
		set_variable = { time_traveler_network = constant:time_traveler_network.none }
	}
}

# COUNTRY (custodian). Empties the registry. Members that leave lose their
# network.
time_traveler_clear_registry = {
	for_each_scope_loop = {
		array = global.time_traveler_interest
		clr_country_flag = time_traveler_interested
		clr_country_flag = time_traveler_rival_chain
		set_variable = { time_traveler_network = constant:time_traveler_network.none }
	}
	clear_array = global.time_traveler_interest
}

# COUNTRY (custodian). Rebuilds the registry for the current Exposure band.
# Strong candidates come first, then the others, up to the band's cap. A
# member that stays keeps its network, a member that leaves loses it.
time_traveler_rebuild_interest_registry = {
	save_event_target_as = time_traveler_scope_country
	set_temp_variable = { time_traveler_slots = constant:time_traveler_band.interest_cap_secret }
	if = {
		limit = { time_traveler_exposure_is_public = yes }
		set_temp_variable = { time_traveler_slots = constant:time_traveler_band.interest_cap_public }
	}
	else_if = {
		limit = { time_traveler_exposure_is_suspected_or_higher = yes }
		set_temp_variable = { time_traveler_slots = constant:time_traveler_band.interest_cap_suspected }
	}
	for_each_scope_loop = {
		array = global.time_traveler_interest
		set_country_flag = time_traveler_previous_member
		clr_country_flag = time_traveler_interested
	}
	clear_array = global.time_traveler_interest
	every_country = {
		limit = {
			exists = yes
			NOT = { has_country_flag = time_traveler_custodian }
			is_special_chaos_country = no
			is_actual_nonhuman_country = no
			NOT = { has_capitulated = yes }
		}
		time_traveler_score_interest_candidate = yes
	}
	for_loop_effect = {
		start = 0
		end = time_traveler_slots
		value = time_traveler_slot_index
		set_temp_variable = { time_traveler_picked = constant:time_traveler_value.zero }
		random_country = {
			limit = {
				has_variable = time_traveler_interest_score
				NOT = { has_country_flag = time_traveler_interested }
				NOT = { check_variable = { time_traveler_interest_score < constant:time_traveler_interest.strong_score } }
			}
			time_traveler_register_member = yes
			set_temp_variable = { time_traveler_picked = constant:time_traveler_value.one }
		}
		if = {
			limit = { check_variable = { time_traveler_picked = constant:time_traveler_value.zero } }
			random_country = {
				limit = {
					has_variable = time_traveler_interest_score
					NOT = { has_country_flag = time_traveler_interested }
					NOT = { check_variable = { time_traveler_interest_score < constant:time_traveler_interest.minimum_score } }
				}
				time_traveler_register_member = yes
			}
		}
	}
	every_country = {
		limit = {
			OR = {
				has_variable = time_traveler_interest_score
				has_country_flag = time_traveler_previous_member
			}
		}
		clear_variable = time_traveler_interest_score
		if = {
			limit = {
				has_country_flag = time_traveler_previous_member
				NOT = { has_country_flag = time_traveler_interested }
			}
			set_variable = { time_traveler_network = constant:time_traveler_network.none }
			clr_country_flag = time_traveler_rival_chain
		}
		clr_country_flag = time_traveler_previous_member
	}
	for_each_scope_loop = {
		array = global.time_traveler_interest
		time_traveler_start_rival_chain = yes
	}
}

# COUNTRY (member). Starts the chain of rival turns of an AI member once and
# informs a human member that it can act.
time_traveler_start_rival_chain = {
	time_traveler_refresh_rival_costs = yes
	if = {
		limit = {
			is_ai = yes
			NOT = { has_country_flag = time_traveler_rival_chain }
		}
		set_country_flag = time_traveler_rival_chain
		set_temp_variable = { time_traveler_delay = constant:time_traveler_timing.rival_first_days }
		country_event = { id = chaosx.nr30.30 days = time_traveler_delay }
	}
	if = {
		limit = {
			is_ai = no
			NOT = { has_country_flag = time_traveler_informed }
		}
		set_country_flag = time_traveler_informed
		country_event = { id = chaosx.nr30.50 days = constant:time_traveler_timing.registry_rebuild_days }
	}
}

# COUNTRY (custodian). Lowers the network of one random member by the reward
# amount, used by counterintelligence successes.
time_traveler_drop_random_network = {
	random_scope_in_array = {
		array = global.time_traveler_interest
		limit = { check_variable = { time_traveler_network > constant:time_traveler_value.zero } }
		subtract_from_variable = { time_traveler_network = constant:time_traveler_reward.counterintelligence_network_drop }
		clamp_variable = { var = time_traveler_network min = constant:time_traveler_value.zero max = constant:time_traveler_network.mature }
	}
}

# COUNTRY (an AI member), called by chaosx.nr30.30. Picks one operation or
# restraint from relations, intelligence capacity, ideology, proximity and the
# network it already holds. A low-chance seizure is never repeated.
time_traveler_rival_turn = {
	set_temp_variable = { time_traveler_weight_infiltrate = constant:time_traveler_ai.rival_infiltrate }
	set_temp_variable = { time_traveler_weight_forge = constant:time_traveler_value.zero }
	set_temp_variable = { time_traveler_weight_asylum = constant:time_traveler_value.zero }
	set_temp_variable = { time_traveler_weight_seize = constant:time_traveler_value.zero }
	set_temp_variable = { time_traveler_weight_restraint = constant:time_traveler_ai.rival_restraint }
	if = {
		limit = { time_traveler_rival_can_infiltrate = yes }
		if = {
			limit = { has_intelligence_agency = yes }
			add_to_temp_variable = { time_traveler_weight_infiltrate = constant:time_traveler_ai.rival_agency }
		}
	}
	else = { set_temp_variable = { time_traveler_weight_infiltrate = constant:time_traveler_value.zero } }
	if = {
		limit = { time_traveler_rival_can_forge = yes }
		set_temp_variable = { time_traveler_weight_forge = constant:time_traveler_ai.rival_forge }
		if = {
			limit = { has_war_with = var:global.time_traveler_custodian }
			add_to_temp_variable = { time_traveler_weight_forge = constant:time_traveler_ai.rival_hostile }
		}
	}
	if = {
		limit = { time_traveler_rival_can_offer_asylum = yes }
		set_temp_variable = { time_traveler_weight_asylum = constant:time_traveler_ai.rival_asylum }
		if = {
			limit = { time_traveler_same_ideology_as_custodian = yes }
			add_to_temp_variable = { time_traveler_weight_asylum = constant:time_traveler_ai.rival_ideology }
		}
	}
	if = {
		limit = {
			time_traveler_rival_can_seize = yes
			NOT = { has_country_flag = time_traveler_seize_tried }
		}
		set_temp_variable = { time_traveler_weight_seize = constant:time_traveler_ai.rival_seize }
	}
	if = {
		limit = { is_in_faction_with = var:global.time_traveler_custodian }
		set_temp_variable = { time_traveler_weight_seize = constant:time_traveler_value.zero }
		add_to_temp_variable = { time_traveler_weight_restraint = constant:time_traveler_ai.rival_ally_restraint }
	}
	set_temp_variable = { time_traveler_weight_total = time_traveler_weight_infiltrate }
	add_to_temp_variable = { time_traveler_weight_total = time_traveler_weight_forge }
	add_to_temp_variable = { time_traveler_weight_total = time_traveler_weight_asylum }
	add_to_temp_variable = { time_traveler_weight_total = time_traveler_weight_seize }
	add_to_temp_variable = { time_traveler_weight_total = time_traveler_weight_restraint }
	set_temp_variable_to_random = { var = time_traveler_pick }
	multiply_temp_variable = { time_traveler_pick = time_traveler_weight_total }
	set_temp_variable = { time_traveler_edge = time_traveler_weight_infiltrate }
	if = {
		limit = { check_variable = { time_traveler_pick < time_traveler_edge } }
		time_traveler_operation_infiltrate = yes
	}
	else = {
		add_to_temp_variable = { time_traveler_edge = time_traveler_weight_forge }
		if = {
			limit = { check_variable = { time_traveler_pick < time_traveler_edge } }
			time_traveler_operation_forge = yes
		}
		else = {
			add_to_temp_variable = { time_traveler_edge = time_traveler_weight_asylum }
			if = {
				limit = { check_variable = { time_traveler_pick < time_traveler_edge } }
				time_traveler_operation_asylum = yes
			}
			else = {
				add_to_temp_variable = { time_traveler_edge = time_traveler_weight_seize }
				if = {
					limit = { check_variable = { time_traveler_pick < time_traveler_edge } }
					time_traveler_operation_seize = yes
				}
			}
		}
	}
}

# COUNTRY (an AI member). Schedules the next rival turn.
time_traveler_next_rival_turn = {
	set_temp_variable = { time_traveler_delay = mtth:time_traveler_rival_interval }
	set_temp_variable_to_random = { var = time_traveler_jitter min = constant:time_traveler_interval.jitter_low max = constant:time_traveler_interval.jitter_high }
	multiply_temp_variable = { time_traveler_delay = time_traveler_jitter }
	round_temp_variable = time_traveler_delay
	country_event = { id = chaosx.nr30.30 days = time_traveler_delay }
}

# COUNTRY (attacker). Records what a human attacker will read in the result
# report. Inputs: temp time_traveler_op_kind and time_traveler_op_result.
time_traveler_report_to_attacker = {
	set_variable = { time_traveler_last_op = time_traveler_op_kind }
	set_variable = { time_traveler_last_op_result = time_traveler_op_result }
	if = {
		limit = { is_ai = no }
		country_event = { id = chaosx.nr30.37 days = constant:time_traveler_timing.report_delay_days }
	}
}

# COUNTRY (attacker). Infiltrate the program. Each success advances the network
# one step: schedules and staff names, then a recruited staff member and a
# Future Fragment, then the archive and the first global theft milestone.
time_traveler_operation_infiltrate = {
	var:global.time_traveler_custodian = { save_event_target_as = time_traveler_target_custodian }
	set_temp_variable = { time_traveler_op_kind = constant:time_traveler_operation.infiltrate }
	set_temp_variable = { time_traveler_chance = constant:time_traveler_chance.infiltrate_base }
	set_temp_variable = { time_traveler_step = time_traveler_network }
	multiply_temp_variable = { time_traveler_step = constant:time_traveler_chance.infiltrate_per_network }
	add_to_temp_variable = { time_traveler_chance = time_traveler_step }
	if = {
		limit = { has_intelligence_agency = yes }
		add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.infiltrate_agency_bonus }
	}
	event_target:time_traveler_target_custodian = {
		if = {
			limit = { has_country_flag = time_traveler_guard_detail }
			subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_chance.infiltrate_guard_penalty }
		}
		if = {
			limit = { has_country_flag = time_traveler_sting_recent }
			subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_chance.infiltrate_sting_penalty }
		}
		if = {
			limit = { time_traveler_arrangement_is_secret = yes }
			subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_chance.infiltrate_secret_penalty }
		}
	}
	if = {
		limit = { has_country_flag = time_traveler_watched }
		subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_double.watched_penalty }
		clr_country_flag = time_traveler_watched
	}
	time_traveler_resolve_chance = yes
	if = {
		limit = { check_variable = { time_traveler_success = constant:time_traveler_value.one } }
		if = {
			limit = { check_variable = { time_traveler_network < constant:time_traveler_network.archive } }
			add_to_variable = { time_traveler_network = constant:time_traveler_value.one }
		}
		set_temp_variable = { time_traveler_op_result = time_traveler_network }
		if = {
			limit = { check_variable = { time_traveler_network = constant:time_traveler_network.recruit } }
			time_traveler_grant_fragment = yes
			event_target:time_traveler_target_custodian = { country_event = { id = chaosx.nr30.31 days = constant:time_traveler_timing.theft_report_days } }
		}
		else_if = {
			limit = { check_variable = { time_traveler_network = constant:time_traveler_network.archive } }
			time_traveler_grant_fragment = yes
			time_traveler_steal_archive = yes
		}
	}
	else = {
		set_temp_variable = { time_traveler_op_result = constant:time_traveler_value.zero }
		time_traveler_roll_percent = yes
		if = {
			limit = {
				check_variable = { time_traveler_roll < constant:time_traveler_chance.exposed_after_failure }
				check_variable = { time_traveler_network > constant:time_traveler_value.zero }
			}
			subtract_from_variable = { time_traveler_network = constant:time_traveler_value.one }
			set_temp_variable = { time_traveler_op_result = constant:time_traveler_value.negative_one }
			event_target:time_traveler_target_custodian = { country_event = { id = chaosx.nr30.36 days = constant:time_traveler_timing.theft_report_days } }
		}
	}
	time_traveler_report_to_attacker = yes
}

# COUNTRY (attacker). The third infiltration step: the archive is copied. The
# attacker gains a research bonus for the specialty copied, the custodian
# learns of the theft and the first theft in the campaign moves Chaos once.
time_traveler_steal_archive = {
	add_tech_bonus = { name = time_traveler_stolen_archive bonus = @time_traveler_stolen_bonus uses = 1 category = industry }
	if = {
		limit = { NOT = { has_global_flag = time_traveler_theft_recorded } }
		set_global_flag = time_traveler_theft_recorded
		set_temp_variable = { chaos_change = constant:time_traveler_chaos.theft }
		set_temp_variable = { chaos_history_reason = constant:chaos_meter_history_reason.special.time_traveler_theft }
		set_temp_variable = { chaos_history_reason_custom = constant:time_traveler_value.one }
		add_chaos_meter_value = yes
	}
	set_global_flag = time_traveler_theft_happened
	event_target:time_traveler_target_custodian = { country_event = { id = chaosx.nr30.35 days = constant:time_traveler_timing.theft_report_days } }
}

# COUNTRY (attacker). Plant forged advice. It needs a network of its own.
time_traveler_operation_forge = {
	var:global.time_traveler_custodian = { save_event_target_as = time_traveler_target_custodian }
	set_temp_variable = { time_traveler_op_kind = constant:time_traveler_operation.forge }
	set_temp_variable = { time_traveler_chance = constant:time_traveler_chance.forge_base }
	set_temp_variable = { time_traveler_step = time_traveler_network }
	multiply_temp_variable = { time_traveler_step = constant:time_traveler_chance.forge_per_network }
	add_to_temp_variable = { time_traveler_chance = time_traveler_step }
	event_target:time_traveler_target_custodian = {
		if = {
			limit = { has_country_flag = time_traveler_sting_recent }
			subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_chance.infiltrate_sting_penalty }
		}
	}
	time_traveler_resolve_chance = yes
	if = {
		limit = { check_variable = { time_traveler_success = constant:time_traveler_value.one } }
		set_temp_variable = { time_traveler_op_result = constant:time_traveler_value.one }
		event_target:time_traveler_target_custodian = {
			set_country_flag = time_traveler_forgery_pending
			country_event = { id = chaosx.nr30.32 days = constant:time_traveler_timing.theft_report_days }
		}
		save_event_target_as = time_traveler_forger
	}
	else = {
		set_temp_variable = { time_traveler_op_result = constant:time_traveler_value.zero }
		subtract_from_variable = { time_traveler_network = constant:time_traveler_value.one }
		clamp_variable = { var = time_traveler_network min = constant:time_traveler_value.zero max = constant:time_traveler_network.mature }
		time_traveler_roll_percent = yes
		if = {
			limit = { check_variable = { time_traveler_roll < constant:time_traveler_chance.forge_exposed_base } }
			set_temp_variable = { time_traveler_op_result = constant:time_traveler_value.negative_one }
			event_target:time_traveler_target_custodian = { country_event = { id = chaosx.nr30.36 days = constant:time_traveler_timing.theft_report_days } }
		}
	}
	time_traveler_report_to_attacker = yes
}

# COUNTRY (attacker). Offer asylum. The traveler weighs his treatment, the
# stability of his host and the offering country.
time_traveler_operation_asylum = {
	var:global.time_traveler_custodian = { save_event_target_as = time_traveler_target_custodian }
	set_temp_variable = { time_traveler_op_kind = constant:time_traveler_operation.asylum }
	set_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_base }
	event_target:time_traveler_target_custodian = {
		if = {
			limit = { time_traveler_arrangement_is_confinement = yes }
			add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_confinement }
		}
		else_if = {
			limit = { time_traveler_arrangement_is_cabinet = yes }
			add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_cabinet }
		}
		else_if = {
			limit = { time_traveler_arrangement_is_secret = yes }
			add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_secret }
		}
		else = { add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_public } }
		if = {
			limit = { time_traveler_credibility_is_at_least_trusted = yes }
			subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_trusted_penalty }
		}
		if = {
			limit = { has_stability < constant:time_traveler_ai.stability_low }
			add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_unstable_bonus }
		}
	}
	if = {
		limit = { time_traveler_same_ideology_as_custodian = yes }
		add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_ideology_bonus }
	}
	if = {
		limit = { is_neighbor_of = var:global.time_traveler_custodian }
		add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.asylum_neighbor_bonus }
	}
	time_traveler_resolve_chance = yes
	if = {
		limit = { check_variable = { time_traveler_success = constant:time_traveler_value.one } }
		set_temp_variable = { time_traveler_op_result = constant:time_traveler_value.one }
		set_temp_variable = { time_traveler_transfer_method = constant:time_traveler_transfer.asylum }
		time_traveler_transfer_custody = yes
	}
	else = {

# COUNTRY (attacker). Seize the traveler. The chance falls with a guard detail
# and rises with the attacker's agency and proximity. A failed attempt may leave
# a captured double, a wounded traveler or an exposed sponsor. Assassination is
# chosen only when the sponsor is at war with the custodian.
time_traveler_operation_seize = {
	var:global.time_traveler_custodian = { save_event_target_as = time_traveler_target_custodian }
	set_country_flag = time_traveler_seize_tried
	set_temp_variable = { time_traveler_op_kind = constant:time_traveler_operation.seize }
	set_temp_variable = { time_traveler_chance = constant:time_traveler_chance.seize_base }
	if = {
		limit = { has_intelligence_agency = yes }
		add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.seize_agency_bonus }
	}
	if = {
		limit = { is_neighbor_of = var:global.time_traveler_custodian }
		add_to_temp_variable = { time_traveler_chance = constant:time_traveler_chance.seize_neighbor_bonus }
	}
	event_target:time_traveler_target_custodian = {
		if = {
			limit = { has_country_flag = time_traveler_guard_detail }
			subtract_from_temp_variable = { time_traveler_chance = constant:time_traveler_chance.seize_guard_penalty }
		}
	}
	time_traveler_resolve_chance = yes
	if = {
		limit = { check_variable = { time_traveler_success = constant:time_traveler_value.one } }
		time_traveler_roll_percent = yes
		if = {
			limit = {
				has_war_with = var:global.time_traveler_custodian
				check_variable = { time_traveler_roll < constant:time_traveler_chance.assassinate_threat }
			}
			set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.assassinated }
		}
		else_if = {
			limit = { check_variable = { time_traveler_roll < constant:time_traveler_chance.vanish_threat } }
			set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.vanished }
		}
		else = { set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.transferred } }
	}
	else = {
		time_traveler_roll_percent = yes
		set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.failed }
		if = {
			limit = { event_target:time_traveler_target_custodian = { has_country_flag = time_traveler_decoy_set } }
			set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.double }
		}
		else_if = {
			limit = { check_variable = { time_traveler_roll < constant:time_traveler_chance.seize_double_chance } }
			set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.exposed }
		}
		else_if = {
			limit = { check_variable = { time_traveler_roll < constant:time_traveler_chance.seize_wound_chance } }
			set_temp_variable = { time_traveler_seizure_outcome = constant:time_traveler_seizure.wounded }
		}
	}
	set_temp_variable = { time_traveler_op_result = time_traveler_seizure_outcome }
	time_traveler_apply_seizure = yes
	time_traveler_report_to_attacker = yes
}

			event_target:time_traveler_share_target = { tag = GER }
			NOT = { has_global_flag = time_traveler_german_variant_used }
			SOV = { exists = yes }
		}
		set_global_flag = time_traveler_german_variant_used
		GER = { country_event = { id = chaosx.nr30.3 days = constant:time_traveler_timing.report_delay_days } }
	}
	if = {
		limit = {
			event_target:time_traveler_share_target = { tag = USA }
			NOT = { has_global_flag = time_traveler_american_variant_used }
			JAP = { exists = yes }
		}
		set_global_flag = time_traveler_american_variant_used
		USA = { country_event = { id = chaosx.nr30.7 days = constant:time_traveler_timing.report_delay_days } }
	}
	time_traveler_refresh_costs = yes
}

# COUNTRY (a human rival). One operation per cooldown, so the decisions match the
# rhythm of an AI member.
time_traveler_set_operation_cooldown = {
	set_temp_variable = { time_traveler_days = constant:time_traveler_timing.operation_cooldown_days }
	set_country_flag = { flag = time_traveler_operation_cooldown days = time_traveler_days }
}

# COUNTRY (the winner of an annexation whose loser held the traveler). Custody
# passes to the winner when it is a country that can hold him and otherwise the
# traveler disappears. Called by on_annex with FROM as the annexed custodian.
time_traveler_custody_after_annexation = {
	if = {
		limit = {
			is_special_chaos_country = no
			is_actual_nonhuman_country = no
			exists = yes
		}
		set_temp_variable = { time_traveler_transfer_method = constant:time_traveler_transfer.negotiated }
		time_traveler_transfer_custody = yes
	}
	else = {
		FROM = { time_traveler_traveler_vanishes = yes }
	}
}
