calculate_economy_scaled_factory_grant = {
	# Input: economy_scaled_factory_grant_step, economy_scaled_factory_grant_min, economy_scaled_factory_grant_cap
	# Output: economy_scaled_factory_grant_count
	set_temp_variable = { economy_scaled_factory_grant_pool = num_of_civilian_factories }
	add_to_temp_variable = { economy_scaled_factory_grant_pool = num_of_military_factories }
	set_temp_variable = { economy_scaled_factory_grant_count = 0 }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = economy_scaled_factory_grant_pool
				value = economy_scaled_factory_grant_step
				compare = greater_than_or_equals
			}
			check_variable = {
				var = economy_scaled_factory_grant_count
				value = economy_scaled_factory_grant_cap
				compare = less_than
			}
		}
		add_to_temp_variable = { economy_scaled_factory_grant_count = 1 }
		subtract_from_temp_variable = { economy_scaled_factory_grant_pool = economy_scaled_factory_grant_step }
	}
	if = {
		limit = {
			check_variable = {
				var = economy_scaled_factory_grant_count
				value = economy_scaled_factory_grant_min
				compare = less_than
			}
		}
		set_temp_variable = { economy_scaled_factory_grant_count = economy_scaled_factory_grant_min }
	}
}

# Apply caller-selected building damage in the current state through meta text.
# Required temporary inputs: dynamic_building_damage_type (building token),
# dynamic_building_damage_amount (number). Inputs are read-only and have no defaults.
damage_state_building_dynamic = {
	meta_effect = {
		text = {
			damage_building = {
				type = [BUILDING_TYPE]
				damage = [BUILDING_DAMAGE]
			}
		}
		BUILDING_TYPE = "[?dynamic_building_damage_type.GetTokenKey]"
		BUILDING_DAMAGE = "[?dynamic_building_damage_amount]"
	}
}

# Configurable building and population damage for randomly selected controlled
# states. Callers may supply the four temporary inputs used below.
damage_buildings_in_random_states = {
	# Input: buildings_to_damage_per_state, percent_of_states_to_target, damage_modifier, state_population_percent

	# THIS refers to the country executing the script.

	# First, count the total number of states.
	set_temp_variable = { num_controlled_states = 0 }
	every_controlled_state = {
		add_to_temp_variable = { num_controlled_states = 1 }
	}

	set_temp_variable = { num_states_to_target = var:num_controlled_states }
	if = {
		limit = {
			check_variable = { percent_of_states_to_target = 0 }
		}
		set_temp_variable = { percent_of_states_to_target = 0.1 } }

		if = {
			limit = {
				check_variable = { num_states_to_target > 0 }
			} # In case target country even has any controlled states.
			# Calculate the number of states to target from the percentage.
			multiply_temp_variable = { num_states_to_target = percent_of_states_to_target }
			if = { limit = { check_variable = { num_states_to_target < 1 } }
			set_temp_variable = { num_states_to_target = 1 }
		}
		round_temp_variable = num_states_to_target
	}

	if = {
		limit = { check_variable = { damage_modifier = 0 } }
		set_temp_variable = { damage_modifier = 0.25 }
	}
	if = {
		limit = { check_variable = { buildings_to_damage_per_state < 1 } }
		set_temp_variable = { buildings_to_damage_per_state = 3 }
	}


	for_loop_effect = {
		end = var:num_states_to_target

		# Select a random controlled state that has at least one of the targetable buildings.
		random_owned_controlled_state = {
			limit = {
				OR = {
					infrastructure > 0
					arms_factory > 0
					industrial_complex > 0
					air_base > 0
					supply_node > 0
					rail_way > 0
					naval_base > 0
					bunker > 0
					coastal_bunker > 0
					dockyard > 0
					anti_air_building > 0
					synthetic_refinery > 0
					fuel_silo > 0
					radar_station > 0
					rocket_site > 0
					nuclear_reactor > 0
					nuclear_reactor_heavy_water > 0
					commercial_nuclear_reactor > 0
				}
			}


			# Optional proportional population loss. A missing or zero
			# state_population_percent leaves population unchanged.
			if = {
				limit = { check_variable = { state_population_percent > 0 } }
				set_temp_variable = { pop_loss = state_population_k }
				multiply_temp_variable = { pop_loss = state_population_percent }
				multiply_temp_variable = { pop_loss = constant:chaos_meter_deaths.people_per_k }
				multiply_temp_variable = { pop_loss = constant:chaos_meter_deaths.negative_one }
				round_temp_variable = pop_loss
				add_manpower = pop_loss
			}


			# 3. Loop and damage a random building each time.
			for_loop_effect = {
				end = buildings_to_damage_per_state # Temp variables work with any scope.

				meta_effect = {
					text = {
						random_list = {
							1 = {
								modifier = {
									factor = 0
									infrastructure < 1
								}
								damage_building = { type = infrastructure damage = [DMG] }
							}
							1 = {
								modifier = {
									factor = 0
									arms_factory < 1
								}
								damage_building = { type = arms_factory damage = [DMG] }
							}
							1 = {
								modifier = {

									industrial_complex < 1

	}
	if = {
		limit = {
			hidden_trigger = { has_global_flag = world_threat_source_sov_nuclear_bombs }

				OR = {
					has_global_flag = fallout_transition_active
					has_global_flag = fallout_active
					has_global_flag = world_end_final_silence
					has_global_flag = world_end_final_silence_completed

		limit = { check_variable = { global.world_threat_source_count > 0 } }
		set_global_flag = world_in_threat
	}
	else = {
		clr_global_flag = world_in_threat
	}
}

# Public country-scope additive technology union. See
# chaosx_dynamic_effects.md for the donor contract and compatibility rules.
union_compatible_researched_technologies_from_donor = {
	for_each_loop = {
		array = event_target:technology_union_donor.researched_techs
		value = technology_union_entry
		if = {
			limit = {
				NOT = { has_tech = var:technology_union_entry }
				if = {
					limit = { check_variable = { technology_union_entry = token:flexible_line } }
					NOT = { has_tech = streamlined_line }
				}
				else_if = {
					limit = { check_variable = { technology_union_entry = token:streamlined_line } }
					NOT = { has_tech = flexible_line }
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { technology_union_entry = token:concentrated_industry }
							check_variable = { technology_union_entry = token:concentrated_industry2 }
							check_variable = { technology_union_entry = token:concentrated_industry3 }
							check_variable = { technology_union_entry = token:concentrated_industry4 }
							check_variable = { technology_union_entry = token:concentrated_industry5 }
						}
					}
					NOT = { has_tech = dispersed_industry }
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { technology_union_entry = token:dispersed_industry }
							check_variable = { technology_union_entry = token:dispersed_industry2 }
							check_variable = { technology_union_entry = token:dispersed_industry3 }
							check_variable = { technology_union_entry = token:dispersed_industry4 }
							check_variable = { technology_union_entry = token:dispersed_industry5 }
						}
					}
					NOT = { has_tech = concentrated_industry }
				}
			}
			set_technology = {
				var:technology_union_entry = 1
				popup = no
			}
		}
	}
}

# Public state-scope exact civilian population transaction. See
# chaosx_dynamic_effects.md for the input/output contract.
apply_state_population_loss_without_recruitable_manpower_gain = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = state_population_transaction_contract_supplied }
				NOT = { check_variable = { state_population_transaction_contract_supplied > constant:chaos_meter_deaths.zero } }
				NOT = { has_variable = state_population_transaction_loss }
			}
		}
		set_temp_variable = { state_population_transaction_loss = constant:chaos_meter_deaths.zero }
	}
	clamp_temp_variable = {
		var = state_population_transaction_loss
		min = constant:chaos_meter_deaths.zero
	}
	round_temp_variable = state_population_transaction_loss

	set_temp_variable = { state_population_transaction_owner_before_k = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_controller_before_k = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_owner_gain = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_controller_gain = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_reconciled_gain = constant:chaos_meter_deaths.zero }

	OWNER = {
		set_temp_variable = { state_population_transaction_owner_before_k = manpower_k }
	}
	if = {
		limit = { NOT = { is_owned_and_controlled_by = OWNER } }
		CONTROLLER = {
			set_temp_variable = { state_population_transaction_controller_before_k = manpower_k }
		}
	}

	set_temp_variable = { state_population_transaction_negative = state_population_transaction_loss }
	multiply_temp_variable = { state_population_transaction_negative = constant:chaos_meter_deaths.negative_one }
	add_manpower = var:state_population_transaction_negative

	OWNER = {
		set_temp_variable = { state_population_transaction_owner_gain = manpower_k }
	}
	subtract_from_temp_variable = {
		state_population_transaction_owner_gain = state_population_transaction_owner_before_k
	}
	multiply_temp_variable = {
		state_population_transaction_owner_gain = constant:chaos_meter_deaths.people_per_k
	}
	round_temp_variable = state_population_transaction_owner_gain
	clamp_temp_variable = {
		var = state_population_transaction_owner_gain
		min = constant:chaos_meter_deaths.zero
		max = state_population_transaction_loss
	}
	if = {
		limit = { check_variable = { state_population_transaction_owner_gain > constant:chaos_meter_deaths.zero } }
		set_temp_variable = { state_population_transaction_owner_correction = state_population_transaction_owner_gain }
		multiply_temp_variable = { state_population_transaction_owner_correction = constant:chaos_meter_deaths.negative_one }
		OWNER = { add_manpower = var:state_population_transaction_owner_correction }

# Shared date-unit conversion. Date variables (global.date and stored copies)
# count one game hour as 0.00001, so a day count must be converted before it
# is added to a stored date, and a date difference must be converted back
# before it feeds a native days = field. Constants live in
# common/script_constants/chaosx_date_units_constants.txt.
#
# Temporary variables created inside a scripted effect may not survive its
# return, so the caller initializes the output variable before each call.

# ANY scope. Input: chaosx_date_days (day count, read only).
# Output: chaosx_date_offset (date-unit offset to add to a stored date).
# Fractional hours round up, so a due date never lands before the requested
# duration has passed. Whole-day inputs are exact.
chaosx_convert_days_to_date_offset = {
	set_temp_variable = { chaosx_date_offset = chaosx_date_days }
	multiply_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.hours_per_day }
	set_temp_variable = { chaosx_date_unrounded_hours = chaosx_date_offset }
	round_temp_variable = chaosx_date_offset
	if = {
		limit = { check_variable = { var = chaosx_date_offset value = chaosx_date_unrounded_hours compare = less_than } }
		add_to_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.hour_step }
	}
	divide_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.hours_per_date_unit }
}

# ANY scope. Input: chaosx_date_offset (a date difference such as a stored
# date minus global.date, read only).
# Output: chaosx_date_days (whole days, rounded up, never below zero), so a
# delayed event or timed effect that waits that many days resolves on or
# after the stored date.
chaosx_convert_date_offset_to_days = {
	set_temp_variable = { chaosx_date_days = chaosx_date_offset }
	multiply_temp_variable = { chaosx_date_days = constant:chaosx_date_units.hours_per_date_unit }
	divide_temp_variable = { chaosx_date_days = constant:chaosx_date_units.hours_per_day }
	set_temp_variable = { chaosx_date_unrounded_days = chaosx_date_days }
	round_temp_variable = chaosx_date_days
	if = {
		limit = { check_variable = { var = chaosx_date_days value = chaosx_date_unrounded_days compare = less_than } }
		add_to_temp_variable = { chaosx_date_days = constant:chaosx_date_units.day_step }
	}
	clamp_temp_variable = {
		var = chaosx_date_days
		min = constant:chaosx_date_units.zero
	}
}

# State scope. Native country-targeted application preserves every human value.
# Temporary inputs: human_state_modifier_request (modifier token) and
# human_state_modifier_requested_days (zero for permanent).
add_human_state_dynamic_modifier = {
	set_temp_variable = { human_state_modifier_active = human_state_modifier_request }
	set_temp_variable = { human_state_modifier_days = human_state_modifier_requested_days }
	if = {
		limit = { NOT = { check_variable = { human_state_modifier_active = constant:chaosx_date_units.zero } } }
		remove_human_state_dynamic_modifier = yes
		meta_effect = {
			text = {
				if = {
					limit = { state_uses_normal_civilian_systems = yes }
					set_variable = { human_state_modifier_[MODIFIER]_until = 0 }
					if = {
						limit = { check_variable = { human_state_modifier_days > 0 } }
						set_temp_variable = { chaosx_date_days = human_state_modifier_days }
						set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
						chaosx_convert_days_to_date_offset = yes
						set_variable = { human_state_modifier_[MODIFIER]_until = global.date }
						add_to_variable = { human_state_modifier_[MODIFIER]_until = chaosx_date_offset }
					}
					if = {
						limit = { NOT = { is_in_array = { array = global.human_state_modifier_states value = THIS } } }

						limit = { exists = yes uses_normal_civilian_systems = yes }
						set_temp_variable = { human_state_modifier_country = THIS }
						event_target:human_state_modifier_state = {
							if = {
								limit = { NOT = { is_in_array = { array = human_state_modifier_countries value = human_state_modifier_country } } }

								limit = { check_variable = { human_state_modifier_days > 0 } }

							}
							else = { add_dynamic_modifier = { modifier = [MODIFIER] scope = var:human_state_modifier_country } }
						}
					}
					force_update_dynamic_modifier = yes
				}
			}
			MODIFIER = "[?human_state_modifier_active.GetTokenKey]"
		}
	}
	set_temp_variable = { human_state_modifier_request = constant:chaosx_date_units.zero }
	set_temp_variable = { human_state_modifier_requested_days = constant:chaosx_date_units.zero }

# Public state-scope positive civilian population transaction, the neutral
# counterpart of the loss helper below. Input state_population_gain_request
# (people); output state_population_gain_applied (measured people) and
# state_population_gain_reconciled (recruitable credit removed). See
# chaosx_dynamic_effects.md for the contract.
apply_state_population_gain_without_recruitable_manpower_gain = {
	set_temp_variable = { state_population_gain_applied = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_gain_reconciled = constant:chaos_meter_deaths.zero }
	if = {
		limit = { NOT = { has_variable = state_population_gain_request } }
		set_temp_variable = { state_population_gain_request = constant:chaos_meter_deaths.zero }
	}
	clamp_temp_variable = { var = state_population_gain_request min = constant:chaos_meter_deaths.zero }
	round_temp_variable = state_population_gain_request
	if = {
		limit = { check_variable = { state_population_gain_request > constant:chaos_meter_deaths.zero } }
		set_temp_variable = { state_population_gain_before = state_population_k }
		multiply_temp_variable = { state_population_gain_before = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = state_population_gain_before
		set_temp_variable = { state_population_gain_owner_before_k = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_population_gain_controller_before_k = constant:chaos_meter_deaths.zero }
		OWNER = { set_temp_variable = { state_population_gain_owner_before_k = manpower_k } }
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = { set_temp_variable = { state_population_gain_controller_before_k = manpower_k } }
		}
		add_manpower = var:state_population_gain_request
		set_temp_variable = { state_population_gain_applied = state_population_k }
		multiply_temp_variable = { state_population_gain_applied = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = state_population_gain_applied
		subtract_from_temp_variable = { state_population_gain_applied = state_population_gain_before }
		clamp_temp_variable = {
			var = state_population_gain_applied
			min = constant:chaos_meter_deaths.zero
			max = state_population_gain_request
		}
		OWNER = {
			set_temp_variable = { state_population_gain_owner_credit = manpower_k }
			subtract_from_temp_variable = { state_population_gain_owner_credit = state_population_gain_owner_before_k }
			multiply_temp_variable = { state_population_gain_owner_credit = constant:chaos_meter_deaths.people_per_k }
			round_temp_variable = state_population_gain_owner_credit
			clamp_temp_variable = {
				var = state_population_gain_owner_credit
				min = constant:chaos_meter_deaths.zero
				max = state_population_gain_applied
			}
			if = {
				limit = { check_variable = { state_population_gain_owner_credit > constant:chaos_meter_deaths.zero } }
				set_temp_variable = { state_population_gain_owner_correction = state_population_gain_owner_credit }
				multiply_temp_variable = { state_population_gain_owner_correction = constant:chaos_meter_deaths.negative_one }
				add_manpower = var:state_population_gain_owner_correction
				add_to_temp_variable = { state_population_gain_reconciled = state_population_gain_owner_credit }
			}
		}
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = {
				set_temp_variable = { state_population_gain_controller_credit = manpower_k }
				subtract_from_temp_variable = { state_population_gain_controller_credit = state_population_gain_controller_before_k }
				multiply_temp_variable = { state_population_gain_controller_credit = constant:chaos_meter_deaths.people_per_k }
				round_temp_variable = state_population_gain_controller_credit
				clamp_temp_variable = {
					var = state_population_gain_controller_credit
					min = constant:chaos_meter_deaths.zero
					max = state_population_gain_applied
				}
				if = {
					limit = { check_variable = { state_population_gain_controller_credit > constant:chaos_meter_deaths.zero } }
					set_temp_variable = { state_population_gain_controller_correction = state_population_gain_controller_credit }
					multiply_temp_variable = { state_population_gain_controller_correction = constant:chaos_meter_deaths.negative_one }
					add_manpower = var:state_population_gain_controller_correction
					add_to_temp_variable = { state_population_gain_reconciled = state_population_gain_controller_credit }
				}
			}
		}
	}
	set_temp_variable = { state_population_gain_request = constant:chaos_meter_deaths.zero }
}
