calculate_economy_scaled_factory_grant = {
	# Input: economy_scaled_factory_grant_step, economy_scaled_factory_grant_min, economy_scaled_factory_grant_cap
	# Output: economy_scaled_factory_grant_count
	set_temp_variable = { economy_scaled_factory_grant_pool = num_of_civilian_factories }
	add_to_temp_variable = { economy_scaled_factory_grant_pool = num_of_military_factories }
	set_temp_variable = { economy_scaled_factory_grant_count = 0 }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = economy_scaled_factory_grant_pool
				value = economy_scaled_factory_grant_step
				compare = greater_than_or_equals
			}
			check_variable = {
				var = economy_scaled_factory_grant_count
				value = economy_scaled_factory_grant_cap
				compare = less_than
			}
		}
		add_to_temp_variable = { economy_scaled_factory_grant_count = 1 }
		subtract_from_temp_variable = { economy_scaled_factory_grant_pool = economy_scaled_factory_grant_step }
	}
	if = {
		limit = {
			check_variable = {
				var = economy_scaled_factory_grant_count
				value = economy_scaled_factory_grant_min
				compare = less_than
			}
		}
		set_temp_variable = { economy_scaled_factory_grant_count = economy_scaled_factory_grant_min }
	}
}

# Can define variables 'buildings_to_damage_per_state', 'percent_of_states_to_target', and 'damage_modifier' to control how many buildings are damaged per state, how many buildings to damage, and how much damage is applied. Also 'state_population_percent' to control how much of population % is killed in the selected random state(s).
damage_buildings_in_random_states = {
    # Input: buildings_to_damage_per_state, percent_of_states_to_target, damage_modifier, state_population_percent

    # THIS refers to the country executing the script.

    # First, count the total number of states.
    set_temp_variable = { num_controlled_states = 1 }
    every_controlled_state = {
        add_to_temp_variable = { num_controlled_states = 1 }
    }

    set_temp_variable = { num_states_to_target = var:num_controlled_states }
    if = {
        limit = {
            check_variable = { percent_of_states_to_target = 0 }
        }
        set_temp_variable = { percent_of_states_to_target = 0.1 } }

        if = {
            limit = {
                check_variable = { num_states_to_target > 0 }
            } # In case target country even has any controlled states.
            # caclulate the number of states to target based on the percentage.
            multiply_temp_variable = { num_states_to_target = percent_of_states_to_target }
            if = { limit = { check_variable = { num_states_to_target < 1 } }
            set_temp_variable = { num_states_to_target = 1 }
        }
        round_temp_variable = num_states_to_target
    }

    if = {
        limit = { check_variable = { damage_modifier = 0 } }
        set_temp_variable = { damage_modifier = 0.25 }
    }
    if = {
        limit = { check_variable = { buildings_to_damage_per_state < 1 } }
        set_temp_variable = { buildings_to_damage_per_state = 3 }
    }


    multiply_temp_variable = { state_population_percent = 1000 }
    if = {
        limit = { check_variable = { state_population_percent < 1 } }
        set_temp_variable = { state_population_percent = -1 }
    }

    for_loop_effect = {
        end = var:num_states_to_target

        # Select a random controlled state that has at least one of the targetable buildings.

            limit = {
                OR = {

            add_manpower = pop_loss


            # 3. Loop and damage a random building each time.
            for_loop_effect = {
                end = buildings_to_damage_per_state # Temp variables work with any scope.

                meta_effect = {
                    text = {
                        random_list = {

# Public country-scope additive technology union. See
# chaosx_dynamic_effects.md for the donor contract and compatibility rules.
union_compatible_researched_technologies_from_donor = {
	for_each_loop = {
		array = event_target:technology_union_donor.researched_techs
		value = technology_union_entry
		if = {
			limit = {
				NOT = { has_tech = var:technology_union_entry }
				if = {
					limit = { check_variable = { technology_union_entry = token:flexible_line } }
					NOT = { has_tech = streamlined_line }
				}
				else_if = {
					limit = { check_variable = { technology_union_entry = token:streamlined_line } }
					NOT = { has_tech = flexible_line }
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { technology_union_entry = token:concentrated_industry }
							check_variable = { technology_union_entry = token:concentrated_industry2 }
							check_variable = { technology_union_entry = token:concentrated_industry3 }
							check_variable = { technology_union_entry = token:concentrated_industry4 }
							check_variable = { technology_union_entry = token:concentrated_industry5 }
						}
					}
					NOT = { has_tech = dispersed_industry }
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { technology_union_entry = token:dispersed_industry }
							check_variable = { technology_union_entry = token:dispersed_industry2 }
							check_variable = { technology_union_entry = token:dispersed_industry3 }
							check_variable = { technology_union_entry = token:dispersed_industry4 }
							check_variable = { technology_union_entry = token:dispersed_industry5 }
						}
					}
					NOT = { has_tech = concentrated_industry }
				}
			}
			set_technology = {
				var:technology_union_entry = 1
				popup = no
			}
		}
	}
}

# Public state-scope exact civilian population transaction. See
# chaosx_dynamic_effects.md for the input/output contract.
apply_state_population_loss_without_recruitable_manpower_gain = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = state_population_transaction_contract_supplied }
				NOT = { check_variable = { state_population_transaction_contract_supplied > constant:chaos_meter_deaths.zero } }
				NOT = { has_variable = state_population_transaction_loss }
			}
		}
		set_temp_variable = { state_population_transaction_loss = constant:chaos_meter_deaths.zero }
	}
	clamp_temp_variable = {
		var = state_population_transaction_loss
		min = constant:chaos_meter_deaths.zero
	}
	round_temp_variable = state_population_transaction_loss

	set_temp_variable = { state_population_transaction_owner_before_k = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_controller_before_k = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_owner_gain = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_controller_gain = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_reconciled_gain = constant:chaos_meter_deaths.zero }

	OWNER = {
		set_temp_variable = { state_population_transaction_owner_before_k = manpower_k }
	}
	if = {
		limit = { NOT = { is_owned_and_controlled_by = OWNER } }
		CONTROLLER = {
			set_temp_variable = { state_population_transaction_controller_before_k = manpower_k }
		}
	}

	set_temp_variable = { state_population_transaction_negative = state_population_transaction_loss }
	multiply_temp_variable = { state_population_transaction_negative = constant:chaos_meter_deaths.negative_one }
	add_manpower = var:state_population_transaction_negative

	OWNER = {
		set_temp_variable = { state_population_transaction_owner_gain = manpower_k }
	}
	subtract_from_temp_variable = {
		state_population_transaction_owner_gain = state_population_transaction_owner_before_k
	}
	multiply_temp_variable = {
		state_population_transaction_owner_gain = constant:chaos_meter_deaths.people_per_k
	}
	round_temp_variable = state_population_transaction_owner_gain
	clamp_temp_variable = {
		var = state_population_transaction_owner_gain
		min = constant:chaos_meter_deaths.zero
		max = state_population_transaction_loss
	}
	if = {
		limit = { check_variable = { state_population_transaction_owner_gain > constant:chaos_meter_deaths.zero } }
		set_temp_variable = { state_population_transaction_owner_correction = state_population_transaction_owner_gain }
		multiply_temp_variable = { state_population_transaction_owner_correction = constant:chaos_meter_deaths.negative_one }
		OWNER = { add_manpower = var:state_population_transaction_owner_correction }
		add_to_temp_variable = { state_population_transaction_reconciled_gain = state_population_transaction_owner_gain }
	}

	if = {
		limit = { NOT = { is_owned_and_controlled_by = OWNER } }
		CONTROLLER = {
			set_temp_variable = { state_population_transaction_controller_gain = manpower_k }
		}
		subtract_from_temp_variable = {
			state_population_transaction_controller_gain = state_population_transaction_controller_before_k
		}
		multiply_temp_variable = {
			state_population_transaction_controller_gain = constant:chaos_meter_deaths.people_per_k
		}
		round_temp_variable = state_population_transaction_controller_gain
		clamp_temp_variable = {
			var = state_population_transaction_controller_gain
			min = constant:chaos_meter_deaths.zero
			max = state_population_transaction_loss
		}
		if = {
			limit = { check_variable = { state_population_transaction_controller_gain > constant:chaos_meter_deaths.zero } }
			set_temp_variable = { state_population_transaction_controller_correction = state_population_transaction_controller_gain }
			multiply_temp_variable = { state_population_transaction_controller_correction = constant:chaos_meter_deaths.negative_one }
			CONTROLLER = { add_manpower = var:state_population_transaction_controller_correction }
			add_to_temp_variable = { state_population_transaction_reconciled_gain = state_population_transaction_controller_gain }
		}
	}
	set_temp_variable = { state_population_transaction_contract_supplied = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_loss = constant:chaos_meter_deaths.zero }
}

apply_exact_state_civilian_population_loss = {
	set_temp_variable = { state_civilian_population_loss_applied = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_civilian_population_loss_result = constant:chaos_meter_deaths.zero }

	if = {
		limit = {
			OR = {
				NOT = { has_variable = state_civilian_population_loss_contract_supplied }
				NOT = { check_variable = { state_civilian_population_loss_contract_supplied > constant:chaos_meter_deaths.zero } }
				NOT = { has_variable = state_civilian_population_loss_requested }
				NOT = { has_variable = state_civilian_population_loss_minimum_remaining }
				NOT = { has_variable = state_civilian_population_loss_reason }
				NOT = { has_variable = state_civilian_population_loss_log_deaths }
				NOT = { has_variable = state_civilian_population_loss_target_country }
				NOT = { has_variable = state_civilian_population_loss_has_target_country }
			}
		}
		set_temp_variable = { state_civilian_population_loss_requested = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.unknown }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:chaos_meter_deaths.zero }
	}
	clamp_temp_variable = {
		var = state_civilian_population_loss_minimum_remaining
		min = constant:chaos_meter_deaths.zero
	}
	round_temp_variable = state_civilian_population_loss_minimum_remaining

	set_temp_variable = { state_civilian_population_loss_available = state_population_k }
	multiply_temp_variable = { state_civilian_population_loss_available = constant:chaos_meter_deaths.people_per_k }
	subtract_from_temp_variable = {
		state_civilian_population_loss_available = state_civilian_population_loss_minimum_remaining
	}
	clamp_temp_variable = {
		var = state_civilian_population_loss_available
		min = constant:chaos_meter_deaths.zero
	}

	set_temp_variable = { state_civilian_population_loss_applied = state_civilian_population_loss_requested }
	clamp_temp_variable = {
		var = state_civilian_population_loss_applied
		min = constant:chaos_meter_deaths.zero
		max = state_civilian_population_loss_available
	}
	round_temp_variable = state_civilian_population_loss_applied

	if = {
		limit = { check_variable = { state_civilian_population_loss_applied > constant:chaos_meter_deaths.zero } }
		set_temp_variable = { state_civilian_population_loss_result = constant:chaos_meter_deaths.one }
		if = {
			limit = {
				OR = {
					NOT = { has_global_flag = settings_chaos_deaths_disabled }
					has_global_flag = fallout_air_cleanliness_disabled
					has_global_flag = fallout_transition_active
					has_global_flag = fallout_active
					has_global_flag = fallout_manual_scenario_active
					has_global_flag = fallout_manual_sweep_active
					has_global_flag = fallout_manual_countdown_active
				}
				check_variable = { state_civilian_population_loss_log_deaths > constant:chaos_meter_deaths.zero }
			}
			set_temp_variable = { chaos_deaths_change = state_civilian_population_loss_applied }
			set_temp_variable = { chaos_deaths_reason = state_civilian_population_loss_reason }
			set_temp_variable = { chaos_deaths_is_civilian = constant:chaos_meter_deaths.one }
			set_temp_variable = { chaos_deaths_apply_state_pop = constant:chaos_meter_deaths.one }
			set_temp_variable = { chaos_deaths_target_country = state_civilian_population_loss_target_country }
			set_temp_variable = { chaos_deaths_has_target_country = state_civilian_population_loss_has_target_country }
			chaos_meter_register_deaths = yes
		}
		else = {
			set_temp_variable = { state_population_transaction_loss = state_civilian_population_loss_applied }
			set_temp_variable = { state_population_transaction_contract_supplied = constant:chaos_meter_deaths.one }
			apply_state_population_loss_without_recruitable_manpower_gain = yes
		}
	}

	set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_civilian_population_loss_requested = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.unknown }
	set_temp_variable = { state_civilian_population_loss_log_deaths = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
	set_temp_variable = { state_civilian_population_loss_has_target_country = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_civilian_population_loss_available = constant:chaos_meter_deaths.zero }
	set_temp_variable = { state_population_transaction_loss = constant:chaos_meter_deaths.zero }
}
