# COUNTRY scope. Input: temporary brilliant_scientist_super_event_id.
brilliant_scientist_queue_current_super_event = {
	add_to_array = {
		array = global.brilliant_scientist_super_event_queue_id_entries
		value = brilliant_scientist_super_event_id
	}
	add_to_array = {
		array = global.brilliant_scientist_super_event_queue_actor_entries
		value = ROOT
	}
	brilliant_scientist_try_dispatch_super_event_queue = yes
	brilliant_scientist_schedule_super_event_queue_retry = yes
}

brilliant_scientist_schedule_super_event_queue_retry = {
	if = {
		limit = {
			check_variable = { global.brilliant_scientist_super_event_queue_id_entries^num > constant:brilliant_scientist_value.zero }
			NOT = { has_global_flag = brilliant_scientist_super_event_queue_retry_pending }
		}
		set_global_flag = brilliant_scientist_super_event_queue_retry_pending
		set_temp_variable = { brilliant_scientist_super_event_retry_days = constant:brilliant_scientist_super_event_runtime.queue_retry_days }
		if = {
			limit = { any_country = { has_country_flag = is_global_host } }
			every_country = {
				limit = { has_country_flag = is_global_host }
				country_event = { id = chaosx.nr16.300 days = brilliant_scientist_super_event_retry_days }
			}
		}
		else_if = {
			limit = { exists = yes }
			country_event = { id = chaosx.nr16.300 days = brilliant_scientist_super_event_retry_days }
		}
	}
}

brilliant_scientist_schedule_super_event_actor_cleanup = {
	if = {
		limit = { NOT = { has_global_flag = brilliant_scientist_super_event_actor_cleanup_pending } }
		set_global_flag = brilliant_scientist_super_event_actor_cleanup_pending
		set_temp_variable = { brilliant_scientist_super_event_actor_cleanup_days = constant:brilliant_scientist_super_event_runtime.visible_days }
		if = {
			limit = { any_country = { has_country_flag = is_global_host } }
			every_country = {
				limit = { has_country_flag = is_global_host }
				country_event = { id = chaosx.nr16.902 days = brilliant_scientist_super_event_actor_cleanup_days }
			}
		}
		else_if = {
			limit = { exists = yes }
			country_event = { id = chaosx.nr16.902 days = brilliant_scientist_super_event_actor_cleanup_days }
		}
	}
}

brilliant_scientist_try_dispatch_super_event_queue = {
	if = {
		limit = {
			check_variable = { global.brilliant_scientist_super_event_queue_id_entries^num > constant:brilliant_scientist_value.zero }
			NOT = { has_global_flag = super_event_visible }
		}
		set_temp_variable = { brilliant_scientist_super_event_dispatch_id = global.brilliant_scientist_super_event_queue_id_entries^0 }
		if = {
			limit = { has_event_target = brilliant_scientist_super_event_actor }
			clear_global_event_target = brilliant_scientist_super_event_actor
		}
		clr_global_flag = brilliant_scientist_super_event_actor_cleanup_pending
		var:global.brilliant_scientist_super_event_queue_actor_entries^0 = {
			save_global_event_target_as = brilliant_scientist_super_event_actor
		}
		remove_from_array = { array = global.brilliant_scientist_super_event_queue_id_entries index = 0 }
		remove_from_array = { array = global.brilliant_scientist_super_event_queue_actor_entries index = 0 }
		set_temp_variable = { brilliant_scientist_super_event_visible_days = constant:brilliant_scientist_super_event_runtime.visible_days }
		meta_effect = {
			text = {
				set_global_flag = {
					flag = super_event_visible
					days = brilliant_scientist_super_event_visible_days
					value = [SUPER_EVENT_ID]
				}
			}
			SUPER_EVENT_ID = "[?brilliant_scientist_super_event_dispatch_id|.0]"
		}
		set_variable = { global.current_super_event_audio_id = brilliant_scientist_super_event_dispatch_id }
		every_country = {
			limit = { is_ai = no }
			play_current_super_event_audio = yes
		}
		brilliant_scientist_schedule_super_event_actor_cleanup = yes
	}
}

		limit = {
			has_event_target = brilliant_scientist_world_threat_opponent_input

				exists = yes
				NOT = { tag = ROOT }
			}
			NOT = {
				is_in_array = {
					array = brilliant_scientist_world_threat_opponent_entries
					value = event_target:brilliant_scientist_world_threat_opponent_input
				}
			}
		}
		add_to_array = {
			array = brilliant_scientist_world_threat_opponent_entries
			value = event_target:brilliant_scientist_world_threat_opponent_input
		}
		set_variable = { brilliant_scientist_world_threat_distinct_opponent_count = brilliant_scientist_world_threat_opponent_entries^num }

			limit = { event_target:brilliant_scientist_world_threat_opponent_input = { is_major = yes } }
			add_to_variable = { brilliant_scientist_world_threat_major_opponent_count = constant:brilliant_scientist_value.one }
		}
	}
	brilliant_scientist_refresh_world_threat_source = yes
}

# COUNTRY scope. This is the canonical event-driven hook used by project,

brilliant_scientist_refresh_world_threat_source = {
	if = {
		limit = { brilliant_scientist_is_kruger_sovereign_country = yes }
		brilliant_scientist_refresh_project_counts = yes
		brilliant_scientist_refresh_project_force_family_count = yes
		every_controlled_state = {
			set_state_flag = brilliant_scientist_krg_historical_territory
		}
		set_variable = { brilliant_scientist_world_threat_current_states = num_of_controlled_states }
		set_variable = { brilliant_scientist_world_threat_current_factories = num_of_factories }
		set_variable = { brilliant_scientist_world_threat_distinct_opponent_count = brilliant_scientist_world_threat_opponent_entries^num }
		if = {
			limit = { check_variable = { brilliant_scientist_world_threat_current_states > brilliant_scientist_world_threat_peak_states } }
			set_variable = { brilliant_scientist_world_threat_peak_states = brilliant_scientist_world_threat_current_states }
		}
		if = {
			limit = { check_variable = { brilliant_scientist_world_threat_current_factories > brilliant_scientist_world_threat_peak_factories } }
			set_variable = { brilliant_scientist_world_threat_peak_factories = brilliant_scientist_world_threat_current_factories }
		}
		if = {
			limit = { check_variable = { brilliant_scientist_project_weaponization_count > brilliant_scientist_world_threat_peak_weaponization_count } }
			set_variable = { brilliant_scientist_world_threat_peak_weaponization_count = brilliant_scientist_project_weaponization_count }
		}
		if = {
			limit = { is_major = yes }
			set_country_flag = brilliant_scientist_world_threat_was_major
		}
		if = {
			limit = {
				OR = {
					has_country_flag = brilliant_scientist_singularity_armed
					check_variable = {
						var = brilliant_scientist_singularity_component_count
						value = constant:brilliant_scientist_world_threat_threshold.late_singularity_components
						compare = greater_than_or_equals
					}
				}
			}
			set_country_flag = brilliant_scientist_singularity_late_construction_ever
		}
		brilliant_scientist_calculate_world_threat_score = yes
		brilliant_scientist_evaluate_formation_super_event = yes
		if = {
			limit = {
				brilliant_scientist_world_threat_activation_is_ready = yes
				NOT = { has_global_flag = world_threat_source_brilliant_scientist }
			}
			set_global_flag = world_threat_source_brilliant_scientist
			set_country_flag = brilliant_scientist_world_threat_active
			brilliant_scientist_refresh_directorate_idea_lifecycle = yes
			set_country_flag = brilliant_scientist_world_threat_ever_active
			set_global_flag = brilliant_scientist_world_threat_ever_active
			if = {
				limit = { NOT = { has_variable = brilliant_scientist_world_threat_start_date } }
				set_variable = { brilliant_scientist_world_threat_start_date = global.date }
			}
			save_global_event_target_as = brilliant_scientist_world_threat_actor
			if = {
				limit = { NOT = { has_global_flag = brilliant_scientist_super_event_global_threat_fired } }
				set_global_flag = brilliant_scientist_super_event_global_threat_fired
				set_variable = { brilliant_scientist_world_threat_reveal_date = global.date }
				if = {
					limit = { NOT = { has_global_flag = brilliant_scientist_news_global_containment_coalition_fired } }
					set_global_flag = brilliant_scientist_news_global_containment_coalition_fired
					news_event = { id = chaosx.nr16.309 days = constant:brilliant_scientist_news_runtime.delay_days }
				}
				set_temp_variable = { brilliant_scientist_super_event_id = constant:brilliant_scientist_super_event.global_kruger_threat }
				brilliant_scientist_queue_current_super_event = yes
			}
		}
		if = {
			limit = {
				has_global_flag = world_threat_source_brilliant_scientist
				brilliant_scientist_world_threat_should_clear = yes
			}
			clr_global_flag = world_threat_source_brilliant_scientist
			clr_country_flag = brilliant_scientist_world_threat_active
			brilliant_scientist_refresh_directorate_idea_lifecycle = yes
			set_variable = { brilliant_scientist_world_threat_end_date = global.date }
			if = {
				limit = { has_variable = brilliant_scientist_world_threat_start_date }
				set_variable = { brilliant_scientist_world_threat_duration_days = global.date }
				subtract_from_variable = { brilliant_scientist_world_threat_duration_days = brilliant_scientist_world_threat_start_date }
			}
		}
	}
	refresh_world_threat_state = yes
}

				}
