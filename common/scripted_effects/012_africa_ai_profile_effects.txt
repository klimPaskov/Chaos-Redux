# - africa_ai_refresh_profile_registry: compose every active profile layer and
#   return fourteen africa_ai_weight_* temps plus risk, partial, retry, layer IDs.

#   refreshes the actor registry and returns africa_ai_requested_family_weight.

#   through the existing africa_action_target event target for member/foreign AI.
#
# Call only from bounded action, focus, event, or AI-decision evaluation. This
# file registers no daily, weekly, monthly, or autonomous world iteration.

	add_to_temp_variable = { africa_ai_weight_accession = africa_ai_profile_input_accession }
	add_to_temp_variable = { africa_ai_weight_regional_congress = africa_ai_profile_input_regional_congress }
	add_to_temp_variable = { africa_ai_weight_integration = africa_ai_profile_input_integration }
	add_to_temp_variable = { africa_ai_weight_economy = africa_ai_profile_input_economy }
	add_to_temp_variable = { africa_ai_weight_diaspora = africa_ai_profile_input_diaspora }
	add_to_temp_variable = { africa_ai_weight_rival_bloc = africa_ai_profile_input_rival_bloc }
	add_to_temp_variable = { africa_ai_weight_high_chaos = africa_ai_profile_input_high_chaos }
	add_to_temp_variable = { africa_ai_weight_scramble = africa_ai_profile_input_scramble }
	add_to_temp_variable = { africa_ai_weight_world_order = africa_ai_profile_input_world_order }
	add_to_temp_variable = { africa_ai_weight_constitutional_crisis = africa_ai_profile_input_constitutional_crisis }
	add_to_temp_variable = { africa_ai_weight_post_unification = africa_ai_profile_input_post_unification }
	add_to_temp_variable = { africa_ai_weight_host_opening = africa_ai_profile_input_host_opening }
	add_to_temp_variable = { africa_ai_weight_regional_restorations = africa_ai_profile_input_regional_restorations }
	if = {
		limit = {
			check_variable = {
				var = africa_ai_profile_input_risk_ceiling
				value = africa_ai_risk_ceiling
				compare = greater_than
			}
		}
		set_temp_variable = { africa_ai_risk_ceiling = africa_ai_profile_input_risk_ceiling }
	}
	if = {
		limit = {
			check_variable = {
				var = africa_ai_profile_input_partial_tolerance
				value = africa_ai_partial_tolerance
				compare = greater_than
			}
		}
		set_temp_variable = { africa_ai_partial_tolerance = africa_ai_profile_input_partial_tolerance }
	}
	if = {
		limit = {
			check_variable = {
				var = africa_ai_profile_input_retry_stance
				value = africa_ai_retry_stance
				compare = greater_than
			}
		}
		set_temp_variable = { africa_ai_retry_stance = africa_ai_profile_input_retry_stance }
	}
	add_to_temp_variable = { africa_ai_active_profile_count = constant:africa_value.one }
	set_temp_variable = { africa_ai_last_applied_profile = africa_ai_profile_input_id }
}

# COUNTRY scope on the host with africa_action_target set. Country targets are
# copied into the existing selected-country cursor and this helper then builds
# a capped state/region cursor from that country's controlled African states.
# It never selects a substitute state, changes ownership, or scans outside the
# already selected country target.
africa_ai_prepare_action_state_cursor = {
	if = {
		limit = { check_variable = { africa_requested_action_id = constant:africa_action.administer_world_regions } }
		africa_ai_prepare_world_region_state_cursor = yes
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { africa_profile_target_mode = constant:africa_action_target_mode.state }
				check_variable = { africa_profile_target_mode = constant:africa_action_target_mode.region }
			}
		}
		africa_clear_selected_action_state_context = yes
		if = {
			limit = { check_variable = { africa_profile_target_mode = constant:africa_action_target_mode.state } }
			for_each_scope_loop = {
				array = africa_selected_country_targets
				every_controlled_state = {
					limit = {
						is_on_continent = africa
						NOT = { has_state_flag = africa_action_project_active }
						ROOT = { check_variable = { africa_selected_action_state_targets^num < constant:africa_value.one } }
					}
					set_state_flag = africa_selected_action_state_candidate
					set_state_flag = africa_selected_action_state
					ROOT = {
						add_to_array = { array = africa_selected_action_state_candidates value = PREV }
						add_to_array = { array = africa_selected_action_state_targets value = PREV }
					}
				}
			}
		}
		else = {
			for_each_scope_loop = {
				array = africa_selected_country_targets
				every_controlled_state = {
					limit = {
						is_on_continent = africa
						NOT = { has_state_flag = africa_action_project_active }
						ROOT = { check_variable = { africa_selected_action_state_targets^num < constant:africa_action_capacity.maximum_region_state_targets } }
					}
					set_state_flag = africa_selected_action_state_candidate
					set_state_flag = africa_selected_action_state
					ROOT = {
						add_to_array = { array = africa_selected_action_state_candidates value = PREV }
						add_to_array = { array = africa_selected_action_state_targets value = PREV }
					}
				}
			}
		}
	}
	else = {
		africa_clear_selected_action_state_context = yes
	}
}

# COUNTRY scope on a foreign participant before africa_world_order.1 options.
# This pre-response classifier uses the same five registry IDs without writing
# a diplomatic disposition before the event choice is made.
africa_ai_classify_scramble_response = {
	set_variable = { africa_ai_scramble_pre_response_profile = constant:africa_ai_profile.none }
	if = {
		limit = { is_ai = yes }
		if = {
			limit = {
				has_country_flag = africa_scramble_colonial_interest
				is_faction_leader = yes
				africa_ai_scramble_expedition_materially_ready = yes
			}
			set_variable = { africa_ai_scramble_pre_response_profile = constant:africa_ai_profile.power_expedition_coalition }
		}
		else_if = {
			limit = {
				has_country_flag = africa_scramble_foreign_base_holder
				OR = {
					has_war_support < constant:africa_ai_policy.low_war_support
					has_stability < constant:africa_ai_policy.low_stability
				}
			}
			set_variable = { africa_ai_scramble_pre_response_profile = constant:africa_ai_profile.power_negotiated_withdrawal }
		}
		else_if = {
			limit = { has_country_flag = africa_scramble_colonial_interest }
			set_variable = { africa_ai_scramble_pre_response_profile = constant:africa_ai_profile.power_containment_sanctions }
		}
		else_if = {
			limit = { has_country_flag = africa_scramble_recognition_candidate }
			set_variable = { africa_ai_scramble_pre_response_profile = constant:africa_ai_profile.power_recognition_partner }
		}
		else = {
			set_variable = { africa_ai_scramble_pre_response_profile = constant:africa_ai_profile.power_opportunist_patron }
		}
		set_country_flag = africa_ai_scramble_pre_response_classified
	}
}
