# ============================================================================
# CHAOS REDUX - EVENT 026 BLACK FRIDAY NATIVE SALE EFFECTS
# ============================================================================
# Overview:
# - Applies and removes the black_friday_native_sale dynamic modifier on one
#   country (the country scope these effects run in).
# - black_friday_native_sale_apply stores one contribution variable per native
#   cost key, adds the modifier when missing, and forces a modifier update so
#   prices change at once. It is safe to call again on the same day: the
#   ordinary factor is read as modifier@<key> minus the stored contribution
#   that is already applied.
# - black_friday_native_sale_remove removes the modifier and clears every
#   stored contribution.
#
# Contribution per key, with r = global.black_friday_discount_ratio / ratio_basis:
#   ordinary     = modifier@key - black_friday_sale_<key> + linked ordinary
#   contribution = (r - 1) x max(0, 1 + ordinary)
# The engine adds the contribution to the ordinary factor, so the price becomes
# base x r x (1 + ordinary): the sale multiplies the current price.
#
# Linked ordinary: when another key can also price the same purchase, the most
# negative of those keys (never above zero) joins the ordinary factor. The
# wiki Modifiers page lists <ledger>_advisor_cost_factor beside the slot cost
# factors, idea_tags slot_ledgers fixes the chief and designer ledgers, and
# the vanilla political advisors that carry a ledger all use the army ledger.
# With no linked discount the result is exact; with one the sale is computed
# from the deeper discount, so no purchase is priced below its sale price or
# becomes free under those unchanged factors, whether the engine adds or
# multiplies the two keys. Other cost-factor changes after this computation
# require another explicit refresh and remain a native acceptance blocker.
#
# The key list mirrors common/dynamic_modifiers/026_black_friday_dynamic_modifiers.txt.
# Keep both files in step when a key is added or removed.
# ============================================================================

black_friday_native_sale_apply = {
	if = {
		limit = {
			has_country_flag = black_friday_native_sale_stored
			NOT = { has_dynamic_modifier = black_friday_native_sale }
		}
		black_friday_native_sale_clear_contributions = yes
	}
	set_temp_variable = { black_friday_native_sale_offset = global.black_friday_discount_ratio }
	divide_temp_variable = { black_friday_native_sale_offset = constant:black_friday_event.ratio_basis }
	subtract_from_temp_variable = { black_friday_native_sale_offset = constant:black_friday_event.one }
	# Caller-owned outputs must survive the linked-factor helper's return.
	set_temp_variable = { black_friday_native_army_ledger_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_navy_ledger_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_air_ledger_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_any_ledger_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_military_leader_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_army_spirit_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_navy_spirit_min = constant:black_friday_event.zero }
	set_temp_variable = { black_friday_native_air_spirit_min = constant:black_friday_event.zero }
	black_friday_native_sale_read_linked_ordinaries = yes
	black_friday_native_sale_store_contributions = yes
	set_country_flag = black_friday_native_sale_stored
	if = {
		limit = { NOT = { has_dynamic_modifier = black_friday_native_sale } }
		add_dynamic_modifier = { modifier = black_friday_native_sale }
	}
	force_update_dynamic_modifier = yes
}

black_friday_native_sale_remove = {
	if = {
		limit = { has_dynamic_modifier = black_friday_native_sale }
		remove_dynamic_modifier = { modifier = black_friday_native_sale }
	}
	if = {
		limit = { has_country_flag = black_friday_native_sale_stored }
		black_friday_native_sale_clear_contributions = yes
		clr_country_flag = black_friday_native_sale_stored
	}
}

# Turns the ordinary factor in black_friday_native_cost into the sale
# contribution. Requires black_friday_native_sale_offset (r - 1).
black_friday_native_sale_contribution = {
	add_to_temp_variable = { black_friday_native_cost = constant:black_friday_event.one }
	clamp_temp_variable = { var = black_friday_native_cost min = constant:black_friday_event.zero }
	multiply_temp_variable = { black_friday_native_cost = black_friday_native_sale_offset }
}
