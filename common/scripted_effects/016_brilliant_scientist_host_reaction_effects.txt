# COUNTRY scope. Schedules the first primary-facility compact report.
brilliant_scientist_try_schedule_primary_facility_reaction = {
	if = {
		limit = {
			brilliant_scientist_is_current_host = yes
			check_variable = {
				var = brilliant_scientist_breakthrough_resolved_count
				value = constant:brilliant_scientist_value.one
				compare = equals
			}
			NOT = { has_country_flag = brilliant_scientist_reaction_facility_pending }
			NOT = { has_country_flag = brilliant_scientist_reaction_facility_resolved }
			KRG_warren_kruger = { NOT = { has_character_flag = brilliant_scientist_personal_reaction_facility_resolved } }
			has_event_target = brilliant_scientist_primary_facility
			event_target:brilliant_scientist_primary_facility = {
				is_owned_by = ROOT
				has_state_flag = brilliant_scientist_primary_facility
			}
		}
		set_country_flag = brilliant_scientist_reaction_facility_pending
		KRG_warren_kruger = { set_character_flag = brilliant_scientist_personal_reaction_facility_pending }
		country_event = { id = chaosx.nr16.7 days = constant:brilliant_scientist_host_reaction_delay.facility_days }
	}
}

# COUNTRY scope on a resolved-operation host while regular actor and host
# targets are still valid. This does not scan countries or create a substitute
# target; the foreign-resolution transaction owns both scopes.
brilliant_scientist_try_fire_foreign_reaction = {
	if = {
		limit = {
			brilliant_scientist_is_current_host = yes
			check_variable = { brilliant_scientist_breakthrough_resolved_count > constant:brilliant_scientist_value.zero }
			has_event_target = brilliant_scientist_foreign_operation_actor
			has_event_target = brilliant_scientist_foreign_operation_host
			event_target:brilliant_scientist_foreign_operation_actor = {
				check_variable = { brilliant_scientist_foreign_last_operation_detected > constant:brilliant_scientist_value.zero }
				NOT = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.counter_program } }
			}
			NOT = { has_country_flag = brilliant_scientist_reaction_foreign_pending }
			NOT = { has_country_flag = brilliant_scientist_reaction_foreign_resolved }
			KRG_warren_kruger = { NOT = { has_character_flag = brilliant_scientist_personal_reaction_foreign_resolved } }
		}
		set_variable = { brilliant_scientist_reaction_foreign_operation = event_target:brilliant_scientist_foreign_operation_actor.brilliant_scientist_foreign_live_operation }
		set_country_flag = brilliant_scientist_reaction_foreign_pending
		KRG_warren_kruger = { set_character_flag = brilliant_scientist_personal_reaction_foreign_pending }
		event_target:brilliant_scientist_foreign_operation_host = {
			country_event = { id = chaosx.nr16.9 days = constant:brilliant_scientist_host_reaction_delay.foreign_days }
		}
	}
}

# COUNTRY scope on a new ordinary-transfer carrier. Resolved consequences stay
# with the old host and its physical state; the fixed character receipt blocks
# replay. Only unresolved `.7` or `.8` ownership and the `.8` family snapshot

			}
		}
		set_variable = { brilliant_scientist_reaction_custody_family = event_target:brilliant_scientist_transfer_old_host.brilliant_scientist_reaction_custody_family }
	}
	if = {
		limit = {
			event_target:brilliant_scientist_transfer_old_host = { has_country_flag = brilliant_scientist_reaction_facility_pending }
			NOT = { has_country_flag = brilliant_scientist_reaction_facility_resolved }
			KRG_warren_kruger = { NOT = { has_character_flag = brilliant_scientist_personal_reaction_facility_resolved } }
			has_event_target = brilliant_scientist_primary_facility
			event_target:brilliant_scientist_primary_facility = { is_owned_by = ROOT has_state_flag = brilliant_scientist_primary_facility }
		}
		set_country_flag = brilliant_scientist_reaction_facility_pending
		KRG_warren_kruger = { set_character_flag = brilliant_scientist_personal_reaction_facility_pending }
		country_event = { id = chaosx.nr16.7 days = constant:brilliant_scientist_host_reaction_delay.facility_days }
	}
	if = {
		limit = {
			event_target:brilliant_scientist_transfer_old_host = { has_country_flag = brilliant_scientist_reaction_custody_pending }
			NOT = { has_country_flag = brilliant_scientist_reaction_custody_resolved }
			KRG_warren_kruger = { NOT = { has_character_flag = brilliant_scientist_personal_reaction_custody_resolved } }
			has_variable = brilliant_scientist_reaction_custody_family
		}
		set_country_flag = brilliant_scientist_reaction_custody_pending
		KRG_warren_kruger = { set_character_flag = brilliant_scientist_personal_reaction_custody_pending }
		country_event = { id = chaosx.nr16.8 days = constant:brilliant_scientist_host_reaction_delay.custody_days }
	}
}
