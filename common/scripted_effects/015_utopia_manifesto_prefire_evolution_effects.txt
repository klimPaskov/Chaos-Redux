utopia_manifesto_select_prefire_district_candidate = {
	if = {
		limit = { NOT = { utopia_manifesto_prefire_has_live_district_candidate = yes } }
		clear_array = utopia_manifesto_prefire_district_candidate_states
		random_owned_controlled_state = {
			limit = {
				is_core_of = ROOT
				NOT = { has_state_flag = utopia_manifesto_district_built }
				NOT = { has_state_flag = utopia_manifesto_district_build_active }
			}
			ROOT = {
				add_to_array = { array = utopia_manifesto_prefire_district_candidate_states value = PREV }
				if = {
					limit = { PREV = { infrastructure > constant:utopia_manifesto_prefire_threshold.stable_candidate_infrastructure } }
					set_country_flag = utopia_manifesto_prefire_district_candidate_infrastructured
					set_temp_variable = { utopia_manifesto_plenty_delta = constant:utopia_manifesto_prefire_change.infrastructure_candidate_plenty }
				}
				else = {
					set_country_flag = utopia_manifesto_prefire_district_candidate_emergency
					set_temp_variable = { utopia_manifesto_need_delta = constant:utopia_manifesto_prefire_change.emergency_candidate_need }
				}
				utopia_manifesto_apply_prepared_ledger_delta = yes
			}
		}
	}
}

utopia_manifesto_apply_prefire_route_resolved_effects = {
	if = {
		limit = {
			utopia_manifesto_has_resolved_route = yes
			utopia_manifesto_prefire_shores_package_is_live = yes
		}
		set_temp_variable = { utopia_manifesto_decision_phase_input = constant:utopia_manifesto_decision_phase.necessary_ground }
		utopia_manifesto_set_decision_phase = yes
		utopia_manifesto_unlock_necessary_ground_decisions = yes
		utopia_manifesto_schedule_prefire_shores_survey_report = yes
	}
	if = {
		limit = {
			utopia_manifesto_has_resolved_route = yes
			utopia_manifesto_prefire_cities_package_is_live = yes
		}
		set_temp_variable = { utopia_manifesto_decision_phase_input = constant:utopia_manifesto_decision_phase.districts }
		utopia_manifesto_set_decision_phase = yes
		utopia_manifesto_unlock_district_decisions = yes
	}
	if = {
		limit = {
			utopia_manifesto_has_resolved_route = yes
			utopia_manifesto_prefire_nowhere_package_is_live = yes
			has_country_flag = utopia_manifesto_prefire_foreign_support_accepted
			NOT = { has_country_flag = utopia_manifesto_prefire_support_route_accounted }
		}
		set_country_flag = utopia_manifesto_prefire_support_route_accounted
		if = {
			limit = {
				OR = {
					utopia_manifesto_is_consent_of_households_route = yes
					utopia_manifesto_is_joke_understood_route = yes
				}
			}
			set_country_flag = utopia_manifesto_prefire_support_publicly_accounted
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_prefire_change.annotated_debate_concord }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		else_if = {
			limit = { utopia_manifesto_is_closed_island_route = yes }
			set_country_flag = utopia_manifesto_prefire_support_recast_as_dependency
			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_prefire_change.foreign_support_assignment }
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_prefire_change.foreign_scrutiny_concord }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		else = {
			set_country_flag = utopia_manifesto_prefire_support_administered
			set_temp_variable = { utopia_manifesto_plenty_delta = constant:utopia_manifesto_prefire_change.infrastructure_candidate_plenty }
			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_prefire_change.foreign_support_assignment }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
	}
	utopia_manifesto_refresh_prefire_project_deadlines = yes
}
