independence_wave_refresh_dahomey_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_dahomey_package = yes }
		remove_ideas = { dah_fragmented_coastal_mandate dah_dahomey_compact }
		if = {
			limit = { has_stable_independence_wave_dahomey_compact = yes }
			add_ideas = dah_dahomey_compact
			set_country_flag = independence_wave_dah_compact_stabilized
		}
		else = {
			add_ideas = dah_fragmented_coastal_mandate
			clr_country_flag = independence_wave_dah_compact_stabilized
		}
	}
}
