independence_wave_iw101_kongo_refresh_ideas = {
	if = {
		limit = { is_independence_wave_iw101_kongo_overlay_active = yes }
		independence_wave_iw101_kongo_remove_ideas = yes

			limit = {
				check_variable = { var = independence_wave_iw101_kongo_river_logistics value = constant:independence_wave_iw101_kongo_value.settlement_threshold compare = greater_than_or_equals }

independence_wave_iw101_kongo_initialize = {
	if = {
		limit = { is_independence_wave_iw101_kongo_route_active = yes NOT = { has_country_flag = independence_wave_iw101_kongo_overlay_ever_activated } }
		set_country_flag = independence_wave_iw101_kongo_overlay_ever_activated
		set_country_flag = independence_wave_iw101_kongo_package_initialized
		set_country_flag = independence_wave_iw101_kongo_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		set_variable = { independence_wave_iw101_kongo_river_logistics = constant:independence_wave_iw101_kongo_value.river_logistics_start }
		set_variable = { independence_wave_iw101_kongo_regency_legitimacy = constant:independence_wave_iw101_kongo_value.regency_legitimacy_start }
		set_variable = { independence_wave_iw101_kongo_veteran_integration = constant:independence_wave_iw101_kongo_value.veteran_integration_start }
		set_variable = { independence_wave_iw101_kongo_guard_hold_days = constant:independence_wave_iw101_kongo_value.minimum }
		set_variable = { independence_wave_anchor_state = 538 }
		set_variable = { independence_wave_package_archetype = constant:independence_wave_package_archetype.river_or_corridor }
		set_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.river_jungle }

	}

independence_wave_iw101_kongo_suspend = {
	if = {
		limit = { has_country_flag = independence_wave_iw101_kongo_overlay_active }
		if = { limit = { has_country_flag = independence_wave_iw101_kongo_guard_running NOT = { has_country_flag = independence_wave_iw101_kongo_guard_completed } } set_country_flag = independence_wave_iw101_kongo_guard_interrupted }
		independence_wave_iw101_kongo_remove_ideas = yes
		clr_country_flag = independence_wave_iw101_kongo_overlay_active
		clr_country_flag = independence_wave_route_overlay_active
		set_country_flag = independence_wave_iw101_kongo_overlay_suspended
	}
}

independence_wave_iw101_kongo_pause = {
	if = {
		limit = { has_country_flag = independence_wave_iw101_kongo_overlay_suspended has_country_flag = independence_wave_iw101_kongo_guard_interrupted has_country_flag = independence_wave_iw101_kongo_guard_running has_active_mission = independence_wave_iw101_hold_kongo_river_corridor }
		set_temp_variable = { independence_wave_iw101_kongo_pause_days = constant:independence_wave_iw_cog_overlay_duration.guard_pause_daily_extension }
		add_days_mission_timeout = { mission = independence_wave_iw101_hold_kongo_river_corridor days = independence_wave_iw101_kongo_pause_days }
	}
}

independence_wave_iw101_kongo_resume = {
	if = {
		limit = { is_independence_wave_iw101_kongo_route_active = yes has_country_flag = independence_wave_iw101_kongo_overlay_ever_activated NOT = { has_country_flag = independence_wave_iw101_kongo_overlay_active } }
		set_country_flag = independence_wave_iw101_kongo_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw101_kongo_overlay_suspended
		independence_wave_iw101_kongo_refresh_ideas = yes
	}
}

independence_wave_iw101_kongo_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw101_kongo_route_active = yes }
		if = { limit = { NOT = { has_country_flag = independence_wave_iw101_kongo_overlay_ever_activated } } independence_wave_iw101_kongo_initialize = yes }
		else_if = { limit = { NOT = { has_country_flag = independence_wave_iw101_kongo_overlay_active } } independence_wave_iw101_kongo_resume = yes }
		else = {
			if = { limit = { has_country_flag = independence_wave_iw101_kongo_guard_interrupted } if = { limit = { has_country_flag = independence_wave_iw101_kongo_guard_running has_active_mission = independence_wave_iw101_hold_kongo_river_corridor } clr_country_flag = independence_wave_iw101_kongo_guard_interrupted } else = { independence_wave_iw101_kongo_fail_guard = yes } }
			else_if = { limit = { has_country_flag = independence_wave_iw101_kongo_guard_running NOT = { has_active_mission = independence_wave_iw101_hold_kongo_river_corridor } } independence_wave_iw101_kongo_fail_guard = yes }
			independence_wave_iw101_kongo_update_guard = yes
			if = { limit = { NOT = { OR = { has_idea = independence_wave_iw101_kongo_contested_river_regency has_idea = independence_wave_iw101_kongo_integrated_river_authority has_idea = independence_wave_iw101_kongo_assembly_compact has_idea = independence_wave_iw101_kongo_veteran_charter } } } independence_wave_iw101_kongo_refresh_ideas = yes }
		}
	}
	else = { if = { limit = { has_country_flag = independence_wave_iw101_kongo_overlay_active } independence_wave_iw101_kongo_suspend = yes } independence_wave_iw101_kongo_pause = yes }
}
