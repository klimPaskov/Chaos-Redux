# ============================================================================
# CHAOS REDUX - EVENT 018 RESOURCES FOUND EFFECTS
# ============================================================================
# Reusable core for persistent resource fields. State scope owns the physical
# field record, six exact event-resource ledgers, field values, contracts, and
# cave-capacity data. Country scope owns selected-field pointers, owned-field
# arrays, and exact terminal-verification inputs.
#
# Package effects are atomic: they acquire a state transaction flag, mutate the
# resource and field ledgers, refresh derived values, then release the flag.
# Full sealing subtracts only the six Event 018 ledgers; resources that existed
# before Event 018, or were added by another system, are never inferred away.
# ============================================================================

# ---------------------------------------------------------------------------
# Transaction, ownership, and field-record helpers
# ---------------------------------------------------------------------------

# STATE scope. Output temp: resources_found_transaction_acquired (0 or 1).
resources_found_begin_field_transaction = {
	set_temp_variable = { resources_found_transaction_acquired = constant:resources_found_value.zero }
	if = {
		limit = { resources_found_field_transaction_available = yes }
		set_state_flag = resources_found_field_transaction_lock
		set_temp_variable = { resources_found_transaction_acquired = constant:resources_found_value.one }
	}
}

# STATE scope.
resources_found_end_field_transaction = {
	clr_state_flag = resources_found_field_transaction_lock
}

# STATE scope, internal. Releases only a lock acquired by the current caller.
resources_found_release_acquired_field_transaction = {
	if = {
		limit = { check_variable = { resources_found_transaction_acquired = constant:resources_found_value.one } }
		resources_found_end_field_transaction = yes
		set_temp_variable = { resources_found_transaction_acquired = constant:resources_found_value.zero }
	}
}

# COUNTRY scope. Clears an invalid selected-field pointer, then selects one
# owned active field if possible. The owned-fields array remains the complete
# registry; the pointer is only the current UI/action selection.
resources_found_validate_country_field_selection = {
	if = {
		limit = {
			has_variable = resources_found_selected_field
			NOT = { resources_found_country_has_valid_selected_field = yes }
		}
		clear_variable = resources_found_selected_field
	}
	if = {
		limit = { NOT = { has_variable = resources_found_selected_field } }
		every_owned_state = {
			limit = {
				resources_found_is_active_field = yes
				ROOT = { NOT = { has_variable = resources_found_selected_field } }
			}
			ROOT = { set_variable = { resources_found_selected_field = PREV } }
		}
	}
}

# STATE scope. Keeps the identity shown by a sealed-field history record current
# without making that state active, selectable for projects, or rediscoverable.
resources_found_refresh_closed_field_identity = {
	if = {
		limit = { has_state_flag = resources_found_field_closed_permanently }
		if = { limit = { OWNER = { exists = yes } } set_variable = { resources_found_recorded_owner = OWNER } }
		if = { limit = { CONTROLLER = { exists = yes } } set_variable = { resources_found_recorded_controller = CONTROLLER } }
	}
}

# STATE scope. Registers THIS with its current owner and selects it when the
# owner's previous selection is absent or invalid.
resources_found_bind_field_to_owner = {
	save_event_target_as = resources_found_binding_state
	if = {
		limit = { OWNER = { exists = yes } }
		set_variable = { resources_found_recorded_owner = OWNER }
		OWNER = {
			set_country_flag = resources_found_field_system_participant
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = resources_found_owned_fields
							value = event_target:resources_found_binding_state
						}
					}
				}
				add_to_array = {
					array = resources_found_owned_fields
					value = event_target:resources_found_binding_state
				}
			}
			if = {
				limit = { NOT = { resources_found_country_has_valid_selected_field = yes } }
				set_variable = { resources_found_selected_field = event_target:resources_found_binding_state }
			}
		}
	}
}

# COUNTRY scope. Requires regular event target resources_found_unbinding_state.
resources_found_remove_field_from_country_registry = {
	if = {
		limit = { has_event_target = resources_found_unbinding_state }
		if = {
			limit = {
				is_in_array = {
					array = resources_found_owned_fields
					value = event_target:resources_found_unbinding_state
				}
			}
			remove_from_array = {
				array = resources_found_owned_fields
				value = event_target:resources_found_unbinding_state
			}
		}
		if = {
			limit = {
				has_variable = resources_found_selected_field
				check_variable = {
					var = resources_found_selected_field
					value = event_target:resources_found_unbinding_state
					compare = equals
				}
			}
			clear_variable = resources_found_selected_field
		}
		resources_found_validate_country_field_selection = yes
		if = {
			limit = {
				check_variable = {
					var = resources_found_owned_fields^num
					value = constant:resources_found_value.zero
					compare = equals
				}
			}
			clr_country_flag = resources_found_field_system_participant
		}
	}
}

# STATE scope. Initializes the physical record but adds no resources. Call one
# package effect after this effect. Existing resources are snapshotted solely
# for audit/UI evidence; exact closure uses the separate added_* ledgers.
resources_found_initialize_field_record = {
	resources_found_begin_field_transaction = yes
	if = {
		limit = {
			check_variable = { resources_found_transaction_acquired = constant:resources_found_value.one }
			NOT = { resources_found_is_active_field = yes }
		}
		if = {
			limit = { NOT = { has_variable = global.resources_found_next_field_sequence } }
			set_variable = { global.resources_found_next_field_sequence = constant:resources_found_value.zero }
		}
		add_to_variable = { global.resources_found_next_field_sequence = constant:resources_found_value.one }
		set_variable = { resources_found_field_sequence = global.resources_found_next_field_sequence }
		set_variable = { resources_found_discovery_date = global.date }
		set_variable = { resources_found_recorded_owner = OWNER }
		set_variable = { resources_found_recorded_controller = CONTROLLER }
		set_variable = { resources_found_discoverer = OWNER }

		set_variable = { resources_found_preexisting_oil = resource@oil }
		set_variable = { resources_found_preexisting_aluminium = resource@aluminium }
		set_variable = { resources_found_preexisting_rubber = resource@rubber }
		set_variable = { resources_found_preexisting_tungsten = resource@tungsten }
		set_variable = { resources_found_preexisting_steel = resource@steel }
		set_variable = { resources_found_preexisting_chromium = resource@chromium }

		set_variable = { resources_found_added_oil = constant:resources_found_value.zero }
		set_variable = { resources_found_added_aluminium = constant:resources_found_value.zero }
		set_variable = { resources_found_added_rubber = constant:resources_found_value.zero }
		set_variable = { resources_found_added_tungsten = constant:resources_found_value.zero }
		set_variable = { resources_found_added_steel = constant:resources_found_value.zero }
		set_variable = { resources_found_added_chromium = constant:resources_found_value.zero }
		set_variable = { resources_found_total_added = constant:resources_found_value.zero }
		set_variable = { resources_found_distinct_resource_count = constant:resources_found_value.zero }
		set_variable = { resources_found_resource_roll_count = constant:resources_found_value.zero }
		set_variable = { resources_found_discovery_count = constant:resources_found_value.zero }

		set_variable = { resources_found_field_stage = constant:resources_found_field_stage.discovery }
		set_variable = { resources_found_field_posture = constant:resources_found_posture.unassigned }
		set_variable = { resources_found_contract_stage = constant:resources_found_contract_stage.none }
		set_variable = { resources_found_commission_stage = constant:resources_found_commission_stage.none }
		set_variable = { resources_found_border_stage = constant:resources_found_border_stage.none }
		set_variable = { resources_found_developed_yield = constant:resources_found_field_value.initial_developed_yield }
		set_variable = { resources_found_excavation_depth = constant:resources_found_field_value.initial_excavation_depth }
		set_variable = { resources_found_workforce_safety = constant:resources_found_field_value.initial_workforce_safety }
		set_variable = { resources_found_foreign_pressure = constant:resources_found_field_value.initial_foreign_pressure }
		set_variable = { resources_found_subsurface_disturbance = constant:resources_found_field_value.initial_disturbance }
		set_variable = { resources_found_breach_pressure = constant:resources_found_field_value.initial_breach }
		set_variable = { resources_found_exploitation_score = constant:resources_found_value.zero }
		set_variable = { resources_found_maximum_developed_yield = constant:resources_found_field_value.initial_developed_yield }
		set_variable = { resources_found_maximum_excavation_depth = constant:resources_found_field_value.initial_excavation_depth }
		set_variable = { resources_found_maximum_recorded_total = constant:resources_found_value.zero }
		set_variable = { resources_found_maximum_extraction_days = constant:resources_found_value.zero }
		set_variable = { resources_found_maximum_extraction_months = constant:resources_found_value.zero }
		clear_variable = resources_found_maximum_extraction_start_date

		set_state_flag = resources_found_field_active
		clr_state_flag = resources_found_field_suspended
		clr_state_flag = resources_found_field_closing
		clr_state_flag = resources_found_field_closed_permanently
		clr_state_flag = resources_found_evolution_iv_prevented
		resources_found_bind_field_to_owner = yes
		resources_found_register_field_achievement_owner = yes
		resources_found_end_field_transaction = yes
	}
	resources_found_release_acquired_field_transaction = yes
}

# STATE scope.
resources_found_recalculate_resource_summary = {
	set_variable = { resources_found_total_added = resources_found_added_oil }
	add_to_variable = { resources_found_total_added = resources_found_added_aluminium }
	add_to_variable = { resources_found_total_added = resources_found_added_rubber }
	add_to_variable = { resources_found_total_added = resources_found_added_tungsten }
	add_to_variable = { resources_found_total_added = resources_found_added_steel }
	add_to_variable = { resources_found_total_added = resources_found_added_chromium }

	set_variable = { resources_found_current_state_resource_total = resource@oil }
	add_to_variable = { resources_found_current_state_resource_total = resource@aluminium }
	add_to_variable = { resources_found_current_state_resource_total = resource@rubber }
	add_to_variable = { resources_found_current_state_resource_total = resource@tungsten }
	add_to_variable = { resources_found_current_state_resource_total = resource@steel }
	add_to_variable = { resources_found_current_state_resource_total = resource@chromium }

	set_variable = { resources_found_distinct_resource_count = constant:resources_found_value.zero }
	if = { limit = { check_variable = { resources_found_added_oil > constant:resources_found_value.zero } } add_to_variable = { resources_found_distinct_resource_count = constant:resources_found_value.one } }
	if = { limit = { check_variable = { resources_found_added_aluminium > constant:resources_found_value.zero } } add_to_variable = { resources_found_distinct_resource_count = constant:resources_found_value.one } }
	if = { limit = { check_variable = { resources_found_added_rubber > constant:resources_found_value.zero } } add_to_variable = { resources_found_distinct_resource_count = constant:resources_found_value.one } }
	if = { limit = { check_variable = { resources_found_added_tungsten > constant:resources_found_value.zero } } add_to_variable = { resources_found_distinct_resource_count = constant:resources_found_value.one } }
	if = { limit = { check_variable = { resources_found_added_steel > constant:resources_found_value.zero } } add_to_variable = { resources_found_distinct_resource_count = constant:resources_found_value.one } }
	if = { limit = { check_variable = { resources_found_added_chromium > constant:resources_found_value.zero } } add_to_variable = { resources_found_distinct_resource_count = constant:resources_found_value.one } }

	clr_state_flag = resources_found_compound_field
	if = {
		limit = { check_variable = { var = resources_found_distinct_resource_count value = constant:resources_found_threshold.compound_distinct_resources compare = greater_than_or_equals } }
		set_state_flag = resources_found_compound_field
	}
	clr_state_flag = resources_found_globally_significant_field
	if = {
		limit = { check_variable = { var = resources_found_total_added value = constant:resources_found_threshold.globally_significant_total compare = greater_than_or_equals } }
		set_state_flag = resources_found_globally_significant_field
	}
}

# STATE scope. Updates the 0..100 exploitation score and scale factors consumed
# by decisions. Country size is read from OWNER; field scale is depth-based.
resources_found_refresh_project_scales = {
	set_variable = { resources_found_exploitation_score = resources_found_developed_yield }
	add_to_variable = { resources_found_exploitation_score = resources_found_excavation_depth }
	divide_variable = { resources_found_exploitation_score = constant:resources_found_value.two }
	if = { limit = { check_variable = { resources_found_exploitation_score > constant:resources_found_field_value.maximum } } set_variable = { resources_found_exploitation_score = constant:resources_found_field_value.maximum } }

	set_variable = { resources_found_project_cost_factor = constant:resources_found_country_factor.standard_cost }
	set_variable = { resources_found_project_duration_factor = constant:resources_found_country_factor.standard_duration }
	if = {
		limit = { OWNER = { num_of_factories < constant:resources_found_country_scale.small_factory_ceiling } }
		set_variable = { resources_found_project_cost_factor = constant:resources_found_country_factor.small_cost }
		set_variable = { resources_found_project_duration_factor = constant:resources_found_country_factor.small_duration }
	}
	else_if = {
		limit = { OWNER = { is_major = yes } }
		set_variable = { resources_found_project_cost_factor = constant:resources_found_country_factor.major_cost }
		set_variable = { resources_found_project_duration_factor = constant:resources_found_country_factor.major_duration }
	}
	else_if = {
		limit = { OWNER = { num_of_factories > constant:resources_found_country_scale.large_factory_floor } }
		set_variable = { resources_found_project_cost_factor = constant:resources_found_country_factor.large_cost }
		set_variable = { resources_found_project_duration_factor = constant:resources_found_country_factor.large_duration }
	}
	if = {
		limit = { OWNER = { has_war = yes } }
		multiply_variable = { resources_found_project_duration_factor = constant:resources_found_field_factor.war_duration }
	}
	if = {
		limit = { NOT = { is_owned_and_controlled_by = OWNER } }
		multiply_variable = { resources_found_project_duration_factor = constant:resources_found_field_factor.occupied_duration }
	}
	if = {
		limit = {
			NOT = { infrastructure > constant:resources_found_country_scale.poor_infrastructure_maximum }
		}
		multiply_variable = { resources_found_project_duration_factor = constant:resources_found_field_factor.poor_infrastructure_duration }
	}
	else_if = {
		limit = {
			NOT = { infrastructure < constant:resources_found_country_scale.strong_infrastructure_minimum }
		}
		multiply_variable = { resources_found_project_duration_factor = constant:resources_found_field_factor.strong_infrastructure_duration }
	}

	set_variable = { resources_found_field_cost_factor = constant:resources_found_field_factor.standard }
	if = { limit = { check_variable = { resources_found_excavation_depth < constant:resources_found_field_value.band_2 } } set_variable = { resources_found_field_cost_factor = constant:resources_found_field_factor.low } }
	else_if = { limit = { check_variable = { resources_found_excavation_depth > constant:resources_found_field_value.band_5 } } set_variable = { resources_found_field_cost_factor = constant:resources_found_field_factor.extreme } }
	else_if = { limit = { check_variable = { resources_found_excavation_depth > constant:resources_found_field_value.band_4 } } set_variable = { resources_found_field_cost_factor = constant:resources_found_field_factor.high } }
}

# STATE scope. Rebuilds the mutually exclusive field-output stage from durable
# lifecycle facts. Suspension replaces the visible output stage without losing
# the underlying development stage used when the field resumes.
resources_found_normalize_field_output_stage = {
	resources_found_clear_field_output_stage_modifiers = yes
	if = {
		limit = { resources_found_is_active_field = yes }
		if = {
			limit = { has_state_flag = resources_found_maximum_shifts_active }
			set_variable = { resources_found_field_stage = constant:resources_found_field_stage.rush }
		}
		else_if = {
			limit = { has_state_flag = resources_found_local_processing_complete }
			set_variable = { resources_found_field_stage = constant:resources_found_field_stage.mature }
		}
		else_if = {
			limit = { has_state_flag = resources_found_primary_works_open }
			set_variable = { resources_found_field_stage = constant:resources_found_field_stage.development }
		}
		else_if = {
			limit = { has_state_flag = resources_found_geological_appraisal_complete }
			set_variable = { resources_found_field_stage = constant:resources_found_field_stage.appraisal }
		}
		else = {
			set_variable = { resources_found_field_stage = constant:resources_found_field_stage.discovery }
		}

		if = {
			limit = { has_state_flag = resources_found_field_suspended }
			add_dynamic_modifier = { modifier = resources_found_field_suspended }
		}
		else_if = {
			limit = { has_state_flag = resources_found_maximum_shifts_active }
			add_dynamic_modifier = { modifier = resources_found_field_maximum_extraction }
		}
		else_if = {
			limit = { has_state_flag = resources_found_local_processing_complete }
			add_dynamic_modifier = { modifier = resources_found_field_industrial }
		}
		else_if = {
			limit = { has_state_flag = resources_found_primary_works_open }
			add_dynamic_modifier = { modifier = resources_found_field_operating }
		}
		else_if = {

			multiply_temp_variable = { resources_found_inverse_amount = constant:resources_found_value.negative_one }

# STATE scope. Exact floor(total standard resources / 10), capped at 10. Cave
# origins always output zero future capacity. Parent spawn logic consumes
# resources_found_captured_state_capacity and owns activation/grace timers.
resources_found_refresh_captured_state_capacity = {
	set_variable = { resources_found_capacity_source_total = resource@oil }
	add_to_variable = { resources_found_capacity_source_total = resource@aluminium }
	add_to_variable = { resources_found_capacity_source_total = resource@rubber }
	add_to_variable = { resources_found_capacity_source_total = resource@tungsten }
	add_to_variable = { resources_found_capacity_source_total = resource@steel }
	add_to_variable = { resources_found_capacity_source_total = resource@chromium }
	set_temp_variable = { resources_found_capacity_remainder = resources_found_capacity_source_total }
	set_temp_variable = { resources_found_capacity_result = constant:resources_found_value.zero }
	while = {
		limit = {
			check_variable = { var = resources_found_capacity_remainder value = constant:resources_found_capacity.resources_per_division compare = greater_than_or_equals }
			check_variable = { resources_found_capacity_result < constant:resources_found_capacity.per_state_cap }
		}
		subtract_from_temp_variable = { resources_found_capacity_remainder = constant:resources_found_capacity.resources_per_division }
		add_to_temp_variable = { resources_found_capacity_result = constant:resources_found_value.one }
	}
	if = {
		limit = { resources_found_is_cave_origin_state = yes }
		set_temp_variable = { resources_found_capacity_result = constant:resources_found_capacity.origin_future_capacity }
	}
	set_variable = { resources_found_captured_state_capacity = resources_found_capacity_result }
	set_variable = { resources_found_capacity_last_refresh_date = global.date }
}

	}

# STATE scope. Derives capped scoring months from completed days plus any live
# interval. This is idempotent and does not end an active extraction interval.
resources_found_finalize_maximum_extraction_duration = {
	set_temp_variable = { resources_found_scored_maximum_extraction_days = constant:resources_found_value.zero }
	if = {
		limit = { has_variable = resources_found_maximum_extraction_days }
		set_temp_variable = { resources_found_scored_maximum_extraction_days = resources_found_maximum_extraction_days }
	}
	if = {
		limit = { has_variable = resources_found_maximum_extraction_start_date }
		set_temp_variable = { resources_found_live_maximum_extraction_days = global.date }
		subtract_from_temp_variable = { resources_found_live_maximum_extraction_days = resources_found_maximum_extraction_start_date }
		clamp_temp_variable = { var = resources_found_live_maximum_extraction_days min = constant:resources_found_value.zero }
		add_to_temp_variable = { resources_found_scored_maximum_extraction_days = resources_found_live_maximum_extraction_days }
	}
	divide_temp_variable = { resources_found_scored_maximum_extraction_days = constant:resources_found_strength.maximum_extraction_month_days }
	clamp_temp_variable = {
		var = resources_found_scored_maximum_extraction_days
		min = constant:resources_found_value.zero
		max = constant:resources_found_strength.maximum_extraction_score_months
	}
	set_variable = { resources_found_maximum_extraction_months = resources_found_scored_maximum_extraction_days }
}

# STATE cave-origin scope. Requires resources_found_cave_country event target.
# Normalizes the mandatory six-resource package, scores persistent exploitation
# and mitigation history, converts each complete five score points to one
# division, adds the minimum six, and hard-caps both score and result at 30.

	save_event_target_as = resources_found_strength_origin_state
	resources_found_recalculate_resource_summary = yes
	resources_found_finalize_maximum_extraction_duration = yes

	clamp_temp_variable = { var = resources_found_strength_score min = constant:resources_found_value.zero }
	divide_temp_variable = { resources_found_strength_score = constant:resources_found_strength.resource_score_divisor }
	multiply_temp_variable = { resources_found_strength_score = constant:resources_found_strength_factor.resource_units }

	set_temp_variable = { resources_found_strength_component = resources_found_distinct_resource_count }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.distinct_resource }
	add_to_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	set_temp_variable = { resources_found_strength_component = resources_found_discovery_count }
	subtract_from_temp_variable = { resources_found_strength_component = constant:resources_found_strength.free_discovery_count }
	clamp_temp_variable = { var = resources_found_strength_component min = constant:resources_found_value.zero }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.discovery }
	add_to_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	set_temp_variable = { resources_found_strength_component = resources_found_maximum_developed_yield }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.maximum_yield }
	add_to_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	set_temp_variable = { resources_found_strength_component = resources_found_maximum_excavation_depth }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.maximum_depth }
	add_to_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	set_temp_variable = { resources_found_strength_component = resources_found_maximum_extraction_months }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.maximum_extraction_month }
	add_to_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	if = {
		limit = { has_variable = resources_found_field_deaths }
		set_temp_variable = { resources_found_strength_component = resources_found_field_deaths }
		divide_temp_variable = { resources_found_strength_component = constant:resources_found_strength.death_score_divisor }
		multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.field_deaths }
		add_to_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	}
	if = { limit = { has_state_flag = resources_found_failed_full_seal } add_to_temp_variable = { resources_found_strength_score = constant:resources_found_strength_factor.failed_full_seal } }
	if = { limit = { has_state_flag = resources_found_military_exploitation } add_to_temp_variable = { resources_found_strength_score = constant:resources_found_strength_factor.military_exploitation } }
	if = { limit = { has_state_flag = resources_found_unsealed_nest } add_to_temp_variable = { resources_found_strength_score = constant:resources_found_strength_factor.unsealed_nest } }

	set_temp_variable = { resources_found_strength_component = resources_found_workforce_safety }
	subtract_from_temp_variable = { resources_found_strength_component = constant:resources_found_strength.safety_reduction_floor }
	clamp_temp_variable = { var = resources_found_strength_component min = constant:resources_found_value.zero }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.workforce_safety_reduction }
	subtract_from_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	if = { limit = { has_state_flag = resources_found_successful_evacuation } subtract_from_temp_variable = { resources_found_strength_score = constant:resources_found_strength_factor.successful_evacuation_reduction } }
	if = { limit = { has_state_flag = resources_found_sealed_network } subtract_from_temp_variable = { resources_found_strength_score = constant:resources_found_strength_factor.sealed_network_reduction } }
	set_temp_variable = { resources_found_strength_component = constant:resources_found_value.zero }
	if = {
		limit = { has_variable = resources_found_total_suspension_days }
		set_temp_variable = { resources_found_strength_component = resources_found_total_suspension_days }
	}
	if = {
		limit = { has_variable = resources_found_suspension_start_date }
		set_temp_variable = { resources_found_live_suspension_days = global.date }
		subtract_from_temp_variable = { resources_found_live_suspension_days = resources_found_suspension_start_date }
		clamp_temp_variable = { var = resources_found_live_suspension_days min = constant:resources_found_value.zero }
		add_to_temp_variable = { resources_found_strength_component = resources_found_live_suspension_days }
	}
	divide_temp_variable = { resources_found_strength_component = constant:resources_found_strength.suspension_score_divisor_days }
	clamp_temp_variable = { var = resources_found_strength_component min = constant:resources_found_value.zero max = constant:resources_found_strength.maximum_suspension_score_periods }
	multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.long_suspension_reduction }
	subtract_from_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	if = {
		limit = { has_variable = resources_found_successful_hunts }
		set_temp_variable = { resources_found_strength_component = resources_found_successful_hunts }
		clamp_temp_variable = { var = resources_found_strength_component min = constant:resources_found_value.zero max = constant:resources_found_strength.maximum_hunts_scored }
		multiply_temp_variable = { resources_found_strength_component = constant:resources_found_strength_factor.successful_hunt_reduction }
		subtract_from_temp_variable = { resources_found_strength_score = resources_found_strength_component }
	}
	clamp_temp_variable = { var = resources_found_strength_score min = constant:resources_found_value.zero max = constant:resources_found_strength.maximum_score }

	set_temp_variable = { resources_found_strength_remainder = resources_found_strength_score }
	set_temp_variable = { resources_found_strength_divisions = constant:resources_found_strength.minimum_divisions }
	while = {
		limit = {
			check_variable = { var = resources_found_strength_remainder value = constant:resources_found_strength.score_per_division compare = greater_than_or_equals }
			check_variable = { resources_found_strength_divisions < constant:resources_found_strength.maximum_divisions }
		}
		subtract_from_temp_variable = { resources_found_strength_remainder = constant:resources_found_strength.score_per_division }
		add_to_temp_variable = { resources_found_strength_divisions = constant:resources_found_value.one }
	}
	set_variable = { resources_found_cave_starting_score = resources_found_strength_score }
	set_variable = { resources_found_cave_starting_divisions = resources_found_strength_divisions }
	if = {
		limit = {
			has_event_target = resources_found_cave_country
			event_target:resources_found_cave_country = { has_country_flag = resources_found_cave_country_created }
		}
		event_target:resources_found_cave_country = {
			set_variable = { resources_found_cave_starting_score = event_target:resources_found_strength_origin_state.resources_found_cave_starting_score }
			set_variable = { resources_found_cave_starting_divisions = event_target:resources_found_strength_origin_state.resources_found_cave_starting_divisions }
		}
	}
}

# COUNTRY scope. Sets/clears the near-control presentation/action flag from the
# exact remaining-state input maintained by the cave layer.
resources_found_refresh_near_control_status = {
	clr_country_flag = resources_found_world_end_near_control
	if = {
		limit = {
			has_variable = resources_found_continent_remaining_state_count
			check_variable = { var = resources_found_continent_remaining_state_count value = constant:resources_found_world_end.near_control_remaining_states compare = less_than_or_equals }
			check_variable = { resources_found_continent_remaining_state_count > constant:resources_found_value.zero }
		}
		set_country_flag = resources_found_world_end_near_control
	}
}

# COUNTRY scope. Starts a reversible 60-day verification only after the exact
# input triggers pass. Completion rechecks the same inputs.
resources_found_start_world_end_verification = {
	if = {
		limit = { resources_found_can_start_world_end_verification = yes }
		set_country_flag = resources_found_world_end_verification_active
		clr_country_flag = resources_found_world_end_verification_complete
		set_variable = { resources_found_world_end_verification_date = global.date }
		set_temp_variable = { resources_found_world_end_verification_delay = constant:resources_found_world_end.verification_days }
		add_to_variable = { resources_found_world_end_verification_date = resources_found_world_end_verification_delay }
	}
}

# COUNTRY scope.
resources_found_cancel_world_end_verification = {
	if = {

		clr_country_flag = resources_found_world_end_verification_active
		clear_variable = resources_found_world_end_verification_date
		add_to_variable = { resources_found_world_end_verification_cancellations = constant:resources_found_value.one }
	}
}

# COUNTRY scope. Marks exact verification complete but deliberately does not set
