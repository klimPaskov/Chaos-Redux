# Called in CBL scope from the capitulation on-action after it saves
# event_target:cannibalism_defeated_country. Each defeated country can issue
# receipts only once per unified runtime, so repeated capitulation callbacks do
# not manufacture battlefield harvest capacity.
cannibalism_unified_record_distinct_enemy_capitulation = {
	if = {
		limit = {
			cannibalism_unified_decisions_are_open = yes
			has_event_target = cannibalism_defeated_country
			event_target:cannibalism_defeated_country = {
				exists = yes
				NOT = { tag = CBL }
				NOT = { is_cannibalism_active_country = yes }
			}
			NOT = {
				is_in_array = {
					array = cannibalism_unified_defeated_enemy_entries
					value = event_target:cannibalism_defeated_country
				}
			}
		}
		add_to_array = {
			array = cannibalism_unified_defeated_enemy_entries
			value = event_target:cannibalism_defeated_country
		}
		add_to_variable = { cannibalism_unified_distinct_enemy_capitulations = constant:cannibalism_unified_profile.one }
		set_temp_variable = { cannibalism_unified_receipt_flag_days = constant:cannibalism_unified_operation.battlefield_receipt_days }
		event_target:cannibalism_defeated_country = {
			set_country_flag = { flag = cannibalism_unified_battlefield_victory_receipted days = cannibalism_unified_receipt_flag_days }
		}
		if = {
			limit = {
				has_country_flag = cannibalism_unified_larder_method_battlefield
				has_country_flag = cannibalism_unified_major_victory_receipts_open
			}
			add_to_variable = { cannibalism_unified_battlefield_receipts = cannibalism_unified_battlefield_receipts_per_victory }
			if = {
				limit = { has_country_flag = cannibalism_unified_battlefield_harvest_caps_required }
				clamp_variable = {
					var = cannibalism_unified_battlefield_receipts
					min = constant:cannibalism_unified_profile.zero
					max = cannibalism_unified_battlefield_receipt_cap
				}
			}
		}
	}
}

cannibalism_unified_execute_state_consumption = {
	set_temp_variable = { cannibalism_unified_consumption_result = 0 }
	if = {
		limit = { has_event_target = cannibalism_unified_decision_state }
		event_target:cannibalism_unified_decision_state = { save_event_target_as = cannibalism_consumption_target_state }
		set_temp_variable = { cannibalism_consumption_exact_request_people = cannibalism_unified_population_cost_k }
		multiply_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_consumption.population_people_per_k }
		set_temp_variable = { cannibalism_consumption_context = cannibalism_unified_consumption_mode }
		cannibalism_prepare_consumption_context = yes
		if = {
			limit = {
				check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied }
				check_variable = { var = cannibalism_population_loss_applied value = cannibalism_consumption_exact_request_people compare = equals }
			}
			if = { limit = { check_variable = { cannibalism_unified_command_cost > 0 } } set_temp_variable = { cannibalism_current_cost = cannibalism_unified_command_cost } cannibalism_pay_current_command_cost = yes }
			if = { limit = { check_variable = { cannibalism_unified_truck_cost > 0 } } set_temp_variable = { cannibalism_current_cost = cannibalism_unified_truck_cost } cannibalism_pay_current_truck_cost = yes }
			if = { limit = { check_variable = { cannibalism_unified_train_cost > 0 } } set_temp_variable = { cannibalism_current_cost = cannibalism_unified_train_cost } cannibalism_pay_current_train_cost = yes }
			if = { limit = { check_variable = { cannibalism_unified_support_cost > 0 } } set_temp_variable = { cannibalism_current_cost = cannibalism_unified_support_cost } cannibalism_pay_current_support_equipment_cost = yes }
			if = { limit = { check_variable = { cannibalism_unified_convoy_cost > 0 } } set_temp_variable = { cannibalism_current_cost = cannibalism_unified_convoy_cost } cannibalism_pay_current_convoy_cost = yes }
			if = { limit = { check_variable = { cannibalism_unified_fuel_cost > 0 } } cannibalism_unified_pay_current_fuel_cost = yes }
			if = {
				limit = { check_variable = { cannibalism_unified_skip_operation_record < 1 } }
				cannibalism_unified_record_larder_operation = yes
			}
			set_temp_variable = { cannibalism_unified_consumption_result = 1 }
		}
		set_temp_variable = { cannibalism_consumption_exact_request_people = 0 }
		set_temp_variable = { cannibalism_unified_skip_operation_record = 0 }
	}
}

	set_temp_variable = { cannibalism_unified_recruitment_result = 0 }
	if = {
		limit = { has_event_target = cannibalism_unified_decision_state }
		save_event_target_as = cannibalism_unified_recruitment_actor
		event_target:cannibalism_unified_decision_state = { save_event_target_as = cannibalism_consumption_target_state }
		set_temp_variable = { cannibalism_consumption_exact_request_people = cannibalism_unified_population_cost_k }
		multiply_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_consumption.population_people_per_k }
		set_temp_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.recruitment }
		cannibalism_prepare_consumption_context = yes
		if = {
			limit = {
				check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied }
				check_variable = { var = cannibalism_population_loss_applied value = cannibalism_consumption_exact_request_people compare = equals }
			}
			set_temp_variable = { cannibalism_unified_larder_cost = cannibalism_unified_recruitment_larder_cost }
			cannibalism_unified_pay_current_larder_cost = yes
			set_temp_variable = { cannibalism_unified_recruitment_manpower = cannibalism_population_loss_applied }
			multiply_temp_variable = { cannibalism_unified_recruitment_manpower = constant:cannibalism_unified_operation.recruitment_manpower_factor }
			round_temp_variable = cannibalism_unified_recruitment_manpower
			add_manpower = cannibalism_unified_recruitment_manpower
			set_temp_variable = { cannibalism_unified_unit_start_experience = constant:cannibalism_unified_operation.unit_start_experience }
			set_temp_variable = { cannibalism_unified_unit_start_equipment = constant:cannibalism_unified_operation.unit_start_equipment }
			set_temp_variable = { cannibalism_unified_unit_start_manpower = constant:cannibalism_unified_operation.unit_start_manpower }
			if = {
				limit = { check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.legion } }
				event_target:cannibalism_unified_decision_state = { cannibalism_unified_spawn_legion = yes }
				add_to_variable = { cannibalism_unified_legions_raised = constant:cannibalism_unified_profile.one }
			}
			else_if = {
				limit = { check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.bone_guard } }
				event_target:cannibalism_unified_decision_state = { cannibalism_unified_spawn_bone_guard = yes }
				add_to_variable = { cannibalism_unified_bone_guards_raised = constant:cannibalism_unified_profile.one }
			}
			else_if = {
				limit = {
					OR = {
						check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.island_reavers }
						check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.siege_eaters }
						check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.march_predation_column }
					}
				}
				event_target:cannibalism_unified_decision_state = { cannibalism_unified_spawn_origin_specialist = yes }
				add_to_variable = { cannibalism_unified_origin_specialists_raised = constant:cannibalism_unified_profile.one }
				if = {
					limit = { check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.island_reavers } }
					add_to_variable = { cannibalism_unified_island_specialists_raised = constant:cannibalism_unified_profile.one }
				}
				else_if = {
					limit = { check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.siege_eaters } }
					add_to_variable = { cannibalism_unified_siege_specialists_raised = constant:cannibalism_unified_profile.one }
				}
				else_if = {
					limit = { check_variable = { cannibalism_unified_recruitment_template = constant:cannibalism_unified_recruitment_template.march_predation_column } }
					add_to_variable = { cannibalism_unified_march_specialists_raised = constant:cannibalism_unified_profile.one }
				}
			}
			event_target:cannibalism_unified_decision_state = {
				set_temp_variable = { cannibalism_unified_recruitment_cooldown_days = constant:cannibalism_unified_operation.recruitment_state_cooldown_days }
				set_state_flag = { flag = cannibalism_recruitment_cooldown days = cannibalism_unified_recruitment_cooldown_days }
			}
			if = {
				limit = { cannibalism_unified_decisions_are_open = yes }
				cannibalism_unified_record_war_machine_operation = yes
				cannibalism_unified_focus_refresh_terminal_route = yes
			}
			else_if = {
				limit = {
					cannibalism_wendigo_command_is_open = yes
					has_country_flag = cannibalism_wendigo_cannibal_recruitment_open
				}
				add_to_variable = { cannibalism_wendigo_cannibal_formations_raised = constant:cannibalism_unified_profile.one }
				cannibalism_unified_refresh_operational_profiles = yes
			}
			set_temp_variable = { cannibalism_unified_recruitment_result = constant:cannibalism_unified_profile.one }

	}
