# ============================================================================
# CHAOS REDUX - EVENT 029 HARM AND ACCOUNTABILITY EFFECTS
# ============================================================================
# STATE-scope wrappers for the finite civilian harm attached to Event 029
# choices and mine collapse. Each wrapper has its own receipt flag, applies
# the requested loss through the shared exact state-population transaction,
# records the loss under its own Deaths reason (mine_disaster for collapses
# and supernatural losses, mine_violence for violence people inflict), and
# adds a public atrocity condemnation entry for the current controller. The helper
# never creates contamination, outbreak state, ordinary map resources, or
# Event 018 cave state. The second half records abuses that stay hidden until
# an audit, a disclosure, a published contract, an occupation, a records raid
# by outsiders or an accepted foreign inspection brings them to light; a
# blocked inspection is a public cover-up of its own.
# ============================================================================

# STATE scope. Inputs: temporary riches_found_harm_deaths,
# riches_found_harm_condemnation, riches_found_harm_reason (a
# chaos_meter_deaths_reason id; mine_violence when absent) and optional
# riches_found_harm_context.
# Outputs: normal state variable riches_found_harm_applied and accumulated
# riches_found_harm_total_deaths. A bounded owner target is retained only for
# the current effect chain so the controller's condemnation names the victim.
riches_found_apply_harm_record = {
	if = {
		limit = {
			has_variable = riches_found_harm_deaths
			check_variable = { var = riches_found_harm_deaths value = constant:riches_found_value.zero compare = greater_than }
		}
		if = {
			limit = { NOT = { has_variable = riches_found_harm_condemnation } }
			set_temp_variable = { riches_found_harm_condemnation = constant:riches_found_value.zero }
		}
		if = {
			limit = { NOT = { has_variable = riches_found_harm_context } }
			set_temp_variable = { riches_found_harm_context = constant:condemnation_context.camp_operation }
		}
		if = {
			limit = { NOT = { has_variable = riches_found_harm_reason } }
			set_temp_variable = { riches_found_harm_reason = constant:chaos_meter_deaths_reason.mine_violence }
		}
		if = {
			limit = { OWNER = { exists = yes } }
			OWNER = { save_event_target_as = riches_found_harm_victim }
		}
		set_temp_variable = { state_civilian_population_loss_requested = riches_found_harm_deaths }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:riches_found_value.zero }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:riches_found_value.one }
		set_temp_variable = { state_civilian_population_loss_reason = riches_found_harm_reason }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:riches_found_value.one }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:riches_found_value.one }
		set_temp_variable = { state_civilian_population_loss_applied = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_result = constant:chaos_meter_deaths.zero }
		apply_exact_state_civilian_population_loss = yes
		set_variable = { riches_found_harm_applied = state_civilian_population_loss_applied }
		if = {
			limit = { NOT = { has_variable = riches_found_harm_total_deaths } }
			set_variable = { riches_found_harm_total_deaths = constant:riches_found_value.zero }
		}
		add_to_variable = { riches_found_harm_total_deaths = riches_found_harm_applied }
		set_state_flag = riches_found_harm_occurred
		if = {
			limit = {
				check_variable = {
					var = riches_found_harm_applied
					value = constant:riches_found_harm.mass_death_gate
					compare = greater_than_or_equals
				}
			}
			set_state_flag = riches_found_mass_death_occurred
			riches_found_record_achievement_mass_death = yes
			# Mass killing by people is a massacre; a collapse has its own source.
			if = {
				limit = { NOT = { check_variable = { var = riches_found_harm_context value = constant:condemnation_context.observer_report compare = equals } } }
				set_temp_variable = { riches_found_chaos_source = constant:riches_found_chaos_source.massacre }
				riches_found_apply_chaos_source = yes
			}
			if = {
				limit = { CONTROLLER = { exists = yes } }
				CONTROLLER = { set_country_flag = riches_found_mass_death_occurred }
			}
		}
		set_temp_variable = { riches_found_harm_severity = constant:condemnation_severity.major }
		if = {
			limit = {
				check_variable = {
					var = riches_found_harm_applied
					value = constant:riches_found_harm.mass_death_gate
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { riches_found_harm_severity = constant:condemnation_severity.mass_death }
		}
		if = {
			limit = {
				check_variable = { var = riches_found_harm_applied value = constant:riches_found_value.zero compare = greater_than }
				CONTROLLER = { exists = yes }
			}
			CONTROLLER = {
				set_temp_variable = { condemnation_source_type = constant:condemnation_source.atrocity }
				set_temp_variable = { condemnation_base_gain = riches_found_harm_condemnation }
				set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.public }
				set_temp_variable = { condemnation_severity_type = riches_found_harm_severity }
				set_temp_variable = { condemnation_civilian_deaths = riches_found_harm_applied }
				set_temp_variable = { condemnation_contamination_value = constant:riches_found_value.zero }
				set_temp_variable = { condemnation_context_type = riches_found_harm_context }
				if = {
					limit = { has_event_target = riches_found_harm_victim }
					set_temp_variable = { condemnation_victim_country = event_target:riches_found_harm_victim }
				}
				condemnation_add_source = yes
			}
		}
	}
}

# closes the workings. Sealing during a severe collapse or a supernatural
# crisis without a completed evacuation entombs the shift still below: one
# bounded loss under the mine_violence reason (a deliberate act), a public
# atrocity, a blow to Local Order and Revenue Legitimacy, and the end of The

# STATE scope. A lethal raid can recur, so its receipt is keyed to the bounded
# raid sequence instead of a permanent boolean. The same raid instance cannot
# apply deaths twice through report delivery, transfer, or pulse reconciliation.

# Hidden abuses and their exposure
# ---------------------------------------------------------------------------
# Some abuses at a mine are not public when they happen: guards who rule the
# district, records burned after deaths, corruption tolerated over a lethal
# history, and soldiers' control of a gold-sick district. Each kind enters the
# responsible controller's Condemnation once per mine as a hidden or suspected
# source (condemnation_add_source), and the mine remembers the hidden amount
# and who holds it. An audit, a disclosure, a published contract, an
# occupation by another country or a records raid by outsiders brings the
# hidden amount to light once through condemnation_expose_hidden_bucket.

# STATE scope. Inputs: temporary riches_found_hidden_source_type,
# riches_found_hidden_visibility, riches_found_hidden_severity,
# riches_found_hidden_context and riches_found_hidden_base_gain. Adds the
# source to the current controller and returns the part that stayed hidden in
# temporary riches_found_hidden_recorded_amount.
riches_found_add_hidden_abuse_source = {
	set_temp_variable = { riches_found_hidden_recorded_amount = constant:riches_found_value.zero }
	if = {
		limit = { CONTROLLER = { exists = yes } }
		if = {
			limit = { OWNER = { exists = yes } }
			OWNER = { save_event_target_as = riches_found_hidden_victim }
		}
		CONTROLLER = {
			set_temp_variable = { condemnation_source_type = riches_found_hidden_source_type }
			set_temp_variable = { condemnation_base_gain = riches_found_hidden_base_gain }
			set_temp_variable = { condemnation_visibility_type = riches_found_hidden_visibility }
			set_temp_variable = { condemnation_severity_type = riches_found_hidden_severity }
			set_temp_variable = { condemnation_civilian_deaths = constant:riches_found_value.zero }
			set_temp_variable = { condemnation_contamination_value = constant:riches_found_value.zero }
			set_temp_variable = { condemnation_context_type = riches_found_hidden_context }
			if = {
				limit = { has_event_target = riches_found_hidden_victim }
				set_temp_variable = { condemnation_victim_country = event_target:riches_found_hidden_victim }
			}
			condemnation_add_source = yes
		}
		set_temp_variable = { riches_found_hidden_recorded_amount = condemnation_gain_hidden }
	}
}

# STATE scope, once per mine. Private guards that answer to no one (an
# enclave, or guards past the private-security gate) abuse the district out
# of sight.
riches_found_record_hidden_security_abuse = {
	if = {
		limit = {
			NOT = { has_state_flag = riches_found_hidden_security_recorded }
			CONTROLLER = { exists = yes }
			OR = {
				has_state_flag = riches_found_private_enclave
				AND = {
					has_state_flag = riches_found_private_guards
					NOT = { has_state_flag = riches_found_private_forces_disarmed }
					check_variable = { var = riches_found_private_security_autonomy value = constant:riches_found_threshold.private_security_gate compare = greater_than_or_equals }
				}
			}
		}
		set_state_flag = riches_found_hidden_security_recorded
		set_temp_variable = { riches_found_hidden_source_type = constant:condemnation_source.atrocity }
		set_temp_variable = { riches_found_hidden_visibility = constant:condemnation_visibility.hidden }
		set_temp_variable = { riches_found_hidden_severity = constant:condemnation_severity.major }
		set_temp_variable = { riches_found_hidden_context = constant:condemnation_context.camp_operation }
		set_temp_variable = { riches_found_hidden_base_gain = constant:riches_found_hidden_abuse.security_gain }
		riches_found_add_hidden_abuse_source = yes
		set_variable = { riches_found_hidden_security_amount = riches_found_hidden_recorded_amount }
		set_variable = { riches_found_hidden_security_holder = CONTROLLER }
	}
}

# COUNTRY scope, the country that holds a hidden entry. Inputs: temporary
# riches_found_hidden_expose_amount, riches_found_hidden_source_type and
# riches_found_hidden_context. Exposes exactly that amount of the matching
# hidden bucket (as a fraction of the bucket) and refreshes the totals and
# tier as condemnation_expose_hidden_evidence does.
riches_found_expose_hidden_amount_in_country = {
	set_temp_variable = { condemnation_hidden_exposure_fraction = constant:riches_found_value.zero }
	if = {
		limit = {
			check_variable = { var = riches_found_hidden_source_type value = constant:condemnation_source.atrocity compare = equals }
			has_variable = condemnation_hidden_atrocity
			check_variable = { condemnation_hidden_atrocity > constant:riches_found_value.zero }
		}
		set_temp_variable = { condemnation_hidden_exposure_fraction = riches_found_hidden_expose_amount }
		divide_temp_variable = { condemnation_hidden_exposure_fraction = condemnation_hidden_atrocity }
	}
	else_if = {
		limit = {
			check_variable = { var = riches_found_hidden_source_type value = constant:condemnation_source.coverup compare = equals }
			has_variable = condemnation_hidden_coverup
			check_variable = { condemnation_hidden_coverup > constant:riches_found_value.zero }
		}
		set_temp_variable = { condemnation_hidden_exposure_fraction = riches_found_hidden_expose_amount }
		divide_temp_variable = { condemnation_hidden_exposure_fraction = condemnation_hidden_coverup }
	}
	clamp_temp_variable = { var = condemnation_hidden_exposure_fraction min = constant:riches_found_value.zero max = constant:riches_found_value.one }
	if = {
		limit = { check_variable = { condemnation_hidden_exposure_fraction > constant:riches_found_value.zero } }
		set_temp_variable = { condemnation_source_type = riches_found_hidden_source_type }
		set_temp_variable = { condemnation_context_type = riches_found_hidden_context }
		condemnation_expose_hidden_bucket = yes
		condemnation_recalculate_hidden_total = yes
		condemnation_recalculate_total = yes
		clamp_variable = { var = condemnation_recent_gain min = constant:riches_found_value.zero max = constant:condemnation_threshold.recent_gain_cap }
		if = {
			limit = { check_variable = { condemnation_recent_gain > constant:riches_found_value.zero } }
			set_temp_variable = { condemnation_recent_gain_days = constant:condemnation_duration_days.recent_gain }
			set_country_flag = { flag = condemnation_recent_gain_visible days = var:condemnation_recent_gain_days }
		}
		condemnation_refresh_tier = yes
	}
}

# STATE scope, from the foreign inspection-demand incident. A mine that still
# holds hidden abuse records either opens its books (a public or state
# administration, or one that disclosed its owners or published its contract),
# which brings every record to light against the country that holds it, or
# its controller turns the inspectors away, which is itself a public cover-up,
# recorded once per mine. Output: temporary riches_found_incident_report_text
# names the outcome for the report; a mine with nothing hidden keeps the
# ordinary report.
riches_found_resolve_inspection_demand = {
	if = {
		limit = { riches_found_mine_holds_hidden_abuse = yes }
		if = {
			limit = { riches_found_mine_opens_books_to_inspection = yes }
			riches_found_expose_hidden_abuse = yes
			set_temp_variable = { riches_found_incident_report_text = constant:riches_found_report_text.inspection_exposed }
		}
		else = {
			riches_found_record_blocked_inspection = yes
			set_temp_variable = { riches_found_incident_report_text = constant:riches_found_report_text.inspection_blocked }
		}
	}
}

# STATE scope, once per mine. The controller blocked a foreign inspection of
# a mine with hidden abuses: one public cover-up source in its Condemnation.
riches_found_record_blocked_inspection = {
	if = {
		limit = {
			NOT = { has_state_flag = riches_found_blocked_inspection_recorded }
			CONTROLLER = { exists = yes }
		}
		set_state_flag = riches_found_blocked_inspection_recorded
		if = {
			limit = { OWNER = { exists = yes } }
			OWNER = { save_event_target_as = riches_found_blocked_inspection_victim }
		}
		CONTROLLER = {
			set_temp_variable = { condemnation_source_type = constant:condemnation_source.coverup }
			set_temp_variable = { condemnation_base_gain = constant:riches_found_hidden_abuse.blocked_inspection_gain }
			set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.public }
			set_temp_variable = { condemnation_severity_type = constant:condemnation_severity.major }
			set_temp_variable = { condemnation_civilian_deaths = constant:riches_found_value.zero }
			set_temp_variable = { condemnation_contamination_value = constant:riches_found_value.zero }
			set_temp_variable = { condemnation_context_type = constant:condemnation_context.blocked_inspection }
			if = {
				limit = { has_event_target = riches_found_blocked_inspection_victim }
				set_temp_variable = { condemnation_victim_country = event_target:riches_found_blocked_inspection_victim }
			}
			condemnation_add_source = yes
		}
	}
}
