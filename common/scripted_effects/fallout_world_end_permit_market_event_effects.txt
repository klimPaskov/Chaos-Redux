			limit = { check_variable = { fallout_event_698_history_stage = constant:fallout_event_runtime.zero } }
			if = { limit = { check_variable = { fallout_event_698_selected_branch value = constant:fallout_event_698_branch.issue_local_passes compare = equals } } set_temp_variable = { fallout_event_698_history_payload = constant:fallout_event_698_log.choice_issue } }

		}
		else_if = {
			limit = { check_variable = { fallout_event_698_history_stage = constant:fallout_event_runtime.one } }
			if = {
				limit = { check_variable = { fallout_event_698_selected_branch value = constant:fallout_event_698_branch.issue_local_passes compare = equals } }
				if = { limit = { check_variable = { fallout_event_698_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_698_history_payload = constant:fallout_event_698_log.issue_success } }

				else = { set_temp_variable = { fallout_event_698_history_payload = constant:fallout_event_698_log.issue_failure } }
			}
			else_if = {
				limit = { check_variable = { fallout_event_698_selected_branch value = constant:fallout_event_698_branch.audit_registry compare = equals } }
				if = { limit = { check_variable = { fallout_event_698_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_698_history_payload = constant:fallout_event_698_log.audit_success } }
