			limit = {
				has_country_flag = camp_rework_usa_authority_active
				OR = {
					has_country_flag = camp_rework_inspection_active
					check_variable = {
						var = court_challenge_pressure
						value = constant:camp_rework_threshold.band_high
						compare = greater_than_or_equals

			limit = { has_country_flag = camp_rework_usa_authority_active }

camp_rework_colonial_register_current_state = {
	if = {
		limit = {
			camp_rework_state_can_accept_responsibility_from_root = yes
			OR = { is_controlled_by = ROOT CONTROLLER = { is_subject_of = ROOT } }
			NOT = { is_valid_camp_active_site_state = yes }
			NOT = { concentration_camp > 0 }
			NOT = { extermination_camp > 0 }
		}
		if = {
			limit = { NOT = { has_variable = genocide_responsible_country } }
			set_variable = { genocide_responsible_country = ROOT }
		}
		set_state_flag = genocide_site_has_responsible_country
		set_state_flag = genocide_concentration_site
		set_state_flag = genocide_atrocity_site_known_to_regime
		add_building_construction = {
			type = concentration_camp
			level = @CAMP_COLONIAL_BUILDING_LEVEL
			instant_build = yes
		}
		chaosx_refresh_camp_visual_anchor = yes
		set_temp_variable = { camp_register_site_type = constant:camp_rework_site_type.detention }
		camp_rework_register_active_site = yes
		add_to_variable = { camp_state_evidence_depth = constant:camp_rework_country_value.evidence_minor }
		add_to_variable = { camp_state_resistance_accumulation = constant:camp_rework_country_value.resistance_minor }
		camp_administration_civil_authorize_subject_site = yes
		camp_admin_initialize_site = yes
	}
}

camp_rework_colonial_select_random_candidate = {
	clear_variable = camp_selected_state_id
	if = {
		limit = { check_variable = { camp_colonial_action_states^num > 0 } }
		random_scope_in_array = {
			array = camp_colonial_action_states
			set_temp_variable = { camp_colonial_selected_state_id = THIS.id }
		}
		set_variable = { camp_selected_state_id = camp_colonial_selected_state_id }
	}
	clear_temp_array = camp_colonial_action_states
}

camp_rework_select_france_unassigned_active_state = {
	clear_temp_array = camp_colonial_action_states
	for_each_scope_loop = {
		array = camp_active_site_states
		if = {
			limit = {
				is_valid_camp_active_site_state = yes
				var:genocide_responsible_country = { tag = ROOT }
				OR = { is_controlled_by = ROOT CONTROLLER = { is_subject_of = ROOT } }
				NOT = { has_variable = camp_site_secondary_responsible_country }
			}
