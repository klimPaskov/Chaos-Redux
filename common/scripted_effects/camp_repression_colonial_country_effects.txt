# Overview:
# - Implements the U.K./Raj, U.S.A., France/Vichy, Italy/Libya, and
#   Belgium/Congo country-kit lifecycles.

#   dispatcher, and registered-country monthly pulse.

camp_rework_update_colonial_country_monthly_bridge = {
	camp_rework_initialize_colonial_country_variables = yes
	if = {
		limit = { original_tag = ENG }
		set_variable = { imperial_detention_reach = camp_active_site_count }
		if = { limit = { check_variable = { indian_autonomy_resistance > camp_policy_resistance_pressure } } set_variable = { camp_policy_resistance_pressure = indian_autonomy_resistance } }
		if = { limit = { check_variable = { colonial_legitimacy_damage > camp_policy_legitimacy } } set_variable = { camp_policy_legitimacy = colonial_legitimacy_damage } }
		if = { limit = { check_variable = { raj_labor_burden > camp_policy_population_loss } } set_variable = { camp_policy_population_loss = raj_labor_burden } }
		if = {
			limit = {
				check_variable = { camp_active_site_count > 0 }
				NOT = { has_country_flag = camp_rework_crisis_exposed }
				OR = {
					check_variable = { var = indian_autonomy_resistance value = constant:camp_rework_threshold.severe_discovery compare = greater_than_or_equals }
					AND = { RAJ = { exists = yes NOT = { is_subject_of = ROOT } } }
				}
			}
			set_country_flag = camp_rework_crisis_exposed
			if = {
				limit = { RAJ = { exists = yes NOT = { is_subject_of = ROOT } } }
				set_country_flag = camp_rework_achievement_uk_colonial_revolt_occurred
			}
			add_to_variable = { colonial_legitimacy_damage = constant:camp_rework_colonial.accountability_major }
			camp_rework_expose_first_colonial_evidence = yes
		}
	}
	else_if = {
		limit = { original_tag = USA }
		set_variable = { wartime_security_reach = camp_active_site_count }
		set_variable = { camp_democratic_legitimacy_damage = democratic_legitimacy_damage }
		if = { limit = { check_variable = { court_challenge_pressure > camp_policy_reform } } set_variable = { camp_policy_reform = court_challenge_pressure } }
		if = { limit = { check_variable = { democratic_legitimacy_damage > camp_policy_legitimacy } } set_variable = { camp_policy_legitimacy = democratic_legitimacy_damage } }
		if = { limit = { check_variable = { relocation_population_disruption > camp_policy_population_loss } } set_variable = { camp_policy_population_loss = relocation_population_disruption } }
		if = {
			limit = {
				has_country_flag = camp_rework_usa_authority_active
				has_war = no
				NOT = { has_country_flag = camp_rework_crisis_exposed }
			}
			set_country_flag = camp_rework_crisis_exposed
			add_to_variable = { court_challenge_pressure = constant:camp_rework_colonial.accountability_major }
			camp_rework_expose_first_colonial_evidence = yes
		}
	}
	else_if = {
		limit = { OR = { original_tag = FRA tag = VIC } }
		set_variable = { vichy_collaboration_reach = camp_active_site_count }
		if = { limit = { check_variable = { refugee_pressure > camp_policy_visibility } } set_variable = { camp_policy_visibility = refugee_pressure } }
		if = { limit = { check_variable = { north_africa_labor_burden > camp_policy_population_loss } } set_variable = { camp_policy_population_loss = north_africa_labor_burden } }
		if = {
			limit = {
				check_variable = { refugee_pressure > constant:camp_rework_threshold.band_medium }
				check_variable = { camp_active_site_count > 0 }
				NOT = { has_active_mission = fr_refugee_pressure_response }
				NOT = { has_country_flag = camp_rework_france_refugee_response_pending }
				NOT = { has_country_flag = camp_rework_france_refugee_response_cooldown }
			}
			camp_rework_begin_france_refugee_response = yes
		}
		if = {
			limit = {
				has_government = democratic
				NOT = { has_country_flag = camp_rework_crisis_exposed }
				OR = {
					check_variable = { camp_active_site_count > 0 }
					any_controlled_state = { is_valid_camp_active_site_state = yes has_variable = genocide_responsible_country var:genocide_responsible_country = { OR = { original_tag = FRA tag = VIC } } }
					any_subject_country = { any_controlled_state = { is_valid_camp_active_site_state = yes has_variable = genocide_responsible_country var:genocide_responsible_country = { OR = { original_tag = FRA tag = VIC } } } }
				}
			}
			set_country_flag = camp_rework_crisis_exposed
			add_to_variable = { refugee_pressure = constant:camp_rework_colonial.accountability_major }
			camp_rework_expose_first_french_inherited_evidence = yes
		}
	}
	else_if = {
		limit = { original_tag = ITA }
		set_variable = { colonial_repression_reach = camp_active_site_count }
		if = { limit = { check_variable = { libyan_resistance_pressure > camp_policy_resistance_pressure } } set_variable = { camp_policy_resistance_pressure = libyan_resistance_pressure } }
		if = { limit = { check_variable = { desert_camp_burden > camp_policy_population_loss } } set_variable = { camp_policy_population_loss = desert_camp_burden } }
		if = { limit = { check_variable = { postwar_colonial_claim_damage > camp_policy_tribunal } } set_variable = { camp_policy_tribunal = postwar_colonial_claim_damage } }
		if = {
			limit = {
				check_variable = { camp_active_site_count > 0 }
				OR = { has_government = democratic has_government = communism }
				NOT = { has_country_flag = camp_rework_crisis_exposed }
			}
			set_country_flag = camp_rework_crisis_exposed
			add_to_variable = { postwar_colonial_claim_damage = constant:camp_rework_colonial.accountability_major }
			camp_rework_expose_first_colonial_evidence = yes
		}
	}
	else_if = {
		limit = { original_tag = BEL }
		set_variable = { congo_extraction_pressure = camp_active_site_count }
		if = { limit = { check_variable = { colonial_unrest_pressure > camp_policy_resistance_pressure } } set_variable = { camp_policy_resistance_pressure = colonial_unrest_pressure } }
		if = { limit = { check_variable = { congo_population_damage > camp_policy_population_loss } } set_variable = { camp_policy_population_loss = congo_population_damage } }
		if = { limit = { check_variable = { postwar_accountability_pressure > camp_policy_tribunal } } set_variable = { camp_policy_tribunal = postwar_accountability_pressure } }
		if = {
			limit = {
				OR = {
					check_variable = { colonial_unrest_pressure > constant:camp_rework_threshold.band_medium }
					check_variable = { camp_resistance_pressure > constant:camp_rework_threshold.band_medium }
				}
				check_variable = { camp_active_site_count > 0 }
				NOT = { has_active_mission = bel_colonial_strike_response }
				NOT = { has_country_flag = camp_rework_belgium_strike_response_pending }
				NOT = { has_country_flag = camp_rework_belgium_strike_response_cooldown }
			}
			camp_rework_begin_belgium_strike_response = yes
		}
		if = {
			limit = {
				check_variable = { camp_active_site_count > 0 }
				NOT = { has_country_flag = camp_rework_crisis_exposed }
				NOT = {
					OR = {
						any_controlled_state = { is_bel_congo_concession_pool_state = yes }
						any_subject_country = { any_controlled_state = { is_bel_congo_concession_pool_state = yes } }
					}
				}
			}
			set_country_flag = camp_rework_crisis_exposed
			add_to_variable = { postwar_accountability_pressure = constant:camp_rework_colonial.accountability_major }
			camp_rework_expose_first_colonial_evidence = yes
		}
	}

	clamp_variable = { var = indian_autonomy_resistance min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = colonial_legitimacy_damage min = 0 max = constant:camp_rework_threshold.max_value }

	if = {
		limit = {
			camp_rework_state_can_accept_responsibility_from_root = yes
			OR = { is_controlled_by = ROOT CONTROLLER = { is_subject_of = ROOT } }
			NOT = { is_valid_camp_active_site_state = yes }
			NOT = { concentration_camp > 0 }
		}
		if = {
			limit = { NOT = { has_variable = genocide_responsible_country } }
			set_variable = { genocide_responsible_country = ROOT }
		}
		set_state_flag = genocide_site_has_responsible_country
		set_state_flag = genocide_concentration_site
		set_state_flag = genocide_atrocity_site_known_to_regime
		add_building_construction = {
			type = concentration_camp
			level = @CAMP_COLONIAL_BUILDING_LEVEL
			instant_build = yes
		}
		chaosx_refresh_camp_visual_anchor = yes
		set_temp_variable = { camp_register_site_type = constant:camp_rework_site_type.detention }
		camp_rework_register_active_site = yes
		add_to_variable = { camp_state_evidence_depth = constant:camp_rework_country_value.evidence_minor }
		add_to_variable = { camp_state_resistance_accumulation = constant:camp_rework_country_value.resistance_minor }

	}
}

# Starts the two country pressure missions without disturbing a human player's
# persistent Repression Ledger selection.
camp_rework_begin_france_refugee_response = {
	set_temp_variable = { camp_colonial_had_saved_selection = 0 }
	if = {
		limit = { has_variable = camp_selected_state_id }
		set_temp_variable = { camp_colonial_had_saved_selection = 1 }
		set_variable = { camp_colonial_saved_selection_id = camp_selected_state_id }
	}
	camp_rework_select_france_active_state = yes
	if = {
		limit = { has_variable = camp_selected_state_id }
		set_variable = { fr_refugee_response_state_id = camp_selected_state_id }
		set_country_flag = camp_rework_france_refugee_response_pending
		clr_country_flag = camp_rework_france_refugee_route_aid
		clr_country_flag = camp_rework_france_refugee_route_reform
		clr_country_flag = camp_rework_france_refugee_route_suppression
		activate_mission = fr_refugee_pressure_response
	}
	if = {
		limit = {
			check_variable = { camp_colonial_had_saved_selection > 0 }
			has_variable = camp_colonial_saved_selection_id
		}
		set_variable = { camp_selected_state_id = camp_colonial_saved_selection_id }
	}
	else = { clear_variable = camp_selected_state_id }
	clear_variable = camp_colonial_saved_selection_id
}

camp_rework_save_current_colonial_subject_administrator = {
	if = {
		limit = { CONTROLLER = { is_subject_of = ROOT } }
		CONTROLLER = { save_event_target_as = camp_rework_colonial_local_administrator }
	}
	else_if = {
		limit = { OWNER = { is_subject_of = ROOT } }
		OWNER = { save_event_target_as = camp_rework_colonial_local_administrator }
	}
}

camp_rework_apply_minor_colonial_subject_pressure = {
	camp_rework_save_current_colonial_subject_administrator = yes
	if = {
		limit = {
			has_event_target = camp_rework_colonial_local_administrator
			OR = { CONTROLLER = { is_subject_of = ROOT } OWNER = { is_subject_of = ROOT } }
		}
		event_target:camp_rework_colonial_local_administrator = {
			if = {
				limit = { ROOT = { original_tag = ENG } }
				if = { limit = { NOT = { has_idea = raj_colonial_labor_burden } } add_ideas = raj_colonial_labor_burden }
				add_autonomy_ratio = {
					value = constant:camp_rework_colonial.subject_autonomy_minor
					localization = camp_rework_uk_raj_autonomy_pressure
				}
			}
			else_if = {
				limit = { ROOT = { OR = { original_tag = FRA tag = VIC } } }
				if = { limit = { NOT = { has_idea = fr_north_africa_labor_burden } } add_ideas = fr_north_africa_labor_burden }
				add_autonomy_ratio = {
					value = constant:camp_rework_colonial.subject_autonomy_minor
					localization = camp_rework_france_colonial_autonomy_pressure
				}
			}
			else_if = {
				limit = { ROOT = { original_tag = BEL } }
				if = { limit = { NOT = { has_idea = congo_concession_labor_burden } } add_ideas = congo_concession_labor_burden }
				add_autonomy_ratio = {
					value = constant:camp_rework_colonial.subject_autonomy_minor
					localization = camp_rework_bel_congo_autonomy_pressure
				}
			}
		}
	}
}
