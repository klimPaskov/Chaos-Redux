fallout_event_677_apply_failure_deaths = {

		limit = { check_variable = { fallout_event_677_selected_branch = constant:fallout_event_677_branch.equal_measure } }
		set_temp_variable = { fallout_event_677_failure_death_percent = constant:fallout_event_677_threshold.equal_measure_failure_death_percent }

	else_if = {
		limit = { check_variable = { fallout_event_677_selected_branch = constant:fallout_event_677_branch.field_requisition } }
		set_temp_variable = { fallout_event_677_failure_death_percent = constant:fallout_event_677_threshold.field_requisition_failure_death_percent }

	}

		limit = {
			has_variable = fallout_event_677_result_state_id
			fallout_event_677_ration_registry_is_current = yes
		}
		var:fallout_event_677_result_state_id = {
			set_temp_variable = { fallout_event_677_death_request = state_population_k }
