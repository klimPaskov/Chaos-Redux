# ============================================================================
# CHAOS REDUX - EVENT 006 INDEPENDENCE WAVE FOCUS EFFECTS
# ============================================================================
# Overview:
# - Assigns the full focus framework only when the release transaction chooses
#   that explicit mode.
# - Activates an additive overlay flag without replacing an existing tree.
# - Publishes package-owned route, host, league, ambition, signature, and
#   formable adapter hooks.
# - Applies auditable bundles to the five public statehood values.
# - Adapts infrastructure, economy, and military rewards to existing package
#   metadata while avoiding free-division loops.
# - Locks five government routes. Patron-client commitment remains owned by
#   Decision DM-38 and the shared government-route effect.
# - Resolves former-host focus outcomes through the shared bilateral ledger.
#
# Scope: country unless stated otherwise. No effect performs a broad periodic
# world iteration.
# ============================================================================

# ---------------------------------------------------------------------------
# Assignment and generation-local cleanup
# ---------------------------------------------------------------------------

# Input: temporary `independence_wave_focus_assignment_input`.
# Full framework mode is reserved for Event 6-created tags or a registered tag
# whose minimal-tree status was explicitly reviewed. Additive modes never call
# load_focus_tree and therefore cannot overwrite a meaningful existing tree.
independence_wave_assign_focus_framework = {
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_valid_independence_wave_focus_assignment_input = yes
		}
		clr_country_flag = independence_wave_full_focus_framework
		clr_country_flag = independence_wave_additive_focus_overlay
		clr_country_flag = independence_wave_post_formation_focus_overlay
		set_variable = { independence_wave_focus_assignment = independence_wave_focus_assignment_input }
		if = {
			limit = { check_variable = { independence_wave_focus_assignment_input = constant:independence_wave_focus_assignment.full_framework } }
			set_country_flag = independence_wave_full_focus_framework
			load_focus_tree = {
				tree = independence_wave_focus_tree
				keep_completed = no
			}
		}
		else_if = {
			limit = { check_variable = { independence_wave_focus_assignment_input = constant:independence_wave_focus_assignment.additive_overlay } }
			set_country_flag = independence_wave_additive_focus_overlay
		}
		else_if = {
			limit = { check_variable = { independence_wave_focus_assignment_input = constant:independence_wave_focus_assignment.post_formation_overlay } }
			set_country_flag = independence_wave_additive_focus_overlay
			set_country_flag = independence_wave_post_formation_focus_overlay
		}
		mark_focus_tree_layout_dirty = yes
	}
}
