	if = {
		limit = { is_independence_wave_iw043_country = yes has_country_flag = independence_wave_iw043_cosmetic_identity_ready }

			limit = { NOT = { has_country_flag = independence_wave_iw043_cosmetic_active } }

			set_country_flag = independence_wave_iw043_cosmetic_active

		}
		if = {
			limit = { has_country_flag = independence_wave_iw043_route_federal NOT = { has_country_flag = independence_wave_iw043_federal_cosmetic_applied } }
			drop_cosmetic_tag = yes
			set_cosmetic_tag = CHU_independence_wave_volga_federationX
			set_country_flag = independence_wave_iw043_federal_cosmetic_applied
			set_country_flag = independence_wave_iw043_cosmetic_active
		}
		if = {
			limit = { has_country_flag = independence_wave_iw043_route_restoration NOT = { has_country_flag = independence_wave_iw043_restoration_cosmetic_applied } }
			drop_cosmetic_tag = yes
			set_cosmetic_tag = CHU_independence_wave_volga_bulgariaX
			set_country_flag = independence_wave_iw043_restoration_cosmetic_applied
			set_country_flag = independence_wave_iw043_cosmetic_active
		}
	}
}

# The guardianship branch may have replaced the cosmetic tag while the route
# flag was active. Restore the last civilian identity using route flags first,
# then the applied-identity receipt, and finally the opening council identity.
# This helper never selects an advisor portrait or changes the package route;
# it only restores the country-level cosmetic surface after guardianship ends.
independence_wave_restore_iw058_preserved_civilian_cosmetic_identity = {
	if = {
		limit = {
			is_independence_wave_iw058_country = yes
			has_country_flag = independence_wave_iw058_cosmetic_identity_ready
		}
		if = {
			limit = { has_country_flag = independence_wave_iw058_cosmetic_active }
			drop_cosmetic_tag = yes
		}
		if = {
			limit = {
				OR = {
					has_country_flag = independence_wave_iw058_route_church_compact
					has_country_flag = independence_wave_iw058_church_cosmetic_applied
				}
			}
			set_cosmetic_tag = ASY_independence_wave_church_compactX
			clr_country_flag = independence_wave_iw058_civic_cosmetic_applied
			clr_country_flag = independence_wave_iw058_guardianship_cosmetic_applied
			set_country_flag = independence_wave_iw058_church_cosmetic_applied
		}
		else_if = {
			limit = {
				OR = {
					has_country_flag = independence_wave_iw058_route_civic_assembly
					has_country_flag = independence_wave_iw058_civic_cosmetic_applied
				}
			}
			set_cosmetic_tag = ASY_independence_wave_civic_federationX
			clr_country_flag = independence_wave_iw058_church_cosmetic_applied
			clr_country_flag = independence_wave_iw058_guardianship_cosmetic_applied
			set_country_flag = independence_wave_iw058_civic_cosmetic_applied
		}
		else = {
			set_cosmetic_tag = ASY_independence_wave_national_councilX
			clr_country_flag = independence_wave_iw058_church_cosmetic_applied
			clr_country_flag = independence_wave_iw058_civic_cosmetic_applied
			clr_country_flag = independence_wave_iw058_guardianship_cosmetic_applied
			set_country_flag = independence_wave_iw058_opening_cosmetic_applied
		}
		set_country_flag = independence_wave_iw058_cosmetic_active
	}
}

# Binds one currently-owned, non-reserve division whose composition matches
# the package's supplied profile. The country-level receipt is only an
# admission gate; it is never treated as the unit identity. The selected
# division is persisted as a generation-scoped global target so every timed
# mission phase can validate and mutate the same formation.
independence_wave_bind_iw043_force_package = {
	set_temp_variable = { independence_wave_iw043_force_binding_result = constant:independence_wave_iw043.zero }
	clear_global_event_target = independence_wave_iw043_bound_force_division
	clear_variable = independence_wave_iw043_bound_force_generation
	if = {
		limit = { can_bind_independence_wave_iw043_force_package = yes }
		random_country_division = {
			limit = {
				has_variable = independence_wave_force_division_package_id
				check_variable = { var = independence_wave_force_division_package_id value = constant:independence_wave_package_id.iw_043 compare = equals }
				has_variable = independence_wave_force_division_generation_id
				check_variable = { var = independence_wave_force_division_generation_id value = ROOT.independence_wave_generation_id compare = equals }
				division_has_majority_template = infantry
				division_has_battalion_in_template = engineer
				division_has_battalion_in_template = recon
				NOT = { division_has_battalion_in_template = artillery_brigade }
				NOT = { is_unit_template_reserves = yes }
				unit_strength > constant:independence_wave_iw043.minimum_supplied_division_strength
				unit_organization > constant:independence_wave_iw043.minimum_supplied_division_organization
			}
			save_global_event_target_as = independence_wave_iw043_bound_force_division
		}
		if = {
			limit = { has_event_target = independence_wave_iw043_bound_force_division }
			set_variable = { independence_wave_iw043_bound_force_generation = independence_wave_generation_id }
			set_country_flag = iw043_force_binding_active
			set_temp_variable = { independence_wave_iw043_force_binding_result = constant:independence_wave_iw043.one }
		}
	}
}

# Costs are supplied as temporary variables by the owning decision. The helper
# validates the complete bundle before spending anything, stores a normal
# transaction receipt, and never refunds a failed timed mission: rollback is a
# ledger close, while the decision's timeout effects own the gameplay penalty.
independence_wave_iw043_begin_paid_transaction = {
	set_temp_variable = { independence_wave_iw043_transaction_result = constant:independence_wave_iw043.zero }
	if = {
		limit = {
			is_independence_wave_iw043_country = yes
			NOT = { has_country_flag = independence_wave_iw043_transaction_active }
			check_variable = { var = independence_wave_iw043_transaction_id value = constant:independence_wave_iw043.zero compare = greater_than }
			OR = {
				check_variable = { var = independence_wave_iw043_command_power_cost value = constant:independence_wave_iw043.zero compare = equals }
				command_power > independence_wave_iw043_command_power_cost
			}
			OR = {
				check_variable = { var = independence_wave_iw043_manpower_cost value = constant:independence_wave_iw043.zero compare = equals }
				has_manpower > independence_wave_iw043_manpower_cost
			}
			OR = {
				check_variable = { var = independence_wave_iw043_infantry_equipment_cost value = constant:independence_wave_iw043.zero compare = equals }
				has_equipment = { infantry_equipment > independence_wave_iw043_infantry_equipment_cost }
			}
			OR = {
				check_variable = { var = independence_wave_iw043_support_equipment_cost value = constant:independence_wave_iw043.zero compare = equals }
				has_equipment = { support_equipment > independence_wave_iw043_support_equipment_cost }
			}
			OR = {
				check_variable = { var = independence_wave_iw043_train_equipment_cost value = constant:independence_wave_iw043.zero compare = equals }
				has_equipment = { train_equipment > independence_wave_iw043_train_equipment_cost }
			}
			OR = {
				check_variable = { var = independence_wave_iw043_convoy_cost value = constant:independence_wave_iw043.zero compare = equals }
				has_equipment = { convoy > independence_wave_iw043_convoy_cost }
			}
		}
		if = {
			limit = { independence_wave_iw043_command_power_cost > constant:independence_wave_iw043.zero }
			set_temp_variable = { independence_wave_iw043_negative_cost = independence_wave_iw043_command_power_cost }
			multiply_temp_variable = { independence_wave_iw043_negative_cost = -1 }
			add_command_power = independence_wave_iw043_negative_cost
		}
		if = {
			limit = { independence_wave_iw043_manpower_cost > constant:independence_wave_iw043.zero }
			set_temp_variable = { independence_wave_iw043_negative_cost = independence_wave_iw043_manpower_cost }
			multiply_temp_variable = { independence_wave_iw043_negative_cost = -1 }
			add_manpower = independence_wave_iw043_negative_cost
		}
		if = {
			limit = { independence_wave_iw043_infantry_equipment_cost > constant:independence_wave_iw043.zero }
			set_temp_variable = { independence_wave_iw043_negative_cost = independence_wave_iw043_infantry_equipment_cost }
			multiply_temp_variable = { independence_wave_iw043_negative_cost = -1 }
			add_equipment_to_stockpile = { type = infantry_equipment amount = independence_wave_iw043_negative_cost }
		}
		if = {
			limit = { independence_wave_iw043_support_equipment_cost > constant:independence_wave_iw043.zero }
			set_temp_variable = { independence_wave_iw043_negative_cost = independence_wave_iw043_support_equipment_cost }
			multiply_temp_variable = { independence_wave_iw043_negative_cost = -1 }
			add_equipment_to_stockpile = { type = support_equipment amount = independence_wave_iw043_negative_cost }
		}
		if = {
			limit = { independence_wave_iw043_train_equipment_cost > constant:independence_wave_iw043.zero }
			set_temp_variable = { independence_wave_iw043_negative_cost = independence_wave_iw043_train_equipment_cost }
			multiply_temp_variable = { independence_wave_iw043_negative_cost = -1 }
			add_equipment_to_stockpile = { type = train_equipment amount = independence_wave_iw043_negative_cost }
		}
		if = {
			limit = { independence_wave_iw043_convoy_cost > constant:independence_wave_iw043.zero }
			set_temp_variable = { independence_wave_iw043_negative_cost = independence_wave_iw043_convoy_cost }
			multiply_temp_variable = { independence_wave_iw043_negative_cost = -1 }
			add_equipment_to_stockpile = { type = convoy amount = independence_wave_iw043_negative_cost }
		}
		set_variable = { independence_wave_iw043_paid_transaction_id = independence_wave_iw043_transaction_id }
		set_variable = { independence_wave_iw043_paid_command_power_cost = independence_wave_iw043_command_power_cost }
		set_variable = { independence_wave_iw043_paid_manpower_cost = independence_wave_iw043_manpower_cost }
		set_variable = { independence_wave_iw043_paid_infantry_equipment_cost = independence_wave_iw043_infantry_equipment_cost }
		set_variable = { independence_wave_iw043_paid_support_equipment_cost = independence_wave_iw043_support_equipment_cost }
		set_variable = { independence_wave_iw043_paid_train_equipment_cost = independence_wave_iw043_train_equipment_cost }
		set_variable = { independence_wave_iw043_paid_convoy_cost = independence_wave_iw043_convoy_cost }
		set_country_flag = independence_wave_iw043_transaction_active
		set_temp_variable = { independence_wave_iw043_transaction_result = constant:independence_wave_iw043.one }
	}
}
