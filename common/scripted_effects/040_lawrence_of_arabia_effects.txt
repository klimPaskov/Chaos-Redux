# Target weighting and pre-fire selection, the campaign opening, per-target
# state, the weekly pulse (drift, band, confirmation windows, incidents),
# the intelligence model with and without La Resistance, AI profiles and
# the cost quotes.
#
# Scope: COUNTRY unless noted. ROOT is the live target inside the pulse, the
# target category and the target events, and ENG inside the British category.
# The pulse is scheduled by the live target through chaosx.nr40.5. No world
# scan runs on a timer: the only whole-world pass is the one-off candidate
# pool when the random pool or the regional delay picks a target.
# Tuning lives in common/script_constants/040_lawrence_of_arabia_constants.txt
# and the weighted logic in common/mtth/040_lawrence_of_arabia_mtth.txt.
# Settlements, detention, defection and cleanup are in
# 040_lawrence_of_arabia_settlement_effects.txt and the decision effects in
# 040_lawrence_of_arabia_action_effects.txt.

# ANY scope. Called by fire_event_by_temp_id_no_cluster before history is
# recorded. Saves lawrence_prefire_target for dispatch and the Event History
# actor and sets temporary lawrence_prefire_ready to 1 on success.
lawrence_prepare_random_event_fire = {
	set_temp_variable = { lawrence_prefire_ready = constant:lawrence_event.zero }
	set_temp_variable = { lawrence_pick_ready = constant:lawrence_event.zero }
	lawrence_pick_weighted_target = yes
	if = {
		limit = { check_variable = { var = lawrence_pick_ready value = constant:lawrence_event.zero compare = greater_than } }
		set_temp_variable = { lawrence_prefire_ready = constant:lawrence_event.one }
	}
}

# ANY scope. Weighted draw among every valid target. Saves the event target
# lawrence_prefire_target and sets temporary lawrence_pick_ready.
lawrence_pick_weighted_target = {
	set_temp_variable = { lawrence_pick_ready = constant:lawrence_event.zero }
	clear_temp_array = lawrence_target_pool
	if = {
		limit = {
			ENG = { lawrence_sponsor_can_act = yes }
			NOT = { has_global_flag = lawrence_campaign_closed }
		}
		every_country = {
			limit = { lawrence_is_valid_target = yes }
			set_temp_variable = { lawrence_target_weight = constant:lawrence_event.zero }
			lawrence_score_target = yes
			for_loop_effect = {
				start = 0
				end = lawrence_target_weight
				value = lawrence_target_pool_index
				add_to_temp_array = { array = lawrence_target_pool value = THIS }
			}
		}
		if = {
			limit = { check_variable = { var = lawrence_target_pool^num value = constant:lawrence_event.zero compare = greater_than } }
			random_scope_in_array = {
				array = lawrence_target_pool
				save_event_target_as = lawrence_prefire_target
				set_temp_variable = { lawrence_pick_ready = constant:lawrence_event.one }
			}
		}
	}
	clear_temp_array = lawrence_target_pool
}

# COUNTRY, the candidate. Output: temporary lawrence_target_weight, an integer
# of at least 1. Every factor is a declared reason and none depends on being a
# player. Weights order candidates and are not probabilities.
lawrence_score_target = {
	set_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.base }
	set_temp_variable = { lawrence_w_stability = stability }
	if = {
		limit = { check_variable = { var = lawrence_w_stability value = constant:lawrence_target_weight.unstable_threshold compare = less_than } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.unstable }
	}
	if = {
		limit = { check_variable = { var = lawrence_w_stability value = constant:lawrence_target_weight.very_unstable_threshold compare = less_than } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.very_unstable }
	}
	if = {
		limit = { check_variable = { var = lawrence_w_stability value = constant:lawrence_target_weight.stable_threshold compare = greater_than } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.stable }
	}
	if = {
		limit = { has_defensive_war = yes }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.threatened }
	}
	if = {
		limit = { has_war_with_major = yes }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.hostile_power }
	}
	if = {
		limit = { has_country_flag = lawrence_requested_protection }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.requested }
	}
	if = {
		limit = { any_owned_state = { is_coastal = yes lawrence_is_registry_state = yes } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.port }
	}
	if = {
		limit = { is_in_faction_with = ENG }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.british_faction }
	}
	if = {
		limit = {
			NOT = { is_in_faction_with = ENG }
			OR = {
				any_neighbor_country = { tag = ENG }
				any_neighbor_country = { is_subject_of = ENG }
				any_neighbor_country = {
					has_country_flag = lawrence_settled
					OR = {
						check_variable = { var = lawrence_settlement value = constant:lawrence_outcome.client compare = equals }
						check_variable = { var = lawrence_settlement value = constant:lawrence_outcome.ally compare = equals }
					}
				}
			}
		}
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.british_neighbour }
	}
	if = {
		limit = { ENG = { has_military_access_to = PREV } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.british_access }
	}
	# Counterintelligence, memory of earlier failures and British overreach.
	lawrence_compute_security = yes
	if = {
		limit = { check_variable = { var = lawrence_security value = constant:lawrence_target_weight.strong_security compare = greater_than_or_equals } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.security }
	}
	if = {
		limit = { has_variable = lawrence_preparedness }
		set_temp_variable = { lawrence_w_prepared = lawrence_preparedness }
		multiply_temp_variable = { lawrence_w_prepared = constant:lawrence_target_weight.prepared_each }
		add_to_temp_variable = { lawrence_target_weight = lawrence_w_prepared }
	}
	if = {
		limit = { ENG = { has_war_with_major = yes } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.british_war }
		if = {
			limit = {
				NOT = { any_owned_state = { is_coastal = yes } }
				NOT = { any_neighbor_country = { is_subject_of = ENG } }
			}
			add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.british_war_remote }
		}
	}
	if = {
		limit = { ENG = { convoy_threat > constant:lawrence_british.overextended_convoy_threat } }
		add_to_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.british_convoys }
	}
	round_temp_variable = lawrence_target_weight
	if = {
		limit = { check_variable = { var = lawrence_target_weight value = constant:lawrence_target_weight.minimum compare = less_than } }
		set_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.minimum }
	}
	if = {
		limit = { check_variable = { var = lawrence_target_weight value = constant:lawrence_target_weight.maximum compare = greater_than } }
		set_temp_variable = { lawrence_target_weight = constant:lawrence_target_weight.maximum }
	}
}

# COUNTRY. Output: variable lawrence_security (0 to 100), the target's practical
# ability to read and break a foreign network. With La Resistance the agency,
# its operatives and decryption progress against Britain count. Without the
# DLC the same decisions and outcomes run on stability, administration,
# encryption, military presence, security spirits and earlier preparedness.
lawrence_compute_security = {
	set_temp_variable = { lawrence_sec = constant:lawrence_security.base }
	set_temp_variable = { lawrence_sec_part = stability }
	multiply_temp_variable = { lawrence_sec_part = constant:lawrence_security.stability_weight }
	add_to_temp_variable = { lawrence_sec = lawrence_sec_part }
	set_temp_variable = { lawrence_sec_part = num_divisions }
	multiply_temp_variable = { lawrence_sec_part = constant:lawrence_security.division_weight }
	clamp_temp_variable = { var = lawrence_sec_part min = 0 max = constant:lawrence_security.division_cap }
	add_to_temp_variable = { lawrence_sec = lawrence_sec_part }
	if = {
		limit = { has_tech = basic_encryption }
		add_to_temp_variable = { lawrence_sec = constant:lawrence_security.encryption_basic }
	}
	if = {
		limit = { has_tech = improved_encryption }
		add_to_temp_variable = { lawrence_sec = constant:lawrence_security.encryption_improved }
	}
	if = {
		limit = { has_tech = basic_decryption }
		add_to_temp_variable = { lawrence_sec = constant:lawrence_security.decryption_basic }
	}
	if = {
		limit = { has_country_flag = lawrence_security_spirit }
		add_to_temp_variable = { lawrence_sec = constant:lawrence_security.security_spirit }
	}
	if = {
		limit = { has_variable = lawrence_preparedness }
		set_temp_variable = { lawrence_sec_part = lawrence_preparedness }
		multiply_temp_variable = { lawrence_sec_part = constant:lawrence_security.prepared_each }
		add_to_temp_variable = { lawrence_sec = lawrence_sec_part }
	}
	if = {
		limit = { has_civil_war = yes }
		add_to_temp_variable = { lawrence_sec = constant:lawrence_security.civil_conflict }
	}
	if = {
		limit = { has_country_flag = lawrence_disorder }
		add_to_temp_variable = { lawrence_sec = constant:lawrence_security.disorder }
	}
	if = {
		limit = {
			has_dlc = "La Resistance"
			has_intelligence_agency = yes
		}
		set_temp_variable = { lawrence_sec_part = agency_upgrade_number }
		multiply_temp_variable = { lawrence_sec_part = constant:lawrence_security.agency_upgrade_weight }
		clamp_temp_variable = { var = lawrence_sec_part min = 0 max = constant:lawrence_security.agency_cap }
		add_to_temp_variable = { lawrence_sec = lawrence_sec_part }
		set_temp_variable = { lawrence_sec_part = num_of_operatives }
		multiply_temp_variable = { lawrence_sec_part = constant:lawrence_security.operative_weight }
		clamp_temp_variable = { var = lawrence_sec_part min = 0 max = constant:lawrence_security.operative_cap }
		add_to_temp_variable = { lawrence_sec = lawrence_sec_part }
		if = {
			limit = { decryption_progress = { target = ENG value > constant:lawrence_security.decryption_threshold } }
			add_to_temp_variable = { lawrence_sec = constant:lawrence_security.decryption_bonus }
		}
		if = {
			limit = {
				intel_level_over = {
					target = ENG
					civilian_intel > constant:lawrence_security.civilian_intel_threshold
				}
			}
			add_to_temp_variable = { lawrence_sec = constant:lawrence_security.civilian_intel_bonus }
		}
	}
	clamp_temp_variable = { var = lawrence_sec min = 0 max = constant:lawrence_influence.maximum }
	set_variable = { lawrence_security = lawrence_sec }
}

# COUNTRY scoped to ENG. Called by chaosx.nr40.1. Resolves the target the
# pre-fire selection saved (or draws one) and starts generation one.
lawrence_open_campaign = {
	if = {
		limit = {
			NOT = { has_global_flag = lawrence_campaign_active }
			NOT = { has_global_flag = lawrence_campaign_closed }
		}
		set_temp_variable = { lawrence_open_target_ok = constant:lawrence_event.zero }
		if = {
			limit = {
				has_event_target = lawrence_prefire_target
				event_target:lawrence_prefire_target = { lawrence_is_valid_target = yes }
			}
			set_temp_variable = { lawrence_open_target_ok = constant:lawrence_event.one }
		}
		else = {
			set_temp_variable = { lawrence_pick_ready = constant:lawrence_event.zero }
			lawrence_pick_weighted_target = yes
			if = {
				limit = { check_variable = { var = lawrence_pick_ready value = constant:lawrence_event.zero compare = greater_than } }
				set_temp_variable = { lawrence_open_target_ok = constant:lawrence_event.one }
			}
		}
		if = {
			limit = { check_variable = { var = lawrence_open_target_ok value = constant:lawrence_event.zero compare = greater_than } }
			lawrence_initialize_campaign_state = yes
			event_target:lawrence_prefire_target = {
				lawrence_begin_target = yes
				lawrence_resolve_opening_evolution = yes
			}
		}
	}
}

# COUNTRY, the target. Starting Influence: enough to create a problem without
# deciding the outcome. Access, instability and a threat raise it, preparedness
# and security lower it. Evolutions raise the floor in later phases.
lawrence_compute_start_influence = {
	set_temp_variable = { lawrence_start = constant:lawrence_influence.start_base }
	if = {
		limit = { is_in_faction_with = ENG }
		add_to_temp_variable = { lawrence_start = constant:lawrence_influence.start_faction }
	}
	if = {
		limit = { ENG = { has_military_access_to = PREV } }
		add_to_temp_variable = { lawrence_start = constant:lawrence_influence.start_access }
	}
	if = {
		limit = {
			OR = {
				any_neighbor_country = { tag = ENG }
				any_neighbor_country = { is_subject_of = ENG }
			}
		}
		add_to_temp_variable = { lawrence_start = constant:lawrence_influence.start_neighbour }
	}
	set_temp_variable = { lawrence_start_stability = stability }
	if = {
		limit = { check_variable = { var = lawrence_start_stability value = constant:lawrence_target_weight.unstable_threshold compare = less_than } }
		add_to_temp_variable = { lawrence_start = constant:lawrence_influence.start_unstable }
	}
	if = {
		limit = { has_defensive_war = yes }
		add_to_temp_variable = { lawrence_start = constant:lawrence_influence.start_threatened }
	}
	set_temp_variable = { lawrence_start_security = lawrence_security }
	multiply_temp_variable = { lawrence_start_security = constant:lawrence_influence.start_security_factor }
	subtract_from_temp_variable = { lawrence_start = lawrence_start_security }
	if = {
		limit = { has_variable = lawrence_preparedness }
		set_temp_variable = { lawrence_start_prepared = lawrence_preparedness }
		multiply_temp_variable = { lawrence_start_prepared = constant:lawrence_influence.start_prepared_each }
		subtract_from_temp_variable = { lawrence_start = lawrence_start_prepared }
	}
	if = {
		limit = { has_global_flag = lawrence_survival_public }
		add_to_temp_variable = { lawrence_start = constant:lawrence_influence.start_public }
	}
	clamp_temp_variable = {
		var = lawrence_start
		min = constant:lawrence_influence.start_minimum
		max = constant:lawrence_influence.start_maximum
	}
	round_temp_variable = lawrence_start
	set_variable = { lawrence_influence = lawrence_start }
	# Officer sympathy opens near the starting access and moves with the
	# officers actions of both sides.
	set_variable = { lawrence_officers = lawrence_influence }
	multiply_variable = { lawrence_officers = constant:lawrence_officers.start_factor }
	round_variable = lawrence_officers
}

# COUNTRY, the target. Names the states the mission needs the target to hold:
# the capital where the ledgers sit, one port where the payments land and one
# state on the British side of the border or coast where the route enters.
lawrence_choose_nodes = {
	clear_variable = lawrence_node_capital
	clear_variable = lawrence_node_port
	clear_variable = lawrence_node_route
	set_variable = { lawrence_node_capital = capital }
	if = {
		limit = {
			any_owned_state = {
				is_coastal = yes
				is_capital = no
				lawrence_is_registry_state = yes
			}
		}
		random_owned_state = {
			limit = {
				is_coastal = yes
				is_capital = no
				lawrence_is_registry_state = yes
			}
			PREV = { set_variable = { lawrence_node_port = PREV } }
		}
	}
	else_if = {
		limit = { any_owned_state = { is_coastal = yes is_capital = no } }
		random_owned_state = {
			limit = { is_coastal = yes is_capital = no }
			PREV = { set_variable = { lawrence_node_port = PREV } }
		}
	}
	if = {
		limit = {
			any_owned_state = {
				is_capital = no
				OR = {
					any_neighbor_state = { is_owned_by = ENG }
					any_neighbor_state = { owner = { is_subject_of = ENG } }
				}
			}
		}
		random_owned_state = {
			limit = {
				is_capital = no
				OR = {
					any_neighbor_state = { is_owned_by = ENG }
					any_neighbor_state = { owner = { is_subject_of = ENG } }
				}
			}
			PREV = { set_variable = { lawrence_node_route = PREV } }
		}
	}
}

# COUNTRY. One primary profile per target. It is reassessed every pulse so a
# coup, a war or a change of government moves the target between profiles.
lawrence_update_ai_profile = {
	set_variable = { lawrence_ai_profile = constant:lawrence_ai.profile_pragmatist }
	set_temp_variable = { lawrence_profile_stability = stability }
	if = {
		limit = {
			OR = {
				check_variable = { var = lawrence_profile_stability value = constant:lawrence_ai.fractured_stability compare = less_than }
				has_civil_war = yes
				has_country_flag = lawrence_disorder
			}
		}
		set_variable = { lawrence_ai_profile = constant:lawrence_ai.profile_fractured }
	}
	else_if = {
		limit = {
			OR = {
				has_government = communism
				has_government = fascism
				has_government = neutrality
			}
			NOT = { is_in_faction_with = ENG }
			check_variable = { var = lawrence_security value = constant:lawrence_ai.challenger_security compare = greater_than_or_equals }
		}
		set_variable = { lawrence_ai_profile = constant:lawrence_ai.profile_challenger }
	}
	else_if = {
		limit = {
			check_variable = { var = lawrence_security value = constant:lawrence_ai.double_security compare = greater_than_or_equals }
			has_country_flag = lawrence_contact_turned
		}
		set_variable = { lawrence_ai_profile = constant:lawrence_ai.profile_double_gamer }
	}
	else_if = {
		limit = {
			check_variable = { var = lawrence_security value = constant:lawrence_ai.counterintelligence_security compare = greater_than_or_equals }
		}
		set_variable = { lawrence_ai_profile = constant:lawrence_ai.profile_counterintelligence }
	}
	else_if = {
		limit = {
			OR = {
				is_in_faction_with = ENG
				has_defensive_war = yes
				ENG = { has_military_access_to = PREV }
			}
			check_variable = { var = lawrence_profile_stability value = constant:lawrence_ai.client_stability_ceiling compare = less_than }
		}
		set_variable = { lawrence_ai_profile = constant:lawrence_ai.profile_client }
	}
}

# COUNTRY scoped to ENG. Quotes scale with the size of the live target.
lawrence_refresh_british_costs = {
	set_temp_variable = { lawrence_quote_base = constant:lawrence_cost.default_base }
	if = {

# COUNTRY, the target. Output: temporary lawrence_drift_total (points per week,
# bounded) and variable lawrence_cause (the largest contributor). Opening
# weeks do not drift until the terms are set or the opening ends.
lawrence_compute_drift = {
	set_temp_variable = { lawrence_d_support = constant:lawrence_event.zero }
	set_temp_variable = { lawrence_d_instability = constant:lawrence_event.zero }
	set_temp_variable = { lawrence_d_officers = constant:lawrence_event.zero }
	set_temp_variable = { lawrence_d_network = constant:lawrence_event.zero }
	set_temp_variable = { lawrence_d_counter = constant:lawrence_event.zero }
	set_temp_variable = { lawrence_d_britain = constant:lawrence_event.zero }
	# British support in force (shipments, route, cadres and the press network).
	if = {
		limit = {
			has_variable = lawrence_support_until
			check_variable = { var = global.num_days value = lawrence_support_until compare = less_than }
		}
		set_temp_variable = { lawrence_d_support = lawrence_support_power }
		multiply_temp_variable = { lawrence_d_support = constant:lawrence_drift.support_each }
	}
	# Government weakness and a threat make British protection credible.
	set_temp_variable = { lawrence_d_stability = stability }
	if = {
		limit = { check_variable = { var = lawrence_d_stability value = constant:lawrence_drift.unstable_threshold compare = less_than } }
		add_to_temp_variable = { lawrence_d_instability = constant:lawrence_drift.unstable }
	}
	if = {
		limit = { check_variable = { var = lawrence_d_stability value = constant:lawrence_drift.very_unstable_threshold compare = less_than } }
		add_to_temp_variable = { lawrence_d_instability = constant:lawrence_drift.very_unstable }
	}
	if = {
		limit = { has_defensive_war = yes }
		add_to_temp_variable = { lawrence_d_instability = constant:lawrence_drift.threatened }
	}
	if = {
		limit = { check_variable = { var = lawrence_d_stability value = constant:lawrence_drift.confident_threshold compare = greater_than } }
		add_to_temp_variable = { lawrence_d_instability = constant:lawrence_drift.confident }
	}
	# British-backed officers pull the army towards the mission.
	set_temp_variable = { lawrence_d_officers = lawrence_officers }
	multiply_temp_variable = { lawrence_d_officers = constant:lawrence_drift.officers_factor }
	subtract_from_temp_variable = { lawrence_d_officers = constant:lawrence_drift.officers_neutral }
	# Terms, press, guarantees and base agreements shape the network.
	if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.broad compare = equals } }
		add_to_temp_variable = { lawrence_d_network = constant:lawrence_drift.terms_broad }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.liaison compare = equals } }
		add_to_temp_variable = { lawrence_d_network = constant:lawrence_drift.terms_liaison }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.escorted compare = equals } }
		add_to_temp_variable = { lawrence_d_network = constant:lawrence_drift.terms_escorted }
	}
	if = {
		limit = { has_country_flag = lawrence_press_active }
		add_to_temp_variable = { lawrence_d_network = constant:lawrence_drift.press }
	}
	if = {
		limit = { has_country_flag = lawrence_guarantee_given }
		add_to_temp_variable = { lawrence_d_network = constant:lawrence_drift.guarantee }
	}
	if = {
		limit = { has_country_flag = lawrence_bases_agreed }
		add_to_temp_variable = { lawrence_d_network = constant:lawrence_drift.bases }
	}
	# Counterintelligence and institutions push the other way.
	set_temp_variable = { lawrence_d_security = lawrence_security }
	multiply_temp_variable = { lawrence_d_security = constant:lawrence_drift.security_factor }
	subtract_from_temp_variable = { lawrence_d_counter = lawrence_d_security }
	if = {
		limit = { has_country_flag = lawrence_file }
		add_to_temp_variable = { lawrence_d_counter = constant:lawrence_drift.file }
	}
	if = {
		limit = { has_country_flag = lawrence_office }
		add_to_temp_variable = { lawrence_d_counter = constant:lawrence_drift.office }
	}
	if = {
		limit = { has_country_flag = lawrence_escorted }
		add_to_temp_variable = { lawrence_d_counter = constant:lawrence_drift.escort }
	}
	if = {
		limit = { has_country_flag = lawrence_custody_taken }
		add_to_temp_variable = { lawrence_d_counter = constant:lawrence_drift.custody }
	}
	if = {
		limit = { has_country_flag = lawrence_contact_turned }
		add_to_temp_variable = { lawrence_d_counter = constant:lawrence_drift.turned }
	}
	if = {
		limit = { has_country_flag = lawrence_nodes_secured }
		add_to_temp_variable = { lawrence_d_counter = constant:lawrence_drift.nodes }
	}
	# Britain's own position.
	if = {
		limit = {
			ENG = {
				OR = {
					has_war_with_major = yes
					convoy_threat > constant:lawrence_british.overextended_convoy_threat
				}
			}
		}
		add_to_temp_variable = { lawrence_d_britain = constant:lawrence_drift.british_overextended }
	}
	if = {
		limit = { NOT = { has_country_flag = lawrence_route_open } }
		add_to_temp_variable = { lawrence_d_britain = constant:lawrence_drift.route_closed }
	}
	if = {
		limit = { check_variable = { var = global.lawrence_credibility value = constant:lawrence_drift.credibility_low compare = less_than } }
		add_to_temp_variable = { lawrence_d_britain = constant:lawrence_drift.credibility_penalty }

	if = {

	}
	# Local agents alone gain less than Lawrence's own network.
	if = {
		limit = { has_global_flag = lawrence_agents_only }
		multiply_temp_variable = { lawrence_d_support = constant:lawrence_drift.agents_factor }
		multiply_temp_variable = { lawrence_d_officers = constant:lawrence_drift.agents_factor }
		multiply_temp_variable = { lawrence_d_network = constant:lawrence_drift.agents_factor }
	}
	set_temp_variable = { lawrence_drift_total = lawrence_d_support }
	add_to_temp_variable = { lawrence_drift_total = lawrence_d_instability }
	add_to_temp_variable = { lawrence_drift_total = lawrence_d_officers }
	add_to_temp_variable = { lawrence_drift_total = lawrence_d_network }
	add_to_temp_variable = { lawrence_drift_total = lawrence_d_counter }
	add_to_temp_variable = { lawrence_drift_total = lawrence_d_britain }
	clamp_temp_variable = {
		var = lawrence_drift_total
		min = constant:lawrence_drift.weekly_minimum
		max = constant:lawrence_drift.weekly_maximum
	}
	if = {
		limit = { lawrence_is_live_target_opening = yes }
		multiply_temp_variable = { lawrence_drift_total = constant:lawrence_drift.opening_factor }
	}
	if = {
		limit = { has_country_flag = lawrence_final_pending }
		set_temp_variable = { lawrence_drift_total = constant:lawrence_event.zero }
	}
	# The largest contributor names the cause shown in the category header.
	set_temp_variable = { lawrence_cause_best = constant:lawrence_event.zero }
	set_variable = { lawrence_cause = constant:lawrence_cause.none }
	lawrence_pick_cause = yes
}

# COUNTRY, the target. Applies the weekly drift and tracks the trend.
lawrence_apply_drift = {
	set_variable = { lawrence_prev_influence = lawrence_influence }
	add_to_variable = { lawrence_influence = lawrence_drift_total }
	clamp_variable = { var = lawrence_influence min = constant:lawrence_influence.minimum max = constant:lawrence_influence.maximum }
	set_temp_variable = { lawrence_trend_delta = lawrence_influence }
	subtract_from_temp_variable = { lawrence_trend_delta = lawrence_prev_influence }
	set_variable = { lawrence_trend = constant:lawrence_event.trend_steady }
	if = {
		limit = { check_variable = { var = lawrence_trend_delta value = constant:lawrence_event.trend_threshold compare = greater_than } }
		set_variable = { lawrence_trend = constant:lawrence_event.trend_rising }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_trend_delta value = constant:lawrence_event.trend_falling_threshold compare = less_than } }
		set_variable = { lawrence_trend = constant:lawrence_event.trend_falling }
	}
}

# COUNTRY, the target. Adds a bounded, explained Influence change (decisions,
# incidents and outcomes). Input: temporary lawrence_change (signed).
lawrence_add_influence = {
	set_temp_variable = { lawrence_change_applied = lawrence_change }
	clamp_temp_variable = {
		var = lawrence_change_applied
		min = constant:lawrence_influence.change_minimum
		max = constant:lawrence_influence.change_maximum
	}
	add_to_variable = { lawrence_influence = lawrence_change_applied }
	clamp_variable = { var = lawrence_influence min = constant:lawrence_influence.minimum max = constant:lawrence_influence.maximum }
	lawrence_update_band = yes
}

# COUNTRY, the target. Band from Influence, with an upward report once each.
lawrence_update_band = {
	set_temp_variable = { lawrence_old_band = constant:lawrence_band.broken }
	if = {
		limit = { has_variable = lawrence_band }
		set_temp_variable = { lawrence_old_band = lawrence_band }
	}
	set_variable = { lawrence_band = constant:lawrence_band.broken }
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.restricted_edge compare = greater_than_or_equals } }
		set_variable = { lawrence_band = constant:lawrence_band.restricted }
	}
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.contested_edge compare = greater_than_or_equals } }
		set_variable = { lawrence_band = constant:lawrence_band.contested }
	}
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.embedded_edge compare = greater_than_or_equals } }
		set_variable = { lawrence_band = constant:lawrence_band.embedded }
	}
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.ascendant_edge compare = greater_than_or_equals } }
		set_variable = { lawrence_band = constant:lawrence_band.ascendant }
	}
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.dominant_edge compare = greater_than_or_equals } }
		set_variable = { lawrence_band = constant:lawrence_band.dominant }
		set_country_flag = lawrence_ach_dominant_reached
	}
	if = {
		limit = { lawrence_is_live_target = yes }
		if = {
			limit = { check_variable = { var = lawrence_band value = constant:lawrence_band.embedded compare = greater_than_or_equals } }
			if = {
				limit = { has_idea = lawrence_british_mission }
				swap_ideas = { remove_idea = lawrence_british_mission add_idea = lawrence_embedded_mission }
			}
		}
		else = {
			if = {
				limit = { has_idea = lawrence_embedded_mission }
				swap_ideas = { remove_idea = lawrence_embedded_mission add_idea = lawrence_british_mission }
			}
		}
	}
	if = {
		limit = {
			lawrence_is_live_target = yes
			check_variable = { var = lawrence_band value = lawrence_old_band compare = greater_than }
			check_variable = { var = lawrence_band value = constant:lawrence_band.embedded compare = greater_than_or_equals }
			NOT = { has_country_flag = lawrence_band_report_cooldown }
		}
		set_temp_variable = { lawrence_flag_days = constant:lawrence_event.report_cooldown_days }
		set_country_flag = { flag = lawrence_band_report_cooldown days = lawrence_flag_days }
		if = {
			limit = { check_variable = { var = lawrence_band value = constant:lawrence_band.dominant compare = equals } }
			country_event = { id = chaosx.nr40.8 }
		}
		else_if = {
			limit = { check_variable = { var = lawrence_band value = constant:lawrence_band.ascendant compare = equals } }
			country_event = { id = chaosx.nr40.8 }
		}
		else = {
			country_event = { id = chaosx.nr40.7 }
		}
	}
	lawrence_publish_header = yes
}

	}
}

# COUNTRY, the target. Shipments arrive once, by receipt.
lawrence_track_deliveries = {
	if = {
		limit = {
			has_country_flag = lawrence_delivery_pending
			has_variable = lawrence_delivery_arrival_day
			check_variable = { var = global.num_days value = lawrence_delivery_arrival_day compare = greater_than_or_equals }
			NOT = { check_variable = { var = lawrence_delivery_received_serial value = lawrence_delivery_serial compare = equals } }
		}
		lawrence_receive_delivery = yes
	}
	# Routes close when Britain no longer holds the access they rely on.
	set_temp_variable = { lawrence_route_ok = constant:lawrence_event.one }
	ENG = {
		if = {
			limit = {
				OR = {
					NOT = { lawrence_sponsor_can_act = yes }
					NOT = { lawrence_sponsor_has_regional_base = yes }
				}
			}
			set_temp_variable = { lawrence_route_ok = constant:lawrence_event.zero }
		}
	}
	if = {
		limit = { check_variable = { var = lawrence_route_ok value = constant:lawrence_event.zero compare = greater_than } }
		set_country_flag = lawrence_route_open
	}
	else = {
		clr_country_flag = lawrence_route_open
	}
}

# COUNTRY, the target. Confirmation windows: a spike never decides a country.
lawrence_track_confirmation = {
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.dominant_edge compare = greater_than_or_equals } }
		add_to_variable = { lawrence_dominant_days = constant:lawrence_event.pulse_days }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.dominant_exit compare = less_than } }
		set_variable = { lawrence_dominant_days = constant:lawrence_event.zero }
	}
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.ascendant_edge compare = greater_than_or_equals } }
		add_to_variable = { lawrence_embedded_days = constant:lawrence_event.pulse_days }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.ascendant_exit compare = less_than } }
		set_variable = { lawrence_embedded_days = constant:lawrence_event.zero }
	}
	if = {
		limit = {
			check_variable = { var = lawrence_influence value = constant:lawrence_influence.contested_edge compare = greater_than_or_equals }
			check_variable = { var = lawrence_influence value = constant:lawrence_influence.ascendant_edge compare = less_than }
		}
		add_to_variable = { lawrence_mid_days = constant:lawrence_event.pulse_days }
	}
	else = {
		set_variable = { lawrence_mid_days = constant:lawrence_event.zero }
	}
	if = {
		limit = { check_variable = { var = lawrence_influence value = constant:lawrence_influence.restricted_edge compare = less_than } }
		add_to_variable = { lawrence_broken_days = constant:lawrence_event.pulse_days }
	}
	else = {
		set_variable = { lawrence_broken_days = constant:lawrence_event.zero }
	}
	# Dominant or embedded network: Britain offers the protectorate terms.
	if = {
		limit = {
			lawrence_is_live_target_contest = yes
			lawrence_status_is_active = yes
			NOT = { has_country_flag = lawrence_final_pending }
			NOT = { has_country_flag = lawrence_terms_refused_recently }
			ENG = { lawrence_sponsor_can_act = yes }
			OR = {
				check_variable = { var = lawrence_dominant_days value = constant:lawrence_confirm.dominant_days compare = greater_than_or_equals }
				check_variable = { var = lawrence_embedded_days value = constant:lawrence_confirm.embedded_days compare = greater_than_or_equals }
			}
		}
		set_country_flag = lawrence_final_pending
		set_variable = { lawrence_phase = constant:lawrence_phase.final }
		country_event = { id = chaosx.nr40.9 }
		ENG = { country_event = { id = chaosx.nr40.11 } }
	}
	# A mission without reach ends: Britain withdraws and keeps the contacts.
	if = {
		limit = {
			lawrence_is_live_target = yes
			check_variable = { var = lawrence_broken_days value = constant:lawrence_confirm.broken_days compare = greater_than_or_equals }
			NOT = { has_country_flag = lawrence_final_pending }
		}
		set_temp_variable = { lawrence_settle_outcome = constant:lawrence_outcome.restricted }
		lawrence_settle = yes
	}
}

# COUNTRY, the target. Baseline incidents on a Mean Time To Happen interval.
# One incident per interval, never a final settlement.
lawrence_track_incidents = {
	if = {
		limit = {
			lawrence_is_live_target_contest = yes
			lawrence_status_is_active = yes
			NOT = { has_country_flag = lawrence_final_pending }
			NOT = { has_country_flag = lawrence_incident_open }
			check_variable = { var = global.num_days value = lawrence_next_incident_day compare = greater_than_or_equals }
		}
		set_temp_variable = { lawrence_incident_delay = mtth:lawrence_incident_interval }
		set_variable = { lawrence_next_incident_day = global.num_days }
		add_to_variable = { lawrence_next_incident_day = lawrence_incident_delay }
		set_country_flag = lawrence_incident_open
		random_list = {
			40 = {
				modifier = { factor = 0.4 NOT = { has_country_flag = lawrence_file } }
				modifier = { factor = 1.5 check_variable = { var = lawrence_influence value = constant:lawrence_influence.contested_edge compare = greater_than_or_equals } }
				country_event = { id = chaosx.nr40.60 }
			}
			35 = {
				modifier = { factor = 0.3 check_variable = { var = lawrence_officers value = constant:lawrence_officers.incident_floor compare = less_than } }
				modifier = { factor = 1.6 check_variable = { var = lawrence_officers value = constant:lawrence_officers.secure_target compare = greater_than_or_equals } }
				country_event = { id = chaosx.nr40.61 }
			}
			25 = {
				modifier = { factor = 0.3 NOT = { has_country_flag = lawrence_route_open } }
				modifier = { factor = 1.4 has_country_flag = lawrence_delivery_pending }
				country_event = { id = chaosx.nr40.62 }
			}
		}
	}
}

# COUNTRY, the target. A British shipment arrives. The receipt serial makes it
# a one-time transaction. The government keeps its custody share and the rest
# moves through Lawrence's network, which raises Influence.
lawrence_receive_delivery = {
	set_variable = { lawrence_delivery_received_serial = lawrence_delivery_serial }
	clr_country_flag = lawrence_delivery_pending
	set_temp_variable = { lawrence_share = constant:lawrence_delivery.network_share }
	if = {
		limit = { has_country_flag = lawrence_delivery_custody }
		set_temp_variable = { lawrence_share = constant:lawrence_delivery.custody_share }
		if = {
			limit = { check_variable = { var = lawrence_officers value = constant:lawrence_officers.capture_threshold compare = greater_than_or_equals } }
			set_temp_variable = { lawrence_share = constant:lawrence_delivery.diverted_share }
		}
	}
	if = {
		limit = { has_country_flag = lawrence_delivery_intercepted }
		set_temp_variable = { lawrence_share = constant:lawrence_delivery.seized_share }
	}
	set_temp_variable = { lawrence_gov_infantry = lawrence_delivery_infantry }
	multiply_temp_variable = { lawrence_gov_infantry = lawrence_share }
	round_temp_variable = lawrence_gov_infantry
	set_temp_variable = { lawrence_gov_support = lawrence_delivery_support }
	multiply_temp_variable = { lawrence_gov_support = lawrence_share }
	round_temp_variable = lawrence_gov_support
	if = {
		limit = { check_variable = { var = lawrence_gov_infantry value = constant:lawrence_event.zero compare = greater_than } }
		add_equipment_to_stockpile = { type = infantry_equipment amount = lawrence_gov_infantry producer = ENG }
	}
	if = {
		limit = { check_variable = { var = lawrence_gov_support value = constant:lawrence_event.zero compare = greater_than } }
		add_equipment_to_stockpile = { type = support_equipment amount = lawrence_gov_support producer = ENG }
	}
	# The share that did not reach government depots sits with the network.
	set_temp_variable = { lawrence_change = constant:lawrence_delivery.network_influence }
	if = {
		limit = { has_country_flag = lawrence_delivery_custody }
		set_temp_variable = { lawrence_change = constant:lawrence_delivery.custody_influence }
	}
	if = {
		limit = { has_country_flag = lawrence_delivery_intercepted }
		set_temp_variable = { lawrence_change = constant:lawrence_delivery.intercepted_influence }
	}
	lawrence_add_influence = yes
	set_variable = { lawrence_support_power = constant:lawrence_support.shipment_power }
	set_variable = { lawrence_support_until = global.num_days }
	add_to_variable = { lawrence_support_until = constant:lawrence_support.shipment_days }
	set_country_flag = lawrence_route_delivered
	set_country_flag = lawrence_ach_aid_received
	clr_country_flag = lawrence_delivery_custody
	clr_country_flag = lawrence_delivery_intercepted
	set_variable = { lawrence_delivery_infantry = constant:lawrence_event.zero }
	set_variable = { lawrence_delivery_support = constant:lawrence_event.zero }
	set_global_flag = lawrence_arms_delivered
	set_temp_variable = { lawrence_report_delay = constant:lawrence_event.report_delay_days }
	country_event = { id = chaosx.nr40.13 days = lawrence_report_delay }
}
