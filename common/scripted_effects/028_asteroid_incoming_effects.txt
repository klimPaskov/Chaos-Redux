# ============================================================================
# Narrow event-owned transaction helpers for Event 028. The parent event owns
# target selection, evolution flags, presentation, decisions, state projects,
# and report dispatch. This file owns only guarded target persistence, bounded
# land-state planning, physical receipts, aggregation, dust bookkeeping,

# ANY SCOPE. Candidate country and state must already be saved as global event
# targets by the parent selector. The two-day lock is a receipt, not a fresh
# selection, and therefore does not reroll when ownership or control changes.
asteroid_incoming_lock_candidate_pair = {
	set_temp_variable = { asteroid_incoming_lock_result = constant:asteroid_incoming_runtime.zero }
	if = {
		limit = {
			asteroid_incoming_target_pair_is_valid = yes
		}
		event_target:asteroid_incoming_candidate_country = {
			save_global_event_target_as = asteroid_incoming_locked_country
		}
		event_target:asteroid_incoming_candidate_state = {
			save_global_event_target_as = asteroid_incoming_locked_state
		}
		add_to_variable = {
			global.asteroid_incoming_transaction_sequence = constant:asteroid_incoming_runtime.loop_increment
		}
		set_variable = {
			global.asteroid_incoming_transaction_id = global.asteroid_incoming_transaction_sequence
		}
		set_variable = {
			global.asteroid_incoming_lock_started_day = global.num_days
		}
		set_variable = {
			global.asteroid_incoming_lock_expires_day = global.num_days
		}
		add_to_variable = {
			global.asteroid_incoming_lock_expires_day = constant:asteroid_incoming_runtime.lock_duration_days
		}
		set_global_flag = asteroid_incoming_target_locked
		clr_global_flag = asteroid_incoming_transaction_resolved
		clr_global_flag = asteroid_incoming_transaction_failed_closed
		set_temp_variable = { asteroid_incoming_lock_result = constant:asteroid_incoming_runtime.one }
	}
}

# ANY SCOPE. Parent-owned near-miss dispatch calls this when the locked target
# fails closed. It creates no physical receipt and leaves no active lock.
asteroid_incoming_fail_closed_transaction = {
	set_global_flag = asteroid_incoming_transaction_failed_closed
	if = {
		limit = {
			has_event_target = asteroid_incoming_original_target_country
			event_target:asteroid_incoming_original_target_country = { exists = yes }
		}
		event_target:asteroid_incoming_original_target_country = {
			remove_ideas = asteroid_incoming_command_continuity
			remove_ideas = asteroid_incoming_transport_dispersion
			remove_ideas = asteroid_incoming_medical_mobilization
		}
	}
	clr_global_flag = asteroid_incoming_target_locked
	clr_global_flag = asteroid_incoming_transaction_active
	clear_global_event_target = asteroid_incoming_locked_country
	clear_global_event_target = asteroid_incoming_locked_state
}

# ANY SCOPE. Called only after every planned state and optional fragment site
# has been resolved or the parent has dispatched the fail-closed outcome.
asteroid_incoming_mark_transaction_resolved = {
	set_global_flag = asteroid_incoming_transaction_resolved
	clr_global_flag = asteroid_incoming_target_locked
}

# STATE SCOPE. Inputs: asteroid_incoming_profile_candidate and
# asteroid_incoming_ring_candidate as positive temporary values. Output:
# asteroid_incoming_profile_recorded as a temporary result. The aligned
# footprint arrays retain only the strongest profile and its shortest ring.
asteroid_incoming_record_footprint_profile = {
	set_temp_variable = { asteroid_incoming_profile_recorded = constant:asteroid_incoming_runtime.zero }
	if = {
		limit = {
			has_variable = asteroid_incoming_profile_candidate
			has_variable = asteroid_incoming_ring_candidate
			check_variable = {
				var = asteroid_incoming_profile_candidate
				value = constant:asteroid_incoming_profile.none
				compare = greater_than
			}
			asteroid_incoming_registry_arrays_are_aligned = yes
		}
		save_event_target_as = asteroid_incoming_profile_state_current
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.asteroid_incoming_footprint_states
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.asteroid_incoming_footprint_states
				value = THIS
			}
			add_to_array = {
				array = global.asteroid_incoming_footprint_profiles
				value = asteroid_incoming_profile_candidate
			}
			add_to_array = {
				array = global.asteroid_incoming_footprint_rings
				value = asteroid_incoming_ring_candidate
			}
			set_variable = { asteroid_incoming_state_profile = asteroid_incoming_profile_candidate }
			set_variable = { asteroid_incoming_state_ring = asteroid_incoming_ring_candidate }
			set_temp_variable = { asteroid_incoming_profile_recorded = constant:asteroid_incoming_runtime.one }
		}
		else = {
			set_temp_variable = { asteroid_incoming_profile_index = constant:asteroid_incoming_runtime.first_array_index }
			while_loop_effect = {
				limit = {
					check_variable = {
						var = asteroid_incoming_profile_recorded
						value = constant:asteroid_incoming_runtime.zero
						compare = equals
					}
					check_variable = {
						var = asteroid_incoming_profile_index
						value = global.asteroid_incoming_footprint_states^num
						compare = less_than
					}
				}
				var:global.asteroid_incoming_footprint_states^asteroid_incoming_profile_index = {
					if = {
						limit = { state = event_target:asteroid_incoming_profile_state_current }
						if = {
							limit = {
								check_variable = {
									var = asteroid_incoming_profile_candidate
									value = global.asteroid_incoming_footprint_profiles^asteroid_incoming_profile_index
									compare = greater_than
								}
							}
							set_variable = {
								global.asteroid_incoming_footprint_profiles^asteroid_incoming_profile_index = asteroid_incoming_profile_candidate
							}
							set_variable = {
								global.asteroid_incoming_footprint_rings^asteroid_incoming_profile_index = asteroid_incoming_ring_candidate
							}
							set_variable = { asteroid_incoming_state_profile = asteroid_incoming_profile_candidate }
							set_variable = { asteroid_incoming_state_ring = asteroid_incoming_ring_candidate }
						}
						else_if = {
							limit = {
								check_variable = {
									var = asteroid_incoming_profile_candidate
									value = global.asteroid_incoming_footprint_profiles^asteroid_incoming_profile_index
									compare = equals
								}
								check_variable = {
									var = asteroid_incoming_ring_candidate
									value = global.asteroid_incoming_footprint_rings^asteroid_incoming_profile_index
									compare = less_than
								}
							}
							set_variable = {
								global.asteroid_incoming_footprint_rings^asteroid_incoming_profile_index = asteroid_incoming_ring_candidate
							}
							set_variable = { asteroid_incoming_state_ring = asteroid_incoming_ring_candidate }
						}
						else = {
							set_variable = {
								asteroid_incoming_state_profile = global.asteroid_incoming_footprint_profiles^asteroid_incoming_profile_index
							}
							set_variable = {
								asteroid_incoming_state_ring = global.asteroid_incoming_footprint_rings^asteroid_incoming_profile_index
							}
						}
						set_temp_variable = { asteroid_incoming_profile_recorded = constant:asteroid_incoming_runtime.one }
					}
				}
				add_to_temp_variable = {
					asteroid_incoming_profile_index = constant:asteroid_incoming_runtime.loop_increment
				}
			}
		}
		if = {
			limit = {
					check_variable = {
						var = global.asteroid_incoming_planner_mode
						value = constant:asteroid_incoming_runtime.planner_main
						compare = equals
					}
				NOT = {
					is_in_array = {
						array = global.asteroid_incoming_main_footprint_states
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.asteroid_incoming_main_footprint_states
				value = THIS
			}
		}
		set_variable = { asteroid_incoming_state_transaction_id = global.asteroid_incoming_transaction_id }
		set_state_flag = asteroid_incoming_state_planned
	}
}

# ANY SCOPE. Clears the bounded planner work arrays, records the locked state
# at ring zero, and advances through rings 1, 2, and 3 by land adjacency.
asteroid_incoming_plan_main_footprint = {
	if = {
		limit = { asteroid_incoming_locked_target_pair_is_valid = yes }
		clear_array = global.asteroid_incoming_footprint_states
		clear_array = global.asteroid_incoming_footprint_profiles
		clear_array = global.asteroid_incoming_footprint_rings
		clear_array = global.asteroid_incoming_main_footprint_states
		clear_array = global.asteroid_incoming_planner_frontier_states
		clear_array = global.asteroid_incoming_planner_next_frontier_states
		clear_array = global.asteroid_incoming_planner_seen_states
		set_variable = { global.asteroid_incoming_planner_mode = constant:asteroid_incoming_runtime.planner_main }
		set_variable = { global.asteroid_incoming_planner_current_ring = constant:asteroid_incoming_runtime.first_array_index }
		event_target:asteroid_incoming_locked_state = {
			set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.main_center }
			set_temp_variable = { asteroid_incoming_ring_candidate = constant:asteroid_incoming_runtime.first_array_index }
			add_to_array = {
				array = global.asteroid_incoming_planner_seen_states
				value = THIS
			}
			add_to_array = {
				array = global.asteroid_incoming_planner_frontier_states
				value = THIS
			}
			asteroid_incoming_record_footprint_profile = yes
		}
		set_temp_variable = { asteroid_incoming_main_ring_loop_end = constant:asteroid_incoming_runtime.maximum_footprint_ring }
		add_to_temp_variable = { asteroid_incoming_main_ring_loop_end = constant:asteroid_incoming_runtime.loop_increment }
		for_loop_effect = {
			start = constant:asteroid_incoming_runtime.loop_increment
			end = asteroid_incoming_main_ring_loop_end
			value = asteroid_incoming_main_ring_loop_index
			asteroid_incoming_plan_next_land_ring = yes
		}
	}
}

# STATE SCOPE. Starts a two-ring breadth-first plan from the already accepted
# fragment center. The combined footprint remains intact so main/fragment
# overlaps are deduplicated by asteroid_incoming_record_footprint_profile.
asteroid_incoming_plan_fragment_footprint = {
	if = {
		limit = {
			has_state_flag = asteroid_incoming_fragment_site
			check_variable = {
				var = global.asteroid_incoming_fragment_site_count
				value = constant:asteroid_incoming_runtime.maximum_fragment_sites
				compare = less_than_or_equals
			}
		}
		clear_array = global.asteroid_incoming_planner_frontier_states
		clear_array = global.asteroid_incoming_planner_next_frontier_states
		clear_array = global.asteroid_incoming_planner_seen_states
		set_variable = { global.asteroid_incoming_planner_mode = constant:asteroid_incoming_runtime.planner_fragment }
		set_variable = { global.asteroid_incoming_planner_current_ring = constant:asteroid_incoming_runtime.first_array_index }
		set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.fragment_center }
		set_temp_variable = { asteroid_incoming_ring_candidate = constant:asteroid_incoming_runtime.first_array_index }
		add_to_array = {
			array = global.asteroid_incoming_planner_seen_states
			value = THIS
		}
		add_to_array = {
			array = global.asteroid_incoming_planner_frontier_states
			value = THIS
		}
		asteroid_incoming_record_footprint_profile = yes
		set_temp_variable = { asteroid_incoming_fragment_ring_loop_end = constant:asteroid_incoming_runtime.maximum_fragment_ring }
		add_to_temp_variable = { asteroid_incoming_fragment_ring_loop_end = constant:asteroid_incoming_runtime.loop_increment }
		for_loop_effect = {
			start = constant:asteroid_incoming_runtime.loop_increment
			end = asteroid_incoming_fragment_ring_loop_end
			value = asteroid_incoming_fragment_ring_loop_index
			asteroid_incoming_plan_next_land_ring = yes
		}
	}
}

# ANY SCOPE. One breadth-first land ring. Maritime adjacency is never used:
# every_neighbor_state follows the map's land-state graph and candidates are
# only appended after the valid-state predicate and seen-set guard pass.
asteroid_incoming_plan_next_land_ring = {
	set_temp_variable = { asteroid_incoming_next_ring = global.asteroid_incoming_planner_current_ring }
	add_to_temp_variable = { asteroid_incoming_next_ring = constant:asteroid_incoming_runtime.loop_increment }
	clear_array = global.asteroid_incoming_planner_next_frontier_states
	for_each_scope_loop = {
		array = global.asteroid_incoming_planner_frontier_states
		every_neighbor_state = {
			limit = { asteroid_incoming_state_is_valid_ring_candidate = yes }
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = global.asteroid_incoming_planner_seen_states
							value = THIS
						}
					}
				}
				add_to_array = {
					array = global.asteroid_incoming_planner_seen_states
					value = THIS
				}
				add_to_array = {
					array = global.asteroid_incoming_planner_next_frontier_states
					value = THIS
				}
				set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.none }
				if = {
					limit = {
					check_variable = {
						var = global.asteroid_incoming_planner_mode
						value = constant:asteroid_incoming_runtime.planner_main
						compare = equals
					}
					}
					if = {
						limit = { check_variable = { var = asteroid_incoming_next_ring value = constant:asteroid_incoming_runtime.ring_one compare = equals } }
						set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.main_ring_one }
					}
					else_if = {
						limit = { check_variable = { var = asteroid_incoming_next_ring value = constant:asteroid_incoming_runtime.ring_two compare = equals } }
						set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.main_ring_two }
					}
					else_if = {
						limit = { check_variable = { var = asteroid_incoming_next_ring value = constant:asteroid_incoming_runtime.ring_three compare = equals } }
						set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.main_ring_three }
					}
				}
				else_if = {
					limit = {
						check_variable = {
							var = global.asteroid_incoming_planner_mode
							value = constant:asteroid_incoming_runtime.planner_fragment
							compare = equals
						}
					}
					if = {
						limit = { check_variable = { var = asteroid_incoming_next_ring value = constant:asteroid_incoming_runtime.ring_one compare = equals } }
						set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.fragment_ring_one }
					}
					else_if = {
						limit = { check_variable = { var = asteroid_incoming_next_ring value = constant:asteroid_incoming_runtime.ring_two compare = equals } }
						set_temp_variable = { asteroid_incoming_profile_candidate = constant:asteroid_incoming_profile.fragment_ring_two }
					}
				}
				set_temp_variable = { asteroid_incoming_ring_candidate = asteroid_incoming_next_ring }
				asteroid_incoming_record_footprint_profile = yes
			}
		}
	}
	clear_array = global.asteroid_incoming_planner_frontier_states
	for_each_scope_loop = {
		array = global.asteroid_incoming_planner_next_frontier_states
		add_to_array = {
			array = global.asteroid_incoming_planner_frontier_states
			value = THIS
		}
	}
	clear_array = global.asteroid_incoming_planner_next_frontier_states
	set_variable = { global.asteroid_incoming_planner_current_ring = asteroid_incoming_next_ring }
}

# STATE SCOPE. Reads asteroid_incoming_state_profile and outputs four
# temporary fractions: asteroid_incoming_population_fraction,
# asteroid_incoming_industry_fraction, asteroid_incoming_logistics_fraction,
# and asteroid_incoming_strategic_fraction. Unknown profiles stay at zero.
asteroid_incoming_load_profile_fractions = {

	set_temp_variable = { asteroid_incoming_industry_fraction = constant:asteroid_incoming_dust.minimum }
	set_temp_variable = { asteroid_incoming_logistics_fraction = constant:asteroid_incoming_dust.minimum }
	set_temp_variable = { asteroid_incoming_strategic_fraction = constant:asteroid_incoming_dust.minimum }
	if = {
		limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_center compare = equals } }
		set_temp_variable = { asteroid_incoming_population_fraction = constant:asteroid_incoming_population_loss.main_center }
		set_temp_variable = { asteroid_incoming_industry_fraction = constant:asteroid_incoming_building_loss.main_center_industry }
		set_temp_variable = { asteroid_incoming_logistics_fraction = constant:asteroid_incoming_building_loss.main_center_logistics }
		set_temp_variable = { asteroid_incoming_strategic_fraction = constant:asteroid_incoming_building_loss.main_center_strategic }
	}
	else_if = {
		limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_one compare = equals } }
		set_temp_variable = { asteroid_incoming_population_fraction = constant:asteroid_incoming_population_loss.main_ring_one }
		set_temp_variable = { asteroid_incoming_industry_fraction = constant:asteroid_incoming_building_loss.main_ring_one_industry }
		set_temp_variable = { asteroid_incoming_logistics_fraction = constant:asteroid_incoming_building_loss.main_ring_one_logistics }
		set_temp_variable = { asteroid_incoming_strategic_fraction = constant:asteroid_incoming_building_loss.main_ring_one_strategic }
	}

# STATE SCOPE. Applies the population fraction through the existing shared
# population transaction helper, then measures the state again. The observed
# loss is the only value sent to the shared Deaths register, with
# chaos_deaths_apply_state_pop set to zero so the state is never debited twice.
asteroid_incoming_apply_population_receipt = {
	set_variable = { asteroid_incoming_state_population_before_k = state_population_k }
	set_variable = { asteroid_incoming_state_population_before_people = state_population_k }
	multiply_variable = {
		asteroid_incoming_state_population_before_people = constant:chaos_meter_deaths.people_per_k
	}
	set_variable = { asteroid_incoming_state_population_after_k = state_population_k }
	set_variable = { asteroid_incoming_state_population_after_people = state_population_k }
	multiply_variable = {
		asteroid_incoming_state_population_after_people = constant:chaos_meter_deaths.people_per_k
	}
	set_temp_variable = { asteroid_incoming_population_available = state_population_k }
	multiply_temp_variable = {
		asteroid_incoming_population_available = constant:chaos_meter_deaths.people_per_k
	}
	set_temp_variable = { asteroid_incoming_population_requested = asteroid_incoming_population_available }
	multiply_temp_variable = {
		asteroid_incoming_population_requested = asteroid_incoming_population_fraction
	}
	clamp_temp_variable = {
		var = asteroid_incoming_population_requested
		min = constant:asteroid_incoming_dust.minimum
		max = asteroid_incoming_population_available
	}
	round_temp_variable = asteroid_incoming_population_requested
	set_temp_variable = { state_population_transaction_loss = asteroid_incoming_population_requested }
	set_temp_variable = { state_population_transaction_contract_supplied = constant:asteroid_incoming_runtime.one }
	apply_state_population_loss_without_recruitable_manpower_gain = yes
	set_variable = { asteroid_incoming_state_population_after_k = state_population_k }
	set_variable = { asteroid_incoming_state_population_after_people = state_population_k }
	multiply_variable = {
		asteroid_incoming_state_population_after_people = constant:chaos_meter_deaths.people_per_k
	}
	set_variable = { asteroid_incoming_state_population_lost_k = asteroid_incoming_state_population_before_k }
	subtract_from_variable = {
		asteroid_incoming_state_population_lost_k = asteroid_incoming_state_population_after_k
	}
	clamp_variable = {
		var = asteroid_incoming_state_population_lost_k
		min = constant:asteroid_incoming_dust.minimum
		max = asteroid_incoming_state_population_before_k
	}
	set_variable = { asteroid_incoming_state_actual_deaths = asteroid_incoming_state_population_before_people }
	subtract_from_variable = {
		asteroid_incoming_state_actual_deaths = asteroid_incoming_state_population_after_people
	}
	clamp_variable = {
		var = asteroid_incoming_state_actual_deaths
		min = constant:asteroid_incoming_dust.minimum
		max = asteroid_incoming_state_population_before_people
	}
	set_variable = { asteroid_incoming_state_population_lost_people = asteroid_incoming_state_actual_deaths }
	if = {
		limit = { check_variable = { var = asteroid_incoming_state_actual_deaths value = constant:asteroid_incoming_dust.minimum compare = greater_than } }
		set_temp_variable = { chaos_deaths_change = asteroid_incoming_state_actual_deaths }
		set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.asteroid_main_impact }
		if = {
			limit = { has_variable = asteroid_incoming_death_reason }
			set_temp_variable = { chaos_deaths_reason = asteroid_incoming_death_reason }
		}
		set_temp_variable = { chaos_deaths_is_civilian = constant:asteroid_incoming_runtime.one }
		set_temp_variable = { chaos_deaths_apply_state_pop = constant:asteroid_incoming_runtime.zero }
		set_temp_variable = { chaos_deaths_record_state_ledger = constant:asteroid_incoming_runtime.one }
		set_temp_variable = { chaos_deaths_target_country = OWNER }
		set_temp_variable = { chaos_deaths_has_target_country = constant:asteroid_incoming_runtime.one }
		chaos_meter_register_deaths = yes
	}
}

# STATE SCOPE. Inputs: asteroid_incoming_selected_building as a token,
# asteroid_incoming_selected_before, and asteroid_incoming_selected_fraction.
# Output: asteroid_incoming_remove_levels. This is an event-owned dynamic
# adapter around the supported remove_building effect; it does not call any
# nuclear or fallout effect.
asteroid_incoming_prepare_selected_building_loss = {
	set_temp_variable = { asteroid_incoming_remove_levels = asteroid_incoming_selected_before }
	multiply_temp_variable = {
		asteroid_incoming_remove_levels = asteroid_incoming_selected_fraction
	}
	clamp_temp_variable = {
		var = asteroid_incoming_remove_levels
		min = constant:asteroid_incoming_dust.minimum
		max = asteroid_incoming_selected_before
	}
	round_temp_variable = asteroid_incoming_remove_levels
}

# STATE SCOPE. Uses the token supplied by the caller and the already prepared
# positive level count. The meta effect is limited to building type injection;
# the before/after receipt remains explicit at each supported building call
# site so actual levels, rather than requested levels, are recorded.
asteroid_incoming_remove_selected_building_levels = {
	if = {
		limit = {
			has_variable = asteroid_incoming_selected_building
			check_variable = {
				var = asteroid_incoming_remove_levels
				value = constant:asteroid_incoming_dust.minimum
				compare = greater_than
			}
		}
		set_temp_variable = { asteroid_incoming_selected_state_id = THIS.id }
		set_temp_variable = { asteroid_incoming_single_building_level = constant:asteroid_incoming_runtime.one }
		for_loop_effect = {
			start = constant:asteroid_incoming_runtime.zero
			end = asteroid_incoming_remove_levels
			value = asteroid_incoming_remove_building_loop_index
			meta_effect = {
				text = {
					remove_building = {
						type = [BUILDING_TYPE]
						level = asteroid_incoming_single_building_level
						state = var:asteroid_incoming_selected_state_id
					}
				}
				BUILDING_TYPE = "[?asteroid_incoming_selected_building.GetTokenKey]"
			}
		}
	}
}

# STATE SCOPE. Inputs: asteroid_incoming_selected_before and
# asteroid_incoming_selected_after. Output: asteroid_incoming_selected_actual
# and the cumulative asteroid_incoming_state_building_levels_lost.
asteroid_incoming_capture_selected_building_receipt = {
	set_temp_variable = { asteroid_incoming_selected_actual = asteroid_incoming_selected_before }
	subtract_from_temp_variable = {
		asteroid_incoming_selected_actual = asteroid_incoming_selected_after
	}
	clamp_temp_variable = {
		var = asteroid_incoming_selected_actual
		min = constant:asteroid_incoming_dust.minimum
		max = asteroid_incoming_selected_before
	}
	add_to_variable = {
		asteroid_incoming_state_building_levels_lost = asteroid_incoming_selected_actual
	}
}

# STATE SCOPE. Applies one deduplicated profile. Optional parent-owned state
# variables asteroid_incoming_state_capital_relocated and
# asteroid_incoming_state_rescue_burden are preserved for the later country
# aggregation pass. Clear state projects, persistent crater modifiers, and
# other presentation effects remain parent-owned.
asteroid_incoming_apply_state_transaction = {
	set_variable = { asteroid_incoming_state_population_lost_k = constant:asteroid_incoming_dust.minimum }
	set_variable = { asteroid_incoming_state_population_lost_people = constant:asteroid_incoming_dust.minimum }
	set_variable = { asteroid_incoming_state_actual_deaths = constant:asteroid_incoming_dust.minimum }
	set_variable = { asteroid_incoming_state_industry_levels_lost = constant:asteroid_incoming_dust.minimum }
	set_variable = { asteroid_incoming_state_logistics_levels_lost = constant:asteroid_incoming_dust.minimum }
	set_variable = { asteroid_incoming_state_strategic_levels_lost = constant:asteroid_incoming_dust.minimum }
	set_variable = { asteroid_incoming_state_building_levels_lost = constant:asteroid_incoming_dust.minimum }
	if = {
		limit = { asteroid_incoming_state_is_resolvable_transaction = yes }
		asteroid_incoming_load_profile_fractions = yes
		asteroid_incoming_apply_population_receipt = yes
		asteroid_incoming_apply_building_receipt = yes
		if = {
			limit = {
				check_variable = {
					var = asteroid_incoming_state_profile
					value = constant:asteroid_incoming_profile.main_center
					compare = equals
				}
			}
			set_state_flag = asteroid_incoming_main_crater
			set_state_flag = asteroid_incoming_mineral_site
			set_global_flag = asteroid_incoming_main_crater_registered
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = global.asteroid_incoming_mineral_site_states
							value = THIS
						}
					}
				}
				add_to_array = {
					array = global.asteroid_incoming_mineral_site_states
					value = THIS
				}
			}
			asteroid_incoming_refresh_site_mineral_control = yes
		}
		set_state_flag = asteroid_incoming_state_transaction_applied
		set_variable = {
			asteroid_incoming_state_transaction_id = global.asteroid_incoming_transaction_id
		}
	}
}

	if = {

	}
	add_to_variable = {
		asteroid_incoming_country_main_state_count = asteroid_incoming_aggregate_main_state_count
	}
	add_to_variable = {

	}

		asteroid_incoming_country_rescue_burden = asteroid_incoming_aggregate_rescue_burden
	}

		limit = {
			check_variable = {
				var = asteroid_incoming_aggregate_worst_profile
				value = asteroid_incoming_country_worst_profile
				compare = greater_than
			}

			asteroid_incoming_country_worst_profile = asteroid_incoming_aggregate_worst_profile

				compare = greater_than
			}
		}

		clr_country_flag = asteroid_incoming_fragment_report_pending
	}
	else = {

	}
	set_country_flag = asteroid_incoming_emergency_phase
}

# STATE SCOPE. Adds this state's observed receipt to its owner and, when
# occupied, its controller. The optional capital and rescue values are
# parent-owned inputs and default to zero when absent.
asteroid_incoming_aggregate_current_state = {
	if = {
		limit = {
			has_state_flag = asteroid_incoming_state_transaction_applied
			NOT = { has_state_flag = asteroid_incoming_state_country_aggregated }
		}
		set_temp_variable = { asteroid_incoming_aggregate_population_lost = asteroid_incoming_state_population_lost_people }
		set_temp_variable = { asteroid_incoming_aggregate_actual_deaths = asteroid_incoming_state_actual_deaths }
		set_temp_variable = { asteroid_incoming_aggregate_building_levels_lost = asteroid_incoming_state_building_levels_lost }
		set_temp_variable = { asteroid_incoming_aggregate_industry_levels_lost = asteroid_incoming_state_industry_levels_lost }
		set_temp_variable = { asteroid_incoming_aggregate_logistics_levels_lost = asteroid_incoming_state_logistics_levels_lost }
		set_temp_variable = { asteroid_incoming_aggregate_strategic_levels_lost = asteroid_incoming_state_strategic_levels_lost }
		set_temp_variable = { asteroid_incoming_aggregate_main_center_count = constant:asteroid_incoming_dust.minimum }
		set_temp_variable = { asteroid_incoming_aggregate_main_state_count = constant:asteroid_incoming_dust.minimum }
		set_temp_variable = { asteroid_incoming_aggregate_fragment_center_count = constant:asteroid_incoming_dust.minimum }
		if = {
			limit = {
				OR = {
					check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_center compare = equals }
					check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_one compare = equals }
					check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_two compare = equals }
					check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_three compare = equals }
				}
			}
			set_temp_variable = { asteroid_incoming_aggregate_main_state_count = constant:asteroid_incoming_runtime.one }
		}
		if = {
			limit = {
				check_variable = {
					var = asteroid_incoming_state_profile
					value = constant:asteroid_incoming_profile.main_center
					compare = equals
				}
			}
			set_temp_variable = { asteroid_incoming_aggregate_main_center_count = constant:asteroid_incoming_runtime.one }
		}
		if = {
			limit = {
				check_variable = {
					var = asteroid_incoming_state_profile
					value = constant:asteroid_incoming_profile.fragment_center
					compare = equals
				}
			}
			set_temp_variable = { asteroid_incoming_aggregate_fragment_center_count = constant:asteroid_incoming_runtime.one }
		}
		set_temp_variable = { asteroid_incoming_aggregate_capital_relocated = constant:asteroid_incoming_dust.minimum }
		if = {
			limit = { has_variable = asteroid_incoming_state_capital_relocated }
			set_temp_variable = { asteroid_incoming_aggregate_capital_relocated = asteroid_incoming_state_capital_relocated }
		}
		set_temp_variable = { asteroid_incoming_aggregate_rescue_burden = constant:asteroid_incoming_dust.minimum }
		if = {
			limit = { has_variable = asteroid_incoming_state_rescue_burden }
			set_temp_variable = { asteroid_incoming_aggregate_rescue_burden = asteroid_incoming_state_rescue_burden }
		}
		set_temp_variable = { asteroid_incoming_aggregate_worst_profile = asteroid_incoming_state_profile }
		OWNER = {
			if = {
				limit = { exists = yes }
				asteroid_incoming_add_state_receipt_to_country = yes
			}
		}
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = {
				if = {
					limit = { exists = yes }
					asteroid_incoming_add_state_receipt_to_country = yes
				}
			}
		}
		set_state_flag = asteroid_incoming_state_country_aggregated
	}
}

# COUNTRY SCOPE. Binds one exact worst affected state for a consolidated
# report. The saved state is preferred under current control, then ownership,
# and finally any affected state that still exists after occupation changes.
asteroid_incoming_bind_report_worst_state = {
	if = {
		limit = {
			any_controlled_state = {
				has_state_flag = asteroid_incoming_affected_state
				check_variable = {
					var = asteroid_incoming_state_profile
					value = PREV.asteroid_incoming_country_worst_profile
					compare = equals
				}
			}
		}
		random_controlled_state = {
			limit = {
				has_state_flag = asteroid_incoming_affected_state
				check_variable = {
					var = asteroid_incoming_state_profile
					value = PREV.asteroid_incoming_country_worst_profile
					compare = equals
				}
			}
			save_event_target_as = asteroid_incoming_report_worst_state
		}
	}
	else_if = {
		limit = {
			any_owned_state = {
				has_state_flag = asteroid_incoming_affected_state
				check_variable = {
					var = asteroid_incoming_state_profile
					value = PREV.asteroid_incoming_country_worst_profile
					compare = equals
				}
			}
		}
		random_owned_state = {
			limit = {
				has_state_flag = asteroid_incoming_affected_state
				check_variable = {
					var = asteroid_incoming_state_profile
					value = PREV.asteroid_incoming_country_worst_profile
					compare = equals
				}
			}
			save_event_target_as = asteroid_incoming_report_worst_state
		}
	}
	else_if = {
		limit = { any_controlled_state = { has_state_flag = asteroid_incoming_affected_state } }
		random_controlled_state = {
			limit = { has_state_flag = asteroid_incoming_affected_state }
			save_event_target_as = asteroid_incoming_report_worst_state
		}
	}
	else_if = {
		limit = { any_owned_state = { has_state_flag = asteroid_incoming_affected_state } }
		random_owned_state = {
			limit = { has_state_flag = asteroid_incoming_affected_state }
			save_event_target_as = asteroid_incoming_report_worst_state
		}
	}
}

# STATE SCOPE. Tentatively registers an accepted fragment center and reserves
# its first-ring states against later centers. The parent commits the complete
# batch to the footprint and mineral registries only after the minimum viable
# center count has been reached.
asteroid_incoming_register_fragment_site = {
	set_temp_variable = { asteroid_incoming_fragment_registration_result = constant:asteroid_incoming_runtime.zero }
	if = {
		limit = { asteroid_incoming_state_is_valid_fragment_site = yes }
		add_to_variable = {
			global.asteroid_incoming_fragment_site_count = constant:asteroid_incoming_runtime.loop_increment
		}
		clamp_variable = {
			var = global.asteroid_incoming_fragment_site_count
			min = constant:asteroid_incoming_runtime.zero
			max = constant:asteroid_incoming_runtime.maximum_fragment_sites
		}
		if = {
			limit = { check_variable = { var = global.asteroid_incoming_fragment_site_count value = constant:asteroid_incoming_runtime.fragment_slot_one compare = equals } }
			save_global_event_target_as = asteroid_incoming_fragment_1_state
		}
		else_if = {
			limit = { check_variable = { var = global.asteroid_incoming_fragment_site_count value = constant:asteroid_incoming_runtime.fragment_slot_two compare = equals } }
			save_global_event_target_as = asteroid_incoming_fragment_2_state
		}
		else_if = {
			limit = { check_variable = { var = global.asteroid_incoming_fragment_site_count value = constant:asteroid_incoming_runtime.fragment_slot_three compare = equals } }
			save_global_event_target_as = asteroid_incoming_fragment_3_state
		}
		else_if = {
			limit = { check_variable = { var = global.asteroid_incoming_fragment_site_count value = constant:asteroid_incoming_runtime.fragment_slot_four compare = equals } }
			save_global_event_target_as = asteroid_incoming_fragment_4_state
		}
		else_if = {
			limit = { check_variable = { var = global.asteroid_incoming_fragment_site_count value = constant:asteroid_incoming_runtime.fragment_slot_five compare = equals } }
			save_global_event_target_as = asteroid_incoming_fragment_5_state
		}
		else_if = {
			limit = { check_variable = { var = global.asteroid_incoming_fragment_site_count value = constant:asteroid_incoming_runtime.fragment_slot_six compare = equals } }
			save_global_event_target_as = asteroid_incoming_fragment_6_state
		}
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.asteroid_incoming_fragment_site_states
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.asteroid_incoming_fragment_site_states
				value = THIS
			}
		}
		OWNER = {
			if = {
				limit = {
					exists = yes
					NOT = {
						is_in_array = {
							array = global.asteroid_incoming_fragment_site_countries
							value = THIS
						}
					}
				}
				add_to_array = {
					array = global.asteroid_incoming_fragment_site_countries
					value = THIS
				}
			}
		}
		every_neighbor_state = {
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = global.asteroid_incoming_fragment_reserved_first_ring_states
							value = THIS
						}
					}
				}
				add_to_array = {
					array = global.asteroid_incoming_fragment_reserved_first_ring_states
					value = THIS
				}
			}
		}
		set_temp_variable = { asteroid_incoming_fragment_registration_result = constant:asteroid_incoming_runtime.one }
	}
}

# STATE SCOPE. Refreshes the persistent controller pointer for a registered
# mineral site. Country ideas are rebuilt by the parent-owned registry refresh;

		clamp_temp_variable = {
			var = asteroid_incoming_dust_state_remaining
			min = constant:asteroid_incoming_dust.minimum
		}
		clamp_temp_variable = {
			var = asteroid_incoming_dust_state_contribution
			max = asteroid_incoming_dust_state_remaining
		}
		add_to_variable = {
			global.asteroid_incoming_dust_state_points = asteroid_incoming_dust_state_contribution
		}
		add_to_variable = {
			global.asteroid_incoming_dust_raw = asteroid_incoming_dust_state_contribution
		}

		if = {
			limit = { NOT = { has_variable = global.asteroid_incoming_dust_fragment_points } }
			set_variable = { global.asteroid_incoming_dust_fragment_points = constant:asteroid_incoming_dust.minimum }
		}
		if = {
			limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.fragment_center compare = equals } }
			set_temp_variable = { asteroid_incoming_dust_fragment_remaining = constant:asteroid_incoming_dust.fragment_point_cap }
			subtract_from_temp_variable = {
				asteroid_incoming_dust_fragment_remaining = global.asteroid_incoming_dust_fragment_points
			}
			clamp_temp_variable = {
				var = asteroid_incoming_dust_fragment_remaining
				min = constant:asteroid_incoming_dust.minimum
			}
			set_temp_variable = { asteroid_incoming_dust_fragment_contribution = constant:asteroid_incoming_dust.fragment_center_point }
			clamp_temp_variable = {
				var = asteroid_incoming_dust_fragment_contribution
				max = asteroid_incoming_dust_fragment_remaining
			}
			add_to_variable = {
				global.asteroid_incoming_dust_fragment_points = asteroid_incoming_dust_fragment_contribution
			}
			add_to_variable = {
				global.asteroid_incoming_dust_raw = asteroid_incoming_dust_fragment_contribution
			}
		}
		if = {
			limit = {
				check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_center compare = equals }
				NOT = { has_global_flag = asteroid_incoming_main_crater_dust_counted }
			}
			add_to_variable = {
				global.asteroid_incoming_dust_raw = constant:asteroid_incoming_dust.main_crater_base
			}
			set_global_flag = asteroid_incoming_main_crater_dust_counted
		}
		set_state_flag = asteroid_incoming_state_dust_counted
	}
}
