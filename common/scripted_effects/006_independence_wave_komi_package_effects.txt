independence_wave_refresh_komi_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_komi_package = yes }
		remove_ideas = { komi_fragmented_taiga_mandate komi_northern_republic_compact }
		if = {
			limit = { has_stable_independence_wave_komi_compact = yes }
			add_ideas = komi_northern_republic_compact
			set_country_flag = independence_wave_komi_compact_stabilized
		}
		else = {
			add_ideas = komi_fragmented_taiga_mandate

		}
	}
}
