independence_wave_refresh_sco_state_lifecycle = {
	if = {
		limit = { is_independence_wave_sco_package = yes }
		remove_ideas = { sco_divided_coastal_command sco_north_atlantic_state_service }
		if = {
			limit = { has_stable_independence_wave_sco_state = yes }
			add_ideas = sco_north_atlantic_state_service
			set_country_flag = independence_wave_sco_state_service_secured
		}
		else = {
			add_ideas = sco_divided_coastal_command
			clr_country_flag = independence_wave_sco_state_service_secured
		}
	}
}

independence_wave_change_sco_shipping_authority = {
	if = {
		limit = { is_independence_wave_sco_package = yes }
		add_to_variable = { independence_wave_sco_shipping_authority = independence_wave_sco_shipping_delta }
		clamp_variable = {
			var = independence_wave_sco_shipping_authority
