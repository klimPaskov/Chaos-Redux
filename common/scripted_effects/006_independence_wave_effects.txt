# Refreshes the state-strip frame indices used by the Statehood Ledger. The
# sheets are not free-running semantic animations: each frame is selected from
# the live ledger state so recognition, dependency, league, and formable art
# cannot display a contradictory phase.
independence_wave_refresh_status_frame_state = {
	if = {
		limit = { is_independence_wave_package_content_active = yes }
		set_variable = { independence_wave_gui_recognition_frame = constant:independence_wave_gui_frame.recognition_unrecognized }
		set_variable = { independence_wave_gui_dependency_frame = constant:independence_wave_gui_frame.dependency_calm }
		set_variable = { independence_wave_gui_league_frame = constant:independence_wave_gui_frame.league_rest }
		set_variable = { independence_wave_gui_formable_frame = constant:independence_wave_gui_frame.formable_hidden }
		if = {
			limit = { check_variable = { var = independence_wave_recognition value = constant:independence_wave_recognition_band.observed compare = greater_than_or_equals } }
			set_variable = { independence_wave_gui_recognition_frame = constant:independence_wave_gui_frame.recognition_observed }
		}
		if = {
			limit = { check_variable = { var = independence_wave_recognition value = constant:independence_wave_recognition_band.de_facto compare = greater_than_or_equals } }
			set_variable = { independence_wave_gui_recognition_frame = constant:independence_wave_gui_frame.recognition_de_facto }
		}
		if = {
			limit = { check_variable = { var = independence_wave_recognition value = constant:independence_wave_recognition_band.treaty_backed compare = greater_than_or_equals } }
			set_variable = { independence_wave_gui_recognition_frame = constant:independence_wave_gui_frame.recognition_treaty_backed }
		}
		if = {
			limit = { check_variable = { var = independence_wave_recognition value = constant:independence_wave_recognition_band.internationally_entrenched compare = greater_than_or_equals } }
			set_variable = { independence_wave_gui_recognition_frame = constant:independence_wave_gui_frame.recognition_entrenched }
		}
		if = {
			limit = { has_independence_wave_patron_warning = yes }
			set_variable = { independence_wave_gui_dependency_frame = constant:independence_wave_gui_frame.dependency_watch }
		}
		if = {
			limit = { has_independence_wave_severe_instability = yes }
			set_variable = { independence_wave_gui_dependency_frame = constant:independence_wave_gui_frame.dependency_danger }
		}
		# League phases are a state machine, not a monotonic progress bar. Keep
		# failed congresses and dissolved networks on the rest frame, and map
		# only the explicit live phase groups to the authored four-frame strip.
		if = {
			limit = {
				has_variable = global.independence_wave_league_phase
				OR = {
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.regional_conferences }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.congress_preparation }
				}
			}
			set_variable = { independence_wave_gui_league_frame = constant:independence_wave_gui_frame.league_drafting }
		}
		if = {
			limit = {
				has_variable = global.independence_wave_league_phase
				check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.charter_vote }
			}
			set_variable = { independence_wave_gui_league_frame = constant:independence_wave_gui_frame.league_vote }
		}
		if = {
			limit = {
				has_variable = global.independence_wave_league_phase
				OR = {
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.consultative_league }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.formal_league }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.durable_league }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.league_crisis }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.reformed_league }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.rival_leagues }
				}
			}
			set_variable = { independence_wave_gui_league_frame = constant:independence_wave_gui_frame.league_activated }
		}
		if = {
			limit = { has_country_flag = independence_wave_formable_discovered }
			set_variable = { independence_wave_gui_formable_frame = constant:independence_wave_gui_frame.formable_discovered }
		}
		if = {
			limit = { has_country_flag = independence_wave_formable_all_required_initial_integration_complete }
			set_variable = { independence_wave_gui_formable_frame = constant:independence_wave_gui_frame.formable_eligible }
		}
		if = {
			limit = {
				has_variable = independence_wave_formable_transaction_state
				check_variable = { independence_wave_formable_transaction_state = constant:independence_wave_formable_transaction.committed }
			}
			set_variable = { independence_wave_gui_formable_frame = constant:independence_wave_gui_frame.formable_proclaimed }
		}
	}
}

# Country scope. Records one factual charter-expulsion ground at the exact
# transaction point. Callers set the temporary ground enum and a zero/one
# `independence_wave_expulsion_has_target_input`; when that input is one they
# also set `independence_wave_expulsion_evidence_target_input` to a surviving
# country scope. The accused country owns the case, violation count, and date.
# A second recorded breach promotes the case to the repeated-violation ground.
independence_wave_record_league_expulsion_ground = {
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_country_flag = independence_wave_league_member
			has_global_flag = independence_wave_charter_anti_puppetry
			OR = {
				check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.formal_league }
				check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.durable_league }
				check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.reformed_league }
			}
		}
		if = {
			limit = { NOT = { has_variable = independence_wave_charter_violation_count } }
			set_variable = { independence_wave_charter_violation_count = constant:independence_wave_value.minimum }
		}
		add_to_variable = { independence_wave_charter_violation_count = constant:independence_wave_plan.loop_increment }
		set_variable = { independence_wave_expulsion_ground = independence_wave_expulsion_ground_input }
		set_variable = { independence_wave_expulsion_ground_date = global.date }
		set_country_flag = independence_wave_expulsion_ground_recorded
		clr_country_flag = independence_wave_expulsion_resolved
		if = {
			limit = { check_variable = { independence_wave_expulsion_has_target_input = constant:independence_wave_plan.loop_increment } }
			set_variable = { independence_wave_expulsion_evidence_target = independence_wave_expulsion_evidence_target_input }
		}
		else = { clear_variable = independence_wave_expulsion_evidence_target }
		if = {
			limit = {
				check_variable = {
					var = independence_wave_charter_violation_count
					value = constant:independence_wave_decision_gate.repeated_charter_violation_threshold
					compare = greater_than_or_equals
				}
			}
			set_variable = { independence_wave_expulsion_ground = constant:independence_wave_expulsion_ground.repeated_charter_violation }
			set_country_flag = independence_wave_expulsion_repeated
		}
	}
}

# Country scope. Clears generation-local case markers while retaining no stale
# target pointer, ground enum, date, or violation count for a later Event 006
# generation.
independence_wave_clear_league_expulsion_case = {
	clr_country_flag = independence_wave_expulsion_ground_recorded
	clr_country_flag = independence_wave_expulsion_repeated
	clr_country_flag = independence_wave_expulsion_resolved
	clear_variable = independence_wave_charter_violation_count
	clear_variable = independence_wave_expulsion_ground
	clear_variable = independence_wave_expulsion_ground_date
	clear_variable = independence_wave_expulsion_evidence_target
}

# Country scope. DM-61's coup action uses the vanilla civil-war effect, with a
# route-selected opposing ideology and a central size constant injected via a
# meta effect because `start_civil_war.size` is not a documented constant field.
independence_wave_start_sponsored_member_coup = {
	set_temp_variable = { independence_wave_coup_size = constant:independence_wave_decision_coup.revolt_size }

		limit = { has_government = fascism }
		meta_effect = {
			text = {
				start_civil_war = {
					ideology = democratic
					ruling_party = fascism
					size = [IW_COUP_SIZE]
				}
			}
			IW_COUP_SIZE = "[?independence_wave_coup_size|.2]"
		}
	}
	else_if = {
		limit = { has_government = democratic }
		meta_effect = {
			text = {
				start_civil_war = {
					ideology = fascism
					ruling_party = democratic
					size = [IW_COUP_SIZE]
				}
			}
			IW_COUP_SIZE = "[?independence_wave_coup_size|.2]"
		}
	}
	else_if = {
		limit = { has_government = communism }
		meta_effect = {
			text = {
				start_civil_war = {
					ideology = democratic
					ruling_party = communism
					size = [IW_COUP_SIZE]
				}
			}
			IW_COUP_SIZE = "[?independence_wave_coup_size|.2]"
		}
	}
	else = {
		meta_effect = {
			text = {
				start_civil_war = {
					ideology = fascism
					ruling_party = neutrality
					size = [IW_COUP_SIZE]

}

# Input: temporary `independence_wave_government_route_input`. Package-aware
# availability is checked by the route decision/focus before this lock effect.
independence_wave_select_government_route = {
	if = {
		limit = {
			is_independence_wave_package_content_active = yes
			has_valid_independence_wave_government_route = yes
			check_variable = { independence_wave_government_route = constant:independence_wave_government_route.undecided }
		}
		set_variable = { independence_wave_government_route = independence_wave_government_route_input }
		set_variable = { independence_wave_government_route_lock_date = global.date }
		set_country_flag = independence_wave_government_route_locked
		# The generic tree is a real political route, not only a ledger label.
		# Apply the shared ideological posture first; researched package adapters
		# below may then restore their documented carrier-specific party surface.
		independence_wave_apply_generic_government_route_politics = yes
		if = { limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.constitutional_republic } } set_country_flag = independence_wave_route_constitutional_republic }
		else_if = { limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.popular_council } } set_country_flag = independence_wave_route_popular_council }
		else_if = { limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.traditional_restoration } } set_country_flag = independence_wave_route_traditional_restoration }
		else_if = { limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.emergency_military } } set_country_flag = independence_wave_route_emergency_military }
		else_if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.patron_client } }
			set_country_flag = independence_wave_route_patron_client
			if = {
				limit = {
					has_country_flag = independence_wave_league_member
					has_global_flag = independence_wave_charter_anti_puppetry
					OR = {
						check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.formal_league }
						check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.durable_league }
						check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.reformed_league }
					}
				}
				set_temp_variable = { independence_wave_expulsion_ground_input = constant:independence_wave_expulsion_ground.patron_client }
				set_temp_variable = { independence_wave_expulsion_has_target_input = constant:independence_wave_value.minimum }
				independence_wave_record_league_expulsion_ground = yes
			}
			set_country_flag = independence_wave_client_route_locked
			# A captured member cannot retain an unspent Defensive Congress
			# authorization while awaiting its charter-governed expulsion vote.
			independence_wave_decision_clear_charter_war_mandate = yes
			independence_wave_unregister_league_founder = yes
			# A captured member remains in the league ledger only until a
			# charter-governed expulsion vote resolves it. New client routes
			# cannot join the league through the normal registration helper.
		}
		else_if = { limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.radical_sovereignty } } set_country_flag = independence_wave_route_radical_sovereignty }
		else_if = { limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.neutral_commission } } set_country_flag = independence_wave_route_neutral_commission }
		# Exact IW-043/IW-058 additive carriers keep their package-specific
		# parties and institutional leaders when a shared government route locks.
		independence_wave_dispatch_iw043_iw058_government_route_politics = yes
		independence_wave_dispatch_iw_012_ice_government_route_politics = yes
		independence_wave_refresh_country_state = yes
	}
}

# Applies the shared ideology/elections posture for the generic Event 006 tree.
# Package adapters that preserve a researched carrier-specific party call after
# this helper and may override the shared posture. The effect is country-scope,
# runs only once when the route locks, and does not create a political-power
# store or a free faction loop.
independence_wave_apply_generic_government_route_politics = {
	if = {
		limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.constitutional_republic } }
		set_politics = { ruling_party = democratic elections_allowed = yes }
	}
	else_if = {
		limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.popular_council } }
		set_politics = { ruling_party = communism elections_allowed = no }
	}
	else_if = {
		limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.radical_sovereignty } }
		set_politics = { ruling_party = fascism elections_allowed = no }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { independence_wave_government_route = constant:independence_wave_government_route.traditional_restoration }
				check_variable = { independence_wave_government_route = constant:independence_wave_government_route.emergency_military }
				check_variable = { independence_wave_government_route = constant:independence_wave_government_route.patron_client }
			}
		}
		set_politics = { ruling_party = neutrality elections_allowed = no }
	}
	else_if = {
		limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.neutral_commission } }
		set_politics = { ruling_party = democratic elections_allowed = no }
	}
}

# Shared package-preparation primitive. Inputs:
# - regular event targets `independence_wave_setup_former_host` and
#   `independence_wave_setup_anchor_state`
# - temporary package, region, depth, territory, force, and archetype values
#
# This preparation phase writes only reversible country-local generation state.
# It deliberately does not publish active, network, host, history, evolution, or
# released-package rows. The frozen-plan executor proves every package before it
# opens those shared registries. The normal Event 006 origin entry and the
# Event 021 adapter entry below are the only public callers.
independence_wave_prepare_origin_state = {
	if = {
		limit = { has_valid_independence_wave_setup_input = yes }
		independence_wave_reset_current_generation = yes
		if = {
			limit = { NOT = { has_country_flag = random_civil_war_event6_adapter_preparing } }
			independence_wave_achievement_reset_country_trackers = yes
		}
		if = {
			limit = { has_country_flag = random_civil_war_event6_adapter_preparing }
			# Event 021 has its own generation counter. Do not advance the Event
			# 006 origin registry or its next-generation sequence for an adapter.
			set_variable = { independence_wave_generation_id = random_civil_war_generation }
		}
		else = {
			if = {
				limit = { NOT = { has_variable = global.independence_wave_next_generation_id } }
				set_variable = { global.independence_wave_next_generation_id = constant:independence_wave_plan.first_package_id }
			}
			else = {
				add_to_variable = { global.independence_wave_next_generation_id = constant:independence_wave_plan.loop_increment }
			}
			independence_wave_registry_record_event6_origin = yes
			set_variable = { independence_wave_generation_id = global.independence_wave_next_generation_id }

			limit = { check_variable = { independence_wave_setup_force_level = constant:liberation_force_level.viable } }
			set_variable = { independence_wave_security = constant:independence_wave_country_value.security_viable_start }
			set_variable = { independence_wave_instability = constant:independence_wave_country_value.instability_viable_start }
		}
		else_if = {
			limit = { check_variable = { independence_wave_setup_force_level = constant:liberation_force_level.armed } }
			set_variable = { independence_wave_security = constant:independence_wave_country_value.security_armed_start }
			set_variable = { independence_wave_instability = constant:independence_wave_country_value.instability_armed_start }
		}
		else_if = {
			limit = { check_variable = { independence_wave_setup_force_level = constant:liberation_force_level.high_chaos } }
			set_variable = { independence_wave_security = constant:independence_wave_country_value.security_high_chaos_start }
			set_variable = { independence_wave_instability = constant:independence_wave_country_value.instability_high_chaos_start }

			limit = { check_variable = { independence_wave_setup_territory_level = constant:liberation_territory_level.anchor } }
			add_to_variable = { independence_wave_capacity = constant:independence_wave_territory_start.anchor_capacity }

			add_to_variable = { independence_wave_instability = constant:independence_wave_territory_start.anchor_instability }

# The released country owns the authoritative values. Its former host receives
# an aligned mirror row so host decisions and the Event Details UI can inspect
# every active bilateral relationship without scanning countries.
independence_wave_sync_former_host_ledger = {
	if = {
		limit = {
			is_independence_wave_package_content_active = yes
			has_independence_wave_living_former_host = yes
			var:independence_wave_former_host = { independence_wave_host_relation_arrays_are_aligned = yes }
		}
		save_event_target_as = independence_wave_relation_country
		set_temp_variable = { independence_wave_relation_generation = independence_wave_generation_id }
		var:independence_wave_former_host = {
			set_temp_variable = { independence_wave_host_row = constant:independence_wave_plan.invalid_index }
			set_temp_variable = { independence_wave_host_index = constant:independence_wave_plan.first_array_index }
			while_loop_effect = {
				limit = {
					check_variable = { independence_wave_host_row = constant:independence_wave_plan.invalid_index }
					check_variable = { var = independence_wave_host_index value = independence_wave_host_country_entries^num compare = less_than }
				}
				set_temp_variable = { independence_wave_host_row_generation = independence_wave_host_generation_entries^independence_wave_host_index }
				var:independence_wave_host_country_entries^independence_wave_host_index = {
					if = {
						limit = {
							tag = event_target:independence_wave_relation_country
							check_variable = { var = independence_wave_host_row_generation value = independence_wave_relation_generation compare = equals }
						}
						set_temp_variable = { independence_wave_host_row = independence_wave_host_index }
					}
				}
				add_to_temp_variable = { independence_wave_host_index = constant:independence_wave_plan.loop_increment }
			}
			if = {
				limit = { check_variable = { independence_wave_host_row = constant:independence_wave_plan.invalid_index } }
				set_temp_variable = { independence_wave_host_row = independence_wave_host_country_entries^num }
				add_to_array = { array = independence_wave_host_country_entries value = event_target:independence_wave_relation_country }
				add_to_array = { array = independence_wave_host_generation_entries value = independence_wave_relation_generation }
				add_to_array = { array = independence_wave_host_claim_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_hostility_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_obligation_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_property_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_population_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_border_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_pressure_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_fear_entries value = constant:independence_wave_value.minimum }
				add_to_array = { array = independence_wave_host_outcome_entries value = constant:independence_wave_host_outcome.unsettled }
			}
			set_variable = { independence_wave_host_claim_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_host_claim_intensity }
			set_variable = { independence_wave_host_hostility_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_hostility }
			set_variable = { independence_wave_host_obligation_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_host_obligations }
			set_variable = { independence_wave_host_property_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_property_dispute }
			set_variable = { independence_wave_host_population_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_population_dispute }
			set_variable = { independence_wave_host_border_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_border_settlement_progress }
			set_variable = { independence_wave_host_pressure_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_host_domestic_pressure }
			set_variable = { independence_wave_host_fear_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_reconquest_fear }
			set_variable = { independence_wave_host_outcome_entries^independence_wave_host_row = event_target:independence_wave_relation_country.independence_wave_host_outcome }
			set_country_flag = independence_wave_former_host_actor
		}
	}
}

# Prunes from the tail so every aligned value remains attached to its country.
# A row survives only for the current origin generation, a living patron, and at
# least one positive relationship value.
independence_wave_prune_patron_ledger = {
	if = {
		limit = { independence_wave_patron_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_patron_row = independence_wave_patron_country_entries^num }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than } }
			subtract_from_temp_variable = { independence_wave_patron_row = constant:independence_wave_plan.loop_increment }
			set_temp_variable = { independence_wave_prune_patron_row = constant:independence_wave_value.minimum }
			set_temp_variable = { independence_wave_patron_row_generation = independence_wave_patron_owner_generation_entries^independence_wave_patron_row }
			if = {
				limit = { NOT = { check_variable = { var = independence_wave_patron_row_generation value = independence_wave_generation_id compare = equals } } }
				set_temp_variable = { independence_wave_prune_patron_row = constant:independence_wave_plan.loop_increment }
			}
			if = {
				limit = { check_variable = { independence_wave_prune_patron_row = constant:independence_wave_value.minimum } }
				var:independence_wave_patron_country_entries^independence_wave_patron_row = {
					if = {
						limit = { NOT = { exists = yes } }
						set_temp_variable = { independence_wave_prune_patron_row = constant:independence_wave_plan.loop_increment }
					}
				}
			}
			if = {
				limit = {
					check_variable = { independence_wave_prune_patron_row = constant:independence_wave_value.minimum }
					NOT = {
						OR = {
							check_variable = { var = independence_wave_patron_influence_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_aid_value_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_recognition_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_arms_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_volunteer_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_industrial_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_intelligence_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_trade_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_ideology_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_loan_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
							check_variable = { var = independence_wave_patron_base_entries^independence_wave_patron_row value = constant:independence_wave_value.minimum compare = greater_than }
						}
					}
				}
				set_temp_variable = { independence_wave_prune_patron_row = constant:independence_wave_plan.loop_increment }
			}

# Any scope. A ten-country Totalen Chaos or World Collapse wave gives each
# armed (or already radical) release at most one real claim on an adjacent
# state still owned by its surviving former host. The exact successful actors
# and distinct claimed states are frozen after the shared plan commits.
independence_wave_prepare_high_chaos_opening_claims = {
	set_variable = { global.independence_wave_presentation_immediate_claim_count = constant:independence_wave_value.minimum }
	set_variable = { global.independence_wave_presentation_immediate_claim_plan_id = global.liberation_plan_id }
	clear_global_event_target = independence_wave_first_opening_claim_actor
	if = {
		limit = {
			independence_wave_plan_metadata_arrays_are_aligned = yes
			check_variable = { global.independence_wave_plan_selected_count = constant:independence_wave_danger_milestone.exact_high_chaos_wave_count }
			check_variable = {
				var = global.independence_wave_plan_chaos_band
				value = constant:independence_wave_chaos_band.totalen_chaos
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { independence_wave_opening_claim_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = independence_wave_opening_claim_index
					value = global.independence_wave_plan_selected_count
					compare = less_than
				}
			}
			set_temp_variable = { independence_wave_execution_package_index = independence_wave_opening_claim_index }
			independence_wave_load_frozen_country_execution_context = yes
			event_target:independence_wave_execution_country = {
				clr_country_flag = independence_wave_opening_claim_actor
				clear_variable = independence_wave_opening_claim_plan_id
				save_event_target_as = independence_wave_opening_claim_actor
				if = {
					limit = {
						is_independence_wave_active_country = yes
						OR = {
							check_variable = {
								var = independence_wave_setup_force_level
								value = constant:liberation_force_level.armed
								compare = greater_than_or_equals
							}
							check_variable = { independence_wave_government_route = constant:independence_wave_government_route.radical_sovereignty }
						}
						has_event_target = independence_wave_execution_former_host
						event_target:independence_wave_execution_former_host = {
							exists = yes
							NOT = { tag = event_target:independence_wave_opening_claim_actor }
							any_owned_state = {
								NOT = { check_variable = { independence_wave_opening_claim_state_plan_id = global.liberation_plan_id } }
								NOT = { is_claimed_by = event_target:independence_wave_opening_claim_actor }
								NOT = { is_core_of = event_target:independence_wave_opening_claim_actor }
								any_neighbor_state = { is_owned_by = event_target:independence_wave_opening_claim_actor }
							}
						}
					}
					event_target:independence_wave_execution_former_host = {
						random_owned_state = {
							limit = {
								NOT = { check_variable = { independence_wave_opening_claim_state_plan_id = global.liberation_plan_id } }
								NOT = { is_claimed_by = event_target:independence_wave_opening_claim_actor }
								NOT = { is_core_of = event_target:independence_wave_opening_claim_actor }
								any_neighbor_state = { is_owned_by = event_target:independence_wave_opening_claim_actor }
							}
							add_claim_by = event_target:independence_wave_opening_claim_actor
							if = {
								limit = { is_claimed_by = event_target:independence_wave_opening_claim_actor }
								set_variable = { independence_wave_opening_claim_state_plan_id = global.liberation_plan_id }
								event_target:independence_wave_opening_claim_actor = {
									set_country_flag = independence_wave_opening_claim_actor
									set_variable = { independence_wave_opening_claim_plan_id = global.liberation_plan_id }
									if = {
										limit = { NOT = { has_event_target = independence_wave_first_opening_claim_actor } }
										save_global_event_target_as = independence_wave_first_opening_claim_actor
									}
								}
								add_to_variable = { global.independence_wave_presentation_immediate_claim_count = constant:independence_wave_plan.loop_increment }
							}
						}
					}
				}
			}
			add_to_temp_variable = { independence_wave_opening_claim_index = constant:independence_wave_plan.loop_increment }
		}
	}
}
