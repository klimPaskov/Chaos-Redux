# Overview:
# - Level 2 layer on the regional framework engine (agent decision D26) for
#   IW-083 Rif (RIF): the tribal levies of the Rif confederation as its
#   starting problem, four paid projects in

# No effect iterates countries or states.

independence_wave_rif_apply_package_layer = {
	add_ideas = independence_wave_rif_tribal_levies
	set_country_flag = independence_wave_framework_package_layer_applied
}

independence_wave_rif_complete_national_assembly = {
	independence_wave_regional_framework_reset_deltas = yes
	set_temp_variable = { independence_wave_decision_legitimacy_delta = constant:independence_wave_decision_effect.value_standard }
	set_temp_variable = { independence_wave_decision_instability_delta = constant:independence_wave_decision_effect.value_minor_loss }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_rif_assembly_convened
}

independence_wave_rif_complete_regular_army = {
	independence_wave_regional_framework_reset_deltas = yes
	remove_ideas = independence_wave_rif_tribal_levies
	add_ideas = independence_wave_rif_regular_army
	set_temp_variable = { independence_wave_decision_security_delta = constant:independence_wave_decision_effect.value_major }
	set_temp_variable = { independence_wave_decision_instability_delta = constant:independence_wave_decision_effect.value_minor }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_rif_levies_unified
}

independence_wave_rif_complete_mountain_redoubts = {
	independence_wave_regional_framework_reset_deltas = yes
	add_ideas = independence_wave_rif_mountain_redoubts
	set_temp_variable = { independence_wave_decision_security_delta = constant:independence_wave_decision_effect.value_standard }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_rif_redoubts_fortified
}

independence_wave_rif_complete_ajdir_network = {
	independence_wave_regional_framework_reset_deltas = yes
	set_temp_variable = { independence_wave_decision_capacity_delta = constant:independence_wave_decision_effect.value_standard }
	set_temp_variable = { independence_wave_decision_recognition_delta = constant:independence_wave_decision_effect.value_minor }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_rif_ajdir_network_restored
}

independence_wave_rif_cleanup_package_layer = {
	if = {
		limit = { original_tag = RIF check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_083 } }
		remove_decision = independence_wave_rif_convene_national_assembly
		remove_decision = independence_wave_rif_unify_tribal_levies
		remove_decision = independence_wave_rif_fortify_mountain_redoubts
		remove_decision = independence_wave_rif_restore_ajdir_network
		remove_ideas = { independence_wave_rif_tribal_levies independence_wave_rif_regular_army independence_wave_rif_mountain_redoubts }
		clr_country_flag = independence_wave_rif_assembly_convened
		clr_country_flag = independence_wave_rif_levies_unified
		clr_country_flag = independence_wave_rif_redoubts_fortified
		clr_country_flag = independence_wave_rif_ajdir_network_restored
	}
}
