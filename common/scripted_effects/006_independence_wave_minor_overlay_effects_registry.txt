# ============================================================================
# Overview:
# - Activates once when the living BEL carrier has the vanilla BEL_flanders
#   identity, then suspends and resumes through the exact route state.
# - Maintains two package values, a resumable corridor mission, and one of two
#   mutually exclusive institutional settlements.
# - Never changes BEL territory, cores, autonomy, characters, focus tree,
#   country history, Event 006 origin state, or FORM-03 membership state.

independence_wave_iw005_remove_flanders_overlay_ideas = {
	remove_ideas = {
		independence_wave_iw005_contested_civic_industrial_command
		independence_wave_iw005_coordinated_civic_industrial_command
		independence_wave_iw005_municipal_industrial_board
		independence_wave_iw005_scheldt_territorial_compact
	}
}
independence_wave_iw005_refresh_flanders_idea_lifecycle = {
	if = {
		limit = { is_independence_wave_iw005_flanders_overlay_active = yes }
		independence_wave_iw005_remove_flanders_overlay_ideas = yes
		if = {
			limit = { has_country_flag = independence_wave_iw005_municipal_industrial_board }
			add_ideas = independence_wave_iw005_municipal_industrial_board
		}
		else_if = {
			limit = { has_country_flag = independence_wave_iw005_scheldt_territorial_compact }
			add_ideas = independence_wave_iw005_scheldt_territorial_compact
		}
		else_if = {
			limit = { has_independence_wave_iw005_flanders_coordinated_command = yes }
			add_ideas = independence_wave_iw005_coordinated_civic_industrial_command
		}
		else = {
			add_ideas = independence_wave_iw005_contested_civic_industrial_command
		}
	}
}

		limit = {

			has_country_flag = independence_wave_iw005_factory_rail_guard_running
			has_independence_wave_iw005_flanders_guard_objective = yes
		}
		add_to_variable = { independence_wave_iw005_guard_hold_days = constant:independence_wave_iw005_flanders_value.guard_hold_daily_gain }
		clamp_variable = {
			var = independence_wave_iw005_guard_hold_days
			min = constant:independence_wave_iw005_flanders_value.minimum
			max = constant:independence_wave_iw005_flanders_value.guard_hold_required
		}
	}
	else_if = {
		limit = {
			is_independence_wave_iw005_flanders_overlay_active = yes
			has_country_flag = independence_wave_iw005_factory_rail_guard_running
		}
		set_variable = { independence_wave_iw005_guard_hold_days = constant:independence_wave_iw005_flanders_value.minimum }
	}
}

independence_wave_iw005_initialize_flanders_overlay = {
	if = {
		limit = {
			is_independence_wave_iw005_flanders_route_active = yes
			NOT = { has_country_flag = independence_wave_iw005_flanders_overlay_ever_activated }
		}
		set_country_flag = independence_wave_iw005_flanders_overlay_ever_activated
		set_country_flag = independence_wave_iw005_flanders_package_initialized
		set_country_flag = independence_wave_iw005_flanders_overlay_active
		clr_country_flag = independence_wave_iw005_flanders_overlay_suspended
		set_variable = { independence_wave_iw005_civic_industrial_coordination = constant:independence_wave_iw005_flanders_value.civic_coordination_start }
		set_variable = { independence_wave_iw005_scheldt_security = constant:independence_wave_iw005_flanders_value.scheldt_security_start }
		set_variable = { independence_wave_iw005_guard_hold_days = constant:independence_wave_iw005_flanders_value.minimum }
		independence_wave_iw005_refresh_flanders_idea_lifecycle = yes
	}
}

independence_wave_iw005_suspend_flanders_overlay = {
	if = {
		limit = { has_country_flag = independence_wave_iw005_flanders_overlay_active }
		if = {
			limit = {
				has_country_flag = independence_wave_iw005_factory_rail_guard_running
				NOT = { has_country_flag = independence_wave_iw005_factory_rail_guard_completed }
			}
			set_country_flag = independence_wave_iw005_factory_rail_guard_interrupted
		}
		independence_wave_iw005_remove_flanders_overlay_ideas = yes
		clr_country_flag = independence_wave_iw005_flanders_overlay_active
		set_country_flag = independence_wave_iw005_flanders_overlay_suspended
	}
}

independence_wave_iw005_pause_flanders_guard_mission = {
	if = {
		limit = {
			has_country_flag = independence_wave_iw005_flanders_overlay_suspended
			has_country_flag = independence_wave_iw005_factory_rail_guard_interrupted
			has_country_flag = independence_wave_iw005_factory_rail_guard_running
			has_active_mission = independence_wave_iw005_hold_scheldt_guard_line
		}
		set_temp_variable = { independence_wave_iw005_guard_pause_days = constant:independence_wave_iw005_flanders_duration.guard_pause_daily_extension }
		add_days_mission_timeout = {
			mission = independence_wave_iw005_hold_scheldt_guard_line
			days = independence_wave_iw005_guard_pause_days
		}
	}
}

independence_wave_iw005_resume_flanders_guard_mission = {
	if = {
		limit = {
			is_independence_wave_iw005_flanders_overlay_active = yes
			has_country_flag = independence_wave_iw005_factory_rail_guard_interrupted
			NOT = { has_country_flag = independence_wave_iw005_factory_rail_guard_completed }
		}
		if = {
			limit = {
				has_country_flag = independence_wave_iw005_factory_rail_guard_running
				has_active_mission = independence_wave_iw005_hold_scheldt_guard_line
			}
			clr_country_flag = independence_wave_iw005_factory_rail_guard_interrupted
		}
		else = {
			independence_wave_iw005_fail_guard_mission = yes
		}
	}
}

independence_wave_iw005_resume_flanders_overlay = {
	if = {
		limit = {
			is_independence_wave_iw005_flanders_route_active = yes
			has_country_flag = independence_wave_iw005_flanders_overlay_ever_activated
			NOT = { has_country_flag = independence_wave_iw005_flanders_overlay_active }
		}
		set_country_flag = independence_wave_iw005_flanders_overlay_active
		clr_country_flag = independence_wave_iw005_flanders_overlay_suspended
		independence_wave_iw005_refresh_flanders_idea_lifecycle = yes
		independence_wave_iw005_resume_flanders_guard_mission = yes
	}
}

# BEL country scope. This is called only from on_daily_BEL.
independence_wave_iw005_refresh_flanders_overlay = {
	if = {
		limit = { is_independence_wave_iw005_flanders_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw005_flanders_overlay_ever_activated } }
			independence_wave_iw005_initialize_flanders_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw005_flanders_overlay_active } }
			independence_wave_iw005_resume_flanders_overlay = yes
		}
		else = {
			clr_country_flag = independence_wave_iw005_flanders_overlay_suspended
			if = {
				limit = { has_country_flag = independence_wave_iw005_factory_rail_guard_interrupted }
				independence_wave_iw005_resume_flanders_guard_mission = yes
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw005_factory_rail_guard_running
					NOT = { has_active_mission = independence_wave_iw005_hold_scheldt_guard_line }
				}
				independence_wave_iw005_fail_guard_mission = yes
			}
			independence_wave_iw005_update_guard_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw005_contested_civic_industrial_command
							has_idea = independence_wave_iw005_coordinated_civic_industrial_command
							has_idea = independence_wave_iw005_municipal_industrial_board
							has_idea = independence_wave_iw005_scheldt_territorial_compact
						}
					}
				}
				independence_wave_iw005_refresh_flanders_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw005_flanders_overlay_active }
			independence_wave_iw005_suspend_flanders_overlay = yes
		}
		independence_wave_iw005_pause_flanders_guard_mission = yes
	}
}

# Called only from the dynamic-country on_daily_Dxx hooks below. There is no
# global country iteration.
independence_wave_iw022_dalmatia_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw022_dalmatia_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw022_dalmatia_overlay_ever_activated } }
			independence_wave_iw022_dalmatia_initialize_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw022_dalmatia_overlay_active } }
			independence_wave_iw022_dalmatia_resume_overlay = yes
		}
		else = {
			clr_country_flag = independence_wave_iw022_dalmatia_overlay_suspended
			if = {
				limit = { has_country_flag = independence_wave_iw022_dalmatia_watch_interrupted }
				if = {
					limit = { has_country_flag = independence_wave_iw022_dalmatia_watch_running has_active_mission = independence_wave_iw022_hold_adriatic_watch }
					clr_country_flag = independence_wave_iw022_dalmatia_watch_interrupted
				}
				else = { independence_wave_iw022_dalmatia_fail_watch_mission = yes }
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw022_dalmatia_watch_running
					NOT = { has_active_mission = independence_wave_iw022_hold_adriatic_watch }
				}
				independence_wave_iw022_dalmatia_fail_watch_mission = yes
			}
			independence_wave_iw022_dalmatia_update_watch_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw022_dalmatia_contested_adriatic_administration
							has_idea = independence_wave_iw022_dalmatia_coordinated_port_authority
							has_idea = independence_wave_iw022_dalmatia_municipal_charter
							has_idea = independence_wave_iw022_dalmatia_coastal_security_compact
						}
					}
				}
				independence_wave_iw022_dalmatia_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw022_dalmatia_overlay_active }
			independence_wave_iw022_dalmatia_suspend_overlay = yes
		}
		independence_wave_iw022_dalmatia_pause_watch_mission = yes
	}
}

# Called only from the dynamic-country on_daily_Dxx hooks below. There is no
# global country iteration.
independence_wave_iw025_vojvodina_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw025_vojvodina_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw025_vojvodina_overlay_ever_activated } }
			independence_wave_iw025_vojvodina_initialize_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw025_vojvodina_overlay_active } }
			independence_wave_iw025_vojvodina_resume_overlay = yes
		}
		else = {
			clr_country_flag = independence_wave_iw025_vojvodina_overlay_suspended
			if = {
				limit = { has_country_flag = independence_wave_iw025_vojvodina_watch_interrupted }
				if = {
					limit = { has_country_flag = independence_wave_iw025_vojvodina_watch_running has_active_mission = independence_wave_iw025_hold_vojvodina_border_watch }
					clr_country_flag = independence_wave_iw025_vojvodina_watch_interrupted
				}
				else = { independence_wave_iw025_vojvodina_fail_watch_mission = yes }
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw025_vojvodina_watch_running
					NOT = { has_active_mission = independence_wave_iw025_hold_vojvodina_border_watch }
				}
				independence_wave_iw025_vojvodina_fail_watch_mission = yes
			}
			independence_wave_iw025_vojvodina_update_watch_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw025_vojvodina_contested_danube_administration
							has_idea = independence_wave_iw025_vojvodina_coordinated_grain_authority
							has_idea = independence_wave_iw025_vojvodina_municipal_minority_charter
							has_idea = independence_wave_iw025_vojvodina_federal_frontier_compact
						}
					}
				}
				independence_wave_iw025_vojvodina_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw025_vojvodina_overlay_active }
			independence_wave_iw025_vojvodina_suspend_overlay = yes
		}
		independence_wave_iw025_vojvodina_pause_watch_mission = yes
	}
}

# Called only from the narrow on_daily_LIT hook. There is no global country
# iteration and the exact cosmetic identity gates every state change.
independence_wave_iw035_livonia_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw035_livonia_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw035_livonia_overlay_ever_activated } }
			independence_wave_iw035_livonia_initialize_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw035_livonia_overlay_active } }
			independence_wave_iw035_livonia_resume_overlay = yes
		}
		else = {
			clr_country_flag = independence_wave_iw035_livonia_overlay_suspended
			if = {
				limit = { has_country_flag = independence_wave_iw035_livonia_watch_interrupted }
				if = {
					limit = { has_country_flag = independence_wave_iw035_livonia_watch_running has_active_mission = independence_wave_iw035_hold_livonian_corridor_watch }
					clr_country_flag = independence_wave_iw035_livonia_watch_interrupted
				}
				else = { independence_wave_iw035_livonia_fail_watch_mission = yes }
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw035_livonia_watch_running
					NOT = { has_active_mission = independence_wave_iw035_hold_livonian_corridor_watch }
				}
				independence_wave_iw035_livonia_fail_watch_mission = yes
			}
			independence_wave_iw035_livonia_update_watch_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw035_livonia_contested_baltic_administration
							has_idea = independence_wave_iw035_livonia_coordinated_rail_authority
							has_idea = independence_wave_iw035_livonia_municipal_baltic_charter
							has_idea = independence_wave_iw035_livonia_federal_coastal_compact
						}
					}
				}
				independence_wave_iw035_livonia_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw035_livonia_overlay_active }
			independence_wave_iw035_livonia_suspend_overlay = yes
		}
		independence_wave_iw035_livonia_pause_watch_mission = yes
	}
}
