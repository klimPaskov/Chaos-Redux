fallout_event_380_apply_failure_deaths = {
	var:fallout_event_380_target_state_id = {
		if = {
			limit = { check_variable = { state_population_k value = constant:fallout_event_runtime.zero compare = greater_than } }
			set_temp_variable = { fallout_event_380_state_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_380_state_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_380_state_death_request = constant:fallout_event_380_threshold.result_failure_death_percent }
			round_temp_variable = fallout_event_380_state_death_request

			set_temp_variable = { state_civilian_population_loss_requested = fallout_event_380_state_death_request }

			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }
			apply_exact_state_civilian_population_loss = yes
