secret_alliance_prepare_target_selection_weight = {
	set_temp_variable = { secret_alliance_target_selection_weight = constant:secret_alliance_target_score.base }
	if = {
		limit = { is_ai = no }
		add_to_temp_variable = { secret_alliance_target_selection_weight = constant:secret_alliance_target_score.human_bonus }
	}
	if = {
		limit = { is_major = yes }
		add_to_temp_variable = { secret_alliance_target_selection_weight = constant:secret_alliance_target_score.major_bonus }
	}
	if = {
		limit = { has_war = yes }
		add_to_temp_variable = { secret_alliance_target_selection_weight = constant:secret_alliance_target_score.war_bonus }
	}
	if = {
		limit = { num_of_factories > constant:secret_alliance_target_score.large_industry_gate }
		add_to_temp_variable = { secret_alliance_target_selection_weight = constant:secret_alliance_target_score.large_industry_bonus }
	}
	round_temp_variable = secret_alliance_target_selection_weight
	clamp_temp_variable = {
		var = secret_alliance_target_selection_weight
		min = constant:secret_alliance_target_score.min_weight
		max = constant:secret_alliance_target_score.max_weight
	}
}

secret_alliance_add_current_target_to_selection_pool = {
	secret_alliance_prepare_target_selection_weight = yes
	set_temp_variable = { secret_alliance_target_pool_weight_remaining = secret_alliance_target_selection_weight }
	while_loop_effect = {
		limit = { check_variable = { secret_alliance_target_pool_weight_remaining > 0 } }
		add_to_temp_array = { secret_alliance_target_selection_pool = THIS }
		add_to_temp_variable = { secret_alliance_target_selection_pool_count = 1 }
		subtract_from_temp_variable = { secret_alliance_target_pool_weight_remaining = 1 }
	}
	set_temp_variable = { secret_alliance_target_pool_weight_remaining = 0 }
	set_temp_variable = { secret_alliance_target_selection_weight = 0 }
}

secret_alliance_select_weighted_target_candidate = {
	clear_temp_array = secret_alliance_target_selection_pool
	set_temp_variable = { secret_alliance_target_selection_pool_count = 0 }
	set_temp_variable = { secret_alliance_weighted_target_selected = 0 }
	every_country = {
		limit = { secret_alliance_can_start_for_target = yes }
		secret_alliance_add_current_target_to_selection_pool = yes
	}
	if = {
		limit = { check_variable = { secret_alliance_target_selection_pool_count > 0 } }
		random_scope_in_array = {
			array = secret_alliance_target_selection_pool
			save_global_event_target_as = secret_alliance_target
			set_temp_variable = { secret_alliance_weighted_target_selected = 1 }
		}
	}
	clear_temp_array = secret_alliance_target_selection_pool
	set_temp_variable = { secret_alliance_target_selection_pool_count = 0 }
}

secret_alliance_prepare_random_event_fire = {
	secret_alliance_sync_evolution_unlocks = yes
	set_variable = { global.secret_alliance_context_ready = 0 }
	if = {
		limit = { is_secret_alliance_automatic_event_available = yes }
		secret_alliance_select_weighted_target_candidate = yes
		if = {
			limit = {
				check_variable = { secret_alliance_weighted_target_selected > 0 }
				has_event_target = secret_alliance_target
			}
			set_variable = { global.secret_alliance_context_ready = 1 }
		}
	}
}

secret_alliance_select_founders_for_current_target = {
	set_temp_variable = { secret_alliance_founders_selected = 0 }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = secret_alliance_founders_selected
				value = constant:secret_alliance_foundation.founder_count
				compare = less_than
			}
		}
		secret_alliance_select_weighted_founder_candidate = yes
		if = {
			limit = { check_variable = { secret_alliance_weighted_founder_selected > 0 } }
			secret_alliance_register_selected_founder = yes
		}
		else = {
			set_temp_variable = { secret_alliance_founders_selected = constant:secret_alliance_foundation.founder_count }
			set_country_flag = secret_alliance_foundation_failed
		}
	}
}

secret_alliance_schedule_next_mission = {
	if = {
		limit = {
			secret_alliance_is_active = yes
			NOT = { secret_alliance_has_active_mission = yes }
		}
		if = {
			limit = {
				NOT = { has_country_flag = secret_alliance_guard_capital_network_mission_seen }
				NOT = { secret_alliance_is_public = yes }
			}
			secret_alliance_start_guard_capital_mission = yes
		}
		else_if = {
			limit = {
				has_country_flag = secret_alliance_evolution_ii
				NOT = { has_country_flag = secret_alliance_industrial_belt_secured }
				NOT = { secret_alliance_is_public = yes }
			}
			secret_alliance_start_industrial_belt_mission = yes
		}
		else_if = {
			limit = {
				has_country_flag = secret_alliance_evolution_ii
				NOT = { has_country_flag = secret_alliance_foreign_route_mission_seen }
				NOT = { secret_alliance_is_public = yes }
			}
			secret_alliance_start_foreign_route_mission = yes
		}
		else_if = {
			limit = {
				has_country_flag = secret_alliance_evolution_ii
				has_event_target = secret_alliance_patron
				NOT = { has_country_flag = secret_alliance_patron_hand_mission_seen }
				NOT = { secret_alliance_is_public = yes }
			}
			secret_alliance_start_patron_hand_mission = yes
		}
		else_if = {
			limit = { secret_alliance_is_public = yes }
			secret_alliance_start_public_border_mission = yes
		}
	}
}

secret_alliance_rebuild_member_arrays = {
	clear_array = global.secret_alliance_members
	clear_array = global.secret_alliance_founders
	clear_array = global.secret_alliance_known_members
	clear_array = global.secret_alliance_public_members
	set_variable = { pact_member_count = 0 }
	set_variable = { pact_known_member_count = 0 }
	set_variable = { pact_public_member_count = 0 }
	every_country = {
		limit = { secret_alliance_member_valid_for_target_root = yes }
		add_to_array = { array = global.secret_alliance_members value = THIS }
		ROOT = { add_to_variable = { pact_member_count = 1 } }
		if = {
			limit = { has_country_flag = secret_alliance_founder }
			add_to_array = { array = global.secret_alliance_founders value = THIS }
		}
		if = {
			limit = { has_country_flag = secret_alliance_exposed }
			add_to_array = { array = global.secret_alliance_known_members value = THIS }
			ROOT = { add_to_variable = { pact_known_member_count = 1 } }
		}
		if = {
			limit = { has_country_flag = secret_alliance_public_member }
			add_to_array = { array = global.secret_alliance_public_members value = THIS }
			ROOT = { add_to_variable = { pact_public_member_count = 1 } }
		}
	}
}

secret_alliance_cleanup_invalid_members = {
	every_country = {
		limit = {
			has_country_flag = secret_alliance_member
			NOT = { secret_alliance_member_valid_for_target_root = yes }
		}
		secret_alliance_clear_member_state = yes
	}
	secret_alliance_rebuild_member_arrays = yes
	if = {
		limit = {
			check_variable = {
				var = pact_member_count
				value = 1
				compare = less_than
			}
		}
		secret_alliance_close_compact_state = yes
	}
}

	secret_alliance_clear_active_mission_flags = yes
	set_country_flag = secret_alliance_industrial_belt_secured
	add_to_variable = { pact_preparedness = constant:secret_alliance_values.preparedness_gain }
	subtract_from_variable = { pact_pressure = constant:secret_alliance_values.pressure_gain }
	secret_alliance_schedule_next_mission = yes
	secret_alliance_clamp_target_values = yes
}

		limit = { has_event_target = secret_alliance_patron }
