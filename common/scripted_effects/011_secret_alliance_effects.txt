secret_alliance_recalculate_member_counts = {
	set_variable = { secret_alliance_core_member_count = 0 }
	set_variable = { secret_alliance_known_members = 0 }
	set_variable = { secret_alliance_known_neighbor_members = 0 }
	set_variable = { secret_alliance_neighbor_member_count = 0 }
	for_each_scope_loop = {
		array = global.secret_alliance_core_members
		if = {
			limit = { secret_alliance_country_is_valid_core_member = yes }
			event_target:secret_alliance_target = {
				add_to_variable = { secret_alliance_core_member_count = 1 }
			}
			if = {
				limit = { has_country_flag = secret_alliance_exposed_member }
				event_target:secret_alliance_target = {
					add_to_variable = { secret_alliance_known_members = 1 }
				}
			}
			if = {
				limit = {
					has_country_flag = secret_alliance_exposed_member
					has_country_flag = secret_alliance_neighbor_member
				}
				event_target:secret_alliance_target = {
					add_to_variable = { secret_alliance_known_neighbor_members = 1 }
				}
			}
			if = {
				limit = { has_country_flag = secret_alliance_neighbor_member }
				event_target:secret_alliance_target = {
					add_to_variable = { secret_alliance_neighbor_member_count = 1 }
				}
			}
		}
		else = {
			clr_country_flag = secret_alliance_core_member
			clr_country_flag = secret_alliance_exposed_member
			remove_ideas = secret_alliance_hidden_compact_discipline
			remove_ideas = secret_alliance_exposed_pact_government
			remove_from_array = {
				array = global.secret_alliance_core_members
				value = THIS
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = secret_alliance_core_member_count
				value = constant:secret_alliance_achievement.ten_signatures_required
				compare = greater_than_or_equals
			}
		}
		set_country_flag = secret_alliance_ten_signatures_eligible
	}
	if = {
		limit = {
			check_variable = {
				var = secret_alliance_known_members
				value = constant:secret_alliance_achievement.ten_signatures_required
				compare = greater_than_or_equals
			}
		}
		set_country_flag = secret_alliance_ten_signatures_eligible
	}
}

	if = {
		limit = {
			has_event_target = secret_alliance_major_patron
			event_target:secret_alliance_major_patron = {
				exists = yes
				NOT = { has_capitulated = yes }
				NOT = { has_war_with = event_target:secret_alliance_target }
				OR = {
					NOT = { is_in_faction = yes }
					is_faction_leader = yes
				}
			}
		}
		event_target:secret_alliance_major_patron = { save_global_event_target_as = secret_alliance_public_leader }
	}
	else_if = {
		limit = {
			has_event_target = secret_alliance_second_major
			event_target:secret_alliance_second_major = {
				exists = yes
				NOT = { has_capitulated = yes }
				NOT = { has_war_with = event_target:secret_alliance_target }
				OR = {
					NOT = { is_in_faction = yes }
					is_faction_leader = yes
				}
			}
		}
		event_target:secret_alliance_second_major = { save_global_event_target_as = secret_alliance_public_leader }
	}
	else_if = {
		limit = {
			any_country = {
				secret_alliance_country_is_valid_core_member = yes
				NOT = { is_in_faction = yes }
			}
		}
		random_country = {
			limit = {
				secret_alliance_country_is_valid_core_member = yes
				NOT = { is_in_faction = yes }
			}
			save_global_event_target_as = secret_alliance_public_leader
		}
	}
	else_if = {
		limit = {
			any_country = {
				secret_alliance_country_is_valid_core_member = yes
				is_faction_leader = yes
			}
		}
		random_country = {
			limit = {
				secret_alliance_country_is_valid_core_member = yes
				is_faction_leader = yes
			}
			save_global_event_target_as = secret_alliance_public_leader
		}
	}
	else = {
		random_country = {
			limit = { secret_alliance_country_is_valid_core_member = yes }
			save_global_event_target_as = secret_alliance_public_leader
		}
	}

secret_alliance_reveal_compact = {
	if = {
		limit = {
			secret_alliance_target_scope_active = yes
			NOT = { has_global_flag = secret_alliance_revealed }
		}
		set_global_flag = secret_alliance_revealed
		set_variable = { secret_alliance_stage = constant:secret_alliance_stage.revealed }
		if = {
			limit = { has_country_flag = secret_alliance_bad_evidence_recovery_eligible }
			set_country_flag = secret_alliance_bad_evidence_backfire_achieved
			clr_country_flag = secret_alliance_bad_evidence_recovery_eligible
		}
		secret_alliance_recalculate_member_counts = yes
		secret_alliance_expose_all_members = yes
		secret_alliance_create_public_faction = yes
		for_each_scope_loop = {
			array = global.secret_alliance_core_members
			if = {
				limit = { secret_alliance_country_is_valid_core_member = yes }
				set_country_flag = secret_alliance_revealed_member
				remove_ideas = secret_alliance_hidden_compact_discipline
				add_ideas = secret_alliance_revealed_compact
				if = {
					limit = {
						has_event_target = secret_alliance_public_leader
						event_target:secret_alliance_public_leader = { exists = yes }
						NOT = { is_in_faction = yes }
					}
					event_target:secret_alliance_public_leader = { add_to_faction = PREV }
				}
				if = {
					limit = { NOT = { has_war_with = event_target:secret_alliance_target } }
					declare_war_on = {
						target = event_target:secret_alliance_target
						type = annex_everything
					}
				}
				add_ai_strategy = { type = conquer id = event_target:secret_alliance_target value = constant:secret_alliance_ai_strategy.revealed_war }
				add_ai_strategy = { type = antagonize id = event_target:secret_alliance_target value = constant:secret_alliance_ai_strategy.revealed_war }
			}
		}
		if = {
			limit = {
				has_event_target = secret_alliance_major_patron
				event_target:secret_alliance_major_patron = {
					exists = yes
					NOT = { has_capitulated = yes }
					NOT = { has_war_with = event_target:secret_alliance_target }
				}
			}
			event_target:secret_alliance_major_patron = {
				add_ideas = secret_alliance_revealed_compact
				declare_war_on = {
					target = event_target:secret_alliance_target
					type = annex_everything
				}
			}
		}
		if = {
			limit = {
				has_event_target = secret_alliance_second_major
				event_target:secret_alliance_second_major = {
					exists = yes
					NOT = { has_capitulated = yes }
					NOT = { has_war_with = event_target:secret_alliance_target }
				}
			}
			event_target:secret_alliance_second_major = {
				add_ideas = secret_alliance_revealed_compact
				declare_war_on = {
					target = event_target:secret_alliance_target
					type = annex_everything
				}
			}
		}
		if = { limit = { has_idea = secret_alliance_counter_pact_bureau } remove_ideas = secret_alliance_counter_pact_bureau }
		add_ideas = secret_alliance_public_war_command
		country_event = { id = chaosx.nr11.30 days = 1 }
		news_event = { id = chaosx.news.11 days = 1 }
	}
}

secret_alliance_check_completion_achievements = {
	if = {
		limit = {
			has_country_flag = secret_alliance_small_shadow_eligible
			has_global_flag = secret_alliance_revealed
			is_major = no
			has_event_target = secret_alliance_major_patron
			event_target:secret_alliance_major_patron = { exists = yes }
			NOT = {
				any_country = {
					has_war_with = ROOT
					OR = {
						secret_alliance_country_is_valid_core_member = yes
						has_country_flag = secret_alliance_major_patron
						has_country_flag = secret_alliance_second_major
					}
				}
			}
			OR = {
				event_target:secret_alliance_major_patron = { has_capitulated = yes }
				event_target:secret_alliance_major_patron = {
					NOT = { has_war_with = ROOT }
				}
			}
			OR = {
				NOT = { has_event_target = secret_alliance_second_major }
				AND = {
					has_event_target = secret_alliance_second_major
					event_target:secret_alliance_second_major = { NOT = { exists = yes } }
				}
				AND = {
					has_event_target = secret_alliance_second_major
					event_target:secret_alliance_second_major = { has_capitulated = yes }
				}
				AND = {
					has_event_target = secret_alliance_second_major
					event_target:secret_alliance_second_major = {
						NOT = { has_war_with = ROOT }
					}
				}
			}
		}
		set_country_flag = secret_alliance_small_shadow_achieved
	}
	if = {
		limit = {
			has_country_flag = secret_alliance_no_factory_loss_eligible
			NOT = { has_capitulated = yes }
			check_variable = {
				var = secret_alliance_stage
				value = constant:secret_alliance_stage.major_patron
				compare = greater_than_or_equals
			}
			check_variable = { secret_alliance_factory_loss_count = 0 }
		}
		set_country_flag = secret_alliance_no_factory_lost_achieved
	}
	if = {
		limit = {
			has_country_flag = secret_alliance_prepared_border_eligible
			NOT = { has_country_flag = secret_alliance_prepared_border_failed }
			secret_alliance_has_covered_exposed_border = yes
			check_variable = {
				var = secret_alliance_known_neighbor_members
				value = constant:secret_alliance_achievement.prepared_border_required
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_preparedness
				value = constant:secret_alliance_threshold.preparedness_border_ready
				compare = greater_than_or_equals
			}
		}
		set_country_flag = secret_alliance_prepared_border_achieved
	}
	if = {
		limit = {
			has_country_flag = secret_alliance_ten_signatures_eligible
			NOT = { has_capitulated = yes }
		}
		set_country_flag = secret_alliance_ten_signatures_achieved
	}
}
