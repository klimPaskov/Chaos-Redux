secret_alliance_prepare_founder_candidate_weight = {
	set_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.base }
	if = {
		limit = { is_major = no }
		add_to_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.minor_bonus }
	}
	if = {
		limit = { is_major = yes }
		subtract_from_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.major_baseline_penalty }
	}
	if = {
		limit = { is_in_faction = yes }
		subtract_from_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.faction_penalty }
	}
	if = {
		limit = {
			has_opinion = {
				target = ROOT
				value < @secret_alliance_negative_opinion_threshold
			}
		}
		add_to_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.negative_opinion_bonus }
	}
	if = {
		limit = { any_neighbor_country = { tag = ROOT } }
		add_to_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.neighbor_bonus }
	}
	if = {
		limit = {
			any_neighbor_country = {
				is_subject_of = ROOT
			}
		}
		add_to_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.target_subject_neighbor_bonus }
	}
	if = {
		limit = {
			ROOT = {
				any_owned_state = {
					OR = {
						is_claimed_by = PREV
						is_core_of = PREV
					}
				}
			}
		}
		add_to_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.claim_or_core_bonus }
	}
	if = {
		limit = {
			ROOT = {
				OR = {
					has_war = yes
					has_stability < constant:secret_alliance_selection_threshold.low_target_stability
					has_war_support < constant:secret_alliance_selection_threshold.low_target_war_support
				}
			}
		}
		add_to_temp_variable = { secret_alliance_candidate_weight = constant:secret_alliance_selection_weight.war_or_tension_bonus }
	}
	round_temp_variable = secret_alliance_candidate_weight
	clamp_temp_variable = {
		var = secret_alliance_candidate_weight
		min = constant:secret_alliance_selection_weight.min
		max = constant:secret_alliance_selection_weight.max
	}
}

secret_alliance_add_current_country_to_founder_pool = {
	secret_alliance_prepare_founder_candidate_weight = yes
	set_temp_variable = { secret_alliance_candidate_weight_remaining = secret_alliance_candidate_weight }
	while_loop_effect = {
		limit = { check_variable = { secret_alliance_candidate_weight_remaining > 0 } }
		add_to_temp_array = { secret_alliance_founder_pool = THIS }
		add_to_temp_variable = { secret_alliance_founder_pool_count = 1 }
		subtract_from_temp_variable = { secret_alliance_candidate_weight_remaining = 1 }
	}
	set_temp_variable = { secret_alliance_candidate_weight_remaining = 0 }
	set_temp_variable = { secret_alliance_candidate_weight = 0 }
}

secret_alliance_prepare_founder_pool = {
	clear_temp_array = secret_alliance_founder_pool
	set_temp_variable = { secret_alliance_founder_pool_count = 0 }
	if = {
		limit = { secret_alliance_can_be_target = yes }
		every_country = {
			limit = { secret_alliance_is_valid_founder_for_root_target = yes }
			secret_alliance_add_current_country_to_founder_pool = yes
		}
	}
}

secret_alliance_select_three_founders = {
	set_temp_variable = { secret_alliance_founders_selected = 0 }
	clear_temp_array = secret_alliance_selected_founders
	secret_alliance_prepare_founder_pool = yes
	if = {
		limit = { check_variable = { secret_alliance_founder_pool_count > 0 } }
		random_scope_in_array = {
			array = secret_alliance_founder_pool
			limit = { secret_alliance_is_valid_founder_for_root_target = yes }
			save_global_event_target_as = secret_alliance_founder_convener
			add_to_temp_array = { secret_alliance_selected_founders = THIS }
			ROOT = { add_to_temp_variable = { secret_alliance_founders_selected = 1 } }
		}
	}
	if = {
		limit = { check_variable = { secret_alliance_founders_selected > 0 } }
		random_scope_in_array = {
			array = secret_alliance_founder_pool
			limit = {
				secret_alliance_is_valid_founder_for_root_target = yes
				NOT = { is_in_array = { array = secret_alliance_selected_founders value = THIS } }
			}
			save_global_event_target_as = secret_alliance_founder_financier
			add_to_temp_array = { secret_alliance_selected_founders = THIS }
			ROOT = { add_to_temp_variable = { secret_alliance_founders_selected = 1 } }
		}
	}
	if = {
		limit = { check_variable = { secret_alliance_founders_selected > 1 } }
		random_scope_in_array = {
			array = secret_alliance_founder_pool
			limit = {
				secret_alliance_is_valid_founder_for_root_target = yes
				NOT = { is_in_array = { array = secret_alliance_selected_founders value = THIS } }
			}
			save_global_event_target_as = secret_alliance_founder_provocateur
			add_to_temp_array = { secret_alliance_selected_founders = THIS }
			ROOT = { add_to_temp_variable = { secret_alliance_founders_selected = 1 } }
		}
	}
	clear_temp_array = secret_alliance_selected_founders
	clear_temp_array = secret_alliance_founder_pool
	set_temp_variable = { secret_alliance_founder_pool_count = 0 }
}

secret_alliance_initialize_pact = {
	set_temp_variable = { secret_alliance_context_ready = 0 }
	if = {
		limit = { secret_alliance_can_be_target = yes }
		set_variable = { secret_alliance_member_count = 0 }
		set_variable = { secret_alliance_member_exit_count = 0 }
		resize_array = { global.secret_alliance_members = 0 }
		resize_array = { global.secret_alliance_founders = 0 }
		secret_alliance_select_three_founders = yes
		if = {
			limit = {
				check_variable = {
					var = secret_alliance_founders_selected
					value = constant:secret_alliance_requirement.founders
					compare = greater_than_or_equals
				}
			}
			save_global_event_target_as = secret_alliance_target
			set_global_flag = secret_alliance_active
			set_country_flag = secret_alliance_target
			if = {
				limit = { NOT = { is_major = yes } }
				set_country_flag = secret_alliance_target_minor_start
			}
			if = {
				limit = { NOT = { is_in_faction = yes } }
				set_country_flag = secret_alliance_target_factionless_start
			}
			set_variable = { global.secret_alliance_phase = constant:secret_alliance_phase.baseline }
			secret_alliance_initialize_values = yes
			secret_alliance_assign_founder_roles = yes
			secret_alliance_calculate_member_cap = yes
			secret_alliance_apply_prefire_evolution_openings = yes
			secret_alliance_calculate_member_cap = yes
			set_temp_variable = { secret_alliance_context_ready = 1 }
		}
	}
	if = {
		limit = { check_variable = { secret_alliance_context_ready < 1 } }
		clr_global_flag = secret_alliance_active
		set_global_flag = secret_alliance_unavailable
		clr_country_flag = secret_alliance_target
		clr_country_flag = secret_alliance_target_minor_start
		clr_country_flag = secret_alliance_target_factionless_start
		secret_alliance_clear_saved_global_event_targets = yes
		resize_array = { global.secret_alliance_members = 0 }
		resize_array = { global.secret_alliance_founders = 0 }
		clear_variable = secret_alliance_member_count
		clear_variable = secret_alliance_member_exit_count
		clear_variable = secret_alliance_member_cap
		clear_variable = secret_alliance_suspicion
		clear_variable = secret_alliance_evidence
		clear_variable = secret_alliance_counter_readiness
		clear_variable = secret_alliance_leverage
		clear_variable = global.secret_alliance_phase
		clear_variable = global.secret_alliance_secrecy
		clear_variable = global.secret_alliance_cohesion
		clear_variable = global.secret_alliance_readiness
	}
}

secret_alliance_register_member_from_current_scope = {
	if = {
		limit = {
			NOT = { is_in_array = { array = global.secret_alliance_members value = THIS } }
		}
		add_to_array = { array = global.secret_alliance_members value = THIS }
		ROOT = { add_to_variable = { secret_alliance_member_count = 1 } }
	}
	set_country_flag = secret_alliance_member
	set_variable = { secret_alliance_member_confidence = constant:secret_alliance_opening_value.member_confidence }
	if = {
		limit = { is_major = yes }
		set_global_flag = secret_alliance_major_patron_joined
		set_country_flag = secret_alliance_role_patron
		set_variable = { secret_alliance_member_role = constant:secret_alliance_role.patron }
		add_ideas = secret_alliance_patron_liaisons
	}
	if = {
		limit = { NOT = { has_variable = secret_alliance_member_role } }
		set_variable = { secret_alliance_member_role = constant:secret_alliance_role.recruit }
	}
}

secret_alliance_calculate_member_cap = {
	set_variable = { secret_alliance_member_cap = constant:secret_alliance_member_cap.baseline }
	if = {
		limit = { has_global_flag = secret_alliance_evolution_i_unlocked }
		set_variable = { secret_alliance_member_cap = constant:secret_alliance_member_cap.evolution_i }
	}
	if = {
		limit = { has_global_flag = secret_alliance_evolution_ii_unlocked }
		set_variable = { secret_alliance_member_cap = constant:secret_alliance_member_cap.evolution_ii }
	}
	if = {
		limit = { has_global_flag = secret_alliance_evolution_iii_unlocked }
		set_variable = { secret_alliance_member_cap = constant:secret_alliance_member_cap.evolution_iii }
	}
	if = {
		limit = { has_global_flag = secret_alliance_major_patron_joined }
		add_to_variable = { secret_alliance_member_cap = constant:secret_alliance_member_cap.major_patron_bonus }
	}
	if = {
		limit = {
			check_variable = {
				var = global.secret_alliance_cohesion
				value = constant:secret_alliance_meter.high
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { secret_alliance_member_cap = constant:secret_alliance_member_cap.high_cohesion_bonus }
	}
}

secret_alliance_try_invite_member = {
	set_temp_variable = { secret_alliance_invite_success = 0 }
	secret_alliance_calculate_member_cap = yes
	if = {
		limit = {
			check_variable = {
				var = secret_alliance_member_count
				value = secret_alliance_member_cap
				compare = less_than
			}
		}
		secret_alliance_prepare_member_candidate_pool = yes
		if = {
			limit = { check_variable = { secret_alliance_founder_pool_count > 0 } }
			random_scope_in_array = {
				array = secret_alliance_founder_pool
				limit = { secret_alliance_is_valid_member_candidate_for_root_target = yes }
				save_event_target_as = secret_alliance_latest_invited_member
				secret_alliance_register_member_from_current_scope = yes
				set_variable = { secret_alliance_member_role = constant:secret_alliance_role.recruit }
				ROOT = { set_temp_variable = { secret_alliance_invite_success = 1 } }
			}
		}
		clear_temp_array = secret_alliance_founder_pool
		set_temp_variable = { secret_alliance_founder_pool_count = 0 }
	}
}

secret_alliance_try_invite_major_patron = {

	}
}

secret_alliance_refresh_member_validity = {
	set_variable = { secret_alliance_member_count = 0 }
	clear_temp_array = secret_alliance_live_members
