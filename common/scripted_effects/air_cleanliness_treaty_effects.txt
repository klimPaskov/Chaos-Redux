# Supported scope: any. A schema migration reconstructs only from the bounded
# Air Winter country registry produced by the existing monthly state pass.
air_cleanliness_treaty_initialize_lifecycle = {
	if = {
		limit = { NOT = { has_variable = global.air_cleanliness_treaty_generation } }
		set_variable = { global.air_cleanliness_treaty_generation = constant:air_cleanliness_treaty_runtime.zero }
	}
	if = {
		limit = { NOT = { has_variable = global.air_cleanliness_treaty_cleaning_day_transaction } }
		set_variable = { global.air_cleanliness_treaty_cleaning_day_transaction = constant:air_cleanliness_treaty_runtime.zero }
	}
	if = {
		limit = { NOT = { has_variable = global.air_cleanliness_treaty_verification_transaction } }
		set_variable = { global.air_cleanliness_treaty_verification_transaction = constant:air_cleanliness_treaty_runtime.zero }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.air_cleanliness_treaty_schema }
				NOT = {
					check_variable = {
						var = global.air_cleanliness_treaty_schema
						value = constant:air_cleanliness_treaty_runtime.schema_version
						compare = equals
					}
				}
			}
		}
		air_cleanliness_treaty_cancel_all_verification_transactions = yes
		air_cleanliness_treaty_cancel_all_filter_convoys = yes
		air_cleanliness_treaty_release_cleaning_day_project = yes

		for_each_scope_loop = {
			array = global.air_cleanliness_treaty_relief_states
			clr_state_flag = air_winter_relief_route
			clr_state_flag = air_cleanliness_treaty_filter_convoy_reserved
			clear_variable = air_cleanliness_treaty_relief_sponsor
			clear_variable = air_cleanliness_treaty_relief_recipient
			clear_variable = air_cleanliness_treaty_relief_generation
			clear_variable = air_cleanliness_treaty_filter_convoy_sponsor
			clear_variable = air_cleanliness_treaty_filter_convoy_recipient
			clear_variable = air_cleanliness_treaty_filter_convoy_generation
		}
		clear_array = global.air_cleanliness_treaty_members
		clear_array = global.air_cleanliness_treaty_violators
		clear_array = global.air_cleanliness_treaty_relief_states
		clear_array = global.air_cleanliness_treaty_member_removals
		clear_array = global.air_cleanliness_treaty_violation_queue
		clear_array = global.air_cleanliness_treaty_route_removals
		clear_array = global.air_cleanliness_treaty_filter_convoy_donors
		clear_array = global.air_cleanliness_treaty_filter_convoy_cancellations
		clear_array = global.air_cleanliness_treaty_active_inspectors
		clear_array = global.air_cleanliness_treaty_verification_cancellations

		if = {
			limit = { has_global_flag = air_cleanliness_treaty_active }
			set_temp_variable = { air_cleanliness_treaty_defer_founder_reconcile = constant:air_cleanliness_treaty_runtime.one }
			for_each_scope_loop = {
				array = global.air_winter_registered_countries
				air_cleanliness_treaty_clear_local_verification_runtime = yes
				if = {
					limit = {
						exists = yes
						has_country_flag = air_cleanliness_treaty_banned
					}
					air_cleanliness_treaty_register_observed_violator = yes
				}
				else_if = {
					limit = {
						has_country_flag = air_cleanliness_treaty_member
						has_used_unconventional_weapons = yes
					}
					clr_country_flag = air_cleanliness_treaty_member
					set_country_flag = air_cleanliness_treaty_banned
					air_cleanliness_treaty_register_observed_violator = yes
				}
				else_if = {
					limit = {
						has_country_flag = air_cleanliness_treaty_member
						air_cleanliness_treaty_country_is_membership_eligible = yes
					}
					air_cleanliness_treaty_register_member = yes
				}
				else_if = {
					limit = { has_country_flag = air_cleanliness_treaty_member }
					air_cleanliness_treaty_unregister_member = yes
				}
			}
			set_temp_variable = { air_cleanliness_treaty_defer_founder_reconcile = constant:air_cleanliness_treaty_runtime.zero }

			if = {
				limit = {
					check_variable = {
						var = global.air_cleanliness_treaty_members^num
						value = constant:air_cleanliness_treaty_runtime.zero
						compare = equals
					}
				}
				clr_global_flag = air_cleanliness_treaty_active
				clr_global_flag = air_cleanliness_treaty_news_fired
				if = {
					limit = { has_event_target = air_cleanliness_treaty_founder }
					clear_global_event_target = air_cleanliness_treaty_founder
				}
			}
			else = {
				if = {
					limit = {
						check_variable = {
							var = global.air_cleanliness_treaty_generation
							value = constant:air_cleanliness_treaty_runtime.zero
							compare = equals
						}
					}
					set_variable = { global.air_cleanliness_treaty_generation = constant:air_cleanliness_treaty_runtime.one }
				}
				air_cleanliness_treaty_reconcile_founder = yes
			}
		}
		else = {
			if = {
				limit = { has_event_target = air_cleanliness_treaty_founder }
				clear_global_event_target = air_cleanliness_treaty_founder
			}
			for_each_scope_loop = {
				array = global.air_winter_registered_countries
				air_cleanliness_treaty_clear_local_verification_runtime = yes
				clr_country_flag = air_cleanliness_treaty_member
				clr_country_flag = air_cleanliness_treaty_invitation_pending
				clr_country_flag = air_cleanliness_treaty_invitation_open
				clear_variable = air_cleanliness_treaty_invitation_attempt_generation
			}
		}
		set_variable = { global.air_cleanliness_treaty_schema = constant:air_cleanliness_treaty_runtime.schema_version }
	}
}

# Supported scope: any. The Air host calls this before the existing state pass
# so an invalid route cannot reduce the next monthly winter pressure result.
air_cleanliness_treaty_pre_winter_pulse = {
	if = {
		limit = {
			OR = {
				has_global_flag = fallout_transition_active
				has_global_flag = fallout_active
				has_global_flag = fallout_air_cleanliness_disabled
			}
		}
		air_cleanliness_treaty_pause_for_fallout = yes
	}
	else = {
		air_cleanliness_treaty_initialize_lifecycle = yes
		if = {
			limit = { has_global_flag = air_cleanliness_treaty_active }
			air_cleanliness_treaty_reconcile_relief_routes = yes
		}
	}
}

# Supported scope: any. The caller is the existing single monthly global host.
air_cleanliness_treaty_host_pulse = {
	if = {
		limit = {
			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_active }
			NOT = { has_global_flag = fallout_air_cleanliness_disabled }
		}
		air_cleanliness_treaty_initialize_lifecycle = yes
	}
	else = { air_cleanliness_treaty_pause_for_fallout = yes }
	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.air_cleanliness_treaty_last_host_date }
				NOT = {
					check_variable = {
						var = global.air_cleanliness_treaty_last_host_date
						value = global.date
						compare = equals
					}
				}
			}
			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_active }
			NOT = { has_global_flag = fallout_air_cleanliness_disabled }
		}
		set_variable = { global.air_cleanliness_treaty_last_host_date = global.date }

		if = {
			limit = { has_global_flag = air_cleanliness_treaty_active }
			air_cleanliness_treaty_reconcile_registry = yes
			air_cleanliness_treaty_reconcile_verification_transactions = yes
		}
		else = { air_cleanliness_treaty_clear_inactive_runtime = yes }

		if = {
			limit = { has_global_flag = air_cleanliness_treaty_active }
			if = {
				limit = {
					OR = {
						NOT = { has_variable = global.air_cleanliness_treaty_next_invitation_scan_date }
						check_variable = {
							var = global.date
							value = global.air_cleanliness_treaty_next_invitation_scan_date
							compare = greater_than_or_equals
						}
					}
				}
				air_cleanliness_treaty_reconcile_observed_countries = yes
				if = {
					limit = { air_cleanliness_treaty_founder_is_valid = yes }
					event_target:air_cleanliness_treaty_founder = {
						air_cleanliness_treaty_issue_invitations = yes
					}
				}
			}
		}

		if = {
			limit = { NOT = { has_global_flag = air_cleanliness_treaty_active } }
			air_cleanliness_treaty_try_form = yes
		}
	}
}

# Supported scope: COUNTRY. Registration is idempotent. Relationship and
# treaty-owned embargo changes happen only at lifecycle edges.
air_cleanliness_treaty_register_member = {
	if = {
		limit = {
			has_global_flag = air_cleanliness_treaty_active
			air_cleanliness_treaty_country_is_membership_eligible = yes
			NOT = {
				is_in_array = {
					array = global.air_cleanliness_treaty_members
					value = THIS
				}
			}
		}
		save_event_target_as = air_cleanliness_treaty_joining_member
		for_each_scope_loop = {
			array = global.air_cleanliness_treaty_members
			if = {
				limit = {
					air_cleanliness_treaty_country_is_live_member = yes
					NOT = { tag = event_target:air_cleanliness_treaty_joining_member }
				}
				save_event_target_as = air_cleanliness_treaty_existing_member
				remove_opinion_modifier = {
					target = event_target:air_cleanliness_treaty_joining_member
					modifier = air_cleanliness_treaty_member_respect
				}
				add_opinion_modifier = {
					target = event_target:air_cleanliness_treaty_joining_member
					modifier = air_cleanliness_treaty_member_respect
				}
				event_target:air_cleanliness_treaty_joining_member = {
					remove_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_existing_member
						modifier = air_cleanliness_treaty_member_respect
					}
					add_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_existing_member
						modifier = air_cleanliness_treaty_member_respect
					}
				}
				set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_joining_member }
				set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
				condemnation_unregister_native_embargo_source = yes
				event_target:air_cleanliness_treaty_joining_member = {
					set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_existing_member }
					set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
					condemnation_unregister_native_embargo_source = yes
				}
			}
		}

		set_country_flag = air_cleanliness_treaty_member
		set_country_flag = air_cleanliness_treaty_ever_member
		clr_country_flag = air_cleanliness_treaty_invitation_pending
		clr_country_flag = air_cleanliness_treaty_invitation_open
		clear_variable = air_cleanliness_treaty_invitation_attempt_generation
		add_to_array = { array = global.air_cleanliness_treaty_members value = THIS }

		for_each_scope_loop = {
			array = global.air_cleanliness_treaty_violators
			if = {
				limit = {
					exists = yes
					has_country_flag = air_cleanliness_treaty_banned
					NOT = { tag = event_target:air_cleanliness_treaty_joining_member }
				}
				save_event_target_as = air_cleanliness_treaty_existing_violator
				event_target:air_cleanliness_treaty_joining_member = {
					remove_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_existing_violator
						modifier = air_cleanliness_treaty_violator
					}
					add_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_existing_violator
						modifier = air_cleanliness_treaty_violator
					}
					if = {
						limit = { NOT = { is_subject_of = event_target:air_cleanliness_treaty_existing_violator } }
						set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_existing_violator }
						set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
						condemnation_register_native_embargo_source = yes
					}
				}
			}
		}
	}
}

# Supported scope: COUNTRY. This removes current membership and every treaty-
# owned bilateral edge while retaining historical membership and betrayal flags.
air_cleanliness_treaty_unregister_member = {
	if = {
		limit = {
			OR = {
				has_country_flag = air_cleanliness_treaty_member
				is_in_array = {
					array = global.air_cleanliness_treaty_members
					value = THIS
				}
			}
		}
		save_event_target_as = air_cleanliness_treaty_departing_member
		for_each_scope_loop = {
			array = global.air_cleanliness_treaty_members
			if = {
				limit = { NOT = { tag = event_target:air_cleanliness_treaty_departing_member } }
				save_event_target_as = air_cleanliness_treaty_remaining_member
				remove_opinion_modifier = {
					target = event_target:air_cleanliness_treaty_departing_member
					modifier = air_cleanliness_treaty_member_respect
				}
				event_target:air_cleanliness_treaty_departing_member = {
					remove_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_remaining_member
						modifier = air_cleanliness_treaty_member_respect
					}
					set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_remaining_member }
					set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
					condemnation_unregister_native_embargo_source = yes
				}
				set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_departing_member }
				set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
				condemnation_unregister_native_embargo_source = yes
			}
		}

		for_each_scope_loop = {
			array = global.air_cleanliness_treaty_violators
			if = {
				limit = { NOT = { tag = event_target:air_cleanliness_treaty_departing_member } }
				save_event_target_as = air_cleanliness_treaty_departure_violator
				event_target:air_cleanliness_treaty_departing_member = {
					remove_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_departure_violator
						modifier = air_cleanliness_treaty_violator
					}
					set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_departure_violator }
					set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
					condemnation_unregister_native_embargo_source = yes
				}
			}
		}

		event_target:air_cleanliness_treaty_departing_member = {
			if = {
				limit = { has_country_flag = air_cleanliness_treaty_cleaning_day_sponsor }
				air_cleanliness_treaty_cancel_cleaning_day_project = yes
			}
			air_cleanliness_treaty_cancel_verification_transactions_for_country = yes
			air_cleanliness_treaty_cancel_filter_convoys_for_country = yes
			air_cleanliness_treaty_revoke_relief_routes_for_country = yes
			while_loop_effect = {
				limit = {
					is_in_array = {
						array = global.air_cleanliness_treaty_members
						value = THIS
					}
				}
				remove_from_array = { array = global.air_cleanliness_treaty_members value = THIS }
			}
			clr_country_flag = air_cleanliness_treaty_member
			clr_country_flag = air_cleanliness_treaty_invitation_pending
			clr_country_flag = air_cleanliness_treaty_invitation_open
			clear_variable = air_cleanliness_treaty_invitation_attempt_generation
			clear_variable = air_cleanliness_treaty_invitation_generation
			clear_variable = air_cleanliness_treaty_invitation_accepted_generation
			clear_variable = air_cleanliness_treaty_invitation_declined_generation
		}

		if = {
			limit = {
				has_event_target = air_cleanliness_treaty_founder
				event_target:air_cleanliness_treaty_founder = { tag = event_target:air_cleanliness_treaty_departing_member }
			}
			clear_global_event_target = air_cleanliness_treaty_founder
		}
		if = {
			limit = { NOT = { check_variable = { air_cleanliness_treaty_defer_founder_reconcile > constant:air_cleanliness_treaty_runtime.zero } } }
			air_cleanliness_treaty_reconcile_founder = yes
		}
	}
}

# Supported scope: COUNTRY. This edge helper records one existing banned
# country and creates every missing treaty-owned sanction against live members.
air_cleanliness_treaty_register_observed_violator = {
	if = {
		limit = {
			exists = yes
			has_country_flag = air_cleanliness_treaty_banned
		}
		save_event_target_as = air_cleanliness_treaty_observed_violator
		if = {
			limit = {
				OR = {
					has_country_flag = air_cleanliness_treaty_member
					is_in_array = {
						array = global.air_cleanliness_treaty_members
						value = THIS
					}
				}
			}
			air_cleanliness_treaty_unregister_member = yes
		}
		set_country_flag = air_cleanliness_treaty_betrayal_memory
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.air_cleanliness_treaty_violators
						value = THIS
					}
				}
			}
			add_to_array = { array = global.air_cleanliness_treaty_violators value = THIS }
			for_each_scope_loop = {
				array = global.air_cleanliness_treaty_members
				if = {
					limit = {
						air_cleanliness_treaty_country_is_live_member = yes
						NOT = { tag = event_target:air_cleanliness_treaty_observed_violator }
					}
					remove_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_observed_violator
						modifier = air_cleanliness_treaty_violator
					}
					add_opinion_modifier = {
						target = event_target:air_cleanliness_treaty_observed_violator
						modifier = air_cleanliness_treaty_violator
					}
					if = {
						limit = { NOT = { is_subject_of = event_target:air_cleanliness_treaty_observed_violator } }
						set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_observed_violator }
						set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
						condemnation_register_native_embargo_source = yes
					}
				}
			}
		}
	}
}

# Supported scope: COUNTRY, the treaty member that breached a member obligation.
# A short-lived country flag distinguishes inspection refusal from the default
# unconventional-weapon cause without changing the shared consequence path.
air_cleanliness_treaty_handle_violation = {
	if = {
		limit = {
			has_global_flag = air_cleanliness_treaty_active
			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_active }
			NOT = { has_global_flag = fallout_air_cleanliness_disabled }
			has_country_flag = air_cleanliness_treaty_member
			NOT = { has_country_flag = air_cleanliness_treaty_banned }
			is_in_array = {
				array = global.air_cleanliness_treaty_members
				value = THIS
			}
		}
		save_event_target_as = air_cleanliness_treaty_violating_member
		set_country_flag = air_cleanliness_treaty_banned
		set_country_flag = air_cleanliness_treaty_betrayal_memory
		air_cleanliness_treaty_unregister_member = yes
		event_target:air_cleanliness_treaty_violating_member = {
			air_cleanliness_treaty_register_observed_violator = yes
		}

		event_target:air_cleanliness_treaty_violating_member = {
			if = {
				limit = {
					NOT = { has_country_flag = air_cleanliness_treaty_inspection_refusal_violation_context }
					NOT = { has_country_flag = air_cleanliness_treaty_violation_news_fired }
				}
				set_country_flag = air_cleanliness_treaty_violation_news_fired
				news_event = { id = chaosx_contamination.8 }
			}
		}
	}
}

# Supported scope: any. Array mutation occurs only while iterating separate
# work queues, never while iterating the member registry itself.
air_cleanliness_treaty_reconcile_registry = {
	clear_array = global.air_cleanliness_treaty_member_removals
	clear_array = global.air_cleanliness_treaty_violation_queue
	set_temp_variable = { air_cleanliness_treaty_defer_founder_reconcile = constant:air_cleanliness_treaty_runtime.one }
	for_each_scope_loop = {
		array = global.air_cleanliness_treaty_members
		if = {
			limit = {
				has_country_flag = air_cleanliness_treaty_member
				NOT = { has_country_flag = air_cleanliness_treaty_banned }
				has_used_unconventional_weapons = yes
			}
			add_to_array = { array = global.air_cleanliness_treaty_violation_queue value = THIS }
		}
		else_if = {
			limit = { NOT = { air_cleanliness_treaty_country_is_live_member = yes } }
			add_to_array = { array = global.air_cleanliness_treaty_member_removals value = THIS }
		}
	}
	for_each_scope_loop = {
		array = global.air_cleanliness_treaty_violation_queue
		air_cleanliness_treaty_handle_violation = yes
	}
	for_each_scope_loop = {
		array = global.air_cleanliness_treaty_member_removals
		air_cleanliness_treaty_unregister_member = yes
	}
	clear_array = global.air_cleanliness_treaty_violation_queue
	clear_array = global.air_cleanliness_treaty_member_removals
	set_temp_variable = { air_cleanliness_treaty_defer_founder_reconcile = constant:air_cleanliness_treaty_runtime.zero }
	air_cleanliness_treaty_reconcile_founder = yes
}

# Supported scope: any. This low-frequency scan repairs legacy flags and adds
# persistent violator memory only among already observed Air Winter countries.
air_cleanliness_treaty_reconcile_observed_countries = {
	for_each_scope_loop = {
		array = global.air_winter_registered_countries
		if = {
			limit = {
				exists = yes
				has_country_flag = air_cleanliness_treaty_banned
			}
			air_cleanliness_treaty_register_observed_violator = yes
		}
		else_if = {
			limit = {
				has_country_flag = air_cleanliness_treaty_member

	}
}

air_cleanliness_treaty_clear_all_relief_routes = {

# Supported scope: any. Each route recovers its two country scopes before the
# bilateral war test. Invalid routes are removed through a separate state queue.
air_cleanliness_treaty_reconcile_relief_routes = {
	clear_array = global.air_cleanliness_treaty_route_removals
	for_each_scope_loop = {
		array = global.air_cleanliness_treaty_relief_states
		set_temp_variable = { air_cleanliness_treaty_route_valid = constant:air_cleanliness_treaty_runtime.one }
		if = {
			limit = { NOT = { air_cleanliness_treaty_relief_route_ledger_is_valid = yes } }
			set_temp_variable = { air_cleanliness_treaty_route_valid = constant:air_cleanliness_treaty_runtime.zero }
		}
		if = {
			limit = {
				has_variable = air_cleanliness_treaty_relief_sponsor
				has_variable = air_cleanliness_treaty_relief_recipient
			}
			var:air_cleanliness_treaty_relief_sponsor = {
				save_event_target_as = air_cleanliness_treaty_route_sponsor
			}
			var:air_cleanliness_treaty_relief_recipient = {
				save_event_target_as = air_cleanliness_treaty_route_recipient
			}
			if = {
				limit = {
					event_target:air_cleanliness_treaty_route_sponsor = {
						has_war_with = event_target:air_cleanliness_treaty_route_recipient
					}
				}
				set_temp_variable = { air_cleanliness_treaty_route_valid = constant:air_cleanliness_treaty_runtime.zero }
			}
		}
		if = {
			limit = {
				check_variable = {
					var = air_cleanliness_treaty_route_valid
					value = constant:air_cleanliness_treaty_runtime.zero
					compare = equals
				}
			}
			add_to_array = { array = global.air_cleanliness_treaty_route_removals value = THIS }
		}
	}
	for_each_scope_loop = {
		array = global.air_cleanliness_treaty_route_removals
		air_cleanliness_treaty_clear_relief_route = yes
	}
	clear_array = global.air_cleanliness_treaty_route_removals
}

	}
}

# Supported scope: COUNTRY, the annexed country from on_annex FROM.
air_cleanliness_treaty_handle_annexed_country = {
	save_event_target_as = air_cleanliness_treaty_annexed_country
	if = {
		limit = {
			OR = {
				has_country_flag = air_cleanliness_treaty_member
				is_in_array = { array = global.air_cleanliness_treaty_members value = THIS }
			}
		}
		air_cleanliness_treaty_unregister_member = yes
	}
	if = {
		limit = {
			is_in_array = {
				array = global.air_cleanliness_treaty_violators
				value = THIS
			}
		}
		for_each_scope_loop = {
			array = global.air_cleanliness_treaty_members
			remove_opinion_modifier = {
				target = event_target:air_cleanliness_treaty_annexed_country
				modifier = air_cleanliness_treaty_violator
			}
			set_temp_variable = { condemnation_embargo_target = event_target:air_cleanliness_treaty_annexed_country }
			set_temp_variable = { condemnation_embargo_source = constant:condemnation_embargo_source.air_cleanliness_treaty }
			condemnation_unregister_native_embargo_source = yes
		}
		while_loop_effect = {
			limit = {
				is_in_array = {
					array = global.air_cleanliness_treaty_violators
					value = THIS
				}
			}
			remove_from_array = { array = global.air_cleanliness_treaty_violators value = THIS }
		}
	}
	if = {
		limit = { has_country_flag = air_cleanliness_treaty_cleaning_day_sponsor }
		air_cleanliness_treaty_release_cleaning_day_project = yes
	}
	air_cleanliness_treaty_revoke_relief_routes_for_country = yes
	air_cleanliness_treaty_cancel_verification_transactions_for_country = yes
	air_cleanliness_treaty_cancel_filter_convoys_for_country = yes
	clr_country_flag = air_cleanliness_treaty_invitation_pending
	clr_country_flag = air_cleanliness_treaty_invitation_open
	clear_variable = air_cleanliness_treaty_invitation_attempt_generation
	air_cleanliness_treaty_reconcile_founder = yes
}
