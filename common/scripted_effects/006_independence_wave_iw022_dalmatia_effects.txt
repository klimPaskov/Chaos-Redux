independence_wave_iw022_dalmatia_remove_overlay_ideas = {
	remove_ideas = {
		independence_wave_iw022_dalmatia_contested_adriatic_administration
		independence_wave_iw022_dalmatia_coordinated_port_authority
		independence_wave_iw022_dalmatia_municipal_charter
		independence_wave_iw022_dalmatia_coastal_security_compact
	}
}

independence_wave_iw022_dalmatia_refresh_idea_lifecycle = {
	if = {
		limit = { is_independence_wave_iw022_dalmatia_overlay_active = yes }
		independence_wave_iw022_dalmatia_remove_overlay_ideas = yes
		if = {
			limit = { has_country_flag = independence_wave_iw022_dalmatia_municipal_charter }
			add_ideas = independence_wave_iw022_dalmatia_municipal_charter
		}
		else_if = {
			limit = { has_country_flag = independence_wave_iw022_dalmatia_coastal_security_compact }
			add_ideas = independence_wave_iw022_dalmatia_coastal_security_compact
		}
		else_if = {
			limit = {
				check_variable = {
					var = independence_wave_iw022_dalmatia_port_coordination
					value = constant:independence_wave_iw022_dalmatia_value.settlement_threshold
					compare = greater_than_or_equals
				}
				check_variable = {
					var = independence_wave_iw022_dalmatia_coastal_security
					value = constant:independence_wave_iw022_dalmatia_value.settlement_threshold
					compare = greater_than_or_equals
				}
			}
			add_ideas = independence_wave_iw022_dalmatia_coordinated_port_authority
		}
		else = {
			add_ideas = independence_wave_iw022_dalmatia_contested_adriatic_administration
		}
	}
}

independence_wave_iw022_dalmatia_cancel_watch_permanent_identity_loss = {

		limit = { has_country_flag = independence_wave_iw022_dalmatia_watch_running }
		set_country_flag = independence_wave_iw022_dalmatia_watch_failed

	if = { limit = { has_active_mission = independence_wave_iw022_hold_adriatic_watch } remove_mission = independence_wave_iw022_hold_adriatic_watch }
	independence_wave_iw022_dalmatia_remove_overlay_ideas = yes
	clr_country_flag = independence_wave_iw022_dalmatia_watch_running
	clr_country_flag = independence_wave_iw022_dalmatia_watch_interrupted
	clr_country_flag = independence_wave_iw022_dalmatia_overlay_active

}

	if = {
		limit = {
			is_independence_wave_iw022_dalmatia_overlay_active = yes
			has_country_flag = independence_wave_iw022_dalmatia_watch_running
			has_independence_wave_iw022_dalmatia_watch_objective = yes
		}
		add_to_variable = { independence_wave_iw022_dalmatia_watch_hold_days = constant:independence_wave_iw022_dalmatia_value.watch_daily_gain }
		clamp_variable = {
			var = independence_wave_iw022_dalmatia_watch_hold_days
			min = constant:independence_wave_iw022_dalmatia_value.minimum
			max = constant:independence_wave_iw022_dalmatia_value.watch_required
		}
	}
	else_if = {
		limit = {
			is_independence_wave_iw022_dalmatia_overlay_active = yes
			has_country_flag = independence_wave_iw022_dalmatia_watch_running
		}
		set_variable = { independence_wave_iw022_dalmatia_watch_hold_days = constant:independence_wave_iw022_dalmatia_value.minimum }
	}
}

independence_wave_iw022_dalmatia_initialize_overlay = {
	if = {
		limit = {
			is_independence_wave_iw022_dalmatia_route_active = yes
			NOT = { has_country_flag = independence_wave_iw022_dalmatia_overlay_ever_activated }
		}
		set_country_flag = independence_wave_iw022_dalmatia_overlay_ever_activated
		set_country_flag = independence_wave_iw022_dalmatia_package_initialized
		set_country_flag = independence_wave_iw022_dalmatia_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw022_dalmatia_overlay_suspended
		set_variable = { independence_wave_iw022_dalmatia_port_coordination = constant:independence_wave_iw022_dalmatia_value.port_coordination_start }
		set_variable = { independence_wave_iw022_dalmatia_coastal_security = constant:independence_wave_iw022_dalmatia_value.coastal_security_start }
		set_variable = { independence_wave_iw022_dalmatia_municipal_legitimacy = constant:independence_wave_iw022_dalmatia_value.municipal_legitimacy_start }
		set_variable = { independence_wave_iw022_dalmatia_watch_hold_days = constant:independence_wave_iw022_dalmatia_value.minimum }
		set_variable = { independence_wave_iw022_dalmatia_watch_suspension_days = constant:independence_wave_iw022_dalmatia_value.minimum }
		set_variable = { independence_wave_anchor_state = 103 }
		set_variable = { independence_wave_package_archetype = constant:independence_wave_package_archetype.port_or_island }
		set_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.coastal_maritime }
		independence_wave_iw022_dalmatia_refresh_idea_lifecycle = yes
	}
}

independence_wave_iw022_dalmatia_suspend_overlay = {
	if = {
		limit = { has_country_flag = independence_wave_iw022_dalmatia_overlay_active }
		if = {
			limit = {
				has_country_flag = independence_wave_iw022_dalmatia_watch_running
				NOT = { has_country_flag = independence_wave_iw022_dalmatia_watch_completed }
			}
			set_country_flag = independence_wave_iw022_dalmatia_watch_interrupted
		}
		independence_wave_iw022_dalmatia_remove_overlay_ideas = yes
		clr_country_flag = independence_wave_iw022_dalmatia_overlay_active
		clr_country_flag = independence_wave_route_overlay_active
		set_country_flag = independence_wave_iw022_dalmatia_overlay_suspended
	}
}

independence_wave_iw022_dalmatia_pause_watch_mission = {
	if = {
		limit = {
			has_country_flag = independence_wave_iw022_dalmatia_overlay_suspended
			has_country_flag = independence_wave_iw022_dalmatia_watch_interrupted
			has_country_flag = independence_wave_iw022_dalmatia_watch_running
			has_active_mission = independence_wave_iw022_hold_adriatic_watch
		}
		set_temp_variable = { independence_wave_iw022_dalmatia_pause_days = constant:independence_wave_iw022_dalmatia_duration.watch_pause_daily_extension }
		add_days_mission_timeout = {
			mission = independence_wave_iw022_hold_adriatic_watch
			days = independence_wave_iw022_dalmatia_pause_days
		}
		add_to_variable = { independence_wave_iw022_dalmatia_watch_suspension_days = constant:independence_wave_iw022_dalmatia_duration.watch_pause_daily_extension }
		if = {
			limit = {
				check_variable = {
					var = independence_wave_iw022_dalmatia_watch_suspension_days
					value = constant:independence_wave_iw022_dalmatia_duration.watch_suspension_grace_days
					compare = greater_than_or_equals
				}
			}
			independence_wave_iw022_dalmatia_cancel_watch_permanent_identity_loss = yes
		}
	}
}

independence_wave_iw022_dalmatia_resume_overlay = {
	if = {
		limit = {
			is_independence_wave_iw022_dalmatia_route_active = yes
			has_country_flag = independence_wave_iw022_dalmatia_overlay_ever_activated
			NOT = { has_country_flag = independence_wave_iw022_dalmatia_overlay_active }
		}
		set_country_flag = independence_wave_iw022_dalmatia_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw022_dalmatia_overlay_suspended
		set_variable = { independence_wave_iw022_dalmatia_watch_suspension_days = constant:independence_wave_iw022_dalmatia_value.minimum }
		independence_wave_iw022_dalmatia_refresh_idea_lifecycle = yes
	}
}

# Called only from the dynamic-country on_daily_Dxx hooks below. There is no
# global country iteration.
independence_wave_iw022_dalmatia_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw022_dalmatia_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw022_dalmatia_overlay_ever_activated } }
			independence_wave_iw022_dalmatia_initialize_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw022_dalmatia_overlay_active } }
			independence_wave_iw022_dalmatia_resume_overlay = yes
		}
		else = {
			clr_country_flag = independence_wave_iw022_dalmatia_overlay_suspended
			if = {
				limit = { has_country_flag = independence_wave_iw022_dalmatia_watch_interrupted }
				if = {
					limit = { has_country_flag = independence_wave_iw022_dalmatia_watch_running has_active_mission = independence_wave_iw022_hold_adriatic_watch }
					clr_country_flag = independence_wave_iw022_dalmatia_watch_interrupted
				}
				else = { independence_wave_iw022_dalmatia_fail_watch_mission = yes }
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw022_dalmatia_watch_running
					NOT = { has_active_mission = independence_wave_iw022_hold_adriatic_watch }
				}
				independence_wave_iw022_dalmatia_fail_watch_mission = yes
			}
			independence_wave_iw022_dalmatia_update_watch_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw022_dalmatia_contested_adriatic_administration
							has_idea = independence_wave_iw022_dalmatia_coordinated_port_authority
							has_idea = independence_wave_iw022_dalmatia_municipal_charter
							has_idea = independence_wave_iw022_dalmatia_coastal_security_compact
						}
					}
				}
				independence_wave_iw022_dalmatia_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw022_dalmatia_overlay_active }
			independence_wave_iw022_dalmatia_suspend_overlay = yes
		}
		independence_wave_iw022_dalmatia_pause_watch_mission = yes
	}
}
