# ============================================================================
# CHAOS REDUX - EVENT 029 HISTORY PAYLOAD EFFECTS
# ============================================================================
# Binds the generic Event Log sequence to the state-bound Riches Found mine.
# The payload is captured from the pre-fire state target before the opening
# event initializes the mine, then resolved only when Event Details requests an
# Event 029 row. A missing or no-longer-valid state pair deliberately leaves
# the payload flag clear so the UI can show the N/A wording.
# ============================================================================

# STATE scope during generic history recording. The opening dispatcher has
# already saved this state as riches_found_prefire_state, but the mine record
# is not initialized until the visible opening event executes.
riches_found_bind_history_payload = {
	if = {
		limit = { has_variable = global.events_log_history_sequence }
		add_to_array = { array = global.riches_found_history_sequence_entries value = global.events_log_history_sequence index = 0 }
		add_to_array = { array = global.riches_found_history_state_entries value = THIS index = 0 }
		set_variable = { riches_found_history_sequence = global.events_log_history_sequence }
		set_state_flag = riches_found_history_payload_bound
	}
}

# COUNTRY scope. Resolves the selected history occurrence, or the latest open
# Event 029 row when Details was opened from the catalog tab. Only a registered
# mine state is accepted as a valid target pair.
riches_found_load_open_history_payload = {
	clear_variable = riches_found_history_detail_state
	clear_variable = riches_found_history_detail_sequence
	clr_country_flag = riches_found_history_payload_loaded
	set_temp_variable = { riches_found_history_target_sequence = constant:riches_found_value.zero }
	if = {
		limit = {
			check_variable = { var = events_log_history_selected_event_id value = constant:riches_found_event.id compare = equals }
			check_variable = { events_log_history_selected_sequence > constant:riches_found_value.zero }
		}
		set_temp_variable = { riches_found_history_target_sequence = events_log_history_selected_sequence }
	}
	else_if = {
		limit = {
			check_variable = { var = global.events_log_open_event_detail_entries^num value = constant:riches_found_value.zero compare = greater_than }
			check_variable = { var = global.events_log_open_event_detail_entries^0 value = constant:riches_found_event.id compare = equals }
			check_variable = { global.events_log_open_event_detail_sequence_entries^0 > constant:riches_found_value.zero }
		}
		set_temp_variable = { riches_found_history_target_sequence = global.events_log_open_event_detail_sequence_entries^0 }
	}
	if = {
		limit = { check_variable = { riches_found_history_target_sequence > constant:riches_found_value.zero } }
		set_temp_variable = { riches_found_history_payload_index = constant:riches_found_value.zero }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = riches_found_history_payload_index
					value = global.riches_found_history_sequence_entries^num
					compare = less_than
				}
				NOT = { has_country_flag = riches_found_history_payload_loaded }
			}
			if = {
				limit = {
					check_variable = {
						var = global.riches_found_history_sequence_entries^riches_found_history_payload_index
						value = riches_found_history_target_sequence
						compare = equals
					}
				}
				set_variable = { riches_found_history_detail_sequence = global.riches_found_history_sequence_entries^riches_found_history_payload_index }
				set_variable = { riches_found_history_detail_state = global.riches_found_history_state_entries^riches_found_history_payload_index }
				if = {
					limit = {
						has_variable = riches_found_history_detail_state
						var:riches_found_history_detail_state = { riches_found_mine_is_registered = yes }
					}
					set_country_flag = riches_found_history_payload_loaded
				}
			}
			add_to_temp_variable = { riches_found_history_payload_index = constant:riches_found_value.one }
		}
	}
}
