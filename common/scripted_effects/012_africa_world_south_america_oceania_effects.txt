# ============================================================================
# CHAOS REDUX - EVENT 012 SOUTH AMERICA AND OCEANIA W2 PLAY LOOPS
# ============================================================================
# Country-scoped contracts for the grounded South American and Oceanian
# packages. The contracts move the existing package variables, open the
# package-owned event stages, extend the shared focus lanes with proof flags,
# and close transient state after sovereign completion. They do not create
# tags, models, high-chaos routes, or recurring world scans.
# ============================================================================

# ---------------------------------------------------------------------------
# South American package contracts
# ---------------------------------------------------------------------------

africa_south_america_open_world_package_congress = {
	if = {
		limit = {
			africa_world_package_is_installed = yes
			NOT = { has_country_flag = africa_world_package_sovereign_complete }
			NOT = { has_country_flag = africa_south_america_congress_event_seen }
		}
		set_country_flag = africa_south_america_three_regions_congress_open
		set_country_flag = africa_south_america_congress_event_seen
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.opening_congress }
		set_variable = { africa_world_package_crisis_result = constant:africa_world_package_crisis_result.none }
		country_event = { id = africa_world_package.500 }
	}
}

# The three named decisions use the existing regional ledger. Each action is
# guarded by a short-lived lock and becomes available again after the founding
# response resolves, which keeps the loop repeatable without a world scan.
africa_south_america_balance_the_three_regions = {
	africa_south_america_maybe_reopen_shared_lane_ratification = yes
	if = {
		limit = {
			africa_world_package_is_installed = yes
			NOT = { has_country_flag = africa_south_america_balance_action_lock }
		}
		set_country_flag = africa_south_america_balance_action_lock
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.constituent_bargaining }
		if = {
			limit = { check_variable = { africa_south_america_andean_voice < constant:africa_measure.medium } }
			add_to_variable = { africa_south_america_andean_voice = constant:africa_world_order.package_major_gain }
		}
		else_if = {
			limit = { check_variable = { africa_south_america_amazon_voice < constant:africa_measure.medium } }
			add_to_variable = { africa_south_america_amazon_voice = constant:africa_world_order.package_major_gain }
		}
		else = {
			add_to_variable = { africa_south_america_plata_voice = constant:africa_world_order.package_major_gain }
		}
		add_to_variable = { africa_south_america_indigenous_representation = constant:africa_world_order.package_minor_gain }
		add_to_variable = { africa_world_package_authority = constant:africa_world_order.package_minor_gain }
		africa_world_clamp_package_mechanic = yes
		africa_south_america_open_founding_dispute = yes
	}
}
