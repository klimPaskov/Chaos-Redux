# COUNTRY scope on one bounded roster target. Rebuilds the six documented
# promotion conditions from current state. The result is inspectable, can be
# refreshed, and never promotes or integrates the country by itself.
africa_priority_member_evaluate_promotion_candidate = {
	africa_priority_member_record_supported_carrier_origin = yes
	clr_country_flag = africa_priority_promotion_candidate
	clr_country_flag = africa_priority_promotion_survey_passed
	clr_country_flag = africa_priority_promotion_survey_deferred
	clr_country_flag = africa_priority_viable_territory
	clr_country_flag = africa_priority_functioning_institution
	clr_country_flag = africa_priority_distinct_role
	clr_country_flag = africa_priority_relationship_unresolved
	clr_country_flag = africa_priority_economic_or_strategic_function
	clr_country_flag = africa_priority_population_and_infrastructure_viable
	set_variable = { africa_priority_promotion_criteria_count = constant:africa_value.zero }
	if = {
		limit = { has_stability > constant:africa_priority_member_promotion.high_support_stability }
		set_variable = { africa_priority_local_support = constant:africa_priority_member_promotion.local_support_high }
	}
	else_if = {
		limit = { has_stability > constant:africa_priority_member_promotion.medium_support_stability }
		set_variable = { africa_priority_local_support = constant:africa_priority_member_promotion.local_support_medium }
	}
	else = {
		set_variable = { africa_priority_local_support = constant:africa_priority_member_promotion.local_support_low }
	}
	if = {
		limit = { africa_priority_member_has_viable_compact_base = yes }
		set_country_flag = africa_priority_viable_territory
		add_to_variable = { africa_priority_promotion_criteria_count = constant:africa_value.one }
	}
	if = {
		limit = { africa_priority_member_has_functioning_institution = yes }
		set_country_flag = africa_priority_functioning_institution
		add_to_variable = { africa_priority_promotion_criteria_count = constant:africa_value.one }
	}
	if = {
		limit = { africa_priority_member_has_documented_strategic_function = yes }
		set_country_flag = africa_priority_economic_or_strategic_function
		add_to_variable = { africa_priority_promotion_criteria_count = constant:africa_value.one }
	}
	if = {
		limit = { africa_priority_member_has_unresolved_league_problem = yes }
		set_country_flag = africa_priority_relationship_unresolved
		add_to_variable = { africa_priority_promotion_criteria_count = constant:africa_value.one }
	}
	if = {
		limit = { africa_priority_member_has_viable_population_and_infrastructure = yes }
		set_country_flag = africa_priority_population_and_infrastructure_viable
		add_to_variable = { africa_priority_promotion_criteria_count = constant:africa_value.one }
	}
	if = {
		limit = { africa_priority_member_has_distinct_package_identity = yes }
		set_country_flag = africa_priority_distinct_role
		add_to_variable = { africa_priority_promotion_criteria_count = constant:africa_value.one }
	}
	if = {
		limit = {
			africa_priority_member_has_valid_origin = yes
			NOT = { has_country_flag = africa_priority_promotion_requalification_pending }
			check_variable = {
				var = africa_priority_promotion_criteria_count
				value = constant:africa_priority_member_promotion.required_criteria
				compare = greater_than_or_equals
			}
			check_variable = {
				var = africa_priority_local_support
				value = constant:africa_action_threshold.minimum_confidence
				compare = greater_than_or_equals
			}
		}
		set_country_flag = africa_priority_promotion_candidate
		set_country_flag = africa_priority_promotion_survey_passed
	}
	else = {
		set_country_flag = africa_priority_promotion_survey_deferred
	}
}

# COUNTRY scope after a negotiated partial Action 102 result. Exactly one
# concrete obstacle is recorded so the compact settlement has a visible route
# back to candidacy rather than silently receiving the complete package.
africa_priority_member_record_partial_requalification_issue = {
	clr_country_flag = africa_priority_requalification_access
	clr_country_flag = africa_priority_requalification_overlap
	clr_country_flag = africa_priority_requalification_local_ratification
	set_country_flag = africa_priority_promotion_requalification_pending
	if = {
		limit = {
			check_variable = { africa_priority_local_support < constant:africa_priority_member_promotion.local_support_medium }
		}
		set_country_flag = africa_priority_requalification_local_ratification
	}
	else_if = {
		limit = {
			NOT = {
				any_owned_state = { infrastructure > constant:africa_priority_member_promotion.infrastructure_floor }
			}
		}
		set_country_flag = africa_priority_requalification_access
	}
	else = {
		set_country_flag = africa_priority_requalification_overlap
	}
}

# COUNTRY scope on the compact candidate. The host decision is available only
# when the recorded obstacle's live requirements are satisfied. Resolution
# clears the obstacle and rebuilds the dossier; it still does not activate the
# package, and Action 102 must produce a later full result.
africa_priority_member_resolve_requalification_issue = {
	if = {
		limit = { africa_priority_member_can_resolve_requalification = yes }
		clr_country_flag = africa_priority_promotion_requalification_pending
		clr_country_flag = africa_priority_requalification_access
		clr_country_flag = africa_priority_requalification_overlap
		clr_country_flag = africa_priority_requalification_local_ratification
		set_country_flag = africa_priority_requalification_resolved
		africa_priority_member_evaluate_promotion_candidate = yes
	}
}

# COUNTRY scope after complete package registration. The seven niche African
# country tags registered by Event 006 may receive the Event 012 priority-member
# tree only after their active Event 006 origin and full Event 006 tree are no
# longer present. While Event 006 owns the active origin or focus tree, record
# the skipped overlay and leave the Event 006 surface intact; Event 012
# decisions, ideas, forces, and AI still apply additively. The reusable carrier
# pool is exposed as `collection:chaosx_country_africa` for future African
# events. Existing vanilla carriers receive this tree only when they still use
# the generic tree.
africa_priority_member_ensure_focus_tree_loaded = {
	if = {
		limit = {
			africa_priority_member_has_package = yes
		}
		clr_country_flag = africa_priority_member_focus_tree_loaded
		clr_country_flag = africa_priority_member_focus_tree_overlay_skipped
		if = {
			limit = {
				OR = {
					has_country_flag = independence_wave_active_origin
					has_focus_tree = independence_wave_focus_tree
					is_independence_wave_registry_soviet_origin = yes
				}
			}
			# Event 006 and Soviet origins retain their meaningful focus surface.
			# The package payload remains available through additive ideas,
			# decisions, forces, and AI; a later lifecycle call may retry the
			# overlay after the protected origin/tree has been cleared.
			set_country_flag = africa_priority_member_focus_tree_overlay_skipped
		}
		else_if = {
			limit = { has_focus_tree = africa_priority_member_focus_tree }
			set_country_flag = africa_priority_member_focus_tree_loaded
		}
		else_if = {
			limit = {
				OR = {
					africa_priority_member_uses_independence_wave_country_tag = yes
					has_focus_tree = generic_focus
				}
			}
			load_focus_tree = {
				tree = africa_priority_member_focus_tree
				keep_completed = yes
			}
			set_country_flag = africa_priority_member_focus_tree_loaded
		}
		else = {
			set_country_flag = africa_priority_member_focus_tree_overlay_skipped
		}
		mark_focus_tree_layout_dirty = yes
	}
}
