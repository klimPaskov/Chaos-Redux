africa_priority_member_advance_one_league_stage = {
	if = {
		limit = { africa_relationship_is_outside = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.protected }
		africa_apply_relationship_transition = yes
	}
	else_if = {
		limit = { africa_relationship_is_protected = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.associate }
		africa_apply_relationship_transition = yes
	}
	else_if = {
		limit = { africa_relationship_is_associate = yes }
		if = {
			limit = { OR = { africa_priority_member_package_is_kanem_bornu = yes africa_priority_member_package_is_aksum = yes africa_priority_member_package_is_harar = yes africa_priority_member_package_is_lunda = yes } }
			set_country_flag = africa_priority_member_preferred_relationship_reached
		}
		else = {
			set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered }
			africa_apply_relationship_transition = yes
		}
	}
	else_if = {
		limit = { africa_relationship_is_chartered = yes }
		if = {
			limit = { OR = { africa_priority_member_package_is_sokoto = yes africa_priority_member_package_is_kilwa = yes } }
			set_country_flag = africa_priority_member_preferred_relationship_reached
		}
		else_if = {
			limit = { africa_member_can_enter_autonomous_federal = yes }
			set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal }
			africa_apply_relationship_transition = yes
			clr_country_flag = africa_priority_member_autonomy_review_pending
			set_country_flag = africa_priority_member_preferred_relationship_reached
		}
		else = { set_country_flag = africa_priority_member_autonomy_review_pending }
	}
	else_if = {
		limit = { africa_relationship_is_resistant = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered }
		africa_apply_relationship_transition = yes
	}
	else_if = {
		limit = { OR = { africa_relationship_is_leaving = yes africa_relationship_is_rival_bloc = yes } }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.associate }
		africa_apply_relationship_transition = yes
	}
	else_if = {
		limit = { OR = { africa_relationship_is_autonomous_federal = yes africa_relationship_is_integrated_region = yes } }
		set_country_flag = africa_priority_member_preferred_relationship_reached
	}
}
