# ============================================================================
# CHAOS REDUX - EVENT 163 DOCTOR WU EFFECTS
# ============================================================================
# One-shot weighted host selection, atomic host lifecycle, and the narrow
# Event 20 notification hook. This package never removes a disease state and
# never bypasses a disease decision's normal target, cost, or phase gates.
# ============================================================================

# COUNTRY scope.
doctor_wu_prepare_host_selection_weight = {
	set_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.base_weight }
	if = {
		limit = { doctor_wu_host_controls_severe_black_plague_state = yes }
		add_to_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.black_plague_severe_bonus }
	}
	if = {
		limit = { has_active_biowarfare_outbreak = yes }
		add_to_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.active_disease_bonus }
	}
	if = {
		limit = { has_tech = tech_field_hospital }
		add_to_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.field_hospital_bonus }
	}
	if = {
		limit = { has_tech = tech_field_hospital2 }
		add_to_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.advanced_field_hospital_bonus }
	}
	if = {
		limit = { num_of_factories > constant:doctor_wu_host_selection.large_industry_threshold }
		add_to_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.large_industry_bonus }
	}
	if = {
		limit = { has_war = yes }
		multiply_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_host_selection.war_penalty_factor }
	}
	round_temp_variable = doctor_wu_host_selection_weight
	clamp_temp_variable = {
		var = doctor_wu_host_selection_weight
		min = constant:doctor_wu_host_selection.minimum_weight
		max = constant:doctor_wu_host_selection.maximum_weight
	}
}

# COUNTRY scope.
doctor_wu_add_current_country_to_host_selection_pool = {
	doctor_wu_prepare_host_selection_weight = yes
	set_temp_variable = { doctor_wu_host_pool_weight_remaining = doctor_wu_host_selection_weight }
	while_loop_effect = {
		limit = { check_variable = { doctor_wu_host_pool_weight_remaining > constant:doctor_wu_value.zero } }
		add_to_temp_array = { doctor_wu_host_selection_pool = THIS }
		add_to_temp_variable = { doctor_wu_host_selection_pool_count = constant:doctor_wu_value.one }
		subtract_from_temp_variable = { doctor_wu_host_pool_weight_remaining = constant:doctor_wu_value.one }
	}
	set_temp_variable = { doctor_wu_host_pool_weight_remaining = constant:doctor_wu_value.zero }
	set_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_value.zero }
}

# Any scope. One explicit fire-once world scan.
doctor_wu_select_weighted_prefire_host = {
	clear_temp_array = doctor_wu_host_selection_pool
	set_temp_variable = { doctor_wu_host_selection_pool_count = constant:doctor_wu_value.zero }
	every_country = {
		limit = { doctor_wu_is_eligible_host = yes }
		doctor_wu_add_current_country_to_host_selection_pool = yes
	}
	if = {
		limit = { check_variable = { doctor_wu_host_selection_pool_count > constant:doctor_wu_value.zero } }
		random_scope_in_array = {
			array = doctor_wu_host_selection_pool
			save_event_target_as = doctor_wu_prefire_host
		}
	}
	clear_temp_array = doctor_wu_host_selection_pool
	set_temp_variable = { doctor_wu_host_selection_pool_count = constant:doctor_wu_value.zero }
	set_temp_variable = { doctor_wu_host_pool_weight_remaining = constant:doctor_wu_value.zero }
	set_temp_variable = { doctor_wu_host_selection_weight = constant:doctor_wu_value.zero }
}

# Any scope. Output is a regular target that persists through the dispatcher chain.
doctor_wu_prepare_random_event_fire = {
	set_temp_variable = { doctor_wu_prefire_ready = constant:doctor_wu_value.zero }
	if = {
		limit = { doctor_wu_automatic_event_is_available = yes }
		doctor_wu_select_weighted_prefire_host = yes
		if = {
			limit = { has_event_target = doctor_wu_prefire_host }
			set_temp_variable = { doctor_wu_prefire_ready = constant:doctor_wu_value.one }
		}
	}
}

# Any scope. Clears only the persistent lifecycle records.
doctor_wu_clear_active_host = {
	if = {
		limit = { has_event_target = doctor_wu_current_host }
		event_target:doctor_wu_current_host = {
			if = {
				limit = { exists = yes }
				clr_country_flag = doctor_wu_current_host
			}
		}
		clear_global_event_target = doctor_wu_current_host
	}
	clr_global_flag = doctor_wu_active
}

# COUNTRY scope. The caller has already validated the new host.
doctor_wu_initialize_host = {
	if = {
		limit = { doctor_wu_is_eligible_host = yes }
		doctor_wu_clear_active_host = yes
		set_country_flag = doctor_wu_current_host
		set_country_flag = doctor_wu_host_history
		save_global_event_target_as = doctor_wu_current_host
		set_global_flag = doctor_wu_active
		set_global_flag = doctor_wu_event_resolved
		if = {
			limit = { has_global_flag = black_plague_system_started }
			black_plague_on_doctor_wu_host_committed = yes
		}
	}
}

# COUNTRY scope. Recipient validation occurs before the old host is touched.
doctor_wu_transfer_host_atomically = {
	if = {
		limit = { doctor_wu_can_receive_transfer = yes }
		doctor_wu_initialize_host = yes
	}
}

# COUNTRY scope opening outcomes.
doctor_wu_accept_public_mandate = {
	set_country_flag = doctor_wu_public_mandate
	doctor_wu_initialize_host = yes
}

doctor_wu_accept_protected_service = {
	set_country_flag = doctor_wu_protected_clinical_service
	doctor_wu_initialize_host = yes
}

doctor_wu_refuse_host = {
	set_country_flag = doctor_wu_host_rejected
	set_global_flag = doctor_wu_event_resolved
}

# COUNTRY scope. Called by the requester-side timed decision while FROM is
# its selected state. The host receives a regular event with both targets.
doctor_wu_send_foreign_access_request = {
	if = {
		limit = {
			doctor_wu_can_request_foreign_access = yes
			doctor_wu_has_current_host = yes
		}
		set_country_flag = doctor_wu_foreign_request_pending
		FROM = { save_event_target_as = doctor_wu_request_state }
		save_event_target_as = doctor_wu_requester
		event_target:doctor_wu_current_host = {
			country_event = { id = chaosx.nr163.3 }
		}
	}
}

# COUNTRY scope of the Doctor Wu host. The requester target is preserved by
# the event chain and gets a bounded access window, not a permanent alliance.
doctor_wu_accept_foreign_access_request = {
	if = {
		limit = {
			has_event_target = doctor_wu_requester
			doctor_wu_can_pay_foreign_access_request = yes
			event_target:doctor_wu_requester = {
				has_country_flag = doctor_wu_foreign_request_pending
			}
		}
		remove_equipment_to_stockpile = { type = support_equipment amount = constant:doctor_wu_access.host_support_equipment }
		remove_equipment_to_stockpile = { type = motorized_equipment amount = constant:doctor_wu_access.host_motorized_equipment }
		set_temp_variable = { doctor_wu_host_fuel_payment = constant:doctor_wu_access.host_fuel }
		subtract_from_fuel = doctor_wu_host_fuel_payment
		event_target:doctor_wu_requester = {
			clr_country_flag = doctor_wu_foreign_request_pending
			set_temp_variable = { doctor_wu_foreign_access_days = constant:doctor_wu_access.foreign_access_days }
			set_country_flag = { flag = doctor_wu_foreign_access_active days = doctor_wu_foreign_access_days }
			set_country_flag = { flag = black_plague_countermeasure_foreign_exchange_active days = doctor_wu_foreign_access_days }
			set_country_flag = { flag = black_plague_doctor_wu_foreign_mission days = doctor_wu_foreign_access_days }
			set_temp_variable = { black_plague_countermeasure_gain = constant:doctor_wu_access.foreign_countermeasure_gain }
			black_plague_record_countermeasure_findings = yes
			black_plague_recompute_country_response_capacity = yes
			country_event = { id = chaosx.nr163.4 }
		}
		if = {
			limit = { has_event_target = doctor_wu_request_state }
			event_target:doctor_wu_request_state = {
				set_temp_variable = { doctor_wu_foreign_treatment_days = constant:doctor_wu_access.foreign_treatment_days }
				set_state_flag = { flag = black_plague_doctor_wu_foreign_treatment days = doctor_wu_foreign_treatment_days }
			}
		}
	}
}

# COUNTRY scope of the Doctor Wu host. Decline still records a cooldown so a
# country cannot spam the same request every day.
doctor_wu_decline_foreign_access_request = {
	if = {
		limit = { has_event_target = doctor_wu_requester }
		event_target:doctor_wu_requester = {
			clr_country_flag = doctor_wu_foreign_request_pending
			set_temp_variable = { doctor_wu_request_cooldown_days = constant:doctor_wu_access.request_cooldown_days }
			set_country_flag = { flag = doctor_wu_foreign_request_cooldown days = doctor_wu_request_cooldown_days }
			country_event = { id = chaosx.nr163.4 }
		}
	}
}
