# Minor/regional custodian scope. The defeat caller saves the defeated KRG as a
# regular event target before entering this effect. Exactly one dominant legacy
# is selected from real project history for event text and persistent history.
brilliant_scientist_aftermath_select_minor_dominant_legacy = {
	if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { has_country_flag = brilliant_scientist_singularity_late_construction_ever } }
		set_country_flag = brilliant_scientist_aftermath_minor_singularity_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_temporal_deployment_completed check_variable = { brilliant_scientist_project_force_temporal_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_temporal_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_alien_arms_deployment_completed check_variable = { brilliant_scientist_project_force_alien_arms_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_alien_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_biological_weapons_deployment_completed has_country_flag = brilliant_scientist_biological_weapons_weaponization_completed } } }
		set_country_flag = brilliant_scientist_aftermath_minor_biological_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_teleportation_deployment_completed check_variable = { brilliant_scientist_project_force_teleportation_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_portal_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_xenobiological_synthesis_deployment_completed check_variable = { brilliant_scientist_project_force_xenobiological_synthesis_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_xenobiological_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_paleogenetics_deployment_completed check_variable = { brilliant_scientist_project_force_paleogenetics_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_paleogenetic_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_robotics_deployment_completed check_variable = { brilliant_scientist_project_force_robotics_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_machine_legacy
	}
	else_if = {
		limit = { event_target:brilliant_scientist_defeated_state_actor = { OR = { has_country_flag = brilliant_scientist_cloning_deployment_completed check_variable = { brilliant_scientist_project_force_cloning_formation_count > constant:brilliant_scientist_value.zero } } } }
		set_country_flag = brilliant_scientist_aftermath_minor_clone_legacy
	}
	else = {
		set_country_flag = brilliant_scientist_aftermath_minor_prototype_archive_legacy
	}
}

	set_variable = { brilliant_scientist_aftermath_inspection_certification_date = global.date }
	add_ideas = brilliant_scientist_aftermath_inspection_authority

		limit = {
			has_event_target = brilliant_scientist_defeated_state_actor

		set_country_flag = brilliant_scientist_aftermath_settlement_completed
