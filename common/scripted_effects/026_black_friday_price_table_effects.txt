# ============================================================================
# CHAOS REDUX - EVENT 026 BLACK FRIDAY SALE PRICE TABLE
# ============================================================================
# Native decision costs cannot be discounted by a modifier, so Chaos Redux
# decisions read their political power price from one global variable per
# registered ordinary amount: cost = global.black_friday_price_<key>.
# The engine computes display, affordability, debit, and AI evaluation from
# that single value, so they cannot disagree.
#
# Outside a sale every registered price equals its ordinary amount. Sale
# activation sets each price to ceil(ordinary x payment ratio / ratio_basis),
# at least 1 for a positive price; expiry and terminal cleanup restore the
# ordinary amounts. Zero prices stay zero. The table is rebuilt whenever
# constant:black_friday_event.price_table_version changes, so running
# campaigns pick up newly registered keys.
#
# Registering a price: add one entry to black_friday_price_table_refresh_entries
# (literal amounts use constant:black_friday_price_table.pp_<N>, owner prices
# use the owner's own constant), bump price_table_version, and set the
# decision's cost to the matching global.black_friday_price_<key>.
# Contract: 026_black_friday_price_table_effects.md.
# ============================================================================

# ANY scope. Inputs: temp black_friday_price_ordinary and temp
# black_friday_price_ratio (payment ratio in basis points). Output: caller-owned
# temp black_friday_price_payable. Non-positive prices and full-price ratios
# return the ordinary amount unchanged.
black_friday_price_table_quote = {
	set_temp_variable = { black_friday_price_payable = black_friday_price_ordinary }
	if = {
		limit = {
			check_variable = { black_friday_price_ordinary > constant:black_friday_event.zero }
			check_variable = {
				var = black_friday_price_ratio
				value = constant:black_friday_event.ratio_basis
				compare = less_than
			}
		}
		set_temp_variable = { black_friday_price_percent = black_friday_price_ratio }
		divide_temp_variable = { black_friday_price_percent = constant:black_friday_event.ratio_per_percent }
		multiply_temp_variable = { black_friday_price_payable = black_friday_price_percent }
		divide_temp_variable = { black_friday_price_payable = constant:black_friday_event.percent_basis }
		set_temp_variable = { black_friday_price_whole = black_friday_price_payable }
		round_temp_variable = black_friday_price_whole
		if = {
			limit = {
				check_variable = {
					var = black_friday_price_whole
					value = black_friday_price_payable
					compare = less_than
				}
			}
			add_to_temp_variable = { black_friday_price_whole = constant:black_friday_event.one }
		}
		set_temp_variable = { black_friday_price_payable = black_friday_price_whole }
		clamp_temp_variable = {
			var = black_friday_price_payable
			min = constant:black_friday_event.one
		}
	}
}

# ANY scope. Sets every registered price to the active sale price snapshotted
# in global.black_friday_discount_ratio.
black_friday_price_table_apply = {
	set_temp_variable = { black_friday_price_ratio = global.black_friday_discount_ratio }
	black_friday_price_table_refresh_entries = yes
}

# ANY scope. Restores every registered price to its ordinary amount.
black_friday_price_table_restore = {
	set_temp_variable = { black_friday_price_ratio = constant:black_friday_event.ratio_basis }
	black_friday_price_table_refresh_entries = yes
}

# ANY scope. Builds the table once per price_table_version: at the active sale
# price while a sale is running, otherwise at ordinary prices.
black_friday_price_table_initialize = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = global.black_friday_price_table_version
					value = constant:black_friday_event.price_table_version
					compare = equals
				}
			}
		}
		if = {
			limit = { black_friday_is_active = yes }
			black_friday_price_table_apply = yes
		}
		else = {
			black_friday_price_table_restore = yes
		}
		set_variable = { global.black_friday_price_table_version = constant:black_friday_event.price_table_version }
	}
}
