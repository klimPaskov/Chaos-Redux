infantry_spawn_achievement_register_current_pretechnology_gate = {
	set_temp_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_gate_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = infantry_spawn_achievement_gate_scan_index value = infantry_spawn_achievement_pretech_unit_uid_entries^num compare = less_than }
		}
		if = {
			limit = {
				check_variable = { var = infantry_spawn_achievement_pretech_unit_uid_entries^infantry_spawn_achievement_gate_scan_index value = infantry_spawn_current_unit_uid compare = equals }
				check_variable = { var = infantry_spawn_achievement_pretech_gate_entries^infantry_spawn_achievement_gate_scan_index value = infantry_spawn_achievement_current_technology_gate compare = equals }
			}
			set_temp_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.one }
		}
		add_to_temp_variable = { infantry_spawn_achievement_gate_scan_index = constant:infantry_spawn_achievement_value.one }
	}
	if = {
		limit = { check_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.zero } }
		add_to_array = { array = infantry_spawn_achievement_pretech_unit_uid_entries value = infantry_spawn_current_unit_uid }
		add_to_array = { array = infantry_spawn_achievement_pretech_gate_entries value = infantry_spawn_achievement_current_technology_gate }
	}
	if = {
		limit = { has_event_target = infantry_spawn_current_live_division }
		infantry_spawn_achievement_mark_current_division_technology_gate = yes
		event_target:infantry_spawn_current_live_division = {
			set_variable = { infantry_spawn_event19_pretechnology_formation = constant:infantry_spawn_achievement_value.one }
		}
	}
}

		limit = { check_variable = { var = infantry_spawn_achievement_unit_scan_index value = infantry_spawn_unit_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_unit_generation_uid_entries^infantry_spawn_achievement_unit_scan_index value = infantry_spawn_achievement_closeout_generation_uid compare = equals }
				NOT = { check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_achievement_unit_scan_index value = constant:infantry_spawn_unit_status.demobilized compare = equals } }
			}
			set_temp_variable = { infantry_spawn_achievement_closeout_all_units_demobilized = constant:infantry_spawn_achievement_value.zero }
		}
		add_to_temp_variable = { infantry_spawn_achievement_unit_scan_index = constant:infantry_spawn_achievement_value.one }
	}

	set_temp_variable = { infantry_spawn_achievement_obligation_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_achievement_obligation_scan_index value = infantry_spawn_obligation_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_obligation_generation_uid_entries^infantry_spawn_achievement_obligation_scan_index value = infantry_spawn_achievement_closeout_generation_uid compare = equals }
				OR = {
					check_variable = { var = infantry_spawn_obligation_status_entries^infantry_spawn_achievement_obligation_scan_index value = constant:infantry_spawn_obligation_state.outstanding compare = equals }
					check_variable = { var = infantry_spawn_obligation_status_entries^infantry_spawn_achievement_obligation_scan_index value = constant:infantry_spawn_obligation_state.servicing compare = equals }
					check_variable = { var = infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_achievement_obligation_scan_index value = constant:infantry_spawn_achievement_threshold.unresolved_debt_epsilon compare = greater_than_or_equals }
				}
			}
			set_temp_variable = { infantry_spawn_achievement_closeout_has_unresolved_obligation = constant:infantry_spawn_achievement_value.one }
		}
		add_to_temp_variable = { infantry_spawn_achievement_obligation_scan_index = constant:infantry_spawn_achievement_value.one }
	}
}

# Country scope. Call inside infantry_spawn_close_resolved_generations directly
# after a row becomes resolved, while infantry_spawn_closeout_generation_uid is
# still the row being closed.
infantry_spawn_achievement_evaluate_closed_generation = {
	set_country_flag = infantry_spawn_event19_achievement_history
	set_temp_variable = { infantry_spawn_achievement_closeout_generation_uid = infantry_spawn_closeout_generation_uid }
	if = {
		limit = { check_variable = { infantry_spawn_achievement_closeout_generation_uid < constant:infantry_spawn_achievement_value.one } }
		set_temp_variable = { infantry_spawn_achievement_closeout_generation_uid = infantry_spawn_last_closed_generation_uid }
	}
	infantry_spawn_achievement_audit_closeout_generation = yes

	if = {
		limit = {
			check_variable = { infantry_spawn_achievement_closeout_generation_row > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_achievement_closeout_generation_unit_count value = constant:infantry_spawn_achievement_threshold.very_large_generation_units compare = greater_than_or_equals }
			check_variable = { infantry_spawn_achievement_closeout_has_unresolved_lot = constant:infantry_spawn_achievement_value.zero }
			check_variable = { infantry_spawn_achievement_closeout_has_unresolved_obligation = constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = infantry_spawn_muster_control value = constant:infantry_spawn_achievement_threshold.high_control compare = greater_than_or_equals }
			check_variable = { var = infantry_spawn_army_congestion value = constant:infantry_spawn_achievement_threshold.low_congestion_exclusive compare = less_than }
			NOT = { has_country_flag = infantry_spawn_claimant_takeover_complete }
			NOT = { has_country_flag = infantry_spawn_achievement_revolt_history }
			NOT = { has_country_flag = infantry_spawn_achievement_forced_setup_used }
			NOT = { has_country_flag = infantry_spawn_achievement_debug_completion_used }
		}
		set_country_flag = infantry_spawn_achievement_every_rifle_accounted_for_ready
	}

	if = {
		limit = {
			check_variable = { infantry_spawn_achievement_closeout_generation_row > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_achievement_closeout_generation_unit_count value = constant:infantry_spawn_achievement_threshold.large_generation_units compare = greater_than_or_equals }
			check_variable = { var = infantry_spawn_achievement_closeout_integrated_unit_count value = constant:infantry_spawn_achievement_threshold.train_integrated_units compare = greater_than_or_equals }
			check_variable = { var = infantry_spawn_achievement_rail_generation_uid value = infantry_spawn_achievement_closeout_generation_uid compare = equals }
			has_country_flag = infantry_spawn_achievement_rail_attempt_active
			has_country_flag = infantry_spawn_achievement_rail_corridor_verified
			NOT = { has_country_flag = infantry_spawn_achievement_rail_corridor_failed }
			NOT = { has_country_flag = infantry_spawn_achievement_emergency_integration_used }
			infantry_spawn_achievement_rail_connection_is_live = yes
		}
		set_country_flag = infantry_spawn_achievement_no_room_on_the_train_ready
		clr_country_flag = infantry_spawn_achievement_rail_attempt_active
		clear_variable = infantry_spawn_achievement_rail_proof_expiry_date
	}

	# Closing the tracked generation consumes the proof even when the generation
	# misses the size or integration thresholds. This prevents a verified rail
	# corridor from keeping the daily continuity event alive indefinitely.
	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_rail_attempt_active
			check_variable = { var = infantry_spawn_achievement_rail_generation_uid value = infantry_spawn_achievement_closeout_generation_uid compare = equals }
		}
		infantry_spawn_achievement_mark_rail_proof_failure = yes
	}

	if = {
		limit = {
			check_variable = { infantry_spawn_achievement_closeout_generation_row > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_achievement_closeout_generation_unit_count value = constant:infantry_spawn_achievement_threshold.very_large_generation_units compare = greater_than_or_equals }
			check_variable = { infantry_spawn_achievement_closeout_all_lots_supervised_demobilized = constant:infantry_spawn_achievement_value.one }
			check_variable = { infantry_spawn_achievement_closeout_all_units_demobilized = constant:infantry_spawn_achievement_value.one }
			check_variable = { infantry_spawn_achievement_closeout_has_unresolved_obligation = constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = infantry_spawn_muster_control value = constant:infantry_spawn_achievement_threshold.high_control compare = greater_than_or_equals }
			NOT = { has_country_flag = infantry_spawn_achievement_revolt_history }
			NOT = { has_country_flag = infantry_spawn_claimant_takeover_complete }
			NOT = { has_country_flag = infantry_spawn_achievement_emergency_integration_used }
			NOT = { has_country_flag = infantry_spawn_achievement_teardown_invariant_failure }
			NOT = { has_country_flag = infantry_spawn_achievement_forced_setup_used }
			NOT = { has_country_flag = infantry_spawn_achievement_debug_completion_used }
		}
		set_country_flag = infantry_spawn_achievement_quiet_demobilisation_ready
	}
}

		limit = { check_variable = { infantry_spawn_lot_row_index > constant:infantry_spawn_event.invalid_index } }
		set_variable = { infantry_spawn_achievement_rail_generation_uid = infantry_spawn_lot_generation_uid_entries^infantry_spawn_lot_row_index }

		set_country_flag = infantry_spawn_achievement_rail_attempt_active
		clr_country_flag = infantry_spawn_achievement_rail_corridor_verified
		clr_country_flag = infantry_spawn_achievement_rail_corridor_failed
		if = {
			limit = { NOT = { infantry_spawn_achievement_rail_connection_is_live = yes } }
			infantry_spawn_achievement_mark_rail_proof_failure = yes
		}
		if = {
			limit = { has_country_flag = infantry_spawn_achievement_rail_attempt_active }
			infantry_spawn_schedule_achievement_continuity_pulse = yes
		}
	}
}

# Lightweight continuity scheduler for rail, claimant-leadership, and scenario
# proof. These checks need daily sampling, but they do not need the full Event
# 19 ledger reconciliation, AI, claimant, family, and pressure pulse.
infantry_spawn_schedule_achievement_continuity_pulse = {
	if = {
		limit = {
			OR = {
				has_country_flag = infantry_spawn_achievement_rail_attempt_active
				AND = {
					has_country_flag = infantry_spawn_achievement_claimant_takeover_started
					has_country_flag = infantry_spawn_achievement_claimant_survival_pending
					NOT = { has_country_flag = infantry_spawn_achievement_claimant_leadership_broken }
				}
				has_country_flag = infantry_spawn_scenario_origin_pulse_active
			}
			NOT = { has_country_flag = infantry_spawn_achievement_continuity_pulse_scheduled }
		}
		set_country_flag = infantry_spawn_achievement_continuity_pulse_scheduled
		set_temp_variable = { infantry_spawn_achievement_continuity_delay_days = constant:infantry_spawn_achievement_threshold.continuity_pulse_days }
		country_event = { id = chaosx.nr19.901 days = infantry_spawn_achievement_continuity_delay_days }
	}
}

infantry_spawn_run_achievement_continuity_pulse = {
	clr_country_flag = infantry_spawn_achievement_continuity_pulse_scheduled
	infantry_spawn_achievement_country_pulse = yes
	infantry_spawn_schedule_achievement_continuity_pulse = yes
}

# Public Event 019 country-pulse hook. It never iterates over other countries.
infantry_spawn_achievement_country_pulse = {
	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_rail_attempt_active
			NOT = { infantry_spawn_achievement_rail_connection_is_live = yes }
		}
		infantry_spawn_achievement_mark_rail_proof_failure = yes
	}
	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_rail_attempt_active
			has_variable = infantry_spawn_achievement_rail_proof_expiry_date
			check_variable = {
				var = global.date
				value = infantry_spawn_achievement_rail_proof_expiry_date
				compare = greater_than_or_equals
			}
		}
		infantry_spawn_achievement_mark_rail_proof_failure = yes
	}
	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_claimant_takeover_started
			NOT = { has_country_flag = infantry_spawn_achievement_claimant_leadership_broken }
			NOT = { infantry_spawn_achievement_claimant_is_still_country_leader = yes }
		}
		set_country_flag = infantry_spawn_achievement_claimant_leadership_broken
		clr_country_flag = infantry_spawn_achievement_claimant_survival_pending
		clr_country_flag = infantry_spawn_achievement_claimant_major_victory
	}
	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_claimant_takeover_started
			has_capitulated = yes

# Country scope for the opponent. The attacker provides unscoped expected
# nonce and country-ID temporaries before entering this scope.
infantry_spawn_achievement_cleanup_combat_trial_opponent = {
	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_combat_trial_opponent
			check_variable = { var = infantry_spawn_achievement_combat_trial_opponent_nonce value = infantry_spawn_achievement_combat_trial_cleanup_expected_nonce compare = equals }
			check_variable = { var = infantry_spawn_achievement_combat_trial_opponent_attacker value = infantry_spawn_achievement_combat_trial_cleanup_expected_attacker compare = equals }
		}
		set_temp_variable = { infantry_spawn_achievement_combat_trial_defender_matches = constant:infantry_spawn_achievement_value.zero }
		every_country_division = {
			limit = {
				check_variable = { var = infantry_spawn_achievement_combat_trial_defender_identity value = constant:infantry_spawn_achievement_value.one compare = equals }
				check_variable = { var = infantry_spawn_achievement_combat_trial_nonce value = infantry_spawn_achievement_combat_trial_cleanup_expected_nonce compare = equals }
				check_variable = { var = infantry_spawn_achievement_combat_trial_attacker_country value = infantry_spawn_achievement_combat_trial_cleanup_expected_attacker compare = equals }
				check_variable = { var = infantry_spawn_achievement_combat_trial_defender_unit_id value = constant:infantry_spawn_achievement_combat_trial.defender_unit_id compare = equals }
			}
			add_to_temp_variable = { infantry_spawn_achievement_combat_trial_defender_matches = constant:infantry_spawn_achievement_value.one }
		}
		if = {
			limit = { check_variable = { var = infantry_spawn_achievement_combat_trial_defender_matches value = constant:infantry_spawn_achievement_combat_trial.defender_divisions compare = equals } }
			set_temp_variable = { infantry_spawn_achievement_combat_trial_defender_unit_id = constant:infantry_spawn_achievement_combat_trial.defender_unit_id }
			delete_unit = {
				division_template = "Event 19 Controlled Trial Detachment"
				id = infantry_spawn_achievement_combat_trial_defender_unit_id
				disband = no
			}
			delete_unit_template_and_units = {
				division_template = "Event 19 Controlled Trial Detachment"
				disband = no
			}
			set_temp_variable = { infantry_spawn_achievement_combat_trial_defender_matches_after_cleanup = constant:infantry_spawn_achievement_value.zero }
			every_country_division = {
				limit = {
					check_variable = { var = infantry_spawn_achievement_combat_trial_defender_identity value = constant:infantry_spawn_achievement_value.one compare = equals }
					check_variable = { var = infantry_spawn_achievement_combat_trial_nonce value = infantry_spawn_achievement_combat_trial_cleanup_expected_nonce compare = equals }
					check_variable = { var = infantry_spawn_achievement_combat_trial_attacker_country value = infantry_spawn_achievement_combat_trial_cleanup_expected_attacker compare = equals }
					check_variable = { var = infantry_spawn_achievement_combat_trial_defender_unit_id value = constant:infantry_spawn_achievement_combat_trial.defender_unit_id compare = equals }
				}
				add_to_temp_variable = { infantry_spawn_achievement_combat_trial_defender_matches_after_cleanup = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = {
					check_variable = { var = infantry_spawn_achievement_combat_trial_defender_matches_after_cleanup value = constant:infantry_spawn_achievement_value.zero compare = equals }
					NOT = { has_template = "Event 19 Controlled Trial Detachment" }
				}
				clr_country_flag = infantry_spawn_achievement_combat_trial_opponent
				clear_variable = infantry_spawn_achievement_combat_trial_opponent_nonce
				clear_variable = infantry_spawn_achievement_combat_trial_opponent_attacker
				clear_variable = infantry_spawn_achievement_combat_trial_opponent_type
			}
			else = {
				set_country_flag = infantry_spawn_achievement_combat_trial_cleanup_quarantined
			}
		}
		else = {
			set_country_flag = infantry_spawn_achievement_combat_trial_cleanup_quarantined
		}
	}
}

# Country scope with FROM equal to the selected attacker state.
infantry_spawn_achievement_start_combat_trial_from_target_state = {
	infantry_spawn_achievement_set_current_combat_trial_costs = yes
	if = {
		limit = {
			infantry_spawn_achievement_combat_trial_country_is_available = yes
			check_variable = { infantry_spawn_achievement_current_combat_trial_army_experience_cost > constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = army_experience value = infantry_spawn_achievement_current_combat_trial_army_experience_cost compare = greater_than_or_equals }
			check_variable = { var = command_power value = infantry_spawn_achievement_current_combat_trial_command_power_cost compare = greater_than_or_equals }
			FROM = {
				infantry_spawn_achievement_combat_trial_attacker_state_is_structurally_safe = yes
				any_state_division = { infantry_spawn_achievement_current_combat_trial_division_is_eligible = yes }
			}
		}
		FROM = {
			save_event_target_as = infantry_spawn_achievement_combat_trial_attacker_state
			random_state_division = {
				limit = { infantry_spawn_achievement_current_combat_trial_division_is_eligible = yes }
				save_event_target_as = infantry_spawn_achievement_battle_division
			}
			random_neighbor_state = {
				limit = { infantry_spawn_achievement_combat_trial_defender_state_is_safe = yes }
				save_event_target_as = infantry_spawn_achievement_combat_trial_defender_state
				OWNER = { save_event_target_as = infantry_spawn_achievement_combat_trial_defender_country }
			}
		}
		if = {
			limit = {
				has_event_target = infantry_spawn_achievement_battle_division
				has_event_target = infantry_spawn_achievement_combat_trial_attacker_state
				has_event_target = infantry_spawn_achievement_combat_trial_defender_state
				has_event_target = infantry_spawn_achievement_combat_trial_defender_country
			}
			infantry_spawn_achievement_prepare_exact_battle_division = yes
			if = {
				limit = { check_variable = { var = infantry_spawn_achievement_exact_division_is_valid value = constant:infantry_spawn_achievement_value.one compare = equals } }
				infantry_spawn_achievement_scan_exact_division_signature = yes
				infantry_spawn_achievement_check_exact_unit_pretechnology_gate = yes
				if = {
					limit = { infantry_spawn_achievement_exact_division_matches_current_combat_trial = yes }
					infantry_spawn_achievement_freeze_current_combat_trial = yes
					infantry_spawn_achievement_create_combat_trial_defender = yes
					if = {
						limit = { infantry_spawn_achievement_combat_trial_frozen_pair_is_intact = yes }
						start_border_war = {
							change_state_after_war = no
							combat_width = @INFANTRY_SPAWN_COMBAT_TRIAL_COMBAT_WIDTH
							minimum_duration_in_days = @INFANTRY_SPAWN_COMBAT_TRIAL_MINIMUM_DURATION_DAYS
							attacker = {
								state = var:infantry_spawn_achievement_combat_trial_attacker_state
								num_provinces = @INFANTRY_SPAWN_COMBAT_TRIAL_NUM_PROVINCES
								on_win = chaosx.nr19.920
								on_lose = chaosx.nr19.921
								on_cancel = chaosx.nr19.922
							}
							defender = {
								state = var:infantry_spawn_achievement_combat_trial_defender_state
								num_provinces = @INFANTRY_SPAWN_COMBAT_TRIAL_NUM_PROVINCES
								on_win = chaosx.nr19.923
								on_lose = chaosx.nr19.924
								on_cancel = chaosx.nr19.925
							}
						}
						if = {
							limit = {
								var:infantry_spawn_achievement_combat_trial_attacker_state = { has_border_war = yes }
								var:infantry_spawn_achievement_combat_trial_defender_state = { has_border_war = yes }
							}
							infantry_spawn_achievement_debit_current_combat_trial_costs = yes
							set_country_flag = infantry_spawn_achievement_combat_trial_started
							activate_mission = infantry_spawn_achievement_combat_trial_mission
						}
						else = { infantry_spawn_achievement_cleanup_combat_trial = yes }
					}
					else = { infantry_spawn_achievement_cleanup_combat_trial = yes }
				}
			}
		}
	}
}
