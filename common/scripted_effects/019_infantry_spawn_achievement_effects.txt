# ============================================================================
# CHAOS REDUX - EVENT 019 ACHIEVEMENT TRACKING EFFECTS
# ============================================================================
# Exact generation, lot, unit, obligation, claimant, derivative, rail, and
# scenario history for the Event 019 achievement set.
#
# Public hooks are documented in the achievement handoff. The four battle
# achievements share one exact-division input contract. The public
# on_army_leader_won_combat callback cannot satisfy that contract because it

# Inputs: infantry_spawn_current_unit_uid and
# infantry_spawn_achievement_current_technology_gate.
infantry_spawn_achievement_register_current_pretechnology_gate = {
	set_temp_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_gate_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = infantry_spawn_achievement_gate_scan_index value = infantry_spawn_achievement_pretech_unit_uid_entries^num compare = less_than }
		}
		if = {
			limit = {
				check_variable = { var = infantry_spawn_achievement_pretech_unit_uid_entries^infantry_spawn_achievement_gate_scan_index value = infantry_spawn_current_unit_uid compare = equals }
				check_variable = { var = infantry_spawn_achievement_pretech_gate_entries^infantry_spawn_achievement_gate_scan_index value = infantry_spawn_achievement_current_technology_gate compare = equals }
			}
			set_temp_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.one }
		}
		add_to_temp_variable = { infantry_spawn_achievement_gate_scan_index = constant:infantry_spawn_achievement_value.one }
	}
	if = {
		limit = { check_variable = { infantry_spawn_achievement_gate_pair_exists = constant:infantry_spawn_achievement_value.zero } }
		add_to_array = { array = infantry_spawn_achievement_pretech_unit_uid_entries value = infantry_spawn_current_unit_uid }
		add_to_array = { array = infantry_spawn_achievement_pretech_gate_entries value = infantry_spawn_achievement_current_technology_gate }
	}
}

# Country scope. Call after an exact Event 019 division has been captured in
# infantry_spawn_current_live_division. It is idempotent and may be called a
# second time after the Evolution III registry finishes scoring its manifest.
infantry_spawn_achievement_register_current_generated_division = {
	if = {
		limit = {
			has_event_target = infantry_spawn_current_live_division
			check_variable = { infantry_spawn_current_unit_uid > constant:infantry_spawn_achievement_value.zero }
		}
		set_country_flag = infantry_spawn_event19_achievement_history

		if = {
			limit = { check_variable = { infantry_spawn_need_light_tank_chassis > constant:infantry_spawn_achievement_value.zero } NOT = { OR = { has_tech = basic_light_tank_chassis has_tech = basic_light_tank } } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.light_armor }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_need_mechanized_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { has_tech = mechanised_infantry } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.mechanised_infantry }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_need_armored_car_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { has_tech = armored_car1 } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.armored_car }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_need_amphibious_mechanized_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { has_tech = amphibious_mechanized_infantry } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.amphibious_mechanized }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_need_helicopter_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { is_special_project_completed = sp:sp_air_helicopter } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.helicopter }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_need_artillery_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { has_tech = gw_artillery } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.artillery }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_need_anti_tank_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { has_tech = interwar_antitank } }
			set_temp_variable = { infantry_spawn_achievement_current_technology_gate = constant:infantry_spawn_achievement_technology_gate.anti_tank }
			infantry_spawn_achievement_register_current_pretechnology_gate = yes

		if = {
			limit = { check_variable = { infantry_spawn_need_anti_air_equipment > constant:infantry_spawn_achievement_value.zero } NOT = { has_tech = interwar_antiair } }

		while_loop_effect = {
			limit = { check_variable = { var = infantry_spawn_achievement_manifest_scan_index value = infantry_spawn_unit_registry_manifest_equipment_profiles^num compare = less_than } }
			set_temp_variable = { var = infantry_spawn_unit_registry_current_equipment_profile value = infantry_spawn_unit_registry_manifest_equipment_profiles^infantry_spawn_achievement_manifest_scan_index }

		}

	}
	else = {
		set_temp_variable = { infantry_spawn_achievement_closeout_generation_unit_count = constant:infantry_spawn_achievement_value.zero }
	}

	set_temp_variable = { infantry_spawn_achievement_closeout_has_unresolved_lot = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_closeout_has_unresolved_obligation = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_closeout_integrated_unit_count = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_closeout_all_lots_supervised_demobilized = constant:infantry_spawn_achievement_value.one }
	set_temp_variable = { infantry_spawn_achievement_closeout_all_units_demobilized = constant:infantry_spawn_achievement_value.one }

	set_temp_variable = { infantry_spawn_achievement_lot_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_achievement_lot_scan_index value = infantry_spawn_lot_uid_entries^num compare = less_than } }
		if = {
			limit = { check_variable = { var = infantry_spawn_lot_generation_uid_entries^infantry_spawn_achievement_lot_scan_index value = infantry_spawn_achievement_closeout_generation_uid compare = equals } }
			set_temp_variable = { infantry_spawn_achievement_closeout_lot_uid = infantry_spawn_lot_uid_entries^infantry_spawn_achievement_lot_scan_index }
			if = {
				limit = {
					OR = {
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.auditing compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.revealed compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.isolated compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.training compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.claimant_controlled compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.anomalous_containment compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.demobilizing compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.revolted compare = equals }
					}
				}
				set_temp_variable = { infantry_spawn_achievement_closeout_has_unresolved_lot = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.integrated compare = equals }
						check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.standardized compare = equals }
					}
				}
				add_to_temp_variable = { var = infantry_spawn_achievement_closeout_integrated_unit_count value = infantry_spawn_lot_unit_count_entries^infantry_spawn_achievement_lot_scan_index }
			}
			if = {
				limit = {
					OR = {
						NOT = { check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_achievement_lot_scan_index value = constant:infantry_spawn_lot_state.demobilized compare = equals } }
						NOT = { is_in_array = { array = infantry_spawn_achievement_supervised_demobilized_lot_uids value = infantry_spawn_achievement_closeout_lot_uid } }
					}
				}
				set_temp_variable = { infantry_spawn_achievement_closeout_all_lots_supervised_demobilized = constant:infantry_spawn_achievement_value.zero }
			}
		}
		add_to_temp_variable = { infantry_spawn_achievement_lot_scan_index = constant:infantry_spawn_achievement_value.one }
	}

	set_temp_variable = { infantry_spawn_achievement_unit_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_achievement_unit_scan_index value = infantry_spawn_unit_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_unit_generation_uid_entries^infantry_spawn_achievement_unit_scan_index value = infantry_spawn_achievement_closeout_generation_uid compare = equals }
				NOT = { check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_achievement_unit_scan_index value = constant:infantry_spawn_unit_status.demobilized compare = equals } }
			}
			set_temp_variable = { infantry_spawn_achievement_closeout_all_units_demobilized = constant:infantry_spawn_achievement_value.zero }
		}
		add_to_temp_variable = { infantry_spawn_achievement_unit_scan_index = constant:infantry_spawn_achievement_value.one }
	}

	set_temp_variable = { infantry_spawn_achievement_obligation_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_achievement_obligation_scan_index value = infantry_spawn_obligation_uid_entries^num compare = less_than } }
		if = {
			limit = {

		}
		add_to_temp_variable = { infantry_spawn_achievement_obligation_scan_index = constant:infantry_spawn_achievement_value.one }
	}
}

# Country scope. Call inside infantry_spawn_close_resolved_generations directly
# after a row becomes resolved, while infantry_spawn_closeout_generation_uid is
# still the row being closed.

	set_temp_variable = { infantry_spawn_achievement_closeout_generation_uid = infantry_spawn_closeout_generation_uid }

	if = {
		limit = {
			check_variable = { infantry_spawn_achievement_closeout_generation_row > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_achievement_closeout_generation_unit_count value = constant:infantry_spawn_achievement_threshold.very_large_generation_units compare = greater_than_or_equals }
			check_variable = { infantry_spawn_achievement_closeout_has_unresolved_lot = constant:infantry_spawn_achievement_value.zero }

			check_variable = { var = infantry_spawn_muster_control value = constant:infantry_spawn_achievement_threshold.high_control compare = greater_than_or_equals }
			check_variable = { var = infantry_spawn_army_congestion value = constant:infantry_spawn_achievement_threshold.low_congestion_exclusive compare = less_than }
			NOT = { has_country_flag = infantry_spawn_claimant_takeover_complete }
			NOT = { has_country_flag = infantry_spawn_achievement_revolt_history }
			NOT = { has_country_flag = infantry_spawn_achievement_forced_setup_used }
			NOT = { has_country_flag = infantry_spawn_achievement_debug_completion_used }
		}
		set_country_flag = infantry_spawn_achievement_every_rifle_accounted_for_ready
	}

		limit = {
			check_variable = { infantry_spawn_achievement_closeout_generation_row > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_achievement_closeout_generation_unit_count value = constant:infantry_spawn_achievement_threshold.large_generation_units compare = greater_than_or_equals }
			check_variable = { var = infantry_spawn_achievement_closeout_integrated_unit_count value = constant:infantry_spawn_achievement_threshold.train_integrated_units compare = greater_than_or_equals }

			has_country_flag = infantry_spawn_achievement_rail_attempt_active
			has_country_flag = infantry_spawn_achievement_rail_corridor_verified
			NOT = { has_country_flag = infantry_spawn_achievement_rail_corridor_failed }
			NOT = { has_country_flag = infantry_spawn_achievement_emergency_integration_used }
			infantry_spawn_achievement_rail_connection_is_live = yes
		}
		set_country_flag = infantry_spawn_achievement_no_room_on_the_train_ready
		clr_country_flag = infantry_spawn_achievement_rail_attempt_active
	}

	if = {
		limit = {
			check_variable = { infantry_spawn_achievement_closeout_generation_row > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_achievement_closeout_generation_unit_count value = constant:infantry_spawn_achievement_threshold.very_large_generation_units compare = greater_than_or_equals }
			check_variable = { infantry_spawn_achievement_closeout_all_lots_supervised_demobilized = constant:infantry_spawn_achievement_value.one }
			check_variable = { infantry_spawn_achievement_closeout_all_units_demobilized = constant:infantry_spawn_achievement_value.one }
			check_variable = { infantry_spawn_achievement_closeout_has_unresolved_obligation = constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = infantry_spawn_muster_control value = constant:infantry_spawn_achievement_threshold.high_control compare = greater_than_or_equals }
			NOT = { has_country_flag = infantry_spawn_achievement_revolt_history }
			NOT = { has_country_flag = infantry_spawn_claimant_takeover_complete }
			NOT = { has_country_flag = infantry_spawn_achievement_emergency_integration_used }
			NOT = { has_country_flag = infantry_spawn_achievement_teardown_invariant_failure }
			NOT = { has_country_flag = infantry_spawn_achievement_forced_setup_used }
			NOT = { has_country_flag = infantry_spawn_achievement_debug_completion_used }

	}

# Public Event 019 country-pulse hook. It never iterates over other countries.

	if = {
		limit = {
			has_country_flag = infantry_spawn_achievement_rail_attempt_active
			NOT = { infantry_spawn_achievement_rail_connection_is_live = yes }
		}
		set_country_flag = infantry_spawn_achievement_rail_corridor_failed
	}
	if = {
		limit = { has_country_flag = infantry_spawn_achievement_scenario_launched }
		if = {
			limit = { is_ai = yes }
			set_country_flag = infantry_spawn_achievement_scenario_tag_switched
			set_country_flag = infantry_spawn_achievement_scenario_tracking_complete
		}
		if = {
			limit = { has_global_flag = world_end }
			set_country_flag = infantry_spawn_achievement_scenario_world_end_seen
			set_country_flag = infantry_spawn_achievement_scenario_tracking_complete
		}
		if = {
			limit = {
				has_global_flag = infantry_spawn_triggerable_scenario_active
				OR = {
					NOT = { check_variable = { var = global.infantry_spawn_scenario_type value = infantry_spawn_achievement_scenario_type compare = equals } }
					NOT = { check_variable = { var = global.infantry_spawn_scenario_intensity value = infantry_spawn_achievement_scenario_intensity compare = equals } }

			}
			set_country_flag = infantry_spawn_achievement_scenario_intensity_changed
			set_country_flag = infantry_spawn_achievement_scenario_tracking_complete
		}
		if = {
			limit = { NOT = { has_country_flag = infantry_spawn_achievement_scenario_survival_pending } }
			set_country_flag = infantry_spawn_achievement_scenario_tracking_complete
		}
	}

# Input: event_target:infantry_spawn_achievement_battle_division.
# Outputs the exact ledger row, original signature fields, and a validity bit.
infantry_spawn_achievement_prepare_exact_battle_division = {
	set_temp_variable = { infantry_spawn_achievement_exact_division_is_valid = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_exact_unit_row = constant:infantry_spawn_event.invalid_index }
	if = {
		limit = {
			has_event_target = infantry_spawn_achievement_battle_division
			event_target:infantry_spawn_achievement_battle_division = {
				has_variable = infantry_spawn_event19_generated_identity
				has_variable = infantry_spawn_event19_unit_uid
				has_variable = infantry_spawn_event19_generation_uid
				has_variable = infantry_spawn_event19_lot_uid
				has_variable = infantry_spawn_event19_template_uid
				check_variable = { var = infantry_spawn_event19_generated_identity value = constant:infantry_spawn_achievement_value.one compare = equals }
			}
		}
		set_temp_variable = { infantry_spawn_achievement_exact_unit_uid = event_target:infantry_spawn_achievement_battle_division.infantry_spawn_event19_unit_uid }
		set_temp_variable = { infantry_spawn_achievement_exact_generation_uid = event_target:infantry_spawn_achievement_battle_division.infantry_spawn_event19_generation_uid }
		set_temp_variable = { infantry_spawn_achievement_exact_lot_uid = event_target:infantry_spawn_achievement_battle_division.infantry_spawn_event19_lot_uid }
		set_temp_variable = { infantry_spawn_achievement_exact_template_uid = event_target:infantry_spawn_achievement_battle_division.infantry_spawn_event19_template_uid }
		set_temp_variable = { infantry_spawn_achievement_exact_lot_profile = event_target:infantry_spawn_achievement_battle_division.infantry_spawn_event19_lot_profile }
		set_temp_variable = { infantry_spawn_achievement_exact_combat_component_count = event_target:infantry_spawn_achievement_battle_division.infantry_spawn_event19_combat_component_count }
		set_temp_variable = { infantry_spawn_lookup_unit_uid = infantry_spawn_achievement_exact_unit_uid }
		infantry_spawn_find_unit_row = yes
		set_temp_variable = { infantry_spawn_achievement_exact_unit_row = infantry_spawn_unit_row_index }
		if = {
			limit = {
				check_variable = { infantry_spawn_achievement_exact_unit_row > constant:infantry_spawn_event.invalid_index }
				check_variable = { var = infantry_spawn_unit_generation_uid_entries^infantry_spawn_achievement_exact_unit_row value = infantry_spawn_achievement_exact_generation_uid compare = equals }
				check_variable = { var = infantry_spawn_unit_lot_uid_entries^infantry_spawn_achievement_exact_unit_row value = infantry_spawn_achievement_exact_lot_uid compare = equals }
				check_variable = { var = infantry_spawn_unit_template_uid_entries^infantry_spawn_achievement_exact_unit_row value = infantry_spawn_achievement_exact_template_uid compare = equals }
				check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_achievement_exact_unit_row value = constant:infantry_spawn_unit_status.active compare = equals }
				is_in_array = { array = infantry_spawn_unit_scope_entries value = event_target:infantry_spawn_achievement_battle_division }
				NOT = { is_in_array = { array = infantry_spawn_achievement_composition_disqualified_unit_uid_entries value = infantry_spawn_achievement_exact_unit_uid } }
			}
			set_temp_variable = { infantry_spawn_achievement_exact_division_is_valid = constant:infantry_spawn_achievement_value.one }
		}
	}
}

# Uses the immutable component ledger and verifies that every generated token
# still exists in the exact live division's current template. It detects
# removals and sanctioned replacement hooks; the remaining engine limitation
# around switching to a different superset template is documented in handoff.
infantry_spawn_achievement_scan_exact_division_signature = {
	clear_temp_array = infantry_spawn_achievement_exact_distinct_combat_tokens
	set_temp_variable = { infantry_spawn_achievement_exact_signature_is_present = constant:infantry_spawn_achievement_value.one }
	set_temp_variable = { infantry_spawn_achievement_babel_has_camelry = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_babel_has_bicycle = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_babel_has_amphibious_tank = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_babel_has_flame_element = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_babel_has_artillery = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_babel_has_engineers = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_component_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_achievement_component_scan_index value = infantry_spawn_template_component_template_uid_entries^num compare = less_than } }
		if = {
			limit = { check_variable = { var = infantry_spawn_template_component_template_uid_entries^infantry_spawn_achievement_component_scan_index value = infantry_spawn_achievement_exact_template_uid compare = equals } }
			set_temp_variable = { infantry_spawn_achievement_current_component_token_id = infantry_spawn_template_component_unit_profile_entries^infantry_spawn_achievement_component_scan_index }
			if = {
				limit = {
					check_variable = { var = infantry_spawn_template_component_slot_kind_entries^infantry_spawn_achievement_component_scan_index value = constant:infantry_spawn_unit_registry_slot_kind.combat compare = equals }
					NOT = { is_in_array = { array = infantry_spawn_achievement_exact_distinct_combat_tokens value = infantry_spawn_achievement_current_component_token_id } }
				}
				add_to_temp_array = { array = infantry_spawn_achievement_exact_distinct_combat_tokens value = infantry_spawn_achievement_current_component_token_id }
			}

			set_temp_variable = { infantry_spawn_unit_registry_current_provider_id = infantry_spawn_achievement_current_component_token_id }
			infantry_spawn_unit_registry_dispatch_static_token = yes
			event_target:infantry_spawn_achievement_battle_division = {
				if = {
					limit = {
						NOT = {
							meta_trigger = {
								text = { division_has_battalion_in_template = [UNIT_TOKEN] }
								UNIT_TOKEN = "[?infantry_spawn_unit_registry_current_static_token.GetTokenKey]"
							}
						}
					}
					set_temp_variable = { infantry_spawn_achievement_exact_signature_is_present = constant:infantry_spawn_achievement_value.zero }
				}
			}

			if = {
				limit = { check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.camelry compare = equals } }
				set_temp_variable = { infantry_spawn_achievement_babel_has_camelry = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = { check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.bicycle_battalion compare = equals } }
				set_temp_variable = { infantry_spawn_achievement_babel_has_bicycle = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.amphibious_armor compare = equals }
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.amphibious_light_armor compare = equals }
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.amphibious_medium_armor compare = equals }
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.amphibious_heavy_armor compare = equals }
					}
				}
				set_temp_variable = { infantry_spawn_achievement_babel_has_amphibious_tank = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.light_flame_tank compare = equals }
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.medium_flame_tank compare = equals }
						check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.heavy_flame_tank compare = equals }
					}
				}
				set_temp_variable = { infantry_spawn_achievement_babel_has_flame_element = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = { check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.artillery_brigade compare = equals } }
				set_temp_variable = { infantry_spawn_achievement_babel_has_artillery = constant:infantry_spawn_achievement_value.one }
			}
			if = {
				limit = { check_variable = { var = infantry_spawn_achievement_current_component_token_id value = constant:infantry_spawn_unit_registry_token_id.engineer compare = equals } }
				set_temp_variable = { infantry_spawn_achievement_babel_has_engineers = constant:infantry_spawn_achievement_value.one }
			}
		}
		add_to_temp_variable = { infantry_spawn_achievement_component_scan_index = constant:infantry_spawn_achievement_value.one }
	}
	set_temp_variable = { infantry_spawn_achievement_exact_distinct_component_count = infantry_spawn_achievement_exact_distinct_combat_tokens^num }
}

infantry_spawn_achievement_check_exact_unit_pretechnology_gate = {
	set_temp_variable = { infantry_spawn_achievement_exact_unit_has_locked_gate = constant:infantry_spawn_achievement_value.zero }
	set_temp_variable = { infantry_spawn_achievement_gate_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_achievement_exact_unit_has_locked_gate = constant:infantry_spawn_achievement_value.zero }
			check_variable = { var = infantry_spawn_achievement_gate_scan_index value = infantry_spawn_achievement_pretech_unit_uid_entries^num compare = less_than }

			limit = { check_variable = { var = infantry_spawn_achievement_pretech_unit_uid_entries^infantry_spawn_achievement_gate_scan_index value = infantry_spawn_achievement_exact_unit_uid compare = equals } }

				set_temp_variable = { infantry_spawn_achievement_exact_unit_has_locked_gate = constant:infantry_spawn_achievement_value.one }
			}
		}
		add_to_temp_variable = { infantry_spawn_achievement_gate_scan_index = constant:infantry_spawn_achievement_value.one }
	}
}
