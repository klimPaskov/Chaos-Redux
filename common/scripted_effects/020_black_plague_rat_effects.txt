# ============================================================================
# CHAOS REDUX - EVENT 020 RAT NATION AND RAT KING EFFECTS
# ============================================================================
# Event-owned two-country runtime for Evolutions I-IV. The state scheduler
# calls black_plague_rat_run_runtime_pulse once per disease pulse; all country
# work is then limited to the maintained rat-country array. No world on_action
# or duplicate disease category is introduced here.

black_plague_rat_set_initial_evolution_ready_day = {
	if = {
		limit = { NOT = { has_variable = global.black_plague_evolution_i_ready_day } }
		set_variable = { global.black_plague_evolution_i_ready_day = global.date }
		add_to_variable = { global.black_plague_evolution_i_ready_day = constant:black_plague_rat_evolution.establishment_days }
	}
}

black_plague_rat_schedule_next_evolution_check = {
	set_temp_variable = { black_plague_rat_evolution_delay_days = constant:black_plague_rat_evolution.check_retry_days }
	if = {
		limit = { NOT = { has_global_flag = black_plague_evolution_i_active } }
		set_temp_variable = { black_plague_rat_evolution_delay_days = mtth:black_plague_rat_evolution_i_interval }
	}
	else_if = {
		limit = { NOT = { has_global_flag = black_plague_evolution_ii_active } }
		set_temp_variable = { black_plague_rat_evolution_delay_days = mtth:black_plague_rat_evolution_ii_interval }
	}
	else_if = {
		limit = { NOT = { has_global_flag = black_plague_evolution_iii_active } }
		set_temp_variable = { black_plague_rat_evolution_delay_days = mtth:black_plague_rat_evolution_iii_interval }
	}
	else_if = {
		limit = { NOT = { has_global_flag = black_plague_evolution_iv_active } }
		set_temp_variable = { black_plague_rat_evolution_delay_days = mtth:black_plague_rat_evolution_iv_interval }
	}
	clamp_temp_variable = {
		var = black_plague_rat_evolution_delay_days
		min = constant:black_plague_rat_evolution.minimum_interval_days
		max = constant:black_plague_rat_evolution.maximum_interval_days
	}
	set_variable = { global.black_plague_rat_next_evolution_check_date = global.date }
	add_to_variable = { global.black_plague_rat_next_evolution_check_date = black_plague_rat_evolution_delay_days }
}

black_plague_rat_schedule_next_growth_pulse = {
	set_variable = { global.black_plague_rat_next_growth_date = global.date }
	add_to_variable = { global.black_plague_rat_next_growth_date = constant:black_plague_rat_pool.rat_pulse_days }
}

# COUNTRY scope with FROM = selected enemy STATE. The royal operation spends
# only rat-specific reserves and a civilian factory slot; it never recruits
# human manpower or consumes ordinary equipment. The canonical exposure effect
# applies the damage so deaths, phases, spread, provenance, and mapmode state
# remain shared with every other Black Plague route.
black_plague_rat_king_send_royal_strike = {
	if = {
		limit = {
			black_plague_rat_country_is_king = yes
			FROM = { black_plague_rat_king_royal_strike_target_is_valid = yes }
			check_variable = { var = black_plague_rat_king_dominion value = constant:black_plague_rat_decision.royal_strike_dominion_cost compare = greater_than_or_equals }
			check_variable = { var = black_plague_rat_brood_mass value = constant:black_plague_rat_decision.royal_strike_brood_mass_cost compare = greater_than_or_equals }
		}
		subtract_from_variable = { black_plague_rat_king_dominion = constant:black_plague_rat_decision.royal_strike_dominion_cost }
		subtract_from_variable = { black_plague_rat_brood_mass = constant:black_plague_rat_decision.royal_strike_brood_mass_cost }
		set_temp_variable = { black_plague_rat_royal_strike_cooldown = constant:black_plague_rat_decision.royal_strike_cooldown_days }
		set_country_flag = { flag = black_plague_rat_king_royal_strike_cooldown days = black_plague_rat_royal_strike_cooldown }
		FROM = {
			set_temp_variable = { black_plague_exposure_amount = constant:black_plague_spread.rat_occupation_base }
			set_temp_variable = { black_plague_exposure_route = constant:black_plague_route.rat_occupation }
			set_temp_variable = { black_plague_exposure_provenance = constant:black_plague_provenance.rat_occupation }
			set_temp_variable = { black_plague_state_strike_cooldown_days = constant:black_plague_rat_decision.royal_strike_cooldown_days }
			black_plague_apply_exposure = yes
			set_state_flag = { flag = black_plague_rat_king_royal_strike_cooldown days = black_plague_state_strike_cooldown_days }
			save_event_target_as = black_plague_royal_strike_state
			CONTROLLER = { country_event = { id = chaosx.nr20.76 } }
		}
		set_global_flag = black_plague_mapmode_dirty
	}
}

black_plague_rat_run_runtime_pulse = {
	black_plague_rat_initialize_runtime = yes
	if = {
		limit = { has_global_flag = black_plague_rat_king_active }
		add_to_variable = { global.black_plague_rat_king_duration_days = constant:black_plague_timing.pulse_days }
	}
	if = {
		limit = {
			has_global_flag = black_plague_evolution_iii_active
			check_variable = {
				var = global.date
				value = global.black_plague_rat_next_growth_date
				compare = greater_than_or_equals
			}
		}
		black_plague_rat_run_country_pulses = yes
		black_plague_rat_schedule_next_growth_pulse = yes
	}
	if = {
		limit = { has_global_flag = black_plague_evolution_iii_active }
		black_plague_rat_run_merger_pulses = yes
	}
	if = {
		limit = { has_global_flag = black_plague_rat_king_active }
		black_plague_rat_king_absorb_one_adjacent_brood = yes
	}
	if = {
		limit = { has_global_flag = black_plague_rat_king_active }
		# Capture transient territorial peaks on every rat pulse, not only on
		# the slower evolution-check cadence used for ordinary progression.
		black_plague_evolution_refresh_king_progress = yes
	}
	if = {
		limit = {
			has_global_flag = black_plague_rat_king_active
			RTX = {
				exists = yes
				OR = {
					NOT = { has_variable = black_plague_rat_king_pulse_block_until }
					check_variable = {
						var = global.date
						value = black_plague_rat_king_pulse_block_until
						compare = greater_than_or_equals
					}
				}
			}
		}
		RTX = {
			if = {
				limit = { exists = yes has_country_flag = black_plague_rat_king_country }
				if = {
					limit = { NOT = { black_plague_rat_country_is_plague_immune = yes } }
					set_country_flag = black_plague_rat_disease_immune
					set_country_flag = black_plague_rat_plague_immune
				}
				add_to_variable = { black_plague_rat_king_dominion = constant:black_plague_rat_pool.king_dominion_pulse_gain }
				add_to_variable = { black_plague_rat_king_sentience = constant:black_plague_rat_pool.king_sentience_pulse_gain }
				add_to_variable = { black_plague_rat_king_cohesion = constant:black_plague_rat_pool.king_cohesion_pulse_gain }
				add_to_variable = { black_plague_rat_king_hunger = constant:black_plague_rat_pool.king_hunger_pulse_gain }
				if = {
					limit = { has_country_flag = black_plague_rat_crown_dominion }
					add_to_variable = { black_plague_rat_king_dominion = constant:black_plague_rat_policy.crown_pulse_dominion_gain }
				}
				if = {
					limit = { has_country_flag = black_plague_rat_council_transparent_hunger }
					add_to_variable = { black_plague_rat_king_cohesion = constant:black_plague_rat_policy.council_pulse_cohesion_gain }
				}
				clamp_variable = { var = black_plague_rat_king_dominion min = constant:black_plague_value.minimum max = constant:black_plague_value.maximum }
				clamp_variable = { var = black_plague_rat_king_sentience min = constant:black_plague_value.minimum max = constant:black_plague_value.maximum }
				clamp_variable = { var = black_plague_rat_king_cohesion min = constant:black_plague_value.minimum max = constant:black_plague_value.maximum }
				clamp_variable = { var = black_plague_rat_king_hunger min = constant:black_plague_value.minimum max = constant:black_plague_value.maximum }
				if = {
					limit = {
						check_variable = {
							var = black_plague_rat_king_hunger
							value = constant:black_plague_rat_pool.king_hunger_crisis_threshold
							compare = greater_than_or_equals
						}
						check_variable = {
							var = black_plague_rat_king_cohesion
							value = constant:black_plague_rat_pool.king_cohesion_crisis_floor
							compare = less_than
						}
						NOT = { has_country_flag = black_plague_rat_hunger_crisis }
					}
					set_country_flag = black_plague_rat_hunger_crisis
					country_event = { id = chaosx.nr20.53 }
					if = {
						limit = {
							has_country_flag = black_plague_rat_crown_dominion
							NOT = { has_country_flag = black_plague_rat_crown_crisis_fired }
						}
						set_country_flag = black_plague_rat_crown_crisis_fired
						country_event = { id = chaosx.nr20.57 }
					}
					if = {
						limit = {
							has_country_flag = black_plague_rat_council_transparent_hunger
							NOT = { has_country_flag = black_plague_rat_council_crisis_fired }
						}
						set_country_flag = black_plague_rat_council_crisis_fired
						country_event = { id = chaosx.nr20.58 }
					}
					if = {
						limit = {
							has_country_flag = black_plague_rat_hierophant_final_omen
							NOT = { has_country_flag = black_plague_rat_hierophancy_crisis_fired }
						}
						set_country_flag = black_plague_rat_hierophancy_crisis_fired
						country_event = { id = chaosx.nr20.59 }
					}
				}
				if = {
					limit = {
						has_country_flag = black_plague_rat_hunger_crisis
						check_variable = {
							var = black_plague_rat_king_hunger
							value = constant:black_plague_rat_pool.king_hunger_crisis_release
							compare = less_than
						}
					}
					clr_country_flag = black_plague_rat_hunger_crisis
				}
			}
		}
	}
	if = {
		limit = {
			has_global_flag = black_plague_rat_king_active
			NOT = { has_global_flag = black_plague_terminal_takeover_active }
			RTX = {
				exists = yes
				num_of_controlled_states < 1
				NOT = { has_country_flag = black_plague_rat_king_defeated }
			}
		}
		black_plague_rat_king_resolve_defeat = yes
	}
}

# STATE scope from on_state_control_changed. An eligible non-Rat-Controlled
# state captured by either rat carrier becomes an established Rat-Controlled infection immediately;
# the shared exposure adapter owns phase, provenance, mapmode, and ledger
# updates, while this hook adds one exact-loss mortality transaction and the
# reusable RTA brood marker without creating another country tag.
black_plague_rat_infect_controlled_state = {
	if = {
		limit = {
			CONTROLLER = { black_plague_rat_country_is_active = yes }
			NOT = { black_plague_state_is_rat_controlled = yes }
			black_plague_state_can_receive_exposure = yes
		}
		set_temp_variable = { black_plague_exposure_amount = constant:black_plague_spread.rat_occupation_base }
		set_temp_variable = { black_plague_exposure_route = constant:black_plague_route.rat_occupation }
		set_temp_variable = { black_plague_exposure_provenance = constant:black_plague_provenance.rat_occupation }
		black_plague_apply_exposure = yes
		black_plague_apply_current_state_mortality_once = yes
		black_plague_apply_current_state_devastation = yes
		if = {
			limit = { CONTROLLER = { black_plague_rat_country_is_active = yes NOT = { has_country_flag = black_plague_rat_king_country } } }
			set_state_flag = black_plague_rat_nation_state
			black_plague_rat_initialize_state_brood_marker = yes
		}
		if = {
			limit = { NOT = { is_in_array = { array = global.black_plague_rat_controlled_states value = THIS } } }
			add_to_array = { array = global.black_plague_rat_controlled_states value = THIS }
			add_to_variable = { global.black_plague_rat_controlled_states_count = constant:black_plague_value.one }
		}
		set_global_flag = black_plague_mapmode_dirty
	}
}

# COUNTRY scope. Track distinct major human governments that reclaim royal
# territory. The array is intentionally finite in practice and is reset when a
# new King is crowned, so the defeat aftermath cannot inherit an old campaign.
black_plague_rat_record_defeat_participant = {
	if = {
		limit = {
			exists = yes
			black_plague_country_is_human_host = yes
			is_major = yes
		}
		if = {
			limit = { NOT = { is_in_array = { array = global.black_plague_rat_king_defeat_participants value = THIS } } }
			add_to_array = { array = global.black_plague_rat_king_defeat_participants value = THIS }
			add_to_variable = { global.black_plague_rat_king_defeat_participant_count = constant:black_plague_value.one }
		}
		save_global_event_target_as = black_plague_rat_king_defeat_actor
	}
}

black_plague_emit_rat_king_defeat_super_event = {
	if = {
		limit = { black_plague_rat_king_defeat_aftermath_is_eligible = yes }
		set_global_flag = black_plague_rat_king_defeat_aftermath_emitted
		set_temp_variable = { black_plague_rat_defeat_super_event_days = constant:black_plague_rat_defeat_aftermath.visible_days }
		set_global_flag = {
			flag = super_event_visible
			days = black_plague_rat_defeat_super_event_days
			value = constant:black_plague_identity.global_defeat_super_event_id
		}
		set_variable = { global.current_super_event_audio_id = constant:black_plague_identity.global_defeat_audio_id }
		every_country = {
			limit = { is_ai = no }
			play_current_super_event_audio = yes
		}
	}
}

# The pulse is the final reconciliation hook for the sovereign. Other scoped
# hooks can call this same idempotent effect later without reviving RTX or
# clearing surviving RTA broods and disease states.
black_plague_rat_king_resolve_defeat = {
	if = {
		limit = { NOT = { has_global_flag = black_plague_rat_king_defeat_resolved } }
		set_global_flag = black_plague_rat_king_defeat_resolved
		set_global_flag = black_plague_rat_king_defeated
		clr_global_flag = black_plague_rat_king_active
		clr_global_flag = black_plague_rat_king_grace_period
		clr_global_flag = black_plague_rat_king_route_completed
		clr_global_flag = black_plague_rat_king_core_route_completed
		clr_global_flag = black_plague_evolution_v_route_open
		set_variable = { global.black_plague_rat_king_terminal_preparation = constant:black_plague_value.zero }
		RTX = {
			set_country_flag = black_plague_rat_king_defeated
			set_country_flag = black_plague_rat_king_country_retired
			clr_country_flag = black_plague_rat_king_country
			clr_country_flag = black_plague_rat_king_active
			clr_country_flag = black_plague_rat_king_source_pending
			clr_country_flag = black_plague_rat_no_normal_recruitment
			clr_country_flag = black_plague_rat_crown_tithe_operation
			clr_country_flag = black_plague_rat_council_audit_operation
			clr_country_flag = black_plague_rat_hierophant_broadcast_operation
			clr_country_flag = black_plague_rat_absolute_crown
			clr_country_flag = black_plague_rat_absolute_crown_route
			clr_country_flag = black_plague_rat_brood_council
			clr_country_flag = black_plague_rat_brood_council_route
			clr_country_flag = black_plague_rat_breath_hierophancy
			clr_country_flag = black_plague_rat_breath_hierophancy_route
			clr_country_flag = black_plague_rat_crown_dominion
			clr_country_flag = black_plague_rat_crown_final_edict
			clr_country_flag = black_plague_rat_crown_warden_protocol
			clr_country_flag = black_plague_rat_council_transparent_hunger
			clr_country_flag = black_plague_rat_hierophant_final_omen
			clr_country_flag = black_plague_rat_hierophant_terminal_omen
			clr_country_flag = black_plague_rat_royal_template_lock
			clr_country_flag = black_plague_rat_terminal_armature
			clr_country_flag = black_plague_rat_terminal_capital_network
			clr_country_flag = black_plague_rat_terminal_refuge_network
			clr_country_flag = black_plague_rat_king_route_completed
			clear_variable = black_plague_rat_king_route
			clear_variable = black_plague_rat_king_dominion
			clear_variable = black_plague_rat_king_sentience
			clear_variable = black_plague_rat_king_cohesion
			clear_variable = black_plague_rat_king_hunger
			clr_country_flag = black_plague_rat_disease_immune
			clr_country_flag = black_plague_rat_plague_immune
			remove_ideas = { black_plague_rat_king_dominion black_plague_rat_no_civilian_economy }
			delete_unit_template_and_units = { division_template = "Rat Brood" disband = no }
			delete_unit_template_and_units = { division_template = "Rat Shock Brood" disband = no }
			delete_unit_template_and_units = { division_template = "Royal Rat Guard" disband = no }
			delete_unit_template_and_units = { division_template = "Rat Burrow Column" disband = no }
			delete_unit_template_and_units = { division_template = "Rat Carrion Guard" disband = no }

		if = {
			limit = { black_plague_rat_state_has_connected_basin_signal = yes black_plague_rat_state_can_spawn = yes }
			add_to_temp_array = { black_plague_rat_spawn_pool = THIS }
		}
	}

		limit = { check_variable = { black_plague_rat_spawn_pool^num > constant:black_plague_value.zero } }
		random_scope_in_array = {
			array = black_plague_rat_spawn_pool

}

# COUNTRY scope. The reusable RTA carrier exposes Brood Mass, Hunger,
# Coherence, and a derived Disease Dominion register. The meters are updated
# only during the existing capped pulse, so the two-tag runtime remains bounded
# and the King keeps its separate Dominion, Sentience, Cohesion, and Hunger
# system.
black_plague_rat_refresh_swarm_meters = {
	if = {
		limit = {
			black_plague_rat_country_is_active = yes
			NOT = { has_country_flag = black_plague_rat_king_country }
		}
		if = {
			limit = { NOT = { has_variable = black_plague_rat_hunger } }
			set_variable = { black_plague_rat_hunger = constant:black_plague_rat_pool.hunger_initial }
		}
		if = {
			limit = { NOT = { has_variable = black_plague_rat_coherence } }
			set_variable = { black_plague_rat_coherence = constant:black_plague_rat_pool.coherence_initial }
		}
		if = {
			limit = { NOT = { has_variable = black_plague_rat_disease_dominion } }
			set_variable = { black_plague_rat_disease_dominion = constant:black_plague_value.zero }
		}

		# Disease Dominion is derived from controlled plague territory and the
		# permanent immune-blood route. It is a display and pacing register, not
		# a second manually purchased resource.
		set_variable = { black_plague_rat_dominion_states = num_of_controlled_states }
		set_variable = { black_plague_rat_disease_dominion = black_plague_rat_dominion_states }
		multiply_variable = { black_plague_rat_disease_dominion = constant:black_plague_rat_pool.disease_dominion_per_controlled_state }
		if = {
			limit = { has_country_flag = black_plague_rat_immune_blood_hardened }
			add_to_variable = { black_plague_rat_disease_dominion = constant:black_plague_rat_pool.disease_dominion_hardened_bonus }
		}
		clamp_variable = {
			var = black_plague_rat_disease_dominion
			min = constant:black_plague_value.minimum
			max = constant:black_plague_value.maximum
		}

		# Hunger rises once per pulse. Rat-controlled territory relieves it,
		# while hierarchy and stable territory improve Coherence.
		add_to_variable = { black_plague_rat_hunger = constant:black_plague_rat_pool.hunger_pulse_gain }
		set_temp_variable = { black_plague_rat_hunger_relief = num_of_controlled_states }
		multiply_temp_variable = { black_plague_rat_hunger_relief = constant:black_plague_rat_pool.hunger_relief_per_controlled_state }
		subtract_from_variable = { black_plague_rat_hunger = black_plague_rat_hunger_relief }
		set_temp_variable = { black_plague_rat_coherence_gain = num_of_controlled_states }
		multiply_temp_variable = { black_plague_rat_coherence_gain = constant:black_plague_rat_pool.coherence_state_gain }
		add_to_variable = { black_plague_rat_coherence = black_plague_rat_coherence_gain }
		if = {
			limit = {
				OR = {
					black_plague_rat_hierarchy_is_distributed = yes
					black_plague_rat_hierarchy_is_dominant = yes
					black_plague_rat_hierarchy_is_emergent = yes
				}
			}
			add_to_variable = { black_plague_rat_coherence = constant:black_plague_rat_pool.coherence_hierarchy_gain }
		}

		# Hunger trades directly against coordination. A settled, high-coherence
		# brood receives a small additional stabilisation pulse.
		if = {
			limit = { black_plague_rat_hunger_is_crisis = yes }
			subtract_from_variable = { black_plague_rat_coherence = constant:black_plague_rat_pool.coherence_hunger_penalty }
		}
		if = {
			limit = {
				black_plague_rat_coherence_is_high = yes
				black_plague_rat_hunger_is_relieved = yes
			}
			add_to_variable = { black_plague_rat_coherence = constant:black_plague_rat_pool.coherence_state_gain }
		}
		clamp_variable = { var = black_plague_rat_hunger min = constant:black_plague_value.minimum max = constant:black_plague_value.maximum }
		clamp_variable = { var = black_plague_rat_coherence min = constant:black_plague_value.minimum max = constant:black_plague_value.maximum }

		if = {
			limit = {
				black_plague_rat_hunger_is_crisis = yes
				NOT = { has_country_flag = black_plague_rat_hunger_crisis }
			}
			set_country_flag = black_plague_rat_hunger_crisis
			country_event = { id = chaosx.nr20.46 }
		}
		if = {
			limit = {
				has_country_flag = black_plague_rat_hunger_crisis
				black_plague_rat_hunger_is_relieved = yes
			}
			clr_country_flag = black_plague_rat_hunger_crisis
		}
	}
}

black_plague_rat_try_absorb_adjacent_brood = {
	# RTA is the sole carrier tag. A state-level brood marker keeps internal
	# broods distinct, so the stronger adjacent marker can absorb one weaker
	# marker and inherit its surviving brood units without inventing another tag.
	set_variable = { black_plague_rat_merger_report_result = constant:black_plague_value.zero }
	if = {
		limit = {
			black_plague_rat_can_absorb_adjacent_brood = yes
		}
		every_owned_state = {
			limit = {
				black_plague_state_is_rat_controlled = yes
				NOT = { has_state_flag = black_plague_rat_brood_absorbed }
				check_variable = { var = ROOT.black_plague_rat_merger_report_result value = constant:black_plague_value.zero compare = equals }
			}
			black_plague_rat_initialize_state_brood_marker = yes
			every_neighbor_state = {
				limit = {
					black_plague_rat_state_is_valid_merger_state = yes
					ROOT = {
						check_variable = {
							var = black_plague_rat_merger_report_result
							value = constant:black_plague_value.zero
							compare = equals
						}
					}
					check_variable = {
						var = black_plague_rat_brood_strength
						value = PREV.black_plague_rat_brood_strength
						compare = less_than
					}
					ROOT = {
						check_variable = {
							var = black_plague_rat_brood_mass
							value = constant:black_plague_rat_decision.merger_mass_cost
							compare = greater_than_or_equals
						}
					}
				}
			black_plague_rat_initialize_state_brood_marker = yes
			set_state_flag = black_plague_rat_merger_reserved
			set_state_flag = black_plague_rat_brood_absorbed
			set_variable = { black_plague_rat_brood_strength = PREV.black_plague_rat_brood_strength }
			set_temp_variable = { black_plague_rat_inherited_brood_units = black_plague_rat_brood_units }
			if = {
				limit = { ROOT = { black_plague_rat_hierarchy_is_dominant = yes } }
				add_to_temp_variable = { black_plague_rat_inherited_brood_units = constant:black_plague_rat_hierarchy_route.dominant_marker_extra_units }
			}
			if = {
				limit = { check_variable = { var = black_plague_rat_inherited_brood_units value = constant:black_plague_rat_pool.brood_units_minimum compare = less_than } }
				set_temp_variable = { black_plague_rat_inherited_brood_units = constant:black_plague_rat_pool.brood_units_minimum }
			}
			ROOT = {
				black_plague_rat_refresh_division_cap = yes
				subtract_from_variable = { black_plague_rat_brood_mass = constant:black_plague_rat_decision.merger_mass_cost }
				set_temp_variable = { black_plague_rat_inherited_brood_units = PREV.black_plague_rat_inherited_brood_units }
				black_plague_rat_create_inherited_divisions = yes
				add_to_variable = { black_plague_rat_division_count = PREV.black_plague_rat_inherited_brood_units }
				clamp_variable = {
					var = black_plague_rat_division_count
					min = constant:black_plague_value.zero
					max = black_plague_rat_division_cap
				}
				add_to_variable = { black_plague_rat_brood_mass = constant:black_plague_rat_pool.brood_mass_state_gain }
				set_variable = { black_plague_rat_merger_report_result = constant:black_plague_value.one }
			}
			clr_state_flag = black_plague_rat_merger_reserved
		}
	}
	if = {
		limit = { check_variable = { black_plague_rat_merger_report_result = constant:black_plague_value.one } }
		set_temp_variable = { black_plague_rat_merger_cooldown_days = constant:black_plague_rat_pool.rat_merger_cooldown_days }
		if = {
			limit = { black_plague_rat_hierarchy_is_distributed = yes }
			set_temp_variable = { black_plague_rat_merger_cooldown_days = constant:black_plague_rat_hierarchy_route.distributed_merger_cooldown_days }
		}
		else_if = {
			limit = { black_plague_rat_hierarchy_is_dominant = yes }
			set_temp_variable = { black_plague_rat_merger_cooldown_days = constant:black_plague_rat_hierarchy_route.dominant_merger_cooldown_days }
		}
		set_variable = { black_plague_rat_merger_ready_date = global.date }
		add_to_variable = { black_plague_rat_merger_ready_date = black_plague_rat_merger_cooldown_days }
	}
	if = {
		limit = {
			check_variable = { black_plague_rat_merger_report_result = constant:black_plague_value.one }
			NOT = { has_global_flag = black_plague_report_rat_absorption }
		}
		set_global_flag = black_plague_report_rat_absorption
		news_event = { id = chaosx.nr20.43 }
	}
}

		limit = { has_global_flag = black_plague_rat_king_active NOT = { has_global_flag = black_plague_rat_king_grace_period } }
		RTX = { set_variable = { black_plague_rat_king_absorption_count = constant:black_plague_value.zero } }

black_plague_rat_select_king_source = {
	set_temp_variable = { black_plague_rat_best_king_score = constant:black_plague_value.zero }
	for_each_scope_loop = {
		array = global.black_plague_rat_countries
		if = {
			limit = { black_plague_rat_country_is_active = yes }
			set_temp_variable = { black_plague_rat_king_score = num_of_controlled_states }
			multiply_temp_variable = { black_plague_rat_king_score = constant:black_plague_rat_pool.state_share_for_king }
			add_to_temp_variable = { black_plague_rat_king_score = black_plague_rat_brood_mass }
			if = {
				limit = { has_variable = black_plague_rat_sentience }
				add_to_temp_variable = { black_plague_rat_king_score = black_plague_rat_sentience }
			}
			if = {
				limit = { black_plague_rat_hierarchy_is_distributed = yes }
				subtract_from_temp_variable = { black_plague_rat_king_score = constant:black_plague_rat_hierarchy_route.distributed_king_candidacy_penalty }
			}
			if = {
				limit = { black_plague_rat_hierarchy_is_dominant = yes }
				add_to_temp_variable = { black_plague_rat_king_score = constant:black_plague_rat_hierarchy_route.dominant_king_candidacy_gain }
			}
			if = {
				limit = { black_plague_rat_hierarchy_is_emergent = yes }
				add_to_temp_variable = { black_plague_rat_king_score = constant:black_plague_rat_hierarchy_route.emergent_king_candidacy_gain }
			}
			if = {
				limit = { has_country_flag = black_plague_rat_king_candidate }
				add_to_temp_variable = { black_plague_rat_king_score = constant:black_plague_rat_hierarchy_route.focus_candidate_king_candidacy_gain }
			}
			if = {
				limit = { check_variable = { var = black_plague_rat_king_score value = constant:black_plague_value.zero compare = less_than } }
				set_temp_variable = { black_plague_rat_king_score = constant:black_plague_value.zero }
			}
			if = {
				limit = { check_variable = { var = black_plague_rat_king_score value = black_plague_rat_best_king_score compare = greater_than } }
				set_temp_variable = { black_plague_rat_best_king_score = black_plague_rat_king_score }
				save_event_target_as = black_plague_rat_king_source
			}
		}
	}
}
