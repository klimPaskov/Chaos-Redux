# Selects coherent feeding-state origins, allocates one of eight verified fixed

# country, state, queue, callback, core, and generation references are cleared.

# ---------------------------------------------------------------------------
# Candidate selection and creation context
# ---------------------------------------------------------------------------

cannibalism_select_warlord_candidate_state = {
	set_temp_variable = { cannibalism_warlord_candidate_found = 0 }
	set_temp_variable = { cannibalism_warlord_best_candidate_score = -1 }
	for_each_scope_loop = {
		array = global.cannibalism_node_states
		if = {
			limit = { cannibalism_state_can_emerge_as_warlord = yes }
			set_temp_variable = { cannibalism_warlord_candidate_score = cannibalism_node_strength }
			set_temp_variable = { cannibalism_warlord_candidate_component = state_population_k }
			divide_temp_variable = { cannibalism_warlord_candidate_component = constant:cannibalism_warlord_creation.candidate_population_divisor_k }
			clamp_temp_variable = {
				var = cannibalism_warlord_candidate_component
				min = 0
				max = constant:cannibalism_warlord_creation.candidate_population_score_cap
			}
			add_to_temp_variable = { cannibalism_warlord_candidate_score = cannibalism_warlord_candidate_component }
			if = {
				limit = { cannibalism_state_can_form_island_host = yes }
				add_to_temp_variable = { cannibalism_warlord_candidate_score = constant:cannibalism_warlord_creation.candidate_island_bonus }
			}
			else_if = {
				limit = { cannibalism_state_can_form_siege_commune = yes }
				add_to_temp_variable = { cannibalism_warlord_candidate_score = constant:cannibalism_warlord_creation.candidate_siege_bonus }
			}
			else = {
				add_to_temp_variable = { cannibalism_warlord_candidate_score = constant:cannibalism_warlord_creation.candidate_march_bonus }
			}
			if = {
				limit = {
					check_variable = {
						var = cannibalism_warlord_candidate_score
						value = cannibalism_warlord_best_candidate_score
						compare = greater_than
					}
				}
				set_temp_variable = { cannibalism_warlord_best_candidate_score = cannibalism_warlord_candidate_score }
				save_event_target_as = cannibalism_warlord_origin_state
				set_temp_variable = { cannibalism_warlord_candidate_found = 1 }
			}
		}
	}
}

# State scope. Origins are derived from geography and operational conditions;
# all eight dormant tags are origin-agnostic reusable slots.
cannibalism_prepare_warlord_creation_context = {
	set_temp_variable = { cannibalism_selected_warlord_origin = constant:cannibalism_origin.none }
	if = {
		limit = {
			cannibalism_state_can_form_island_host = yes
		}
		set_temp_variable = { cannibalism_selected_warlord_origin = constant:cannibalism_origin.island_host }
	}
	else_if = {
		limit = {
			cannibalism_state_can_form_siege_commune = yes
		}
		set_temp_variable = { cannibalism_selected_warlord_origin = constant:cannibalism_origin.siege_commune }
	}
	else_if = {
		limit = {
			cannibalism_state_can_form_march_host = yes
		}
		set_temp_variable = { cannibalism_selected_warlord_origin = constant:cannibalism_origin.march_host }
	}

	set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.unknown }
	if = { limit = { is_on_continent = middle_east } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.middle_east } }
	else_if = { limit = { is_on_continent = europe } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.europe } }
	else_if = { limit = { is_on_continent = asia } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.asia } }
	else_if = { limit = { is_on_continent = africa } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.africa } }
	else_if = { limit = { is_on_continent = north_america } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.north_america } }
	else_if = { limit = { is_on_continent = south_america } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.south_america } }
	else_if = { limit = { is_on_continent = australia } set_temp_variable = { cannibalism_selected_warlord_region = constant:cannibalism_region.oceania } }

	set_temp_variable = { cannibalism_warlord_starting_units = constant:cannibalism_warlord_start.weak_units }
	set_temp_variable = { cannibalism_warlord_starting_experience = constant:cannibalism_warlord_start.weak_experience }
	set_temp_variable = { cannibalism_warlord_starting_frenzy = constant:cannibalism_warlord_start.weak_frenzy }
	if = {
		limit = {
			check_variable = {
				var = state_population_k
				value = constant:cannibalism_warlord_start.severe_population_k
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { cannibalism_warlord_starting_units = constant:cannibalism_warlord_start.severe_units }
		set_temp_variable = { cannibalism_warlord_starting_experience = constant:cannibalism_warlord_start.severe_experience }
		set_temp_variable = { cannibalism_warlord_starting_frenzy = constant:cannibalism_warlord_start.severe_frenzy }
	}
	else_if = {
		limit = {
			check_variable = {
				var = state_population_k
				value = constant:cannibalism_warlord_start.normal_population_k
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { cannibalism_warlord_starting_units = constant:cannibalism_warlord_start.normal_units }
		set_temp_variable = { cannibalism_warlord_starting_experience = constant:cannibalism_warlord_start.normal_experience }
		set_temp_variable = { cannibalism_warlord_starting_frenzy = constant:cannibalism_warlord_start.normal_frenzy }
	}
	if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:cannibalism_warlord_start.high_chaos
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { cannibalism_warlord_starting_units = constant:cannibalism_warlord_start.high_chaos_units }
		set_temp_variable = { cannibalism_warlord_starting_experience = constant:cannibalism_warlord_start.high_chaos_experience }
		set_temp_variable = { cannibalism_warlord_starting_frenzy = constant:cannibalism_warlord_start.high_chaos_frenzy }
	}
	clamp_temp_variable = { var = cannibalism_warlord_starting_units min = constant:cannibalism_slot.minimum max = constant:cannibalism_warlord_start.maximum_units }

	save_event_target_as = cannibalism_warlord_origin_state
	OWNER = { save_event_target_as = cannibalism_warlord_former_owner }
	CONTROLLER = { save_event_target_as = cannibalism_warlord_former_controller }
}

cannibalism_prepare_current_warlord_starting_force = {
	set_temp_variable = { cannibalism_warlord_population_unit_capacity = cannibalism_population_loss_applied }
	divide_temp_variable = { cannibalism_warlord_population_unit_capacity = constant:cannibalism_consumption.population_people_per_k }
	divide_temp_variable = { cannibalism_warlord_population_unit_capacity = constant:cannibalism_warlord_creation.population_per_unit_k }
	round_temp_variable = cannibalism_warlord_population_unit_capacity
	clamp_temp_variable = {
		var = cannibalism_warlord_population_unit_capacity
		min = constant:cannibalism_slot.minimum
		max = constant:cannibalism_warlord_start.maximum_units
	}
	if = {
		limit = {
			check_variable = {
				var = cannibalism_warlord_starting_units
				value = cannibalism_warlord_population_unit_capacity
				compare = greater_than
			}
		}
		set_temp_variable = { cannibalism_warlord_starting_units = cannibalism_warlord_population_unit_capacity }
	}
	set_variable = { cannibalism_warlord_starting_unit_count = cannibalism_warlord_starting_units }
	set_variable = { cannibalism_warlord_reinforcement_unit_cap = cannibalism_warlord_starting_units }

	multiply_temp_variable = { cannibalism_warlord_starting_manpower = constant:cannibalism_warlord_creation.recruitment_manpower_factor }

	add_manpower = cannibalism_warlord_starting_manpower

# State scope. The selected origin and at most two connected active-node states
# transfer together; unrelated neighboring territory is never swept into a Host.
cannibalism_create_selected_warlord_country_from_current_state = {
	set_temp_variable = { cannibalism_warlord_creation_result = 0 }
	if = {
		limit = {
			cannibalism_state_can_emerge_as_warlord = yes
			check_variable = { cannibalism_selected_warlord_origin > constant:cannibalism_origin.none }
			check_variable = { cannibalism_warlord_slot_index > constant:cannibalism_slot.none }
		}
		set_temp_variable = { cannibalism_warlord_formation_lock_days = constant:cannibalism_warlord_creation.formation_lock_days }
		set_state_flag = { flag = cannibalism_warlord_formation_pending days = cannibalism_warlord_formation_lock_days }
		set_state_flag = cannibalism_warlord_transfer_selected
		set_temp_variable = { cannibalism_warlord_neighbor_state_count = 0 }
		var:cannibalism_new_warlord_country = {
			transfer_state = PREV
			add_state_core = PREV
		}
		every_neighbor_state = {
			limit = {
				check_variable = { cannibalism_warlord_neighbor_state_count < constant:cannibalism_warlord_creation.maximum_neighbor_states }
				is_controlled_by = event_target:cannibalism_warlord_former_controller
				cannibalism_state_has_active_cell = yes
				NOT = { cannibalism_state_is_unusable_larder = yes }
				NOT = { has_state_flag = cannibalism_recovery_active }
				NOT = { has_state_flag = cannibalism_warlord_transfer_selected }
				OR = {
					NOT = { check_variable = { cannibalism_selected_warlord_origin = constant:cannibalism_origin.island_host } }
					is_island_state = yes
					is_coastal = yes
				}
			}
			set_state_flag = cannibalism_warlord_transfer_selected
			var:cannibalism_new_warlord_country = {
				transfer_state = PREV
				add_state_core = PREV
			}
			add_to_temp_variable = { cannibalism_warlord_neighbor_state_count = 1 }
		}
		var:cannibalism_new_warlord_country = {
			save_event_target_as = cannibalism_warlord_new_country
			set_variable = { cannibalism_warlord_slot_index = cannibalism_warlord_slot_index }
			set_capital = { state = event_target:cannibalism_warlord_origin_state remember_old_capital = no }
			cannibalism_setup_current_warlord_country = yes
		}
		if = {
			limit = { check_variable = { cannibalism_warlord_setup_result > 0 } }
			event_target:cannibalism_warlord_new_country = { cannibalism_declare_current_warlord_origin_wars = yes }
			if = {
				limit = { has_event_target = cannibalism_warlord_former_controller }
				event_target:cannibalism_warlord_former_controller = { country_event = { id = chaosx.nr14.30 } }
			}
			set_temp_variable = { cannibalism_chaos_change = constant:cannibalism_delta.warlord_chaos }
			set_temp_variable = { cannibalism_chaos_reason = constant:chaos_meter_history_reason.special.cannibalism_warlord }
			cannibalism_apply_chaos_change = yes
			set_temp_variable = { cannibalism_warlord_creation_result = 1 }
		}
		else = {
			event_target:cannibalism_warlord_new_country = { cannibalism_rollback_current_warlord_creation = yes }
		}
	}
}

# Country scope. Rows are invalidated through the canonical ID-safe API and
# remain tombstones until the normal reverse compactor runs.

	cannibalism_invalidate_spread_entries_for_current_country_lifecycle = yes

# State scope. A defeated Host's cells enter the full liberation-recovery
# sequence before their slot/core/source references are cleared.
cannibalism_retire_current_state_from_warlord_slot = {
	if = {
		limit = {
			has_variable = cannibalism_node_source_country
			check_variable = {
				var = cannibalism_node_source_country
				value = event_target:cannibalism_warlord_release_country
				compare = equals
			}
		}
		if = {
			limit = {
				has_state_flag = cannibalism_active_node
				has_variable = cannibalism_node_id
				has_variable = cannibalism_node_generation
				is_in_array = { array = global.cannibalism_node_states value = THIS }
			}
			set_temp_variable = { cannibalism_expected_node_id = cannibalism_node_id }
			set_temp_variable = { cannibalism_expected_node_generation = cannibalism_node_generation }
			cannibalism_begin_current_state_recovery = yes
		}
		else = {
			if = {
				limit = { is_in_array = { array = global.cannibalism_node_states value = THIS } }
				remove_from_array = { array = global.cannibalism_node_states value = THIS }
			}
			clear_variable = cannibalism_node_id
			clear_variable = cannibalism_node_generation
			clear_variable = cannibalism_node_type
			clear_variable = cannibalism_node_strength
			clear_variable = cannibalism_node_source_country
			clear_variable = cannibalism_node_source_country_generation
			clr_state_flag = cannibalism_active_node
			if = {
				limit = { cannibalism_state_has_active_cell = yes }
				if = {
					limit = { NOT = { is_in_array = { array = global.cannibalism_recovery_states value = THIS } } }
					add_to_array = { array = global.cannibalism_recovery_states value = THIS }
				}
				set_state_flag = cannibalism_recovery_active
				set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.liberated_emergency }
				cannibalism_set_current_state_stage = yes
				set_variable = { cannibalism_recovery_next_date = global.date }
				add_to_variable = { cannibalism_recovery_next_date = constant:cannibalism_timing.liberated_emergency_days }
			}
		}
	}
	if = {
		limit = { is_core_of = event_target:cannibalism_warlord_release_country }
		remove_state_core = event_target:cannibalism_warlord_release_country
	}
	clr_state_flag = cannibalism_warlord_state
	clr_state_flag = cannibalism_warlord_transfer_selected
	clr_state_flag = cannibalism_warlord_formation_pending
	clr_state_flag = cannibalism_commune_state
	clr_state_flag = cannibalism_network_node
	clear_variable = cannibalism_warlord_slot_index
	clear_variable = cannibalism_warlord_slot_generation
}

# Country scope. Called from the annexed reusable tag after pulse rehoming.
cannibalism_begin_current_warlord_slot_release = {
	cannibalism_validate_spread_queue = yes
	if = {
		limit = {
			check_variable = { cannibalism_spread_queue_validation_result > 0 }
			is_cannibalism_reusable_warlord_slot_country = yes
			has_country_flag = cannibalism_warlord_slot_in_use
		}
		save_event_target_as = cannibalism_warlord_release_country
		set_country_flag = cannibalism_warlord_slot_release_pending
		clr_country_flag = cannibalism_warlord_integration_release_pending
		clr_country_flag = cannibalism_warlord_decisions_open
		clr_country_flag = cannibalism_containment_decisions_open
		clr_country_flag = cannibalism_pending_absorption
		clr_country_flag = cannibalism_foreign_seeding_active
		cannibalism_warlord_focus_reset_contracts = yes
		cannibalism_remove_current_country_from_spread_ledger = yes
		cannibalism_reset_current_country_incarnation_state = yes
		remove_ideas = {
			cannibalism_closed_muster_rolls
			cannibalism_warlord_survival_burdens
			cannibalism_island_host_origin
			cannibalism_siege_commune_origin
			cannibalism_march_host_origin
			cannibalism_archipelago_hunt
			cannibalism_city_that_eats
			cannibalism_moving_front
		}
		delete_unit_template_and_units = { division_template = "Scavenger Warband" disband = no }
		delete_unit_template_and_units = { division_template = "Feast Guard" disband = no }
		delete_unit_template_and_units = { division_template = "Feast Cohort" disband = no }
		delete_unit_template_and_units = { division_template = "Bone Guard" disband = no }
		delete_unit_template_and_units = { division_template = "Island Reavers" disband = no }
		delete_unit_template_and_units = { division_template = "Siege Eaters" disband = no }
		delete_unit_template_and_units = { division_template = "March Predation Column" disband = no }
		delete_unit_template_and_units = { division_template = "Network Cadre" disband = no }
		if = {
			limit = { is_in_array = { array = global.cannibalism_convergence_actor_countries value = THIS } }
			remove_from_array = { array = global.cannibalism_convergence_actor_countries value = THIS }
		}
		if = {
			limit = {
				has_event_target = cannibalism_latest_actor
				event_target:cannibalism_latest_actor = { tag = event_target:cannibalism_warlord_release_country }
			}
			clear_global_event_target = cannibalism_latest_actor
		}
		if = {
			limit = {
				has_event_target = cannibalism_evolution_i_mtth_actor
				event_target:cannibalism_evolution_i_mtth_actor = { tag = event_target:cannibalism_warlord_release_country }
			}
			clear_global_event_target = cannibalism_evolution_i_mtth_actor
			clear_variable = global.cannibalism_evolution_i_mtth_actor_generation
		}
		if = {
			limit = {
				has_event_target = cannibalism_evolution_ii_mtth_actor
				event_target:cannibalism_evolution_ii_mtth_actor = { tag = event_target:cannibalism_warlord_release_country }
			}
			clear_global_event_target = cannibalism_evolution_ii_mtth_actor
			clear_variable = global.cannibalism_evolution_ii_mtth_actor_generation
		}
		every_state = {
			limit = {
				OR = {
					is_core_of = event_target:cannibalism_warlord_release_country
					AND = {
						has_variable = cannibalism_node_source_country
						check_variable = {
							var = cannibalism_node_source_country
							value = event_target:cannibalism_warlord_release_country
							compare = equals
						}
					}
				}
			}
			cannibalism_retire_current_state_from_warlord_slot = yes
		}
		if = {
			limit = { has_variable = cannibalism_actor_generation }
			set_temp_variable = { cannibalism_expected_actor_generation = cannibalism_actor_generation }
			cannibalism_unregister_current_country_actor = yes
		}
		set_variable = { cannibalism_warlord_slot_reference_count = 0 }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_frontline_crisis } } remove_dynamic_modifier = { modifier = cannibalism_frontline_crisis } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_emergency_logistics_burden } } remove_dynamic_modifier = { modifier = cannibalism_emergency_logistics_burden } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_formation_screening_burden } } remove_dynamic_modifier = { modifier = cannibalism_formation_screening_burden } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_terror_battalion_deployed } } remove_dynamic_modifier = { modifier = cannibalism_terror_battalion_deployed } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_synchronized_warlord_attack } } remove_dynamic_modifier = { modifier = cannibalism_synchronized_warlord_attack } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_island_convoy_ambush_operation } } remove_dynamic_modifier = { modifier = cannibalism_island_convoy_ambush_operation } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_siege_relief_ambush_operation } } remove_dynamic_modifier = { modifier = cannibalism_siege_relief_ambush_operation } }
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_march_supply_raid_operation } } remove_dynamic_modifier = { modifier = cannibalism_march_supply_raid_operation } }
	}
}

cannibalism_finalize_current_warlord_slot_release = {
	if = {
		limit = {
			has_country_flag = cannibalism_warlord_slot_release_pending
			cannibalism_current_warlord_slot_references_are_clear = yes
		}
		set_variable = { cannibalism_warlord_slot_reuse_date = global.date }
		add_to_variable = { cannibalism_warlord_slot_reuse_date = constant:cannibalism_warlord_creation.reuse_quarantine_days }
		set_variable = { global.cannibalism_warlord_slot_reuse_date_entries^cannibalism_warlord_slot_index = cannibalism_warlord_slot_reuse_date }
		if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cba } } clr_global_flag = cannibalism_slot_cba_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbb } } clr_global_flag = cannibalism_slot_cbb_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbc } } clr_global_flag = cannibalism_slot_cbc_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbd } } clr_global_flag = cannibalism_slot_cbd_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbe } } clr_global_flag = cannibalism_slot_cbe_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbf } } clr_global_flag = cannibalism_slot_cbf_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbg } } clr_global_flag = cannibalism_slot_cbg_in_use }
		else_if = { limit = { check_variable = { cannibalism_warlord_slot_index = constant:cannibalism_slot.cbh } } clr_global_flag = cannibalism_slot_cbh_in_use }
		clr_country_flag = cannibalism_warlord_slot_in_use
		clr_country_flag = cannibalism_warlord_slot_release_pending
		clr_country_flag = cannibalism_warlord_integration_release_pending
		clr_country_flag = cannibalism_event_created_warlord
		clr_country_flag = cannibalism_origin_island_host
		clr_country_flag = cannibalism_origin_siege_commune
		clr_country_flag = cannibalism_origin_march_host
		set_country_flag = cannibalism_warlord_slot_release_verified
		clear_variable = cannibalism_warlord_slot_reference_count
	}
}
