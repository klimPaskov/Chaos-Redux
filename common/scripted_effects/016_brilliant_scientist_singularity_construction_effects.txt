# Country-scoped payment and callback transaction wrappers for the existing
# Singularity command-core and power-link decisions. The caller supplies the
# selected state through the regular event target
# `brilliant_scientist_singularity_site_selected_state`, plus temporary
# `brilliant_scientist_singularity_site_selected_role` and
# `brilliant_scientist_singularity_site_selected_quantity = 1`.
#
# The physical-site ledger owns the selected state row. This file owns the
# private resource receipt and the shared timed-stage flag. It never changes
# the existing stage durations, AI weights, or component counts, and it never
# adds or removes civilian factories. Native decision `civilian_factory_use`
# remains the sole owner of temporary CIC occupation and release.

# COUNTRY scope. Clears only this transaction's private receipt metadata. The
# caller must snapshot any refund before invoking this effect. Selected event
# targets and temporary callback inputs are intentionally left untouched for
# the physical-site completion/cancellation effect that follows.
brilliant_scientist_singularity_construction_clear_receipt = {
	clr_country_flag = brilliant_scientist_singularity_construction_receipt_active
	clear_variable = brilliant_scientist_singularity_construction_receipt_state
	clear_variable = brilliant_scientist_singularity_construction_receipt_role
	clear_variable = brilliant_scientist_singularity_construction_receipt_quantity
	clear_variable = brilliant_scientist_singularity_construction_receipt_political_power
	clear_variable = brilliant_scientist_singularity_construction_receipt_support_equipment
	clear_variable = brilliant_scientist_singularity_construction_receipt_trucks
	clear_variable = brilliant_scientist_singularity_construction_receipt_fuel
	clear_variable = brilliant_scientist_singularity_construction_receipt_manpower
}

# COUNTRY scope. Copies the private receipt into temporary positive refund
# amounts before the normal receipt variables are cleared. Missing receipt
# fields make the snapshot incomplete; cleanup then clears the stale receipt
# without inventing a replacement cost profile.
brilliant_scientist_singularity_construction_snapshot_receipt = {
	set_temp_variable = { brilliant_scientist_singularity_construction_refund_snapshot_complete = constant:brilliant_scientist_value.one }
	set_temp_variable = { brilliant_scientist_singularity_construction_refund_political_power = brilliant_scientist_singularity_construction_receipt_political_power }
	set_temp_variable = { brilliant_scientist_singularity_construction_refund_support_equipment = brilliant_scientist_singularity_construction_receipt_support_equipment }
	set_temp_variable = { brilliant_scientist_singularity_construction_refund_trucks = brilliant_scientist_singularity_construction_receipt_trucks }
	set_temp_variable = { brilliant_scientist_singularity_construction_refund_fuel = brilliant_scientist_singularity_construction_receipt_fuel }
	set_temp_variable = { brilliant_scientist_singularity_construction_refund_manpower = brilliant_scientist_singularity_construction_receipt_manpower }
	if = {
		limit = { NOT = { has_variable = brilliant_scientist_singularity_construction_receipt_political_power } }
		set_temp_variable = { brilliant_scientist_singularity_construction_refund_snapshot_complete = constant:brilliant_scientist_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = brilliant_scientist_singularity_construction_receipt_support_equipment } }
		set_temp_variable = { brilliant_scientist_singularity_construction_refund_snapshot_complete = constant:brilliant_scientist_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = brilliant_scientist_singularity_construction_receipt_trucks } }
		set_temp_variable = { brilliant_scientist_singularity_construction_refund_snapshot_complete = constant:brilliant_scientist_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = brilliant_scientist_singularity_construction_receipt_fuel } }
		set_temp_variable = { brilliant_scientist_singularity_construction_refund_snapshot_complete = constant:brilliant_scientist_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = brilliant_scientist_singularity_construction_receipt_manpower } }
		set_temp_variable = { brilliant_scientist_singularity_construction_refund_snapshot_complete = constant:brilliant_scientist_value.zero }
	}
}

# COUNTRY scope. Refunds one complete positive snapshot and nothing else. The
# caller has already proved the private receipt was active and has cleared it;
# no amount is recomputed from current constants during a callback.
brilliant_scientist_singularity_construction_refund_snapshot = {
	if = {
		limit = {
			check_variable = {
				var = brilliant_scientist_singularity_construction_refund_snapshot_complete
				value = constant:brilliant_scientist_value.one
				compare = equals
			}
		}
		add_political_power = brilliant_scientist_singularity_construction_refund_political_power
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = brilliant_scientist_singularity_construction_refund_support_equipment
		}
		add_equipment_to_stockpile = {
			type = motorized_equipment
			amount = brilliant_scientist_singularity_construction_refund_trucks
		}
		add_fuel = brilliant_scientist_singularity_construction_refund_fuel
		add_manpower = brilliant_scientist_singularity_construction_refund_manpower
	}
}

# COUNTRY scope. Finish core for either native timer. Construction-receipt
# identity is checked before any receipt clearing, physical mutation, refund,
# or shared stage-flag mutation. A valid current host and intact selected
# facility gets exactly one physical completion and no refund; every other
# matching callback is cancelled and refunded from its snapshot. Physical
# pending-row identity is part of the success gate, not the outer refund
# identity, so a destruction hook cannot strand a paid receipt.
brilliant_scientist_singularity_construction_finish_core = {
	set_temp_variable = { brilliant_scientist_singularity_construction_finish_applied = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_singularity_construction_finish_refunded = constant:brilliant_scientist_value.zero }
	if = {
		limit = {
			has_country_flag = brilliant_scientist_singularity_construction_receipt_active
			has_event_target = brilliant_scientist_singularity_site_selected_state
			OR = {
				NOT = { brilliant_scientist_singularity_site_pending_receipt_is_present = yes }
				brilliant_scientist_singularity_site_pending_callback_matches_receipt = yes
			}
			has_variable = brilliant_scientist_singularity_construction_receipt_state
			has_variable = brilliant_scientist_singularity_construction_receipt_role
			has_variable = brilliant_scientist_singularity_construction_receipt_quantity
			has_variable = brilliant_scientist_singularity_construction_receipt_political_power
			has_variable = brilliant_scientist_singularity_construction_receipt_support_equipment
			has_variable = brilliant_scientist_singularity_construction_receipt_trucks
			has_variable = brilliant_scientist_singularity_construction_receipt_fuel
			has_variable = brilliant_scientist_singularity_construction_receipt_manpower
			check_variable = {
				var = brilliant_scientist_singularity_site_selected_role
				value = brilliant_scientist_singularity_construction_expected_role
				compare = equals
			}
			check_variable = {
				var = brilliant_scientist_singularity_construction_receipt_state
				value = event_target:brilliant_scientist_singularity_site_selected_state
				compare = equals
			}
			check_variable = {
				var = brilliant_scientist_singularity_construction_receipt_role
				value = brilliant_scientist_singularity_construction_expected_role
				compare = equals
			}
			check_variable = {
				var = brilliant_scientist_singularity_construction_receipt_quantity
				value = brilliant_scientist_singularity_site_selected_quantity
				compare = equals
			}
			check_variable = {
				var = brilliant_scientist_singularity_construction_receipt_quantity
				value = constant:brilliant_scientist_value.one
				compare = equals
			}
		}
		if = {
			limit = {
				brilliant_scientist_project_context_is_valid = yes
				brilliant_scientist_singularity_site_pending_callback_is_valid = yes
			}
			brilliant_scientist_singularity_construction_snapshot_receipt = yes
			brilliant_scientist_singularity_construction_clear_receipt = yes
			brilliant_scientist_singularity_site_complete_pending_construction = yes
			if = {
				limit = {
					check_variable = {
						var = brilliant_scientist_singularity_site_completion_applied
						value = constant:brilliant_scientist_value.one
						compare = equals
					}
				}
				clr_country_flag = brilliant_scientist_project_stage_in_progress
				set_temp_variable = { brilliant_scientist_singularity_construction_finish_applied = constant:brilliant_scientist_value.one }
			}
			else = {
				brilliant_scientist_singularity_site_cancel_pending_construction = yes
				brilliant_scientist_singularity_construction_refund_snapshot = yes
				clr_country_flag = brilliant_scientist_project_stage_in_progress
				if = {
					limit = {
						check_variable = {
							var = brilliant_scientist_singularity_construction_refund_snapshot_complete
							value = constant:brilliant_scientist_value.one
							compare = equals
						}
					}
					set_temp_variable = { brilliant_scientist_singularity_construction_finish_refunded = constant:brilliant_scientist_value.one }
				}
			}
		}
		else = {
			brilliant_scientist_singularity_construction_snapshot_receipt = yes
			brilliant_scientist_singularity_construction_clear_receipt = yes
			brilliant_scientist_singularity_site_cancel_pending_construction = yes
			brilliant_scientist_singularity_construction_refund_snapshot = yes
			clr_country_flag = brilliant_scientist_project_stage_in_progress
			if = {
				limit = {
					check_variable = {
						var = brilliant_scientist_singularity_construction_refund_snapshot_complete
						value = constant:brilliant_scientist_value.one
						compare = equals
					}
				}
				set_temp_variable = { brilliant_scientist_singularity_construction_finish_refunded = constant:brilliant_scientist_value.one }
			}
		}
	}
}
