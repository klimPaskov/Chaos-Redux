# ============================================================================
# CHAOS REDUX - STRATEGIC BIOLOGICAL RAID SCRIPTED EFFECTS
# ============================================================================
# Overview:
# - Resolves the four native raid engine levels into the six accepted ordinary
#   biological outcomes from one exact raid-instance contract.
# - Treats native essential_equipment collection as the payload reservation,
#   consumes a bounded share, and refunds only the unused matching agent model.
# - Dispatches every actual release through biological_lifecycle_effects.txt.
# - Records failed-delivery evidence without seeding contamination and routes an
#   attacker accident only into the actor's designated staging-complex state.
# - Uses state-owned delayed evidence decay. It creates no periodic world pass,
#   launch-state estimate, aircraft-presence proxy, or ordinary-air hook.
#
# Public input contract:
# - raid-instance scope
# - bio_raid_agent: one bio_lifecycle_agent code matching the raid type
# - bio_raid_engine_outcome: one bio_strategic_raid_engine_outcome code
# - native var:actor_country, var:victim_country, and var:target_state
# ============================================================================

bio_strategic_raid_set_reservation_internal = {
	set_temp_variable = { bio_raid_payload_required = constant:bio_lifecycle_math.zero }
	set_temp_variable = { bio_raid_command_cost = constant:bio_lifecycle_math.zero }

	if = {
		limit = { check_variable = { var = bio_raid_agent value = constant:bio_lifecycle_agent.anthrax compare = equals } }
		set_temp_variable = { bio_raid_payload_required = constant:bio_strategic_raid_reservation.anthrax }
		set_temp_variable = { bio_raid_command_cost = constant:bio_strategic_raid_command.anthrax_cost }
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_agent value = constant:bio_lifecycle_agent.plague compare = equals } }
		set_temp_variable = { bio_raid_payload_required = constant:bio_strategic_raid_reservation.plague }
		set_temp_variable = { bio_raid_command_cost = constant:bio_strategic_raid_command.plague_cost }
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_agent value = constant:bio_lifecycle_agent.tularemia compare = equals } }
		set_temp_variable = { bio_raid_payload_required = constant:bio_strategic_raid_reservation.tularemia }
		set_temp_variable = { bio_raid_command_cost = constant:bio_strategic_raid_command.tularemia_cost }
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_agent value = constant:bio_lifecycle_agent.smallpox compare = equals } }
		set_temp_variable = { bio_raid_payload_required = constant:bio_strategic_raid_reservation.smallpox }
		set_temp_variable = { bio_raid_command_cost = constant:bio_strategic_raid_command.smallpox_cost }
	}
}

bio_strategic_raid_select_outcome_internal = {
	set_temp_variable = { bio_raid_outcome = constant:bio_strategic_raid_outcome.rejected }
	if = {
		limit = { check_variable = { var = bio_raid_engine_outcome value = constant:bio_strategic_raid_engine_outcome.failure compare = equals } }
		bio_strategic_raid_select_failure_outcome_internal = yes
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_engine_outcome value = constant:bio_strategic_raid_engine_outcome.limited_success compare = equals } }
		set_temp_variable = { bio_raid_outcome = constant:bio_strategic_raid_outcome.partial_contamination }
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_engine_outcome value = constant:bio_strategic_raid_engine_outcome.success compare = equals } }
		bio_strategic_raid_select_success_outcome_internal = yes
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_engine_outcome value = constant:bio_strategic_raid_engine_outcome.critical_success compare = equals } }
		set_temp_variable = { bio_raid_outcome = constant:bio_strategic_raid_outcome.successful_outbreak_seed }
	}
}

bio_strategic_raid_set_consumption_internal = {
	set_temp_variable = { bio_raid_consumption_ratio = constant:bio_lifecycle_math.zero }
	if = {
		limit = { bio_strategic_raid_outcome_has_no_release = yes }
		set_temp_variable_to_random = {
			var = bio_raid_consumption_ratio
			min = constant:bio_strategic_raid_consumption.failed_min
			max = constant:bio_strategic_raid_consumption.failed_max
		}
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_outcome value = constant:bio_strategic_raid_outcome.partial_contamination compare = equals } }
		set_temp_variable_to_random = {
			var = bio_raid_consumption_ratio
			min = constant:bio_strategic_raid_consumption.partial_min
			max = constant:bio_strategic_raid_consumption.partial_max
		}
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_outcome value = constant:bio_strategic_raid_outcome.hidden_contamination compare = equals } }
		set_temp_variable = { bio_raid_consumption_ratio = constant:bio_strategic_raid_consumption.hidden }
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_outcome value = constant:bio_strategic_raid_outcome.detected_outbreak_seed compare = equals } }
		set_temp_variable = { bio_raid_consumption_ratio = constant:bio_strategic_raid_consumption.detected }
	}
	else_if = {
		limit = { check_variable = { var = bio_raid_outcome value = constant:bio_strategic_raid_outcome.successful_outbreak_seed compare = equals } }
		set_temp_variable = { bio_raid_consumption_ratio = constant:bio_strategic_raid_consumption.successful }
	}
	else_if = {
		limit = { bio_strategic_raid_outcome_is_attacker_accident = yes }
		set_temp_variable = { bio_raid_consumption_ratio = constant:bio_strategic_raid_consumption.attacker_accident }
	}

	set_temp_variable = { bio_raid_payload_consumed = bio_raid_payload_required }
	multiply_temp_variable = { bio_raid_payload_consumed = bio_raid_consumption_ratio }
	round_temp_variable = bio_raid_payload_consumed
	clamp_temp_variable = {
		var = bio_raid_payload_consumed
		min = constant:bio_lifecycle_math.one
		max = bio_raid_payload_required
	}
}

# State scope. Only the currently dominant actor lead is retained.
bio_strategic_raid_record_failed_attempt_evidence_internal = {
	set_temp_variable = { bio_raid_attempt_evidence_points = constant:bio_strategic_raid_outcome_profile.failed_delivery.evidence }
	set_temp_variable = { bio_raid_attempt_lead_matches_actor = constant:bio_lifecycle_proof.missing }
	set_temp_variable = { bio_raid_attempt_evidence_applied = constant:bio_lifecycle_math.zero }
	event_target:bio_raid_actor = {
		if = {
			limit = { cbrn_designer_has_stable_weaponization = yes }
			multiply_temp_variable = { bio_raid_attempt_evidence_points = constant:cbrn_biosecurity_designer_effect.stable_weaponization_evidence_mult }
		}
		if = {
			limit = { cbrn_designer_has_strategic_dissemination_program = yes }
			multiply_temp_variable = { bio_raid_attempt_evidence_points = constant:cbrn_biosecurity_designer_effect.strategic_dissemination_evidence_mult }
		}
	}

	if = {
		limit = { NOT = { has_variable = bio_strategic_raid_attempt_actor } }
		set_variable = { bio_strategic_raid_attempt_actor = event_target:bio_raid_actor }
		set_variable = { bio_strategic_raid_attempt_evidence = bio_raid_attempt_evidence_points }
		set_temp_variable = { bio_raid_attempt_evidence_applied = bio_raid_attempt_evidence_points }
		set_temp_variable = { bio_raid_attempt_lead_matches_actor = constant:bio_lifecycle_proof.supplied }
	}
	else_if = {
		limit = { var:bio_strategic_raid_attempt_actor = { tag = event_target:bio_raid_actor } }
		add_to_variable = { bio_strategic_raid_attempt_evidence = bio_raid_attempt_evidence_points }
		set_temp_variable = { bio_raid_attempt_evidence_applied = bio_raid_attempt_evidence_points }
		set_temp_variable = { bio_raid_attempt_lead_matches_actor = constant:bio_lifecycle_proof.supplied }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = bio_strategic_raid_attempt_evidence }
				check_variable = { var = bio_raid_attempt_evidence_points value = bio_strategic_raid_attempt_evidence compare = greater_than }
			}
		}
		set_variable = { bio_strategic_raid_attempt_actor = event_target:bio_raid_actor }
		set_variable = { bio_strategic_raid_attempt_evidence = bio_raid_attempt_evidence_points }
		set_temp_variable = { bio_raid_attempt_evidence_applied = bio_raid_attempt_evidence_points }
		set_temp_variable = { bio_raid_attempt_lead_matches_actor = constant:bio_lifecycle_proof.supplied }
	}

	if = {
		limit = { has_variable = bio_strategic_raid_attempt_evidence }
		clamp_variable = {
			var = bio_strategic_raid_attempt_evidence
			min = constant:bio_strategic_raid_attempt_evidence.minimum
			max = constant:bio_strategic_raid_attempt_evidence.maximum
		}
	}
	set_state_flag = bio_strategic_raid_failed_attempt_history
	if = {
		limit = {
			check_variable = {
				var = bio_raid_attempt_lead_matches_actor
				value = constant:bio_lifecycle_proof.supplied
				compare = equals
			}
		}
		set_variable = { bio_strategic_raid_attempt_agent = bio_raid_agent }
		set_variable = { bio_strategic_raid_attempt_date = global.num_days }
		set_variable = { bio_strategic_raid_attempt_evidence_decay_due_day = global.num_days }
		add_to_variable = { bio_strategic_raid_attempt_evidence_decay_due_day = constant:bio_strategic_raid_attempt_evidence.decay_days }
		if = {
			limit = { NOT = { has_state_flag = bio_strategic_raid_attempt_evidence_decay_scheduled } }
			set_state_flag = bio_strategic_raid_attempt_evidence_decay_scheduled
			set_temp_variable = { bio_raid_attempt_decay_days = constant:bio_strategic_raid_attempt_evidence.decay_days }
			round_temp_variable = bio_raid_attempt_decay_days
			meta_effect = {
				text = {
					state_event = { id = cbrn_bio_lifecycle.21 days = [BIO_RAID_ATTEMPT_DECAY_DAYS] }
				}
				BIO_RAID_ATTEMPT_DECAY_DAYS = "[?bio_raid_attempt_decay_days|.0]"
			}
		}
	}
}

# Actor country scope. Failed delivery is an attempt consequence, never use.
bio_strategic_raid_apply_failed_attempt_condemnation_internal = {
	set_temp_variable = { bio_raid_attempt_condemnation_mult = constant:bio_lifecycle_math.one }
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.biological }
	set_temp_variable = { condemnation_base_gain = constant:bio_strategic_raid_outcome_profile.failed_delivery.condemnation }
	set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.hidden }
	if = {
		limit = {
			check_variable = { var = bio_raid_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals }
			event_target:bio_raid_target_state = {
				check_variable = { var = bio_strategic_raid_attempt_evidence value = constant:bio_lifecycle_evidence.confirmed_threshold compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.public }
	}
	else_if = {
		limit = {
			check_variable = { var = bio_raid_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals }
			event_target:bio_raid_target_state = {
				check_variable = { var = bio_strategic_raid_attempt_evidence value = constant:bio_lifecycle_evidence.probable_threshold compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.discovered }
	}
	else_if = {
		limit = {
			check_variable = { var = bio_raid_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals }
			event_target:bio_raid_target_state = {
				check_variable = { var = bio_strategic_raid_attempt_evidence value = constant:bio_lifecycle_evidence.suspicious_threshold compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.suspected }
	}
	# Doctrine modifies Condemnation only after the evidence record has produced attribution.
	# It never changes evidence, death, contamination, or historical state.
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = condemnation_visibility_type
					value = constant:condemnation_visibility.hidden
					compare = equals
				}
			}
			has_idea = cbrn_terminal_hazard_doctrine_spirit
		}
		set_temp_variable = { bio_raid_attempt_condemnation_mult = constant:bio_lifecycle_doctrine.terminal_hazard.condemnation_mult }
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					var = condemnation_visibility_type
					value = constant:condemnation_visibility.hidden
					compare = equals
				}
			}
			has_idea = cbrn_theater_contamination_doctrine_spirit
		}
		set_temp_variable = { bio_raid_attempt_condemnation_mult = constant:bio_lifecycle_doctrine.theater_contamination.condemnation_mult }
	}
	if = {
		limit = { cbrn_designer_has_strategic_dissemination_program = yes }
		multiply_temp_variable = { bio_raid_attempt_condemnation_mult = constant:cbrn_biosecurity_designer_effect.strategic_dissemination_condemnation_mult }
	}
	multiply_temp_variable = { condemnation_base_gain = bio_raid_attempt_condemnation_mult }
	set_temp_variable = { condemnation_severity_type = constant:condemnation_severity.minor }
	set_temp_variable = { condemnation_civilian_deaths = constant:bio_lifecycle_math.zero }
	set_temp_variable = { condemnation_contamination_value = constant:bio_lifecycle_math.zero }
	set_temp_variable = { condemnation_context_type = constant:condemnation_context.biological_strike }
	set_temp_variable = { condemnation_source_is_no_release_attempt = constant:condemnation_tier.concern }
	set_temp_variable = { condemnation_victim_country = event_target:bio_raid_victim }
	condemnation_add_source = yes
}

# State scope. Invoked only by cbrn_bio_lifecycle.21.
bio_strategic_raid_decay_attempt_evidence = {
	if = {
		limit = {
			has_state_flag = bio_strategic_raid_attempt_evidence_decay_scheduled
			has_variable = bio_strategic_raid_attempt_evidence_decay_due_day
			check_variable = { var = bio_strategic_raid_attempt_evidence_decay_due_day value = global.num_days compare = greater_than }
		}
		set_temp_variable = { bio_raid_attempt_decay_days = bio_strategic_raid_attempt_evidence_decay_due_day }
		subtract_from_temp_variable = { bio_raid_attempt_decay_days = global.num_days }
		round_temp_variable = bio_raid_attempt_decay_days
		clamp_temp_variable = { var = bio_raid_attempt_decay_days min = constant:bio_lifecycle_schedule.minimum_event_delay_days }
		meta_effect = {
			text = {
				state_event = { id = cbrn_bio_lifecycle.21 days = [BIO_RAID_ATTEMPT_DECAY_DAYS] }
			}
			BIO_RAID_ATTEMPT_DECAY_DAYS = "[?bio_raid_attempt_decay_days|.0]"
		}
	}
	else_if = {
		limit = { has_state_flag = bio_strategic_raid_attempt_evidence_decay_scheduled }
		clr_state_flag = bio_strategic_raid_attempt_evidence_decay_scheduled
		clear_variable = bio_strategic_raid_attempt_evidence_decay_due_day
		if = {
			limit = { has_variable = bio_strategic_raid_attempt_evidence }
			subtract_from_variable = { bio_strategic_raid_attempt_evidence = constant:bio_strategic_raid_attempt_evidence.decay_amount }
			clamp_variable = { var = bio_strategic_raid_attempt_evidence min = constant:bio_strategic_raid_attempt_evidence.minimum max = constant:bio_strategic_raid_attempt_evidence.maximum }
		}
		if = {
			limit = { check_variable = { bio_strategic_raid_attempt_evidence > constant:bio_strategic_raid_attempt_evidence.minimum } }
			set_variable = { bio_strategic_raid_attempt_evidence_decay_due_day = global.num_days }
			add_to_variable = { bio_strategic_raid_attempt_evidence_decay_due_day = constant:bio_strategic_raid_attempt_evidence.decay_days }
			set_state_flag = bio_strategic_raid_attempt_evidence_decay_scheduled
			set_temp_variable = { bio_raid_attempt_decay_days = constant:bio_strategic_raid_attempt_evidence.decay_days }
			round_temp_variable = bio_raid_attempt_decay_days
			meta_effect = {
				text = {
					state_event = { id = cbrn_bio_lifecycle.21 days = [BIO_RAID_ATTEMPT_DECAY_DAYS] }
				}
				BIO_RAID_ATTEMPT_DECAY_DAYS = "[?bio_raid_attempt_decay_days|.0]"
			}
		}
		else = {
			clear_variable = bio_strategic_raid_attempt_actor
			clear_variable = bio_strategic_raid_attempt_evidence
			clear_variable = bio_strategic_raid_attempt_agent
			clear_variable = bio_strategic_raid_attempt_date
		}
	}
}

# State scope. Called only for the exact state delivered by
# on_state_control_changed. Losing control clears this state's marker and the
# former actor's pointer only when that pointer still names this same state.
bio_strategic_raid_cleanup_staging_after_control_change = {
	save_event_target_as = bio_strategic_raid_control_changed_state
	if = {
		limit = {
			has_state_flag = bio_strategic_raid_staging_complex
			OR = {
				NOT = { has_variable = bio_strategic_raid_staging_actor }
				var:bio_strategic_raid_staging_actor = { exists = no }
				NOT = { is_controlled_by = var:bio_strategic_raid_staging_actor }
			}
		}
		if = {
			limit = {
				has_variable = bio_strategic_raid_staging_actor
				var:bio_strategic_raid_staging_actor = { exists = yes }
			}
			var:bio_strategic_raid_staging_actor = {
				if = {
					limit = {
						has_variable = bio_strategic_raid_staging_state
						var:bio_strategic_raid_staging_state = { state = event_target:bio_strategic_raid_control_changed_state }
					}
					clear_variable = bio_strategic_raid_staging_state
					set_country_flag = bio_strategic_raid_staging_lost_history
					set_variable = { bio_last_strategic_raid_staging_loss_date = global.num_days }
				}
			}
		}
		clr_state_flag = bio_strategic_raid_staging_complex
		clear_variable = bio_strategic_raid_staging_actor
		clear_variable = bio_strategic_raid_staging_designated_date
	}
}
