gods_of_africa_floor_temporary_value = {
	# Caller owns input/output storage; round then correct an upward rounding.
	set_temp_variable = { gods_of_africa_floor_output = constant:gods_of_africa_value.zero }
	if = {
		limit = { exists = yes }
		set_temp_variable = { gods_of_africa_floor_output = gods_of_africa_floor_input }
		round_temp_variable = gods_of_africa_floor_output
		if = {
			limit = { check_variable = { gods_of_africa_floor_output > gods_of_africa_floor_input } }
			subtract_from_temp_variable = { gods_of_africa_floor_output = constant:gods_of_africa_value.one }
		}
	}
}

gods_of_africa_normalize_paid_quantity_milli = {
	set_temp_variable = { gods_of_africa_normalize_output_milli = constant:gods_of_africa_value.zero }
	if = {
		limit = {
			exists = yes
			check_variable = { gods_of_africa_normalize_denominator > constant:gods_of_africa_value.zero }
			check_variable = { gods_of_africa_normalize_bound_milli > constant:gods_of_africa_value.zero }
		}
		# One safe scale avoids raw-unit intermediates above native fixed-point
		# range and preserves one manpower as 0.1 milli-package.
		set_temp_variable = { gods_of_africa_normalize_scale = constant:gods_of_africa_contribution_units.units_per_thousand }
		multiply_temp_variable = { gods_of_africa_normalize_scale = constant:gods_of_africa_contribution_units.milli_packages_per_package }
		divide_temp_variable = { gods_of_africa_normalize_scale = gods_of_africa_normalize_denominator }
		set_temp_variable = { gods_of_africa_normalize_cap_k = gods_of_africa_normalize_bound_milli }
		divide_temp_variable = { gods_of_africa_normalize_cap_k = gods_of_africa_normalize_scale }
		set_temp_variable = { gods_of_africa_normalize_output_milli = gods_of_africa_normalize_paid_k }
		clamp_temp_variable = { var = gods_of_africa_normalize_output_milli min = constant:gods_of_africa_value.zero max = gods_of_africa_normalize_cap_k }
		multiply_temp_variable = { gods_of_africa_normalize_output_milli = gods_of_africa_normalize_scale }
	}
}

gods_of_africa_calculate_settlement_entitlement = {
	if = {

			gods_of_africa_full_grade_inputs_are_valid = yes
			gods_of_africa_final_settlement_value_is_valid = yes
			# Parent calls this once before committing its source/owed receipt.
			OR = {
				NOT = { has_variable = gods_of_africa_reward_equipment_entitlement }
				NOT = { has_variable = gods_of_africa_reward_fuel_entitlement }
			}
		}
		set_temp_variable = { gods_of_africa_entitlement_strength = gods_of_africa_settlement_strength }
		clamp_temp_variable = { var = gods_of_africa_entitlement_strength min = constant:gods_of_africa_value.zero max = constant:gods_of_africa_value.maximum_strength }
		set_temp_variable = { gods_of_africa_entitlement_contribution_milli = gods_of_africa_grade_material_milli }
		set_temp_variable = { gods_of_africa_entitlement_premium_milli = gods_of_africa_grade_voluntary_milli }
		multiply_temp_variable = { gods_of_africa_entitlement_premium_milli = constant:gods_of_africa_reward_scale.voluntary_premium }
		add_to_temp_variable = { gods_of_africa_entitlement_contribution_milli = gods_of_africa_entitlement_premium_milli }

		# Direct raw Strength coefficients retain fractional capability until final floor.
		set_temp_variable = { gods_of_africa_entitlement_coefficient = constant:gods_of_africa_value.zero }
		set_temp_variable = { gods_of_africa_entitlement_denominator = constant:gods_of_africa_reward_scale.favored_denominator }

		set_temp_variable = { gods_of_africa_entitlement_fuel_ceiling = constant:gods_of_africa_value.zero }
		if = {
			limit = { check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.favored } }
			set_temp_variable = { gods_of_africa_entitlement_coefficient = gods_of_africa_entitlement_strength }
			add_to_temp_variable = { gods_of_africa_entitlement_coefficient = constant:gods_of_africa_reward_scale.favored_coefficient_base }
			if = {
				limit = { check_variable = { gods_of_africa_settlement_doctrine = constant:gods_of_africa_doctrine.reciprocal_covenant } }
				multiply_temp_variable = { gods_of_africa_entitlement_coefficient = constant:gods_of_africa_reward_scale.reciprocal_coefficient_multiplier }

			}
			set_temp_variable = { gods_of_africa_entitlement_equipment_ceiling = constant:gods_of_africa_reward_units.favored_equipment_ceiling }
			set_temp_variable = { gods_of_africa_entitlement_fuel_ceiling = constant:gods_of_africa_reward_units.favored_fuel_ceiling }
		}
		else_if = {
			limit = { check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.respected } }
			set_temp_variable = { gods_of_africa_entitlement_coefficient = gods_of_africa_entitlement_strength }
			multiply_temp_variable = { gods_of_africa_entitlement_coefficient = constant:gods_of_africa_reward_scale.respected_coefficient_strength_multiplier }
			add_to_temp_variable = { gods_of_africa_entitlement_coefficient = constant:gods_of_africa_reward_scale.respected_coefficient_base }

			set_temp_variable = { gods_of_africa_entitlement_fuel_ceiling = constant:gods_of_africa_reward_units.respected_fuel_ceiling }
		}

		set_temp_variable = { gods_of_africa_entitlement_equipment_completed = gods_of_africa_scaled_entitlement_output }

		clamp_temp_variable = { var = gods_of_africa_entitlement_fuel_completed min = constant:gods_of_africa_value.zero max = gods_of_africa_entitlement_fuel_ceiling }
		# Publish only after both components are complete. A partial pair is repairable.
		set_variable = { gods_of_africa_reward_equipment_entitlement = gods_of_africa_entitlement_equipment_completed }
