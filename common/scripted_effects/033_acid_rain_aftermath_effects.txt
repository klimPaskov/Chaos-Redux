# Integrate actual union-active time once. A paid emergency water interval is
# clipped at its captured absolute expiry, even if it expires between workers.
acid_rain_advance_consequence_interval = {
	set_temp_variable = { acid_rain_consequence_elapsed = global.date }
	subtract_from_temp_variable = { acid_rain_consequence_elapsed = acid_rain_consequence_clock_day }
	# Date differences count one hour as 0.00001. Integrate exact fractional days so
	# several partial-day intervals do not each round up to a whole day.
	multiply_temp_variable = { acid_rain_consequence_elapsed = constant:chaosx_date_units.hours_per_date_unit }
	divide_temp_variable = { acid_rain_consequence_elapsed = constant:chaosx_date_units.hours_per_day }
	if = {
		limit = { has_state_flag = acid_rain_consequence_interval_active check_variable = { var = acid_rain_consequence_elapsed value = constant:acid_rain_aftermath.zero compare = greater_than } }
		add_to_variable = { acid_rain_window_days = acid_rain_consequence_elapsed }
		set_temp_variable = { acid_rain_water_integral = acid_rain_window_water_tier }
		multiply_temp_variable = { acid_rain_water_integral = acid_rain_consequence_elapsed }
		if = {
			limit = { check_variable = { var = acid_rain_window_water_tier value = constant:acid_rain_aftermath.water_tier_max compare = less_than } }
			set_temp_variable = { acid_rain_water_bonus_days = acid_rain_window_water_until }
			clamp_temp_variable = { var = acid_rain_water_bonus_days max = global.date }
			subtract_from_temp_variable = { acid_rain_water_bonus_days = acid_rain_consequence_clock_day }
			multiply_temp_variable = { acid_rain_water_bonus_days = constant:chaosx_date_units.hours_per_date_unit }
			divide_temp_variable = { acid_rain_water_bonus_days = constant:chaosx_date_units.hours_per_day }
			clamp_temp_variable = { var = acid_rain_water_bonus_days min = constant:acid_rain_aftermath.zero max = acid_rain_consequence_elapsed }
			multiply_temp_variable = { acid_rain_water_bonus_days = constant:acid_rain_aftermath.emergency_water_tier_bonus }
			add_to_temp_variable = { acid_rain_water_integral = acid_rain_water_bonus_days }
		}
		add_to_variable = { acid_rain_window_water_days = acid_rain_water_integral }
	}
	set_variable = { acid_rain_consequence_clock_day = global.date }
}

# Entry freezes CURRENT population for an ordinary episode. Its six-day gap
# is independent of outstanding debt and of other concurrently active layers.
acid_rain_open_ordinary_episode = {
	acid_rain_advance_consequence_interval = yes
	set_temp_variable = { chaosx_date_days = constant:acid_rain_aftermath.ordinary_inactive_gap_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_temp_variable = { acid_rain_ordinary_gap_offset = chaosx_date_offset }
	set_temp_variable = { acid_rain_ordinary_gap = global.date }
	if = { limit = { has_variable = acid_rain_ordinary_inactive_day } subtract_from_temp_variable = { acid_rain_ordinary_gap = acid_rain_ordinary_inactive_day } }
	if = {
		limit = { OR = { NOT = { has_variable = acid_rain_ordinary_cap_population_people } AND = { has_variable = acid_rain_ordinary_inactive_day check_variable = { var = acid_rain_ordinary_gap value = acid_rain_ordinary_gap_offset compare = greater_than_or_equals } } } }
		acid_rain_assess_window_aftermath = yes
		add_to_variable = { acid_rain_ordinary_episode_id = constant:acid_rain_aftermath.one }
		set_variable = { acid_rain_ordinary_cap_population_people = state_population_k }
		multiply_variable = { acid_rain_ordinary_cap_population_people = constant:chaos_meter_deaths.people_per_k }
		set_variable = { acid_rain_ordinary_episode_loss = constant:acid_rain_aftermath.zero }
		clr_state_flag = acid_rain_opening_pulse_recorded
		acid_rain_start_consequence_window = yes
	}
}

# Called only from the accepted new-key population gateway. Forecast population
# for achievements stays separate. All global/superstorm packets share one cap.
acid_rain_load_episode_cap_ledger = {
	set_temp_variable = { acid_rain_cap_baseline_people = constant:acid_rain_aftermath.zero }
	set_temp_variable = { acid_rain_cap_applied_people = constant:acid_rain_aftermath.zero }
	if = {
		limit = { check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.ordinary compare = equals } }
		set_temp_variable = { acid_rain_cap_baseline_people = acid_rain_ordinary_cap_population_people }
		set_temp_variable = { acid_rain_cap_applied_people = acid_rain_ordinary_episode_loss }
	}
	else_if = {
		limit = { check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.severe compare = equals } }
		if = {
			limit = { NOT = { check_variable = { var = acid_rain_severe_cap_episode_id value = acid_rain_severe_cell_id compare = equals } } }
			set_variable = { acid_rain_severe_cap_episode_id = acid_rain_severe_cell_id }
			set_variable = { acid_rain_severe_cap_population_people = state_population_k }
			multiply_variable = { acid_rain_severe_cap_population_people = constant:chaos_meter_deaths.people_per_k }
			set_variable = { acid_rain_severe_episode_loss = constant:acid_rain_aftermath.zero }
		}
		set_temp_variable = { acid_rain_cap_baseline_people = acid_rain_severe_cap_population_people }
		set_temp_variable = { acid_rain_cap_applied_people = acid_rain_severe_episode_loss }
	}
	else = {
		set_temp_variable = { acid_rain_cap_baseline_people = acid_rain_global_cap_population_people }
		set_temp_variable = { acid_rain_cap_applied_people = acid_rain_global_episode_loss }
	}
}

		acid_rain_floor_consequence_component = yes

		add_to_temp_variable = { acid_rain_consequence_score = acid_rain_window_severe_points }

		add_to_temp_variable = { acid_rain_consequence_score = acid_rain_floor_value }

		if = { limit = { acid_rain_state_has_proven_transport_damage = yes } add_to_temp_variable = { acid_rain_consequence_score = constant:acid_rain_aftermath.unresolved_transport_points } }
		set_temp_variable = { acid_rain_floor_value = constant:acid_rain_aftermath.zero }
		if = {
			limit = { check_variable = { var = acid_rain_window_days value = constant:acid_rain_aftermath.zero compare = greater_than } }
			set_temp_variable = { acid_rain_floor_value = acid_rain_window_water_days }

			divide_temp_variable = { acid_rain_floor_value = constant:acid_rain_aftermath.water_tier_per_credit }
			acid_rain_floor_consequence_component = yes
			clamp_temp_variable = { var = acid_rain_floor_value max = constant:acid_rain_aftermath.water_credit_max }
		}
		subtract_from_temp_variable = { acid_rain_consequence_score = acid_rain_floor_value }
		clamp_temp_variable = { var = acid_rain_consequence_score min = constant:acid_rain_aftermath.zero }
		set_variable = { acid_rain_aftermath_score = acid_rain_consequence_score }
		set_temp_variable = { acid_rain_assessed_tier = constant:acid_rain_aftermath.zero }
		if = { limit = { check_variable = { var = acid_rain_consequence_score value = constant:acid_rain_aftermath.tier_three_score compare = greater_than_or_equals } } set_temp_variable = { acid_rain_assessed_tier = constant:acid_rain_aftermath.tier_max } }

		subtract_from_temp_variable = { acid_rain_new_window_charge = acid_rain_window_charged_tier }
