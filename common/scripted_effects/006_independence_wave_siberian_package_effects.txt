independence_wave_refresh_altai_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_altai_package = yes }
		remove_ideas = { alt_fragmented_mountain_mandate alt_altai_frontier_compact }
		if = {
			limit = { has_stable_independence_wave_altai_compact = yes }
			add_ideas = alt_altai_frontier_compact
			set_country_flag = independence_wave_altai_compact_stabilized

		else = {
			add_ideas = alt_fragmented_mountain_mandate

	}
}
