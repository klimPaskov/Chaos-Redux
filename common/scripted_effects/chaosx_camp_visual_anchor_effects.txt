# Chaos Redux legacy camp map-visual anchor cleanup.
# Camp gameplay buildings are direct map consumers. This state-scoped helper
# removes anchors created by earlier revisions without changing either camp.

@CHAOSX_CAMP_VISUAL_ANCHOR_REMOVE_LEVEL = 0

chaosx_refresh_camp_visual_anchor = {
	set_building_level = {
		type = concentration_camp_site
		level = @CHAOSX_CAMP_VISUAL_ANCHOR_REMOVE_LEVEL
		province = {
			all_provinces = yes
			level > 0
		}
	}
	set_building_level = {
		type = extermination_camp_site
		level = @CHAOSX_CAMP_VISUAL_ANCHOR_REMOVE_LEVEL
		province = {
			all_provinces = yes
			level > 0
		}
	}
}
