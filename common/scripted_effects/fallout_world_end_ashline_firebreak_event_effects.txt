		else_if = { limit = { check_variable = { fallout_event_554_selected_branch = constant:fallout_event_554_branch.military_cordon } check_variable = { fallout_event_554_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_554_history_payload = constant:fallout_event_554_log.cordon_success } }

fallout_event_554_calculate_outcome = {
	set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.failure }
	set_temp_variable = { fallout_event_554_viability = fallout_event_554_frozen_adaptation }
	multiply_temp_variable = { fallout_event_554_viability = constant:fallout_event_554_threshold.adaptation_weight }
	divide_temp_variable = { fallout_event_554_viability = constant:fallout_event_554_threshold.weight_divisor }
	set_temp_variable = { fallout_event_554_component = fallout_event_554_frozen_reclamation }
	multiply_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.reclamation_weight }
	divide_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.weight_divisor }
	add_to_temp_variable = { fallout_event_554_viability = fallout_event_554_component }
	set_temp_variable = { fallout_event_554_component = fallout_event_554_frozen_supply }
	multiply_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.supply_weight }
	divide_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.weight_divisor }
	add_to_temp_variable = { fallout_event_554_viability = fallout_event_554_component }
	set_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.maximum_supply }
	subtract_from_temp_variable = { fallout_event_554_component = fallout_event_554_frozen_exposure }
	if = { limit = { check_variable = { fallout_event_554_component value = constant:fallout_event_runtime.zero compare = less_than } } set_temp_variable = { fallout_event_554_component = constant:fallout_event_runtime.zero } }
	multiply_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.exposure_weight }
	divide_temp_variable = { fallout_event_554_component = constant:fallout_event_554_threshold.weight_divisor }
	add_to_temp_variable = { fallout_event_554_viability = fallout_event_554_component }
	round_temp_variable = fallout_event_554_viability
	clamp_temp_variable = { var = fallout_event_554_viability min = constant:fallout_event_554_threshold.minimum_viability max = constant:fallout_event_554_threshold.maximum_viability }
	if = {
		limit = { check_variable = { fallout_event_554_requested_branch = constant:fallout_event_554_branch.cut_firebreak } }
		if = { limit = { check_variable = { fallout_event_554_viability value = constant:fallout_event_554_threshold.cut_success_viability compare = greater_than_or_equals } } set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.success } }
		else_if = { limit = { check_variable = { fallout_event_554_viability value = constant:fallout_event_554_threshold.cut_partial_viability compare = greater_than_or_equals } } set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	else_if = {
		limit = { check_variable = { fallout_event_554_requested_branch = constant:fallout_event_554_branch.seal_roofs } }
		if = { limit = { check_variable = { fallout_event_554_viability value = constant:fallout_event_554_threshold.seal_success_viability compare = greater_than_or_equals } } set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.success } }
		else_if = { limit = { check_variable = { fallout_event_554_viability value = constant:fallout_event_554_threshold.seal_partial_viability compare = greater_than_or_equals } } set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	else = {
		if = { limit = { check_variable = { fallout_event_554_viability value = constant:fallout_event_554_threshold.cordon_success_viability compare = greater_than_or_equals } } set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.success } }
		else_if = { limit = { check_variable = { fallout_event_554_viability value = constant:fallout_event_554_threshold.cordon_partial_viability compare = greater_than_or_equals } } set_temp_variable = { fallout_event_554_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	clear_temp_variable = fallout_event_554_component
}
