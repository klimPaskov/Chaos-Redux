# Event 021 active-war evolution provenance and owned transaction helpers.
# Opening provenance is captured before mutation and committed with the crisis.
# Active-war review must keep its receipt separate from the opening template.

# ANY scope. Snapshot actual opening stage into serial transaction scratch.
# No actor, country history, generation, weight, or timing is changed here.
event021_capture_pending_opening_evolution_stage = {
	set_variable = { global.random_civil_war_pending_opening_evolution_stage = constant:random_civil_war_value.zero }

		limit = { has_global_flag = random_civil_war_evolution_i_active }
		set_variable = { global.random_civil_war_pending_opening_evolution_stage = constant:random_civil_war_event.evolution_i_stage }
	}

		limit = { has_global_flag = random_civil_war_evolution_ii_active }
		set_variable = { global.random_civil_war_pending_opening_evolution_stage = constant:random_civil_war_event.evolution_ii_stage }
	}
	if = {
		limit = { has_global_flag = random_civil_war_evolution_iii_active }
		set_variable = { global.random_civil_war_pending_opening_evolution_stage = constant:random_civil_war_event.evolution_iii_stage }
	}
}

# -----------------------------------------------------------------------------
# Evolution I review of wars that were already running
# -----------------------------------------------------------------------------
# When Evolution I activates, every live crisis that opened before it is queued
# once on its canonical government. The daily global-host pulse dispatches a
# bounded number of reviews through hidden callback chaosx.nr21.20, so ROOT is
# that government during the review. Each review snapshots the crisis's live
# registered fronts, asks every source side whether it can give up a coherent
# region (or, for the government, a complete Event 006 package) and still keep
# a connected remnant with its capital and forces, and opens each proven side
# at once through the existing secondary adapters. The receipt rows live on the
# government; they never use the global opening template. Temporary blockers
# defer the review on the centralized retry interval instead of closing it.

# ANY scope. Queues one review on the government of every live crisis that
# opened before Evolution I and unlocks its Evolution I management phase.
# Traverses only the bounded active-theater registry.
event021_parent_attach_active_evolution_i_reviews = {
	for_each_scope_loop = {
		array = global.random_civil_war_active_theaters
		if = {
			limit = {
				event021_active_review_baseline_host = yes
				NOT = { has_country_flag = random_civil_war_active_evolution_i_pending }
			}
			set_country_flag = random_civil_war_active_evolution_i_pending
			set_variable = { random_civil_war_active_evolution_i_epoch = global.random_civil_war_evolution_i_date }
			set_variable = { random_civil_war_active_evolution_i_crisis_id = random_civil_war_crisis_id }
			clear_array = random_civil_war_active_evolution_i_reviewed_front_ids
			clear_variable = random_civil_war_active_evolution_i_next_review_date
			event021_parent_grant_evolution_i_management = yes
		}
	}
}

# COUNTRY scope on a crisis government. Gives the government and every live
# registered side of the same crisis the Evolution I management receipt.
event021_parent_grant_evolution_i_management = {
	set_country_flag = random_civil_war_evolution_i_management_ready
	set_temp_variable = { event021_management_crisis_id = random_civil_war_crisis_id }
	for_each_scope_loop = {
		array = global.random_civil_war_front_actors
		if = {
			limit = {
				exists = yes
				has_country_flag = random_civil_war_active
				has_variable = random_civil_war_crisis_id
				check_variable = { var = random_civil_war_crisis_id value = event021_management_crisis_id compare = equals }
				OR = {
					has_country_flag = random_civil_war_government_side
					has_country_flag = random_civil_war_opposition_side
				}
			}
			set_country_flag = random_civil_war_evolution_i_management_ready
		}
	}
}

# COUNTRY scope on the existing Chaos global host. Visits a bounded, rotating
# slice of the active-theater registry and dispatches each due queued review.
# Inspected rows count against the budget whether or not they are queued.
event021_parent_process_active_evolution_i_reviews = {
	if = {
		limit = { has_global_flag = random_civil_war_evolution_i_active }
		if = {
			limit = { NOT = { has_variable = global.random_civil_war_active_evolution_i_cursor } }
			set_variable = { global.random_civil_war_active_evolution_i_cursor = constant:event021_scheduler.first_array_index }
		}
		set_temp_variable = { event021_active_review_index = global.random_civil_war_active_evolution_i_cursor }
		if = {
			limit = { check_variable = { var = event021_active_review_index value = global.random_civil_war_active_theaters^num compare = greater_than_or_equals } }
			set_temp_variable = { event021_active_review_index = constant:event021_scheduler.first_array_index }
		}
		set_temp_variable = { event021_active_review_scanned = constant:random_civil_war_value.zero }
		while_loop_effect = {
			limit = {
				check_variable = { var = event021_active_review_scanned value = constant:event021_active_review.pulse_theater_scan compare = less_than }
				check_variable = { var = event021_active_review_index value = global.random_civil_war_active_theaters^num compare = less_than }
			}
			var:global.random_civil_war_active_theaters^event021_active_review_index = {
				if = {
					limit = {
						has_country_flag = random_civil_war_active_evolution_i_pending
						OR = {
							NOT = { has_variable = random_civil_war_active_evolution_i_next_review_date }
							check_variable = { var = global.date value = random_civil_war_active_evolution_i_next_review_date compare = greater_than_or_equals }
						}
					}
					country_event = { id = chaosx.nr21.20 }
				}
			}
			add_to_temp_variable = { event021_active_review_scanned = constant:random_civil_war_value.one }
			add_to_temp_variable = { event021_active_review_index = constant:random_civil_war_value.one }
		}
		set_variable = { global.random_civil_war_active_evolution_i_cursor = event021_active_review_index }
		if = {
			limit = { check_variable = { var = global.random_civil_war_active_evolution_i_cursor value = global.random_civil_war_active_theaters^num compare = greater_than_or_equals } }
			set_variable = { global.random_civil_war_active_evolution_i_cursor = constant:event021_scheduler.first_array_index }
		}
	}
}

# COUNTRY scope on a queued crisis government; ROOT is the same country
# through chaosx.nr21.20. Closes, defers, or runs the review.
event021_parent_review_active_evolution_i_theater = {
	if = {
		limit = { has_country_flag = random_civil_war_active_evolution_i_pending }
		clear_variable = random_civil_war_active_evolution_i_next_review_date
		if = {
			limit = {
				OR = {
					NOT = { event021_active_review_host_valid = yes }

			}
			event021_parent_finish_active_evolution_i_review = yes
		}

				event021_parent_defer_active_evolution_i_review = yes
			}
			else = { event021_parent_review_active_evolution_i_sources = yes }
		}
	}
}

# release it. The owner's own release helper keeps every terminal-row,
# callback, and durable-copy check; an unproven template stays published.

			NOT = { has_global_flag = random_civil_war_manual_scenario_launch_in_progress }

# COUNTRY scope on a queued crisis government. Snapshots the crisis's live
# registered front IDs, reviews each source once, and either finishes the
# review or defers it while any source is temporarily blocked. Sides created
# by this pass are marked reviewed, so they cannot split again in this epoch.
event021_parent_review_active_evolution_i_sources = {
	save_event_target_as = random_civil_war_active_review_host
	set_temp_variable = { event021_active_review_crisis_id = random_civil_war_crisis_id }
	set_temp_variable = { event021_active_review_deferred_count = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_active_review_opened_count = constant:random_civil_war_value.zero }
	clear_array = random_civil_war_active_evolution_i_sources
	if = {
		limit = {
			check_variable = { var = global.random_civil_war_front_crisis_ids^num value = global.random_civil_war_front_ids^num compare = equals }
			check_variable = { var = global.random_civil_war_front_statuses^num value = global.random_civil_war_front_ids^num compare = equals }
		}
		for_each_loop = {
			array = global.random_civil_war_front_ids
			value = event021_active_review_registry_id
			index = event021_active_review_registry_index
			if = {
				limit = {
					check_variable = { var = event021_active_review_registry_id value = constant:random_civil_war_value.zero compare = greater_than }
					check_variable = { var = global.random_civil_war_front_crisis_ids^event021_active_review_registry_index value = event021_active_review_crisis_id compare = equals }
					check_variable = { var = global.random_civil_war_front_statuses^event021_active_review_registry_index value = constant:random_civil_war_front_status.active compare = equals }
					NOT = { is_in_array = { array = random_civil_war_active_evolution_i_reviewed_front_ids value = event021_active_review_registry_id } }
				}
				add_to_array = { array = random_civil_war_active_evolution_i_sources value = event021_active_review_registry_id }
			}
		}
	}
	for_each_loop = {
		array = random_civil_war_active_evolution_i_sources
		value = event021_active_review_source_id
		event021_parent_review_active_evolution_i_source = yes
	}
	clear_array = random_civil_war_active_evolution_i_sources
	if = {
		limit = { check_variable = { var = event021_active_review_deferred_count value = constant:random_civil_war_value.zero compare = greater_than } }
		event021_parent_defer_active_evolution_i_review = yes
	}
	else = { event021_parent_finish_active_evolution_i_review = yes }
}

# COUNTRY scope on the reviewing government. Input:
# `event021_active_review_source_id`. Resolves that exact registered row to a
# live side of this crisis: the government itself or one of its claimants.
# Outputs `event021_active_review_source_found`, the row generation, and the
# chain-local target `random_civil_war_active_review_source`.
event021_parent_resolve_active_review_source = {
	set_temp_variable = { event021_active_review_source_found = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_active_review_source_break = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_active_review_row_generation = constant:random_civil_war_value.zero }
	if = {
		limit = {
			check_variable = { var = global.random_civil_war_front_actors^num value = global.random_civil_war_front_ids^num compare = equals }
			check_variable = { var = global.random_civil_war_front_crisis_ids^num value = global.random_civil_war_front_ids^num compare = equals }
			check_variable = { var = global.random_civil_war_front_statuses^num value = global.random_civil_war_front_ids^num compare = equals }
			check_variable = { var = global.random_civil_war_front_generations^num value = global.random_civil_war_front_ids^num compare = equals }
		}
		for_each_loop = {
			array = global.random_civil_war_front_ids
			value = event021_active_review_resolve_id
			index = event021_active_review_resolve_index
			break = event021_active_review_source_break
			if = {
				limit = {
					check_variable = { var = event021_active_review_resolve_id value = event021_active_review_source_id compare = equals }
					check_variable = { var = global.random_civil_war_front_crisis_ids^event021_active_review_resolve_index value = event021_active_review_crisis_id compare = equals }
					check_variable = { var = global.random_civil_war_front_statuses^event021_active_review_resolve_index value = constant:random_civil_war_front_status.active compare = equals }
				}
				set_temp_variable = { event021_active_review_source_break = constant:random_civil_war_value.one }
				set_temp_variable = { event021_active_review_row_generation = global.random_civil_war_front_generations^event021_active_review_resolve_index }
				var:global.random_civil_war_front_actors^event021_active_review_resolve_index = {
					if = {
						limit = {
							exists = yes
							has_country_flag = random_civil_war_active
							has_country_flag = random_civil_war_front_registered
							has_variable = random_civil_war_front_id
							check_variable = { var = random_civil_war_front_id value = event021_active_review_source_id compare = equals }
							has_variable = random_civil_war_crisis_id
							check_variable = { var = random_civil_war_crisis_id value = event021_active_review_crisis_id compare = equals }
							OR = {
								AND = {
									tag = event_target:random_civil_war_active_review_host
									has_country_flag = random_civil_war_government_side
								}
								AND = {
									has_country_flag = random_civil_war_opposition_actor
									has_variable = random_civil_war_crisis_host_scope
									var:random_civil_war_crisis_host_scope = { tag = event_target:random_civil_war_active_review_host }
								}
							}
						}
						save_event_target_as = random_civil_war_active_review_source
						set_temp_variable = { event021_active_review_source_found = constant:random_civil_war_value.one }
					}
				}
			}
		}
	}
}

# COUNTRY scope on the reviewing government. Reviews one source front ID and
# records its result: opened, no side, deferred, or ineligible. Every result
# except deferral marks that source reviewed for this epoch.
event021_parent_review_active_evolution_i_source = {
	set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.ineligible }
	event021_parent_resolve_active_review_source = yes
	if = {
		limit = { check_variable = { var = event021_active_review_source_found value = constant:random_civil_war_value.one compare = equals } }
		event_target:random_civil_war_active_review_source = {
			set_temp_variable = { event021_active_review_generation_rounded = constant:random_civil_war_value.zero }
			if = {
				limit = { has_variable = random_civil_war_generation }
				set_temp_variable = { event021_active_review_generation_rounded = random_civil_war_generation }
				round_temp_variable = event021_active_review_generation_rounded
			}
			if = {
				limit = { NOT = { event021_active_review_source_generation_valid = yes } }
				set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.ineligible }
			}
			else_if = {
				limit = { event021_active_review_source_deferred = yes }
				set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.deferred }
			}
			else_if = {
				limit = { NOT = { event021_active_review_source_can_split = yes } }
				set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.no_side }
			}
			else = { set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.none } }
		}
		if = {
			limit = {
				check_variable = { var = event021_active_source_result value = constant:event021_active_review_result.none compare = equals }
				NOT = { random_civil_war_front_capacity_available = yes }
			}
			set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.deferred }
		}
		if = {
			limit = { check_variable = { var = event021_active_source_result value = constant:event021_active_review_result.none compare = equals } }
			event021_parent_collect_active_front_candidates = yes
			event021_parent_build_active_front_rows = yes
			set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.no_side }
			if = {
				limit = { has_country_flag = random_civil_war_active_front_transaction_ready }
				event021_parent_commit_active_front_rows = yes
				if = {
					limit = { check_variable = { var = event021_active_commit_opened value = constant:random_civil_war_value.zero compare = greater_than } }
					set_temp_variable = { event021_active_source_result = constant:event021_active_review_result.opened }
				}
			}
			event021_parent_clear_active_front_rows = yes
		}
	}
	if = {
		limit = { check_variable = { var = event021_active_source_result value = constant:event021_active_review_result.deferred compare = equals } }
		add_to_temp_variable = { event021_active_review_deferred_count = constant:random_civil_war_value.one }
	}
	else = {
		add_to_array = { array = random_civil_war_active_evolution_i_reviewed_front_ids value = event021_active_review_source_id }
		if = {
			limit = { check_variable = { var = event021_active_source_result value = constant:event021_active_review_result.opened compare = equals } }
			add_to_temp_variable = { event021_active_review_opened_count = constant:random_civil_war_value.one }
		}
	}
}

# COUNTRY scope on the reviewing government with a resolved
# `random_civil_war_active_review_source`. Builds the same aligned candidate
# scratch as the opening collector from the source's live map: its eligible
# held states, its own route evidence, its live divisions, the shared spare
# front capacity, and, when the source is this government, complete Event 006
# packages. The source keeps its capital, a share of its states, and a share
# of its divisions; no opening reservation or receipt is read.
event021_parent_collect_active_front_candidates = {
	save_event_target_as = random_civil_war_host_country
	event021_parent_reset_dynamic_optional_front_candidates = yes
	set_temp_variable = { event021_parent_dynamic_active_mode = constant:random_civil_war_value.one }
	set_temp_variable = { event021_parent_dynamic_optional_front_valid = constant:random_civil_war_value.one }
	set_temp_variable = { event021_parent_dynamic_optional_front_count = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_parent_dynamic_ordinary_front_count = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_parent_dynamic_optional_front_capacity = global.random_civil_war_max_active_fronts }
	if = {
		limit = { has_variable = global.random_civil_war_active_front_count }
		subtract_from_temp_variable = { event021_parent_dynamic_optional_front_capacity = global.random_civil_war_active_front_count }
	}
	if = {
		limit = { check_variable = { var = event021_parent_dynamic_optional_front_capacity value = constant:random_civil_war_value.zero compare = less_than } }
		set_temp_variable = { event021_parent_dynamic_optional_front_capacity = constant:random_civil_war_value.zero }
	}
	event_target:random_civil_war_active_review_source = {
		save_event_target_as = random_civil_war_dynamic_candidate_host
		capital_scope = { save_event_target_as = random_civil_war_dynamic_primary_anchor }

# COUNTRY scope on the reviewing government. Floods from the source capital
# across source-held states outside `event021_active_excluded_states`.
# Outputs `event021_active_remnant_reached` and the temporary array
# `event021_active_remnant_reach`; the working state flag is cleared again.
event021_parent_measure_active_source_remnant = {
	clear_temp_array = event021_active_remnant_reach
	event_target:random_civil_war_active_review_source = {
		capital_scope = {
			if = {
				limit = {
					is_owned_by = event_target:random_civil_war_active_review_source
					is_controlled_by = event_target:random_civil_war_active_review_source
					NOT = { is_in_array = { array = event021_active_excluded_states value = THIS } }
				}
				set_state_flag = event021_active_remnant_reach
				add_to_temp_array = { array = event021_active_remnant_reach value = THIS }
			}
		}
		set_temp_variable = { event021_active_remnant_growth = event021_active_remnant_reach^num }
		while_loop_effect = {
			limit = { check_variable = { var = event021_active_remnant_growth value = constant:random_civil_war_value.zero compare = greater_than } }
			set_temp_variable = { event021_active_remnant_growth = constant:random_civil_war_value.zero }
			every_owned_state = {
				limit = {
					NOT = { has_state_flag = event021_active_remnant_reach }
					is_controlled_by = event_target:random_civil_war_active_review_source
					NOT = { is_in_array = { array = event021_active_excluded_states value = THIS } }
					any_neighbor_state = { has_state_flag = event021_active_remnant_reach }
				}
				set_state_flag = event021_active_remnant_reach
				add_to_temp_array = { array = event021_active_remnant_reach value = THIS }
				add_to_temp_variable = { event021_active_remnant_growth = constant:random_civil_war_value.one }
			}
		}
	}
	for_each_scope_loop = {
		array = event021_active_remnant_reach
		clr_state_flag = event021_active_remnant_reach
	}
	set_temp_variable = { event021_active_remnant_reached = event021_active_remnant_reach^num }
}

# COUNTRY scope on a source side inside the active commit lock. The caller
# has counted `event021_secondary_marker_count` and saved
# `event021_secondary_receipt_source`. Proves the marked states are exactly
# the pending row's span held by this source and derives the army share from
# the row's planned divisions over the source's live divisions.
event021_parent_validate_active_secondary_transfer_receipt = {
	set_temp_variable = { event021_active_receipt_row_found = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_secondary_receipt_verified_count = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_secondary_receipt_count = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_secondary_receipt_divisions = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_secondary_receipt_live_divisions = num_divisions }
	set_temp_variable = { event021_active_receipt_source_front_id = constant:random_civil_war_value.zero }
	if = {
		limit = { has_variable = random_civil_war_front_id }
		set_temp_variable = { event021_active_receipt_source_front_id = random_civil_war_front_id }
	}
	event_target:random_civil_war_active_front_commit_owner = {
		for_each_loop = {
			array = random_civil_war_active_front_ids
			value = event021_active_receipt_pending_id
			index = event021_active_receipt_pending_row
			break = event021_active_receipt_row_found
			if = {
				limit = {
					check_variable = { var = event021_active_receipt_pending_id value = global.random_civil_war_pending_secondary_front_id compare = equals }
					check_variable = { var = random_civil_war_active_front_source_ids^event021_active_receipt_pending_row value = event021_active_receipt_source_front_id compare = equals }
					check_variable = { var = random_civil_war_active_front_actor_types^event021_active_receipt_pending_row value = constant:random_civil_war_planned_actor_type.ordinary_claimant compare = equals }
				}
				set_temp_variable = { event021_active_receipt_row_found = constant:random_civil_war_value.one }
				set_temp_variable = { event021_secondary_receipt_offset = random_civil_war_active_front_state_offsets^event021_active_receipt_pending_row }
				set_temp_variable = { event021_secondary_receipt_count = random_civil_war_active_front_state_counts^event021_active_receipt_pending_row }
				set_temp_variable = { event021_secondary_receipt_end = event021_secondary_receipt_offset }
				add_to_temp_variable = { event021_secondary_receipt_end = event021_secondary_receipt_count }
				set_temp_variable = { event021_secondary_receipt_divisions = random_civil_war_active_front_force_divisions^event021_active_receipt_pending_row }
				set_temp_variable = { event021_active_receipt_state_index = constant:random_civil_war_value.zero }
				for_each_scope_loop = {
					array = random_civil_war_active_front_state_entries
					if = {
						limit = {
							check_variable = { var = event021_active_receipt_state_index value = event021_secondary_receipt_offset compare = greater_than_or_equals }
							check_variable = { var = event021_active_receipt_state_index value = event021_secondary_receipt_end compare = less_than }
							is_owned_by = event_target:event021_secondary_receipt_source
							is_controlled_by = event_target:event021_secondary_receipt_source
							has_state_flag = random_civil_war_secondary_transfer_state
						}
						add_to_temp_variable = { event021_secondary_receipt_verified_count = constant:random_civil_war_value.one }
					}
					add_to_temp_variable = { event021_active_receipt_state_index = constant:random_civil_war_value.one }
				}
			}
		}
	}
	if = {
		limit = {
			check_variable = { var = event021_active_receipt_row_found value = constant:random_civil_war_value.one compare = equals }
			check_variable = { var = event021_secondary_receipt_verified_count value = event021_secondary_receipt_count compare = equals }
			num_of_controlled_states > event021_secondary_receipt_count
			check_variable = { var = event021_secondary_marker_count value = event021_secondary_receipt_count compare = equals }
			check_variable = { var = event021_secondary_receipt_count value = constant:random_civil_war_value.zero compare = greater_than }
			check_variable = { var = event021_secondary_receipt_divisions value = constant:random_civil_war_value.zero compare = greater_than }
			check_variable = { var = event021_secondary_receipt_divisions value = event021_secondary_receipt_live_divisions compare = less_than }
		}
		set_temp_variable = { event021_secondary_transfer_valid = constant:random_civil_war_value.one }
		set_variable = { global.random_civil_war_pending_secondary_opening_state_count = event021_secondary_receipt_verified_count }
		set_variable = { random_civil_war_secondary_army_ratio = event021_secondary_receipt_divisions }
		divide_variable = { random_civil_war_secondary_army_ratio = event021_secondary_receipt_live_divisions }
	}
