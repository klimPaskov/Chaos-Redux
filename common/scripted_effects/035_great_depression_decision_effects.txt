# ============================================================================
# CHAOS REDUX - EVENT 035 DECISION AND MISSION EFFECTS
# ============================================================================
# Overview:
# - Binds every Event 35 decision and mission identifier to a concrete,
#   receipt-backed outcome.
# - All national outcomes pay one declared cost family and pass Severity
#   through one clamped public path.
# - State and country targets retain their scope-specific consequences while
#   ROOT owns the national ledger and public lifecycle.
# - Missions use flags as compact objective receipts and never refund a cost.
# ============================================================================

# Timed flag fields require literal days. Mirrors the central action/event tuning.
@great_depression_action_lock_days = 7
@great_depression_foreign_support_cooldown_days = 30

# COUNTRY scope. Input temp: great_depression_action_delta. The one public
# Severity path. Its tooltip names the change from the same temporary value the
# effect adds; band, phase, spirit and modifier upkeep stay out of the tooltip.
great_depression_apply_public_action_delta = {
	if = {
		limit = { NOT = { check_variable = { var = great_depression_action_delta value = constant:great_depression_action.no_change compare = equals } } }
		custom_effect_tooltip = great_depression_severity_change_tt
	}
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = great_depression_depression_severity } }
			set_variable = { great_depression_depression_severity = constant:great_depression_api.zero }
		}
		add_to_variable = { great_depression_depression_severity = great_depression_action_delta }
		great_depression_clamp_public_severity = yes
		# An action that leaves Severity unchanged keeps the measured trend.
		if = {
			limit = { NOT = { check_variable = { var = great_depression_action_delta value = constant:great_depression_action.no_change compare = equals } } }
			set_variable = { great_depression_severity_trend = great_depression_action_delta }
		}
		if = {
			limit = { check_variable = { var = great_depression_depression_severity value = great_depression_episode_peak_severity compare = greater_than } }
			set_variable = { great_depression_episode_peak_severity = great_depression_depression_severity }
		}
		great_depression_set_qualitative_band = yes
		great_depression_update_phase_hysteresis = yes
		great_depression_apply_active_ideas = yes
		great_depression_refresh_national_modifier = yes
	}
}

great_depression_record_action_receipt = {
	if = {
		limit = { has_variable = great_depression_successful_action_count }
		add_to_variable = { great_depression_successful_action_count = constant:great_depression_api.one }
	}
	else = {
		set_variable = { great_depression_successful_action_count = constant:great_depression_api.one }
	}
	set_variable = { great_depression_last_action_day = global.num_days }
	set_country_flag = { flag = great_depression_action_lock days = @great_depression_action_lock_days }
}

# Rebuild the public mission-capacity ledger from the engine's active mission
# state. The activation triggers remain the authoritative hard gate; this
# ledger is a bounded receipt for UI, audits, and post-load synchronization.
great_depression_refresh_mission_capacity_ledger = {
	if = {
		limit = {
			has_active_mission = great_depression_reopen_depression_center
			has_variable = great_depression_selected_center_state
			NOT = { has_variable = great_depression_mission_reopen_center_state_id }
		}
		set_variable = { great_depression_mission_reopen_center_state_id = great_depression_selected_center_state }
		set_variable = { great_depression_mission_reopen_center_episode_id = great_depression_active_episode_id }
	}
	if = {
		limit = {
			has_active_mission = great_depression_national_employment_guarantee
			NOT = { has_variable = great_depression_employment_guarantee_due_day }
		}
		set_variable = { great_depression_employment_guarantee_due_day = global.num_days }
		add_to_variable = { great_depression_employment_guarantee_due_day = constant:great_depression_event.employment_guarantee_proof_days }
	}
	set_variable = { great_depression_active_main_mission_count = constant:great_depression_api.zero }
	set_variable = { great_depression_active_support_mission_count = constant:great_depression_api.zero }
	set_variable = { great_depression_active_mission_count = constant:great_depression_api.zero }
	if = { limit = { has_active_mission = great_depression_halt_the_panic } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_reopen_depression_center } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_prevent_a_relapse } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_maximum_severity_emergency } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_prevent_national_breakdown } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_contain_financial_contagion } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_prevent_general_strike } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_international_reconstruction } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_coordinate_international_rescue_mission } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_emergency_government_mandate } add_to_variable = { great_depression_active_main_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_keep_essential_freight_moving } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_national_employment_guarantee } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_complete_national_recovery_plan } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_certify_cleared_economy } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	if = { limit = { has_active_mission = great_depression_restore_essential_production } add_to_variable = { great_depression_active_support_mission_count = constant:great_depression_api.one } }
	add_to_variable = { great_depression_active_mission_count = great_depression_active_main_mission_count }
	add_to_variable = { great_depression_active_mission_count = great_depression_active_support_mission_count }
	if = {
		limit = {
			check_variable = {
				var = great_depression_active_mission_count
				value = constant:great_depression_event.maximum_active_missions
				compare = greater_than_or_equals
			}
		}
		set_country_flag = great_depression_mission_capacity_committed
	}
	else = {
		clr_country_flag = great_depression_mission_capacity_committed
	}
}

# COUNTRY scope. The decision's custom cost text already shows the payment, so
# only the Severity change reaches the tooltip.
great_depression_commit_country_action = {
	hidden_effect = { great_depression_pay_action_cost = yes }
	great_depression_apply_public_action_delta = yes
	hidden_effect = {
		great_depression_record_action_receipt = yes
		clear_variable = great_depression_cost_family
	}
}

# COUNTRY scope. Exposed and worldwide-pressure countries can pay for bounded
# response and reconstruction actions without silently receiving a national
# Depression Severity ledger or the full Event 35 crisis surface.
great_depression_commit_external_action = {
	if = {
		limit = { great_depression_active_crisis_is_coherent = yes }
		great_depression_commit_country_action = yes
	}
	else = {
		hidden_effect = {
			great_depression_pay_action_cost = yes
			great_depression_record_action_receipt = yes
			clear_variable = great_depression_cost_family
		}
	}
}

# contagion link that great_depression_commit_partner_action opens when this
# crisis has reached Financial Contagion and the partner is not yet exposed.

	if = {
		limit = {
			great_depression_active_crisis_is_coherent = yes
			has_country_flag = great_depression_evolution_financial_contagion
			FROM = { great_depression_is_valid_contagion_network_target = yes }
			NOT = { is_in_array = { array = great_depression_contagion_exposure_targets value = FROM } }
		}

	}
}

great_depression_record_center_positive_resolution = {
	# A positive treatment resolves a centre once, even if later pressure changes
	# its qualitative status. The receipt prevents repeated decisions from
	# farming the recovery gate.
	if = {
		limit = { NOT = { has_state_flag = great_depression_center_positive_resolution_receipt } }
		set_state_flag = great_depression_center_positive_resolution_receipt
		if = {
			limit = { has_state_flag = great_depression_inherited_center }
			ROOT = { set_country_flag = great_depression_inherited_region_rescued }
		}
		ROOT = {
			set_country_flag = great_depression_center_mission_complete
			if = {
				limit = { has_variable = great_depression_centers_restructured_count }
				add_to_variable = { great_depression_centers_restructured_count = constant:great_depression_api.one }
			}
			else = {
				set_variable = { great_depression_centers_restructured_count = constant:great_depression_api.one }
			}
		}
	}
}

great_depression_remove_center_positive_resolution_credit = {
	if = {
		limit = { has_state_flag = great_depression_center_positive_resolution_receipt }
		ROOT = {
			if = {
				limit = {
					has_variable = great_depression_centers_restructured_count
					check_variable = { var = great_depression_centers_restructured_count value = constant:great_depression_api.zero compare = greater_than }
				}
				subtract_from_variable = { great_depression_centers_restructured_count = constant:great_depression_api.one }
			}
		}
		clr_state_flag = great_depression_center_positive_resolution_receipt
	}
}

		limit = {

			great_depression_global_commitments_are_open = yes
		}
		custom_effect_tooltip = great_depression_global_contribution_tt
		if = {
			limit = {
				OR = {
					NOT = { has_variable = great_depression_contribution_global_episode_id }
					NOT = { check_variable = { var = great_depression_contribution_global_episode_id value = global.great_depression_global_episode_id compare = equals } }

			clr_country_flag = great_depression_global_major_recipient_improved

			clr_country_flag = great_depression_international_debt_conference_complete
			clr_country_flag = great_depression_global_recovery_conference_complete
			clr_country_flag = great_depression_emergency_trade_clearing_complete
			clr_country_flag = great_depression_reconstruction_employment_fund
			clr_country_flag = great_depression_domestic_market_protected
			clr_country_flag = great_depression_industrial_demand_coordinated
			clr_country_flag = great_depression_regional_clearing_contribution
			clr_country_flag = great_depression_supplier_compact_contribution
			clr_country_flag = great_depression_international_reconstruction_complete
		}
		if = {
			limit = { NOT = { is_in_array = { array = global.great_depression_global_contribution_countries value = THIS } } }
			add_to_array = { array = global.great_depression_global_contribution_countries value = THIS }
			set_temp_variable = { great_depression_contribution_recorded_value = num_of_civilian_factories }

			add_to_temp_variable = { great_depression_contribution_recorded_value = great_depression_contribution_naval_weight }

				var = great_depression_contribution_recorded_value
				min = constant:great_depression_api.one
				max = constant:great_depression_event.public_value_maximum
			}
			add_to_array = { array = global.great_depression_global_contribution_values value = great_depression_contribution_recorded_value }
			if = {
				limit = { has_variable = global.great_depression_global_contribution_industrial_weight }

		set_variable = { great_depression_global_contribution_receipt_id = great_depression_last_receipt_id }
		add_to_array = { array = global.great_depression_global_contribution_receipts value = great_depression_global_contribution_receipt_id }
		add_to_array = { array = great_depression_global_contribution_receipt_ids value = great_depression_global_contribution_receipt_id }
		add_to_variable = { great_depression_global_contribution_count = constant:great_depression_api.one }
		set_country_flag = great_depression_global_contribution_made
		if = {
			limit = {
				is_ai = no
				NOT = { is_debug = yes }
				NOT = { has_country_flag = chaosx_test_country_initialized }
				NOT = { has_country_flag = force_trigger_mode_enabled }

			set_variable = { great_depression_achievement_recovery_of_nations_episode_id = global.great_depression_global_episode_id }
		}
	}
}

# COUNTRY scope. Keep Essential Freight Moving restores basic transport and
# supply for the selected center: it starts Transport Integration there, or
# shortens the project already running there once.
great_depression_apply_freight_to_selected_center = {
	set_temp_variable = { great_depression_freight_center_done = constant:great_depression_api.zero }
	if = {
		limit = {
			has_variable = great_depression_selected_center_state
			var:great_depression_selected_center_state = { great_depression_center_transport_target_is_valid = yes }
		}
		set_variable = { great_depression_paid_center_project = constant:great_depression_center_project.transport_integration }
		var:great_depression_selected_center_state = { great_depression_start_paid_center_project = yes }
		set_temp_variable = { great_depression_freight_center_done = constant:great_depression_api.one }
	}

# COUNTRY actor scope. Puts timed devaluation pressure on a bounded set of the
# actor's partners: first the partners its own crisis already reaches, then
# countries under worldwide pressure that it has a proved trade relationship
# with. Each partner keeps one window; a later devaluation refreshes it.
great_depression_apply_devaluation_partner_pressure = {
	set_temp_variable = { great_depression_currency_pressure_new_due_day = global.num_days }
	add_to_temp_variable = { great_depression_currency_pressure_new_due_day = constant:great_depression_currency_action.devaluation_partner_pressure_days }
	set_temp_variable = { great_depression_devaluation_partner_count = constant:great_depression_api.zero }
	set_temp_variable = { great_depression_devaluation_partner_break = constant:great_depression_api.zero }
	if = {
		limit = {
			has_variable = great_depression_contagion_exposure_targets^num
			check_variable = { var = great_depression_contagion_exposure_targets^num value = constant:great_depression_api.zero compare = greater_than }
		}
		for_each_scope_loop = {
			array = great_depression_contagion_exposure_targets
			break = great_depression_devaluation_partner_break
			great_depression_receive_devaluation_pressure = yes
		}
	}
	if = {
		limit = {
			check_variable = { var = great_depression_devaluation_partner_break value = constant:great_depression_api.zero compare = equals }
			has_variable = global.great_depression_global_pressure_countries^num
			check_variable = { var = global.great_depression_global_pressure_countries^num value = constant:great_depression_api.zero compare = greater_than }
		}
		for_each_scope_loop = {

				}

			}
		}
	}
}

# COUNTRY scope. Removes this country's row from the global contribution ledger
# for the current global episode: the country, its recorded industrial weight
# (subtracted from the global total), its contribution receipts, and its
# contribution flags. Safe when the country never contributed.
great_depression_remove_global_contribution = {
	if = {
		limit = {
			has_variable = global.great_depression_global_contribution_countries^num
			is_in_array = { array = global.great_depression_global_contribution_countries value = THIS }
		}
		set_temp_variable = { great_depression_withdrawal_row_found = constant:great_depression_api.zero }
		for_loop_effect = {
			start = 0
			end = global.great_depression_global_contribution_countries^num
			value = great_depression_withdrawal_row_index
			var:global.great_depression_global_contribution_countries^great_depression_withdrawal_row_index = {
				if = {
					limit = { tag = ROOT }
					set_temp_variable = { great_depression_withdrawal_row = great_depression_withdrawal_row_index }
					set_temp_variable = { great_depression_withdrawal_row_found = constant:great_depression_api.one }
				}
			}
		}
		if = {
			limit = { check_variable = { var = great_depression_withdrawal_row_found value = constant:great_depression_api.one compare = equals } }
			if = {
				limit = { has_variable = global.great_depression_global_contribution_industrial_weight }
				subtract_from_variable = { global.great_depression_global_contribution_industrial_weight = global.great_depression_global_contribution_values^great_depression_withdrawal_row }
				if = {
					limit = { check_variable = { var = global.great_depression_global_contribution_industrial_weight value = constant:great_depression_api.zero compare = less_than } }
					set_variable = { global.great_depression_global_contribution_industrial_weight = constant:great_depression_api.zero }
				}
			}
			remove_from_array = { array = global.great_depression_global_contribution_countries index = great_depression_withdrawal_row }
			remove_from_array = { array = global.great_depression_global_contribution_values index = great_depression_withdrawal_row }
		}
	}
	if = {
		limit = {
			has_variable = great_depression_global_contribution_receipt_ids^num
			check_variable = { var = great_depression_global_contribution_receipt_ids^num value = constant:great_depression_api.zero compare = greater_than }
		}
		for_loop_effect = {
			start = 0
			end = great_depression_global_contribution_receipt_ids^num
			value = great_depression_withdrawal_receipt_index
			remove_from_array = { array = global.great_depression_global_contribution_receipts value = great_depression_global_contribution_receipt_ids^great_depression_withdrawal_receipt_index }
		}
	}
	clear_array = great_depression_global_contribution_receipt_ids
	set_variable = { great_depression_global_contribution_count = constant:great_depression_api.zero }
	clr_country_flag = great_depression_global_contribution_made
	clr_country_flag = great_depression_global_public_works_complete
	# The debts rescheduled at an International Debt Conference stay
	# rescheduled, so its flag, which also spares the Debt Overhang scar, stays.
	clr_country_flag = great_depression_global_recovery_conference_complete
	clr_country_flag = great_depression_emergency_trade_clearing_complete
	# Leaving the common fund ends the jobs it paid for.
	clr_country_flag = great_depression_reconstruction_employment_fund
	clr_country_flag = great_depression_domestic_market_protected
	clr_country_flag = great_depression_industrial_demand_coordinated
	clr_country_flag = great_depression_regional_clearing_contribution
	clr_country_flag = great_depression_supplier_compact_contribution
	clr_country_flag = great_depression_international_reconstruction_complete
	clr_country_flag = great_depression_achievement_recovery_of_nations_candidate
	great_depression_remove_conference_participant = yes
}

# COUNTRY actor scope. Mirrors a completed partner transaction into the exact
# sparse contagion row on both countries. Negative deltas are proved aid or
# diversification; positive deltas are deliberate abandonment or exploitation.
great_depression_record_partner_contagion_outcome = {
	if = {
		limit = {
			great_depression_active_crisis_is_coherent = yes
			has_country_flag = great_depression_evolution_financial_contagion
			has_event_target = great_depression_partner_action_target
			is_in_array = { array = great_depression_contagion_exposure_targets value = event_target:great_depression_partner_action_target }
		}
		set_temp_variable = { great_depression_partner_row_due_day = global.num_days }
		add_to_temp_variable = { great_depression_partner_row_due_day = constant:great_depression_contagion_row.pair_cooldown_days }
		set_temp_variable = { great_depression_partner_row_aid_state = constant:great_depression_contagion_aid_state.aid_received }
		set_temp_variable = { great_depression_partner_row_material_source = constant:great_depression_contagion_material_source.partner_aid }
		if = {
			limit = {
				OR = {
					check_variable = { var = great_depression_action_delta value = constant:great_depression_api.zero compare = greater_than }
					check_variable = { var = great_depression_partner_outcome_hostile value = constant:great_depression_api.one compare = equals }
				}
			}
			set_temp_variable = { great_depression_partner_row_aid_state = constant:great_depression_contagion_aid_state.abandoned }
			set_temp_variable = { great_depression_partner_row_material_source = constant:great_depression_contagion_material_source.abandonment }
		}
		for_loop_effect = {
			start = 0
			end = great_depression_contagion_exposure_targets^num
			value = great_depression_partner_outgoing_row_index
			var:great_depression_contagion_exposure_targets^great_depression_partner_outgoing_row_index = {
				if = {
					limit = { tag = event_target:great_depression_partner_action_target }
					ROOT = {
						set_variable = { great_depression_contagion_target_aid_state_entries^great_depression_partner_outgoing_row_index = great_depression_partner_row_aid_state }
						set_variable = { great_depression_contagion_target_material_source_entries^great_depression_partner_outgoing_row_index = great_depression_partner_row_material_source }
						set_variable = { great_depression_contagion_target_cooldown_day_entries^great_depression_partner_outgoing_row_index = great_depression_partner_row_due_day }
					}
				}
			}
		}
		event_target:great_depression_partner_action_target = {
			if = {
				limit = { has_variable = great_depression_contagion_source_countries^num }
				for_loop_effect = {
					start = 0
					end = great_depression_contagion_source_countries^num
					value = great_depression_partner_incoming_row_index
					var:great_depression_contagion_source_countries^great_depression_partner_incoming_row_index = {
						if = {
							limit = { tag = ROOT }
							PREV = {
								set_variable = { great_depression_contagion_source_aid_state_entries^great_depression_partner_incoming_row_index = great_depression_partner_row_aid_state }
								set_variable = { great_depression_contagion_source_material_source_entries^great_depression_partner_incoming_row_index = great_depression_partner_row_material_source }
								set_variable = { great_depression_contagion_source_cooldown_day_entries^great_depression_partner_incoming_row_index = great_depression_partner_row_due_day }
							}
						}
					}
				}
			}
		}
	}
}

# COUNTRY actor scope with the partner in great_depression_partner_action_target.
# Raises the actor's existing contagion row towards that partner by
# great_depression_partner_link_raise stages (temporary input), capped at
# conversion readiness, and applies the raised row once through
# great_depression_apply_exposure so the partner's stage and conversion rules
# follow it. Every exposure input it writes is cleared again.
great_depression_raise_partner_exposure_link = {
	set_temp_variable = { great_depression_raise_row_found = constant:great_depression_api.zero }
	for_loop_effect = {
		start = 0
		end = great_depression_contagion_exposure_targets^num
		value = great_depression_raise_row_index
		var:great_depression_contagion_exposure_targets^great_depression_raise_row_index = {
			if = {
				limit = { tag = event_target:great_depression_partner_action_target }
				set_temp_variable = { great_depression_raise_row = great_depression_raise_row_index }
				set_temp_variable = { great_depression_raise_row_found = constant:great_depression_api.one }
			}
		}
	}
	if = {
		limit = { check_variable = { var = great_depression_raise_row_found value = constant:great_depression_api.one compare = equals } }
		set_variable = { great_depression_exposure_stage = great_depression_contagion_target_stage_entries^great_depression_raise_row }
		add_to_variable = { great_depression_exposure_stage = great_depression_partner_link_raise }
		clamp_variable = {
			var = great_depression_exposure_stage
			min = constant:great_depression_exposure_stage.observed
			max = constant:great_depression_exposure_stage.conversion_ready
		}
		set_variable = { great_depression_exposure_relationship_type = great_depression_contagion_target_relationship_entries^great_depression_raise_row }
		set_variable = { great_depression_exposure_relationship_proof = constant:great_depression_api.one }
		set_variable = { great_depression_exposure_aid_state = constant:great_depression_contagion_aid_state.abandoned }
		set_variable = { great_depression_exposure_material_source = constant:great_depression_contagion_material_source.abandonment }
		set_variable = { great_depression_exposure_conversion_receipt = great_depression_contagion_target_conversion_receipt_entries^great_depression_raise_row }
		set_variable = { great_depression_exposure_propagation_depth = great_depression_contagion_target_propagation_depth_entries^great_depression_raise_row }
		set_variable = { great_depression_exposure_cooldown_day = great_depression_contagion_target_cooldown_day_entries^great_depression_raise_row }
		event_target:great_depression_partner_action_target = { save_event_target_as = great_depression_exposure_target }
		hidden_effect = { great_depression_apply_exposure = yes }
		clear_variable = great_depression_exposure_stage
		clear_variable = great_depression_exposure_relationship_type
		clear_variable = great_depression_exposure_relationship_proof
		clear_variable = great_depression_exposure_aid_state
		clear_variable = great_depression_exposure_material_source
		clear_variable = great_depression_exposure_conversion_receipt
		clear_variable = great_depression_exposure_propagation_depth
		clear_variable = great_depression_exposure_cooldown_day
	}
}

great_depression_commit_partner_action = {
	if = {
		limit = { has_event_target = great_depression_partner_action_target }
		event_target:great_depression_partner_action_target = {
			save_event_target_as = great_depression_last_partner
		}
		if = {
			limit = {
				great_depression_active_crisis_is_coherent = yes
				has_country_flag = great_depression_evolution_financial_contagion
				event_target:great_depression_partner_action_target = { great_depression_is_valid_contagion_network_target = yes }
				NOT = { is_in_array = { array = great_depression_contagion_exposure_targets value = event_target:great_depression_partner_action_target } }
			}
			event_target:great_depression_partner_action_target = { save_event_target_as = great_depression_exposure_target }
			set_variable = { great_depression_exposure_stage = constant:great_depression_exposure_stage.pressured }
			if = {
				limit = { check_variable = { var = great_depression_action_delta value = constant:great_depression_api.zero compare = greater_than } }
				set_variable = { great_depression_exposure_stage = constant:great_depression_exposure_stage.dependent }
			}
			# A market exit names the stage at which the cut link opens.
			if = {
				limit = { check_variable = { var = great_depression_partner_link_stage value = constant:great_depression_api.zero compare = greater_than } }
				set_variable = { great_depression_exposure_stage = great_depression_partner_link_stage }
			}
			set_variable = { great_depression_exposure_relationship_type = constant:great_depression_exposure_relationship.event_agreement }
			set_variable = { great_depression_exposure_relationship_proof = constant:great_depression_api.one }
			set_variable = { great_depression_exposure_aid_state = constant:great_depression_contagion_aid_state.aid_received }
			set_variable = { great_depression_exposure_material_source = constant:great_depression_contagion_material_source.partner_aid }
			if = {
				limit = {
					OR = {
						check_variable = { var = great_depression_action_delta value = constant:great_depression_api.zero compare = greater_than }
						check_variable = { var = great_depression_partner_outcome_hostile value = constant:great_depression_api.one compare = equals }
					}
				}
				set_variable = { great_depression_exposure_aid_state = constant:great_depression_contagion_aid_state.abandoned }
				set_variable = { great_depression_exposure_material_source = constant:great_depression_contagion_material_source.abandonment }
			}
			set_variable = { great_depression_exposure_cooldown_day = global.num_days }
			add_to_variable = { great_depression_exposure_cooldown_day = constant:great_depression_contagion_row.pair_cooldown_days }
			hidden_effect = { great_depression_apply_exposure = yes }
			clear_variable = great_depression_exposure_stage
			clear_variable = great_depression_exposure_relationship_type
			clear_variable = great_depression_exposure_relationship_proof
			clear_variable = great_depression_exposure_aid_state
			clear_variable = great_depression_exposure_material_source
			clear_variable = great_depression_exposure_cooldown_day
		}
		else_if = {
			# Abandoning a partner already exposed to this crisis deepens its
			# existing link once; the link's own conversion rules still apply.
			limit = {
				check_variable = { var = great_depression_partner_link_raise value = constant:great_depression_api.zero compare = greater_than }
				great_depression_active_crisis_is_coherent = yes
				has_country_flag = great_depression_evolution_financial_contagion
				event_target:great_depression_partner_action_target = {
					great_depression_is_valid_contagion_network_target = yes
					NOT = { has_country_flag = great_depression_contagion_contained }
				}
				is_in_array = { array = great_depression_contagion_exposure_targets value = event_target:great_depression_partner_action_target }
			}
			great_depression_raise_partner_exposure_link = yes
		}
		great_depression_record_partner_contagion_outcome = yes
		event_target:great_depression_partner_action_target = { great_depression_register_partner_pair_cooldown = yes }
		if = {
			limit = { has_global_flag = great_depression_worldwide_episode_active }
			set_country_flag = great_depression_global_recipient_action_made
			if = {
				limit = {
					check_variable = {
						var = great_depression_action_delta
						value = constant:great_depression_api.zero
						compare = less_than
					}
				}
				if = {
					limit = { great_depression_active_crisis_is_coherent = yes }
					set_country_flag = great_depression_global_recipient_improved
					if = {
						limit = { is_major = yes }
						set_country_flag = great_depression_global_major_recipient_improved
					}
				}
				else = {
					event_target:great_depression_partner_action_target = {
						set_country_flag = great_depression_global_recipient_improved
						ROOT = { set_country_flag = great_depression_global_recipient_improved }
						if = {
							limit = { is_major = yes }
							ROOT = { set_country_flag = great_depression_global_major_recipient_improved }
						}
					}
				}
			}
		}
	}
	# Outside the partner block so the decision tooltip always shows the cost
	# and Severity change; the callers always save the partner first.
	great_depression_commit_external_action = yes
}

# Establish a Clearing Bloc, Regional Clearing Agreement and the clearing-bloc
# incident for a fixed term, and the protectionist bloc, joined for the rest
# of the worldwide depression. A country leaves both when a new national
# episode starts there, so no membership carries relief into a later episode.
# -----------------------------------------------------------------------------

# COUNTRY scope. Joins the clearing bloc, or renews the term of a member.
# A non-member joins only while the bloc has room.
great_depression_join_clearing_bloc = {
	if = {
		limit = {
			exists = yes
			OR = {
				is_in_array = { array = global.great_depression_clearing_bloc_members value = THIS }
				NOT = { has_variable = global.great_depression_clearing_bloc_members^num }
				check_variable = { var = global.great_depression_clearing_bloc_members^num value = constant:great_depression_bloc.member_cap compare = less_than }
			}
		}
		if = {
			limit = { NOT = { is_in_array = { array = global.great_depression_clearing_bloc_members value = THIS } } }
			add_to_array = { array = global.great_depression_clearing_bloc_members value = THIS }
		}
		set_country_flag = great_depression_clearing_bloc_member
		set_variable = { great_depression_clearing_bloc_due_day = global.num_days }
		add_to_variable = { great_depression_clearing_bloc_due_day = constant:great_depression_bloc.clearing_term_days }
		set_temp_variable = { great_depression_clearing_bloc_term = constant:great_depression_bloc.clearing_term_days }
		add_dynamic_modifier = { modifier = great_depression_clearing_bloc_membership days = great_depression_clearing_bloc_term }
	}
}

# COUNTRY scope. Ends this country's clearing bloc membership. Safe for a
# country that is not a member.
great_depression_leave_clearing_bloc = {
	if = {
		limit = { is_in_array = { array = global.great_depression_clearing_bloc_members value = THIS } }
		remove_from_array = { array = global.great_depression_clearing_bloc_members value = THIS }
	}
	clr_country_flag = great_depression_clearing_bloc_member
	clear_variable = great_depression_clearing_bloc_due_day
	remove_dynamic_modifier = { modifier = great_depression_clearing_bloc_membership }
	great_depression_reset_clearing_bloc_announcement_if_small = yes
}

# ANY scope. Surviving clearing agreements are not a new bloc when the next
# worldwide episode begins. A bloc below formation size can form anew later.
great_depression_reset_clearing_bloc_announcement_if_small = {
	if = {
		limit = { check_variable = { var = global.great_depression_clearing_bloc_members^num value = constant:great_depression_bloc.formation_member_minimum compare = less_than } }
		clr_global_flag = great_depression_clearing_bloc_formation_announced
	}
}

# COUNTRY scope. Joins the protectionist bloc for the current global episode
# while the bloc has room.
great_depression_join_protectionist_bloc = {
	if = {
		limit = {
			exists = yes
			has_global_flag = great_depression_worldwide_episode_active
			NOT = { is_in_array = { array = global.great_depression_protectionist_bloc_members value = THIS } }
			OR = {
				NOT = { has_variable = global.great_depression_protectionist_bloc_members^num }
				check_variable = { var = global.great_depression_protectionist_bloc_members^num value = constant:great_depression_bloc.member_cap compare = less_than }
			}
		}
		add_to_array = { array = global.great_depression_protectionist_bloc_members value = THIS }
		set_country_flag = great_depression_protectionist_bloc_member
		set_variable = { great_depression_protectionist_bloc_global_episode_id = global.great_depression_global_episode_id }
		add_dynamic_modifier = { modifier = great_depression_protectionist_bloc_membership }
	}
}

# COUNTRY scope, the country whose action grew the bloc. Input temp:
# great_depression_bloc_kind. Counts the bloc's surviving members and majors;
# at the formation size with a major member it asks for the bloc's world news,
# which great_depression_queue_global_news sends once per global episode.
great_depression_check_bloc_formation = {
	set_temp_variable = { great_depression_bloc_member_count = constant:great_depression_api.zero }
	set_temp_variable = { great_depression_bloc_major_count = constant:great_depression_api.zero }
	if = {
		limit = { check_variable = { var = great_depression_bloc_kind value = constant:great_depression_bloc.kind_clearing compare = equals } }
		if = {
			limit = { has_variable = global.great_depression_clearing_bloc_members^num }
			for_each_scope_loop = {
				array = global.great_depression_clearing_bloc_members
				if = {
					limit = { exists = yes }
					add_to_temp_variable = { great_depression_bloc_member_count = constant:great_depression_api.one }
					if = {
						limit = { is_major = yes }
						add_to_temp_variable = { great_depression_bloc_major_count = constant:great_depression_api.one }
					}
				}
			}
		}
	}
	else = {
		if = {
			limit = { has_variable = global.great_depression_protectionist_bloc_members^num }
			for_each_scope_loop = {
				array = global.great_depression_protectionist_bloc_members
				if = {
					limit = { exists = yes }
					add_to_temp_variable = { great_depression_bloc_member_count = constant:great_depression_api.one }
					if = {
						limit = { is_major = yes }
						add_to_temp_variable = { great_depression_bloc_major_count = constant:great_depression_api.one }
					}
				}
			}
		}
	}
	if = {
		limit = {
			has_global_flag = great_depression_worldwide_episode_active
			check_variable = { var = great_depression_bloc_member_count value = constant:great_depression_bloc.formation_member_minimum compare = greater_than_or_equals }
			check_variable = { var = great_depression_bloc_major_count value = constant:great_depression_bloc.formation_major_minimum compare = greater_than_or_equals }
		}
		if = {
			limit = { check_variable = { var = great_depression_bloc_kind value = constant:great_depression_bloc.kind_clearing compare = equals } }
			set_temp_variable = { great_depression_global_news_kind = constant:great_depression_global_news.clearing_bloc }
		}
		else = {
			set_temp_variable = { great_depression_global_news_kind = constant:great_depression_global_news.protectionist_bloc }
		}
		great_depression_queue_global_news = yes
	}
}

# Partner scope with ROOT as the buyer. The partner's broken supply adds to its
# own depression and leaves resentment of ROOT; ROOT secures industrial inputs
# for the current episode, at the price of social backlash at home while
# Social Collapse is active.
great_depression_exploit_supply_gap = {
	save_event_target_as = great_depression_partner_action_target
	if = {
		limit = { great_depression_active_crisis_is_coherent = yes }
		set_temp_variable = { great_depression_action_delta = constant:great_depression_market_exit.exploit_partner_pressure }
		great_depression_apply_public_action_delta = yes
	}
	add_opinion_modifier = { target = ROOT modifier = great_depression_supply_gap_exploited }
	ROOT = {
		set_variable = { great_depression_cost_family = constant:great_depression_cost_family.contagion }
		set_temp_variable = { great_depression_action_delta = constant:great_depression_action.no_change }
		set_temp_variable = { great_depression_partner_outcome_hostile = constant:great_depression_api.one }
		hidden_effect = { great_depression_commit_partner_action = yes }
		set_variable = { great_depression_supply_gap_relief_due_day = global.num_days }
		add_to_variable = { great_depression_supply_gap_relief_due_day = constant:great_depression_market_exit.exploit_relief_days }
		set_variable = { great_depression_supply_gap_relief_episode_id = great_depression_active_episode_id }
	}
}

		clear_variable = great_depression_mission_cost_family
		great_depression_refresh_mission_capacity_ledger = yes
