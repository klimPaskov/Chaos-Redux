# Any indexed mutation first calls this executable audit. It checks alignment,
# positive row identities, legal route/status values, live-row scope payloads,
# exact state metadata, unique live IDs, and one live row per target state.
# A malformed ledger is left untouched and exposes a durable failure flag.
cannibalism_validate_spread_queue = {
	set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
	if = {
		limit = {
			NOT = { has_global_flag = cannibalism_spread_queue_invariant_failure }
			cannibalism_spread_queue_arrays_are_aligned = yes
		}
		set_temp_variable = { cannibalism_spread_queue_validation_result = 1 }
		set_temp_variable = { cannibalism_spread_validation_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = { cannibalism_spread_queue_validation_result > 0 }
				check_variable = {
					var = cannibalism_spread_validation_index
					value = global.cannibalism_spread_queue_id_entries^num
					compare = less_than
				}
			}
			set_temp_variable = { cannibalism_spread_queue_index = cannibalism_spread_validation_index }
			set_temp_variable = { cannibalism_spread_route = global.cannibalism_spread_queue_route_entries^cannibalism_spread_validation_index }
			if = {
				limit = {
					OR = {
						NOT = { check_variable = { global.cannibalism_spread_queue_id_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { check_variable = { global.cannibalism_spread_queue_source_country_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { check_variable = { global.cannibalism_spread_queue_source_state_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { check_variable = { global.cannibalism_spread_queue_target_country_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { check_variable = { global.cannibalism_spread_queue_target_state_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { check_variable = { global.cannibalism_spread_queue_source_generation_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { check_variable = { global.cannibalism_spread_queue_due_date_entries^cannibalism_spread_validation_index > 0 } }
						NOT = { cannibalism_spread_route_value_is_valid = yes }
						NOT = {
							OR = {
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_validation_index = constant:cannibalism_spread_status.queued }
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_validation_index = constant:cannibalism_spread_status.in_transit }
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_validation_index = constant:cannibalism_spread_status.arrived }
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_validation_index = constant:cannibalism_spread_status.contained }
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_validation_index = constant:cannibalism_spread_status.invalidated }
							}
						}
					}
				}
				set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
			}
			if = {
				limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
				set_temp_variable = { cannibalism_spread_id_validation_index = cannibalism_spread_validation_index }
				add_to_temp_variable = { cannibalism_spread_id_validation_index = 1 }
				while_loop_effect = {
					limit = {
						check_variable = { cannibalism_spread_queue_validation_result > 0 }
						check_variable = {
							var = cannibalism_spread_id_validation_index
							value = global.cannibalism_spread_queue_id_entries^num
							compare = less_than
						}
					}
					if = {
						limit = {
							check_variable = {
								var = global.cannibalism_spread_queue_id_entries^cannibalism_spread_id_validation_index
								value = global.cannibalism_spread_queue_id_entries^cannibalism_spread_validation_index
								compare = equals
							}
						}
						set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
					}
					add_to_temp_variable = { cannibalism_spread_id_validation_index = 1 }
				}
			}
			if = {
				limit = {
					check_variable = { cannibalism_spread_queue_validation_result > 0 }
					cannibalism_loaded_spread_status_is_live = yes
				}
				cannibalism_load_spread_entry_context = yes
				if = {
					limit = {
						OR = {
							NOT = { check_variable = { cannibalism_spread_source_country_context_loaded > 0 } }
							NOT = { check_variable = { cannibalism_spread_source_state_context_loaded > 0 } }
							NOT = { check_variable = { cannibalism_spread_target_country_context_loaded > 0 } }
							NOT = { check_variable = { cannibalism_spread_target_state_context_loaded > 0 } }
							NOT = {
								event_target:cannibalism_spread_target_state = {
									has_state_flag = cannibalism_spread_inbound_active
									has_variable = cannibalism_inbound_spread_id
									has_variable = cannibalism_inbound_spread_route
									has_variable = cannibalism_inbound_spread_source_country
									has_variable = cannibalism_inbound_spread_source_generation
									check_variable = { var = cannibalism_inbound_spread_id value = cannibalism_expected_spread_id compare = equals }
									check_variable = { var = cannibalism_inbound_spread_route value = cannibalism_spread_route compare = equals }
									check_variable = { var = cannibalism_inbound_spread_source_country value = event_target:cannibalism_spread_source_country compare = equals }
									check_variable = { var = cannibalism_inbound_spread_source_generation value = cannibalism_spread_source_generation compare = equals }
									OR = {
										NOT = { has_state_flag = cannibalism_spread_warning_event_open }
										AND = {
											has_variable = cannibalism_warning_expected_spread_id
											check_variable = { var = cannibalism_warning_expected_spread_id value = cannibalism_expected_spread_id compare = equals }
										}
									}
								}
							}
						}
					}
					set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
				}
				set_temp_variable = { cannibalism_spread_duplicate_index = cannibalism_spread_validation_index }
				add_to_temp_variable = { cannibalism_spread_duplicate_index = 1 }
				while_loop_effect = {
					limit = {
						check_variable = { cannibalism_spread_queue_validation_result > 0 }
						check_variable = {
							var = cannibalism_spread_duplicate_index
							value = global.cannibalism_spread_queue_id_entries^num
							compare = less_than
						}
					}
					if = {
						limit = {
							OR = {
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_duplicate_index = constant:cannibalism_spread_status.queued }
								check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_duplicate_index = constant:cannibalism_spread_status.in_transit }
							}
							var:global.cannibalism_spread_queue_target_state_entries^cannibalism_spread_duplicate_index = {
								state = event_target:cannibalism_spread_target_state
							}
						}
						set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
					}
					add_to_temp_variable = { cannibalism_spread_duplicate_index = 1 }
				}
			}
			else_if = {
				limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
				cannibalism_load_spread_entry_context = yes
				if = {
					limit = {
						check_variable = { cannibalism_spread_target_state_context_loaded > 0 }
						event_target:cannibalism_spread_target_state = {
							has_variable = cannibalism_inbound_spread_id
							check_variable = { var = cannibalism_inbound_spread_id value = cannibalism_expected_spread_id compare = equals }
						}
					}
					set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
				}
			}
			add_to_temp_variable = { cannibalism_spread_validation_index = 1 }
		}
	}
	if = {
		limit = { check_variable = { cannibalism_spread_queue_validation_result < 1 } }
		set_global_flag = cannibalism_spread_queue_invariant_failure
	}
}

# Normalized target-state scope is event_target:cannibalism_spread_target_state.
# This authoritative scan rejects a duplicate even if the state's cache flag is
# missing. A cache flag without a matching live row is exposed as corruption.
cannibalism_check_spread_target_state_availability = {
	set_temp_variable = { cannibalism_spread_target_available = 0 }
	set_temp_variable = { cannibalism_spread_target_live_row_count = 0 }
	cannibalism_validate_spread_queue = yes
	if = {
		limit = {
			check_variable = { cannibalism_spread_queue_validation_result > 0 }
			has_event_target = cannibalism_spread_target_state
		}
		set_temp_variable = { cannibalism_spread_queue_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_spread_queue_index
					value = global.cannibalism_spread_queue_status_entries^num
					compare = less_than
				}
			}
			if = {
				limit = {
					cannibalism_loaded_spread_status_is_live = yes
					var:global.cannibalism_spread_queue_target_state_entries^cannibalism_spread_queue_index = {
						state = event_target:cannibalism_spread_target_state
					}
				}
				add_to_temp_variable = { cannibalism_spread_target_live_row_count = 1 }
			}
			add_to_temp_variable = { cannibalism_spread_queue_index = 1 }
		}
		if = {
			limit = {
				check_variable = { cannibalism_spread_target_live_row_count < 1 }
				event_target:cannibalism_spread_target_state = {
					NOT = { has_state_flag = cannibalism_spread_inbound_active }
					NOT = { has_state_flag = cannibalism_spread_warning_event_open }
					NOT = { has_state_flag = cannibalism_spread_screening_active }
					NOT = { has_variable = cannibalism_inbound_spread_id }
					NOT = { has_variable = cannibalism_inbound_spread_route }
					NOT = { has_variable = cannibalism_inbound_spread_source_country }
					NOT = { has_variable = cannibalism_inbound_spread_source_generation }
				}
			}
			set_temp_variable = { cannibalism_spread_target_available = 1 }
		}
		else_if = {
			limit = {
				check_variable = { cannibalism_spread_target_live_row_count < 1 }
				event_target:cannibalism_spread_target_state = {
					OR = {
						has_state_flag = cannibalism_spread_inbound_active
						has_state_flag = cannibalism_spread_screening_active
						has_variable = cannibalism_inbound_spread_id
						has_variable = cannibalism_inbound_spread_route
						has_variable = cannibalism_inbound_spread_source_country
						has_variable = cannibalism_inbound_spread_source_generation
					}
				}
			}
			set_global_flag = cannibalism_spread_queue_invariant_failure
		}
	}
}

# Rebuilds every spread cache reachable from the canonical queue. It never runs
# against an invalid array shape or invalid live-row identity.
cannibalism_rebuild_spread_pending_ledgers = {
	cannibalism_validate_spread_queue = yes
	if = {
		limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
		for_each_scope_loop = {
			array = global.cannibalism_spread_queue_source_country_entries
			clear_variable = cannibalism_outbound_spread_count
			clr_country_flag = cannibalism_foreign_seeding_active
		}
		for_each_scope_loop = {
			array = global.cannibalism_spread_queue_target_country_entries
			clear_variable = cannibalism_inbound_spread_count
			clr_country_flag = cannibalism_foreign_route_warning_active
		}
		for_each_scope_loop = {
			array = global.cannibalism_spread_queue_target_state_entries
			clr_state_flag = cannibalism_spread_inbound_active
			clear_variable = cannibalism_inbound_spread_id
			clear_variable = cannibalism_inbound_spread_route
			clear_variable = cannibalism_inbound_spread_source_country
			clear_variable = cannibalism_inbound_spread_source_generation
		}
		set_variable = { global.cannibalism_pending_spread_count = 0 }
		set_temp_variable = { cannibalism_spread_queue_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_spread_queue_index
					value = global.cannibalism_spread_queue_status_entries^num
					compare = less_than
				}
			}
			if = {
				limit = { cannibalism_loaded_spread_status_is_live = yes }
				cannibalism_load_spread_entry_context = yes
				add_to_variable = { global.cannibalism_pending_spread_count = 1 }
				if = {
					limit = {
						check_variable = { cannibalism_spread_source_country_context_loaded > 0 }
						event_target:cannibalism_spread_source_country = {
							exists = yes
							has_variable = cannibalism_actor_generation
							check_variable = { var = cannibalism_actor_generation value = cannibalism_spread_source_generation compare = equals }
						}
					}
					event_target:cannibalism_spread_source_country = {
						add_to_variable = { cannibalism_outbound_spread_count = 1 }
						set_country_flag = cannibalism_foreign_seeding_active
					}
				}
				if = {
					limit = {
						check_variable = { cannibalism_spread_target_country_context_loaded > 0 }
						event_target:cannibalism_spread_target_country = { exists = yes }
					}
					event_target:cannibalism_spread_target_country = {
						add_to_variable = { cannibalism_inbound_spread_count = 1 }
						set_country_flag = cannibalism_foreign_route_warning_active
					}
				}
				if = {
					limit = { check_variable = { cannibalism_spread_target_state_context_loaded > 0 } }
					event_target:cannibalism_spread_target_state = {
						set_state_flag = cannibalism_spread_inbound_active
						set_variable = { cannibalism_inbound_spread_id = cannibalism_expected_spread_id }
						set_variable = { cannibalism_inbound_spread_route = cannibalism_spread_route }
						set_variable = { cannibalism_inbound_spread_source_country = event_target:cannibalism_spread_source_country }
						set_variable = { cannibalism_inbound_spread_source_generation = cannibalism_spread_source_generation }
					}
				}
			}
			add_to_temp_variable = { cannibalism_spread_queue_index = 1 }
		}
	}
}

cannibalism_enqueue_spread_entry = {
	set_temp_variable = { cannibalism_spread_enqueue_result = 0 }
	cannibalism_set_spread_parameters_from_route = yes
	cannibalism_check_spread_target_state_availability = yes
	if = {
		limit = {
			check_variable = { cannibalism_spread_parameters_valid > 0 }
			check_variable = { cannibalism_spread_queue_validation_result > 0 }
			check_variable = { cannibalism_spread_target_available > 0 }
			cannibalism_spread_context_is_valid = yes
		}
		add_to_variable = { global.cannibalism_next_spread_id = 1 }
		set_temp_variable = { cannibalism_spread_due_date = global.date }
		add_to_temp_variable = { cannibalism_spread_due_date = cannibalism_spread_delay_days }

		add_to_array = { array = global.cannibalism_spread_queue_id_entries value = global.cannibalism_next_spread_id }
		add_to_array = { array = global.cannibalism_spread_queue_source_country_entries value = event_target:cannibalism_spread_source_country }
		add_to_array = { array = global.cannibalism_spread_queue_source_state_entries value = event_target:cannibalism_spread_source_state }
		add_to_array = { array = global.cannibalism_spread_queue_target_country_entries value = event_target:cannibalism_spread_target_country }
		add_to_array = { array = global.cannibalism_spread_queue_target_state_entries value = event_target:cannibalism_spread_target_state }
		add_to_array = { array = global.cannibalism_spread_queue_route_entries value = cannibalism_spread_route }
		add_to_array = { array = global.cannibalism_spread_queue_source_generation_entries value = event_target:cannibalism_spread_source_country.cannibalism_actor_generation }
		add_to_array = { array = global.cannibalism_spread_queue_due_date_entries value = cannibalism_spread_due_date }
		add_to_array = { array = global.cannibalism_spread_queue_status_entries value = constant:cannibalism_spread_status.in_transit }

		event_target:cannibalism_spread_source_country = {
			add_to_variable = { cannibalism_outbound_spread_count = 1 }
			set_country_flag = cannibalism_foreign_seeding_active
		}
		event_target:cannibalism_spread_target_country = {
			add_to_variable = { cannibalism_inbound_spread_count = 1 }
			set_country_flag = cannibalism_foreign_route_warning_active
		}
		event_target:cannibalism_spread_target_state = {
			set_state_flag = cannibalism_spread_inbound_active
			set_variable = { cannibalism_inbound_spread_id = global.cannibalism_next_spread_id }
			set_variable = { cannibalism_inbound_spread_route = cannibalism_spread_route }
			set_variable = { cannibalism_inbound_spread_source_country = event_target:cannibalism_spread_source_country }
			set_variable = { cannibalism_inbound_spread_source_generation = event_target:cannibalism_spread_source_country.cannibalism_actor_generation }
		}
		add_to_variable = { global.cannibalism_pending_spread_count = 1 }
		set_temp_variable = { cannibalism_spread_enqueue_result = 1 }
		event_target:cannibalism_spread_target_country = {
			country_event = { id = chaosx.nr14.60 }
		}
	}
}

cannibalism_clear_loaded_spread_pending_state = {
	set_temp_variable = { cannibalism_spread_pending_clear_result = 0 }
	if = {
		limit = {
			check_variable = { cannibalism_spread_source_country_context_loaded > 0 }
			check_variable = { cannibalism_spread_target_state_context_loaded > 0 }
			event_target:cannibalism_spread_target_state = {
				has_state_flag = cannibalism_spread_inbound_active
				has_variable = cannibalism_inbound_spread_id
				has_variable = cannibalism_inbound_spread_route
				has_variable = cannibalism_inbound_spread_source_country
				has_variable = cannibalism_inbound_spread_source_generation
				check_variable = { var = cannibalism_inbound_spread_id value = cannibalism_expected_spread_id compare = equals }
				check_variable = { var = cannibalism_inbound_spread_route value = cannibalism_spread_route compare = equals }
				check_variable = { var = cannibalism_inbound_spread_source_country value = event_target:cannibalism_spread_source_country compare = equals }
				check_variable = { var = cannibalism_inbound_spread_source_generation value = cannibalism_spread_source_generation compare = equals }
			}
		}
		event_target:cannibalism_spread_target_state = {
			clr_state_flag = cannibalism_spread_inbound_active
			clr_state_flag = cannibalism_spread_screening_active
			clr_state_flag = cannibalism_spread_screening_humane
			clr_state_flag = cannibalism_spread_screening_hard
			clear_variable = cannibalism_inbound_spread_id
			clear_variable = cannibalism_inbound_spread_route
			clear_variable = cannibalism_inbound_spread_source_country
			clear_variable = cannibalism_inbound_spread_source_generation
			if = {
				limit = { check_variable = { cannibalism_spread_clear_warning_instance_on_invalidation > 0 } }
				clr_state_flag = cannibalism_spread_warning_event_open
				clear_variable = cannibalism_warning_expected_spread_id
			}
		}
		set_temp_variable = { cannibalism_spread_pending_clear_result = 1 }
	}
	else = {
		set_temp_variable = { cannibalism_spread_queue_validation_result = 0 }
		set_global_flag = cannibalism_spread_queue_invariant_failure
	}
	if = {
		limit = {
			check_variable = { cannibalism_spread_pending_clear_result > 0 }
			has_event_target = cannibalism_spread_source_country
			event_target:cannibalism_spread_source_country = {
				exists = yes
				has_variable = cannibalism_actor_generation
				check_variable = {
					var = cannibalism_actor_generation
					value = cannibalism_spread_source_generation
					compare = equals
				}
			}
		}
		event_target:cannibalism_spread_source_country = {
			subtract_from_variable = { cannibalism_outbound_spread_count = 1 }
			if = {
				limit = { check_variable = { cannibalism_outbound_spread_count < 1 } }
				clear_variable = cannibalism_outbound_spread_count
				clr_country_flag = cannibalism_foreign_seeding_active
			}
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_spread_pending_clear_result > 0 }
			has_event_target = cannibalism_spread_target_country
			event_target:cannibalism_spread_target_country = { exists = yes }
		}
		event_target:cannibalism_spread_target_country = {
			subtract_from_variable = { cannibalism_inbound_spread_count = 1 }
			if = {
				limit = { check_variable = { cannibalism_inbound_spread_count < 1 } }
				clear_variable = cannibalism_inbound_spread_count
				clr_country_flag = cannibalism_foreign_route_warning_active
			}
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_spread_pending_clear_result > 0 }
			check_variable = { global.cannibalism_pending_spread_count > 0 }
		}
		subtract_from_variable = { global.cannibalism_pending_spread_count = 1 }
	}
}

# Inputs: cannibalism_spread_queue_index, cannibalism_expected_spread_id, and
# cannibalism_spread_terminal_status. The ID is re-read before status changes;
# status changes first, so a repeated call cannot clear any ledger twice.
cannibalism_finalize_loaded_spread_entry = {
	set_temp_variable = { cannibalism_spread_finalize_result = 0 }
	set_temp_variable = { cannibalism_spread_requested_id = cannibalism_expected_spread_id }
	if = {
		limit = {
			check_variable = { cannibalism_spread_queue_validation_result > 0 }
			check_variable = {
				var = cannibalism_spread_queue_index
				value = global.cannibalism_spread_queue_status_entries^num
				compare = less_than
			}
			check_variable = { cannibalism_spread_requested_id > 0 }
			check_variable = {
				var = global.cannibalism_spread_queue_id_entries^cannibalism_spread_queue_index
				value = cannibalism_spread_requested_id
				compare = equals
			}
			cannibalism_loaded_spread_status_is_live = yes
			OR = {
				check_variable = { cannibalism_spread_terminal_status = constant:cannibalism_spread_status.arrived }
				check_variable = { cannibalism_spread_terminal_status = constant:cannibalism_spread_status.contained }
				check_variable = { cannibalism_spread_terminal_status = constant:cannibalism_spread_status.invalidated }
			}
		}
		cannibalism_load_spread_entry_context = yes
		if = {
			limit = {
				check_variable = {
					var = cannibalism_expected_spread_id
					value = cannibalism_spread_requested_id
					compare = equals
				}
			}
			set_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_queue_index = cannibalism_spread_terminal_status }
			cannibalism_clear_loaded_spread_pending_state = yes
			if = {
				limit = { check_variable = { cannibalism_spread_pending_clear_result > 0 } }
				set_temp_variable = { cannibalism_spread_finalize_result = 1 }
			}
		}
	}
}

			check_variable = { cannibalism_spread_parameters_valid > 0 }

		}

			limit = {
				event_target:cannibalism_spread_target_state = { has_state_flag = cannibalism_spread_screening_active }

			set_temp_variable = { cannibalism_spread_terminal_status = constant:cannibalism_spread_status.contained }
		}
		else = {
			cannibalism_resolve_loaded_spread_arrival = yes
			if = {
				limit = { check_variable = { cannibalism_spread_arrival_applied > 0 } }
				set_temp_variable = { cannibalism_spread_terminal_status = constant:cannibalism_spread_status.arrived }
			}
		}
	}
	cannibalism_finalize_loaded_spread_entry = yes
}

cannibalism_process_spread_queue = {
	cannibalism_validate_spread_queue = yes
	if = {
		limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
		set_temp_variable = { cannibalism_spread_queue_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = { cannibalism_spread_queue_validation_result > 0 }
				check_variable = {
					var = cannibalism_spread_queue_index
					value = global.cannibalism_spread_queue_status_entries^num
					compare = less_than
				}
			}
			if = {
				limit = {
					cannibalism_loaded_spread_status_is_live = yes
					check_variable = {
						var = global.cannibalism_spread_queue_due_date_entries^cannibalism_spread_queue_index
						value = global.date
						compare = less_than_or_equals
					}
				}
				cannibalism_load_spread_entry_context = yes
				cannibalism_resolve_loaded_spread_entry = yes
			}
			add_to_temp_variable = { cannibalism_spread_queue_index = 1 }
		}
	}
}

cannibalism_compact_spread_queue = {
	cannibalism_validate_spread_queue = yes
	if = {
		limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
		set_temp_variable = { cannibalism_spread_compaction_index = global.cannibalism_spread_queue_status_entries^num }
		while_loop_effect = {
			limit = { check_variable = { cannibalism_spread_compaction_index > 0 } }
			subtract_from_temp_variable = { cannibalism_spread_compaction_index = 1 }
			set_temp_variable = { cannibalism_expected_spread_id = global.cannibalism_spread_queue_id_entries^cannibalism_spread_compaction_index }
			if = {
				limit = {
					check_variable = { cannibalism_expected_spread_id > 0 }
					check_variable = {
						var = global.cannibalism_spread_queue_id_entries^cannibalism_spread_compaction_index
						value = cannibalism_expected_spread_id
						compare = equals
					}
					OR = {
						check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_compaction_index = constant:cannibalism_spread_status.arrived }
						check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_compaction_index = constant:cannibalism_spread_status.contained }
						check_variable = { global.cannibalism_spread_queue_status_entries^cannibalism_spread_compaction_index = constant:cannibalism_spread_status.invalidated }
					}
				}
				remove_from_array = { array = global.cannibalism_spread_queue_id_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_source_country_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_source_state_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_target_country_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_target_state_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_route_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_source_generation_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_due_date_entries index = cannibalism_spread_compaction_index }
				remove_from_array = { array = global.cannibalism_spread_queue_status_entries index = cannibalism_spread_compaction_index }
			}
		}
	}
}

# New-controller country scope for on_state_control_changed. An ordinary
# country retaking a live node starts the full liberation-recovery
# sequence in that state. If another controlled state is available, returning
# personnel also create an occupation-turnover warning there; containment can
# then stop the rear-area route without erasing the liberated state's trauma.
cannibalism_handle_state_control_recovery = {
	set_temp_variable = { cannibalism_control_change_recovery_context_valid = 0 }
	if = {
		limit = {
			cannibalism_spread_target_country_is_valid = yes
			FROM.FROM = {
				has_state_flag = cannibalism_active_node
				NOT = { has_state_flag = cannibalism_recovery_active }
				has_variable = cannibalism_node_id
				has_variable = cannibalism_node_generation
				has_variable = cannibalism_node_source_country
				has_variable = cannibalism_node_source_country_generation
				var:cannibalism_node_source_country = {
					has_variable = cannibalism_actor_generation
					check_variable = {
						var = cannibalism_actor_generation
						value = PREV.cannibalism_node_source_country_generation
						compare = equals
					}
				}
			}
		}
		FROM.FROM = {
			save_event_target_as = cannibalism_control_change_state
			save_event_target_as = cannibalism_spread_source_state
			var:cannibalism_node_source_country = { save_event_target_as = cannibalism_control_change_node_source_country }
			set_temp_variable = { cannibalism_expected_node_id = cannibalism_node_id }
			set_temp_variable = { cannibalism_expected_node_generation = cannibalism_node_generation }
			set_temp_variable = { cannibalism_control_change_node_source_generation = cannibalism_node_source_country_generation }
			set_temp_variable = { cannibalism_control_change_was_warlord_state = 0 }
			if = {
				limit = { has_state_flag = cannibalism_warlord_state }
				set_temp_variable = { cannibalism_control_change_was_warlord_state = 1 }
			}
			cannibalism_begin_current_state_recovery = yes
			if = {
				limit = {
					check_variable = { cannibalism_control_change_was_warlord_state > 0 }
					check_variable = { cannibalism_recovery_start_allowed > 0 }
				}
				event_target:cannibalism_control_change_node_source_country = {
					subtract_from_variable = { cannibalism_warlord_slot_reference_count = 1 }
					clamp_variable = { var = cannibalism_warlord_slot_reference_count min = 0 max = constant:cannibalism_slot.maximum }
				}
				clr_state_flag = cannibalism_warlord_state
				clr_state_flag = cannibalism_commune_state
				clr_state_flag = cannibalism_network_node
				clear_variable = cannibalism_warlord_slot_index
				clear_variable = cannibalism_warlord_slot_generation
			}
		}
		if = {
			limit = { check_variable = { cannibalism_recovery_start_allowed > 0 } }
			set_temp_variable = { cannibalism_control_change_recovery_context_valid = 1 }
			cannibalism_record_current_country_as_defeat_contributor = yes
		}
	}
}

				add_to_variable = { cannibalism_larder = constant:cannibalism_warlord_contract.network_manipulation_larder_gain }
			}
			if = {
				limit = { has_country_flag = cannibalism_warlord_diverted_couriers_open }
				add_to_variable = { cannibalism_larder = constant:cannibalism_warlord_contract.diverted_courier_larder_gain }
			}
			if = {
				limit = {
					has_country_flag = cannibalism_warlord_officer_corruption_open
					has_event_target = cannibalism_spread_target_country
				}
				event_target:cannibalism_spread_target_country = {
					add_stability = constant:cannibalism_warlord_contract.officer_corruption_stability
				}
			}
			clamp_variable = { var = cannibalism_frenzy min = constant:cannibalism_consumption.frenzy_min max = constant:cannibalism_consumption.frenzy_max }
			clamp_variable = { var = cannibalism_larder min = constant:cannibalism_consumption.larder_min max = constant:cannibalism_consumption.larder_max }
			clamp_variable = { var = global.cannibalism_network_reach min = constant:cannibalism_network.min max = constant:cannibalism_network.max }
			set_temp_variable = { cannibalism_deliberate_seed_result = 1 }
		}
	}
}

# Country scope. Matches both sides of every live row and leaves terminal
# tombstones for the normal post-iteration compactor.
cannibalism_invalidate_spread_entries_for_current_country_lifecycle = {
	save_event_target_as = cannibalism_spread_lifecycle_country
	cannibalism_validate_spread_queue = yes
	if = {
		limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
		set_temp_variable = { cannibalism_spread_queue_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = { cannibalism_spread_queue_validation_result > 0 }
				check_variable = {
					var = cannibalism_spread_queue_index
					value = global.cannibalism_spread_queue_status_entries^num
					compare = less_than
				}
			}
			if = {
				limit = { cannibalism_loaded_spread_status_is_live = yes }
				set_temp_variable = { cannibalism_spread_lifecycle_row_matches = 0 }
				set_temp_variable = { cannibalism_spread_clear_warning_instance_on_invalidation = 0 }
				if = {
					limit = {
						var:global.cannibalism_spread_queue_source_country_entries^cannibalism_spread_queue_index = {
							tag = event_target:cannibalism_spread_lifecycle_country
						}
					}
					set_temp_variable = { cannibalism_spread_lifecycle_row_matches = 1 }
				}
				if = {
					limit = {
						var:global.cannibalism_spread_queue_target_country_entries^cannibalism_spread_queue_index = {
							tag = event_target:cannibalism_spread_lifecycle_country
						}
					}
					set_temp_variable = { cannibalism_spread_lifecycle_row_matches = 1 }
					set_temp_variable = { cannibalism_spread_clear_warning_instance_on_invalidation = 1 }
				}
				if = {
					limit = { check_variable = { cannibalism_spread_lifecycle_row_matches > 0 } }
					set_temp_variable = { cannibalism_expected_spread_id = global.cannibalism_spread_queue_id_entries^cannibalism_spread_queue_index }
					cannibalism_invalidate_loaded_spread_entry = yes
				}
			}
			add_to_temp_variable = { cannibalism_spread_queue_index = 1 }
		}
		cannibalism_rebuild_spread_pending_ledgers = yes
	}
}

# State scope. Used only by release/puppet lifecycle hooks to remove cache
# residue that has no authoritative live target row.

	save_event_target_as = cannibalism_spread_lifecycle_state
	set_temp_variable = { cannibalism_spread_state_live_row_count = 0 }
	cannibalism_validate_spread_queue = yes
	if = {
		limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
		set_temp_variable = { cannibalism_spread_queue_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_spread_queue_index
					value = global.cannibalism_spread_queue_status_entries^num
					compare = less_than
				}
			}
			if = {
				limit = {
					cannibalism_loaded_spread_status_is_live = yes
					var:global.cannibalism_spread_queue_target_state_entries^cannibalism_spread_queue_index = {
						state = event_target:cannibalism_spread_lifecycle_state
					}
				}
				add_to_temp_variable = { cannibalism_spread_state_live_row_count = 1 }
			}
			add_to_temp_variable = { cannibalism_spread_queue_index = 1 }
		}
		if = {
			limit = { check_variable = { cannibalism_spread_state_live_row_count < 1 } }
			clr_state_flag = cannibalism_spread_inbound_active
			clr_state_flag = cannibalism_spread_screening_active

			clr_state_flag = cannibalism_spread_screening_hard

			clear_variable = cannibalism_inbound_spread_route
			clear_variable = cannibalism_inbound_spread_source_country
			clear_variable = cannibalism_inbound_spread_source_generation
			clear_variable = cannibalism_warning_expected_spread_id
		}
	}
}

				NOT = { has_variable = cannibalism_inbound_spread_count }
				NOT = { has_variable = cannibalism_outbound_spread_count }
			}
			clr_country_flag = cannibalism_foreign_route_warning_active
			clr_country_flag = cannibalism_foreign_seeding_active
		}
	}
}

# Contract for the unification owner:
# - regular target cannibalism_runtime_migration_source_country = old actor;
# - regular target cannibalism_runtime_migration_destination_country = host;
# - destination has a positive current cannibalism_actor_generation.
# The helper updates every live source country/generation and target country
# reference, plus exact target-state source metadata, then rebuilds all caches.
cannibalism_migrate_spread_entries_between_countries = {
	set_temp_variable = { cannibalism_spread_migration_result = 0 }
	cannibalism_validate_spread_queue = yes
	if = {
		limit = {
			check_variable = { cannibalism_spread_queue_validation_result > 0 }
			has_event_target = cannibalism_runtime_migration_source_country
			has_event_target = cannibalism_runtime_migration_destination_country
			event_target:cannibalism_runtime_migration_source_country = { exists = yes }
			event_target:cannibalism_runtime_migration_destination_country = {
				exists = yes
				has_variable = cannibalism_actor_generation
				check_variable = { cannibalism_actor_generation > 0 }
				NOT = { tag = event_target:cannibalism_runtime_migration_source_country }
			}
		}
		set_temp_variable = { cannibalism_spread_migration_destination_generation = event_target:cannibalism_runtime_migration_destination_country.cannibalism_actor_generation }
		event_target:cannibalism_runtime_migration_source_country = {
			clear_variable = cannibalism_outbound_spread_count
			clear_variable = cannibalism_inbound_spread_count
			clr_country_flag = cannibalism_foreign_seeding_active
			clr_country_flag = cannibalism_foreign_route_warning_active
		}
		set_temp_variable = { cannibalism_spread_queue_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_spread_queue_index
					value = global.cannibalism_spread_queue_status_entries^num
					compare = less_than
				}
			}
			if = {
				limit = { cannibalism_loaded_spread_status_is_live = yes }
				set_temp_variable = { cannibalism_spread_migration_requested_id = global.cannibalism_spread_queue_id_entries^cannibalism_spread_queue_index }
				cannibalism_load_spread_entry_context = yes
				set_temp_variable = { cannibalism_spread_migration_source_match = 0 }
				set_temp_variable = { cannibalism_spread_migration_target_match = 0 }
				if = {
					limit = {
						var:global.cannibalism_spread_queue_source_country_entries^cannibalism_spread_queue_index = {
							tag = event_target:cannibalism_runtime_migration_source_country
						}
					}
					set_temp_variable = { cannibalism_spread_migration_source_match = 1 }
				}
				if = {
					limit = {
						var:global.cannibalism_spread_queue_target_country_entries^cannibalism_spread_queue_index = {
							tag = event_target:cannibalism_runtime_migration_source_country
						}
					}
					set_temp_variable = { cannibalism_spread_migration_target_match = 1 }
				}
				if = {
					limit = {
						check_variable = { cannibalism_spread_migration_source_match > 0 }
						check_variable = { var = cannibalism_expected_spread_id value = cannibalism_spread_migration_requested_id compare = equals }
					}
					event_target:cannibalism_spread_target_state = {
						set_variable = { cannibalism_inbound_spread_source_country = event_target:cannibalism_runtime_migration_destination_country }
						set_variable = { cannibalism_inbound_spread_source_generation = cannibalism_spread_migration_destination_generation }
					}
					set_variable = { global.cannibalism_spread_queue_source_country_entries^cannibalism_spread_queue_index = event_target:cannibalism_runtime_migration_destination_country }
					set_variable = { global.cannibalism_spread_queue_source_generation_entries^cannibalism_spread_queue_index = cannibalism_spread_migration_destination_generation }
				}
				if = {
					limit = {
						check_variable = { cannibalism_spread_migration_target_match > 0 }
						check_variable = { var = cannibalism_expected_spread_id value = cannibalism_spread_migration_requested_id compare = equals }
					}
					set_variable = { global.cannibalism_spread_queue_target_country_entries^cannibalism_spread_queue_index = event_target:cannibalism_runtime_migration_destination_country }
				}
			}
			add_to_temp_variable = { cannibalism_spread_queue_index = 1 }
		}
		cannibalism_rebuild_spread_pending_ledgers = yes
		if = {
			limit = { check_variable = { cannibalism_spread_queue_validation_result > 0 } }
			set_temp_variable = { cannibalism_spread_migration_result = 1 }
		}
	}
}
