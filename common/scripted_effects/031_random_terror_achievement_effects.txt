random_terror_achievement_initialize_country = {
	if = {
		limit = { NOT = { has_country_flag = random_terror_achievement_ledger_initialized } }
		set_country_flag = random_terror_achievement_ledger_initialized
		set_variable = { random_terror_achievement_automatic_firings = constant:random_terror_value.zero }
		set_variable = { random_terror_achievement_cleared_firings = constant:random_terror_value.zero }
		set_variable = { random_terror_achievement_minimum_legitimacy = constant:random_terror_value.maximum }
		set_variable = { random_terror_achievement_movement_days = constant:random_terror_value.zero }
		set_variable = { random_terror_achievement_corridors_broken = constant:random_terror_value.zero }
		set_variable = { random_terror_achievement_actor_defeats = constant:random_terror_value.zero }

}

			}
		}

}

random_terror_achievement_record_incident = {
	random_terror_achievement_initialize_country = yes
	if = {
		limit = {
			OR = {
				has_global_flag = random_terror_global_jihad_setup_in_progress
				has_global_flag = random_terror_scenario_setup_bypass
			}
		}
		set_country_flag = random_terror_achievement_manual_scenario_disqualified
		set_country_flag = random_terror_achievement_short_crisis_disqualified
		set_country_flag = random_terror_achievement_long_watch_disqualified
	}
	if = {
		limit = { has_global_flag = random_terror_automatic_wave_active }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = random_terror_achievement_last_automatic_wave }
					NOT = {
						check_variable = {
							var = random_terror_achievement_last_automatic_wave
							value = global.random_terror_automatic_wave_id
							compare = equals
						}
					}
				}
			}
			set_variable = { random_terror_achievement_last_automatic_wave = global.random_terror_automatic_wave_id }
			if = {
				limit = { has_country_flag = random_terror_achievement_short_crisis_tracking }
				set_country_flag = random_terror_achievement_short_crisis_disqualified
			}
			if = {
				limit = { has_country_flag = random_terror_achievement_automatic_crisis_open }
				set_country_flag = random_terror_achievement_long_watch_disqualified
			}
			add_to_variable = { random_terror_achievement_automatic_firings = constant:random_terror_value.one }
			set_country_flag = random_terror_achievement_automatic_crisis_open
			set_variable = { random_terror_achievement_network_reach_start = global.random_terror_network_reach }
		}
		random_terror_achievement_count_active_states = yes
		if = {
			limit = {
				NOT = { has_country_flag = random_terror_achievement_short_crisis_tracking }
				NOT = { has_country_flag = random_terror_achievement_short_crisis_disqualified }
				check_variable = { var = random_terror_pressure value = constant:random_terror_tuning.achievement_crisis_pressure_gate compare = greater_than_or_equals }
				check_variable = { var = random_terror_achievement_active_states value = constant:random_terror_tuning.achievement_crisis_active_state_gate compare = greater_than_or_equals }
			}
			random_terror_achievement_begin_short_crisis = yes
		}
	}
	else = {
		set_country_flag = random_terror_achievement_force_trigger_disqualified
		set_country_flag = random_terror_achievement_short_crisis_disqualified
		set_country_flag = random_terror_achievement_long_watch_disqualified
	}
	if = {
		limit = {
			has_event_target = random_terror_current_state
			event_target:random_terror_current_state = { has_state_flag = random_terror_cannibal_clash_recorded }
		}
		set_country_flag = random_terror_achievement_event14_rivalry_proof
	}
}

random_terror_achievement_finalize_country = {
	# No Second Blast.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_short_crisis_tracking
			NOT = { has_country_flag = random_terror_achievement_short_crisis_disqualified }
			NOT = { has_country_flag = random_terror_achievement_actor_formed_against_country }
			NOT = { has_country_flag = random_terror_achievement_government_civilian_death }
			NOT = { has_country_flag = random_terror_achievement_abusive_failure }
			check_variable = { var = global.date value = random_terror_achievement_short_crisis_deadline compare = less_than_or_equals }
			check_variable = { random_terror_pressure = constant:random_terror_value.zero }
			NOT = { random_terror_country_has_active_state = yes }
		}
		set_country_flag = random_terror_achievement_no_second_blast
		clr_country_flag = random_terror_achievement_short_crisis_tracking
		every_owned_state = { clr_state_flag = random_terror_achievement_short_crisis_initial_state }
	}

	# The Long Watch.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_automatic_crisis_open
			check_variable = { random_terror_pressure = constant:random_terror_value.zero }
			NOT = { random_terror_country_has_active_state = yes }
		}
		clr_country_flag = random_terror_achievement_automatic_crisis_open
		add_to_variable = { random_terror_achievement_cleared_firings = constant:random_terror_value.one }
	}
	if = {
		limit = {
			check_variable = { var = random_terror_achievement_automatic_firings value = constant:random_terror_tuning.achievement_long_watch_firings compare = greater_than_or_equals }
			check_variable = { var = random_terror_achievement_cleared_firings value = constant:random_terror_tuning.achievement_long_watch_firings compare = greater_than_or_equals }
			check_variable = { random_terror_pressure = constant:random_terror_value.zero }
			NOT = { has_country_flag = random_terror_achievement_long_watch_disqualified }
			NOT = { has_country_flag = random_terror_achievement_state_lost_to_actor }
			NOT = { has_country_flag = random_terror_achievement_successful_coup }
		}
		set_country_flag = random_terror_achievement_the_long_watch
	}

	# The City Still Stands.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_city_mission_succeeded
			check_variable = { var = global.date value = random_terror_achievement_city_recovery_deadline compare = less_than_or_equals }
			check_variable = { random_terror_pressure < constant:random_terror_tuning.achievement_city_pressure_recovery }
			capital_scope = { has_state_flag = random_terror_achievement_original_capital }
			NOT = { has_country_flag = random_terror_achievement_city_disqualified }
		}
		set_country_flag = random_terror_achievement_the_city_still_stands
	}

	# Cut Every Route.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_route_crisis_tracking
			check_variable = { var = random_terror_achievement_corridors_broken value = constant:random_terror_tuning.achievement_network_corridors compare = greater_than_or_equals }
			has_country_flag = random_terror_achievement_sponsor_neutralized
			has_country_flag = random_terror_achievement_safe_haven_eliminated
			check_variable = {
				var = global.random_terror_network_reach
				value = random_terror_achievement_network_reach_target
				compare = less_than_or_equals
			}
			NOT = { random_terror_country_has_active_state = yes }
			NOT = { has_country_flag = random_terror_achievement_route_cleanup_only }
		}
		set_country_flag = random_terror_achievement_cut_every_route
	}

	# The False Claim Rejected.
	if = {
		limit = {
			has_global_flag = random_terror_evolution_4_unlocked
			has_country_flag = random_terror_muslim_majority_reaction_registry
			has_country_flag = random_terror_muslim_government_opposition_recorded
			has_country_flag = random_terror_achievement_public_rejection_complete
			check_variable = { var = random_terror_response_legitimacy value = constant:random_terror_tuning.achievement_legitimacy_no_collective compare = greater_than_or_equals }
			NOT = {
				any_owned_state = {
					OR = {
						has_state_flag = random_terror_community_rejection_recorded
						has_state_flag = random_terror_muslim_community_opposition_recorded
					}
					NOT = { has_state_flag = random_terror_achievement_community_objective_protected }
				}
			}
			has_country_flag = random_terror_achievement_jihadist_leader_defeated
			is_subject = no
			NOT = { has_country_flag = random_terror_achievement_joined_jihadists }
			NOT = { has_country_flag = random_terror_achievement_hostile_cooperation }
			NOT = { has_country_flag = random_terror_achievement_collective_punishment }
			NOT = { has_country_flag = random_terror_government_takeover_recorded }
		}
		set_country_flag = random_terror_achievement_false_claim_rejected
	}

	# No Collective Punishment.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_event31_material_contribution
			has_country_flag = random_terror_achievement_actor_war_resolved
			NOT = { has_country_flag = random_terror_achievement_no_collective_disqualified }
			NOT = { has_country_flag = random_terror_achievement_abusive_failure }
			NOT = { has_country_flag = random_terror_achievement_atrocity_or_coverup }
			NOT = {
				any_owned_state = {
					has_state_flag = random_terror_achievement_state_lost_to_actor
					NOT = { has_state_flag = random_terror_civil_authority_restored }
				}
			}
		}
		set_country_flag = random_terror_achievement_no_collective_punishment
	}

	# The Enemy of My Enemy Is Still My Enemy.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_event31_material_contribution
			has_country_flag = random_terror_achievement_event14_material_contribution
			has_country_flag = random_terror_achievement_event14_rivalry_proof
			NOT = { has_country_flag = random_terror_achievement_hostile_cooperation }
		}
		set_country_flag = random_terror_achievement_enemy_of_my_enemy
	}

	# Fracture from Within.
	if = {
		limit = {
			is_ai = no
			has_global_flag = random_terror_evolution_4_unlocked
			has_country_flag = random_terror_achievement_actor_jihad_route
			has_country_flag = random_terror_achievement_actor_rejected_leader
			has_country_flag = random_terror_achievement_fracture_proven
			is_subject = no
			NOT = { has_global_flag = world_end }
			NOT = { has_country_flag = random_terror_final_actor }
			NOT = { has_country_flag = random_terror_achievement_submitted_to_final }
		}
		set_country_flag = random_terror_achievement_fracture_from_within
	}

	# Maximum Survivor.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_maximum_player
			has_country_flag = random_terror_achievement_maximum_ui_launch
			check_variable = { random_terror_achievement_maximum_launch_intensity = constant:random_terror_intensity.maximum }
			check_variable = { random_terror_pressure = constant:random_terror_value.zero }
			NOT = { has_country_flag = random_terror_achievement_maximum_disqualified }
			NOT = { random_terror_has_unresolved_actor = yes }
			has_global_flag = random_terror_defeat_resolved
			OR = {
				NOT = { has_country_flag = random_terror_achievement_original_capital_lost }
				has_country_flag = random_terror_achievement_original_capital_restored
			}
		}
		set_country_flag = random_terror_achievement_maximum_survivor
	}

	# The False Revelation Denied.
	if = {
		limit = {
			has_global_flag = random_terror_false_revelation_denied
			has_global_flag = random_terror_defeat_resolved
			has_country_flag = random_terror_achievement_terminal_command_capital
			has_country_flag = random_terror_achievement_terminal_corridor
			has_country_flag = random_terror_achievement_terminal_restoration
			has_country_flag = random_terror_achievement_terminal_final_defeat
			has_country_flag = random_terror_achievement_terminal_material_contribution
			NOT = { has_country_flag = random_terror_achievement_false_revelation_disqualified }
			NOT = { has_capitulated = yes }
		}
		set_country_flag = random_terror_achievement_false_revelation_denied
	}

	# Victims Before Victory.
	if = {
		limit = {
			has_country_flag = random_terror_major_civilian_loss_recorded
			has_country_flag = random_terror_achievement_actor_war_resolved
			check_variable = { var = random_terror_response_legitimacy value = constant:random_terror_tuning.achievement_legitimacy_victims compare = greater_than_or_equals }
			NOT = { has_country_flag = random_terror_achievement_victims_disqualified }
			NOT = {
				any_owned_state = {
					has_state_flag = random_terror_major_civilian_loss_recorded
					NOT = { has_state_flag = random_terror_achievement_victim_chain_complete }
				}
			}
		}
		set_country_flag = random_terror_achievement_victims_before_victory
	}

	# War Without a Capital.
	if = {
		limit = {
			has_country_flag = random_terror_achievement_original_capital_lost
			has_country_flag = random_terror_achievement_backup_capital_established
			has_country_flag = random_terror_achievement_continued_capital_war
			has_country_flag = random_terror_achievement_original_capital_retaken
			has_country_flag = random_terror_achievement_original_capital_restored
			check_variable = { var = global.date value = random_terror_achievement_capital_restore_deadline compare = less_than_or_equals }
			check_variable = { random_terror_pressure < constant:random_terror_tuning.achievement_capital_pressure_recovery }
			NOT = { has_country_flag = random_terror_achievement_war_without_capital_disqualified }
			NOT = { has_capitulated = yes }
		}
		set_country_flag = random_terror_achievement_war_without_a_capital
	}
}

random_terror_achievement_daily_tick = {
	if = {
		limit = {
			OR = {
				has_country_flag = random_terror_affected_country
				has_country_flag = random_terror_actor
				has_country_flag = random_terror_achievement_ledger_initialized
			}
		}
		random_terror_achievement_initialize_country = yes
		random_terror_achievement_record_legitimacy = yes
		if = {
			limit = { has_country_flag = random_terror_movement_restrictions_active }
			add_to_variable = { random_terror_achievement_movement_days = constant:random_terror_value.one }
			if = {
				limit = { check_variable = { random_terror_achievement_movement_days > constant:random_terror_tuning.achievement_movement_restriction_days } }
				set_country_flag = random_terror_achievement_no_collective_disqualified
			}
		}
		else_if = {
			limit = { has_country_flag = random_terror_achievement_movement_tracking }
			clr_country_flag = random_terror_achievement_movement_tracking
			set_variable = { random_terror_achievement_movement_days = constant:random_terror_value.zero }
		}
		if = {
			limit = {
				NOT = { has_country_flag = random_terror_actor }
				any_country = {
					random_terror_is_actor = yes
					has_war_with = ROOT
				}
			}
			set_country_flag = random_terror_achievement_actor_war_tracked
		}
		if = {
			limit = {
				has_country_flag = random_terror_achievement_original_capital_lost
				NOT = { has_country_flag = random_terror_achievement_backup_capital_established }
				NOT = { has_capitulated = yes }
				capital_scope = {
					is_controlled_by = ROOT
					NOT = { has_state_flag = random_terror_achievement_original_capital }
				}
			}
			set_country_flag = random_terror_achievement_backup_capital_established
			set_country_flag = random_terror_achievement_continued_capital_war
		}
		if = {
			limit = {
				has_country_flag = random_terror_achievement_short_crisis_tracking
				check_variable = { var = global.date value = random_terror_achievement_short_crisis_deadline compare = greater_than }
			}
			set_country_flag = random_terror_achievement_short_crisis_disqualified
			clr_country_flag = random_terror_achievement_short_crisis_tracking
		}
		if = {
			limit = {
				has_country_flag = random_terror_achievement_city_mission_succeeded
				check_variable = { var = global.date value = random_terror_achievement_city_recovery_deadline compare = greater_than }
			}
			set_country_flag = random_terror_achievement_city_disqualified
		}
		if = {
			limit = {
				has_country_flag = random_terror_achievement_city_tracking
				NOT = { capital_scope = { has_state_flag = random_terror_achievement_original_capital } }
			}
			set_country_flag = random_terror_achievement_city_disqualified
		}
		if = {
			limit = {
				has_country_flag = random_terror_achievement_event14_rivalry_proof
				has_country_flag = cannibalism_global_defeat_contributor
				NOT = { has_country_flag = random_terror_achievement_event14_material_contribution }
			}
			random_terror_achievement_record_event14_actor_defeat_contribution = yes
		}
		random_terror_achievement_finalize_country = yes
	}
}
