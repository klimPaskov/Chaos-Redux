# ============================================================================
# CHAOS REDUX - EVENT 020 EVOLUTION RUNTIME EFFECTS
# ============================================================================
# Overview:
# - Owns the five true Black Plague evolutions and their Event Log rows.
# - Keeps ordinary disease phases separate from evolution history.
# - Maintains the Rat King terminal-progress contract without granting the
#   terminal route until every explicit gate is satisfied.
# - Is called from the existing Event 020 pulse; no whole-world on_action is
#   introduced.
# ============================================================================

black_plague_evolution_initialize_terminal_targets = {
	clear_global_event_target = black_plague_rat_evolution_actor
	set_variable = { global.black_plague_terminal_capital_target_count = constant:black_plague_value.zero }
	set_variable = { global.black_plague_terminal_refuge_target_count = constant:black_plague_value.zero }
	every_state = {
		clr_state_flag = black_plague_rat_terminal_capital_target
		clr_state_flag = black_plague_rat_terminal_refuge_node
		if = {
			limit = {
				black_plague_state_can_receive_exposure = yes
				is_capital = yes
			}
			set_state_flag = black_plague_rat_terminal_capital_target
			add_to_variable = { global.black_plague_terminal_capital_target_count = constant:black_plague_value.one }
		}
		if = {
			limit = {
				black_plague_state_can_receive_exposure = yes
				NOT = { is_capital = yes }
				OR = {
					has_state_category = large_city
					has_state_category = metropolis
					has_state_category = megalopolis
				}
				OR = {
					is_coastal = yes
					black_plague_state_has_strategic_transport = yes
				}
			}
			set_state_flag = black_plague_rat_terminal_refuge_node
			add_to_variable = { global.black_plague_terminal_refuge_target_count = constant:black_plague_value.one }
		}
	}
}

black_plague_evolution_select_actor = {
	if = {
		limit = {
			has_global_flag = black_plague_triggerable_scenario_bootstrap
			check_variable = {
				var = black_plague_evolution_stage
				value = constant:black_plague_evolution_runtime.evolution_i_stage
				compare = equals
			}
		}
		# SCN-012 has no single origin owner. Leave Evolution I's actor empty
		# rather than attributing the strain milestone to an arbitrary human host.
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					var = black_plague_evolution_stage
					value = constant:black_plague_evolution_runtime.evolution_i_stage
					compare = equals
				}
			}
			has_event_target = black_plague_rat_evolution_actor
			event_target:black_plague_rat_evolution_actor = { exists = yes }
		}
		event_target:black_plague_rat_evolution_actor = { save_event_target_as = black_plague_evolution_actor }
	}
	else_if = {
		limit = {
			has_event_target = black_plague_origin_controller
			event_target:black_plague_origin_controller = { exists = yes }
		}
		event_target:black_plague_origin_controller = { save_event_target_as = black_plague_evolution_actor }
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = {
					var = black_plague_evolution_stage
					value = constant:black_plague_evolution_runtime.evolution_i_stage
					compare = equals
				}
			}
			RTX = { exists = yes }
		}
		RTX = { save_event_target_as = black_plague_evolution_actor }
	}
	else = {
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = black_plague_evolution_stage
						value = constant:black_plague_evolution_runtime.evolution_i_stage
						compare = equals
					}
				}
			}
			every_country = {
				limit = { black_plague_country_is_human_host = yes }
				save_event_target_as = black_plague_evolution_actor
			}
		}
	}
}

black_plague_evolution_record_stage = {
	black_plague_evolution_select_actor = yes
	set_temp_variable = { events_log_evolution_event_id = constant:black_plague_identity.event_id }
	set_temp_variable = { events_log_evolution_type = constant:black_plague_identity.evolution_type }
	set_temp_variable = { events_log_evolution_stage = black_plague_evolution_stage }
	set_temp_variable = { events_log_evolution_tier = black_plague_evolution_stage }
	if = {
		limit = {
			has_global_flag = black_plague_triggerable_scenario_bootstrap
			check_variable = {
				var = black_plague_evolution_stage
				value = constant:black_plague_evolution_runtime.evolution_i_stage
				compare = equals
			}
		}
		set_temp_variable = { events_log_evolution_has_actor = constant:black_plague_value.zero }
	}
	else_if = {
		limit = { has_event_target = black_plague_evolution_actor }
		event_target:black_plague_evolution_actor = { save_event_target_as = events_log_evolution_actor }
		set_temp_variable = { events_log_evolution_has_actor = constant:black_plague_value.one }
	}
	else = {
		set_temp_variable = { events_log_evolution_has_actor = constant:black_plague_value.zero }
	}
	# Evolution I in SCN-012 is deliberately an originless strain milestone,
	# but every later row must carry a verified country actor. Do not write a
	# misleading actorless row when a natural destination or rat carrier could
	# not be resolved.
	if = {
		limit = {
			OR = {
				check_variable = {
					var = black_plague_evolution_stage
					value = constant:black_plague_evolution_runtime.evolution_i_stage
					compare = equals
				}
				check_variable = {
					var = events_log_evolution_has_actor
					value = constant:black_plague_value.one
					compare = equals
				}
			}
		}
		record_events_log_evolution_entry = yes
	}
}

black_plague_activate_evolution_i = {
	if = {
		limit = {
			black_plague_evolution_i_is_enabled = yes
			NOT = { has_global_flag = black_plague_evolution_i_active }
		}
		set_global_flag = black_plague_evolution_i_active
		set_global_flag = black_plague_evolution_i_recorded
		set_global_flag = black_plague_strain_strengthened
		set_variable = { global.black_plague_evolution_i_date = global.date }
		set_variable = { global.black_plague_strain_generation = constant:black_plague_evolution_runtime.evolution_i_stage }
		set_temp_variable = { black_plague_evolution_stage = constant:black_plague_evolution_runtime.evolution_i_stage }
		black_plague_evolution_record_stage = yes
		if = { limit = { NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap } } ROOT = { news_event = { id = chaosx.nr20.20 } } }
	}
}

black_plague_activate_evolution_ii = {
	if = {
		limit = {
			black_plague_evolution_ii_is_enabled = yes
			has_global_flag = black_plague_evolution_i_active
			NOT = { has_global_flag = black_plague_evolution_ii_active }
		}
		set_global_flag = black_plague_evolution_ii_active
		set_global_flag = black_plague_overseas_spread_enabled
		set_variable = { global.black_plague_evolution_ii_date = global.date }
		set_variable = { global.black_plague_overseas_route_generation = constant:black_plague_evolution_runtime.evolution_ii_stage }
		set_temp_variable = { black_plague_evolution_stage = constant:black_plague_evolution_runtime.evolution_ii_stage }
		clear_global_event_target = black_plague_rat_evolution_actor
		clr_global_flag = black_plague_evolution_ii_actor_captured
		clr_global_flag = black_plague_evolution_ii_actor_pending
		every_state = {
			limit = {
				OR = {
					AND = {
						has_global_flag = black_plague_triggerable_scenario_bootstrap
						black_plague_state_is_established = yes
						black_plague_rat_state_has_dock_archetype_signal = yes
					}
					check_variable = {
						var = black_plague_last_exposure_route
						value = constant:black_plague_route.overseas_port
						compare = equals
					}
				}
				CONTROLLER = { black_plague_country_is_human_host = yes }
			}
			# The milestone belongs to the first verified overseas destination,
			# not to whichever qualifying port happens to be iterated last.
			if = {
				limit = { NOT = { has_global_flag = black_plague_evolution_ii_actor_captured } }
				CONTROLLER = { save_global_event_target_as = black_plague_rat_evolution_actor }
				set_global_flag = black_plague_evolution_ii_actor_captured
			}
		}
		if = {
			limit = { has_event_target = black_plague_rat_evolution_actor }
			set_global_flag = black_plague_evolution_ii_recorded
			black_plague_evolution_record_stage = yes
			if = { limit = { NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap } } ROOT = { news_event = { id = chaosx.nr20.30 } } }
		}
		else = {
			set_global_flag = black_plague_evolution_ii_actor_pending
		}
	}
}

black_plague_activate_evolution_iii = {
	if = {
		limit = {
			black_plague_evolution_iii_is_enabled = yes
			has_global_flag = black_plague_evolution_ii_active
			has_global_flag = black_plague_evolution_ii_recorded
			NOT = { has_global_flag = black_plague_evolution_iii_active }
		}
		black_plague_rat_activate_evolution_iii = yes
		if = {
			limit = { has_global_flag = black_plague_evolution_iii_active }
			# SCN-012 keeps all internal broods on the reusable RTA carrier, but the
			# first carrier is still the meaningful Evolution III actor. The rat
			# runtime selects that carrier before this wrapper records the row.
			set_temp_variable = { black_plague_evolution_stage = constant:black_plague_evolution_runtime.evolution_iii_stage }
			black_plague_evolution_record_stage = yes
			if = { limit = { NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap } } ROOT = { news_event = { id = chaosx.nr20.40 } } }
		}
	}
}

black_plague_activate_evolution_iv = {
	if = {
		limit = {
			black_plague_evolution_iv_is_enabled = yes
			has_global_flag = black_plague_evolution_iii_active
			NOT = { has_global_flag = black_plague_evolution_iv_active }
		}
		black_plague_rat_activate_evolution_iv = yes
		if = {
			limit = { has_global_flag = black_plague_evolution_iv_active }
			# The source RTA is only a transfer candidate. The evolution belongs
			# to the separate sentient country, including when an already-active
			# RTX is preserved by SCN-012.
			clear_global_event_target = black_plague_rat_evolution_actor
			if = {
				limit = { RTX = { black_plague_rat_country_is_king = yes } }
				RTX = { save_global_event_target_as = black_plague_rat_evolution_actor }
				set_temp_variable = { black_plague_evolution_stage = constant:black_plague_evolution_runtime.evolution_iv_stage }
				black_plague_evolution_record_stage = yes
				if = { limit = { NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap } } ROOT = { news_event = { id = chaosx.nr20.50 } } }
			}
			else = {
				# A stale global King flag must not consume Evolution IV without a
				# real RTX country to own the milestone.
				clr_global_flag = black_plague_evolution_iv_active
				clr_global_flag = black_plague_evolution_iv_recorded
				set_global_flag = black_plague_rat_king_activation_failed
			}
		}
	}
	else_if = {
		limit = {
			black_plague_evolution_iv_is_enabled = yes
			has_global_flag = black_plague_evolution_iv_active
			NOT = { has_global_flag = black_plague_rat_king_active }
			NOT = { has_global_flag = world_end }
		}
		# A non-terminal campaign may relaunch SCN-012 after the sentient King
		# was defeated. Reuse the existing Evolution IV row while rebuilding the
		# separate RTX country instead of trying to record a sixth evolution.
		black_plague_rat_activate_evolution_iv = yes
	}
}

				if = { limit = { OR = { black_plague_state_is_established = yes black_plague_state_is_rat_controlled = yes } } add_to_variable = { global.black_plague_rat_king_south_america_controlled = constant:black_plague_value.one } }

				limit = { is_on_continent = australia }

				if = { limit = { OR = { black_plague_state_is_established = yes black_plague_state_is_rat_controlled = yes } } add_to_variable = { global.black_plague_rat_king_australia_controlled = constant:black_plague_value.one } }

				limit = { is_controlled_by = RTX }
				add_to_variable = { global.black_plague_rat_king_controlled_states = constant:black_plague_value.one }
			}
			if = {
				limit = {
					is_controlled_by = RTX
					has_state_flag = black_plague_rat_terminal_capital_target

				add_to_variable = { global.black_plague_rat_king_capital_targets_controlled = constant:black_plague_value.one }

			limit = {
				check_variable = {
					var = global.black_plague_rat_king_controlled_states

		if = {
			limit = {
				OR = {
					check_variable = {
						var = global.black_plague_rat_king_best_continent_states
						value = global.black_plague_rat_king_peak_continent_states
						compare = greater_than

						check_variable = {
							var = global.black_plague_rat_king_best_continent_states
							value = global.black_plague_rat_king_peak_continent_states
							compare = equals
						}
						check_variable = {
							var = global.black_plague_rat_king_best_continent_ratio
							value = global.black_plague_rat_king_peak_continent_ratio
							compare = greater_than
						}
					}
				}
			}
			set_variable = { global.black_plague_rat_king_peak_continent_states = global.black_plague_rat_king_best_continent_states }
			set_variable = { global.black_plague_rat_king_peak_continent_ratio = global.black_plague_rat_king_best_continent_ratio }
		}
	}
}

black_plague_evolution_compare_king_continent = {
	set_temp_variable = { black_plague_king_scan_ratio = constant:black_plague_value.zero }
	if = {
		limit = { check_variable = { global.black_plague_rat_king_europe_total > constant:black_plague_value.zero } }
		set_temp_variable = { black_plague_king_scan_ratio = global.black_plague_rat_king_europe_controlled }
		divide_temp_variable = { black_plague_king_scan_ratio = global.black_plague_rat_king_europe_total }
		if = { limit = { check_variable = { var = black_plague_king_scan_ratio value = global.black_plague_rat_king_best_continent_ratio compare = greater_than } } set_variable = { global.black_plague_rat_king_best_continent_ratio = black_plague_king_scan_ratio } set_variable = { global.black_plague_rat_king_best_continent_states = global.black_plague_rat_king_europe_controlled } }
	}
	if = {
		limit = { check_variable = { global.black_plague_rat_king_asia_total > constant:black_plague_value.zero } }
		set_temp_variable = { black_plague_king_scan_ratio = global.black_plague_rat_king_asia_controlled }
		divide_temp_variable = { black_plague_king_scan_ratio = global.black_plague_rat_king_asia_total }
		if = { limit = { check_variable = { var = black_plague_king_scan_ratio value = global.black_plague_rat_king_best_continent_ratio compare = greater_than } } set_variable = { global.black_plague_rat_king_best_continent_ratio = black_plague_king_scan_ratio } set_variable = { global.black_plague_rat_king_best_continent_states = global.black_plague_rat_king_asia_controlled } }
	}
	if = {

# STATE scope. The terminal route consumes the remaining population of each
# human state exactly once before transfer. This deliberately uses the shared
# exact-loss contract so the ordinary deaths ledger and Chaos accounting stay
# synchronized with every other Black Plague mortality path.
black_plague_apply_terminal_takeover_loss_once = {
	if = {
		limit = {
			NOT = { has_state_flag = black_plague_terminal_takeover_death_recorded }
			check_variable = {
				var = state_population_k
				value = constant:black_plague_value.zero
				compare = greater_than
			}
		}
		set_temp_variable = { black_plague_terminal_loss_requested = state_population_k }
		multiply_temp_variable = { black_plague_terminal_loss_requested = constant:black_plague_value.people_per_thousand }
		set_temp_variable = { state_civilian_population_loss_requested = black_plague_terminal_loss_requested }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:black_plague_mortality.terminal_population_floor }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.black_plague }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:black_plague_value.one }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:black_plague_value.one }
		set_temp_variable = { state_civilian_population_loss_applied = constant:black_plague_value.zero }
		set_temp_variable = { state_civilian_population_loss_result = constant:black_plague_value.zero }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:black_plague_value.one }
		apply_exact_state_civilian_population_loss = yes
		if = {
			limit = { check_variable = { state_civilian_population_loss_applied > constant:black_plague_value.zero } }
			add_to_variable = { black_plague_deaths_applied = state_civilian_population_loss_applied }
			add_to_variable = { global.black_plague_deaths_total = state_civilian_population_loss_applied }
		}
		set_state_flag = black_plague_terminal_takeover_death_recorded
	}
}

# Rebuild the natural Evolution III basin count immediately before the

# Infected/Severe/Collapsed states that participate in a three-state path, so
# the fixed-point minimum remains a real tuning input instead of an unused
# design note. SCN-012's bootstrap path does not use this gate.

		}
		else_if = {
			limit = { black_plague_evolution_ii_is_ready = yes }
			black_plague_activate_evolution_ii = yes
			if = {
				limit = { has_global_flag = black_plague_evolution_ii_active }
				set_temp_variable = { black_plague_evolution_advanced_this_check = constant:black_plague_value.one }
			}
		}
		else_if = {
			limit = { black_plague_evolution_iii_is_ready = yes }
			black_plague_activate_evolution_iii = yes
			if = {
				limit = { has_global_flag = black_plague_evolution_iii_active }
				set_temp_variable = { black_plague_evolution_advanced_this_check = constant:black_plague_value.one }
			}
		}
		else_if = {
			limit = { black_plague_evolution_iv_is_ready = yes }
			black_plague_activate_evolution_iv = yes
			if = {
				limit = { has_global_flag = black_plague_evolution_iv_active }
				set_temp_variable = { black_plague_evolution_advanced_this_check = constant:black_plague_value.one }
			}
		}

			}

		}
		black_plague_rat_schedule_next_evolution_check = yes
	}
}
