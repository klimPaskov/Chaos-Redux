# ============================================================================
# CHAOS REDUX - CHAOS UI WEAR EFFECTS
# ============================================================================
# toggle_chaos_ui_wear flips the player setting that hides every overlay. The
# decals read the live chaos_tier global flag, so no tier hook is needed.
# See docs/systems/chaos_ui_wear.md.
# ============================================================================

toggle_chaos_ui_wear = {
	if = {
		limit = { has_global_flag = chaos_ui_wear_disabled }
		clr_global_flag = chaos_ui_wear_disabled
	}
	else = { set_global_flag = chaos_ui_wear_disabled }
}
