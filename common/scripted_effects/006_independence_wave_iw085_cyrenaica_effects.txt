	remove_ideas = {

independence_wave_iw085_cyrenaica_refresh_idea_lifecycle = {
	if = {
		limit = { is_independence_wave_iw085_cyrenaica_overlay_active = yes }
		independence_wave_iw085_cyrenaica_remove_overlay_ideas = yes
		if = {
			limit = { has_country_flag = independence_wave_iw085_cyrenaica_assembly_compact }
			add_ideas = independence_wave_iw085_cyrenaica_assembly_compact
		}
		else_if = {
			limit = { has_country_flag = independence_wave_iw085_cyrenaica_coastal_defense_charter }
			add_ideas = independence_wave_iw085_cyrenaica_coastal_defense_charter
		}
		else_if = {
			limit = {
				check_variable = {
					var = independence_wave_iw085_cyrenaica_desert_mobility
					value = constant:independence_wave_iw085_cyrenaica_value.settlement_threshold
					compare = greater_than_or_equals
				}
				check_variable = {
					var = independence_wave_iw085_cyrenaica_regency_legitimacy
					value = constant:independence_wave_iw085_cyrenaica_value.settlement_threshold
					compare = greater_than_or_equals
				}
			}
			add_ideas = independence_wave_iw085_cyrenaica_desert_regency
		}
		else = { add_ideas = independence_wave_iw085_cyrenaica_contested_regency }
	}
}

independence_wave_iw085_cyrenaica_initialize_overlay = {
	if = {
		limit = { is_independence_wave_iw085_cyrenaica_route_active = yes NOT = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_ever_activated } }
		set_country_flag = independence_wave_iw085_cyrenaica_overlay_ever_activated
		set_country_flag = independence_wave_iw085_cyrenaica_package_initialized
		set_country_flag = independence_wave_iw085_cyrenaica_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		set_variable = { independence_wave_iw085_cyrenaica_desert_mobility = constant:independence_wave_iw085_cyrenaica_value.desert_mobility_start }
		set_variable = { independence_wave_iw085_cyrenaica_regency_legitimacy = constant:independence_wave_iw085_cyrenaica_value.regency_legitimacy_start }
		set_variable = { independence_wave_iw085_cyrenaica_coastal_security = constant:independence_wave_iw085_cyrenaica_value.coastal_security_start }
		set_variable = { independence_wave_iw085_cyrenaica_guard_hold_days = constant:independence_wave_iw085_cyrenaica_value.minimum }
		set_variable = { independence_wave_anchor_state = 663 }
		set_variable = { independence_wave_package_archetype = constant:independence_wave_package_archetype.port_or_island }
		set_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.desert_nomadic }
		independence_wave_iw085_cyrenaica_refresh_idea_lifecycle = yes
	}
}

independence_wave_iw085_cyrenaica_suspend_overlay = {
	if = {
		limit = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_active }
		if = { limit = { has_country_flag = independence_wave_iw085_cyrenaica_guard_running NOT = { has_country_flag = independence_wave_iw085_cyrenaica_guard_completed } } set_country_flag = independence_wave_iw085_cyrenaica_guard_interrupted }
		independence_wave_iw085_cyrenaica_remove_overlay_ideas = yes
		clr_country_flag = independence_wave_iw085_cyrenaica_overlay_active
		clr_country_flag = independence_wave_route_overlay_active
		set_country_flag = independence_wave_iw085_cyrenaica_overlay_suspended
	}
}

independence_wave_iw085_cyrenaica_pause_guard_mission = {
	if = {
		limit = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_suspended has_country_flag = independence_wave_iw085_cyrenaica_guard_interrupted has_country_flag = independence_wave_iw085_cyrenaica_guard_running has_active_mission = independence_wave_iw085_hold_cyrenaica_coastal_guard }
		set_temp_variable = { independence_wave_iw085_cyrenaica_pause_days = constant:independence_wave_iw085_cyrenaica_duration.guard_pause_daily_extension }
		add_days_mission_timeout = { mission = independence_wave_iw085_hold_cyrenaica_coastal_guard days = independence_wave_iw085_cyrenaica_pause_days }
	}
}

independence_wave_iw085_cyrenaica_resume_overlay = {
	if = {
		limit = { is_independence_wave_iw085_cyrenaica_route_active = yes has_country_flag = independence_wave_iw085_cyrenaica_overlay_ever_activated NOT = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_active } }
		set_country_flag = independence_wave_iw085_cyrenaica_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw085_cyrenaica_overlay_suspended
		independence_wave_iw085_cyrenaica_refresh_idea_lifecycle = yes
	}
}

# Called only from on_daily_LBA. The exact LBA autonomy gate prevents a global
# loop and prevents an independent Cyrenaica package from being fabricated.
independence_wave_iw085_cyrenaica_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw085_cyrenaica_route_active = yes }
		if = { limit = { NOT = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_ever_activated } } independence_wave_iw085_cyrenaica_initialize_overlay = yes }
		else_if = { limit = { NOT = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_active } } independence_wave_iw085_cyrenaica_resume_overlay = yes }
		else = {
			if = {
				limit = { has_country_flag = independence_wave_iw085_cyrenaica_guard_interrupted }
				if = { limit = { has_country_flag = independence_wave_iw085_cyrenaica_guard_running has_active_mission = independence_wave_iw085_hold_cyrenaica_coastal_guard } clr_country_flag = independence_wave_iw085_cyrenaica_guard_interrupted }
				else = { independence_wave_iw085_cyrenaica_fail_guard_mission = yes }
			}
			else_if = { limit = { has_country_flag = independence_wave_iw085_cyrenaica_guard_running NOT = { has_active_mission = independence_wave_iw085_hold_cyrenaica_coastal_guard } } independence_wave_iw085_cyrenaica_fail_guard_mission = yes }
			independence_wave_iw085_cyrenaica_update_guard_hold = yes
			if = {
				limit = { NOT = { OR = { has_idea = independence_wave_iw085_cyrenaica_contested_regency has_idea = independence_wave_iw085_cyrenaica_desert_regency has_idea = independence_wave_iw085_cyrenaica_assembly_compact has_idea = independence_wave_iw085_cyrenaica_coastal_defense_charter } } }
				independence_wave_iw085_cyrenaica_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = { limit = { has_country_flag = independence_wave_iw085_cyrenaica_overlay_active } independence_wave_iw085_cyrenaica_suspend_overlay = yes }
		independence_wave_iw085_cyrenaica_pause_guard_mission = yes
	}
}
