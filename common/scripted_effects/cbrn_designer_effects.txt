# ============================================================================
# CHAOS REDUX - CBRN DESIGNER MODULE UNLOCK EFFECTS
# ============================================================================
# Overview:
# - Grants hidden technologies for exact chemical aircraft rack variants.
# - Country scope is required. Agent access and completed MIO traits are both
#   checked before a module technology is granted.
# - Trait on-complete effects call the matching one-direction helper so the
#   just-completed trait never depends on completion-trigger timing. Agent
#   technologies and special-project outputs resynchronize the other side.
# - Effects are idempotent, create no equipment, and never run from a periodic
#   all-country action.
# ============================================================================

cbrn_unlock_lightweight_aerosol_modules_for_available_agents = {
	if = {
		limit = {
			has_tech = chlorine_gas
			NOT = { has_tech = cbrn_air_rack_chlorine_lightweight_tech }
		}
		set_technology = { cbrn_air_rack_chlorine_lightweight_tech = 1 popup = no }

	if = {

	if = {

	if = {

		}

	}
	if = {

		set_technology = { cbrn_air_rack_tabun_lightweight_tech = 1 popup = no }
	}

		limit = {

cbrn_sync_aerosol_designer_module_unlocks = {
	if = {
		limit = { cbrn_designer_has_lightweight_payload_assemblies = yes }
		cbrn_unlock_lightweight_aerosol_modules_for_available_agents = yes
	}
	if = {
		limit = { cbrn_designer_has_long_range_payload = yes }
		cbrn_unlock_long_range_aerosol_modules_for_available_agents = yes
	}
}

cbrn_unlock_malodor_aerosol_modules_after_project = {
	if = {
		limit = { cbrn_designer_has_lightweight_payload_assemblies = yes }
		if = {
			limit = { NOT = { has_tech = cbrn_air_rack_malodor_lightweight_tech } }
			set_technology = { cbrn_air_rack_malodor_lightweight_tech = 1 popup = no }
		}
	}
	if = {
		limit = { cbrn_designer_has_long_range_payload = yes }
		if = {
			limit = { NOT = { has_tech = cbrn_air_rack_malodor_long_range_tech } }
			set_technology = { cbrn_air_rack_malodor_long_range_tech = 1 popup = no }
		}
	}
	if = {
		limit = {
			cbrn_designer_has_lightweight_payload_assemblies = yes
			cbrn_designer_has_long_range_payload = yes
		}
		if = {
			limit = { NOT = { has_tech = cbrn_air_rack_malodor_lightweight_long_range_tech } }
			set_technology = { cbrn_air_rack_malodor_lightweight_long_range_tech = 1 popup = no }
		}
	}
}
