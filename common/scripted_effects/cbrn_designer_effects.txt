# ============================================================================
# CHAOS REDUX - CBRN DESIGNER EFFECTS
# ============================================================================
# Overview:
# - Grants hidden technologies for exact chemical aircraft rack variants and
#   refreshes exact contaminated-state designer modifiers.
# - Country scope is required. Agent access and completed MIO traits are both
#   checked before a module technology is granted.
# - Trait on-complete effects call the matching one-direction helper so the
#   just-completed trait never depends on completion-trigger timing. Agent
#   technologies and special-project outputs resynchronize the other side.
# - Effects are idempotent, create no equipment, and never run from a periodic
#   all-country action.
# ============================================================================

cbrn_sync_aerosol_designer_module_unlocks = {
	if = {
		limit = { cbrn_designer_has_lightweight_payload_assemblies = yes }
		cbrn_unlock_lightweight_aerosol_modules_for_available_agents = yes
	}
	if = {
		limit = { cbrn_designer_has_long_range_payload = yes }
		cbrn_unlock_long_range_aerosol_modules_for_available_agents = yes
	}
}

cbrn_unlock_malodor_aerosol_modules_after_project = {
	if = {
		limit = { cbrn_designer_has_lightweight_payload_assemblies = yes }
		if = {
			limit = { NOT = { has_tech = cbrn_air_rack_malodor_lightweight_tech } }
			set_technology = { cbrn_air_rack_malodor_lightweight_tech = 1 popup = no }
		}
	}
	if = {
		limit = { cbrn_designer_has_long_range_payload = yes }
		if = {
			limit = { NOT = { has_tech = cbrn_air_rack_malodor_long_range_tech } }
			set_technology = { cbrn_air_rack_malodor_long_range_tech = 1 popup = no }
		}
	}

		limit = {
			cbrn_designer_has_lightweight_payload_assemblies = yes

		}
		if = {

# State scope. Recomputes only the legacy dynamic modifier's truck-attrition
# input from the exact contamination strength and the current controller's
# completed Vehicle Recovery Teams trait. It is called when contamination is
# applied and by the exact state-control-change hook, so no stale controller
# benefit survives conquest.
cbrn_designer_refresh_contaminated_state_vehicle_attrition = {
	if = {
		limit = { has_variable = cbrn_chemical_effective_strength }
		set_temp_variable = { cbrn_designer_truck_attrition = cbrn_chemical_effective_strength }
		multiply_temp_variable = { cbrn_designer_truck_attrition = constant:chem_state.mult.truck_attrition }
		if = {
			limit = { CONTROLLER = { cbrn_designer_has_vehicle_recovery_teams = yes } }
			multiply_temp_variable = { cbrn_designer_truck_attrition = constant:cbrn_decontamination_designer_effect.vehicle_recovery_attrition_mult }
		}
		set_variable = { cbrn_chemical_truck_attrition = cbrn_designer_truck_attrition }
		if = {
			limit = { has_dynamic_modifier = { modifier = chem_state_contamination } }
			force_update_dynamic_modifier = yes
		}
	}
}

# State scope. Applies the current controller's completed Burn Treatment trait
# only to continuing blister-agent deaths in the existing contamination death
# pass. Initial release deaths and non-blister contamination are unchanged.
cbrn_designer_apply_blister_continuing_death_treatment = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = chem_state_contamination }
			has_variable = cbrn_last_chemical_agent_class
			check_variable = { var = cbrn_last_chemical_agent_class value = constant:cbrn_agent_class.blister compare = equals }
			CONTROLLER = { cbrn_designer_has_burn_treatment = yes }
		}
		multiply_temp_variable = { chaos_state_deaths_mult = constant:cbrn_medical_designer_effect.burn_treatment_continuing_death_mult }
	}
}
