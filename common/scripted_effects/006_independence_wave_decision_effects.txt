# CHAOS REDUX - EVENT 006 INDEPENDENCE WAVE DECISION EFFECTS
# ============================================================================
# Overview:
# - Applies auditable five-value, former-host, patron, network, and league
#   transactions for DM-01 through DM-62.
# - Consumes real material resources through reusable cost palettes.
# - Publishes bounded state targets for integration and future-wave support.
# - Removes decision-layer missions and state when Event 006 origin ends.
# - Never creates a free repeatable unit or performs a periodic world scan.
# ============================================================================

# Country scope. At most one unconsumed defensive-charter war mandate is kept
# per member. Clearing it through the stored country scope prevents a later tag
# reuse from inheriting an authorization aimed at an earlier government.
independence_wave_decision_clear_charter_war_mandate = {
	if = {
		limit = { has_variable = independence_wave_last_charter_war_mandate_target }
		var:independence_wave_last_charter_war_mandate_target = {
			ROOT = { clr_country_flag = independence_wave_charter_war_authorized_@PREV }
		}
		clear_variable = independence_wave_last_charter_war_mandate_target
		clear_variable = independence_wave_last_charter_war_mandate_date
	}
}

# Country scope. DM-60 rechecks both leader authority and the stored accused
# country's factual ground before removing membership. The caller owns pointer
# cleanup after either success or failure.
independence_wave_decision_resolve_charter_expulsion_vote = {
	if = {
		limit = {
			is_independence_wave_charter_expulsion_authority = yes
			has_valid_independence_wave_active_league_decision_target = yes
			var:independence_wave_active_league_target_country = {
				is_valid_independence_wave_charter_expulsion_target = yes
			}
		}
		var:independence_wave_active_league_target_country = {
			independence_wave_expel_league_member = yes
		}
		independence_wave_achievement_record_member_expulsion = yes
		set_temp_variable = { independence_wave_decision_league_cohesion_delta = constant:independence_wave_decision_effect.league_expulsion_cohesion_loss }
		set_temp_variable = { independence_wave_decision_league_common_cause_delta = constant:independence_wave_decision_effect.league_expulsion_common_cause_loss }
		set_temp_variable = { independence_wave_decision_league_patron_capture_delta = constant:independence_wave_decision_effect.league_expulsion_patron_capture_loss }
		set_temp_variable = { independence_wave_decision_league_reserve_delta = constant:independence_wave_decision_effect.league_expulsion_reserve_loss }
		set_temp_variable = { independence_wave_decision_league_confidence_delta = constant:independence_wave_decision_effect.league_expulsion_confidence_loss }
		independence_wave_decision_apply_league_deltas = yes
		set_variable = { independence_wave_last_charter_expulsion_vote_date = global.date }
		independence_wave_enter_league_crisis = yes
	}
	else = { independence_wave_decision_fail_charter_expulsion_vote = yes }
}

# Creates the DM-22 emergency formations from the researched opening-force
# template at the persisted origin anchor. The decision pays the major security
# package first; the factors below deliberately keep the two formations
# understrength so their created manpower and equipment do not exceed that
# material commitment. This effect is called once per origin generation.
independence_wave_decision_raise_emergency_formations = {
	if = {
		limit = {
			has_country_flag = independence_wave_force_package_applied
			has_variable = independence_wave_anchor_state
		}
		set_temp_variable = { independence_wave_decision_emergency_experience = constant:independence_wave_decision_force.emergency_experience_factor }
		set_temp_variable = { independence_wave_decision_emergency_equipment = constant:independence_wave_decision_force.emergency_equipment_factor }
		set_temp_variable = { independence_wave_decision_emergency_manpower = constant:independence_wave_decision_force.emergency_manpower_factor }
		set_temp_variable = { independence_wave_decision_emergency_count = constant:independence_wave_decision_force.emergency_division_count }
		set_temp_variable = { independence_wave_decision_emergency_unit_id = constant:independence_wave_decision_force.emergency_unit_id }
		meta_effect = {
			text = {
				var:independence_wave_anchor_state = {
					create_unit = {
						division = "division_template = \"[IW_TEMPLATE]\" start_experience_factor = [IW_EXPERIENCE] start_equipment_factor = [IW_EQUIPMENT] start_manpower_factor = [IW_MANPOWER]"
						owner = ROOT
						allow_spawning_on_enemy_provs = no
						count = [IW_COUNT]
						id = [IW_ID]
					}
				}
			}
			IW_TEMPLATE = "[This.GetIndependenceWaveForceTemplateName]"
			IW_EXPERIENCE = "[?independence_wave_decision_emergency_experience|.2]"
			IW_EQUIPMENT = "[?independence_wave_decision_emergency_equipment|.2]"
			IW_MANPOWER = "[?independence_wave_decision_emergency_manpower|.2]"
			IW_COUNT = "[?independence_wave_decision_emergency_count|.0]"
			IW_ID = "[?independence_wave_decision_emergency_unit_id|.0]"
		}
	}
}

# Demobilizes only the formations created by DM-22 and refunds their remaining
# manpower and equipment. The fixed unit id is local to the owner and is never
# reused by another Event 006 action.
independence_wave_decision_demobilize_emergency_formations = {
	set_temp_variable = { independence_wave_decision_emergency_unit_id = constant:independence_wave_decision_force.emergency_unit_id }
	meta_effect = {
		text = {
			delete_unit = {
				id = [IW_ID]
				disband = yes
			}
		}

	}
	clr_country_flag = independence_wave_emergency_units_raised
	clr_country_flag = independence_wave_emergency_formations_active
}

# Input: temporary `independence_wave_decision_patron_reduction_delta`.
independence_wave_decision_reduce_all_patron_influence = {
	if = {
		limit = { independence_wave_patron_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_decision_patron_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = independence_wave_decision_patron_index
					value = independence_wave_patron_country_entries^num
					compare = less_than
				}
			}
			add_to_variable = {
				independence_wave_patron_influence_entries^independence_wave_decision_patron_index = independence_wave_decision_patron_reduction_delta
			}
			set_temp_variable = { independence_wave_patron_row = independence_wave_decision_patron_index }
			independence_wave_clamp_patron_row = yes
			add_to_temp_variable = { independence_wave_decision_patron_index = constant:independence_wave_plan.loop_increment }
		}
		independence_wave_refresh_strongest_patron = yes
		independence_wave_refresh_idea_lifecycles = yes
	}
}

# Input: FROM is the selected patron's capital state and the reduction delta is
# temporary. Only the matching current-generation row is changed.
independence_wave_decision_reduce_targeted_patron_influence = {
	FROM = { owner = { save_event_target_as = independence_wave_decision_selected_patron } }
	if = {
		limit = { independence_wave_patron_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_decision_patron_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = independence_wave_decision_patron_index
					value = independence_wave_patron_country_entries^num
					compare = less_than
				}
			}
			set_temp_variable = { independence_wave_decision_patron_generation = independence_wave_patron_owner_generation_entries^independence_wave_decision_patron_index }
			var:independence_wave_patron_country_entries^independence_wave_decision_patron_index = {
				if = {
					limit = {
						tag = event_target:independence_wave_decision_selected_patron
						check_variable = {
							var = independence_wave_decision_patron_generation
							value = ROOT.independence_wave_generation_id
							compare = equals
						}
					}
					ROOT = {
						add_to_variable = {
							independence_wave_patron_influence_entries^independence_wave_decision_patron_index = independence_wave_decision_patron_reduction_delta
						}
						set_temp_variable = { independence_wave_patron_row = independence_wave_decision_patron_index }
						independence_wave_clamp_patron_row = yes
					}
				}
			}
			add_to_temp_variable = { independence_wave_decision_patron_index = constant:independence_wave_plan.loop_increment }
		}
		independence_wave_refresh_strongest_patron = yes
		independence_wave_refresh_idea_lifecycles = yes
	}
}

# Consumes the first safe surplus channel and records exactly one reserve
# contribution. No channel can be counted twice.
independence_wave_decision_contribute_safe_reserve = {
	if = {
		limit = { has_equipment = { infantry_equipment > constant:independence_wave_decision_cost.infantry_major } }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = constant:independence_wave_decision_cost.infantry_standard_spend
		}
		set_temp_variable = { independence_wave_decision_reserve_gain = constant:independence_wave_decision_effect.league_reserve_standard }
	}
	else_if = {
		limit = { has_equipment = { train_equipment > constant:independence_wave_decision_cost.train_major } }
		add_equipment_to_stockpile = {
			type = train_equipment
			amount = constant:independence_wave_decision_cost.train_standard_spend
		}
		set_temp_variable = { independence_wave_decision_reserve_gain = constant:independence_wave_decision_effect.league_reserve_light }
	}
	else_if = {
		limit = { has_equipment = { convoy > constant:independence_wave_decision_cost.convoy_major } }
		add_equipment_to_stockpile = {
			type = convoy
			amount = constant:independence_wave_decision_cost.convoy_standard_spend
		}
		set_temp_variable = { independence_wave_decision_reserve_gain = constant:independence_wave_decision_effect.league_reserve_light }
	}
	else = {
		add_fuel = constant:independence_wave_decision_cost.fuel_standard_spend
		set_temp_variable = { independence_wave_decision_reserve_gain = constant:independence_wave_decision_effect.league_reserve_light }
	}
	set_temp_variable = { independence_wave_decision_league_cohesion_delta = constant:independence_wave_decision_effect.league_minor }
	set_temp_variable = { independence_wave_decision_league_common_cause_delta = constant:independence_wave_value.minimum }
	set_temp_variable = { independence_wave_decision_league_patron_capture_delta = constant:independence_wave_value.minimum }
	set_temp_variable = { independence_wave_decision_league_reserve_delta = independence_wave_decision_reserve_gain }
	set_temp_variable = { independence_wave_decision_league_confidence_delta = constant:independence_wave_decision_effect.league_minor }
	independence_wave_decision_apply_league_deltas = yes
	set_temp_variable = { independence_wave_member_contribution_delta = independence_wave_decision_reserve_gain }
	set_temp_variable = { independence_wave_member_confidence_delta = constant:independence_wave_decision_effect.league_minor }
	independence_wave_change_own_league_record = yes
}

# State scope. Adds one durable integration result and marks the state against
# repeat use by the current country.
independence_wave_decision_integrate_settled_state = {
	if = {
		limit = {
			is_controlled_by = ROOT
			NOT = { is_core_of = ROOT }
			NOT = {
				AND = {
					has_state_flag = independence_wave_dm52_integrated_by@ROOT
					check_variable = { var = independence_wave_dm52_generation@ROOT value = ROOT.independence_wave_generation_id compare = equals }
				}
			}
		}
		add_core_of = ROOT
		set_state_flag = independence_wave_dm52_integrated_by@ROOT
		set_variable = { independence_wave_dm52_generation@ROOT = ROOT.independence_wave_generation_id }
		set_state_flag = independence_wave_integrated_origin_region
		add_building_construction = {
			type = infrastructure
			level = constant:independence_wave_decision_effect.construction_infrastructure
			instant_build = yes
		}
	}
}

# State scope. Publishes the complete frozen DM-57 record. Selection does not
# consume this queue; only a confirmed shared-plan commit may do so.
independence_wave_decision_publish_breakaway_sponsorship = {
	if = {
		limit = { is_valid_independence_wave_breakaway_sponsorship_state = yes }
		set_state_flag = independence_wave_dm57_sponsorship_active
		set_variable = { independence_wave_sponsoring_country = ROOT }
		set_variable = { independence_wave_sponsorship_generation = ROOT.independence_wave_generation_id }
		set_variable = { independence_wave_sponsored_opening_strength = constant:independence_wave_decision_gate.sponsorship_opening_strength }
		set_variable = { independence_wave_sponsored_route = constant:independence_wave_government_route.radical_sovereignty }
		if = {
			limit = {
				ROOT = {
					is_independence_wave_active_country = yes
					has_country_flag = independence_wave_league_member
				}
				OR = {
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.formal_league }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.durable_league }
					check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.reformed_league }
				}
			}
			set_state_flag = independence_wave_dm57_league_sponsorship_active
		}
		else = { clr_state_flag = independence_wave_dm57_league_sponsorship_active }
		if = {
			limit = { NOT = { is_in_array = { array = global.independence_wave_sponsored_breakaway_states value = THIS } } }
			add_to_array = { array = global.independence_wave_sponsored_breakaway_states value = THIS }
		}
	}
}

# DM-58 resolver. Supported scope: the activating Event 006 country. The
# resolver first enumerates the frozen member ledger without changing claims,
# flags, wargoals, or costs. It keeps searching until one exact three-member,
# three-distinct-owner witness is found, then writes that witness to the
# aligned member/state/owner arrays. A second pass revalidates and applies the
# already-selected rows. This keeps activation feasibility and paid execution
# on the same witness instead of drawing a new random state for each member.
independence_wave_execute_reclamation_front = {
	set_temp_variable = { independence_wave_reclamation_front_witness_valid = constant:independence_wave_value.minimum }
	clear_array = global.independence_wave_reclamation_front_states
	clear_array = global.independence_wave_reclamation_front_targets
	clear_array = global.independence_wave_reclamation_front_members
	set_variable = { global.independence_wave_reclamation_front_count = constant:independence_wave_value.minimum }
	set_temp_variable = { independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum }
	set_temp_variable = { independence_wave_reclamation_front_witness_member_one_break = constant:independence_wave_value.minimum }
	set_temp_variable = { independence_wave_reclamation_front_witness_member_two_break = constant:independence_wave_value.minimum }
	set_temp_variable = { independence_wave_reclamation_front_witness_member_three_break = constant:independence_wave_value.minimum }

	# The three nested member/state searches are the effect-side equivalent of
	# the activation trigger's existential matcher. Every state search is guarded
	# by the completion variable, so a found witness stops all later mutation-
	# free scans through the same effect chain.
	for_each_scope_loop = {
		array = global.independence_wave_league_member_country_entries
		break = independence_wave_reclamation_front_witness_member_one_break
		if = {
			limit = {
				check_variable = {
					independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum
				}
				is_independence_wave_reclamation_front_member_eligible = yes
			}
			save_event_target_as = independence_wave_reclamation_front_witness_member_one
			event_target:independence_wave_reclamation_front_witness_member_one = {
				save_event_target_as = independence_wave_reclamation_front_member
			}
			every_state = {
				limit = {
					check_variable = {
						independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum
					}
					is_valid_independence_wave_reclamation_front_state = yes
				}
				save_event_target_as = independence_wave_reclamation_front_witness_state_one
				owner = {
					save_event_target_as = independence_wave_reclamation_front_witness_owner_one
				}

				for_each_scope_loop = {
					array = global.independence_wave_league_member_country_entries
					break = independence_wave_reclamation_front_witness_member_two_break
					if = {
						limit = {
							check_variable = {
								independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum
							}
							is_independence_wave_reclamation_front_member_eligible = yes
							NOT = { tag = event_target:independence_wave_reclamation_front_witness_member_one }
						}
						save_event_target_as = independence_wave_reclamation_front_witness_member_two
						event_target:independence_wave_reclamation_front_witness_member_two = {
							save_event_target_as = independence_wave_reclamation_front_member
						}
						every_state = {
							limit = {
								check_variable = {
									independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum
								}
								is_valid_independence_wave_reclamation_front_state = yes
								owner = {
									NOT = { tag = event_target:independence_wave_reclamation_front_witness_owner_one }
								}
							}
							save_event_target_as = independence_wave_reclamation_front_witness_state_two
							owner = {
								save_event_target_as = independence_wave_reclamation_front_witness_owner_two
							}

							for_each_scope_loop = {
								array = global.independence_wave_league_member_country_entries
								break = independence_wave_reclamation_front_witness_member_three_break
								if = {
									limit = {
										check_variable = {
											independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum
										}
										is_independence_wave_reclamation_front_member_eligible = yes
										NOT = { tag = event_target:independence_wave_reclamation_front_witness_member_one }
										NOT = { tag = event_target:independence_wave_reclamation_front_witness_member_two }
									}
									save_event_target_as = independence_wave_reclamation_front_witness_member_three
									event_target:independence_wave_reclamation_front_witness_member_three = {
										save_event_target_as = independence_wave_reclamation_front_member
									}
									every_state = {
										limit = {
											check_variable = {
												independence_wave_reclamation_front_witness_complete = constant:independence_wave_value.minimum
											}
											is_valid_independence_wave_reclamation_front_state = yes
											owner = {
												NOT = { tag = event_target:independence_wave_reclamation_front_witness_owner_one }
												NOT = { tag = event_target:independence_wave_reclamation_front_witness_owner_two }
											}
										}
										save_event_target_as = independence_wave_reclamation_front_witness_state_three
										owner = {
											save_event_target_as = independence_wave_reclamation_front_witness_owner_three
										}

										# Write the complete witness only after all six
										# member/owner inequality checks have passed.
										event_target:independence_wave_reclamation_front_witness_member_one = {
											add_to_array = { array = global.independence_wave_reclamation_front_members value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_member_two = {
											add_to_array = { array = global.independence_wave_reclamation_front_members value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_member_three = {
											add_to_array = { array = global.independence_wave_reclamation_front_members value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_state_one = {
											add_to_array = { array = global.independence_wave_reclamation_front_states value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_state_two = {
											add_to_array = { array = global.independence_wave_reclamation_front_states value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_state_three = {
											add_to_array = { array = global.independence_wave_reclamation_front_states value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_owner_one = {
											add_to_array = { array = global.independence_wave_reclamation_front_targets value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_owner_two = {
											add_to_array = { array = global.independence_wave_reclamation_front_targets value = THIS }
										}
										event_target:independence_wave_reclamation_front_witness_owner_three = {
											add_to_array = { array = global.independence_wave_reclamation_front_targets value = THIS }
										}
										set_temp_variable = { independence_wave_reclamation_front_witness_complete = constant:independence_wave_plan.loop_increment }
										set_temp_variable = { independence_wave_reclamation_front_witness_member_one_break = constant:independence_wave_plan.loop_increment }
										set_temp_variable = { independence_wave_reclamation_front_witness_member_two_break = constant:independence_wave_plan.loop_increment }
										set_temp_variable = { independence_wave_reclamation_front_witness_member_three_break = constant:independence_wave_plan.loop_increment }
									}
								}
							}
							# The inner member loop leaves the generic target on
							# its last probe. Restore member two before the next
							# state-two iteration reuses the shared state trigger.
							event_target:independence_wave_reclamation_front_witness_member_two = {
								save_event_target_as = independence_wave_reclamation_front_member
							}
						}
					}
				}
				# The member-two loop likewise leaves the generic target
				# on its last probe. Restore member one before the next
				# state-one iteration.
				event_target:independence_wave_reclamation_front_witness_member_one = {
					save_event_target_as = independence_wave_reclamation_front_member
				}
			}
		}
	}

	# The arrays are still only a witness until this second pass succeeds. Any
	# live-state change between activation and this effect fails closed and the
	# caller invokes the same provenance-aware pre-cost rollback as before.
	if = {
		limit = {
			check_variable = {
				independence_wave_reclamation_front_witness_complete = constant:independence_wave_plan.loop_increment
			}
		}
		independence_wave_validate_reclamation_front_witness = yes
	}
}

# DM-01 material commitment. The force-level equipment tier follows the
# garrison tier, while an isolated capital consumes one train or one truck
# reserve. The cost is sunk when the mission opens and is never refunded by a
# cancellation caused by capital loss or a broken garrison.
independence_wave_decision_pay_provisional_capital = {
	if = {
		limit = { check_variable = { var = independence_wave_force_level value = constant:liberation_force_level.fragile compare = equals } }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = constant:independence_wave_decision_cost.infantry_light_spend
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = constant:independence_wave_decision_cost.support_light_spend
		}
	}
	else_if = {
		limit = { check_variable = { var = independence_wave_force_level value = constant:liberation_force_level.viable compare = equals } }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = constant:independence_wave_decision_cost.infantry_standard_spend
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = constant:independence_wave_decision_cost.support_standard_spend
		}
	}
	else_if = {
		# Legacy or incomplete package setup has no published force level. The
		# matching trigger fallback is the standard material tier, so it must
		# consume the same standard commitment rather than the major tier.
		limit = { NOT = { has_variable = independence_wave_force_level } }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = constant:independence_wave_decision_cost.infantry_standard_spend
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = constant:independence_wave_decision_cost.support_standard_spend
		}
	}
	else = {
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = constant:independence_wave_decision_cost.infantry_major_spend
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = constant:independence_wave_decision_cost.support_major_spend
		}
	}
	if = {
		limit = { NOT = { capital_scope = { supply_node > 0 } } }
		if = {
			limit = { has_equipment = { train_equipment > constant:independence_wave_decision_cost.provisional_capital_train_gate } }
			add_equipment_to_stockpile = {
				type = train_equipment
				amount = constant:independence_wave_decision_cost.provisional_capital_train_spend
			}
		}
		else = {
			add_equipment_to_stockpile = {
				type = motorized_equipment
				amount = constant:independence_wave_decision_cost.provisional_capital_motorized_spend
			}
		}
	}
}

		var:global.independence_wave_reclamation_front_members^independence_wave_reclamation_front_witness_index = {
			save_event_target_as = independence_wave_reclamation_front_member
		}
		var:global.independence_wave_reclamation_front_targets^independence_wave_reclamation_front_witness_index = {
			save_event_target_as = independence_wave_reclamation_front_target
		}
		var:global.independence_wave_reclamation_front_states^independence_wave_reclamation_front_witness_index = {
			save_event_target_as = independence_wave_reclamation_front_state
			if = {
				limit = { is_valid_independence_wave_reclamation_front_witness_slot = yes }
				set_state_flag = independence_wave_dm58_reclamation_front_used
				if = {
					limit = { NOT = { is_claimed_by = event_target:independence_wave_reclamation_front_member } }
					add_claim_by = event_target:independence_wave_reclamation_front_member
					set_state_flag = independence_wave_dm58_reclamation_front_claim_added
				}
				event_target:independence_wave_reclamation_front_member = {
					set_temp_variable = { independence_wave_reclamation_front_expire_days = constant:independence_wave_decision_duration.reclamation_front }
					meta_effect = {
						text = {
							create_wargoal = {
								type = take_state_focus
								target = event_target:independence_wave_reclamation_front_target
								generator = { [IW_RECLAMATION_STATE] }
								expire = [IW_RECLAMATION_EXPIRE]
							}
						}
						IW_RECLAMATION_STATE = "[independence_wave_reclamation_front_state.GetID]"
						IW_RECLAMATION_EXPIRE = "[?independence_wave_reclamation_front_expire_days|.0]"
					}
					set_country_flag = independence_wave_dm58_reclamation_front_staged
					set_temp_variable = { independence_wave_reclamation_front_days = constant:independence_wave_decision_duration.reclamation_front }
					set_country_flag = { flag = independence_wave_reclamation_front_ready days = independence_wave_reclamation_front_days }
				}
				event_target:independence_wave_reclamation_front_state = {
					set_state_flag = independence_wave_dm58_reclamation_front_wargoal_added
				}
				add_to_variable = { global.independence_wave_reclamation_front_count = constant:independence_wave_plan.loop_increment }
			}
		}
	}
}

# The resolver is called before DM-58 charges any strategic or security cost.
# If witness application cannot freeze the minimum number of distinct, war-legal
# state/owner pairs, this effect removes only transaction-marked claims and
# finite wargoals, clears all staging receipts, and restores the operation
# ledger without charging the player.
independence_wave_rollback_reclamation_front_staging = {
	set_temp_variable = { independence_wave_reclamation_front_rollback_index = global.independence_wave_reclamation_front_states^num }
	while_loop_effect = {
		limit = {
			check_variable = {
				independence_wave_reclamation_front_rollback_index = constant:independence_wave_value.minimum
				compare = greater_than
			}
		}
		subtract_from_temp_variable = { independence_wave_reclamation_front_rollback_index = constant:independence_wave_plan.loop_increment }
		var:global.independence_wave_reclamation_front_members^independence_wave_reclamation_front_rollback_index = {
			save_event_target_as = independence_wave_reclamation_front_rollback_member
		}
		var:global.independence_wave_reclamation_front_targets^independence_wave_reclamation_front_rollback_index = {
			save_event_target_as = independence_wave_reclamation_front_rollback_target
		}
		var:global.independence_wave_reclamation_front_states^independence_wave_reclamation_front_rollback_index = {
			if = {
				limit = { has_state_flag = independence_wave_dm58_reclamation_front_claim_added }
				remove_claim_by = event_target:independence_wave_reclamation_front_rollback_member
			}
			clr_state_flag = independence_wave_dm58_reclamation_front_claim_added
			if = {
				limit = { has_state_flag = independence_wave_dm58_reclamation_front_wargoal_added }
				event_target:independence_wave_reclamation_front_rollback_member = {
					remove_wargoal = {
						type = take_state_focus
						target = event_target:independence_wave_reclamation_front_rollback_target
					}
				}
			}
			clr_state_flag = independence_wave_dm58_reclamation_front_wargoal_added
			clr_state_flag = independence_wave_dm58_reclamation_front_used
		}
		event_target:independence_wave_reclamation_front_rollback_member = {
			clr_country_flag = independence_wave_dm58_reclamation_front_staged
			clr_country_flag = independence_wave_reclamation_front_ready
			clear_variable = independence_wave_reclamation_front_days
		}
	}
	clear_array = global.independence_wave_reclamation_front_states
	clear_array = global.independence_wave_reclamation_front_targets
	clear_array = global.independence_wave_reclamation_front_members
	set_variable = { global.independence_wave_reclamation_front_count = constant:independence_wave_value.minimum }
}
