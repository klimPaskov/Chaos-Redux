# ============================================================================
# CHAOS REDUX - EVENT 006 AUTOMATIC PACKAGE ALLOCATOR
# ============================================================================
# Overview:
# - Recomputes every regional package weight before every draw.
# - Selects regions by their summed viable package weight, then selects one
#   package inside that region, preserving a single global weighted draw.
# - Keeps rerolling rejected candidates without reducing the accepted wave
#   count. If the exact count cannot be frozen, the contribution fails and no
#   ownership mutation is permitted.
# - Assumes the shared release coordinator is in its allocation phase and the
#   Event 006 participant flag is present.
# ============================================================================

independence_wave_prepare_all_automatic_weights = {

	independence_wave_prepare_region_01_automatic_weights = yes
	independence_wave_prepare_region_02_automatic_weights = yes
	independence_wave_prepare_region_03_automatic_weights = yes
	independence_wave_prepare_region_04_automatic_weights = yes
	independence_wave_prepare_region_05_automatic_weights = yes
	independence_wave_prepare_region_06_automatic_weights = yes
	independence_wave_prepare_region_07_automatic_weights = yes
	independence_wave_prepare_region_08_automatic_weights = yes
	independence_wave_prepare_region_09_automatic_weights = yes
	independence_wave_prepare_region_10_automatic_weights = yes
	independence_wave_prepare_region_11_automatic_weights = yes
	independence_wave_prepare_region_12_automatic_weights = yes
	independence_wave_prepare_region_13_automatic_weights = yes
	independence_wave_prepare_region_14_automatic_weights = yes
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_01_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_02_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_03_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_04_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_05_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_06_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_07_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_08_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_09_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_10_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_11_total_weight }

	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_13_total_weight }
	add_to_temp_variable = { independence_wave_total_automatic_weight = independence_wave_region_14_total_weight }
}

independence_wave_select_one_automatic_package = {
	independence_wave_prepare_all_automatic_weights = yes
	if = {
		limit = {
			check_variable = {
				var = independence_wave_total_automatic_weight
				value = constant:independence_wave_allocation_factor.zero
				compare = greater_than
			}
		}
		random_list = {
			independence_wave_region_01_total_weight = { independence_wave_select_region_01_automatic_package = yes }
			independence_wave_region_02_total_weight = { independence_wave_select_region_02_automatic_package = yes }
			independence_wave_region_03_total_weight = { independence_wave_select_region_03_automatic_package = yes }
			independence_wave_region_04_total_weight = { independence_wave_select_region_04_automatic_package = yes }
			independence_wave_region_05_total_weight = { independence_wave_select_region_05_automatic_package = yes }
			independence_wave_region_06_total_weight = { independence_wave_select_region_06_automatic_package = yes }
			independence_wave_region_07_total_weight = { independence_wave_select_region_07_automatic_package = yes }
			independence_wave_region_08_total_weight = { independence_wave_select_region_08_automatic_package = yes }
			independence_wave_region_09_total_weight = { independence_wave_select_region_09_automatic_package = yes }
			independence_wave_region_10_total_weight = { independence_wave_select_region_10_automatic_package = yes }
			independence_wave_region_11_total_weight = { independence_wave_select_region_11_automatic_package = yes }
			independence_wave_region_12_total_weight = { independence_wave_select_region_12_automatic_package = yes }
			independence_wave_region_13_total_weight = { independence_wave_select_region_13_automatic_package = yes }
			independence_wave_region_14_total_weight = { independence_wave_select_region_14_automatic_package = yes }
		}
	}
	else = {
		set_global_flag = independence_wave_plan_pool_exhausted
	}
}

independence_wave_allocate_automatic_packages = {
	clr_global_flag = independence_wave_plan_pool_exhausted
	clr_global_flag = independence_wave_plan_contribution_ready
	clr_global_flag = independence_wave_plan_exact_count_failed
	if = {
		limit = {
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.allocating }
			has_global_flag = liberation_release_plan_includes_event6
		}
		independence_wave_prepare_evolution_for_incident = yes
		independence_wave_freeze_evolution_plan_state = yes
		independence_wave_begin_plan_contribution = yes
		while_loop_effect = {
			limit = {
				NOT = { has_global_flag = independence_wave_plan_pool_exhausted }
				check_variable = {
					var = global.independence_wave_plan_selected_count
					value = global.independence_wave_plan_target_count
					compare = less_than
				}
				check_variable = {
					var = global.independence_wave_plan_attempt_count
					value = constant:independence_wave_planner.maximum_attempts
					compare = less_than
				}
			}
			independence_wave_select_one_automatic_package = yes
		}
		if = {
			limit = {
				check_variable = {
					var = global.independence_wave_plan_selected_count
					value = global.independence_wave_plan_target_count
					compare = equals
				}
			}
			if = {
				limit = { independence_wave_plan_metadata_arrays_are_aligned = yes }
				set_global_flag = independence_wave_plan_contribution_ready
			}
			else = {
				set_global_flag = independence_wave_plan_exact_count_failed
				set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.aligned_array_failure }
			}
		}
		else = {
			set_global_flag = independence_wave_plan_exact_count_failed
			set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.insufficient_pool }
		}
	}
	else = {
		set_global_flag = independence_wave_plan_exact_count_failed
		set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.stale_plan }
	}
}
