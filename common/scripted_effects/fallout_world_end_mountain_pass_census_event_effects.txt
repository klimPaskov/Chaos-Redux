fallout_event_635_pay_branch_cost = {

			limit = { check_variable = { fallout_event_635_selected_branch = constant:fallout_event_635_branch.local_citizenship } }
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_635_cost.local_food }

			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_635_cost.local_recognition }

			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_635_cost.return_food }

			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_635_cost.return_recognition }

			limit = { check_variable = { fallout_event_635_selected_branch = constant:fallout_event_635_branch.military_colony } }
			subtract_from_variable = { fallout_survival_current_entries^5 = constant:fallout_event_635_cost.military_fuel }
