# ============================================================================
# CHAOS REDUX - EVENT 033 ACID RAIN GUI PROJECTION
# ============================================================================
# COUNTRY SCOPE. This file copies managed Event 33 state into presentation
# variables. Open/close flags and the viewing generation belong to the caller.
# Country counts come from the runtime owner's controller-aware edge publisher.
# It never selects a state, changes a front, spends a cost, applies mortality,
# adds contamination, advances a timer, or writes a gameplay receipt.

@acid_rain_gui_floor_half = 0.5

acid_rain_gui_open_from_event_details = {
	acid_rain_gui_open_world_window = yes
	if = {
		limit = { has_country_flag = acid_rain_gui_window_open }
		set_country_flag = acid_rain_event_details_gui_open
	}
}

acid_rain_gui_open_from_category = {
	acid_rain_gui_open_world_window = yes
}

# No runtime setup or repair is allowed here. Missing/invalid records fail closed.
acid_rain_gui_open_world_window = {
	acid_rain_gui_close_event_details_view = yes
	if = {
		limit = {
			exists = yes
			is_ai = no
			has_variable = global.acid_rain_generation_id
			check_variable = { global.acid_rain_generation_id > constant:acid_rain_runtime.zero }
			check_variable = { var = global.acid_rain_phase value = constant:acid_rain_phase.formation compare = greater_than_or_equals }
			check_variable = { var = global.acid_rain_phase value = constant:acid_rain_phase.closed compare = less_than_or_equals }
			NOT = { has_global_flag = acid_rain_migration_failed_closed }
		}
		acid_rain_gui_refresh_read_only = yes
		set_variable = { acid_rain_gui_view_generation = global.acid_rain_generation_id }
		set_country_flag = acid_rain_gui_window_open
	}
}

acid_rain_gui_close_event_details_view = {
	clr_country_flag = acid_rain_event_details_gui_open
	clr_country_flag = acid_rain_gui_window_open
	clear_variable = acid_rain_gui_view_generation
}

# COUNTRY SCOPE. Closes the world map and opens the Event Log on the Event 33
# Event Details page through the shared Event Log entry points. Presentation
# only: no weather, timer or gameplay value changes.
acid_rain_gui_open_event_log_details = {
	acid_rain_gui_close_event_details_view = yes
	if = {
		limit = {
			exists = yes
			is_ai = no
		}
		open_events_log_events_tab = yes
		set_variable = { events_log_selected_event_id = constant:acid_rain_event.id }
		set_temp_variable = { event_id = events_log_selected_event_id }
		set_temp_variable = { event_type = constant:event_system_event_type.unknown }
		get_event_type = yes
		set_variable = { events_log_selected_event_type = event_type }
		events_log_add_event_details_entry = yes
		clr_country_flag = events_log_evolution_details_open
	}
}

# COUNTRY SCOPE. A presentation preference only; absent flag means motion on.
# Keep the preference across close/reopen and episode generations.
acid_rain_gui_toggle_motion = {
	if = {
		limit = {
			exists = yes
			is_ai = no
			has_country_flag = acid_rain_gui_window_open
			has_variable = acid_rain_gui_view_generation
			has_variable = global.acid_rain_generation_id
			check_variable = { global.acid_rain_generation_id > constant:acid_rain_runtime.zero }
			check_variable = { var = acid_rain_gui_view_generation value = global.acid_rain_generation_id compare = equals }
			check_variable = { var = global.acid_rain_phase value = constant:acid_rain_phase.formation compare = greater_than_or_equals }
			check_variable = { var = global.acid_rain_phase value = constant:acid_rain_phase.closed compare = less_than_or_equals }
			NOT = { has_global_flag = acid_rain_migration_failed_closed }
		}
		if = {
			limit = { has_country_flag = acid_rain_gui_motion_disabled }
			clr_country_flag = acid_rain_gui_motion_disabled
		}
		else = { set_country_flag = acid_rain_gui_motion_disabled }
	}
	else = { acid_rain_gui_close_event_details_view = yes }
}
