# ============================================================================
# CHAOS REDUX - EVENT 014 IRREGULAR SUBUNIT ACTIVATION EFFECTS
# ============================================================================
# Package-owned bridge effects for the nine inactive Event 014 line subunits.
# Each caller owns the existing Event 014 package, focus, or transition gate;
# these effects only grant the hidden technology that makes the matching
# subunit valid for its already-paid and locked-template spawn transaction.
# ============================================================================

cannibalism_unlock_event014_warlord_package_subunits = {
	set_technology = {
		cannibalism_scavenger_warband_tech = 1
		cannibalism_feast_guard_tech = 1
		popup = no
	}
	if = {
		limit = { check_variable = { cannibalism_warlord_origin = constant:cannibalism_origin.island_host } }
		set_technology = { cannibalism_island_reavers_tech = 1 popup = no }
	}
	else_if = {
		limit = { check_variable = { cannibalism_warlord_origin = constant:cannibalism_origin.siege_commune } }
		set_technology = { cannibalism_siege_eaters_tech = 1 popup = no }
	}
	else_if = {
		limit = { check_variable = { cannibalism_warlord_origin = constant:cannibalism_origin.march_host } }
		set_technology = { cannibalism_march_predation_column_tech = 1 popup = no }
	}
}
