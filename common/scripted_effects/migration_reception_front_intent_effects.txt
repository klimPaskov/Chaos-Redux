# move people, bind a cohort/destination, grant route permission, change services
# or capacity, reveal a category, register a job, or apply an outflow modifier.
# Complete local scope iterators run only at real transaction/lifecycle seams.

	clr_state_flag = migration_reception_front_intent_cache_active
	clear_variable = migration_reception_front_intent_cache_origin_id
	clear_variable = migration_reception_front_intent_cache_generation

# First valid publication only. Partial/malformed previous metadata is not
# repaired or replaced with an empty registry.
migration_reception_front_intent_initialize_records = {
	if = {
		limit = {
			NOT = { has_variable = migration_reception_front_intent_schema_version }
			NOT = { has_state_flag = migration_reception_front_intent_cache_active }
			check_variable = { migration_reception_front_intent_origins^num = constant:migration_reception_front_intent.zero }
			check_variable = { migration_reception_front_intent_generations^num = constant:migration_reception_front_intent.zero }
			check_variable = { migration_reception_front_intent_requesters^num = constant:migration_reception_front_intent.zero }

			check_variable = { migration_reception_front_intent_cohorts^num = constant:migration_reception_front_intent.zero }
			check_variable = { migration_reception_front_intent_phases^num = constant:migration_reception_front_intent.zero }
		}
		clear_array = migration_reception_front_intent_origins
		clear_array = migration_reception_front_intent_generations
		clear_array = migration_reception_front_intent_requesters
		clear_array = migration_reception_front_intent_receivers
		clear_array = migration_reception_front_intent_cohorts
		clear_array = migration_reception_front_intent_phases
		set_variable = { migration_reception_front_intent_schema_version = constant:migration_reception_front_intent.schema_version }
	}
}

	clear_temp_array = migration_reception_front_intent_seen_origins

		limit = { migration_reception_front_intent_arrays_are_aligned = yes }
		for_each_scope_loop = {
			array = migration_reception_front_intent_origins
			break = migration_reception_front_intent_walk_break
			migration_reception_front_intent_load_walk_identity = yes
			if = {
				limit = {

					check_variable = { migration_reception_front_intent_identity_generation > constant:migration_reception_front_intent.zero }

				}
				if = {
					limit = {
						migration_reception_front_intent_identity_is_current = yes
						event_target:migration_reception_front_intent_front = { migration_reception_front_intent_identity_front_is_current = yes }
					}
					if = {
						limit = { is_in_array = { array = migration_reception_front_intent_seen_origins value = THIS } }
						set_temp_variable = { migration_reception_front_intent_walk_valid = constant:migration_reception_front_intent.zero }
					}
					else = {
						add_to_temp_array = { array = migration_reception_front_intent_seen_origins value = THIS }
						if = {
							limit = {
								OR = {
									check_variable = { migration_reception_front_intent_best_origin_id = constant:migration_reception_front_intent.zero }
									check_variable = { migration_reception_front_intent_identity_phase < migration_reception_front_intent_best_phase }
									AND = {
										check_variable = { migration_reception_front_intent_identity_phase = migration_reception_front_intent_best_phase }
										check_variable = { migration_reception_front_intent_identity_generation > migration_reception_front_intent_best_generation }
									}
									AND = {
										check_variable = { migration_reception_front_intent_identity_phase = migration_reception_front_intent_best_phase }
										check_variable = { migration_reception_front_intent_identity_generation = migration_reception_front_intent_best_generation }
										check_variable = { migration_reception_front_intent_identity_origin_id < migration_reception_front_intent_best_origin_id }
									}
								}
							}
							set_temp_variable = { migration_reception_front_intent_best_origin_id = migration_reception_front_intent_identity_origin_id }
							set_temp_variable = { migration_reception_front_intent_best_generation = migration_reception_front_intent_identity_generation }
							set_temp_variable = { migration_reception_front_intent_best_requester = migration_reception_front_intent_identity_requester }

					}
				}
			}
			else = { set_temp_variable = { migration_reception_front_intent_walk_valid = constant:migration_reception_front_intent.zero } }
			add_to_temp_variable = { migration_reception_front_intent_walk_index = constant:humanitarian_runtime.array_index_increment }
			add_to_temp_variable = { migration_reception_front_intent_walk_visits = constant:humanitarian_runtime.one }
		}
		if = {
			limit = {
				check_variable = { migration_reception_front_intent_walk_valid = constant:migration_reception_front_intent.one }
				check_variable = { migration_reception_front_intent_walk_visits = migration_reception_front_intent_walk_expected }
			}
			if = {
				limit = { check_variable = { migration_reception_front_intent_best_origin_id > constant:migration_reception_front_intent.zero } }
				set_variable = { migration_reception_front_intent_cache_origin_id = migration_reception_front_intent_best_origin_id }
				set_variable = { migration_reception_front_intent_cache_generation = migration_reception_front_intent_best_generation }
				set_variable = { migration_reception_front_intent_cache_requester = migration_reception_front_intent_best_requester }

		}

		}

	}

	set_temp_variable = { migration_reception_front_intent_clear_receiver = constant:migration_reception_front_intent.zero }

# FRONT STATE after its owner or controller changed. Retires only the rows
# whose recorded receiver no longer owns and controls this state, walking from
# the last row so earlier indices stay stable. Rows for a receiver that still
# holds the state survive. Malformed metadata is left untouched and only the
# cache is cleared, so map readers stay neutral.
migration_reception_front_intent_handle_control_change = {
	if = {
		limit = { has_variable = migration_reception_front_intent_schema_version }
		if = {
			limit = { migration_reception_front_intent_arrays_are_aligned = yes }
			set_temp_variable = { migration_reception_front_intent_prune_index = migration_reception_front_intent_origins^num }
			while_loop_effect = {
				limit = { check_variable = { migration_reception_front_intent_prune_index > constant:migration_reception_front_intent.zero } }
				subtract_from_temp_variable = { migration_reception_front_intent_prune_index = constant:humanitarian_runtime.array_index_increment }
				set_temp_variable = { migration_reception_front_intent_prune_receiver = migration_reception_front_intent_receivers^migration_reception_front_intent_prune_index }
				set_temp_variable = { migration_reception_front_intent_prune_row = constant:migration_reception_front_intent.one }
				if = {
					limit = { check_variable = { migration_reception_front_intent_prune_receiver > constant:migration_reception_front_intent.zero } }
					if = {
						limit = {
							var:migration_reception_front_intent_prune_receiver = { humanitarian_country_is_valid = yes }
							is_owned_and_controlled_by = var:migration_reception_front_intent_prune_receiver
						}
						set_temp_variable = { migration_reception_front_intent_prune_row = constant:migration_reception_front_intent.zero }
					}
				}
				if = {
					limit = { check_variable = { migration_reception_front_intent_prune_row = constant:migration_reception_front_intent.one } }
					remove_from_array = { array = migration_reception_front_intent_origins index = migration_reception_front_intent_prune_index }
					remove_from_array = { array = migration_reception_front_intent_generations index = migration_reception_front_intent_prune_index }
					remove_from_array = { array = migration_reception_front_intent_requesters index = migration_reception_front_intent_prune_index }
					remove_from_array = { array = migration_reception_front_intent_receivers index = migration_reception_front_intent_prune_index }
					remove_from_array = { array = migration_reception_front_intent_cohorts index = migration_reception_front_intent_prune_index }
					remove_from_array = { array = migration_reception_front_intent_phases index = migration_reception_front_intent_prune_index }
				}
			}
			migration_refresh_reception_front_intent_projection = yes
		}
		else = { migration_reception_front_intent_clear_cache = yes }
	}
}
