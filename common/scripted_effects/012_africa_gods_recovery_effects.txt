# Event 012 Gods original-owner mandate recovery.
# All public calls use = yes. Inputs and recount outputs are caller-set temps.
# This file never selects a host, changes shared generation, uncaps a country,
# changes doctrine, or initializes from a queued event. See the matching .md.

# Actual setup caller alone supplies this one-call authorization input.
# COUNTRY object scope; ROOT independent. No legacy/wake migration is allowed.
gods_of_africa_authorize_initial_epoch = {
	if = {
		limit = {
			check_variable = { gods_of_africa_initial_epoch_setup = constant:gods_of_africa_value.one }
			gods_of_africa_authority_owner_object_is_current = yes
			gods_of_africa_authority_owner_has_physical_return = yes
			NOT = { gods_of_africa_continent_is_secured = yes }
			NOT = { has_variable = gods_of_africa_authority_epoch }
			NOT = { has_variable = gods_of_africa_loss_sequence }
			NOT = { has_variable = gods_of_africa_loss_epoch }
			NOT = { has_country_flag = gods_of_africa_cancelled_on_unifier_loss }
			NOT = { has_country_flag = gods_of_africa_unifier_loss_recorded }
			NOT = { has_country_flag = gods_of_africa_recovery_epoch }
			NOT = { has_country_flag = gods_of_africa_original_mandate_restored }
			NOT = { has_country_flag = gods_of_africa_active }
			NOT = { has_country_flag = gods_of_africa_activation_committed }
		}
		set_variable = { gods_of_africa_authority_epoch = constant:gods_of_africa_value.one }
		set_variable = { gods_of_africa_authority_host_id = THIS.id }
		set_variable = { gods_of_africa_authority_origin_host_id = africa_origin_host_id }
		set_variable = { gods_of_africa_authority_host_generation = africa_host_generation }
		set_country_flag = gods_of_africa_epoch_authorized
	}
	# Consume the caller token even when an invalid or repeated call is rejected.
	set_temp_variable = { gods_of_africa_initial_epoch_setup = constant:gods_of_africa_value.zero }
}

# Member COUNTRY. Private request state only; durable history is untouched.
gods_of_africa_clear_recovery_member_request = {
	clr_country_flag = gods_of_africa_recovery_invited
	clr_country_flag = gods_of_africa_recovery_approved
	clr_country_flag = gods_of_africa_recovery_declined
	clear_variable = gods_of_africa_recovery_request_host
	clear_variable = gods_of_africa_recovery_request_host_id
	clear_variable = gods_of_africa_recovery_request_origin_host_id
	clear_variable = gods_of_africa_recovery_request_host_generation
	clear_variable = gods_of_africa_recovery_request_loss_sequence
	clear_variable = gods_of_africa_recovery_request_cancelled_epoch
	clear_variable = gods_of_africa_recovery_answer_day
}

# Host object. Cleanup validates ownership receipts even after physical loss.
# Member cleanup does not mutate this host's invitation array while iterating.
gods_of_africa_close_recovery_requests = {
	for_each_scope_loop = {
		array = gods_of_africa_recovery_invited_members
		if = {
			limit = {
				has_variable = gods_of_africa_recovery_request_host_id
				check_variable = { var = gods_of_africa_recovery_request_host_id value = PREV.id compare = equals }
			}
			gods_of_africa_clear_recovery_member_request = yes
		}
	}
	clear_array = gods_of_africa_recovery_invited_members
	clear_array = gods_of_africa_recovery_eligible_members
	clear_array = gods_of_africa_recovery_approved_members
	clear_array = gods_of_africa_mandate_approved_members
	clr_country_flag = gods_of_africa_recovery_open
	clear_variable = gods_of_africa_recovery_host_id
	clear_variable = gods_of_africa_recovery_origin_host_id
	clear_variable = gods_of_africa_recovery_host_generation
	clear_variable = gods_of_africa_recovery_loss_sequence
	clear_variable = gods_of_africa_recovery_cancelled_epoch
	clear_variable = gods_of_africa_recovery_start_day
}

# Departing host object BEFORE parent roster/target cleanup. Can run capitulated
# or after a terminal flag was set. Never needs an unrelated native ROOT.
gods_of_africa_revoke_current_epoch = {
	if = {
		limit = {
			has_variable = gods_of_africa_authority_epoch
			has_variable = gods_of_africa_authority_host_id
			check_variable = { var = gods_of_africa_authority_host_id value = THIS.id compare = equals }
		}
		if = {
			limit = {
				has_country_flag = gods_of_africa_epoch_authorized
				NOT = {
					AND = {
						has_variable = gods_of_africa_loss_epoch
						check_variable = { var = gods_of_africa_loss_epoch value = gods_of_africa_authority_epoch compare = equals }
					}
				}
			}
			if = {
				limit = { NOT = { has_variable = gods_of_africa_loss_sequence } }
				set_variable = { gods_of_africa_loss_sequence = constant:gods_of_africa_value.zero }
			}
			add_to_variable = { gods_of_africa_loss_sequence = constant:gods_of_africa_value.one }
			set_variable = { gods_of_africa_loss_epoch = gods_of_africa_authority_epoch }
			set_variable = { gods_of_africa_loss_day = global.num_days }
		}
		clr_country_flag = gods_of_africa_epoch_authorized
		set_country_flag = gods_of_africa_cancelled_on_unifier_loss
		gods_of_africa_close_recovery_requests = yes
		clr_country_flag = gods_of_africa_recovery_epoch
		clr_country_flag = gods_of_africa_activation_clock_scheduled
		clr_country_flag = gods_of_africa_owner_clock_scheduled
		clr_country_flag = gods_of_africa_registry_clock_scheduled
		clear_variable = gods_of_africa_consolidation_start_day
		clear_variable = gods_of_africa_consolidation_due_day
		clear_variable = gods_of_africa_next_owner_tick_day
		clear_variable = gods_of_africa_next_registry_review_day
		clear_variable = gods_of_africa_last_clock_repair_day
	}
}

					}
				}
			}
		}
	}
}

# Native targeted decision: host ROOT/THIS, member FROM. Read the live registry
# so a member joining during this congress can also be invited without restart.
gods_of_africa_invite_recovery_member = {
	if = {
		limit = {
			gods_of_africa_recovery_can_invite = yes
			FROM = {
				gods_of_africa_recovery_member_is_eligible = yes
				NOT = { gods_of_africa_recovery_member_receipt_is_current = yes }
			}
		}
		FROM = {
			gods_of_africa_clear_recovery_member_request = yes
			set_variable = { gods_of_africa_recovery_request_host = ROOT }
			set_variable = { gods_of_africa_recovery_request_host_id = ROOT.id }
			set_variable = { gods_of_africa_recovery_request_origin_host_id = ROOT.africa_origin_host_id }
			set_variable = { gods_of_africa_recovery_request_host_generation = ROOT.africa_host_generation }
			set_variable = { gods_of_africa_recovery_request_loss_sequence = ROOT.gods_of_africa_loss_sequence }
			set_variable = { gods_of_africa_recovery_request_cancelled_epoch = ROOT.gods_of_africa_loss_epoch }
			set_country_flag = gods_of_africa_recovery_invited
		}
		if = {
			limit = { NOT = { is_in_array = { array = gods_of_africa_recovery_invited_members value = FROM } } }
			add_to_array = { array = gods_of_africa_recovery_invited_members value = FROM }
		}
		if = {
			limit = { NOT = { is_in_array = { array = gods_of_africa_recovery_eligible_members value = FROM } } }
			add_to_array = { array = gods_of_africa_recovery_eligible_members value = FROM }
		}
	}
}

# Member COUNTRY; caller temp gods_of_africa_recovery_answer is zero (decline)
# or one (approve). No default; callers must initialize before every call.
gods_of_africa_record_recovery_member_answer = {
	if = {
		limit = {
			gods_of_africa_recovery_member_request_is_current = yes
			OR = {
				check_variable = { gods_of_africa_recovery_answer = constant:gods_of_africa_value.zero }
				check_variable = { gods_of_africa_recovery_answer = constant:gods_of_africa_value.one }
			}
		}
		clr_country_flag = gods_of_africa_recovery_approved
		clr_country_flag = gods_of_africa_recovery_declined
		set_variable = { gods_of_africa_recovery_answer_day = global.num_days }
		if = {
			limit = { check_variable = { gods_of_africa_recovery_answer = constant:gods_of_africa_value.one } }
			set_country_flag = gods_of_africa_recovery_approved
			event_target:africa_host = {
				if = {
					limit = {
						has_country_flag = gods_of_africa_recovery_open
						NOT = { is_in_array = { array = gods_of_africa_recovery_approved_members value = PREV } }
					}
					add_to_array = { array = gods_of_africa_recovery_approved_members value = PREV }
				}
				if = {
					limit = {
						has_country_flag = gods_of_africa_recovery_epoch
						NOT = { is_in_array = { array = gods_of_africa_mandate_approved_members value = PREV } }
					}
					add_to_array = { array = gods_of_africa_mandate_approved_members value = PREV }
				}
			}
		}
		else = {
			set_country_flag = gods_of_africa_recovery_declined
			event_target:africa_host = {
				remove_from_array = { array = gods_of_africa_recovery_approved_members value = PREV }
				remove_from_array = { array = gods_of_africa_mandate_approved_members value = PREV }
			}
		}
	}
}

# COUNTRY participant. Parent cancellation clears two durable settlement flags;
# retain their exact pre-call presence rather than acknowledging old reports.
# The normal settlement/victory receipt fields are not cleared by that closure.
gods_of_africa_cancel_recovery_participant_preserving_receipts = {
	# Africa's defeat ends the defiance a participant survived.
	gods_of_africa_achievement_complete_survive_defiance = yes
	set_temp_variable = { gods_of_africa_recovery_keep_final_outcome = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_recovery_keep_settlement_report = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_recovery_keep_victor_report = constant:gods_of_africa_value.zero }
	if = {
		limit = { has_country_flag = gods_of_africa_final_outcome_committed }
		set_temp_variable = { gods_of_africa_recovery_keep_final_outcome = constant:gods_of_africa_value.one }
	}
	if = {
		limit = { has_country_flag = gods_of_africa_settlement_report_pending }
		set_temp_variable = { gods_of_africa_recovery_keep_settlement_report = constant:gods_of_africa_value.one }
	}
	if = {
		limit = { has_country_flag = gods_of_africa_victor_report_pending }
		set_temp_variable = { gods_of_africa_recovery_keep_victor_report = constant:gods_of_africa_value.one }
	}
	gods_of_africa_cancel_participant = yes
	if = {
		limit = { check_variable = { gods_of_africa_recovery_keep_final_outcome = constant:gods_of_africa_value.one } }
		set_country_flag = gods_of_africa_final_outcome_committed
	}

		set_country_flag = gods_of_africa_settlement_report_pending
	}
	if = {
		limit = { check_variable = { gods_of_africa_recovery_keep_victor_report = constant:gods_of_africa_value.one } }
		set_country_flag = gods_of_africa_victor_report_pending
	}
}

# Trusted host scheduling only, not a wake handler. Caller sets kind/due/days.
# Match the caller due to the present owner date. Does not change a due date.
gods_of_africa_schedule_current_clock = {
	if = {
		limit = {
			tag = ROOT
			gods_of_africa_epoch_permission_is_current = yes
		}
		set_temp_variable = { gods_of_africa_clock_days = gods_of_africa_clock_due }
		subtract_from_temp_variable = { gods_of_africa_clock_days = global.num_days }
		if = {
			limit = { check_variable = { gods_of_africa_clock_days < constant:gods_of_africa_value.one } }
			set_temp_variable = { gods_of_africa_clock_days = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
				check_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_recovery_clock.activation }
				has_country_flag = gods_of_africa_consolidation_pending
				NOT = { has_country_flag = gods_of_africa_active }
				NOT = { has_country_flag = gods_of_africa_activation_committed }
				has_variable = gods_of_africa_consolidation_due_day
				check_variable = { var = gods_of_africa_clock_due value = gods_of_africa_consolidation_due_day compare = equals }
				OR = {
					NOT = { has_country_flag = gods_of_africa_recovery_epoch }
					gods_of_africa_recovery_mandate_is_current = yes
				}
			}
			set_country_flag = gods_of_africa_activation_clock_scheduled
			country_event = { id = chaosx.nr12.600 days = var:gods_of_africa_clock_days }
		}
		else_if = {
			limit = {
				check_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_recovery_clock.owner }
				has_country_flag = gods_of_africa_active
				has_country_flag = gods_of_africa_activation_committed
				has_variable = gods_of_africa_next_owner_tick_day
				check_variable = { var = gods_of_africa_clock_due value = gods_of_africa_next_owner_tick_day compare = equals }
			}
			set_country_flag = gods_of_africa_owner_clock_scheduled
			country_event = { id = chaosx.nr12.606 days = var:gods_of_africa_clock_days }
		}
		else_if = {
			limit = {
				check_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_recovery_clock.registry }
				has_country_flag = gods_of_africa_active
				has_country_flag = gods_of_africa_activation_committed
				has_variable = gods_of_africa_next_registry_review_day
				check_variable = { var = gods_of_africa_clock_due value = gods_of_africa_next_registry_review_day compare = equals }
			}
			set_country_flag = gods_of_africa_registry_clock_scheduled
			country_event = { id = chaosx.nr12.607 days = var:gods_of_africa_clock_days }
		}
	}
}

# One current-day repair after authorization, receipt, due and quorum checks.
# Retains the elapsed due day and authority epoch. No fresh consolidation.

	if = {

		set_variable = { gods_of_africa_last_clock_repair_day = global.num_days }
		set_temp_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_value.zero }
		set_temp_variable = { gods_of_africa_clock_due = constant:gods_of_africa_value.zero }
		if = {
			limit = { gods_of_africa_activation_wake_is_due = yes }
			set_temp_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_recovery_clock.activation }
			set_temp_variable = { gods_of_africa_clock_due = gods_of_africa_consolidation_due_day }
		}
		else_if = {
			limit = { gods_of_africa_owner_wake_is_due = yes }
			set_temp_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_recovery_clock.owner }
			set_temp_variable = { gods_of_africa_clock_due = gods_of_africa_next_owner_tick_day }
		}
		else_if = {
			limit = { gods_of_africa_registry_wake_is_due = yes }
			set_temp_variable = { gods_of_africa_clock_kind = constant:gods_of_africa_recovery_clock.registry }
			set_temp_variable = { gods_of_africa_clock_due = gods_of_africa_next_registry_review_day }
		}
		set_temp_variable = { gods_of_africa_clock_days = constant:gods_of_africa_value.zero }
		gods_of_africa_schedule_current_clock = yes
	}
