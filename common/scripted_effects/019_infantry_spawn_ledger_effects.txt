infantry_spawn_initialize_country_system = {
	if = {
		limit = {
			NOT = { has_country_flag = infantry_spawn_participant }
			NOT = { is_infantry_spawn_derivative_country = yes }
			NOT = { has_country_flag = infantry_spawn_annex_cleanup_invariant_failure }
		}
		clr_country_flag = infantry_spawn_country_cleanup_complete
		clr_country_flag = infantry_spawn_defeat_cleanup_in_progress
		set_country_flag = infantry_spawn_participant
		set_variable = { infantry_spawn_generation = 0 }
		set_variable = { infantry_spawn_active_lot_count = 0 }
		set_variable = { infantry_spawn_active_division_count = 0 }
		set_variable = { infantry_spawn_destroyed_division_count = 0 }
		set_variable = { infantry_spawn_unaccounted_division_count = 0 }
		set_variable = { infantry_spawn_unresolved_generation_count = 0 }
		set_variable = { infantry_spawn_equipment_debt = 0 }
		set_variable = { infantry_spawn_manpower_liability = 0 }
		set_variable = { infantry_spawn_anomalous_saturation = constant:infantry_spawn_saturation.minimum }
		set_variable = { infantry_spawn_request_count_generation = 0 }
		set_variable = { infantry_spawn_anomalous_request_count = 0 }
		set_variable = { infantry_spawn_active_claimant_count = 0 }
		set_variable = { infantry_spawn_ai_profile = constant:infantry_spawn_ai_profile.professionalizer }
		set_variable = { infantry_spawn_selected_lot_index = constant:infantry_spawn_event.invalid_index }

		set_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.opening_base }
		set_temp_variable = { infantry_spawn_opening_stability = stability }
		multiply_temp_variable = { infantry_spawn_opening_stability = constant:infantry_spawn_control.opening_stability_factor }
		add_to_variable = { infantry_spawn_muster_control = infantry_spawn_opening_stability }
		if = {
			limit = { has_war = yes }
			subtract_from_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.opening_war_penalty }
		}
		clamp_variable = {
			var = infantry_spawn_muster_control
			min = constant:infantry_spawn_control.minimum
			max = constant:infantry_spawn_control.maximum
		}
		set_variable = { infantry_spawn_army_congestion = constant:infantry_spawn_congestion.minimum }
	}
	infantry_spawn_reconcile_participant_evolution_memberships = yes
}

infantry_spawn_mark_ledger_invariant_failure = {
	set_country_flag = infantry_spawn_ledger_invariant_failure
	set_country_flag = infantry_spawn_processing_deferred
}

infantry_spawn_validate_all_ledgers = {
	if = {
		limit = {
			OR = {
				infantry_spawn_all_ledgers_are_aligned = no
				infantry_spawn_claimant_ledgers_are_aligned = no
			}
		}
		infantry_spawn_mark_ledger_invariant_failure = yes
	}
}

# Country-pulse reconciliation for live division scopes. HOI4 exposes no
# division-destruction on_action, so active Event 19 countries take one bounded
# snapshot of their own divisions. A missing exact UID is recorded once as
# unaccounted: its material and manpower obligations stay outstanding, while
# the absent formation leaves the live-unit counts. Empty affected lots remain
# unresolved and cannot be integrated or demobilized to erase their debt.
# Combat casualties remain solely in the shared deaths pipeline; this helper
# never writes deaths, wars, or parent-event progression.
infantry_spawn_reconcile_live_unit_ledger = {
	clear_temp_array = infantry_spawn_live_unit_uid_snapshot
	every_country_division = {
		limit = {
			has_variable = infantry_spawn_event19_generated_identity
			has_variable = infantry_spawn_event19_unit_uid
			check_variable = { infantry_spawn_event19_generated_identity = 1 }
		}
		add_to_temp_array = { array = infantry_spawn_live_unit_uid_snapshot value = infantry_spawn_event19_unit_uid }
	}
	set_temp_variable = { infantry_spawn_reconcile_lost_unit_count = 0 }
	set_temp_variable = { infantry_spawn_reconcile_unit_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_reconcile_unit_index value = infantry_spawn_unit_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_reconcile_unit_index value = constant:infantry_spawn_unit_status.untracked compare = greater_than }
				check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_reconcile_unit_index value = constant:infantry_spawn_unit_status.transferred_out compare = less_than }
				NOT = {
					AND = {
						has_country_flag = infantry_spawn_derivative_creation_in_progress
						check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_reconcile_unit_index value = constant:infantry_spawn_unit_status.transfer_staged compare = equals }
					}
				}
				NOT = {
					is_in_array = {
						array = infantry_spawn_live_unit_uid_snapshot
						value = infantry_spawn_unit_uid_entries^infantry_spawn_reconcile_unit_index
					}
				}
			}
			set_temp_variable = { infantry_spawn_lookup_lot_uid = infantry_spawn_unit_lot_uid_entries^infantry_spawn_reconcile_unit_index }
			infantry_spawn_find_lot_row = yes
			if = {
				limit = { check_variable = { infantry_spawn_lot_row_index > constant:infantry_spawn_event.invalid_index } }
				set_variable = { infantry_spawn_unit_previous_status_entries^infantry_spawn_reconcile_unit_index = infantry_spawn_unit_status_entries^infantry_spawn_reconcile_unit_index }
				set_variable = { infantry_spawn_unit_status_entries^infantry_spawn_reconcile_unit_index = constant:infantry_spawn_unit_status.unaccounted }
				if = {
					limit = { check_variable = { infantry_spawn_active_division_count > 0 } }
					subtract_from_variable = { infantry_spawn_active_division_count = 1 }
				}
				if = {
					limit = { check_variable = { infantry_spawn_lot_unit_count_entries^infantry_spawn_lot_row_index > 0 } }
					subtract_from_variable = { infantry_spawn_lot_unit_count_entries^infantry_spawn_lot_row_index = 1 }
				}
				add_to_variable = { infantry_spawn_unaccounted_division_count = 1 }
				set_country_flag = infantry_spawn_unaccounted_formation_detected
				add_to_temp_variable = { infantry_spawn_reconcile_lost_unit_count = 1 }
			}
			else = { infantry_spawn_mark_ledger_invariant_failure = yes }
		}
		add_to_temp_variable = { infantry_spawn_reconcile_unit_index = 1 }
	}
	set_temp_variable = { infantry_spawn_reconcile_lot_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_reconcile_lot_index value = infantry_spawn_lot_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_reconcile_lot_index value = constant:infantry_spawn_lot_state.demobilized compare = less_than }
				check_variable = { infantry_spawn_lot_unit_count_entries^infantry_spawn_reconcile_lot_index = 0 }
			}
			set_variable = { infantry_spawn_lot_previous_status_entries^infantry_spawn_reconcile_lot_index = infantry_spawn_lot_status_entries^infantry_spawn_reconcile_lot_index }
			set_variable = { infantry_spawn_lot_status_entries^infantry_spawn_reconcile_lot_index = constant:infantry_spawn_lot_state.unaccounted }
		}
		add_to_temp_variable = { infantry_spawn_reconcile_lot_index = 1 }
	}
	if = {
		limit = { check_variable = { infantry_spawn_reconcile_lost_unit_count > 0 } }
		infantry_spawn_refresh_claimant_loyal_lot_counts = yes
		infantry_spawn_close_resolved_generations = yes
	}
	clear_temp_array = infantry_spawn_live_unit_uid_snapshot
}

infantry_spawn_find_generation_row = {
	set_temp_variable = { infantry_spawn_generation_row_index = constant:infantry_spawn_event.invalid_index }
	set_temp_variable = { infantry_spawn_ledger_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_generation_row_index < constant:infantry_spawn_event.first_index }
			check_variable = { var = infantry_spawn_ledger_scan_index value = infantry_spawn_generation_uid_entries^num compare = less_than }
		}
		if = {
			limit = { check_variable = { var = infantry_spawn_generation_uid_entries^infantry_spawn_ledger_scan_index value = infantry_spawn_lookup_generation_uid compare = equals } }
			set_temp_variable = { infantry_spawn_generation_row_index = infantry_spawn_ledger_scan_index }
		}
		add_to_temp_variable = { infantry_spawn_ledger_scan_index = 1 }
	}
}

infantry_spawn_append_current_generation_row = {
	if = {
		limit = { infantry_spawn_generation_ledger_is_aligned = no }
		infantry_spawn_mark_ledger_invariant_failure = yes
	}
	else = {
		set_temp_variable = { infantry_spawn_lookup_generation_uid = infantry_spawn_current_generation_uid }
		infantry_spawn_find_generation_row = yes
		if = {
			limit = { check_variable = { infantry_spawn_generation_row_index > constant:infantry_spawn_event.invalid_index } }
			infantry_spawn_mark_ledger_invariant_failure = yes
		}
		else = {
			add_to_array = { array = infantry_spawn_generation_uid_entries value = infantry_spawn_current_generation_uid }
			add_to_array = { array = infantry_spawn_generation_created_date_entries value = global.date }
			add_to_array = { array = infantry_spawn_generation_evolution_stage_entries value = infantry_spawn_current_evolution_stage }
			add_to_array = { array = infantry_spawn_generation_profile_entries value = infantry_spawn_current_generation_profile }
			add_to_array = { array = infantry_spawn_generation_eligible_state_count_entries value = infantry_spawn_current_eligible_state_count }
			add_to_array = { array = infantry_spawn_generation_selected_state_count_entries value = infantry_spawn_current_selected_state_count }
			add_to_array = { array = infantry_spawn_generation_lot_count_entries value = infantry_spawn_current_generation_lot_count }
			add_to_array = { array = infantry_spawn_generation_unit_count_entries value = infantry_spawn_current_generation_unit_count }
			add_to_array = { array = infantry_spawn_generation_status_entries value = infantry_spawn_current_generation_status }
		}
	}
}

# Inputs: infantry_spawn_obligation_payment_index,
# infantry_spawn_obligation_payment_lot_index, and
# infantry_spawn_obligation_payment_amount. This updates one exact obligation
# row and both country/lot accounting surfaces as one ledger transaction.
infantry_spawn_apply_exact_obligation_payment = {
	set_temp_variable = { infantry_spawn_obligation_payment_applied = 0 }
	set_temp_variable = { infantry_spawn_obligation_payment_accounting_valid = 0 }
	if = {
		limit = {
			check_variable = { infantry_spawn_obligation_payment_index > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_obligation_payment_index value = infantry_spawn_obligation_uid_entries^num compare = less_than }
			check_variable = { infantry_spawn_obligation_payment_lot_index > constant:infantry_spawn_event.invalid_index }
			check_variable = { var = infantry_spawn_obligation_payment_lot_index value = infantry_spawn_lot_uid_entries^num compare = less_than }
			check_variable = { infantry_spawn_obligation_payment_amount > 0 }
			check_variable = { var = infantry_spawn_obligation_payment_amount value = infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_obligation_payment_index compare = less_than_or_equals }
			OR = {
				check_variable = { infantry_spawn_obligation_status_entries^infantry_spawn_obligation_payment_index = constant:infantry_spawn_obligation_state.outstanding }
				check_variable = { infantry_spawn_obligation_status_entries^infantry_spawn_obligation_payment_index = constant:infantry_spawn_obligation_state.servicing }
			}
		}
		set_temp_variable = { infantry_spawn_obligation_payment_profile = infantry_spawn_obligation_resource_profile_entries^infantry_spawn_obligation_payment_index }
		if = {
			limit = { check_variable = { infantry_spawn_obligation_payment_profile = constant:infantry_spawn_equipment_family.manpower } }
			if = {
				limit = {
					check_variable = { var = infantry_spawn_manpower_liability value = infantry_spawn_obligation_payment_amount compare = greater_than_or_equals }
					check_variable = { var = infantry_spawn_lot_outstanding_manpower_entries^infantry_spawn_obligation_payment_lot_index value = infantry_spawn_obligation_payment_amount compare = greater_than_or_equals }
				}
				set_temp_variable = { infantry_spawn_obligation_payment_accounting_valid = 1 }
				subtract_from_variable = { infantry_spawn_manpower_liability = infantry_spawn_obligation_payment_amount }
				subtract_from_variable = { infantry_spawn_lot_outstanding_manpower_entries^infantry_spawn_obligation_payment_lot_index = infantry_spawn_obligation_payment_amount }
			}
		}
		else = {
			set_temp_variable = { infantry_spawn_obligation_payment_debt_value = infantry_spawn_obligation_outstanding_debt_value_entries^infantry_spawn_obligation_payment_index }
			if = {
				limit = { check_variable = { var = infantry_spawn_obligation_payment_amount value = infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_obligation_payment_index compare = less_than } }
				multiply_temp_variable = { infantry_spawn_obligation_payment_debt_value = infantry_spawn_obligation_payment_amount }
				divide_temp_variable = { var = infantry_spawn_obligation_payment_debt_value value = infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_obligation_payment_index if_zero = constant:infantry_spawn_factor.standard }
			}
			if = {
				limit = {
					check_variable = { var = infantry_spawn_obligation_outstanding_debt_value_entries^infantry_spawn_obligation_payment_index value = infantry_spawn_obligation_payment_debt_value compare = greater_than_or_equals }
					check_variable = { var = infantry_spawn_lot_outstanding_debt_entries^infantry_spawn_obligation_payment_lot_index value = infantry_spawn_obligation_payment_debt_value compare = greater_than_or_equals }
					check_variable = { var = infantry_spawn_equipment_debt value = infantry_spawn_obligation_payment_debt_value compare = greater_than_or_equals }
				}
				set_temp_variable = { infantry_spawn_obligation_payment_accounting_valid = 1 }
				subtract_from_variable = { infantry_spawn_obligation_outstanding_debt_value_entries^infantry_spawn_obligation_payment_index = infantry_spawn_obligation_payment_debt_value }
				subtract_from_variable = { infantry_spawn_lot_outstanding_debt_entries^infantry_spawn_obligation_payment_lot_index = infantry_spawn_obligation_payment_debt_value }
				subtract_from_variable = { infantry_spawn_equipment_debt = infantry_spawn_obligation_payment_debt_value }
			}
		}
		if = {
			limit = { check_variable = { infantry_spawn_obligation_payment_accounting_valid > 0 } }
			subtract_from_variable = { infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_obligation_payment_index = infantry_spawn_obligation_payment_amount }
			add_to_variable = { infantry_spawn_obligation_paid_amount_entries^infantry_spawn_obligation_payment_index = infantry_spawn_obligation_payment_amount }
			if = {
				limit = { check_variable = { infantry_spawn_obligation_payment_is_material > 0 } }
				add_to_variable = { infantry_spawn_obligation_salvageable_paid_amount_entries^infantry_spawn_obligation_payment_index = infantry_spawn_obligation_payment_amount }
			}
			if = {
				limit = { check_variable = { infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_obligation_payment_index < 1 } }
				set_variable = { infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_obligation_payment_index = 0 }
				set_variable = { infantry_spawn_obligation_outstanding_debt_value_entries^infantry_spawn_obligation_payment_index = 0 }
				set_variable = { infantry_spawn_obligation_status_entries^infantry_spawn_obligation_payment_index = constant:infantry_spawn_obligation_state.settled }
			}
			else = { set_variable = { infantry_spawn_obligation_status_entries^infantry_spawn_obligation_payment_index = constant:infantry_spawn_obligation_state.servicing } }
			set_temp_variable = { infantry_spawn_obligation_payment_applied = 1 }
		}
	}
	if = {
		limit = { check_variable = { infantry_spawn_obligation_payment_applied = 0 } }
		infantry_spawn_mark_ledger_invariant_failure = yes
	}
}

				add_to_array = { array = infantry_spawn_obligation_outstanding_amount_entries value = infantry_spawn_current_obligation_outstanding }
				add_to_array = { array = infantry_spawn_obligation_outstanding_debt_value_entries value = infantry_spawn_current_obligation_debt_value }
				add_to_array = { array = infantry_spawn_obligation_status_entries value = infantry_spawn_current_obligation_status }
				if = {
					limit = { check_variable = { infantry_spawn_current_obligation_resource_profile = constant:infantry_spawn_equipment_family.manpower } }
					add_to_variable = { infantry_spawn_manpower_liability = infantry_spawn_current_obligation_outstanding }
					add_to_variable = { infantry_spawn_lot_outstanding_manpower_entries^infantry_spawn_lot_row_index = infantry_spawn_current_obligation_outstanding }
				}
				else = {
					add_to_variable = { infantry_spawn_equipment_debt = infantry_spawn_current_obligation_debt_value }
					add_to_variable = { infantry_spawn_lot_outstanding_debt_entries^infantry_spawn_lot_row_index = infantry_spawn_current_obligation_debt_value }
				}
			}
		}
		else = { infantry_spawn_mark_ledger_invariant_failure = yes }
	}
}

# Provider callbacks build a temporary, aligned obligation manifest before

	clear_temp_array = infantry_spawn_provider_manifest_profile_entries
	clear_temp_array = infantry_spawn_provider_manifest_per_battalion_entries
	clear_temp_array = infantry_spawn_provider_manifest_component_count_entries
	clear_temp_array = infantry_spawn_provider_manifest_scale_kind_entries
	set_temp_variable = { infantry_spawn_provider_manifest_invalid = 0 }
}

# Inputs: infantry_spawn_provider_manifest_profile,
# infantry_spawn_provider_manifest_per_battalion,
# infantry_spawn_provider_manifest_component_count, and
# infantry_spawn_provider_manifest_scale_kind. The row is omitted only when
# its computed component_count * per_battalion amount is zero. A negative or
# unknown scale is rejected so no malformed row can reach the persistent
# ledger.
infantry_spawn_queue_provider_obligation = {
	set_temp_variable = { infantry_spawn_provider_manifest_queued_amount = infantry_spawn_provider_manifest_component_count }
	multiply_temp_variable = { infantry_spawn_provider_manifest_queued_amount = infantry_spawn_provider_manifest_per_battalion }
	if = {
		limit = { check_variable = { infantry_spawn_provider_manifest_queued_amount > 0 } }
		if = {
			limit = {
				OR = {
					check_variable = { infantry_spawn_provider_manifest_scale_kind = constant:infantry_spawn_provider_manifest_scale_kind.manpower }
					check_variable = { infantry_spawn_provider_manifest_scale_kind = constant:infantry_spawn_provider_manifest_scale_kind.equipment }
				}
			}
			add_to_temp_array = { array = infantry_spawn_provider_manifest_profile_entries value = infantry_spawn_provider_manifest_profile }
			add_to_temp_array = { array = infantry_spawn_provider_manifest_per_battalion_entries value = infantry_spawn_provider_manifest_per_battalion }
			add_to_temp_array = { array = infantry_spawn_provider_manifest_component_count_entries value = infantry_spawn_provider_manifest_component_count }
			add_to_temp_array = { array = infantry_spawn_provider_manifest_scale_kind_entries value = infantry_spawn_provider_manifest_scale_kind }
		}
		else = { set_temp_variable = { infantry_spawn_provider_manifest_invalid = 1 } }
	}
	else_if = {
		limit = { check_variable = { infantry_spawn_provider_manifest_queued_amount < 0 } }
		set_temp_variable = { infantry_spawn_provider_manifest_invalid = 1 }
	}
}

# Commits every queued manifest row as one exact obligation. The provider's
# current start factors are applied only here, so all resources use the same
# scaling semantics as ordinary unit needs.
infantry_spawn_record_provider_obligation_manifest = {
	if = {
		limit = { check_variable = { infantry_spawn_provider_manifest_invalid > 0 } }
		infantry_spawn_mark_ledger_invariant_failure = yes
	}
	# Keep the manifest's committed shape and accounting totals available to the
	# owning Event 19 transaction after the callback clears its working arrays.
	# Rows with a zero scaled amount are intentionally omitted, so this count is
	# the exact number of persistent rows the manifest can publish.
	set_temp_variable = { infantry_spawn_provider_manifest_committed_row_count = constant:infantry_spawn_event.first_index }
	set_temp_variable = { infantry_spawn_provider_manifest_committed_manpower = constant:infantry_spawn_event.first_index }
	set_temp_variable = { infantry_spawn_provider_manifest_committed_equipment_debt = constant:infantry_spawn_event.first_index }
	set_temp_variable = { infantry_spawn_provider_manifest_commit_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = infantry_spawn_provider_manifest_commit_index
				value = infantry_spawn_provider_manifest_profile_entries^num
				compare = less_than
			}
		}
		set_temp_variable = { infantry_spawn_current_obligation_debt_value_override = 0 }
		set_temp_variable = { infantry_spawn_current_obligation_resource_profile = infantry_spawn_provider_manifest_profile_entries^infantry_spawn_provider_manifest_commit_index }
		set_temp_variable = { infantry_spawn_current_obligation_issued = infantry_spawn_provider_manifest_component_count_entries^infantry_spawn_provider_manifest_commit_index }
		multiply_temp_variable = { infantry_spawn_current_obligation_issued = infantry_spawn_provider_manifest_per_battalion_entries^infantry_spawn_provider_manifest_commit_index }
		if = {
			limit = { check_variable = { infantry_spawn_provider_manifest_scale_kind_entries^infantry_spawn_provider_manifest_commit_index = constant:infantry_spawn_provider_manifest_scale_kind.manpower } }
			multiply_temp_variable = { infantry_spawn_current_obligation_issued = infantry_spawn_current_start_manpower }
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_provider_manifest_scale_kind_entries^infantry_spawn_provider_manifest_commit_index = constant:infantry_spawn_provider_manifest_scale_kind.equipment } }
			multiply_temp_variable = { infantry_spawn_current_obligation_issued = infantry_spawn_current_start_equipment }
		}
		else = { infantry_spawn_mark_ledger_invariant_failure = yes }
		set_temp_variable = { infantry_spawn_current_obligation_paid = 0 }
		set_temp_variable = { infantry_spawn_current_obligation_outstanding = infantry_spawn_current_obligation_issued }
		set_temp_variable = { infantry_spawn_current_obligation_status = constant:infantry_spawn_obligation_state.outstanding }
		if = {
			limit = { check_variable = { infantry_spawn_current_obligation_issued > 0 } }
			# Provider 508 cohorts are materialized and paid by the shared
			# alien-infantry API. Do not append a stale Event 019 obligation row
			# after that API transaction; ordinary provider-neutral request
			# overhead remains owned by Event 019 outside this row ledger.
			if = {
				limit = {
					NOT = {
						check_variable = {
							var = infantry_spawn_current_registry_provider_id
							value = constant:chaos_unit_family_event16_alien_infantry.provider_id
							compare = equals
						}
					}
				}
				infantry_spawn_append_current_obligation_row = yes
			}
			if = {
				limit = { NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure } }
				add_to_temp_variable = { infantry_spawn_provider_manifest_committed_row_count = constant:infantry_spawn_muster_gui_value.one }
				if = {
					limit = { check_variable = { infantry_spawn_current_obligation_resource_profile = constant:infantry_spawn_equipment_family.manpower } }
					add_to_temp_variable = { infantry_spawn_provider_manifest_committed_manpower = infantry_spawn_current_obligation_outstanding }
				}
				else = { add_to_temp_variable = { infantry_spawn_provider_manifest_committed_equipment_debt = infantry_spawn_current_obligation_debt_value } }
			}
		}
		add_to_temp_variable = { infantry_spawn_provider_manifest_commit_index = 1 }
	}
}

infantry_spawn_compaction_commit_candidate_generation = {
	if = {
		limit = {
			infantry_spawn_ledger_compaction_may_advance = yes
			infantry_spawn_compaction_lot_snapshot_matches = yes
			infantry_spawn_compaction_candidate_generation_is_detached_summary = yes
			check_variable = { infantry_spawn_compaction_candidate_generation_proven = constant:infantry_spawn_ledger_compaction.one }
		}
		set_temp_variable = { infantry_spawn_compaction_expected_generation_rows = infantry_spawn_compaction_snapshot_generation_rows }
		subtract_from_temp_variable = { infantry_spawn_compaction_expected_generation_rows = constant:infantry_spawn_ledger_compaction.one }
		set_temp_variable = { infantry_spawn_compaction_remove_index = infantry_spawn_compaction_candidate_generation_index }
		infantry_spawn_compaction_remove_generation_row = yes
		if = {
			limit = {
				infantry_spawn_all_ledgers_are_aligned = yes
				infantry_spawn_claimant_ledgers_are_aligned = yes
				infantry_spawn_compaction_auxiliary_evidence_is_aligned = yes
				check_variable = { var = infantry_spawn_generation_uid_entries^num value = infantry_spawn_compaction_expected_generation_rows compare = equals }
				check_variable = { var = infantry_spawn_selected_state_generation_uid_entries^num value = infantry_spawn_compaction_snapshot_selected_state_rows compare = equals }
				check_variable = { var = infantry_spawn_lot_uid_entries^num value = infantry_spawn_compaction_snapshot_lot_rows compare = equals }
				check_variable = { var = infantry_spawn_template_uid_entries^num value = infantry_spawn_compaction_snapshot_template_rows compare = equals }
				check_variable = { var = infantry_spawn_template_component_template_uid_entries^num value = infantry_spawn_compaction_snapshot_component_rows compare = equals }
				check_variable = { var = infantry_spawn_unit_uid_entries^num value = infantry_spawn_compaction_snapshot_unit_rows compare = equals }
				check_variable = { var = infantry_spawn_obligation_uid_entries^num value = infantry_spawn_compaction_snapshot_obligation_rows compare = equals }
				check_variable = { var = infantry_spawn_locked_template_uid_entries^num value = infantry_spawn_compaction_snapshot_locked_template_rows compare = equals }
				check_variable = { var = infantry_spawn_spawn_only_template_uid_entries^num value = infantry_spawn_compaction_snapshot_spawn_only_template_rows compare = equals }
				check_variable = { var = infantry_spawn_technology_locked_template_uid_entries^num value = infantry_spawn_compaction_snapshot_technology_template_rows compare = equals }
				check_variable = { var = infantry_spawn_technology_locked_unit_uid_entries^num value = infantry_spawn_compaction_snapshot_technology_unit_rows compare = equals }
				check_variable = { var = infantry_spawn_transfer_eligible_unit_uid_entries^num value = infantry_spawn_compaction_snapshot_transfer_unit_rows compare = equals }
				check_variable = { var = infantry_spawn_pretechnology_template_uid_entries^num value = infantry_spawn_compaction_snapshot_pretechnology_template_rows compare = equals }
				check_variable = { var = infantry_spawn_pretechnology_unit_uid_entries^num value = infantry_spawn_compaction_snapshot_pretechnology_unit_rows compare = equals }
				check_variable = { var = infantry_spawn_achievement_pretech_unit_uid_entries^num value = infantry_spawn_compaction_snapshot_achievement_pretech_rows compare = equals }
				check_variable = { var = infantry_spawn_achievement_composition_disqualified_unit_uid_entries^num value = infantry_spawn_compaction_snapshot_composition_evidence_rows compare = equals }
				check_variable = { var = infantry_spawn_achievement_supervised_demobilized_lot_uids^num value = infantry_spawn_compaction_snapshot_supervised_demobilization_rows compare = equals }
			}
			set_temp_variable = { infantry_spawn_compaction_restart_index = infantry_spawn_compaction_candidate_generation_index }
			set_temp_variable = { infantry_spawn_compaction_removed_generation_index = infantry_spawn_compaction_candidate_generation_index }
			set_temp_variable = { infantry_spawn_compaction_removed_generation_uid = infantry_spawn_compaction_candidate_generation_uid }
			add_to_variable = { infantry_spawn_compaction_generations_removed_this_cycle = constant:infantry_spawn_ledger_compaction.one }
			infantry_spawn_compaction_clear_generation_candidate = yes
			set_variable = { infantry_spawn_compaction_generation_search_cursor = infantry_spawn_compaction_restart_index }
			if = {
				limit = { check_variable = { var = infantry_spawn_compaction_generations_removed_this_cycle value = constant:infantry_spawn_ledger_compaction.maximum_generations_per_cycle compare = less_than } }
				set_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_find_generation }
			}
			else = { set_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_finish } }
			infantry_spawn_compaction_adjust_muster_board_after_generation_commit = yes
		}
		else = { infantry_spawn_compaction_fail_closed = yes }
	}
	else = { infantry_spawn_compaction_skip_generation_candidate = yes }
}

# COUNTRY scope. Called only from the existing Event 19 country pulse. A due or
# dirty cycle advances exactly one bounded scan phase (or one capped commit).
infantry_spawn_run_lifetime_ledger_compaction = {
	if = {
		limit = { NOT = { has_variable = infantry_spawn_compaction_phase } }
		set_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_idle }
	}
	if = {
		limit = { NOT = { has_variable = infantry_spawn_compaction_due_date } }
		set_variable = { infantry_spawn_compaction_due_date = global.date }
		set_country_flag = infantry_spawn_ledger_compaction_dirty
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = infantry_spawn_compaction_observed_generation_rows }
				NOT = { has_variable = infantry_spawn_compaction_observed_lot_rows }
				NOT = { has_variable = infantry_spawn_compaction_observed_unit_rows }
				NOT = { has_variable = infantry_spawn_compaction_observed_obligation_rows }
				NOT = { has_variable = infantry_spawn_compaction_observed_active_lots }
				NOT = { check_variable = { var = infantry_spawn_compaction_observed_generation_rows value = infantry_spawn_generation_uid_entries^num compare = equals } }
				NOT = { check_variable = { var = infantry_spawn_compaction_observed_lot_rows value = infantry_spawn_lot_uid_entries^num compare = equals } }
				NOT = { check_variable = { var = infantry_spawn_compaction_observed_unit_rows value = infantry_spawn_unit_uid_entries^num compare = equals } }
				NOT = { check_variable = { var = infantry_spawn_compaction_observed_obligation_rows value = infantry_spawn_obligation_uid_entries^num compare = equals } }
				NOT = { check_variable = { var = infantry_spawn_compaction_observed_active_lots value = infantry_spawn_active_lot_count compare = equals } }
			}
		}
		set_country_flag = infantry_spawn_ledger_compaction_dirty
	}
	if = {
		limit = {
			NOT = { has_country_flag = infantry_spawn_ledger_compaction_active }
			NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
			OR = {
				has_country_flag = infantry_spawn_ledger_compaction_dirty
				check_variable = { var = infantry_spawn_compaction_due_date value = global.date compare = less_than_or_equals }
			}
		}
		infantry_spawn_compaction_start_cycle = yes
	}

	if = {
		limit = {
			has_country_flag = infantry_spawn_ledger_compaction_active
			OR = {
				has_country_flag = infantry_spawn_ledger_invariant_failure
				infantry_spawn_all_ledgers_are_aligned = no
				infantry_spawn_claimant_ledgers_are_aligned = no
				infantry_spawn_compaction_auxiliary_evidence_is_aligned = no
			}
		}
		infantry_spawn_compaction_fail_closed = yes
	}
	else_if = {
		limit = {
			has_country_flag = infantry_spawn_ledger_compaction_active
			infantry_spawn_ledger_compaction_may_advance = yes
		}
		if = {
			limit = {
				check_variable = { var = infantry_spawn_compaction_phase value = constant:infantry_spawn_ledger_compaction.phase_find_lot compare = greater_than }
				check_variable = { var = infantry_spawn_compaction_phase value = constant:infantry_spawn_ledger_compaction.phase_find_generation compare = less_than }
				infantry_spawn_compaction_lot_snapshot_matches = no
			}
			infantry_spawn_compaction_restart_lot_proof = yes
		}
		else_if = {
			limit = {
				check_variable = { var = infantry_spawn_compaction_phase value = constant:infantry_spawn_ledger_compaction.phase_find_generation compare = greater_than }
				check_variable = { var = infantry_spawn_compaction_phase value = constant:infantry_spawn_ledger_compaction.phase_finish compare = less_than }
				infantry_spawn_compaction_lot_snapshot_matches = no
			}
			infantry_spawn_compaction_restart_generation_proof = yes
		}

		if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_find_lot } }
			infantry_spawn_compaction_find_lot_candidate = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_lot_generation } }
			infantry_spawn_compaction_scan_candidate_lot_generation = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_lot_template_references } }
			infantry_spawn_compaction_scan_lot_template_references = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_units } }
			infantry_spawn_compaction_scan_candidate_units = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_obligations } }
			infantry_spawn_compaction_scan_candidate_obligations = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_selected_states } }
			infantry_spawn_compaction_scan_candidate_selected_states = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_templates } }
			infantry_spawn_compaction_scan_candidate_template = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_components } }
			infantry_spawn_compaction_scan_candidate_components = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_achievement_evidence } }
			infantry_spawn_compaction_scan_candidate_achievement_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_technology_unit_evidence } }
			infantry_spawn_compaction_scan_candidate_technology_unit_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_transfer_unit_evidence } }
			infantry_spawn_compaction_scan_candidate_transfer_unit_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_pretechnology_unit_evidence } }
			infantry_spawn_compaction_scan_candidate_pretechnology_unit_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_composition_evidence } }
			infantry_spawn_compaction_scan_candidate_composition_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_supervised_demobilization_evidence } }
			infantry_spawn_compaction_scan_candidate_supervised_demobilization_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_locked_template_evidence } }
			infantry_spawn_compaction_scan_candidate_locked_template_evidence = yes
		}
		else_if = {

			infantry_spawn_compaction_scan_candidate_spawn_only_template_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_technology_template_evidence } }
			infantry_spawn_compaction_scan_candidate_technology_template_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_pretechnology_template_evidence } }
			infantry_spawn_compaction_scan_candidate_pretechnology_template_evidence = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_commit_lot } }
			infantry_spawn_compaction_commit_candidate_lot = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_find_generation } }
			infantry_spawn_compaction_find_generation_candidate = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_generation_lots } }
			infantry_spawn_compaction_scan_generation_lot_references = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_generation_units } }
			infantry_spawn_compaction_scan_generation_unit_references = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_compaction_phase = constant:infantry_spawn_ledger_compaction.phase_scan_generation_obligations } }
			infantry_spawn_compaction_scan_generation_obligation_references = yes
