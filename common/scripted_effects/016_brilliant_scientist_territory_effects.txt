# ============================================================================
# CHAOS REDUX - EVENT 016 KRUGER STATE TERRITORY EFFECTS
# ============================================================================
# Overview:
# - Builds a deterministic, bounded territory plan from live Event 016 facility
#   markers and directly adjacent strategic support states.
# - Freezes membership with state flags and persistent host/capital targets;
#   ownership is never changed by this file.
# - Preserves a viable former host before accepting any candidate state.
# - Commits verification last and supplies an immediate, non-rebuilding
#   revalidation effect for the parent-owned KRG formation transaction.
# - Invalid plans clear every state mark and target and expose a stable reason;
#   no route substitution or arbitrary-state fallback exists here.
# ============================================================================

# COUNTRY scope. The one-time world scan is intentional: selected states may
# have changed owner since an earlier plan. This effect is never called by a
# recurring on-action.
brilliant_scientist_clear_formation_territory_state_marks = {
	every_state = {
		limit = {
			OR = {
				has_state_flag = brilliant_scientist_formation_territory_selected
				has_state_flag = brilliant_scientist_formation_territory_capital
				has_state_flag = brilliant_scientist_formation_territory_connected_facility
				has_state_flag = brilliant_scientist_formation_territory_multi_site_facility
				has_state_flag = brilliant_scientist_formation_territory_support
			}
		}
		clr_state_flag = brilliant_scientist_formation_territory_selected
		clr_state_flag = brilliant_scientist_formation_territory_capital
		clr_state_flag = brilliant_scientist_formation_territory_connected_facility
		clr_state_flag = brilliant_scientist_formation_territory_multi_site_facility
		clr_state_flag = brilliant_scientist_formation_territory_support
	}
}

# COUNTRY scope. Called only after the source route flags have been proven
# mutually exclusive.
brilliant_scientist_prepare_formation_territory_route_identity = {
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_charter_accepted
				has_country_flag = brilliant_scientist_sovereignty_response_charter
			}
		}
		set_country_flag = brilliant_scientist_formation_territory_route_charter
		set_variable = { brilliant_scientist_formation_territory_route = constant:brilliant_scientist_territory_route.charter }
	}
	else_if = {
		limit = { has_country_flag = brilliant_scientist_enclave_settlement }
		set_country_flag = brilliant_scientist_formation_territory_route_enclave
		set_variable = { brilliant_scientist_formation_territory_route = constant:brilliant_scientist_territory_route.enclave }
	}
	else_if = {
		limit = { has_country_flag = brilliant_scientist_rebellion_committed }
		set_country_flag = brilliant_scientist_formation_territory_route_rebellion
		set_variable = { brilliant_scientist_formation_territory_route = constant:brilliant_scientist_territory_route.rebellion }
	}
}

# COUNTRY scope. Captures host survival capacity and a route-specific upper
# bound derived from the actual Event 016 facility count and project history.
brilliant_scientist_initialize_formation_territory_capacity = {
	brilliant_scientist_refresh_project_counts = yes
	set_variable = { brilliant_scientist_formation_host_remaining_state_count = num_owned_states }
	set_variable = { brilliant_scientist_formation_host_remaining_factory_count = num_of_factories }
	set_variable = { brilliant_scientist_formation_host_remaining_core_state_count = constant:brilliant_scientist_territory_plan.zero }
	set_variable = { brilliant_scientist_formation_eligible_facility_state_count = constant:brilliant_scientist_territory_plan.zero }
	every_owned_state = {
		limit = { is_core_of = ROOT }
		ROOT = {
			add_to_variable = { brilliant_scientist_formation_host_remaining_core_state_count = constant:brilliant_scientist_territory_plan.one }
		}
	}
	every_owned_state = {
		limit = {
			brilliant_scientist_territory_state_is_host_candidate = yes
			brilliant_scientist_territory_state_has_event_facility = yes
		}
		ROOT = {
			add_to_variable = { brilliant_scientist_formation_eligible_facility_state_count = constant:brilliant_scientist_territory_plan.one }
		}
	}
	if = {
		limit = { has_country_flag = brilliant_scientist_formation_territory_route_enclave }
		set_variable = { brilliant_scientist_formation_target_state_count = constant:brilliant_scientist_territory_limit.enclave_maximum_states }
	}
	else_if = {
		limit = { has_country_flag = brilliant_scientist_formation_territory_route_charter }
		set_variable = { brilliant_scientist_formation_target_state_count = brilliant_scientist_formation_eligible_facility_state_count }
		add_to_variable = { brilliant_scientist_formation_target_state_count = constant:brilliant_scientist_territory_limit.base_support_states }
		clamp_variable = {
			var = brilliant_scientist_formation_target_state_count
			min = constant:brilliant_scientist_territory_limit.charter_minimum_states
			max = constant:brilliant_scientist_territory_limit.charter_maximum_states
		}
	}
	else_if = {
		limit = { has_country_flag = brilliant_scientist_formation_territory_route_rebellion }
		set_variable = { brilliant_scientist_formation_target_state_count = brilliant_scientist_formation_eligible_facility_state_count }
		add_to_variable = { brilliant_scientist_formation_target_state_count = constant:brilliant_scientist_territory_limit.base_support_states }
		if = {
			limit = {
				check_variable = {
					var = brilliant_scientist_project_deployment_count
					value = constant:brilliant_scientist_territory_plan.zero
					compare = greater_than
				}
			}
			add_to_variable = { brilliant_scientist_formation_target_state_count = constant:brilliant_scientist_territory_limit.advanced_project_support_states }
		}
		if = {
			limit = {
				check_variable = {
					var = brilliant_scientist_project_weaponization_count
					value = constant:brilliant_scientist_territory_plan.zero
					compare = greater_than
				}
			}
			add_to_variable = { brilliant_scientist_formation_target_state_count = constant:brilliant_scientist_territory_limit.advanced_project_support_states }
		}
		if = {
			limit = { has_country_flag = brilliant_scientist_multi_site_network_verified }
			add_to_variable = { brilliant_scientist_formation_target_state_count = constant:brilliant_scientist_territory_limit.multi_site_support_states }
		}
		clamp_variable = {
			var = brilliant_scientist_formation_target_state_count
			min = constant:brilliant_scientist_territory_limit.rebellion_minimum_states
			max = constant:brilliant_scientist_territory_limit.rebellion_maximum_states
		}
	}
}

# STATE scope. ROOT must be the former host. The helper reserves nothing unless
# the host remains above every state, core, and factory floor after selection.
brilliant_scientist_try_reserve_formation_territory_state = {
	set_temp_variable = { brilliant_scientist_territory_candidate_selected = constant:brilliant_scientist_territory_plan.zero }
	set_temp_variable = { brilliant_scientist_territory_candidate_factory_count = building_level@industrial_complex }
	add_to_temp_variable = { brilliant_scientist_territory_candidate_factory_count = building_level@arms_factory }
	add_to_temp_variable = { brilliant_scientist_territory_candidate_factory_count = building_level@dockyard }
	ROOT = {
		set_temp_variable = { brilliant_scientist_territory_host_states_after_candidate = brilliant_scientist_formation_host_remaining_state_count }
		set_temp_variable = { brilliant_scientist_territory_host_cores_after_candidate = brilliant_scientist_formation_host_remaining_core_state_count }
		set_temp_variable = { brilliant_scientist_territory_host_factories_after_candidate = brilliant_scientist_formation_host_remaining_factory_count }
	}
	subtract_from_temp_variable = { brilliant_scientist_territory_host_states_after_candidate = constant:brilliant_scientist_territory_plan.one }
	subtract_from_temp_variable = { brilliant_scientist_territory_host_factories_after_candidate = brilliant_scientist_territory_candidate_factory_count }
	if = {
		limit = { is_core_of = ROOT }
		subtract_from_temp_variable = { brilliant_scientist_territory_host_cores_after_candidate = constant:brilliant_scientist_territory_plan.one }
	}
	if = {
		limit = {
			brilliant_scientist_territory_state_is_host_candidate = yes
			NOT = { has_state_flag = brilliant_scientist_formation_territory_selected }
			ROOT = {
				check_variable = {
					var = brilliant_scientist_formation_selected_state_count
					value = brilliant_scientist_formation_target_state_count
					compare = less_than
				}
			}
			check_variable = {
				var = brilliant_scientist_territory_host_states_after_candidate
				value = constant:brilliant_scientist_formation.host_retained_states
				compare = greater_than_or_equals
			}
			check_variable = {
				var = brilliant_scientist_territory_host_cores_after_candidate
				value = constant:brilliant_scientist_territory_limit.host_retained_core_states
				compare = greater_than_or_equals
			}
			check_variable = {
				var = brilliant_scientist_territory_host_factories_after_candidate
				value = constant:brilliant_scientist_formation.host_retained_factories
				compare = greater_than_or_equals
			}
		}
		set_state_flag = brilliant_scientist_formation_territory_selected
		ROOT = {
			add_to_variable = { brilliant_scientist_formation_selected_state_count = constant:brilliant_scientist_territory_plan.one }
			set_variable = { brilliant_scientist_formation_host_remaining_state_count = brilliant_scientist_territory_host_states_after_candidate }
			set_variable = { brilliant_scientist_formation_host_remaining_core_state_count = brilliant_scientist_territory_host_cores_after_candidate }
			set_variable = { brilliant_scientist_formation_host_remaining_factory_count = brilliant_scientist_territory_host_factories_after_candidate }
		}
		set_temp_variable = { brilliant_scientist_territory_candidate_selected = constant:brilliant_scientist_territory_plan.one }
	}
}

		limit = { brilliant_scientist_territory_state_is_viable_capital = yes }
		brilliant_scientist_try_reserve_formation_territory_state = yes

	if = {
		limit = {
			check_variable = {
				var = brilliant_scientist_territory_candidate_selected
				value = constant:brilliant_scientist_territory_plan.one
				compare = equals
			}
		}
		set_state_flag = brilliant_scientist_formation_territory_capital
		save_global_event_target_as = brilliant_scientist_formation_capital_state
	}
}

# COUNTRY scope. Capital priority follows the country-package specification:
# current primary, marked primary, current secondary, marked secondary, joint
# laboratory, transport facility, supplied facility, then any viable facility.
brilliant_scientist_select_formation_territory_capital = {
	if = {
		limit = {
			NOT = { has_event_target = brilliant_scientist_formation_capital_state }
			has_event_target = brilliant_scientist_primary_facility
			event_target:brilliant_scientist_primary_facility = {
				brilliant_scientist_territory_state_is_viable_capital = yes
			}
		}
		event_target:brilliant_scientist_primary_facility = {
			brilliant_scientist_try_add_formation_territory_capital = yes
		}
	}
	every_owned_state = {
		limit = {
			ROOT = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
			has_state_flag = brilliant_scientist_primary_facility
			brilliant_scientist_territory_state_is_viable_capital = yes
		}
		brilliant_scientist_try_add_formation_territory_capital = yes
	}
	if = {
		limit = {
			NOT = { has_event_target = brilliant_scientist_formation_capital_state }
			has_event_target = brilliant_scientist_secondary_facility
			event_target:brilliant_scientist_secondary_facility = {
				brilliant_scientist_territory_state_is_viable_capital = yes
			}
		}
		event_target:brilliant_scientist_secondary_facility = {
			brilliant_scientist_try_add_formation_territory_capital = yes
		}
	}
	every_owned_state = {
		limit = {
			ROOT = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
			has_state_flag = brilliant_scientist_secondary_facility
			brilliant_scientist_territory_state_is_viable_capital = yes
		}
		brilliant_scientist_try_add_formation_territory_capital = yes
	}
	if = {
		limit = {
			NOT = { has_event_target = brilliant_scientist_formation_capital_state }
			has_event_target = brilliant_scientist_joint_laboratory_site
			event_target:brilliant_scientist_joint_laboratory_site = {
				brilliant_scientist_territory_state_is_viable_capital = yes
			}
		}
		event_target:brilliant_scientist_joint_laboratory_site = {
			brilliant_scientist_try_add_formation_territory_capital = yes
		}
	}
	every_owned_state = {
		limit = {
			ROOT = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
			has_state_flag = brilliant_scientist_joint_laboratory_site
			brilliant_scientist_territory_state_is_viable_capital = yes
		}
		brilliant_scientist_try_add_formation_territory_capital = yes
	}
	every_owned_state = {
		limit = {
			ROOT = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
			OR = {
				has_state_flag = brilliant_scientist_quantum_transit_terminal
				has_state_flag = brilliant_scientist_temporal_anchor
			}
			brilliant_scientist_territory_state_is_viable_capital = yes
		}
		brilliant_scientist_try_add_formation_territory_capital = yes
	}
	every_owned_state = {
		limit = {
			ROOT = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
			brilliant_scientist_territory_state_is_viable_capital = yes
			OR = {
				supply_node > constant:brilliant_scientist_territory_plan.zero
				rail_way > constant:brilliant_scientist_territory_plan.zero
				industrial_complex > constant:brilliant_scientist_territory_plan.zero
				arms_factory > constant:brilliant_scientist_territory_plan.zero
				dockyard > constant:brilliant_scientist_territory_plan.zero
			}
		}
		brilliant_scientist_try_add_formation_territory_capital = yes
	}
	every_owned_state = {
		limit = {
			ROOT = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
			brilliant_scientist_territory_state_is_viable_capital = yes
		}
		brilliant_scientist_try_add_formation_territory_capital = yes
	}
}

# COUNTRY scope. Connected facility passes are repeated so state iteration order
# cannot prevent a valid facility chain from growing outward from the capital.
brilliant_scientist_expand_connected_formation_territory = {
	set_temp_variable = { brilliant_scientist_territory_growth_pass = constant:brilliant_scientist_territory_plan.zero }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = brilliant_scientist_territory_growth_pass
				value = constant:brilliant_scientist_territory_plan.contiguous_growth_passes
				compare = less_than
			}
		}
		every_owned_state = {
			limit = {
				has_state_flag = brilliant_scientist_secondary_facility
				brilliant_scientist_territory_state_is_connected_facility_candidate = yes
			}
			brilliant_scientist_try_add_connected_formation_facility = yes
		}
		every_owned_state = {
			limit = {
				has_state_flag = brilliant_scientist_joint_laboratory_site
				brilliant_scientist_territory_state_is_connected_facility_candidate = yes
			}
			brilliant_scientist_try_add_connected_formation_facility = yes
		}
		every_owned_state = {
			limit = {
				OR = {
					has_state_flag = brilliant_scientist_quantum_transit_terminal
					has_state_flag = brilliant_scientist_temporal_anchor
				}
				brilliant_scientist_territory_state_is_connected_facility_candidate = yes
			}
			brilliant_scientist_try_add_connected_formation_facility = yes
		}
		every_owned_state = {
			limit = { brilliant_scientist_territory_state_is_connected_facility_candidate = yes }
			brilliant_scientist_try_add_connected_formation_facility = yes
		}
		add_to_temp_variable = { brilliant_scientist_territory_growth_pass = constant:brilliant_scientist_territory_plan.one }
	}
}

# COUNTRY scope. Support grows only through directly adjacent core states with
# a concrete supply, rail, port, or factory role.
brilliant_scientist_add_formation_territory_support = {
	set_temp_variable = { brilliant_scientist_territory_support_pass = constant:brilliant_scientist_territory_plan.zero }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = brilliant_scientist_territory_support_pass
				value = constant:brilliant_scientist_territory_plan.support_growth_passes
				compare = less_than
			}
		}
		every_owned_state = {
			limit = {
				brilliant_scientist_territory_state_is_support_candidate = yes
				OR = {
					supply_node > constant:brilliant_scientist_territory_plan.zero
					rail_way > constant:brilliant_scientist_territory_plan.zero
					AND = {
						is_coastal = yes
						naval_base > constant:brilliant_scientist_territory_plan.zero
					}
				}
			}
			brilliant_scientist_try_add_formation_territory_support = yes
		}
		every_owned_state = {
			limit = {
				brilliant_scientist_territory_state_is_support_candidate = yes
				OR = {
					industrial_complex > constant:brilliant_scientist_territory_plan.zero
					arms_factory > constant:brilliant_scientist_territory_plan.zero
					dockyard > constant:brilliant_scientist_territory_plan.zero
				}
			}
			brilliant_scientist_try_add_formation_territory_support = yes
		}
		every_owned_state = {
			limit = { brilliant_scientist_territory_state_is_support_candidate = yes }
			brilliant_scientist_try_add_formation_territory_support = yes
		}
		add_to_temp_variable = { brilliant_scientist_territory_support_pass = constant:brilliant_scientist_territory_plan.one }
	}
}

# COUNTRY scope. Rebuilds all plan metrics from current state marks. Orphaned
# role marks, changed ownership/control, invalid roles, or multiple capitals
# clear the scope-scan proof used by every route contract.
brilliant_scientist_refresh_formation_territory_metrics = {
	set_country_flag = brilliant_scientist_formation_territory_scope_scan_valid
	brilliant_scientist_clear_formation_territory_runtime_metrics = yes
	set_variable = { brilliant_scientist_formation_host_remaining_state_count = num_owned_states }
	set_variable = { brilliant_scientist_formation_host_remaining_factory_count = num_of_factories }
	every_owned_state = {
		limit = { is_core_of = ROOT }
		ROOT = {
			add_to_variable = { brilliant_scientist_formation_host_remaining_core_state_count = constant:brilliant_scientist_territory_plan.one }
		}
	}
	every_state = {
		limit = {
			OR = {
				has_state_flag = brilliant_scientist_formation_territory_selected
				has_state_flag = brilliant_scientist_formation_territory_capital
				has_state_flag = brilliant_scientist_formation_territory_connected_facility
				has_state_flag = brilliant_scientist_formation_territory_multi_site_facility
				has_state_flag = brilliant_scientist_formation_territory_support
			}
		}
		if = {
			limit = { has_state_flag = brilliant_scientist_formation_territory_selected }
			ROOT = {
				add_to_variable = { brilliant_scientist_formation_territory_marked_state_count = constant:brilliant_scientist_territory_plan.one }
			}
			if = {
				limit = { brilliant_scientist_territory_selected_state_is_current = yes }
				set_temp_variable = { brilliant_scientist_territory_selected_factory_count = building_level@industrial_complex }
				add_to_temp_variable = { brilliant_scientist_territory_selected_factory_count = building_level@arms_factory }
				add_to_temp_variable = { brilliant_scientist_territory_selected_factory_count = building_level@dockyard }
				ROOT = {
					add_to_variable = { brilliant_scientist_formation_selected_state_count = constant:brilliant_scientist_territory_plan.one }
					add_to_variable = { brilliant_scientist_formation_selected_factory_count = brilliant_scientist_territory_selected_factory_count }
					add_to_variable = { brilliant_scientist_formation_territory_score = constant:brilliant_scientist_formation.state_score }
					subtract_from_variable = { brilliant_scientist_formation_host_remaining_state_count = constant:brilliant_scientist_territory_plan.one }
					subtract_from_variable = { brilliant_scientist_formation_host_remaining_factory_count = brilliant_scientist_territory_selected_factory_count }
				}
				if = {
					limit = { brilliant_scientist_territory_state_has_event_facility = yes }
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_facility_state_count = constant:brilliant_scientist_territory_plan.one }
						add_to_variable = { brilliant_scientist_formation_territory_score = constant:brilliant_scientist_formation.facility_state_score }
					}
				}
				if = {
					limit = { is_core_of = ROOT }
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_selected_core_state_count = constant:brilliant_scientist_territory_plan.one }
						add_to_variable = { brilliant_scientist_formation_territory_score = constant:brilliant_scientist_formation.core_state_score }
						subtract_from_variable = { brilliant_scientist_formation_host_remaining_core_state_count = constant:brilliant_scientist_territory_plan.one }
					}
				}
				if = {
					limit = { brilliant_scientist_territory_state_has_logistics_access = yes }
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_logistics_state_count = constant:brilliant_scientist_territory_plan.one }
					}
				}
				if = {
					limit = { has_state_flag = brilliant_scientist_formation_territory_multi_site_facility }
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_multi_site_state_count = constant:brilliant_scientist_territory_plan.one }
					}
				}
				else = {
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_connected_state_count = constant:brilliant_scientist_territory_plan.one }
						add_to_variable = { brilliant_scientist_formation_territory_score = constant:brilliant_scientist_formation.connected_state_score }
					}
				}
				if = {
					limit = { has_state_flag = brilliant_scientist_formation_territory_support }
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_support_state_count = constant:brilliant_scientist_territory_plan.one }
					}
				}
				if = {
					limit = { has_state_flag = brilliant_scientist_formation_territory_capital }
					ROOT = {
						add_to_variable = { brilliant_scientist_formation_territory_capital_mark_count = constant:brilliant_scientist_territory_plan.one }
					}
				}
			}
			else = { ROOT = { clr_country_flag = brilliant_scientist_formation_territory_scope_scan_valid } }
		}
		else = { ROOT = { clr_country_flag = brilliant_scientist_formation_territory_scope_scan_valid } }
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = brilliant_scientist_formation_territory_marked_state_count
					value = brilliant_scientist_formation_selected_state_count
					compare = equals
				}
			}
		}
		clr_country_flag = brilliant_scientist_formation_territory_scope_scan_valid
	}
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = brilliant_scientist_formation_territory_capital_mark_count
					value = constant:brilliant_scientist_territory_plan.one
					compare = equals
				}
			}
		}
		clr_country_flag = brilliant_scientist_formation_territory_scope_scan_valid
	}
}

# COUNTRY scope. The caller sets a nonzero invalid reason first. Attempted
# metrics survive for reporting, while all transfer-capable marks and targets
# are removed before the stable invalid flag is written.
brilliant_scientist_invalidate_formation_territory_plan = {
	if = {
		limit = {
			check_variable = {
				var = brilliant_scientist_formation_territory_invalid_reason
				value = constant:brilliant_scientist_territory_invalid_reason.none
				compare = equals
			}
		}
		set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.route_contract }
	}
	set_variable = { brilliant_scientist_formation_attempted_state_count = brilliant_scientist_formation_selected_state_count }
	set_variable = { brilliant_scientist_formation_attempted_facility_state_count = brilliant_scientist_formation_facility_state_count }
	set_variable = { brilliant_scientist_formation_attempted_territory_score = brilliant_scientist_formation_territory_score }
	clr_country_flag = brilliant_scientist_formation_territory_verified
	clr_country_flag = brilliant_scientist_formation_territory_prepared
	clr_country_flag = brilliant_scientist_formation_territory_revalidated
	clr_country_flag = brilliant_scientist_formation_territory_scope_scan_valid
	clr_country_flag = brilliant_scientist_formation_territory_planning
	brilliant_scientist_clear_formation_territory_state_marks = yes
	if = {
		limit = { has_event_target = brilliant_scientist_formation_capital_state }
		clear_global_event_target = brilliant_scientist_formation_capital_state
	}
	if = {
		limit = { has_event_target = brilliant_scientist_formation_territory_host }
		clear_global_event_target = brilliant_scientist_formation_territory_host
	}
	brilliant_scientist_clear_formation_territory_runtime_metrics = yes
	set_country_flag = brilliant_scientist_formation_territory_invalid
}

# COUNTRY scope. Public planner entry point used by Event 016 confrontation and
# charter resolution. Verification is committed only after the complete route
# contract passes against the freshly rescanned state marks.
brilliant_scientist_prepare_formation_territory_plan = {
	brilliant_scientist_clear_formation_territory_plan = yes
	set_country_flag = brilliant_scientist_formation_territory_planning
	if = {
		limit = { NOT = { brilliant_scientist_has_single_territory_route_request = yes } }
		set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.ambiguous_route }
		brilliant_scientist_invalidate_formation_territory_plan = yes
	}
	else = {
		brilliant_scientist_prepare_formation_territory_route_identity = yes
		save_global_event_target_as = brilliant_scientist_formation_territory_host
		if = {
			limit = { NOT = { brilliant_scientist_territory_plan_host_is_eligible = yes } }
			set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.invalid_host }
			brilliant_scientist_invalidate_formation_territory_plan = yes
		}
		else_if = {
			limit = { NOT = { brilliant_scientist_project_mode_history_is_coherent = yes } }
			set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.incoherent_project_mode }
			brilliant_scientist_invalidate_formation_territory_plan = yes
		}
		else = {
			brilliant_scientist_initialize_formation_territory_capacity = yes
			if = {
				limit = {
					check_variable = {
						var = brilliant_scientist_formation_eligible_facility_state_count
						value = constant:brilliant_scientist_territory_limit.minimum_facility_states
						compare = less_than
					}
				}
				set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.no_eligible_facility }
				brilliant_scientist_invalidate_formation_territory_plan = yes
			}
			else = {
				brilliant_scientist_select_formation_territory_capital = yes
				if = {
					limit = { NOT = { has_event_target = brilliant_scientist_formation_capital_state } }
					set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.no_viable_capital }
					brilliant_scientist_invalidate_formation_territory_plan = yes
				}
				else = {
					brilliant_scientist_expand_connected_formation_territory = yes
					brilliant_scientist_add_verified_multi_site_formation_territory = yes
					brilliant_scientist_add_formation_territory_support = yes
					brilliant_scientist_refresh_formation_territory_metrics = yes
					if = {
						limit = {
							OR = {
								brilliant_scientist_charter_territory_plan_meets_contract = yes
								brilliant_scientist_enclave_territory_plan_meets_contract = yes
								brilliant_scientist_rebellion_territory_plan_meets_contract = yes
							}
						}
						brilliant_scientist_snapshot_formation_territory_plan = yes
						set_country_flag = brilliant_scientist_formation_territory_prepared
						set_country_flag = brilliant_scientist_formation_territory_revalidated
						clr_country_flag = brilliant_scientist_formation_territory_planning
						set_country_flag = brilliant_scientist_formation_territory_verified
					}
					else = {
						if = {
							limit = { NOT = { brilliant_scientist_territory_plan_capital_is_current = yes } }
							set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.no_viable_capital }
						}
						else_if = {
							limit = { NOT = { brilliant_scientist_territory_plan_preserves_host = yes } }
							set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.host_survival }
						}
						else = {
							set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.route_contract }
						}
						brilliant_scientist_invalidate_formation_territory_plan = yes
					}
				}
			}
		}
	}
}

# COUNTRY scope. Must be called by the parent-owned formation effect immediately
# before its first core, ownership, control, capital, or country mutation. This
# effect rescans the frozen plan and never selects replacement territory.
brilliant_scientist_revalidate_formation_territory_plan = {
	clr_country_flag = brilliant_scientist_formation_territory_verified
	clr_country_flag = brilliant_scientist_formation_territory_revalidated
	set_country_flag = brilliant_scientist_formation_territory_planning
	set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.none }
	if = {
		limit = { NOT = { has_country_flag = brilliant_scientist_formation_territory_prepared } }
		set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.missing_plan }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_event_target = brilliant_scientist_formation_territory_host }
				NOT = { event_target:brilliant_scientist_formation_territory_host = { tag = ROOT } }
			}
		}
		set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.stale_host }
	}
	else_if = {
		limit = { NOT = { brilliant_scientist_territory_plan_route_matches_request = yes } }
		set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.stale_route }
	}
	else_if = {
		limit = { NOT = { brilliant_scientist_project_mode_history_is_coherent = yes } }
		set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.incoherent_project_mode }
	}
	else = {
		brilliant_scientist_refresh_formation_territory_metrics = yes
		if = {
			limit = { NOT = { has_country_flag = brilliant_scientist_formation_territory_scope_scan_valid } }
			set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.stale_state }
		}
		else_if = {
			limit = { NOT = { brilliant_scientist_territory_plan_capital_is_current = yes } }
			set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.stale_capital }
		}
		else_if = {
			limit = { NOT = { brilliant_scientist_territory_plan_snapshot_is_current = yes } }
			set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.plan_changed }
		}
		else_if = {
			limit = {
				NOT = {
					OR = {
						brilliant_scientist_charter_territory_plan_meets_contract = yes
						brilliant_scientist_enclave_territory_plan_meets_contract = yes
						brilliant_scientist_rebellion_territory_plan_meets_contract = yes
					}
				}
			}
			set_variable = { brilliant_scientist_formation_territory_invalid_reason = constant:brilliant_scientist_territory_invalid_reason.route_contract }
		}
	}
	if = {
		limit = {
			check_variable = {
				var = brilliant_scientist_formation_territory_invalid_reason
				value = constant:brilliant_scientist_territory_invalid_reason.none
				compare = greater_than
			}
		}
		brilliant_scientist_invalidate_formation_territory_plan = yes
	}
	else = {
		set_country_flag = brilliant_scientist_formation_territory_revalidated
		clr_country_flag = brilliant_scientist_formation_territory_planning
		set_country_flag = brilliant_scientist_formation_territory_verified
	}
}
