# scheduler. It installs only three manually authored global-survival rows.
# Food has a country target. Water and rail carry one selected state target only
# when their native source proof is present. The producer never sets a scheduler
# activation flag and never fires an event by itself.
# ============================================================================

# Supported scope: COUNTRY coordinator. Builds one generation-bound candidate
# registry row set per Fallout country. State targets are selected by the
# lowest valid native state id, and a missing native source omits that row.
fallout_event_build_pilot_candidate_registries = {
	for_each_scope_loop = {
		array = global.fallout_event_registry_countries
		if = {
			limit = {
				has_country_flag = fallout_event_registry_member
				fallout_event_registry_country_row_is_current = yes
				OR = {
					NOT = { has_country_flag = fallout_event_candidate_registry_reviewed }
					NOT = { has_variable = fallout_event_candidate_registry_generation }
					NOT = { check_variable = { var = fallout_event_candidate_registry_generation value = fallout_event_runtime_generation compare = equals } }
				}
			}
			fallout_event_clear_candidate_rows_for_rebuild = yes

			# The country-level food row is always available after the numerical
			# registry is current. Its chain carries no invented state target.
			fallout_event_initialize_pilot_candidate_row = yes
			set_temp_variable = { fallout_event_pilot_candidate_id = constant:fallout_event_candidate_pilot.food_candidate_id }
			set_temp_variable = { fallout_event_pilot_transaction_key = constant:fallout_event_candidate_pilot.food_transaction_key }
			set_temp_variable = { fallout_event_pilot_cooldown_family = constant:fallout_event_cooldown_family.food_security }
			set_temp_variable = { fallout_event_pilot_human_token = constant:fallout_event_id.last_inventory_human }
			set_temp_variable = { fallout_event_pilot_ai_token = constant:fallout_event_id.last_inventory_ai }
			set_temp_variable = { fallout_event_pilot_resource_index = constant:fallout_survival_resource.food }
			set_temp_variable = { fallout_event_pilot_route = constant:fallout_event_candidate_pilot.food_route }
			fallout_event_append_pilot_candidate_row = yes

			# Water row. The state trigger proves an existing produced water source,
			# a current Fallout state row, and the native owner relationship.
			set_temp_variable = { fallout_event_pilot_water_target = constant:fallout_event_runtime.zero }
			every_owned_state = {
				set_temp_variable = { fallout_event_pilot_state_candidate_id = THIS.id }
				if = {
					limit = {
						fallout_event_pilot_water_state_is_current = yes
						OR = {
							check_variable = { fallout_event_pilot_water_target = constant:fallout_event_runtime.zero }
							check_variable = { var = fallout_event_pilot_state_candidate_id value = fallout_event_pilot_water_target compare = less_than }
						}
					}
					set_temp_variable = { fallout_event_pilot_water_target = fallout_event_pilot_state_candidate_id }
				}
			}
			if = {
				limit = { check_variable = { fallout_event_pilot_water_target > constant:fallout_event_runtime.zero } }
				fallout_event_initialize_pilot_candidate_row = yes
				set_temp_variable = { fallout_event_pilot_candidate_id = constant:fallout_event_candidate_pilot.water_candidate_id }
				set_temp_variable = { fallout_event_pilot_transaction_key = constant:fallout_event_candidate_pilot.water_transaction_key }
				set_temp_variable = { fallout_event_pilot_cooldown_family = constant:fallout_event_cooldown_family.water_security }
				set_temp_variable = { fallout_event_pilot_human_token = constant:fallout_event_id.river_intake_human }
				set_temp_variable = { fallout_event_pilot_ai_token = constant:fallout_event_id.river_intake_ai }
				set_temp_variable = { fallout_event_pilot_branch = constant:fallout_event_107_branch.close_intake }
				set_temp_variable = { fallout_event_pilot_resource_index = constant:fallout_survival_resource.clean_water }
				set_temp_variable = { fallout_event_pilot_route = constant:fallout_event_candidate_pilot.water_route }
				set_temp_variable = { fallout_event_pilot_target_type = constant:fallout_event_subject_type.state }
				set_temp_variable = { fallout_event_pilot_target = fallout_event_pilot_water_target }
				fallout_event_append_pilot_candidate_row = yes
			}

			# Rail row. A state enters only when its native railway is present. There
			# is no infrastructure or supply-node substitute for this pilot.
			set_temp_variable = { fallout_event_pilot_rail_target = constant:fallout_event_runtime.zero }
			every_owned_state = {
				set_temp_variable = { fallout_event_pilot_state_candidate_id = THIS.id }
				if = {
					limit = {
						fallout_event_pilot_rail_state_is_current = yes
						OR = {
							check_variable = { fallout_event_pilot_rail_target = constant:fallout_event_runtime.zero }
							check_variable = { var = fallout_event_pilot_state_candidate_id value = fallout_event_pilot_rail_target compare = less_than }
						}
					}
					set_temp_variable = { fallout_event_pilot_rail_target = fallout_event_pilot_state_candidate_id }
				}
			}
			if = {
				limit = { check_variable = { fallout_event_pilot_rail_target > constant:fallout_event_runtime.zero } }
				fallout_event_initialize_pilot_candidate_row = yes
				set_temp_variable = { fallout_event_pilot_candidate_id = constant:fallout_event_candidate_pilot.rail_candidate_id }
				set_temp_variable = { fallout_event_pilot_transaction_key = constant:fallout_event_candidate_pilot.rail_transaction_key }
				set_temp_variable = { fallout_event_pilot_cooldown_family = constant:fallout_event_cooldown_family.transport_recovery }
				set_temp_variable = { fallout_event_pilot_human_token = constant:fallout_event_id.rail_crew_human }
				set_temp_variable = { fallout_event_pilot_ai_token = constant:fallout_event_id.rail_crew_ai }
				set_temp_variable = { fallout_event_pilot_branch = constant:fallout_event_114_branch.send_protected_crews }
				set_temp_variable = { fallout_event_pilot_resource_index = constant:fallout_survival_resource.fuel }
				set_temp_variable = { fallout_event_pilot_route = constant:fallout_event_candidate_pilot.rail_route }
				set_temp_variable = { fallout_event_pilot_target_type = constant:fallout_event_subject_type.state }
				set_temp_variable = { fallout_event_pilot_target = fallout_event_pilot_rail_target }
				fallout_event_append_pilot_candidate_row = yes
			}

			set_variable = { fallout_event_candidate_registry_generation = fallout_event_runtime_generation }
			set_variable = { fallout_event_candidate_registry_day = global.num_days }
			set_country_flag = fallout_event_candidate_registry_reviewed
			if = {
				limit = { NOT = { fallout_event_candidate_registry_is_reviewed = yes } }
				clr_country_flag = fallout_event_candidate_registry_reviewed
				clear_variable = fallout_event_candidate_registry_generation
				clear_variable = fallout_event_candidate_registry_day
			}
		}
	}
}
