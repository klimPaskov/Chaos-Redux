# Supported scope: COUNTRY coordinator. Builds one generation-bound candidate
# registry row set per Fallout country. State targets are selected by the
# lowest valid native state id, and a missing native source omits that row.
fallout_event_build_pilot_candidate_registries = {
	for_each_scope_loop = {
		array = global.fallout_event_registry_countries
		if = {
			limit = {
				has_country_flag = fallout_event_registry_member
				fallout_event_registry_country_row_is_current = yes
				OR = {
					NOT = { has_country_flag = fallout_event_candidate_registry_reviewed }
					NOT = { has_variable = fallout_event_candidate_registry_generation }
					NOT = { check_variable = { var = fallout_event_candidate_registry_generation value = fallout_event_runtime_generation compare = equals } }
				}
			}
			fallout_event_clear_candidate_rows_for_rebuild = yes

			# The country-level food row is always available after the numerical
			# registry is current. Its chain carries no invented state target.
			fallout_event_initialize_pilot_candidate_row = yes
			set_temp_variable = { fallout_event_pilot_candidate_id = constant:fallout_event_candidate_pilot.food_candidate_id }
			set_temp_variable = { fallout_event_pilot_transaction_key = constant:fallout_event_candidate_pilot.food_transaction_key }
			set_temp_variable = { fallout_event_pilot_cooldown_family = constant:fallout_event_cooldown_family.food_security }
			set_temp_variable = { fallout_event_pilot_human_token = constant:fallout_event_id.last_inventory_human }
			set_temp_variable = { fallout_event_pilot_ai_token = constant:fallout_event_id.last_inventory_ai }
			set_temp_variable = { fallout_event_pilot_resource_index = constant:fallout_survival_resource.food }
			set_temp_variable = { fallout_event_pilot_route = constant:fallout_event_candidate_pilot.food_route }
			fallout_event_append_pilot_candidate_row = yes

			# Names for the Missing is admitted only after a country has recorded a
			# material civilian death count and its recognition ledger still shows an
			# incomplete census. The row has no state target or invented actor.
			if = {
				limit = { fallout_event_269_country_is_current = yes }
				fallout_event_initialize_pilot_candidate_row = yes
				set_temp_variable = { fallout_event_pilot_candidate_id = constant:fallout_event_candidate_pilot.names_missing_candidate_id }
				set_temp_variable = { fallout_event_pilot_transaction_key = constant:fallout_event_candidate_pilot.names_missing_transaction_key }
				set_temp_variable = { fallout_event_pilot_class = constant:fallout_event_class.crisis_incident }
				set_temp_variable = { fallout_event_pilot_cooldown_family = constant:fallout_event_cooldown_family.cause_memory }
				set_temp_variable = { fallout_event_pilot_human_token = constant:fallout_event_id.names_missing_human }
				set_temp_variable = { fallout_event_pilot_ai_token = constant:fallout_event_id.names_missing_ai }
				set_temp_variable = { fallout_event_pilot_phase = constant:fallout_event_phase.first_winter_year }
				set_temp_variable = { fallout_event_pilot_visible_budget_cost = constant:fallout_event_269_transaction.visible_budget_cost }
				set_temp_variable = { fallout_event_269_candidate_severity = chaos_meter_deaths_country_civilian_total }
				multiply_temp_variable = { fallout_event_269_candidate_severity = constant:fallout_event_269_threshold.deaths_to_severity }
				clamp_temp_variable = { var = fallout_event_269_candidate_severity min = constant:fallout_event_runtime.zero max = constant:fallout_event_269_threshold.maximum_candidate_severity }
				set_temp_variable = { fallout_event_pilot_mechanic_pressure = constant:fallout_event_runtime.zero }
				set_temp_variable = { fallout_event_pilot_severity = fallout_event_269_candidate_severity }
				set_temp_variable = { fallout_event_pilot_state_value = fallout_survival_current_entries^9 }
				set_temp_variable = { fallout_event_pilot_state_value_source = constant:fallout_survival_resource.recognition }
				set_temp_variable = { fallout_event_pilot_mechanic_pressure_source = constant:fallout_event_pressure_source_type.survival_resource }
				set_temp_variable = { fallout_event_pilot_severity_source = constant:fallout_event_pressure_source_type.survival_resource }
				set_temp_variable = { fallout_event_pilot_resource_index = constant:fallout_survival_resource.recognition }
				set_temp_variable = { fallout_event_pilot_required_resource = constant:fallout_survival_resource.recognition }
				set_temp_variable = { fallout_event_pilot_route = constant:fallout_event_candidate_pilot.names_missing_route }
				fallout_event_append_pilot_candidate_row = yes
				clear_temp_variable = fallout_event_269_candidate_severity
			}

			# First Safe Birth is a country-level generation-change incident. Its
			# severity is the reviewed Deaths score, while its state value is the
			# current Cohesion ledger. No invented state or actor target is stored.
			if = {
				limit = { fallout_event_282_country_is_current = yes }
				fallout_event_initialize_pilot_candidate_row = yes
				set_temp_variable = { fallout_event_pilot_candidate_id = constant:fallout_event_candidate_pilot.first_safe_birth_candidate_id }
				set_temp_variable = { fallout_event_pilot_transaction_key = constant:fallout_event_candidate_pilot.first_safe_birth_transaction_key }
				set_temp_variable = { fallout_event_pilot_class = constant:fallout_event_class.routine_incident }
				set_temp_variable = { fallout_event_pilot_cooldown_family = constant:fallout_event_cooldown_family.generation_change }
				set_temp_variable = { fallout_event_pilot_human_token = constant:fallout_event_id.first_safe_birth_human }
				set_temp_variable = { fallout_event_pilot_ai_token = constant:fallout_event_id.first_safe_birth_ai }
				set_temp_variable = { fallout_event_pilot_phase_mode = constant:fallout_event_review_phase_mode.preferred }
				set_temp_variable = { fallout_event_pilot_phase = constant:fallout_event_phase.first_winter_year }
				set_temp_variable = { fallout_event_pilot_secondary_phase = constant:fallout_event_phase.consolidation }
				set_temp_variable = { fallout_event_pilot_visible_budget_cost = constant:fallout_event_282_transaction.visible_budget_cost }
				set_temp_variable = { fallout_event_282_candidate_severity = chaos_meter_deaths_country_civilian_total }
				multiply_temp_variable = { fallout_event_282_candidate_severity = constant:fallout_event_282_threshold.deaths_to_severity }
				clamp_temp_variable = { var = fallout_event_282_candidate_severity min = constant:fallout_event_runtime.zero max = constant:fallout_event_282_threshold.maximum_candidate_severity }
				set_temp_variable = { fallout_event_pilot_mechanic_pressure = constant:fallout_event_runtime.zero }
				set_temp_variable = { fallout_event_pilot_severity = fallout_event_282_candidate_severity }
