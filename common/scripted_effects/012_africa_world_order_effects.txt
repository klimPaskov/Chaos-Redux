# COUNTRY scope on a candidate. Input: temporary africa_world_candidate_id.
africa_world_register_current_candidate = {
	if = {
		limit = {
			africa_world_package_candidate_base_is_valid = yes
			check_variable = { africa_world_candidate_id > constant:africa_world_continent.none }
			check_variable = {
				var = africa_world_candidate_id
				value = constant:africa_world_continent.oceania
				compare = less_than_or_equals
			}
		}
		set_country_flag = africa_world_package_candidate
		set_variable = { africa_world_continent_id = africa_world_candidate_id }
		set_variable = { africa_world_package_status = constant:africa_world_package_status.candidate }
		set_variable = { africa_world_package_legitimacy = constant:africa_world_order.package_start_legitimacy }
		set_variable = { africa_world_package_capacity = constant:africa_world_order.package_start_capacity }
		set_variable = { africa_world_package_authority = constant:africa_world_order.package_start_authority }
		if = {
			limit = { check_variable = { africa_world_candidate_id = constant:africa_world_continent.middle_east } }
			set_country_flag = africa_world_package_implementation_ready
		}
		else_if = {
			limit = { check_variable = { africa_world_candidate_id = constant:africa_world_continent.europe } }
			set_country_flag = africa_world_package_implementation_ready
		}
		else_if = {
			limit = { check_variable = { africa_world_candidate_id = constant:africa_world_continent.asia } }
			set_country_flag = africa_world_package_implementation_ready
		}
		event_target:africa_host = {
			if = {
				limit = { NOT = { is_in_array = { array = africa_world_package_candidates value = PREV } } }
				add_to_array = { array = africa_world_package_candidates value = PREV }
				add_to_variable = { africa_world_package_candidate_count = constant:africa_value.one }
			}
		}
	}
}

# COUNTRY scope on the Event 12 host. This is a deliberate, player-visible
# one-time nomination pass, not a periodic world scan.

		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.middle_east }
		random_country = {
			limit = { africa_world_is_middle_east_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.europe } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.europe } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.europe }
		random_country = {
			limit = { africa_world_is_europe_candidate = yes }
			africa_world_register_current_candidate = yes

	}
	if = {

		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.north_america } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.north_america } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.north_america }
		random_country = {
			limit = { africa_world_is_north_america_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.south_america } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.south_america } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.south_america }
		random_country = {
			limit = { africa_world_is_south_america_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {

			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.oceania } } }

				compare = less_than

# COUNTRY scope on a foreign participant.
africa_scramble_register_participant = {
	if = {
		limit = { NOT = { has_country_flag = africa_scramble_participant } }
		set_country_flag = africa_scramble_participant
		set_country_flag = africa_scramble_response_invited
		event_target:africa_host = {
			add_to_variable = { africa_scramble_participant_count = constant:africa_value.one }
			add_to_array = { array = africa_scramble_participants value = PREV }
		}
		if = {
			limit = { any_owned_state = { is_on_continent = africa } }
			set_country_flag = africa_scramble_colonial_interest
			set_country_flag = africa_scramble_foreign_base_holder
			set_country_flag = africa_scramble_ultimatum_candidate
		}
		if = {
			limit = { OR = { is_major = yes is_faction_leader = yes } }
			set_country_flag = africa_scramble_major_power
			set_country_flag = africa_scramble_recognition_candidate
		}
		country_event = { id = africa_world_order.1 }
	}
}

# COUNTRY scope on the Event 12 host. Called once from chaosx.nr12.309.
africa_initialize_scramble_and_world_packages = {
	if = {
		limit = {
			africa_is_current_host = yes
			has_global_flag = africa_is_one
			NOT = { has_country_flag = africa_scramble_state_machine_initialised }
		}
		set_country_flag = africa_scramble_state_machine_initialised
		set_variable = { africa_scramble_phase = constant:africa_scramble_phase.shock_and_recognition }
		set_variable = { africa_scramble_participant_count = constant:africa_value.zero }
		set_variable = { africa_scramble_recognition_count = constant:africa_value.zero }
		set_variable = { africa_scramble_conditional_recognition_count = constant:africa_value.zero }
		set_variable = { africa_scramble_sanction_count = constant:africa_value.zero }
		set_variable = { africa_scramble_ultimatum_count = constant:africa_value.zero }
		set_variable = { africa_scramble_expedition_count = constant:africa_value.zero }
		set_variable = { africa_scramble_coalition_break_count = constant:africa_value.zero }
		set_variable = { africa_scramble_settlement_score = constant:africa_value.zero }
		set_variable = { africa_scramble_pressure = constant:africa_measure.medium }
		set_variable = { africa_world_package_candidate_count = constant:africa_value.zero }
		set_variable = { africa_world_packages_installed_count = constant:africa_value.zero }
		set_variable = { africa_world_packages_settled_count = constant:africa_value.zero }
		set_variable = { africa_world_packages_sovereign_complete_count = constant:africa_value.zero }
		set_variable = { africa_world_terminal_resolved_count = constant:africa_value.zero }
		set_variable = { africa_world_sponsorship_overextension = constant:africa_value.zero }
		set_variable = { africa_world_order_route = constant:africa_world_order_route.uncommitted }
		clear_array = africa_scramble_participants
		clear_array = africa_world_package_candidates
		clear_array = africa_world_package_actors
		clear_array = africa_world_sponsorship_targets

		# This explicit post-unification census is the only global participant
		# discovery pass. It never repeats on a daily/weekly/monthly on_action.
		every_country = {
			limit = {
				exists = yes
				has_capitulated = no
				NOT = { tag = ROOT }
				NOT = { is_actual_nonhuman_country = yes }
				NOT = { is_special_chaos_country = yes }
				capital_scope = { NOT = { is_on_continent = africa } }
				OR = {
					is_major = yes
					is_faction_leader = yes
					any_owned_state = { is_on_continent = africa }
				}
			}
			africa_scramble_register_participant = yes
		}

		set_variable = { africa_scramble_recognition_target = africa_scramble_participant_count }
		divide_variable = { africa_scramble_recognition_target = constant:africa_world_order.recognition_divisor }
		clamp_variable = {
			var = africa_scramble_recognition_target
			min = constant:africa_world_order.minimum_recognition_target
			max = constant:africa_measure.maximum
		}
		if = {
			limit = {
				check_variable = {
					var = africa_scramble_recognition_target
					value = africa_scramble_participant_count
					compare = greater_than
				}
			}
			set_variable = { africa_scramble_recognition_target = africa_scramble_participant_count }
		}
		if = {
			limit = { check_variable = { africa_scramble_recognition_target = constant:africa_value.zero } }
			set_variable = { africa_scramble_recognition_target = constant:africa_value.one }
		}

		africa_world_nominate_missing_package_candidates = yes
		activate_mission = africa_scramble_recognition_window
		country_event = { id = africa_world_order.2 }
	}
}

# COUNTRY scope on the host. Input: temporary africa_scramble_timeout_result,
# where zero means a voluntary transition and one means deadline pressure.
africa_scramble_advance_to_coalition_phase = {
	if = {
		limit = { africa_scramble_phase_is_shock = yes }
		set_variable = { africa_scramble_phase = constant:africa_scramble_phase.coalition_and_ultimatum }
		clr_country_flag = africa_scramble_recognition_phase
		set_country_flag = africa_scramble_coalition_phase
		if = {
			limit = { check_variable = { africa_scramble_timeout_result > constant:africa_value.zero } }
			set_country_flag = africa_scramble_recognition_window_expired
			add_to_variable = { africa_scramble_pressure = constant:africa_world_order.scramble_pressure_gain }
			africa_scramble_clamp_values = yes
		}
		activate_mission = africa_scramble_coalition_window
		country_event = { id = africa_world_order.4 }
	}
}

}

africa_scramble_launch_expedition_if_unresolved = {
	set_temp_variable = { africa_scramble_expedition_launched = constant:africa_value.zero }
	if = {
		limit = {
			africa_scramble_phase_is_intervention = yes
			NOT = { africa_scramble_intervention_requirement_met = yes }
		}
		random_country = {
			limit = {
				has_country_flag = africa_scramble_expedition_planner
				has_country_flag = africa_scramble_coalition_member
				has_capitulated = no
				NOT = { has_war_with = event_target:africa_host }
				can_declare_war_on = event_target:africa_host
			}
			set_country_flag = africa_scramble_expedition_leader
			set_country_flag = africa_scramble_intervention_war_actor
			set_temp_variable = { africa_scramble_expedition_launched = constant:africa_value.one }
			set_global_flag = africa_scramble_intervention_war_active
			declare_war_on = {
				target = event_target:africa_host
				type = topple_government
			}
			event_target:africa_host = {
				set_country_flag = africa_scramble_expedition_launched
				set_country_flag = africa_scramble_defensive_war_active
				if = {
					limit = { NOT = { has_country_flag = africa_scramble_war_news_sent } }
					set_country_flag = africa_scramble_war_news_sent
					news_event = { id = africa_world_order.101 }
				}
			}
		}
	}
	if = {
		limit = { check_variable = { africa_scramble_expedition_launched = constant:africa_value.zero } }
		set_country_flag = africa_scramble_expedition_planning_collapsed
		set_country_flag = africa_intervention_coalition_withdrawn
	}
}

# COUNTRY scope on a nominated package candidate. Inputs are temporary
# africa_world_install_status and africa_world_install_outcome values.
africa_world_install_current_package = {
	if = {
		limit = {
			africa_world_package_is_candidate = yes
			check_variable = { africa_world_install_status > constant:africa_world_package_status.candidate }
			check_variable = {
				var = africa_world_install_status
				value = constant:africa_world_package_status.rival
				compare = less_than_or_equals
			}
		}
		clr_country_flag = africa_world_package_candidate
		set_country_flag = africa_world_package_actor
		set_country_flag = africa_world_package_installed
		set_variable = { africa_world_package_status = africa_world_install_status }
		set_variable = { africa_world_package_install_outcome = africa_world_install_outcome }
		set_variable = { africa_world_package_step = constant:africa_world_package_step.none }
		if = {
			limit = { check_variable = { africa_world_install_status = constant:africa_world_package_status.sponsored } }
			set_country_flag = africa_world_package_sponsored
			set_country_flag = africa_world_order_partner
			set_country_flag = africa_world_sponsorship_obligation_due
		}
		else_if = {
			limit = { check_variable = { africa_world_install_status = constant:africa_world_package_status.independent } }
			set_country_flag = africa_world_package_independent
		}
		else = {
			set_country_flag = africa_world_package_rival
			set_country_flag = africa_world_order_rivalry_intensified
		}

		event_target:africa_host = {
			if = {
				limit = { NOT = { is_in_array = { array = africa_world_package_actors value = PREV } } }
				add_to_array = { array = africa_world_package_actors value = PREV }
				add_to_variable = { africa_world_packages_installed_count = constant:africa_value.one }
			}
			if = {
				limit = { PREV = { has_country_flag = africa_world_package_sponsored } }
				add_to_variable = { africa_world_sponsorship_overextension = constant:africa_world_order.sponsorship_overextension_per_package }
				add_to_array = { array = africa_world_sponsorship_targets value = PREV }
				activate_targeted_decision = { target = PREV decision = africa_world_sponsorship_obligation }
			}
		}

		if = {
			limit = { check_variable = { africa_world_continent_id = constant:africa_world_continent.middle_east } }
			set_country_flag = africa_world_middle_east_package
			load_focus_tree = { tree = africa_middle_east_world_focus_tree keep_completed = no }
		}
		else_if = {
