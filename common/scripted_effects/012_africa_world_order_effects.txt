# COUNTRY scope on the current Africa host. The four named wrappers below are
# the only Event 012 super-event emitters. They share one bounded presenter so
# the global display flag, unique audio id, and human-player playback cannot
# drift between roles.
africa_world_emit_super_event = {
	set_temp_variable = { africa_super_event_visible_days = constant:africa_world_super_event.visible_days }
	meta_effect = {
		text = {
			set_global_flag = {
				flag = super_event_visible
				days = africa_super_event_visible_days
				value = [SUPER_EVENT_ID]
			}
		}
		SUPER_EVENT_ID = "[?africa_world_super_event_slot|.0]"
	}
	set_variable = { global.current_super_event_audio_id = africa_world_super_event_audio }
	every_country = {
		limit = { is_ai = no }
		play_current_super_event_audio = yes
	}
	clear_variable = africa_world_super_event_slot
	clear_variable = africa_world_super_event_audio
}

# COUNTRY scope on the Event 012 host. This is an explicit source registrar
# for the four accepted terminal super-event roles. A role receives five
# global receipts only when its slot, image, unique audio id, localisation/text
# records, and runtime consumer are all named below. No fallback presentation
# or generic global claim is used; removing any role's source mapping leaves
# africa_the_world_super_event_package_ready unset.
africa_world_register_terminal_super_event_runtime_surfaces = {
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { has_global_flag = africa_the_world_super_event_package_ready }
			NOT = { has_global_flag = world_end }
		}
		# Role 101: Africa Is One. Slot 101/audio id 58; the exact image sprite
		# and DDS, audio track/wrappers, localisation records, scripted text
		# selectors, and africa_world_emit_africa_is_one_super_event consumer are
		# all present in the Event 012 source package.
		if = {
			limit = { NOT = { has_global_flag = africa_world_super_event_101_slot_receipt } }
			set_global_flag = africa_world_super_event_101_slot_receipt
			set_global_flag = africa_world_super_event_101_image_receipt
			set_global_flag = africa_world_super_event_101_audio_receipt
			set_global_flag = africa_world_super_event_101_localisation_receipt
			set_global_flag = africa_world_super_event_101_consumer_receipt
		}
		# Role 102: Scramble Response. Slot 102/audio id 59; the exact image
		# sprite/DDS, audio track/wrappers, chaosx_super_event.102.t/q/a/d and
		# scripted selectors, and africa_world_emit_scramble_response_super_event
		# consumer are all present in the Event 012 source package.
		if = {
			limit = { NOT = { has_global_flag = africa_world_super_event_102_slot_receipt } }
			set_global_flag = africa_world_super_event_102_slot_receipt
			set_global_flag = africa_world_super_event_102_image_receipt
			set_global_flag = africa_world_super_event_102_audio_receipt
			set_global_flag = africa_world_super_event_102_localisation_receipt
			set_global_flag = africa_world_super_event_102_consumer_receipt
		}
		# Role 103: Continental Wars. Slot 103/audio id 60; the exact image
		# sprite/DDS, audio track/wrappers, chaosx_super_event.103.t/q/a/d and
		# scripted selectors, and africa_world_emit_continental_wars_super_event
		# consumer are all present in the Event 012 source package.
		if = {
			limit = { NOT = { has_global_flag = africa_world_super_event_103_slot_receipt } }
			set_global_flag = africa_world_super_event_103_slot_receipt
			set_global_flag = africa_world_super_event_103_image_receipt
			set_global_flag = africa_world_super_event_103_audio_receipt
			set_global_flag = africa_world_super_event_103_localisation_receipt
			set_global_flag = africa_world_super_event_103_consumer_receipt
		}
		# Role 104: The World. Slot 104/audio id 61; the exact image
		# sprite/DDS, audio track/wrappers, chaosx_super_event.104.t/q/a/d and
		# scripted selectors, and africa_world_emit_the_world_super_event consumer
		# are all present in the Event 012 source package.
		if = {
			limit = { NOT = { has_global_flag = africa_world_super_event_104_slot_receipt } }
			set_global_flag = africa_world_super_event_104_slot_receipt
			set_global_flag = africa_world_super_event_104_image_receipt
			set_global_flag = africa_world_super_event_104_audio_receipt
			set_global_flag = africa_world_super_event_104_localisation_receipt
			set_global_flag = africa_world_super_event_104_consumer_receipt
		}
	}
}

# COUNTRY scope on the Event 012 host. The readiness writer is intentionally
# separate from the source registrar above. It can set the terminal package
# flag only after the trigger has evaluated every role receipt and it does not
# write any receipt itself.
africa_world_certify_terminal_super_event_runtime_surfaces = {
	if = {
		limit = {
			africa_world_terminal_super_event_runtime_surfaces_are_registered = yes
			NOT = { has_global_flag = africa_the_world_super_event_package_ready }
		}
		set_global_flag = africa_the_world_super_event_package_ready
	}
}

# COUNTRY scope on a candidate. Input: temporary africa_world_candidate_id.
africa_world_register_current_candidate = {
	if = {
		limit = {
			africa_world_package_candidate_base_is_valid = yes
			check_variable = { africa_world_candidate_id > constant:africa_world_continent.none }
			check_variable = {
				var = africa_world_candidate_id
				value = constant:africa_world_continent.oceania
				compare = less_than_or_equals
			}
		}
		set_country_flag = africa_world_package_candidate
		set_variable = { africa_world_continent_id = africa_world_candidate_id }
		set_variable = { africa_world_package_status = constant:africa_world_package_status.candidate }
		set_variable = { africa_world_package_legitimacy = constant:africa_world_order.package_start_legitimacy }
		set_variable = { africa_world_package_capacity = constant:africa_world_order.package_start_capacity }
		set_variable = { africa_world_package_authority = constant:africa_world_order.package_start_authority }
		# Continent packages are nominated for the post-unification ledger. The
		# one-time source certification below opens installation only after all
		# six complete regional packages have been registered in the frozen roster.
		event_target:africa_host = {
			if = {
				limit = { NOT = { is_in_array = { array = africa_world_package_candidates value = PREV } } }
				add_to_array = { array = africa_world_package_candidates value = PREV }
				add_to_variable = { africa_world_package_candidate_count = constant:africa_value.one }
			}
		}
	}
}

# COUNTRY scope on the Event 12 host. This is a deliberate, player-visible
# one-time nomination pass, not a periodic world scan.
africa_world_nominate_missing_package_candidates = {
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.middle_east } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.middle_east } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.middle_east }
		random_country = {
			limit = { africa_world_is_middle_east_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.europe } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.europe } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.europe }
		random_country = {
			limit = { africa_world_is_europe_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.asia } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.asia } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.asia }
		random_country = {
			limit = { africa_world_is_asia_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.north_america } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.north_america } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.north_america }
		random_country = {
			limit = { africa_world_is_north_america_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.south_america } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.south_america } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.south_america }
		random_country = {
			limit = { africa_world_is_south_america_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { any_country = { has_country_flag = africa_world_package_candidate check_variable = { africa_world_continent_id = constant:africa_world_continent.oceania } } }
			NOT = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.oceania } } }
		}
		set_temp_variable = { africa_world_candidate_id = constant:africa_world_continent.oceania }
		random_country = {
			limit = { africa_world_is_oceania_candidate = yes }
			africa_world_register_current_candidate = yes
		}
	}

	africa_world_finalize_package_roster = yes
}

# COUNTRY scope on the Event 12 host. Records one explicit disposition for
# each external continent after the one-time nomination pass. Installed actors
# are resolved, candidates remain pending, and an empty slot is documented as
# absent; no placeholder country is manufactured.
africa_world_finalize_package_roster = {
	if = {
		limit = { africa_is_current_host = yes }
		clear_array = africa_world_package_pending_continents
		clear_array = africa_world_package_absent_continents
		clear_array = africa_world_package_resolved_continents
		set_variable = { africa_world_package_roster_disposition_count = constant:africa_value.zero }
		set_variable = { africa_world_package_pending_count = constant:africa_value.zero }
		set_variable = { africa_world_package_absent_count = constant:africa_value.zero }
		set_variable = { africa_world_package_resolved_count = constant:africa_value.zero }
		clr_country_flag = africa_world_package_roster_partial
		clr_country_flag = africa_world_package_roster_absence_documented
		clr_country_flag = africa_world_package_roster_complete

		if = {
			limit = {
				any_country = {
					OR = {
						has_country_flag = africa_world_package_candidate
						has_country_flag = africa_world_package_actor
					}
					check_variable = { africa_world_continent_id = constant:africa_world_continent.middle_east }
				}
			}
			if = {
				limit = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.middle_east } } }
				add_to_array = { array = africa_world_package_resolved_continents value = constant:africa_world_continent.middle_east }
				add_to_variable = { africa_world_package_resolved_count = constant:africa_value.one }
			}
			else = {
				add_to_array = { array = africa_world_package_pending_continents value = constant:africa_world_continent.middle_east }
				add_to_variable = { africa_world_package_pending_count = constant:africa_value.one }
			}
		}
		else = {
			add_to_array = { array = africa_world_package_absent_continents value = constant:africa_world_continent.middle_east }
			add_to_variable = { africa_world_package_absent_count = constant:africa_value.one }
		}
		add_to_variable = { africa_world_package_roster_disposition_count = constant:africa_value.one }

		if = {
			limit = {
				any_country = {
					OR = {
						has_country_flag = africa_world_package_candidate
						has_country_flag = africa_world_package_actor
					}
					check_variable = { africa_world_continent_id = constant:africa_world_continent.europe }
				}
			}
			if = {
				limit = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.europe } } }
				add_to_array = { array = africa_world_package_resolved_continents value = constant:africa_world_continent.europe }
				add_to_variable = { africa_world_package_resolved_count = constant:africa_value.one }
			}
			else = {
				add_to_array = { array = africa_world_package_pending_continents value = constant:africa_world_continent.europe }
				add_to_variable = { africa_world_package_pending_count = constant:africa_value.one }
			}
		}
		else = {
			add_to_array = { array = africa_world_package_absent_continents value = constant:africa_world_continent.europe }
			add_to_variable = { africa_world_package_absent_count = constant:africa_value.one }
		}
		add_to_variable = { africa_world_package_roster_disposition_count = constant:africa_value.one }

		if = {
			limit = {
				any_country = {
					OR = {
						has_country_flag = africa_world_package_candidate
						has_country_flag = africa_world_package_actor
					}
					check_variable = { africa_world_continent_id = constant:africa_world_continent.asia }
				}
			}
			if = {
				limit = { any_country = { has_country_flag = africa_world_package_actor check_variable = { africa_world_continent_id = constant:africa_world_continent.asia } } }
				add_to_array = { array = africa_world_package_resolved_continents value = constant:africa_world_continent.asia }

		limit = { africa_world_constituent_withdrawal_notice_is_open = yes }
		if = {
			limit = {
				OR = {
					has_war = yes
					is_subject = yes
					has_country_flag = africa_world_constituent_withdrawal_contested
				}
			}
			africa_world_clear_constituent_withdrawal_notice = yes
			set_country_flag = africa_world_constituent_withdrawal_contested
			set_country_flag = africa_world_constituent_counterterms_rejected

			add_to_variable = { africa_world_constituent_grievance = constant:africa_world_package_protocol.grievance_gain }
		else = { africa_world_record_constituent_withdrawal = yes }
	}
}

# COUNTRY scope on a package actor after capitulation or annexation. A single
# successor candidate is nominated if one exists; otherwise the ledger opens a
# visible exile or breakup certification path.
africa_world_handle_package_actor_loss = {
	if = {
		limit = {
			africa_world_package_is_installed = yes
			NOT = { has_country_flag = africa_world_package_successor_recorded }
			NOT = { has_country_flag = africa_world_package_exile_path_resolved }
			NOT = { has_country_flag = africa_world_package_breakup_recorded }
		}
		# A sponsored package actor has lost the sponsored movement at the
		# exact capitulation/annexation witness. Successor review may preserve
		# a continent slot, but it cannot restore the original sponsorship proof.
		africa_world_record_sponsored_movement_collapse = yes
		if = {
			limit = { africa_world_sponsorship_active_cleanup_is_needed = yes }
			africa_world_sponsorship_cleanup_active_state = yes
		}
		africa_world_cleanup_package_constituent_protocol = yes
		set_country_flag = africa_world_package_successor_review_open
		africa_world_refresh_package_polity_proof = yes
		set_variable = { africa_world_package_resolution = constant:africa_world_package_resolution.broken }
		clear_array = africa_world_package_successor_candidates
			random_country = {
			limit = { africa_world_package_successor_candidate_is_valid = yes }
			set_country_flag = africa_world_package_successor_candidate
			set_variable = { africa_world_constituent_status = constant:africa_world_constituent_status.successor }
			ROOT = { add_to_array = { array = africa_world_package_successor_candidates value = PREV } }
		}
		if = {
			limit = {
				NOT = {
					any_of_scopes = {
						array = africa_world_package_successor_candidates
						has_country_flag = africa_world_package_successor_candidate
					}
				}
			}
			if = {
				limit = { is_government_in_exile = yes }
				set_country_flag = africa_world_package_exile_path_open
			}
			else = {
				set_country_flag = africa_world_package_breakup_review_open
			}
		}
	}
}

# COUNTRY scope on a nominated package candidate. Inputs are temporary
# africa_world_install_status and africa_world_install_outcome values.
africa_world_install_current_package = {
	if = {
		limit = {
			africa_world_package_is_candidate = yes
			has_country_flag = africa_world_package_implementation_ready
			check_variable = { africa_world_install_status > constant:africa_world_package_status.candidate }
			check_variable = {
				var = africa_world_install_status
				value = constant:africa_world_package_status.rival
				compare = less_than_or_equals
			}
		}
		clr_country_flag = africa_world_package_candidate
		set_country_flag = africa_world_package_actor
		set_country_flag = africa_world_package_installed
		set_variable = { africa_world_package_status = africa_world_install_status }
		set_variable = { africa_world_package_install_outcome = africa_world_install_outcome }
		set_variable = { africa_world_package_step = constant:africa_world_package_step.none }
		set_variable = { africa_world_package_completed_step_count = constant:africa_value.zero }
		if = {
			limit = { check_variable = { africa_world_install_status = constant:africa_world_package_status.sponsored } }
			set_country_flag = africa_world_package_sponsored
			set_country_flag = africa_world_order_partner
			set_country_flag = africa_world_sponsorship_obligation_due
		}
		else_if = {
			limit = { check_variable = { africa_world_install_status = constant:africa_world_package_status.independent } }
			set_country_flag = africa_world_package_independent
		}
		else = {
			set_country_flag = africa_world_package_rival
			set_country_flag = africa_world_order_rivalry_intensified
		}

		event_target:africa_host = {
			if = {
				limit = { NOT = { is_in_array = { array = africa_world_package_actors value = PREV } } }
				add_to_array = { array = africa_world_package_actors value = PREV }
				add_to_variable = { africa_world_packages_installed_count = constant:africa_value.one }
			}

# COUNTRY scope on a failed postwar region actor. Each unresolved actor enters
# the pressure ledger once; four simultaneous failures reach the public global
# revolt threshold and the registered actors themselves open the wars.
africa_world_record_region_revolt = {
	if = {
		limit = {
			NOT = { has_country_flag = africa_world_region_revolt }
			has_event_target = africa_host
		}
		set_country_flag = africa_world_region_revolt
		event_target:africa_host = {
			add_to_variable = { africa_integration_burden = constant:africa_world_order.world_region_revolt_pressure }
			add_to_variable = { africa_world_global_revolt_pressure = constant:africa_world_order.world_region_revolt_pressure }
			if = {
				limit = {
					check_variable = {
						var = africa_world_global_revolt_pressure
						value = constant:africa_world_order.world_region_revolt_threshold
						compare = greater_than_or_equals
					}
					NOT = { has_global_flag = africa_world_global_revolt_active }
				}
				set_global_flag = africa_world_global_revolt_active
				set_global_flag = africa_achievement_global_revolt_threshold_reached
				for_each_scope_loop = {
					array = africa_world_package_actors
					if = {
						limit = {
							exists = yes
							has_country_flag = africa_world_region_revolt
							NOT = { has_war_with = event_target:africa_host }
							can_declare_war_on = event_target:africa_host
						}
						declare_war_on = { target = event_target:africa_host type = topple_government }
					}
				}
			}
		}
	}
}
