# ============================================================================
# CHAOS REDUX - EVENT 006 FROZEN PLAN EXECUTION
# ============================================================================
# Overview:
# - Validates the complete Event 006 metadata ledger before ownership changes.
# - Transfers every reserved Event 006 state in one effect chain after the
#   shared coordinator enters its execution phase.
# - Initializes creator provenance and the five country/eight host values from
#   the frozen package row, never from transient data on an absent tag.
# - Commits only when every selected country was created and initialized.
# - The standalone wrapper accepts Event 006-only plans. Joint Liberations
#   execution is owned by the separate Event 005/Event 006 joint dispatcher.
# ============================================================================

independence_wave_validate_execution_metadata = {
	set_temp_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.none }
	if = {
		limit = {
			OR = {
				NOT = { is_liberation_release_plan_locked = yes }
				NOT = { has_global_flag = independence_wave_plan_contribution_ready }
				NOT = { independence_wave_plan_metadata_arrays_are_aligned = yes }
				NOT = { check_variable = { var = global.independence_wave_plan_selected_count value = global.independence_wave_plan_target_count compare = equals } }
			}
		}
		set_temp_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.aligned_array_failure }
	}
	set_temp_variable = { independence_wave_execution_package_index = constant:independence_wave_plan.first_array_index }
	while_loop_effect = {
		limit = {
			check_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.none }
			check_variable = { var = independence_wave_execution_package_index value = global.independence_wave_plan_selected_count compare = less_than }
		}
		set_temp_variable = { independence_wave_execution_country_row = global.independence_wave_plan_country_row_indices^independence_wave_execution_package_index }
		if = {
			limit = {
				OR = {
					check_variable = { var = independence_wave_execution_country_row value = constant:independence_wave_plan.first_array_index compare = less_than }
					NOT = { check_variable = { var = independence_wave_execution_country_row value = global.liberation_plan_countries^num compare = less_than } }
				}
			}
			set_temp_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.aligned_array_failure }
		}
		if = {
			limit = { check_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.none } }
			var:global.liberation_plan_countries^independence_wave_execution_country_row = { save_event_target_as = independence_wave_execution_country }
			var:global.liberation_plan_anchor_states^independence_wave_execution_country_row = { save_event_target_as = independence_wave_execution_anchor }
			var:global.liberation_plan_primary_hosts^independence_wave_execution_country_row = { save_event_target_as = independence_wave_execution_former_host }
			set_temp_variable = { independence_wave_execution_package_id = global.independence_wave_plan_selected_package_ids^independence_wave_execution_package_index }
			set_temp_variable = { independence_wave_setup_package_id = independence_wave_execution_package_id }
			independence_wave_probe_force_package_mapping = yes
			if = {
				limit = {
					OR = {
						NOT = { has_event_target = independence_wave_execution_country }
						NOT = { has_event_target = independence_wave_execution_anchor }
						NOT = { has_event_target = independence_wave_execution_former_host }
						NOT = { check_variable = { var = global.liberation_plan_country_owners^independence_wave_execution_country_row value = constant:liberation_plan_owner.independence_wave compare = equals } }
						NOT = { check_variable = { var = global.liberation_plan_package_ids^independence_wave_execution_country_row value = independence_wave_execution_package_id compare = equals } }
						NOT = { has_independence_wave_runtime_package_adapter_for_execution_id = yes }
						NOT = { has_ready_independence_wave_force_package_mapping_probe = yes }
						NOT = {
							event_target:independence_wave_execution_country = {
								exists = no
								is_liberation_release_current_reserved_country = yes
								is_independence_wave_runtime_package_preflight_ready = yes
							}
						}
						NOT = { event_target:independence_wave_execution_anchor = { is_liberation_release_current_reserved_state = yes is_owned_by = event_target:independence_wave_execution_former_host } }
						NOT = { event_target:independence_wave_execution_former_host = { exists = yes is_in_array = { array = global.independence_wave_plan_hosts value = THIS } } }
					}
				}
				set_temp_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.invalid_scope }
			}
		}
		add_to_temp_variable = { independence_wave_execution_package_index = constant:independence_wave_plan.loop_increment }
	}
	set_temp_variable = { independence_wave_execution_sponsorship_index = constant:independence_wave_plan.first_array_index }
	while_loop_effect = {
		limit = {
			check_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.none }
			check_variable = {
				var = independence_wave_execution_sponsorship_index
				value = global.independence_wave_plan_sponsorship_count
				compare = less_than
			}
		}
		set_temp_variable = { independence_wave_execution_sponsorship_country_row = global.independence_wave_plan_sponsorship_country_rows^independence_wave_execution_sponsorship_index }
		set_temp_variable = { independence_wave_execution_sponsorship_generation = global.independence_wave_plan_sponsorship_generations^independence_wave_execution_sponsorship_index }
		set_temp_variable = { independence_wave_execution_sponsorship_opening_strength = global.independence_wave_plan_sponsorship_opening_strengths^independence_wave_execution_sponsorship_index }
		set_temp_variable = { independence_wave_execution_sponsorship_route = global.independence_wave_plan_sponsorship_routes^independence_wave_execution_sponsorship_index }
		if = {
			limit = {
				OR = {
					check_variable = { var = independence_wave_execution_sponsorship_country_row value = constant:independence_wave_plan.first_array_index compare = less_than }
					NOT = { check_variable = { var = independence_wave_execution_sponsorship_country_row value = global.liberation_plan_countries^num compare = less_than } }
				}
			}
			set_temp_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.aligned_array_failure }
		}
		else = {
			var:global.liberation_plan_countries^independence_wave_execution_sponsorship_country_row = { save_event_target_as = independence_wave_execution_sponsorship_country }
			var:global.liberation_plan_anchor_states^independence_wave_execution_sponsorship_country_row = { save_event_target_as = independence_wave_execution_sponsorship_anchor }
			var:global.independence_wave_plan_sponsorship_states^independence_wave_execution_sponsorship_index = { save_event_target_as = independence_wave_execution_sponsorship_state }
			var:global.independence_wave_plan_sponsorship_sponsors^independence_wave_execution_sponsorship_index = { save_event_target_as = independence_wave_execution_sponsorship_sponsor }
			if = {
				limit = {
					OR = {
						NOT = { has_event_target = independence_wave_execution_sponsorship_country }
						NOT = { has_event_target = independence_wave_execution_sponsorship_anchor }
						NOT = { has_event_target = independence_wave_execution_sponsorship_state }
						NOT = { has_event_target = independence_wave_execution_sponsorship_sponsor }
						NOT = {
							event_target:independence_wave_execution_sponsorship_country = {
								exists = no
								is_liberation_release_current_reserved_country = yes
								has_country_flag = independence_wave_pending_package
								has_country_flag = independence_wave_pending_breakaway_sponsorship
								check_variable = { var = independence_wave_pending_plan_id value = global.liberation_plan_id compare = equals }
								check_variable = { var = independence_wave_pending_sponsorship_plan_id value = global.liberation_plan_id compare = equals }
								check_variable = { var = independence_wave_pending_sponsorship_state value = event_target:independence_wave_execution_sponsorship_anchor compare = equals }
								check_variable = { var = independence_wave_pending_sponsorship_state value = event_target:independence_wave_execution_sponsorship_state compare = equals }
								check_variable = { var = independence_wave_pending_sponsor_country value = event_target:independence_wave_execution_sponsorship_sponsor compare = equals }
								check_variable = { var = independence_wave_pending_sponsor_generation value = independence_wave_execution_sponsorship_generation compare = equals }
								check_variable = { var = independence_wave_pending_sponsorship_opening_strength value = independence_wave_execution_sponsorship_opening_strength compare = equals }
								check_variable = { var = independence_wave_pending_sponsorship_route value = independence_wave_execution_sponsorship_route compare = equals }
							}
						}
						NOT = {
							event_target:independence_wave_execution_sponsorship_state = {
								has_valid_independence_wave_breakaway_sponsorship_record = yes
								check_variable = { var = independence_wave_sponsoring_country value = event_target:independence_wave_execution_sponsorship_sponsor compare = equals }
								check_variable = { var = independence_wave_sponsorship_generation value = independence_wave_execution_sponsorship_generation compare = equals }
								check_variable = { var = independence_wave_sponsored_opening_strength value = independence_wave_execution_sponsorship_opening_strength compare = equals }
								check_variable = { var = independence_wave_sponsored_route value = independence_wave_execution_sponsorship_route compare = equals }
							}
						}
					}
				}
				set_temp_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.invalid_scope }
			}
		}
		add_to_temp_variable = { independence_wave_execution_sponsorship_index = constant:independence_wave_plan.loop_increment }
	}
	set_variable = { global.independence_wave_execution_last_failure = independence_wave_execution_validation_failure }
	if = {
		limit = { check_variable = { independence_wave_execution_validation_failure = constant:liberation_plan_reject_reason.none } }
		set_global_flag = independence_wave_execution_metadata_valid
	}
	else = { clr_global_flag = independence_wave_execution_metadata_valid }
}

# Inputs are the current package event targets and
# `independence_wave_execution_package_id`. The target's unplanned historical
# cores are masked only for the fixed-tag release call, then restored. Planned
# states become permanent cores. This avoids relying on ownership assignment to
# instantiate an absent country and prevents a reused registered tag from
# taking territory outside the frozen package.
independence_wave_release_one_frozen_country = {
	clear_array = global.independence_wave_execution_masked_core_states
	clear_array = global.independence_wave_execution_added_plan_cores
	event_target:independence_wave_execution_country = {
		every_core_state = {
			limit = {
				NOT = {
					is_liberation_release_current_reserved_state = yes
					check_variable = {
						var = liberation_release_plan_owner
						value = constant:liberation_plan_owner.independence_wave
						compare = equals
					}
					check_variable = {
						var = liberation_release_package_id
						value = independence_wave_execution_package_id
						compare = equals
					}
					check_variable = {
						var = liberation_release_target_country
						value = event_target:independence_wave_execution_country
						compare = equals
					}
				}
			}
			add_to_array = { array = global.independence_wave_execution_masked_core_states value = THIS }
			remove_core_of = event_target:independence_wave_execution_country
		}
	}
	for_each_scope_loop = {
		array = global.liberation_plan_states
		if = {
			limit = {
				check_variable = {
					var = liberation_release_plan_owner
					value = constant:liberation_plan_owner.independence_wave
					compare = equals
				}
				check_variable = {
					var = liberation_release_package_id
					value = independence_wave_execution_package_id
					compare = equals
				}
				check_variable = {
					var = liberation_release_target_country
					value = event_target:independence_wave_execution_country
					compare = equals
				}
			}
			if = {
				limit = { NOT = { is_core_of = event_target:independence_wave_execution_country } }
				add_to_array = { array = global.independence_wave_execution_added_plan_cores value = THIS }
			}
			add_core_of = event_target:independence_wave_execution_country
		}
	}
	event_target:independence_wave_execution_former_host = {
		release = event_target:independence_wave_execution_country
		if = {
			limit = {
				event_target:independence_wave_execution_country = {
					is_subject_of = event_target:independence_wave_execution_former_host
				}
			}
			set_autonomy = {
				target = event_target:independence_wave_execution_country
				autonomy_state = autonomy_free
			}
		}
	}
	for_each_scope_loop = {
		array = global.independence_wave_execution_masked_core_states
		add_core_of = event_target:independence_wave_execution_country
	}
	if = {
		limit = { event_target:independence_wave_execution_country = { exists = yes } }
		add_to_temp_variable = { independence_wave_execution_instantiated_count = constant:independence_wave_plan.loop_increment }
	}
	else = {
		for_each_scope_loop = {
			array = global.independence_wave_execution_added_plan_cores
			remove_core_of = event_target:independence_wave_execution_country
		}
	}
	clear_array = global.independence_wave_execution_masked_core_states
	clear_array = global.independence_wave_execution_added_plan_cores
}

			}
		}

independence_wave_transfer_frozen_states = {

	set_temp_variable = { independence_wave_execution_expected_state_count = constant:independence_wave_planner.no_candidates }

	set_temp_variable = { independence_wave_execution_transfer_failure = constant:liberation_plan_reject_reason.none }
	while_loop_effect = {
		limit = {
			check_variable = { var = independence_wave_execution_state_index value = global.liberation_plan_states^num compare = less_than }
		}
		if = {
			limit = {
				check_variable = { var = global.liberation_plan_state_country_owners^independence_wave_execution_state_index value = constant:liberation_plan_owner.independence_wave compare = equals }
			}
			add_to_temp_variable = { independence_wave_execution_expected_state_count = constant:independence_wave_plan.loop_increment }
			var:global.liberation_plan_state_target_countries^independence_wave_execution_state_index = { save_event_target_as = independence_wave_execution_state_target }
			var:global.liberation_plan_states^independence_wave_execution_state_index = {
				set_state_owner_to = event_target:independence_wave_execution_state_target
				set_state_controller_to = event_target:independence_wave_execution_state_target
				if = {
					limit = {
						is_owned_by = event_target:independence_wave_execution_state_target
						is_controlled_by = event_target:independence_wave_execution_state_target
					}
					add_to_temp_variable = { independence_wave_execution_transferred_state_count = constant:independence_wave_plan.loop_increment }
				}
				else = {
					set_temp_variable = { independence_wave_execution_transfer_failure = constant:liberation_plan_reject_reason.unsafe_instantiation }
				}
			}
		}
		add_to_temp_variable = { independence_wave_execution_state_index = constant:independence_wave_plan.loop_increment }
	}
}

	set_temp_variable = { independence_wave_setup_region = global.independence_wave_plan_selected_regions^independence_wave_execution_package_index }

	event_target:independence_wave_execution_anchor = { save_event_target_as = independence_wave_setup_anchor_state }
}

# Pass 1: prepare origin state, then apply the exact package after the shared
# finalization barrier. Package setup must prove before any durable Event 006
# history row is opened.
independence_wave_prepare_frozen_country_packages = {
	set_temp_variable = { independence_wave_execution_prepared_count = constant:independence_wave_planner.no_candidates }
	set_temp_variable = { independence_wave_execution_package_index = constant:independence_wave_plan.first_array_index }
	while_loop_effect = {
		limit = { check_variable = { var = independence_wave_execution_package_index value = global.independence_wave_plan_selected_count compare = less_than } }
		independence_wave_load_frozen_country_execution_context = yes
		event_target:independence_wave_execution_country = {
			set_capital = {
				state = event_target:independence_wave_execution_anchor
				remember_old_capital = no
			}
			set_temp_variable = { independence_wave_package_setup_success = constant:independence_wave_planner.no_candidates }
			independence_wave_prepare_country_origin = yes
			if = {
				limit = { has_country_flag = independence_wave_origin_prepared }
				independence_wave_apply_pending_breakaway_sponsorship = yes
				independence_wave_dispatch_package_setup = yes
			}
			if = {
				limit = {
					has_country_flag = independence_wave_origin_prepared
					has_country_flag = independence_wave_package_setup_complete
					check_variable = { independence_wave_package_setup_success = constant:independence_wave_planner.selected }
				}
				add_to_temp_variable = { independence_wave_execution_prepared_count = constant:independence_wave_plan.loop_increment }
			}
		}
		add_to_temp_variable = { independence_wave_execution_package_index = constant:independence_wave_plan.loop_increment }
	}
}

# Package setup includes politics, technology inheritance, units, stockpiles,
# and bounded transfers of host air and naval assets. It therefore runs only
# after the shared coordinator enters `finalizing`; no package mutation is part
# of the compensating ownership rollback.

	set_temp_variable = { independence_wave_execution_prepared_count = constant:independence_wave_planner.no_candidates }

	if = {
		limit = { is_liberation_release_plan_finalizing = yes }
		independence_wave_prepare_frozen_country_packages = yes
		if = {
			limit = { check_variable = { var = independence_wave_execution_prepared_count value = global.independence_wave_plan_selected_count compare = equals } }
			independence_wave_activate_frozen_country_origins = yes
		}
		if = {
			limit = { check_variable = { var = independence_wave_execution_activated_count value = global.independence_wave_plan_selected_count compare = equals } }
			independence_wave_validate_frozen_country_packages = yes
		}
	}
}

# Durable Event 006 history is appended only after every package setup and live
# registry assertion has succeeded. The caller still verifies the initialized

independence_wave_finalize_frozen_country_origins = {

	set_temp_variable = { independence_wave_execution_initialized_count = constant:independence_wave_planner.no_candidates }
	independence_wave_commit_frozen_country_origins = yes
}

# Event-owned rollback acknowledgement. The execution pipeline performs no
# Event 006 package or force mutation before finalization, so a pre-finalizer
# ownership rollback has no Event 006 registry state to unwind.
independence_wave_rollback_reversible_frozen_country_packages = {
	clr_global_flag = liberation_release_event6_rollback_cleanup_validated
	if = {
		limit = {
			NOT = { has_global_flag = liberation_release_finalizer_started }
			NOT = { has_global_flag = liberation_release_event6_origin_commit_started }
		}
		set_global_flag = liberation_release_event6_rollback_cleanup_validated
	}
	if = {
		limit = {
			NOT = { has_global_flag = liberation_release_event6_rollback_cleanup_validated }
			check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
		}
		set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.event6_package_cleanup }
	}
}

independence_wave_execute_standalone_frozen_plan = {
	clr_global_flag = independence_wave_execution_metadata_valid
	clr_global_flag = independence_wave_execution_failed
	if = {
		limit = {
			check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.independence_wave }
			has_global_flag = liberation_release_plan_includes_event6
			NOT = { has_global_flag = liberation_release_plan_includes_event5 }
			has_global_flag = independence_wave_plan_contribution_ready
		}
		liberation_release_lock_plan = yes
		independence_wave_validate_execution_metadata = yes
		if = {
			limit = { has_global_flag = independence_wave_execution_metadata_valid }
			liberation_release_prepare_host_capitals_for_execution = yes
			if = {
				limit = { check_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.none } }
				liberation_release_begin_execution = yes
				if = {
					limit = { is_liberation_release_plan_executing = yes }
					set_temp_variable = { independence_wave_execution_instantiated_count = constant:independence_wave_planner.no_candidates }
					set_temp_variable = { independence_wave_execution_expected_state_count = constant:independence_wave_planner.no_candidates }
					set_temp_variable = { independence_wave_execution_transferred_state_count = constant:independence_wave_planner.no_candidates }
					set_temp_variable = { independence_wave_execution_transfer_failure = constant:liberation_plan_reject_reason.none }
					set_temp_variable = { independence_wave_execution_validated_count = constant:independence_wave_planner.no_candidates }
					if = {
						limit = {
							check_variable = {
								var = global.independence_wave_plan_selected_count
								value = global.independence_wave_plan_target_count
								compare = equals
							}
						}
						independence_wave_instantiate_frozen_countries = yes
					}
					if = {
						limit = {
							check_variable = {
								var = independence_wave_execution_instantiated_count
								value = global.independence_wave_plan_selected_count
								compare = equals
							}
						}
						independence_wave_transfer_frozen_states = yes
					}
					if = {
						limit = {
							check_variable = { var = independence_wave_execution_instantiated_count value = global.independence_wave_plan_selected_count compare = equals }
							check_variable = { independence_wave_execution_transfer_failure = constant:liberation_plan_reject_reason.none }
							check_variable = { var = independence_wave_execution_transferred_state_count value = independence_wave_execution_expected_state_count compare = equals }
							liberation_release_execution_matches_frozen_ownership = yes
						}
						liberation_release_begin_finalization = yes
					}
					if = {
						limit = { is_liberation_release_plan_finalizing = yes }
						independence_wave_prepare_finalizer_frozen_country_origins = yes
						if = {
							limit = {
								check_variable = { var = independence_wave_execution_validated_count value = global.independence_wave_plan_selected_count compare = equals }
							}
							independence_wave_finalize_frozen_country_origins = yes
							if = {
								limit = {
									check_variable = { var = independence_wave_execution_initialized_count value = global.independence_wave_plan_selected_count compare = equals }
								}
								set_global_flag = liberation_release_event6_finalization_validated
								independence_wave_commit_wave_history = yes
								liberation_release_commit_plan = yes
								if = {
									limit = { check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.committed } }
									independence_wave_consume_committed_breakaway_sponsorships = yes
								}
								else = {
									set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.invalid_phase }
									liberation_release_mark_finalization_failed = yes
								}
							}
							else = {
								set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.event6_origin_commit }
								liberation_release_mark_finalization_failed = yes
							}
						}
						else = {
							set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.event6_package_setup }
							liberation_release_mark_finalization_failed = yes
						}
					}
					if = {
						limit = { is_liberation_release_plan_executing = yes }
						set_global_flag = independence_wave_execution_failed
						set_variable = { global.independence_wave_execution_last_failure = constant:liberation_plan_reject_reason.unsafe_instantiation }
					}
					else_if = {
						limit = { has_liberation_release_plan_finalization_failed = yes }
						set_global_flag = independence_wave_execution_failed
						set_variable = { global.independence_wave_execution_last_failure = constant:liberation_plan_reject_reason.unsafe_instantiation }
					}
				}
				else = {
					set_global_flag = independence_wave_execution_failed
					set_variable = { global.independence_wave_execution_last_failure = global.liberation_plan_last_failure }
				}
			}
			else = {
				set_global_flag = independence_wave_execution_failed
				set_variable = { global.independence_wave_execution_last_failure = liberation_host_capital_relocation_failure }
			}
		}
		else = { set_global_flag = independence_wave_execution_failed }
	}
	else = {
		set_global_flag = independence_wave_execution_failed
		set_variable = { global.independence_wave_execution_last_failure = constant:liberation_plan_reject_reason.stale_plan }
	}
}

# Complete standalone Event 006 transaction. Presentation is deliberately
# outside this effect so a failed allocation can never display a release that
# did not happen. Pre-mutation failures restore host capitals and clear the
# Event 006 contribution; failures after the execution barrier run the exact
# compensating ledger before any reservation metadata can be discarded.
independence_wave_prepare_and_execute_standalone_incident = {
	clr_global_flag = independence_wave_standalone_incident_committed
	clr_global_flag = independence_wave_standalone_incident_cancelled_before_mutation
	clr_global_flag = independence_wave_standalone_incident_failed_after_mutation
	clr_global_flag = independence_wave_standalone_incident_failed_during_finalization
	clr_global_flag = independence_wave_standalone_incident_rolled_back_after_mutation
	set_temp_variable = { independence_wave_standalone_plan_begun = constant:independence_wave_planner.no_candidates }

	independence_wave_capture_wave_tuning = yes
	set_temp_variable = { liberation_call_mode = constant:liberation_plan_mode.automatic }
	set_temp_variable = { liberation_call_plan_owner = constant:liberation_plan_owner.independence_wave }
	set_temp_variable = { liberation_call_expected_country_count = independence_wave_wave_target_count }
	liberation_release_begin_plan = yes
	if = {
		limit = {
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.collecting }
			check_variable = { global.liberation_plan_mode = constant:liberation_plan_mode.automatic }
			check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.independence_wave }
		}
		set_temp_variable = { independence_wave_standalone_plan_begun = constant:independence_wave_planner.selected }
	}

	if = {
		limit = { check_variable = { independence_wave_standalone_plan_begun = constant:independence_wave_planner.selected } }
		liberation_release_enter_allocation_phase = yes
	}
	if = {
		limit = { check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.allocating } }
		independence_wave_allocate_automatic_packages = yes
	}
	if = {
		limit = { has_global_flag = independence_wave_plan_contribution_ready }
		independence_wave_expand_selected_optional_territory = yes
	}
	if = {
		limit = {
			has_global_flag = independence_wave_plan_contribution_ready
			NOT = { has_global_flag = independence_wave_plan_optional_expansion_failed }
		}
		independence_wave_execute_standalone_frozen_plan = yes
	}

	if = {
		limit = { check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.committed } }
		set_global_flag = independence_wave_standalone_incident_committed
	}
	else_if = {
		limit = {
			check_variable = { independence_wave_standalone_plan_begun = constant:independence_wave_planner.selected }
			has_liberation_release_plan_finalization_failed = yes
		}
		set_global_flag = independence_wave_standalone_incident_failed_after_mutation
		set_global_flag = independence_wave_standalone_incident_failed_during_finalization
	}
	else_if = {
		limit = {
			check_variable = { independence_wave_standalone_plan_begun = constant:independence_wave_planner.selected }
			has_global_flag = liberation_release_execution_started
		}
		set_global_flag = independence_wave_standalone_incident_failed_after_mutation
		liberation_release_execute_compensating_rollback = yes
		if = {
			limit = { has_global_flag = liberation_release_plan_rollback_completed }
			set_global_flag = independence_wave_standalone_incident_rolled_back_after_mutation
		}
	}
	else_if = {
		limit = { check_variable = { independence_wave_standalone_plan_begun = constant:independence_wave_planner.selected } }
		liberation_release_restore_host_capitals_before_execution = yes
		if = {
			limit = { NOT = { has_global_flag = liberation_release_capital_restore_failed } }
			independence_wave_clear_plan_contribution = yes
			clr_global_flag = independence_wave_plan_contribution_ready
			clr_global_flag = independence_wave_plan_exact_count_failed
			clr_global_flag = independence_wave_plan_pool_exhausted
			clr_global_flag = independence_wave_plan_optional_expansion_failed
			liberation_release_abort_plan = yes
			set_global_flag = independence_wave_standalone_incident_cancelled_before_mutation
		}
		else = { liberation_release_abort_plan = yes }
	}
}
