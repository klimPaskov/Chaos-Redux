# ---------------------------------------------------------------------------
# Persisted cohort ledger
# ---------------------------------------------------------------------------
# Every row keeps aligned identity, ownership, amount, and lifecycle arrays:
# original state, current host, and (only after a real destination scope is
# entered) destination. A missing destination remains unbound and is never
# replaced with a random neighbour.

migration_record_displaced_cohort = {
	humanitarian_initialize_runtime = yes
	set_temp_variable = { migration_cohort_record_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_id_result = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			migration_cohort_record_request_is_valid = yes
			migration_cohort_arrays_are_aligned = yes
			NOT = { has_global_flag = migration_runtime_needs_repair }
		}
		set_temp_variable = { migration_cohort_new_id = global.migration_next_cohort_id }
		add_to_array = { array = global.migration_cohort_ids value = migration_cohort_new_id }
		add_to_array = { array = global.migration_cohort_origins value = THIS }
		add_to_array = { array = global.migration_cohort_hosts value = THIS }
		# The origin placeholder is not a destination identity. Status controls
		# whether this slot may ever be used as a route endpoint.
		add_to_array = { array = global.migration_cohort_destinations value = THIS }
		add_to_array = { array = global.migration_cohort_owners value = OWNER }
		add_to_array = { array = global.migration_cohort_amounts value = migration_cohort_record_amount }
		add_to_array = { array = global.migration_cohort_sources value = migration_cohort_record_source }
		add_to_array = { array = global.migration_cohort_statuses value = constant:migration_cohort_status.active }
		add_to_variable = { global.migration_next_cohort_id = constant:humanitarian_runtime.array_index_increment }
		set_variable = { global.migration_cohort_count = global.migration_cohort_ids^num }
		set_temp_variable = { migration_cohort_id_result = migration_cohort_new_id }
		set_temp_variable = { migration_cohort_record_result = constant:civilian_transfer_route_result.valid }
		set_temp_variable = { migration_cohort_history_id_request = migration_cohort_new_id }
		migration_record_cohort_history_visit = yes
		migration_reconcile_state_cohort_selection = yes
		OWNER = { migration_reconcile_country_cohort_selection = yes }
	}
	set_temp_variable = { migration_cohort_record_request_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_cohort_record_amount = constant:humanitarian_population.zero }
	set_temp_variable = { migration_cohort_record_owner_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_cohort_record_source = constant:humanitarian_pressure_source.unknown }
}


# Bind a destination only from inside that actual destination state scope.
# The regular route target is saved for the current effect chain so callers
# may return to origin safely; the global row remains the durable identity.
migration_save_cohort_previous_host = {
	set_temp_variable = { migration_previous_host_break = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
			migration_cohort_arrays_are_aligned = yes
		}
		set_temp_variable = { migration_previous_host_index = global.migration_cohort_ids^num }
		while_loop_effect = {
			break = migration_previous_host_break
			limit = {
				check_variable = { var = migration_previous_host_index value = constant:humanitarian_runtime.zero compare = greater_than }
				check_variable = { var = migration_previous_host_break value = constant:humanitarian_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { migration_previous_host_index = constant:humanitarian_runtime.array_index_increment }
			if = {
				limit = { check_variable = { var = global.migration_cohort_ids^migration_previous_host_index value = migration_cohort_id_request compare = equals } }
				var:global.migration_cohort_hosts^migration_previous_host_index = { save_event_target_as = migration_cohort_previous_host }
				set_temp_variable = { migration_previous_host_break = constant:humanitarian_runtime.one }
			}
		}
	}
}

		limit = {
			migration_cohort_destination_bind_request_is_valid = yes

		set_temp_variable = { civilian_transfer_bind_receipt_result = constant:civilian_transfer_route_result.invalid }

	set_temp_variable = { migration_cohort_id_request = constant:humanitarian_runtime.zero }
}

# Resolve a row's original state and current host. A destination target is
# returned only for a bound row; unsafe binding remains visible to the forced
# return contract but is rejected by the normal destination safety trigger.
migration_resolve_cohort_origin = {
	humanitarian_initialize_runtime = yes
	set_temp_variable = { migration_cohort_origin_resolution_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_destination_resolution_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_resolution_detail = constant:migration_core_reconciliation.selector_zero }
	set_temp_variable = { migration_cohort_resolution_explicit = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_cohort_resolution_count = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_cohort_resolved_amount = constant:humanitarian_population.zero }
	set_temp_variable = { migration_cohort_resolved_source = constant:humanitarian_pressure_source.unknown }
	set_temp_variable = { migration_cohort_resolved_status = constant:migration_cohort_status.none }
	if = {
		limit = {
			has_variable = migration_cohort_id_request
			check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
		}
		set_temp_variable = { migration_cohort_resolution_explicit = constant:humanitarian_runtime.one }
	}
	if = {
		limit = { check_variable = { var = migration_cohort_resolution_explicit value = constant:humanitarian_runtime.zero compare = equals } }
		if = {
			limit = {
				humanitarian_state_is_valid = yes
				NOT = { has_state_flag = migration_cohort_selection_ambiguous }
				has_variable = migration_current_cohort_id
			}
			set_temp_variable = { migration_cohort_id_request = migration_current_cohort_id }
		}
		else_if = {
			limit = {
				humanitarian_country_is_valid = yes
				NOT = { has_country_flag = migration_country_cohort_selection_ambiguous }
				has_variable = migration_current_country_cohort_id
			}
			set_temp_variable = { migration_cohort_id_request = migration_current_country_cohort_id }
		}
	}
	if = {
		limit = {
			check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
			migration_cohort_arrays_are_aligned = yes
			NOT = { has_global_flag = migration_runtime_needs_repair }
		}
		set_temp_variable = { migration_cohort_resolve_index = global.migration_cohort_ids^num }
		while_loop_effect = {
			limit = { check_variable = { var = migration_cohort_resolve_index value = constant:humanitarian_runtime.zero compare = greater_than } }
			subtract_from_temp_variable = { migration_cohort_resolve_index = constant:humanitarian_runtime.array_index_increment }
			set_temp_variable = { migration_cohort_row_id = global.migration_cohort_ids^migration_cohort_resolve_index }
			if = {
				limit = {
					check_variable = { var = migration_cohort_row_id value = migration_cohort_id_request compare = equals }
					OR = {
						check_variable = { var = global.migration_cohort_statuses^migration_cohort_resolve_index value = constant:migration_cohort_status.active compare = equals }
						check_variable = { var = global.migration_cohort_statuses^migration_cohort_resolve_index value = constant:migration_cohort_status.destination_bound compare = equals }
						check_variable = { var = global.migration_cohort_statuses^migration_cohort_resolve_index value = constant:migration_cohort_status.destination_bound_unsafe compare = equals }
					}
					check_variable = { var = global.migration_cohort_amounts^migration_cohort_resolve_index value = constant:humanitarian_population.zero compare = greater_than }
				}
				add_to_temp_variable = { migration_cohort_resolution_count = constant:humanitarian_runtime.one }
				if = {
					limit = { check_variable = { var = migration_cohort_resolution_count value = constant:humanitarian_runtime.one compare = equals } }
					set_temp_variable = { migration_cohort_resolved_amount = global.migration_cohort_amounts^migration_cohort_resolve_index }
					set_temp_variable = { migration_cohort_resolved_source = global.migration_cohort_sources^migration_cohort_resolve_index }
					set_temp_variable = { migration_cohort_resolved_status = global.migration_cohort_statuses^migration_cohort_resolve_index }
					var:global.migration_cohort_origins^migration_cohort_resolve_index = { save_event_target_as = migration_cohort_origin }
					var:global.migration_cohort_hosts^migration_cohort_resolve_index = { save_event_target_as = migration_cohort_host }
					var:global.migration_cohort_owners^migration_cohort_resolve_index = { save_event_target_as = migration_cohort_owner }
				}
				if = {
					limit = {
						OR = {
							check_variable = { var = migration_cohort_resolved_status value = constant:migration_cohort_status.destination_bound compare = equals }
							check_variable = { var = migration_cohort_resolved_status value = constant:migration_cohort_status.destination_bound_unsafe compare = equals }
						}
					}
					if = {
						limit = { check_variable = { var = migration_cohort_resolution_count value = constant:humanitarian_runtime.one compare = equals } }
						var:global.migration_cohort_destinations^migration_cohort_resolve_index = { save_event_target_as = migration_cohort_destination }
						set_temp_variable = { migration_cohort_destination_resolution_result = constant:civilian_transfer_route_result.valid }
					}
				}
			}
		}
		if = {
			limit = { check_variable = { var = migration_cohort_resolution_count value = constant:humanitarian_runtime.one compare = equals } }
			set_temp_variable = { migration_cohort_origin_resolution_result = constant:civilian_transfer_route_result.valid }
			set_temp_variable = { migration_cohort_resolution_detail = constant:migration_core_reconciliation.selector_one }
		}
		if = {
			limit = { check_variable = { var = migration_cohort_resolution_count value = constant:humanitarian_runtime.one compare = greater_than } }
			set_temp_variable = { migration_cohort_origin_resolution_result = constant:civilian_transfer_route_result.invalid }
			set_temp_variable = { migration_cohort_destination_resolution_result = constant:civilian_transfer_route_result.invalid }
			set_temp_variable = { migration_cohort_resolution_detail = constant:migration_core_reconciliation.selector_ambiguous }
		}
	}
	if = {
		limit = { NOT = { migration_cohort_arrays_are_aligned = yes } }
		set_temp_variable = { migration_cohort_resolution_detail = constant:migration_core_reconciliation.selector_misaligned }
	}
}


# Remove a completed or invalid row by id. All aligned arrays are removed at
# the same index, keeping identity/amount/owner conservation intact.
migration_cleanup_cohort_record = {
	humanitarian_initialize_runtime = yes
	set_temp_variable = { migration_cohort_cleanup_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_cleanup_break = constant:humanitarian_runtime.zero }
	if = {
		limit = { migration_cohort_cleanup_request_is_valid = yes }
		set_temp_variable = { migration_cohort_cleanup_id_match_count = constant:humanitarian_runtime.zero }
		set_temp_variable = { migration_cohort_cleanup_match_index = global.migration_cohort_ids^num }
		while_loop_effect = {
			limit = { check_variable = { var = migration_cohort_cleanup_match_index value = constant:humanitarian_runtime.zero compare = greater_than } }
			subtract_from_temp_variable = { migration_cohort_cleanup_match_index = constant:humanitarian_runtime.array_index_increment }
			if = {
				limit = { check_variable = { var = global.migration_cohort_ids^migration_cohort_cleanup_match_index value = migration_cohort_id_request compare = equals } }
				add_to_temp_variable = { migration_cohort_cleanup_id_match_count = constant:humanitarian_runtime.one }
			}
		}
		set_temp_variable = { migration_cohort_cleanup_index = global.migration_cohort_ids^num }
		while_loop_effect = {
			break = migration_cohort_cleanup_break
			limit = {
				check_variable = { var = migration_cohort_cleanup_index value = constant:humanitarian_runtime.zero compare = greater_than }
				check_variable = { var = migration_cohort_cleanup_break value = constant:humanitarian_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { migration_cohort_cleanup_index = constant:humanitarian_runtime.array_index_increment }
			set_temp_variable = { migration_cohort_row_id = global.migration_cohort_ids^migration_cohort_cleanup_index }
			set_temp_variable = { migration_cohort_row_status = global.migration_cohort_statuses^migration_cohort_cleanup_index }
			set_temp_variable = { migration_cohort_row_amount = global.migration_cohort_amounts^migration_cohort_cleanup_index }
			if = {
				limit = {
					check_variable = { var = migration_cohort_cleanup_id_match_count value = constant:humanitarian_runtime.one compare = equals }
					check_variable = { var = migration_cohort_row_id value = migration_cohort_id_request compare = equals }
					OR = {
						check_variable = { var = migration_cohort_row_status value = constant:migration_cohort_status.completed compare = equals }
						check_variable = { var = migration_cohort_row_status value = constant:migration_cohort_status.invalid compare = equals }
					}
					check_variable = { var = migration_cohort_row_amount value = constant:humanitarian_population.zero compare = equals }
				}
				if = {
					limit = {
						check_variable = { var = migration_cohort_row_id value = migration_cohort_id_request compare = equals }
						OR = {
							check_variable = { var = migration_cohort_row_status value = constant:migration_cohort_status.completed compare = equals }
							check_variable = { var = migration_cohort_row_status value = constant:migration_cohort_status.invalid compare = equals }
						}
						check_variable = { var = migration_cohort_row_amount value = constant:humanitarian_population.zero compare = equals }
					}
					set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.one }
					# Resolve every exact persisted row scope before removal. This remains
					# fail-closed when any old-save target is invalid or still carries a
					# reception, trapped, mission, corridor, or selector obligation.
					set_temp_variable = { migration_cleanup_origin_scope_resolved = constant:humanitarian_runtime.zero }
					set_temp_variable = { migration_cleanup_host_scope_resolved = constant:humanitarian_runtime.zero }
					set_temp_variable = { migration_cleanup_destination_scope_resolved = constant:humanitarian_runtime.zero }
					set_temp_variable = { migration_cleanup_host_country_scope_resolved = constant:humanitarian_runtime.zero }
					set_temp_variable = { migration_cleanup_owner_country_scope_resolved = constant:humanitarian_runtime.zero }
					var:global.migration_cohort_origins^migration_cohort_cleanup_index = {
						set_temp_variable = { migration_cleanup_origin_scope_resolved = constant:humanitarian_runtime.one }
						if = { limit = { NOT = { migration_terminal_state_references_clear = yes } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
						save_event_target_as = migration_cleanup_origin_state
					}
					var:global.migration_cohort_hosts^migration_cohort_cleanup_index = {
						set_temp_variable = { migration_cleanup_host_scope_resolved = constant:humanitarian_runtime.one }
						if = { limit = { NOT = { migration_terminal_state_references_clear = yes } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
						save_event_target_as = migration_cleanup_host_state
						OWNER = {
							set_temp_variable = { migration_cleanup_host_country_scope_resolved = constant:humanitarian_runtime.one }
							if = { limit = { NOT = { migration_terminal_country_references_clear = yes } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
							save_event_target_as = migration_cleanup_host_country
						}
					}
					var:global.migration_cohort_destinations^migration_cohort_cleanup_index = {
						set_temp_variable = { migration_cleanup_destination_scope_resolved = constant:humanitarian_runtime.one }
						if = { limit = { NOT = { migration_terminal_state_references_clear = yes } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
						save_event_target_as = migration_cleanup_destination_state
					}
					var:global.migration_cohort_owners^migration_cohort_cleanup_index = {
						set_temp_variable = { migration_cleanup_owner_country_scope_resolved = constant:humanitarian_runtime.one }
						if = { limit = { NOT = { migration_terminal_country_references_clear = yes } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
						save_event_target_as = migration_cleanup_persisted_owner
					}
					if = { limit = { NOT = { check_variable = { var = migration_cleanup_origin_scope_resolved value = constant:humanitarian_runtime.one compare = equals } } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
					if = { limit = { NOT = { check_variable = { var = migration_cleanup_host_scope_resolved value = constant:humanitarian_runtime.one compare = equals } } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
					if = { limit = { NOT = { check_variable = { var = migration_cleanup_destination_scope_resolved value = constant:humanitarian_runtime.one compare = equals } } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }
					if = { limit = { NOT = { check_variable = { var = migration_cleanup_host_country_scope_resolved value = constant:humanitarian_runtime.one compare = equals } } } set_temp_variable = { migration_cohort_cleanup_references_proven = constant:humanitarian_runtime.zero } }

# State scope. Roll back exactly one active positive cohort row staged in this
# effect chain when the atomic transfer never debited population. This is not
# terminal cleanup: it also removes the origin visit created by row staging and

				var:global.migration_cohort_hosts^migration_staged_cohort_abort_match_index = { state = THIS }
				var:global.migration_cohort_destinations^migration_staged_cohort_abort_match_index = { state = THIS }
			}

			remove_from_array = { array = global.migration_cohort_ids index = migration_staged_cohort_abort_match_index }

			remove_from_array = { array = global.migration_cohort_statuses index = migration_staged_cohort_abort_match_index }
			set_variable = { global.migration_cohort_count = global.migration_cohort_ids^num }
			migration_reconcile_state_cohort_selection = yes
			OWNER = { migration_reconcile_country_cohort_selection = yes }
			set_temp_variable = { migration_staged_cohort_abort_result = constant:civilian_transfer_route_result.valid }
		}
	}
	set_temp_variable = { migration_staged_cohort_abort_proven = constant:humanitarian_runtime.zero }

# STATE scope. Reveal the migration category only after the accepted
# repeated/large thresholds are met. This projection is invoked only for an

				OWNER = {
					check_variable = { var = migration_incident_count value = constant:migration_decision_threshold.category_incidents compare = greater_than_or_equals }
				}
				check_variable = { var = migration_flight_pressure value = constant:migration_decision_threshold.category_flight_people compare = greater_than_or_equals }

		}
		OWNER = {
			migration_register_active_country = yes
			if = {
				limit = {
					NOT = { has_country_flag = migration_decision_active }
					NOT = { has_country_flag = migration_decision_resolution }
				}
				set_country_flag = migration_decision_emerging
				clr_country_flag = migration_decision_dormant
			}
		}
	}
}

	set_temp_variable = { migration_pressure_request_result = constant:humanitarian_runtime.zero }
	if = {
		limit = { migration_pressure_request_is_valid = yes }
		set_temp_variable = { migration_new_incident = constant:humanitarian_runtime.zero }

			limit = {
				OR = {
					NOT = { has_variable = migration_flight_pressure }
					check_variable = { var = migration_flight_pressure value = constant:humanitarian_population.zero compare = less_than_or_equals }
				}
			}
			set_temp_variable = { migration_new_incident = constant:humanitarian_runtime.one }
		}
		migration_initialize_state = yes
		add_to_variable = { migration_flight_pressure = migration_pressure_request_amount }
		add_to_variable = { migration_state_flight_population = migration_pressure_request_amount }
		set_variable = { migration_last_pressure_source = migration_pressure_request_source }

			migration_register_active_country = yes

				limit = { check_variable = { var = migration_new_incident value = constant:humanitarian_runtime.one compare = equals } }
				add_to_variable = { migration_incident_count = constant:humanitarian_runtime.one }
			}
		}
		migration_register_initial_incident = yes
		migration_refresh_decision_phase_from_state = yes
		# A famine-owned stage may supply a generation-bound request receipt.
		# Only this migration owner writes the migration request-generation and
		# accepted-total ledgers; ordinary pressure callers omit the optional
		# field and retain their existing behavior.
		if = {
			limit = {
				has_variable = migration_pressure_request_generation
				check_variable = { var = migration_pressure_request_generation value = constant:humanitarian_runtime.zero compare = greater_than }
			}
			if = { limit = { NOT = { has_variable = migration_flight_request_accepted_total } } set_variable = { migration_flight_request_accepted_total = constant:humanitarian_population.zero } }
			add_to_variable = { migration_flight_request_accepted_total = migration_pressure_request_amount }
			set_variable = { migration_flight_request_generation = migration_pressure_request_generation }
		}
		set_temp_variable = { migration_pressure_request_result = constant:humanitarian_runtime.one }
	}
	set_temp_variable = { migration_pressure_request_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_pressure_request_amount = constant:humanitarian_population.zero }
	set_temp_variable = { migration_pressure_request_source = constant:humanitarian_pressure_source.unknown }
	set_temp_variable = { migration_pressure_request_actor_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_pressure_request_generation = constant:humanitarian_runtime.zero }
}

# Called only by the deferred finalizer. It is the sole transfer-time owner of
# cohort destination, host, amount, and status mutation. A surviving row moves
# to the exact route destination and records one visit; an all-death row stays
# at its current host with zero amount and completed status.
migration_update_cohort_host_after_transfer = {
	set_temp_variable = { migration_cohort_host_update_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_host_update_break = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			civilian_transfer_finalize_request_is_valid = yes
			check_variable = { var = civilian_transfer_finalize_cohort_id value = constant:humanitarian_runtime.zero compare = greater_than }
			migration_cohort_arrays_are_aligned = yes
		}
		set_temp_variable = { migration_cohort_host_update_index = global.migration_cohort_ids^num }
		while_loop_effect = {
			break = migration_cohort_host_update_break
			limit = {
				check_variable = { var = migration_cohort_host_update_index value = constant:humanitarian_runtime.zero compare = greater_than }
				check_variable = { var = migration_cohort_host_update_break value = constant:humanitarian_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { migration_cohort_host_update_index = constant:humanitarian_runtime.array_index_increment }
			set_temp_variable = { migration_cohort_row_id = global.migration_cohort_ids^migration_cohort_host_update_index }
			if = {
				limit = { check_variable = { var = migration_cohort_row_id value = civilian_transfer_finalize_cohort_id compare = equals } }
				if = {
					limit = { check_variable = { var = civilian_transfer_survivor_credit value = constant:humanitarian_population.zero compare = greater_than } }
					remove_from_array = { array = global.migration_cohort_destinations index = migration_cohort_host_update_index }
					add_to_array = { array = global.migration_cohort_destinations value = event_target:civilian_transfer_route_destination index = migration_cohort_host_update_index }
					remove_from_array = { array = global.migration_cohort_hosts index = migration_cohort_host_update_index }
					add_to_array = { array = global.migration_cohort_hosts value = event_target:civilian_transfer_route_destination index = migration_cohort_host_update_index }
					remove_from_array = { array = global.migration_cohort_amounts index = migration_cohort_host_update_index }
					add_to_array = { array = global.migration_cohort_amounts value = civilian_transfer_survivor_credit index = migration_cohort_host_update_index }
					remove_from_array = { array = global.migration_cohort_statuses index = migration_cohort_host_update_index }
					if = {
						limit = {
							OR = {
								check_variable = { var = civilian_transfer_finalize_bind_mode value = constant:migration_core_reconciliation.transfer_bind_forced compare = equals }
								check_variable = { var = civilian_transfer_finalize_bind_mode value = constant:migration_core_reconciliation.transfer_rebind_forced compare = equals }
							}
						}
						add_to_array = { array = global.migration_cohort_statuses value = constant:migration_cohort_status.destination_bound_unsafe index = migration_cohort_host_update_index }
					}
					else = {
						add_to_array = { array = global.migration_cohort_statuses value = constant:migration_cohort_status.destination_bound index = migration_cohort_host_update_index }
					}
					event_target:civilian_transfer_route_destination = {
						set_temp_variable = { migration_cohort_history_id_request = civilian_transfer_finalize_cohort_id }
						migration_record_cohort_history_visit = yes
					}
				}
				else = {
					# A zero-survivor result is a terminal row, not permission to
					# erase the aligned ledger during the transfer transaction.
					remove_from_array = { array = global.migration_cohort_amounts index = migration_cohort_host_update_index }
					add_to_array = { array = global.migration_cohort_amounts value = constant:humanitarian_population.zero index = migration_cohort_host_update_index }
					remove_from_array = { array = global.migration_cohort_statuses index = migration_cohort_host_update_index }
					add_to_array = { array = global.migration_cohort_statuses value = constant:migration_cohort_status.completed index = migration_cohort_host_update_index }
				}
				if = { limit = { has_event_target = civilian_transfer_physical_origin } event_target:civilian_transfer_physical_origin = { migration_reconcile_state_cohort_selection = yes } }
				if = { limit = { has_event_target = civilian_transfer_route_destination } event_target:civilian_transfer_route_destination = { migration_reconcile_state_cohort_selection = yes } }
				if = { limit = { has_event_target = migration_cohort_owner } event_target:migration_cohort_owner = { migration_reconcile_country_cohort_selection = yes } }
				if = {
					limit = { has_event_target = civilian_transfer_physical_origin }
					event_target:civilian_transfer_physical_origin = { OWNER = { migration_reconcile_country_cohort_selection = yes } }
				}
				if = {
					limit = { has_event_target = civilian_transfer_route_destination }
					event_target:civilian_transfer_route_destination = { OWNER = { migration_reconcile_country_cohort_selection = yes } }
				}
				set_temp_variable = { migration_cohort_host_update_result = constant:civilian_transfer_route_result.valid }
				set_temp_variable = { migration_cohort_host_update_break = constant:humanitarian_runtime.one }
			}
		}
	}
}

# State scope. Rebuilds reception/overcrowding projection flags from the
# authoritative state load and the owner-country capacity/load contract.
# Country load is aggregate and is never inferred by scanning other states.
migration_refresh_reception_context = {
	if = {
		limit = { humanitarian_state_is_valid = yes }
		save_event_target_as = migration_reception_context_state
		if = {
			limit = { check_variable = { var = migration_state_reception_load value = constant:humanitarian_population.zero compare = greater_than } }
			set_state_flag = migration_reception_context_active
		}
		else = { clr_state_flag = migration_reception_context_active }
		OWNER = {
			if = {
				limit = {
					has_variable = migration_reception_capacity
					has_variable = migration_reception_load
					check_variable = { var = migration_reception_load value = migration_reception_capacity compare = greater_than }
				}
				event_target:migration_reception_context_state = { set_state_flag = migration_overcrowded_context_active }
			}
			else = { event_target:migration_reception_context_state = { clr_state_flag = migration_overcrowded_context_active } }
		}
		migration_mark_reception_observation_breach = yes
		migration_refresh_dynamic_modifiers = yes
		migration_presentation_refresh_state = yes
	}
}


# State scope. Observes only an armed exact reception subject. A host outbreak
# or capacity breach permanently invalidates the current medical-reception
# attempt; the non-selectable mission consumes the breach on its next bounded
# lifecycle evaluation. The observer never creates an outbreak or population.
migration_mark_reception_observation_breach = {
	if = {
		limit = {
			has_state_flag = migration_mission_reception_subject
			has_state_flag = migration_mission_reception_observation_pending
			OR = {
				black_plague_state_is_infected_or_worse = yes
				has_state_flag = migration_overcrowded_context_active
				OWNER = {
					has_variable = migration_reception_capacity
					has_variable = migration_reception_load
					check_variable = {
						var = migration_reception_load
						value = migration_reception_capacity
						compare = greater_than_or_equals
					}
				}
			}
		}
		set_state_flag = migration_mission_reception_observation_breach
		OWNER = { migration_achievement_fail_medical_observation = yes }
	}
}


# State scope. Applies one positive, already-proven actual survivor credit or
# debit to both the state-local load and its owner-country aggregate. Debit
# requests fail closed unless both ledgers contain the full requested amount;
# this makes the helper conservation-safe and prevents double counting. The
# caller must invoke it once per exact transfer transaction.
migration_apply_reception_delta = {
	set_temp_variable = { migration_reception_delta_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_reception_delta_applied = constant:humanitarian_population.zero }
	if = {
		limit = { migration_reception_delta_request_is_valid = yes }
		migration_initialize_state = yes
		if = {
			limit = { check_variable = { var = migration_reception_delta_mode value = constant:migration_reception_delta.credit compare = equals } }
			add_to_variable = { migration_state_reception_load = migration_reception_delta_amount }
			set_state_flag = migration_displacement_active
			migration_register_active_state = yes
			OWNER = {
				add_to_variable = { migration_reception_load = migration_reception_delta_amount }
				migration_register_active_country = yes
			}
		}
		else_if = {
			limit = { check_variable = { var = migration_reception_delta_mode value = constant:migration_reception_delta.debit compare = equals } }
			subtract_from_variable = { migration_state_reception_load = migration_reception_delta_amount }
			OWNER = { subtract_from_variable = { migration_reception_load = migration_reception_delta_amount } }
		}
		set_temp_variable = { migration_reception_delta_applied = migration_reception_delta_amount }
		migration_register_reception_candidate_state = yes
		OWNER = { migration_mark_reception_capacity_dirty = yes }
		if = {
			limit = { NOT = { check_variable = { var = migration_reception_delta_defer_refresh value = constant:humanitarian_runtime.one compare = equals } } }
			migration_refresh_reception_context = yes
			migration_refresh_decision_phase_from_state = yes
		}
		set_temp_variable = { migration_reception_delta_result = constant:civilian_transfer_route_result.valid }
	}
	set_temp_variable = { migration_reception_delta_request_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_delta_amount = constant:humanitarian_population.zero }
	set_temp_variable = { migration_reception_delta_mode = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_delta_defer_refresh = constant:humanitarian_runtime.zero }
}

# STATE scope. Native state projections are refreshed only for a state already
# present in one of the sparse registries. Railway damage is an availability
# signal, not proof of a route or distant geometry. Strategic-bombing recency
# is state-local and deliberately carries no attacker attribution.
migration_refresh_native_state_safety_projections = {
	if = {
		limit = { humanitarian_state_is_valid = yes }
		if = {
			limit = { days_since_last_strategic_bombing < constant:chaos_meter_deaths.strategic_bombing_days_recent }
			if = {
				limit = { NOT = { has_state_flag = migration_bombing_active } }
				set_state_flag = migration_bombing_active
				if = { limit = { has_state_flag = migration_reception_candidate_active } OWNER = { migration_mark_reception_capacity_dirty = yes } }
			}
		}
		else_if = {
			limit = { has_state_flag = migration_bombing_active }
			clr_state_flag = migration_bombing_active
			if = { limit = { has_state_flag = migration_reception_candidate_active } OWNER = { migration_mark_reception_capacity_dirty = yes } }
		}
	}
}

		limit = {
			OR = {
				check_variable = { var = migration_cleanup_requested value = constant:humanitarian_runtime.zero compare = greater_than }
				NOT = { humanitarian_country_is_valid = yes }
			}
		}
		if = {
			limit = { has_variable = humanitarian_corridor_offer_origin_state_id }
			var:humanitarian_corridor_offer_origin_state_id = {
				set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.invalid_owner }
				set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.invalidated }
				humanitarian_corridor_cleanup = yes
			}
		}
		else_if = {
			limit = { has_variable = humanitarian_corridor_origin_state_id }
			var:humanitarian_corridor_origin_state_id = {
				set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.invalid_owner }
				set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.invalidated }
				humanitarian_corridor_cleanup = yes
			}
		}
		migration_reconcile_country_cohort_selection = yes
		migration_recount_active_mission_slots = yes
		migration_unregister_active_country = yes
		clear_variable = migration_war_reassessment_pending
		clear_variable = migration_peace_reassessment_pending
		# Country retirement is likewise non-destructive. Durable integrated and
		# resettled outcome flags are history and are never erased here.
		if = {
			limit = { NOT = { migration_country_has_obligations = yes } }
			clr_country_flag = migration_border_policy_active
			clr_country_flag = migration_voluntary_return_eligible
			clr_country_flag = migration_forced_return_pending
			clr_country_flag = humanitarian_corridor_mission_active
			clr_country_flag = migration_mission_evacuation_active
			clr_country_flag = migration_mission_reception_active
			clr_country_flag = migration_medical_reception_active
			clr_country_flag = migration_mission_return_active
			set_variable = { migration_active_mission_count = constant:humanitarian_runtime.zero }
			clear_variable = migration_border_policy
			clear_variable = migration_current_country_cohort_id
			clear_variable = migration_cohort_resolution_transaction_proven
			clr_country_flag = migration_country_cohort_selection_ambiguous
		}
		clear_variable = migration_cleanup_requested
	}
}

migration_consume_registered_state_reassessment = {
	if = {
		limit = { humanitarian_state_is_valid = yes }
		if = { limit = { has_state_flag = migration_control_change_pending } clr_state_flag = migration_control_change_pending }
		if = { limit = { has_state_flag = migration_nuclear_reassessment_pending } clr_state_flag = migration_nuclear_reassessment_pending }
	}
}

migration_handle_state_control_change = {
	migration_cleanup_mission_subjects_after_control_change = yes
	if = {
		limit = { humanitarian_state_is_valid = yes }
		set_state_flag = migration_control_change_pending
		if = { limit = { has_state_flag = migration_displacement_active } migration_register_active_state = yes }
		if = { limit = { has_state_flag = migration_reception_candidate_active } OWNER = { migration_mark_reception_capacity_dirty = yes } }
		if = { limit = { has_state_flag = migration_reception_candidate_active has_event_target = migration_previous_controller } event_target:migration_previous_controller = { migration_mark_reception_capacity_dirty = yes } }
		migration_refresh_dynamic_modifiers = yes
		migration_presentation_refresh_state = yes
	}
	else = {
		migration_cleanup_cohort_records_for_state = yes
		migration_cleanup_state_registration = yes
	}
}

migration_cleanup_mission_subjects_after_control_change = {
	if = {
		limit = { has_state_flag = humanitarian_corridor_mission_subject }
		clr_state_flag = humanitarian_corridor_mission_subject
		clr_state_flag = humanitarian_corridor_mission_success_proven
		clear_variable = humanitarian_corridor_mission_cohort_id
	}
	if = {
		limit = { has_state_flag = migration_mission_evacuation_subject }
		clr_state_flag = migration_mission_evacuation_subject
		clr_state_flag = migration_mission_evacuation_success_proven
		clear_variable = migration_mission_evacuation_cohort_id
		clear_variable = migration_mission_evacuation_deadline
	}
	if = {
		limit = { has_state_flag = migration_mission_reception_subject }
		if = {
			limit = { has_event_target = migration_previous_controller }
			event_target:migration_previous_controller = { migration_achievement_fail_medical_observation = yes }
		}
		clr_state_flag = migration_mission_reception_subject
		clr_state_flag = migration_mission_reception_observation_pending
		clr_state_flag = migration_mission_reception_observation_breach
		clear_variable = migration_mission_reception_cohort_id
		clear_variable = migration_mission_reception_observation_deadline
	}
	if = {
		limit = { has_state_flag = migration_mission_return_subject }
		clr_state_flag = migration_mission_return_subject
		clear_variable = migration_mission_return_deadline
	}
}
