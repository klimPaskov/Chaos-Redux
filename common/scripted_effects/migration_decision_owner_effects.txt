# Decision-owned adapters for bounded migration actions.
#
# These effects do not select destinations, mutate ledgers, or replace the
# authoritative transfer core. They only convert a selected destination's
# affirmative destination contract into the route-safety marker consumed by
# the exact transfer request. A missing, stale, or invalid destination leaves
# the marker at zero and therefore fails the transfer closed.

# STATE scope. Mission deadlines are stored dates. Date variables count one
# game hour as 0.00001, so each day count passes through
# chaosx_convert_days_to_date_offset before it is added to global.date.

# Set migration_mission_evacuation_deadline evacuation_protection_days after today.
migration_set_mission_evacuation_deadline = {
	set_temp_variable = { chaosx_date_days = constant:migration_mission_timing.evacuation_protection_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { migration_mission_evacuation_deadline = global.date }
	add_to_variable = { migration_mission_evacuation_deadline = chaosx_date_offset }
}

# STATE scope at the selected physical origin. Stage the new organized-
# evacuation cohort before any population debit so the atomic preflight can
# prove an exact current host. A failed preflight/zero-debit attempt removes
# only that same-chain provisional row and its origin visit.
migration_decision_stage_trapped_obligation = {
	set_temp_variable = { civilian_transfer_trapped_obligation_proven = constant:humanitarian_runtime.zero }
	if = {
		limit = { check_variable = { var = migration_trapped_population value = constant:humanitarian_population.zero compare = greater_than } }
		set_temp_variable = { civilian_transfer_trapped_obligation_proven = constant:humanitarian_runtime.one }
	}
}

migration_decision_refresh_trapped_state_flag = {
	if = {
		limit = { check_variable = { var = migration_trapped_population value = constant:humanitarian_population.zero compare = greater_than } }
		set_state_flag = migration_population_trapped
	}
	else = {
		clr_state_flag = migration_population_trapped
	}
}

# STATE scope at the exact destination after a finalized organized evacuation.
# Migration owns preparation cleanup, mission subject/cohort/deadline state,
# the mission-slot ledger, wartime reporting, and corridor-subject resolution.
migration_decision_finalize_organized_evacuation = {
	if = {
		limit = { NOT = { check_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one } } }
		set_temp_variable = { humanitarian_action_actor = ROOT }
		set_temp_variable = { humanitarian_action_target = FROM }
	}
	migration_decision_finalize_organized_evacuation_in_context = yes
}

migration_decision_finalize_organized_evacuation_in_context = {
	set_temp_variable = { migration_decision_evacuation_finalize_result = constant:humanitarian_runtime.zero }
		if = {
			limit = {
				check_variable = { var = civilian_transfer_finalize_result value = constant:migration_core_reconciliation.transfer_finalize_valid compare = equals }
				check_variable = { var = civilian_transfer_actual_origin_debit value = constant:humanitarian_population.zero compare = greater_than }
				check_variable = { var = civilian_transfer_survivor_credit value = constant:humanitarian_population.zero compare = greater_than }
				check_variable = { var = civilian_transfer_projection_result value = constant:civilian_transfer_route_result.valid compare = equals }
				check_variable = { var = migration_decision_atomic_transfer_result value = constant:civilian_transfer_route_result.valid compare = equals }
			}
			clr_state_flag = migration_evacuation_prepared
			if = {
				limit = {
					check_variable = { var = migration_cohort_record_result value = constant:civilian_transfer_route_result.valid compare = equals }
					check_variable = { var = civilian_transfer_survivor_credit value = constant:migration_mission_contract.minimum_survivor_credit compare = greater_than }
					NOT = { var:humanitarian_action_actor = { has_country_flag = migration_mission_evacuation_active } }
					var:humanitarian_action_actor = { migration_country_has_free_mission_slot = yes }
				}
				set_state_flag = migration_mission_evacuation_subject
				set_variable = { migration_mission_evacuation_cohort_id = migration_cohort_id_result }
				migration_set_mission_evacuation_deadline = yes
				var:humanitarian_action_actor = {
					migration_claim_mission_slot = yes
					set_country_flag = migration_mission_evacuation_active
					if = { limit = { has_war = yes } migration_show_report_wartime_evacuation = yes }
				}
			}
			if = {
				limit = { has_state_flag = humanitarian_corridor_mission_subject }
				set_state_flag = humanitarian_corridor_mission_success_proven
				migration_decision_refresh_trapped_state_flag = yes
			}
			set_temp_variable = { migration_decision_evacuation_finalize_result = constant:humanitarian_runtime.one }
		}
}

migration_decision_execute_new_origin_organized_evacuation = {
	set_temp_variable = { migration_decision_atomic_transfer_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_record_request_proven = constant:humanitarian_runtime.one }
	set_temp_variable = { migration_cohort_record_amount = civilian_transfer_people }
	set_temp_variable = { migration_cohort_record_owner_proven = constant:humanitarian_runtime.one }
	set_temp_variable = { migration_cohort_record_source = constant:humanitarian_pressure_source.evacuation }
	set_temp_variable = { migration_cohort_record_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_id_result = constant:humanitarian_runtime.zero }
	migration_record_displaced_cohort = yes
	if = {
		limit = { check_variable = { var = migration_cohort_record_result value = constant:civilian_transfer_route_result.valid compare = equals } }
		set_temp_variable = { migration_presentation_organized_denominator = state_population_k }
		multiply_temp_variable = { migration_presentation_organized_denominator = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = migration_presentation_organized_denominator
		set_temp_variable = { migration_presentation_input_organized_evacuation_proven = constant:migration_presentation.one }
		set_temp_variable = { migration_presentation_input_organized_evacuation_people = civilian_transfer_people }
		set_temp_variable = { migration_presentation_input_organized_evacuation_denominator = migration_presentation_organized_denominator }
		set_temp_variable = { migration_presentation_input_cohort_id = migration_cohort_id_result }
		set_temp_variable = { migration_presentation_input_generation = global.civilian_transfer_generation }
		set_temp_variable = { migration_presentation_input_role = constant:migration_core_reconciliation.transfer_projection_role_organized_evacuation }
		migration_presentation_record_organized_evacuation_projection = yes
		migration_refresh_dynamic_modifiers = yes
		migration_presentation_refresh_state = yes
		event_target:civilian_transfer_route_destination = {
			set_temp_variable = { migration_cohort_destination_food_safe_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_destination_reception_proven = constant:humanitarian_runtime.one }
		}
		set_temp_variable = { migration_cohort_id_request = migration_cohort_id_result }
		set_temp_variable = { civilian_transfer_role_request = constant:migration_core_reconciliation.transfer_projection_role_organized_evacuation }
		set_temp_variable = { civilian_transfer_bind_mode = constant:migration_core_reconciliation.transfer_bind_safe }
		set_temp_variable = { civilian_transfer_source_reception_required = constant:migration_core_reconciliation.transfer_source_reception_not_required }
		civilian_transfer_execute_transaction = yes
		if = {
			limit = { check_variable = { var = civilian_transfer_finalize_result value = constant:migration_core_reconciliation.transfer_finalize_valid compare = equals } }
			set_temp_variable = { migration_decision_atomic_transfer_result = constant:civilian_transfer_route_result.valid }
		}
		else_if = {
			limit = { NOT = { check_variable = { var = civilian_transfer_result value = constant:civilian_transfer_route_result.valid compare = equals } } }
			migration_presentation_clear_preparing_projection = yes
			migration_presentation_clear_organized_evacuation_projection = yes
			migration_refresh_dynamic_modifiers = yes
			migration_presentation_refresh_state = yes
			set_temp_variable = { migration_staged_cohort_abort_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_id_request = migration_cohort_id_result }
			migration_abort_staged_cohort_record = yes
		}
	}
}

# STATE scope at an exact current host. Callers stage the concrete transfer
# role and route proofs; this adapter adds matched source-reception ownership
# and a role-correct safe/forced rebind receipt before invoking the atomic core.
migration_decision_execute_hosted_transfer = {
	set_temp_variable = { migration_decision_atomic_transfer_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_id_request = migration_current_cohort_id }
	if = {
		limit = { check_variable = { var = civilian_transfer_role_request value = constant:migration_core_reconciliation.transfer_projection_role_forced_return compare = equals } }
		set_temp_variable = { civilian_transfer_bind_mode = constant:migration_core_reconciliation.transfer_rebind_forced }
		event_target:civilian_transfer_route_destination = {
			set_temp_variable = { migration_cohort_forced_rebind_request_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_forced_rebind_actor_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_forced_rebind_policy_proven = constant:humanitarian_runtime.one }
		}
	}
	else = {
		set_temp_variable = { civilian_transfer_bind_mode = constant:migration_core_reconciliation.transfer_rebind_safe }
		event_target:civilian_transfer_route_destination = {
			set_temp_variable = { migration_cohort_resettlement_rebind_request_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_resettlement_food_safe_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_resettlement_reception_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_resettlement_route_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_cohort_resettlement_actor_proven = constant:humanitarian_runtime.one }
		}
	}
	set_temp_variable = { civilian_transfer_source_reception_required = constant:migration_core_reconciliation.transfer_source_reception_required }
	civilian_transfer_execute_transaction = yes
	if = {
		limit = { check_variable = { var = civilian_transfer_finalize_result value = constant:migration_core_reconciliation.transfer_finalize_valid compare = equals } }
		set_temp_variable = { migration_decision_atomic_transfer_result = constant:civilian_transfer_route_result.valid }
	}
}

migration_decision_prove_route_contract = {
	if = {
		limit = { NOT = { check_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one } } }
		set_temp_variable = { humanitarian_action_actor = ROOT }
		set_temp_variable = { humanitarian_action_target = FROM }
	}
	migration_decision_prove_route_contract_in_context = yes
}

migration_decision_prove_route_contract_in_context = {
	set_temp_variable = { civilian_transfer_route_safety_proven = constant:humanitarian_runtime.zero }
		if = {
			limit = {
				humanitarian_state_is_valid = yes
				infrastructure > constant:migration_destination_selection_threshold.minimum_route_infrastructure
				check_variable = { var = civilian_transfer_route_border_proven value = constant:humanitarian_runtime.zero compare = greater_than }
				check_variable = { var = civilian_transfer_route_transport_proven value = constant:humanitarian_runtime.zero compare = greater_than }
				has_event_target = civilian_transfer_route_destination
				any_neighbor_state = { state = event_target:civilian_transfer_route_destination }
				event_target:civilian_transfer_route_destination = {
					migration_destination_is_valid = yes
					famine_to_migration_food_safety_is_valid = yes
					is_controlled_by = OWNER
					infrastructure > constant:migration_destination_selection_threshold.minimum_route_infrastructure
					check_variable = { var = famine_to_migration_food_stage value = constant:famine_food_stage.stable compare = equals }
					NOT = { has_state_flag = migration_persecution_active }
					NOT = { has_state_flag = migration_bombing_active }
					NOT = { has_state_flag = migration_population_trapped }
					NOT = { genocide_state_has_active_camps = yes }
					NOT = { is_valid_camp_active_site_state = yes }
					NOT = { cbrn_state_has_chemical_contamination = yes }
					NOT = { has_dynamic_modifier = { modifier = nuclear_fallout_state } }
					NOT = { black_plague_state_is_infected_or_worse = yes }
					OWNER = {
						humanitarian_country_is_valid = yes
						migration_reception_capacity_is_valid = yes
						migration_decision_reception_policy_allows_admission = yes
					}
					CONTROLLER = { NOT = { has_war_with = var:humanitarian_action_actor } }
				}
			}
			set_temp_variable = { civilian_transfer_route_safety_proven = constant:humanitarian_runtime.one }
		}
}

# Return lanes need their own affirmative physical route.  Forced return may
# point at a hazardous historical origin, so this adapter proves only exact
# state identity, adjacency, control, infrastructure, destination capacity,
# and a current destination policy; it does not infer that the route is safe.
migration_decision_prove_exact_return_route_contract = {
	set_temp_variable = { civilian_transfer_route_safety_proven = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			infrastructure > constant:migration_destination_selection_threshold.minimum_route_infrastructure
			check_variable = { var = civilian_transfer_route_border_proven value = constant:humanitarian_runtime.zero compare = greater_than }
			check_variable = { var = civilian_transfer_route_transport_proven value = constant:humanitarian_runtime.zero compare = greater_than }
			has_event_target = civilian_transfer_route_destination
			any_neighbor_state = { state = event_target:civilian_transfer_route_destination }
			event_target:civilian_transfer_route_destination = {
				humanitarian_state_is_valid = yes
				is_controlled_by = OWNER
				infrastructure > constant:migration_destination_selection_threshold.minimum_route_infrastructure
				OWNER = {
					humanitarian_country_is_valid = yes
					migration_reception_capacity_is_valid = yes
					migration_decision_reception_policy_is_valid = yes
				}
			}
		}
		set_temp_variable = { civilian_transfer_route_safety_proven = constant:humanitarian_runtime.one }
	}
}

# Ordinary distribution/transit preflight. The core history writer still owns
# the durable receipt, but this bounded pass prevents a selected destination
# from being used when the exact cohort already visited it. A missing or
# misaligned history registry fails closed; terminal return/integration lanes
# deliberately do not call this adapter.
migration_decision_preflight_destination_history = {
	set_temp_variable = { migration_decision_history_preflight_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { civilian_transfer_result = constant:civilian_transfer_route_result.invalid }
	# This is a read-only transaction guard. Do not initialize or repair the
	# history registry here: a missing registry on an old save must fail closed,
	# and a rejected A->B->A attempt must not mutate global history state.
	set_temp_variable = { migration_decision_history_match_count = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			has_event_target = civilian_transfer_route_destination
			has_variable = migration_cohort_id_request
			check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
			has_global_flag = migration_cohort_history_initialized
			has_variable = global.migration_cohort_history_count
			check_variable = { var = global.migration_cohort_history_count value = constant:humanitarian_runtime.zero compare = greater_than }
			migration_cohort_history_arrays_are_aligned = yes
			migration_cohort_history_live_ledger_arrays_are_aligned = yes
		}
		set_temp_variable = { migration_decision_history_preflight_result = constant:civilian_transfer_route_result.valid }
		set_temp_variable = { migration_decision_history_index = global.migration_cohort_history_ids^num }
		while_loop_effect = {
			limit = {
				check_variable = { var = migration_decision_history_index value = constant:humanitarian_runtime.zero compare = greater_than }
			}
			subtract_from_temp_variable = { migration_decision_history_index = constant:humanitarian_runtime.array_index_increment }
			set_temp_variable = { migration_decision_history_row_id = global.migration_cohort_history_ids^migration_decision_history_index }
			if = {
				limit = { check_variable = { var = migration_decision_history_row_id value = migration_cohort_id_request compare = equals } }
				add_to_temp_variable = { migration_decision_history_match_count = constant:humanitarian_runtime.one }
				var:global.migration_cohort_history_states^migration_decision_history_index = {
					if = {
						limit = { state = event_target:civilian_transfer_route_destination }
						set_temp_variable = { migration_decision_history_preflight_result = constant:civilian_transfer_route_result.invalid }
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = migration_decision_history_match_count
						value = constant:humanitarian_runtime.zero
						compare = greater_than
					}
				}
			}
			set_temp_variable = { migration_decision_history_preflight_result = constant:civilian_transfer_route_result.invalid }
		}
	}
}

# Country-scope policy adapter.  Departure restriction and reception
# admission are persisted separately; the core Border Policy is recomputed as
# the more restrictive supporting value so legacy selectors and localisation
# remain truthful while decision lanes consume the correct dimension.
migration_decision_set_policy = {
	# Authorized external subsystem requests select native laws through their normal on_add publisher.
	# Zero means no request. Validate both dimensions before changing either native slot.
	set_temp_variable = { migration_decision_policy_result = constant:civilian_transfer_route_result.invalid }
	if = {
		limit = {
			national_policy_country_uses_civilian_laws = yes
			OR = {
				check_variable = { migration_departure_policy_request = constant:humanitarian_runtime.zero }
				check_variable = { migration_departure_policy_request = constant:migration_border_policy.humanitarian_open }
				check_variable = { migration_departure_policy_request = constant:migration_border_policy.controlled }
				check_variable = { migration_departure_policy_request = constant:migration_border_policy.closed }
			}
			OR = {
				check_variable = { migration_reception_policy_request = constant:humanitarian_runtime.zero }
				migration_decision_reception_policy_request_is_valid = yes
			}
		}
		national_policy_initialize_country = yes
		if = {
			limit = { check_variable = { migration_departure_policy_request = constant:migration_border_policy.humanitarian_open } }
			if = { limit = { NOT = { has_idea = national_policy_departure_free } } add_ideas = national_policy_departure_free }
		}
		else_if = {
			limit = { check_variable = { migration_departure_policy_request = constant:migration_border_policy.controlled } }
			if = { limit = { NOT = { has_idea = national_policy_departure_permit } } add_ideas = national_policy_departure_permit }
		}
		else_if = {
			limit = { check_variable = { migration_departure_policy_request = constant:migration_border_policy.closed } }
			if = { limit = { NOT = { has_idea = national_policy_departure_prohibited } } add_ideas = national_policy_departure_prohibited }
		}
		if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.humanitarian_open } }
			if = { limit = { NOT = { has_idea = national_policy_reception_humanitarian } } add_ideas = national_policy_reception_humanitarian }
		}
		else_if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.controlled } }
			if = { limit = { NOT = { has_idea = national_policy_reception_controlled } } add_ideas = national_policy_reception_controlled }
		}
		else_if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.transit_only } }
			if = { limit = { NOT = { has_idea = national_policy_reception_transit } } add_ideas = national_policy_reception_transit }
		}
		else_if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.quarantine } }
			if = { limit = { NOT = { has_idea = national_policy_reception_quarantine } } add_ideas = national_policy_reception_quarantine }
		}
		else_if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.closed } }
			if = { limit = { NOT = { has_idea = national_policy_reception_closed } } add_ideas = national_policy_reception_closed }
		}
		else_if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.violent } }
			if = { limit = { NOT = { has_idea = national_policy_reception_violent } } add_ideas = national_policy_reception_violent }
		}
		else_if = {
			limit = { check_variable = { migration_reception_policy_request = constant:migration_border_policy.forced_return } }
			if = { limit = { NOT = { has_idea = national_policy_reception_forced_return } } add_ideas = national_policy_reception_forced_return }
		}
		national_policy_reconcile_country = yes
		set_temp_variable = { migration_decision_policy_result = constant:civilian_transfer_route_result.valid }
	}
	set_temp_variable = { migration_departure_policy_request = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_policy_request = constant:humanitarian_runtime.zero }
}

# State-scope arrival receipt.  The caller enters the exact destination state
# only after the shared transfer, destination bind, reception credit, and
# history visit have all returned valid.  The origin is read from the exact
# persisted cohort target, and infection is captured now as an exposure fact;
# later recovery or current-host infection cannot rewrite this receipt.
migration_decision_record_medical_reception_receipt = {
	set_temp_variable = { migration_decision_reception_receipt_result = constant:civilian_transfer_route_result.invalid }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			migration_cohort_arrays_are_aligned = yes
			has_event_target = migration_cohort_origin
			has_variable = migration_decision_reception_receipt_cohort_id_request
			check_variable = { var = migration_decision_reception_receipt_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
			has_variable = migration_decision_reception_receipt_generation_request
			check_variable = { var = migration_decision_reception_receipt_generation_request value = constant:humanitarian_runtime.zero compare = greater_than }
			check_variable = { var = migration_projection_last_cohort_id value = migration_decision_reception_receipt_cohort_id_request compare = equals }
			check_variable = { var = migration_projection_last_generation value = migration_decision_reception_receipt_generation_request compare = equals }
			check_variable = { var = migration_cohort_history_record_result value = constant:civilian_transfer_route_result.valid compare = equals }
			event_target:migration_cohort_origin = {
				humanitarian_state_is_valid = yes
				black_plague_state_is_infected_or_worse = yes
			}
		}
		set_state_flag = migration_mission_reception_exposure_receipt
		set_variable = { migration_mission_reception_receipt_cohort_id = migration_decision_reception_receipt_cohort_id_request }
		set_variable = { migration_mission_reception_receipt_origin_state_id = event_target:migration_cohort_origin.id }
		set_variable = { migration_mission_reception_receipt_host_state_id = THIS.id }
		set_variable = { migration_mission_reception_receipt_generation = migration_decision_reception_receipt_generation_request }
		set_variable = { migration_mission_reception_receipt_transaction = migration_decision_reception_receipt_generation_request }
		set_variable = { migration_mission_reception_receipt_exposure_class = constant:humanitarian_pressure_source.outbreak }
		set_temp_variable = { migration_decision_reception_receipt_result = constant:civilian_transfer_route_result.valid }
	}
	set_temp_variable = { migration_decision_reception_receipt_cohort_id_request = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_decision_reception_receipt_generation_request = constant:humanitarian_runtime.zero }
}
