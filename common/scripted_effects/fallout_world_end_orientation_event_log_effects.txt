# CHAOS REDUX - FALLOUT ASH-WEEK ORIENTATION EVENT LOG EFFECTS
# ============================================================================
# Dormant, receipt-authenticated event-log writer for the five Ash-week
# orientation components. It appends one Fallout memory row only for a newly
# applied result. The payload preserves component, branch, and outcome in a
# fixed numeric matrix so the Event Log can render concrete text without
# borrowing ordinary event identifiers.
#
# Scope: COUNTRY.
# Inputs: fallout_orientation_transaction_component,
# fallout_orientation_transaction_branch, fallout_orientation_frozen_outcome.
# The caller supplies fallout_orientation_history_should_record as one only
# when the result was newly applied.
# ============================================================================

fallout_orientation_record_history = {
	if = {
		limit = {
			check_variable = { fallout_orientation_history_should_record = constant:fallout_orientation_runtime.one }
			check_variable = { fallout_orientation_transaction_component > constant:fallout_event_orientation_component.none }
			check_variable = { fallout_orientation_transaction_component < constant:fallout_event_orientation_component.upper_bound }
			check_variable = { fallout_orientation_transaction_branch > constant:fallout_orientation_runtime.zero }
			check_variable = { fallout_orientation_transaction_branch < constant:fallout_orientation_runtime.upper_bound }
			check_variable = { fallout_orientation_frozen_outcome > constant:fallout_orientation_runtime.zero }
			check_variable = { fallout_orientation_frozen_outcome < constant:fallout_event_outcome_type.aborted_after_choice }
		}
		set_temp_variable = { fallout_orientation_history_payload = fallout_orientation_transaction_component }
		multiply_temp_variable = { fallout_orientation_history_payload = constant:fallout_orientation_log.component_multiplier }
		set_temp_variable = { fallout_orientation_history_branch_payload = fallout_orientation_transaction_branch }
		multiply_temp_variable = { fallout_orientation_history_branch_payload = constant:fallout_orientation_log.branch_multiplier }
		add_to_temp_variable = { fallout_orientation_history_payload = fallout_orientation_history_branch_payload }
		add_to_temp_variable = { fallout_orientation_history_payload = fallout_orientation_frozen_outcome }
		set_temp_variable = { events_log_system_event_id = constant:fallout_orientation_log.history_id }
		set_temp_variable = { events_log_system_event_type = constant:event_system_event_type.fallout_country_memory }
		set_temp_variable = { events_log_system_payload = fallout_orientation_history_payload }
		set_temp_variable = { events_log_system_actor = ROOT.id }
		set_temp_variable = { events_log_system_has_actor = constant:fallout_orientation_runtime.one }
		set_temp_variable = { events_log_system_secondary_actor = constant:fallout_orientation_runtime.zero }
		set_temp_variable = { events_log_system_has_secondary_actor = constant:fallout_orientation_runtime.zero }
		record_events_log_system_history_entry = yes
		refresh_events_log_system_history_views = yes
		set_variable = { fallout_orientation_history_generation = global.fallout_transition_generation }
		set_variable = { fallout_orientation_history_component = fallout_orientation_transaction_component }
		clear_temp_variable = fallout_orientation_history_payload
		clear_temp_variable = fallout_orientation_history_branch_payload
		clear_temp_variable = events_log_system_event_id
		clear_temp_variable = events_log_system_event_type
		clear_temp_variable = events_log_system_payload
		clear_temp_variable = events_log_system_actor
		clear_temp_variable = events_log_system_has_actor
		clear_temp_variable = events_log_system_secondary_actor
		clear_temp_variable = events_log_system_has_secondary_actor
	}
	clear_temp_variable = fallout_orientation_history_should_record
}
