# Civil and colonial administration callbacks.
# Cohorts describe finite membership already present in host-state civilians.
# The foundation alone owns physical population receipts, support and output.

camp_administration_civil_initialize = {
	set_variable = { camp_admin_civil_policy_support_cost = constant:camp_admin_civil.policy_support }
	set_variable = { camp_admin_civil_policy_pp_cost = constant:camp_admin_civil.policy_pp }
	set_variable = { camp_admin_civil_review_support_cost = constant:camp_admin_civil.review_support }
	set_variable = { camp_admin_civil_review_pp_cost = constant:camp_admin_civil.review_pp }
	set_variable = { camp_admin_civil_claims_support_cost = constant:camp_admin_civil.claims_support }
	set_variable = { camp_admin_civil_claims_pp_cost = constant:camp_admin_civil.claims_pp }
	set_variable = { camp_admin_civil_review_months = constant:camp_admin_civil.review_months }
	set_variable = { camp_admin_civil_claims_months = constant:camp_admin_civil.claims_months }
	if = {
		limit = { camp_administration_civil_country_supported = yes NOT = { has_country_flag = camp_admin_civil_initialized } }
		set_country_flag = camp_admin_civil_initialized
		set_variable = { camp_admin_civil_admitted_k = constant:camp_admin_civil.zero }
		set_variable = { camp_admin_civil_review_progress = constant:camp_admin_civil.zero }
		set_variable = { camp_admin_civil_claims_progress = constant:camp_admin_civil.zero }
		set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.zero }
		if = { limit = { original_tag = ENG } set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.uk_intake_ceiling_k } }
		else_if = { limit = { original_tag = RAJ } set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.raj_intake_ceiling_k } }
		else_if = { limit = { original_tag = USA } set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.usa_intake_ceiling_k } }
		else_if = { limit = { OR = { original_tag = FRA tag = VIC } } set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.france_intake_ceiling_k } }
		else_if = { limit = { original_tag = ITA } set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.italy_intake_ceiling_k } set_country_flag = camp_admin_civil_italian_colonial_legacy }
		else_if = { limit = { OR = { original_tag = BEL original_tag = COG } } set_variable = { camp_admin_civil_intake_ceiling_k = constant:camp_admin_civil.congo_intake_ceiling_k } }
		for_each_scope_loop = { array = camp_active_site_states camp_administration_civil_authorize_subject_site = yes }
		camp_administration_civil_migrate_projects = yes
	}
}

# Narrow pre-existing colonial legal authority, not a generic overlord privilege.
camp_administration_civil_authorize_subject_site = {
	if = {
		limit = {
			has_variable = genocide_responsible_country
			OR = {
				AND = { var:genocide_responsible_country = { original_tag = BEL } is_bel_congo_concession_pool_state = yes CONTROLLER = { original_tag = COG is_subject_of = var:PREV.genocide_responsible_country } }
				AND = { var:genocide_responsible_country = { original_tag = ENG } is_uk_raj_detention_pool_state = yes CONTROLLER = { original_tag = RAJ is_subject_of = var:PREV.genocide_responsible_country } }
			}
		}
		set_state_flag = camp_admin_subject_authority
	}
}

# Call only within an explicit historically eligible host state.
# A reopened or captured location retains its exhausted source-membership budget.
camp_administration_civil_admit_host_cohort = {
	if = {
		limit = {
			camp_administration_civil_host_state = yes
			camp_admin_site_is_operational = yes
			ROOT = { camp_administration_civil_custody_authority = yes has_country_flag = camp_admin_civil_policy_authorized }
		}
		if = {
			limit = { NOT = { has_state_flag = camp_admin_civil_host_membership_declared } }
			set_state_flag = camp_admin_civil_host_membership_declared
			set_variable = { camp_admin_civil_host_remaining_k = constant:camp_admin_civil.local_membership_k }
		}
		camp_administration_civil_classify_host = yes
		set_temp_variable = { camp_admin_civil_intake_room_k = ROOT.camp_admin_civil_intake_ceiling_k }
		subtract_from_temp_variable = { camp_admin_civil_intake_room_k = ROOT.camp_admin_civil_admitted_k }
		clamp_temp_variable = { var = camp_admin_civil_intake_room_k min = constant:camp_admin_civil.zero max = constant:camp_admin_civil.monthly_intake_k }
		set_temp_variable = { camp_admin_admission_requested_k = camp_admin_civil_intake_room_k }
		set_temp_variable = { camp_admin_admission_source_k = camp_admin_civil_host_remaining_k }
		set_temp_variable = { camp_admin_admission_proven = constant:camp_admin_civil.one }
		camp_admin_admit_detainees = yes
		subtract_from_variable = { camp_admin_civil_host_remaining_k = camp_admin_admission_accepted_k }
		ROOT = { add_to_variable = { camp_admin_civil_admitted_k = camp_admin_admission_accepted_k } }
	}
}

# State roles distinguish civilian internment, political detention and coerced labor.
# Evidence labels persist; control loss never changes the original authority pointer.
camp_administration_civil_classify_host = {
	if = { limit = { ROOT = { original_tag = ENG } } set_state_flag = camp_admin_civil_british_domestic set_variable = { camp_admin_role = constant:camp_admin_civil.role_custody } }
	else_if = { limit = { ROOT = { original_tag = USA } } set_state_flag = camp_admin_civil_american_incarceration set_variable = { camp_admin_role = constant:camp_admin_civil.role_custody } set_variable = { camp_admin_site_output_scale = constant:camp_admin_civil.usa_works_limit } }
	else_if = {
		limit = { ROOT = { original_tag = RAJ } }
		set_state_flag = camp_admin_civil_raj_political_detention
		set_variable = { camp_admin_role = constant:camp_admin_civil.role_labor }
	}
	else_if = {
		limit = { ROOT = { OR = { original_tag = FRA tag = VIC } } }
		if = { limit = { is_france_north_africa_labor_pool_state = yes ROOT = { camp_administration_civil_french_collaboration = yes } } set_state_flag = camp_admin_civil_north_africa_labor set_variable = { camp_admin_role = constant:camp_admin_civil.role_labor } }
		else = { set_state_flag = camp_admin_civil_french_refugee_internment set_variable = { camp_admin_role = constant:camp_admin_civil.role_custody } }
		if = { limit = { ROOT = { camp_administration_civil_french_collaboration = yes } } set_state_flag = camp_admin_civil_vichy_responsibility }
	}
	else_if = { limit = { ROOT = { original_tag = ITA } } set_state_flag = camp_admin_civil_italian_wartime_site set_variable = { camp_admin_role = constant:camp_admin_civil.role_labor } }
	else_if = { limit = { ROOT = { OR = { original_tag = BEL original_tag = COG } } } set_state_flag = camp_admin_civil_congo_concession set_variable = { camp_admin_role = constant:camp_admin_civil.role_labor } }
}

camp_administration_civil_monthly = {
	camp_administration_civil_initialize = yes
	camp_admin_prepare_month_key = yes

		limit = {
			tag = ROOT
			camp_administration_civil_country_supported = yes
			NOT = { check_variable = { camp_admin_civil_last_month = camp_admin_month_key } }
		}
		set_variable = { camp_admin_civil_last_month = camp_admin_month_key }
		if = {

			}
		}
		if = {
			limit = { has_country_flag = camp_admin_civil_review_active }

			if = { limit = { check_variable = { var = camp_admin_civil_review_progress value = constant:camp_admin_civil.review_months compare = greater_than_or_equals } } camp_administration_civil_finish_review = yes }
		}
		if = {
			limit = { has_country_flag = camp_admin_civil_claims_active }
			if = {
				limit = { original_tag = USA has_government = democratic date > 1948.7.1 has_country_flag = camp_rework_usa_authority_terminated }

					limit = { check_variable = { var = camp_admin_civil_claims_progress value = constant:camp_admin_civil.claims_months compare = greater_than_or_equals } }

					set_country_flag = camp_rework_usa_redress_complete
					set_variable = { redress_pressure = constant:camp_admin_civil.zero }
					set_variable = { relocation_population_disruption = constant:camp_admin_civil.zero }
				}
			}
			else = { clr_country_flag = camp_admin_civil_claims_active clr_country_flag = camp_admin_institution_slot_active set_variable = { camp_admin_civil_claims_progress = constant:camp_admin_civil.zero } }

	}
	else_if = {
		limit = { OR = { original_tag = FRA tag = VIC } }
		set_variable = { north_africa_labor_burden = constant:camp_admin_civil.zero }
		set_variable = { camp_admin_civil_refugee_burden = constant:camp_admin_civil.zero }
		for_each_scope_loop = {
			array = camp_active_site_states
			if = {
				limit = { camp_admin_site_is_operational = yes }
				set_temp_variable = { camp_admin_civil_local_burden = camp_admin_detainees_k }
				multiply_temp_variable = { camp_admin_civil_local_burden = constant:camp_admin_civil.pressure_per_thousand }
				if = { limit = { has_state_flag = camp_admin_civil_north_africa_labor } ROOT = { add_to_variable = { north_africa_labor_burden = camp_admin_civil_local_burden } } }
				else = { ROOT = { add_to_variable = { camp_admin_civil_refugee_burden = camp_admin_civil_local_burden } } }
			}
		}
		if = { limit = { check_variable = { camp_admin_civil_refugee_burden > refugee_pressure } } set_variable = { refugee_pressure = camp_admin_civil_refugee_burden } }
		if = { limit = { has_country_flag = camp_admin_civil_review_complete } subtract_from_variable = { refugee_pressure = constant:camp_admin_civil.monthly_relief } clamp_variable = { var = refugee_pressure min = constant:camp_admin_civil.zero } }
	}
	else_if = {
		limit = { original_tag = ITA }
		set_variable = { desert_camp_burden = constant:camp_admin_civil.zero }
		for_each_scope_loop = {
			array = camp_active_site_states
			if = {
				limit = { camp_admin_site_is_operational = yes is_core_of = LBA }
				set_temp_variable = { camp_admin_civil_local_burden = camp_admin_detainees_k }
				multiply_temp_variable = { camp_admin_civil_local_burden = constant:camp_admin_civil.pressure_per_thousand }
				ROOT = { add_to_variable = { desert_camp_burden = camp_admin_civil_local_burden } }
			}
		}
	}
	else_if = {
		limit = { OR = { original_tag = BEL original_tag = COG } }
		set_variable = { concession_labor_burden = camp_admin_civil_current_burden }
		if = { limit = { has_country_flag = camp_admin_civil_concession_transport NOT = { has_country_flag = camp_admin_civil_review_active } NOT = { has_country_flag = camp_admin_civil_review_complete } } add_to_variable = { colonial_unrest_pressure = camp_admin_civil_current_burden } }
		clamp_variable = { var = colonial_unrest_pressure min = constant:camp_admin_civil.zero max = constant:camp_admin_civil.pressure_ceiling }
		if = { limit = { check_variable = { colonial_unrest_pressure > constant:camp_admin_civil.strike_pressure } } set_country_flag = camp_admin_civil_strike_active }
		if = { limit = { has_country_flag = camp_admin_civil_review_complete } clr_country_flag = camp_admin_civil_strike_active set_variable = { colonial_unrest_pressure = constant:camp_admin_civil.zero } }
	}
}

# At most one additional host per funded month, using the country's controlled pool.
# Existing historical cohort ceilings, not unrest, determine remaining membership.
camp_administration_civil_open_next_host = {
	set_temp_variable = { camp_admin_civil_open_membership_k = constant:camp_admin_civil.zero }
	for_each_scope_loop = {
		array = camp_active_site_states
		if = {
			limit = { camp_admin_site_is_operational = yes camp_administration_civil_host_state = yes }
			if = { limit = { has_state_flag = camp_admin_civil_host_membership_declared } add_to_temp_variable = { camp_admin_civil_open_membership_k = camp_admin_civil_host_remaining_k } }
			else = { add_to_temp_variable = { camp_admin_civil_open_membership_k = constant:camp_admin_civil.local_membership_k } }
		}
	}
	set_temp_variable = { camp_admin_civil_unassigned_k = camp_admin_civil_intake_ceiling_k }
	subtract_from_temp_variable = { camp_admin_civil_unassigned_k = camp_admin_civil_admitted_k }
	if = {
		limit = { check_variable = { camp_admin_civil_open_membership_k < camp_admin_civil_unassigned_k } }
		clear_variable = camp_admin_civil_next_host_id
		every_controlled_state = {
			limit = { camp_administration_civil_host_state = yes NOT = { is_valid_camp_active_site_state = yes } NOT = { has_state_flag = camp_admin_civil_host_membership_declared } camp_rework_state_can_accept_responsibility_from_root = yes }
			if = { limit = { ROOT = { NOT = { has_variable = camp_admin_civil_next_host_id } } } ROOT = { set_variable = { camp_admin_civil_next_host_id = PREV } } }
		}
		every_subject_country = {
			every_controlled_state = {
				limit = { camp_administration_civil_host_state = yes NOT = { is_valid_camp_active_site_state = yes } NOT = { has_state_flag = camp_admin_civil_host_membership_declared } camp_rework_state_can_accept_responsibility_from_root = yes }
				if = { limit = { ROOT = { NOT = { has_variable = camp_admin_civil_next_host_id } } } ROOT = { set_variable = { camp_admin_civil_next_host_id = PREV } } }
			}
		}
		if = {
			limit = { has_variable = camp_admin_civil_next_host_id }
			genocide_activate_crisis = yes
			var:camp_admin_civil_next_host_id = { camp_rework_colonial_register_current_state = yes }
		}
		clear_variable = camp_admin_civil_next_host_id
	}
}
