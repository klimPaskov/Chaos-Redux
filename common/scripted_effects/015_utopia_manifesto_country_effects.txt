# ============================================================================
# CHAOS REDUX - EVENT 015 UTOPIA MANIFESTO COUNTRY PACKAGE EFFECTS

# decision, and event call sites use these effects so a concept has exactly one
# current form and failures remain correctable instead of stacking indefinitely.

utopia_manifesto_apply_route_property_settlement = {
	utopia_manifesto_clear_inherited_order_idea_stages = yes
	if = {
		limit = {
			NOT = {
				OR = {
					has_country_flag = utopia_manifesto_garden_district_stage_founding
					has_country_flag = utopia_manifesto_garden_district_stage_mitigated
					has_country_flag = utopia_manifesto_garden_district_stage_failure
					has_country_flag = utopia_manifesto_garden_district_stage_final
				}
			}
		}
		if = {
			limit = { has_country_flag = utopia_manifesto_route_consent_of_households }
			add_ideas = utopia_manifesto_household_land_trusts
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_common_table }
			add_ideas = utopia_manifesto_common_ownership
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_guardians_of_measure }
			add_ideas = utopia_manifesto_measured_tenure
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_closed_island }
			add_ideas = utopia_manifesto_state_allotment
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_joke_understood }
			add_ideas = utopia_manifesto_mixed_property_order
		}
	}
}

utopia_manifesto_restore_route_institution_stage = {
	if = {
		limit = { has_country_flag = utopia_manifesto_route_institution_stage_final }
		utopia_manifesto_apply_route_institution_final = yes
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_institution_stage_failure }
		utopia_manifesto_apply_route_institution_failure = yes
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_institution_stage_mitigated }
		utopia_manifesto_apply_route_institution_mitigation = yes
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_institution_stage_founding }
		utopia_manifesto_apply_route_institution_founding = yes
	}
}

utopia_manifesto_restore_garden_district_stage = {
	if = {
		limit = { has_country_flag = utopia_manifesto_garden_district_stage_final }
		utopia_manifesto_complete_garden_district_network = yes
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_garden_district_stage_failure }
		utopia_manifesto_break_garden_district_network = yes
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_garden_district_stage_mitigated }
		utopia_manifesto_link_garden_district_network = yes
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_garden_district_stage_founding }
		utopia_manifesto_found_garden_district_network = yes
	}
}
