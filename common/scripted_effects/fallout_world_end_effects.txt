# ============================================================================
# CHAOS REDUX - FALLOUT WORLD END EFFECTS
# ============================================================================
# One idempotent request coordinator owns Fallout entry, blackout sequencing,
# save reconciliation, deterministic state grading, and the first rewrite pass.
# Public callers provide temporary request source and intensity variables.
# ============================================================================

# Timed flag duration fields do not accept script-constant tokens reliably.
# Keep this file-scoped literal aligned with
# constant:fallout_transition_timing.recovery_watch_days.
@FALLOUT_TRANSITION_RECOVERY_WATCH_DAYS = 1

# ---------------------------------------------------------------------------
# Request intake and validation
# ---------------------------------------------------------------------------

# Supported scope: COUNTRY
# Inputs:
# - fallout_request_source_input
# - fallout_request_intensity_input
# - optional COUNTRY event target fallout_request_actor_input
# - optional event target fallout_request_state_input
fallout_request_aftermath = {
	set_temp_variable = { fallout_request_dispatch_source = fallout_request_source_input }
	set_temp_variable = { fallout_request_dispatch_intensity = fallout_request_intensity_input }
	if = {
		limit = { NOT = { check_variable = { fallout_request_dispatch_source > constant:fallout_request_source.none } } }
		set_temp_variable = { fallout_request_dispatch_source = constant:fallout_request_source.gradual_air_collapse }
	}
	if = {
		limit = { NOT = { check_variable = { fallout_request_dispatch_intensity > constant:fallout_runtime.zero } } }
		set_temp_variable = { fallout_request_dispatch_intensity = constant:fallout_request_intensity.standard }
	}

	if = {
		limit = { any_country = { has_country_flag = is_global_host } }
		every_country = {
			limit = {
				has_country_flag = is_global_host
				fallout_request_ledger_is_free = yes
			}
			set_variable = { global.fallout_request_source = fallout_request_dispatch_source }
			set_variable = { global.fallout_request_intensity = fallout_request_dispatch_intensity }
			fallout_accept_request_envelope = yes
		}
	}
	else_if = {
		limit = {
			exists = yes
			fallout_request_ledger_is_free = yes
		}
		set_variable = { global.fallout_request_source = fallout_request_dispatch_source }
		set_variable = { global.fallout_request_intensity = fallout_request_dispatch_intensity }
		fallout_queue_request_envelope = yes
	}
}

# Supported scope: COUNTRY
fallout_accept_request_envelope = {
	fallout_queue_request_envelope = yes
	save_global_event_target_as = fallout_transition_coordinator
	fallout_validate_pending_request = yes
}

# Supported scope: COUNTRY. This records one request without appointing the
# caller as host. The existing host pulse claims and validates the envelope.
fallout_queue_request_envelope = {
	clear_global_event_target = fallout_request_actor
	clear_global_event_target = fallout_request_state
	set_global_flag = fallout_request_pending
	set_variable = { global.fallout_request_date = global.date }
	if = {
		limit = { has_event_target = fallout_request_actor_input }
		event_target:fallout_request_actor_input = { save_global_event_target_as = fallout_request_actor }
	}
	if = {
		limit = { has_event_target = fallout_request_state_input }
		event_target:fallout_request_state_input = { save_global_event_target_as = fallout_request_state }
	}
}

# Supported scope: COUNTRY
fallout_validate_pending_request = {
	if = {
		limit = {
			has_global_flag = fallout_request_pending
			fallout_current_country_is_coordinator = yes
			fallout_pending_request_is_valid = yes
		}
		fallout_lock_transition = yes
	}
	else_if = {
		limit = { has_global_flag = fallout_request_pending }
		clr_global_flag = fallout_request_pending
		clear_variable = global.fallout_request_source
		clear_variable = global.fallout_request_intensity
		clear_variable = global.fallout_request_date
		clear_global_event_target = fallout_request_actor
		clear_global_event_target = fallout_request_state
		clear_global_event_target = fallout_transition_coordinator
	}
}

# Supported scope: COUNTRY
fallout_evaluate_air_contamination_request = {
	if = {
		limit = {
			NOT = { has_global_flag = world_end }
			NOT = { has_global_flag = world_end_disabled }
			world_end_fallout_scenario_enabled = yes
			check_variable = {
				var = global.air_contamination_bp
				value = constant:air_contamination_threshold_bp.irreversible_100
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { fallout_request_source_input = constant:fallout_request_source.gradual_air_collapse }
		set_temp_variable = { fallout_request_intensity_input = constant:fallout_request_intensity.terminal }
		fallout_request_aftermath = yes
	}
}

# Supported scope: COUNTRY. Dedicated system-owned settings control. The
# shared disabled-scenario array remains the single authoritative toggle ledger.
fallout_toggle_system_scenario_disabled = {
	if = {
		limit = { is_in_array = { global.disabled_world_end_scenarios = constant:world_end_scenario_id.fallout } }
		remove_from_array = { global.disabled_world_end_scenarios = constant:world_end_scenario_id.fallout }
		fallout_evaluate_air_contamination_request = yes
	}
	else = {
		add_to_array = { global.disabled_world_end_scenarios = constant:world_end_scenario_id.fallout }
	}
}

# Supported scope: COUNTRY
fallout_dispatch_blackout_audio = {
	every_country = {
		limit = { is_ai = no }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = fallout_blackout_audio_generation }
					check_variable = {
						var = fallout_blackout_audio_generation
						value = constant:fallout_world_end_schema.audio_generation
						compare = less_than
					}
				}
			}
			set_variable = { fallout_blackout_audio_generation = constant:fallout_world_end_schema.audio_generation }
			fallout_play_blackout_audio = yes
		}
	}
}

	}
}

# Supported scope: COUNTRY
fallout_play_blackout_audio = {
	if = {
		limit = {
			has_variable = settings_super_event_audio_volume
			check_variable = { settings_super_event_audio_volume = constant:settings_super_event_audio_volume.x0 }
		}
	}
	else_if = {
		limit = {
			has_variable = settings_super_event_audio_mode
			check_variable = { settings_super_event_audio_mode = constant:settings_super_event_audio_mode.sound }
		}
		meta_effect = {
			text = { scoped_sound_effect = "fallout_world_end_blackout_sound_[VOLUME_SUFFIX]" }
			VOLUME_SUFFIX = "[GetFalloutWorldEndAudioVolumeToken]"
		}
	}
	else = {
		meta_effect = {
			text = { scoped_play_song = "fallout_world_end_blackout_[VOLUME_SUFFIX]" }
			VOLUME_SUFFIX = "[GetFalloutWorldEndAudioVolumeToken]"
		}
	}
}

# Supported scope: COUNTRY
fallout_world_end_migrate_save = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.fallout_transition_schema_version }
				check_variable = {
					var = global.fallout_transition_schema_version
					value = constant:fallout_world_end_schema.version
					compare = less_than
				}
			}
		}
		if = {
			limit = {
				has_global_flag = world_end_fallout
				has_global_flag = fallout_transition_active
				NOT = { has_global_flag = fallout_active }
				NOT = { has_global_flag = fallout_transition_complete }
				NOT = { has_global_flag = fallout_transition_map_return_applied }
				has_global_flag = fallout_transition_map_return_blocked
				has_global_flag = fallout_transition_error
				check_variable = {
					var = global.fallout_transition_schema_version
					value = constant:fallout_world_end_schema.legacy_map_return_error_version
					compare = equals
				}
				check_variable = {
					var = global.fallout_transition_phase
					value = constant:fallout_transition_phase.map_return
					compare = equals
				}
				check_variable = {
					var = global.fallout_transition_error_code
					value = constant:fallout_transition_error_code.map_postcondition_failed
					compare = equals
				}
				check_variable = {
					var = global.fallout_transition_error_count
					value = constant:fallout_runtime.one
					compare = equals
				}
			}
			clr_global_flag = fallout_transition_error
			clr_global_flag = fallout_transition_schema_migration_blocked
			clr_global_flag = fallout_transition_phase_event_scheduled
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.none }
			set_variable = { global.fallout_transition_schema_version = constant:fallout_world_end_schema.version }
		}
		else_if = {
			limit = {
				has_global_flag = world_end_fallout
				NOT = { has_global_flag = fallout_transition_active }
				OR = {
					has_global_flag = fallout_active
					has_global_flag = fallout_transition_complete
				}
			}
			set_global_flag = fallout_active
			set_global_flag = fallout_transition_complete
			set_global_flag = fallout_transition_map_return_applied
			set_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.complete }
			clr_global_flag = fallout_transition_schema_migration_blocked
			set_variable = { global.fallout_transition_schema_version = constant:fallout_world_end_schema.version }
		}
		else_if = {
			limit = {
				has_global_flag = world_end_fallout
				NOT = { has_global_flag = fallout_active }
				NOT = { has_global_flag = fallout_transition_complete }
				NOT = { has_variable = global.fallout_transition_schema_version }
				NOT = { has_global_flag = fallout_transition_schema_migration_blocked }
			}
			set_global_flag = fallout_request_locked
			set_global_flag = fallout_transition_active
			set_global_flag = fallout_transition_schema_migration_blocked
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.snapshot }
			set_variable = { global.fallout_transition_dirty = constant:fallout_runtime.one }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.legacy_schema_state_unproven }
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
			if = {
				limit = { NOT = { has_variable = global.fallout_request_source } }
				set_variable = { global.fallout_request_source = constant:fallout_request_source.legacy_fallout }
			}
			if = {
				limit = { NOT = { has_variable = global.fallout_request_intensity } }
				set_variable = { global.fallout_request_intensity = constant:fallout_request_intensity.terminal }
			}
			if = {
				limit = { NOT = { has_variable = global.fallout_transition_generation } }
				set_variable = { global.fallout_transition_generation = global.date }
			}
			save_global_event_target_as = fallout_transition_coordinator
			clr_global_flag = fallout_transition_phase_event_scheduled
			fallout_mark_blackout_gui_dirty = yes
			fallout_dispatch_blackout_audio = yes
		}
		else_if = {
			limit = {
				has_global_flag = world_end_fallout
				has_global_flag = fallout_transition_schema_migration_blocked
			}
			set_global_flag = fallout_request_locked
			set_global_flag = fallout_transition_active
			save_global_event_target_as = fallout_transition_coordinator
			clr_global_flag = fallout_transition_phase_event_scheduled
			fallout_mark_blackout_gui_dirty = yes
		}
		else_if = {
			limit = {
				has_global_flag = world_end_fallout
			}
			set_global_flag = fallout_request_locked
			set_global_flag = fallout_transition_active
			set_global_flag = fallout_transition_schema_migration_blocked
			set_global_flag = fallout_transition_error
			if = {
				limit = { has_global_flag = fallout_transition_destructive_started }
				set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.schema_migration_after_destruction }
			}
			else = {
				set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.legacy_schema_state_unproven }
			}
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
			if = {
				limit = { NOT = { has_variable = global.fallout_request_source } }
				set_variable = { global.fallout_request_source = constant:fallout_request_source.legacy_fallout }
			}
			if = {
				limit = { NOT = { has_variable = global.fallout_request_intensity } }
				set_variable = { global.fallout_request_intensity = constant:fallout_request_intensity.terminal }
			}
			if = {
				limit = { NOT = { has_variable = global.fallout_transition_generation } }
				set_variable = { global.fallout_transition_generation = global.date }
			}
			save_global_event_target_as = fallout_transition_coordinator
			clr_global_flag = fallout_transition_phase_event_scheduled
			set_variable = { global.fallout_transition_dirty = constant:fallout_runtime.one }
			fallout_mark_blackout_gui_dirty = yes
			fallout_dispatch_blackout_audio = yes
		}
		else = {
			set_variable = { global.fallout_transition_schema_version = constant:fallout_world_end_schema.version }
		}
	}
}

# Supported scope: COUNTRY. Called from the existing single host daily pulse.
fallout_reconcile_coordinator = {
	if = {
		limit = { has_country_flag = is_global_host }
		fallout_manual_reconcile = yes
		fallout_world_end_migrate_save = yes
		if = {
			limit = {
				has_global_flag = fallout_request_pending
				NOT = { has_event_target = fallout_transition_coordinator }
			}
			save_global_event_target_as = fallout_transition_coordinator
		}
		if = {
			limit = {
				has_global_flag = fallout_transition_active
				NOT = { has_event_target = fallout_transition_coordinator }
			}
			save_global_event_target_as = fallout_transition_coordinator
			clr_global_flag = fallout_transition_phase_event_scheduled
		}
		else_if = {
			limit = {
				has_global_flag = fallout_transition_active
				NOT = {
					event_target:fallout_transition_coordinator = {
						exists = yes
						has_country_flag = is_global_host
					}
				}
			}
			clear_global_event_target = fallout_transition_coordinator
			save_global_event_target_as = fallout_transition_coordinator
			clr_global_flag = fallout_transition_phase_event_scheduled
		}
		if = {
			limit = { has_global_flag = fallout_transition_active }
			fallout_dispatch_blackout_audio = yes
		}
		if = {
			limit = {
				has_global_flag = fallout_transition_active
				fallout_current_country_is_coordinator = yes
				NOT = { has_global_flag = fallout_transition_schema_migration_blocked }
				NOT = { has_global_flag = fallout_transition_phase_event_scheduled }
			}
			fallout_schedule_current_transition_phase = yes
		}
		if = {
			limit = {
				has_global_flag = fallout_request_pending
				fallout_current_country_is_coordinator = yes
			}
			fallout_validate_pending_request = yes
		}
	}
}

# Supported scope: COUNTRY
fallout_schedule_current_transition_phase = {
	if = {
		limit = {
			has_global_flag = fallout_transition_active
			NOT = { has_global_flag = fallout_transition_schema_migration_blocked }
		}
		set_temp_variable = { fallout_transition_event_delay_hours = constant:fallout_transition_timing.beat_hours }
		set_global_flag = {
			flag = fallout_transition_phase_event_scheduled
			days = @FALLOUT_TRANSITION_RECOVERY_WATCH_DAYS
		}
		if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.snapshot } }
			country_event = { id = chaosx.fallout.1001 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.state_grading } }
			country_event = { id = chaosx.fallout.1002 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.population_loss } }
			country_event = { id = chaosx.fallout.1003 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.physical_collapse } }
			country_event = { id = chaosx.fallout.1004 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.government_sorting } }
			country_event = { id = chaosx.fallout.1005 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.survivor_allocation } }
			country_event = { id = chaosx.fallout.1006 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.player_continuation } }
			country_event = { id = chaosx.fallout.1007 hours = fallout_transition_event_delay_hours }
		}
		else_if = {
			limit = { check_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.map_return } }
			country_event = { id = chaosx.fallout.1008 hours = fallout_transition_event_delay_hours }
		}
		else = {
			clr_global_flag = fallout_transition_phase_event_scheduled
		}
	}
}

		else_if = { limit = { has_state_category = metropolis } add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.metropolis } }
		else_if = { limit = { has_state_category = megalopolis } add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.megalopolis } }
		else_if = { limit = { OR = { has_state_category = enclave has_state_category = tiny_island has_state_category = small_island } } add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.remote_category_relief } }

		set_temp_variable = { fallout_grade_factory_count = building_level@industrial_complex }
		add_to_temp_variable = { fallout_grade_factory_count = building_level@arms_factory }
		if = {
			limit = {
				check_variable = {
					var = fallout_grade_factory_count
					value = constant:fallout_grade_trigger.industrial_center_factories
					compare = greater_than
				}
			}
			add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.industrial_center }
		}
		if = {
			limit = {
				check_variable = {
					var = fallout_grade_factory_count
					value = constant:fallout_grade_trigger.major_industrial_center_factories
					compare = greater_than
				}
			}
			add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.major_industrial_center }
		}
		if = {
			limit = {
				has_variable = air_winter_adaptation
				check_variable = { air_winter_adaptation > constant:fallout_grade_trigger.high_adaptation }
			}
			add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.high_adaptation_relief }
		}
		if = {
			limit = {
				has_variable = air_winter_shelter_capacity
				check_variable = { air_winter_shelter_capacity > constant:fallout_grade_trigger.high_shelter }
			}
			add_to_variable = { fallout_grade_score = constant:fallout_grade_weight.high_shelter_relief }
		}

		clamp_variable = {
			var = fallout_grade_score
			min = constant:fallout_runtime.zero
			max = constant:fallout_grade_threshold.maximum_score
		}

		set_variable = { fallout_state_grade = constant:fallout_state_grade.remote_refuge }
		if = {
			limit = { check_variable = { var = fallout_grade_score value = constant:fallout_grade_threshold.vitrified_zone compare = greater_than_or_equals } }
			set_variable = { fallout_state_grade = constant:fallout_state_grade.vitrified_zone }
		}
		else_if = {
			limit = { check_variable = { var = fallout_grade_score value = constant:fallout_grade_threshold.wasteland compare = greater_than_or_equals } }
			set_variable = { fallout_state_grade = constant:fallout_state_grade.wasteland }
		}
		else_if = {
			limit = { check_variable = { var = fallout_grade_score value = constant:fallout_grade_threshold.dead_city compare = greater_than_or_equals } }
			set_variable = { fallout_state_grade = constant:fallout_state_grade.dead_city }
		}
		else_if = {
			limit = { check_variable = { var = fallout_grade_score value = constant:fallout_grade_threshold.ash_zone compare = greater_than_or_equals } }
			set_variable = { fallout_state_grade = constant:fallout_state_grade.ash_zone }
		}
		else_if = {
			limit = { check_variable = { var = fallout_grade_score value = constant:fallout_grade_threshold.scarred_province compare = greater_than_or_equals } }
			set_variable = { fallout_state_grade = constant:fallout_state_grade.scarred_province }
		}

		set_variable = { fallout_state_subtype = constant:fallout_state_subtype.none }
		if = {
			limit = {
				OR = {
					has_state_flag = fallout_legacy_altered_grade_memory
					AND = {
						OR = {
							ROOT = { fallout_request_source_is_biological = yes }
							ROOT = { fallout_request_source_is_mixed = yes }
						}
						check_variable = {
							var = fallout_grade_score
							value = constant:fallout_grade_threshold.altered_biosphere
							compare = greater_than_or_equals
						}
					}
				}
			}
			set_variable = { fallout_state_subtype = constant:fallout_state_subtype.altered_biosphere }
			set_state_flag = fallout_altered_biosphere
			clr_state_flag = fallout_legacy_altered_grade_memory
		}
		fallout_calculate_state_survival_value = yes
		set_state_flag = fallout_state_graded
		set_variable = { fallout_state_grade_date = global.date }
	}
}

# Supported scope: STATE. This 0 to 100 ledger is intentionally independent
# from the damage grade and is used by continuation and successor selection.
fallout_calculate_state_survival_value = {
	set_variable = { fallout_survival_value = constant:fallout_survival_value.maximum }
	set_temp_variable = { fallout_survival_grade_penalty = fallout_state_grade }
	multiply_temp_variable = { fallout_survival_grade_penalty = constant:fallout_survival_value.grade_penalty }
	subtract_from_variable = { fallout_survival_value = fallout_survival_grade_penalty }

	if = { limit = { fallout_state_is_direct_manual_target = yes } subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.manual_strike_penalty } }
	if = { limit = { fallout_state_has_live_nuclear_fallout = yes } subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.nuclear_fallout_penalty } }
	if = { limit = { has_dynamic_modifier = { modifier = chem_state_contamination } } subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.chemical_penalty } }
	if = {
		limit = {
			OR = {
				has_state_flag = bioweapon_contaminated
				has_state_flag = bio_agent_1_active
				has_state_flag = bio_agent_2_active
				has_state_flag = bio_agent_3_active
				has_state_flag = bio_agent_4_active
			}
		}
		subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.biological_penalty }
	}
	if = {
		limit = { has_variable = air_winter_exposure }
		set_temp_variable = { fallout_survival_winter_penalty = air_winter_exposure }
		multiply_temp_variable = { fallout_survival_winter_penalty = constant:fallout_survival_value.winter_exposure_factor }
		subtract_from_variable = { fallout_survival_value = fallout_survival_winter_penalty }
	}
	if = {
		limit = { has_variable = air_winter_adaptation }
		set_temp_variable = { fallout_survival_adaptation_bonus = air_winter_adaptation }
		multiply_temp_variable = { fallout_survival_adaptation_bonus = constant:fallout_survival_value.adaptation_factor }
		add_to_variable = { fallout_survival_value = fallout_survival_adaptation_bonus }
	}
	if = {
		limit = { has_variable = air_winter_shelter_capacity }
		set_temp_variable = { fallout_survival_shelter_bonus = air_winter_shelter_capacity }
		multiply_temp_variable = { fallout_survival_shelter_bonus = constant:fallout_survival_value.shelter_factor }
		add_to_variable = { fallout_survival_value = fallout_survival_shelter_bonus }
	}
	if = {
		limit = { has_variable = air_winter_food_reserve }
		set_temp_variable = { fallout_survival_food_bonus = air_winter_food_reserve }
		multiply_temp_variable = { fallout_survival_food_bonus = constant:fallout_survival_value.food_factor }
		add_to_variable = { fallout_survival_value = fallout_survival_food_bonus }
	}
	if = {
		limit = { has_variable = air_winter_reclamation }
		set_temp_variable = { fallout_survival_reclamation_bonus = air_winter_reclamation }
		multiply_temp_variable = { fallout_survival_reclamation_bonus = constant:fallout_survival_value.reclamation_factor }
		add_to_variable = { fallout_survival_value = fallout_survival_reclamation_bonus }
	}
	if = { limit = { is_coastal = yes } add_to_variable = { fallout_survival_value = constant:fallout_survival_value.coastal_bonus } }
	if = {
		limit = { check_variable = { building_level@supply_node > constant:fallout_runtime.zero } }
		add_to_variable = { fallout_survival_value = constant:fallout_survival_value.supply_node_bonus }
	}
	set_temp_variable = { fallout_survival_infrastructure_bonus = building_level@infrastructure }
	multiply_temp_variable = { fallout_survival_infrastructure_bonus = constant:fallout_survival_value.infrastructure_per_level }
	add_to_variable = { fallout_survival_value = fallout_survival_infrastructure_bonus }
	set_temp_variable = { fallout_survival_railway_bonus = building_level@rail_way }
	multiply_temp_variable = { fallout_survival_railway_bonus = constant:fallout_survival_value.railway_per_level }
	add_to_variable = { fallout_survival_value = fallout_survival_railway_bonus }

	set_temp_variable = { fallout_survival_damaged_factory_penalty = damaged_building_level@industrial_complex }
	add_to_temp_variable = { fallout_survival_damaged_factory_penalty = damaged_building_level@arms_factory }
	multiply_temp_variable = { fallout_survival_damaged_factory_penalty = constant:fallout_survival_value.damaged_factory_penalty }
	subtract_from_variable = { fallout_survival_value = fallout_survival_damaged_factory_penalty }
	if = {
		limit = { check_variable = { state_population_k > constant:fallout_survival_value.very_high_population_threshold_k } }
		subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.very_high_population_penalty }
	}
	else_if = {
		limit = { check_variable = { state_population_k > constant:fallout_survival_value.high_population_threshold_k } }
		subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.high_population_penalty }
	}
	if = {
		limit = { CONTROLLER = { has_war = yes } }
		subtract_from_variable = { fallout_survival_value = constant:fallout_survival_value.active_war_penalty }
	}
	clamp_variable = {
		var = fallout_survival_value
		min = constant:fallout_survival_value.minimum
		max = constant:fallout_survival_value.maximum
	}
	round_variable = fallout_survival_value
	set_variable = { fallout_survival_value_date = global.date }
}

# Supported scope: COUNTRY. The former capital wins when it is still hostable.
# Otherwise the highest survival value wins with the lowest state id as the
# deterministic final tie breaker.
fallout_select_player_primary_capital = {
	set_temp_variable = { fallout_player_capital_found = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_best_survival = constant:fallout_runtime.negative_one }
	set_temp_variable = { fallout_player_best_state_id = constant:fallout_snapshot_limit.maximum_database_id }
	if = {
		limit = { has_variable = fallout_pretransition_capital_state }
		var:fallout_pretransition_capital_state = {
			if = {
				limit = {
					is_owned_by = PREV
					fallout_state_can_host_government = yes
				}
				set_temp_variable = { fallout_player_primary_capital_scope = THIS }
				set_temp_variable = { fallout_player_capital_found = constant:fallout_runtime.one }
			}
		}
	}
	if = {
		limit = { check_variable = { fallout_player_capital_found = constant:fallout_runtime.zero } }
		every_owned_state = {
			limit = { fallout_state_can_host_government = yes }
			set_temp_variable = { fallout_player_candidate_survival = fallout_survival_value }
			set_temp_variable = { fallout_player_candidate_state_id = THIS.id }
			if = {
				limit = {
					OR = {
						check_variable = {
							var = fallout_player_candidate_survival
							value = fallout_player_best_survival
							compare = greater_than
						}
						AND = {
							check_variable = { fallout_player_candidate_survival = fallout_player_best_survival }
							check_variable = {
								var = fallout_player_candidate_state_id
								value = fallout_player_best_state_id
								compare = less_than
							}
						}
					}
				}
				set_temp_variable = { fallout_player_best_survival = fallout_player_candidate_survival }
				set_temp_variable = { fallout_player_best_state_id = fallout_player_candidate_state_id }
				set_temp_variable = { fallout_player_primary_capital_scope = THIS }
				set_temp_variable = { fallout_player_capital_found = constant:fallout_runtime.one }
			}
		}
	}
	if = {
		limit = { check_variable = { fallout_player_capital_found = constant:fallout_runtime.one } }
		set_country_flag = fallout_player_primary_capital_reserved
		set_variable = { fallout_player_primary_capital_state = fallout_player_primary_capital_scope }
		var:fallout_player_primary_capital_scope = {
			set_state_flag = fallout_player_continuation_reserved
			set_state_flag = fallout_player_primary_capital_reserved
			set_variable = { fallout_player_reservation_origin = PREV }
			set_variable = { fallout_player_reservation_generation = global.fallout_transition_generation }
		}
	}
}

# Supported scope: COUNTRY. Pass one claims every structurally addressable
# existing and proposed target before strict validation. No player switch
# occurs unless every snapshotted human has one ready and unique target.
fallout_preflight_player_commit_targets = {
	clr_global_flag = fallout_player_commit_preflight_complete
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}
	set_temp_variable = { fallout_player_preflight_existing_count = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_preflight_ready_count = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.zero }

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = { has_country_flag = fallout_player_continuation_committed }
			if = {
				limit = {
					has_variable = fallout_player_final_target
					has_variable = fallout_player_commit_generation
					check_variable = {
						var = fallout_player_commit_generation
						value = global.fallout_transition_generation
						compare = equals
					}
					var:fallout_player_final_target = { exists = yes }
				}
				var:fallout_player_final_target = {
					if = {
						limit = { has_country_flag = fallout_player_assignment_claimed }
						set_country_flag = fallout_player_assignment_collision_target
						set_temp_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.one }
					}
					else = { set_country_flag = fallout_player_assignment_claimed }
				}
			}
			if = {
				limit = { fallout_player_continuation_commit_is_valid = yes }
				add_to_variable = { fallout_player_preflight_existing_count = constant:fallout_runtime.one }
			}
			else = {
				set_temp_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.one }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
			}
		}
	}

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = { NOT = { has_country_flag = fallout_player_continuation_committed } }
			if = {
				limit = {
					has_variable = fallout_player_primary_target
					check_variable = {
						var = fallout_player_snapshot_generation
						value = global.fallout_transition_generation
						compare = equals
					}
					var:fallout_player_primary_target = { exists = yes }
				}
				var:fallout_player_primary_target = {
					if = {
						limit = { has_country_flag = fallout_player_assignment_claimed }
						set_country_flag = fallout_player_assignment_collision_target
						set_temp_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.one }
					}
					else = { set_country_flag = fallout_player_assignment_claimed }
				}
			}
			if = {
				limit = { fallout_player_primary_target_is_commit_ready = yes }
				add_to_variable = { fallout_player_preflight_ready_count = constant:fallout_runtime.one }
			}
			else = {
				set_temp_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.one }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
			}
		}
	}
	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				OR = {
					AND = {
						has_country_flag = fallout_player_continuation_committed
						has_variable = fallout_player_final_target
						var:fallout_player_final_target = { has_country_flag = fallout_player_assignment_collision_target }
					}
					AND = {
						NOT = { has_country_flag = fallout_player_continuation_committed }
						has_variable = fallout_player_primary_target
						var:fallout_player_primary_target = { has_country_flag = fallout_player_assignment_collision_target }
					}
				}
			}
			set_country_flag = fallout_player_continuation_error
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
			set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
	}

	set_temp_variable = { fallout_player_preflight_total_count = fallout_player_preflight_existing_count }
	add_to_temp_variable = { fallout_player_preflight_total_count = fallout_player_preflight_ready_count }
	if = {
		limit = {
			check_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.zero }
			check_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.zero }
			check_variable = {
				var = fallout_player_preflight_total_count
				value = global.fallout_player_snapshot_count
				compare = equals
			}
		}
		set_global_flag = fallout_player_commit_preflight_complete
	}
	else = {
		set_global_flag = fallout_transition_error
		set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
		if = {
			limit = { check_variable = { fallout_player_preflight_collision_found > constant:fallout_runtime.zero } }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
		else = {
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
		}
	}
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}
}

		set_country_flag = fallout_player_continuation_committed
		clr_country_flag = fallout_player_continuation_error
		clear_variable = fallout_player_error_code
		set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.committed }
	}
	else = {

			clr_country_flag = fallout_player_assignment_committed
			clear_variable = fallout_player_assignment_origin
			clear_variable = fallout_player_assignment_generation
		}
		clr_country_flag = fallout_player_continuation_committed
		clear_variable = fallout_player_final_target
		clear_variable = fallout_player_commit_generation
		set_country_flag = fallout_player_continuation_error
		set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
		set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_tag_switch_failed }
		set_global_flag = fallout_transition_error
		set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
		set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_tag_switch_failed }
	}
}

# Supported scope: COUNTRY. Rebuilds committed targets from persisted player
# records and proves that no two players claim the same final country.
fallout_rebuild_player_commit_ledger = {
	clear_array = global.fallout_player_committed_targets
	set_variable = { global.fallout_player_committed_count = constant:fallout_runtime.zero }
	clr_global_flag = fallout_player_continuation_commit_complete
	clr_global_flag = fallout_player_assignment_uniqueness_verified
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}
	set_temp_variable = { fallout_player_invalid_commit_found = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_assignment_collision_found = constant:fallout_runtime.zero }

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				has_country_flag = fallout_player_continuation_committed
				has_variable = fallout_player_final_target
				has_variable = fallout_player_commit_generation
				check_variable = {
					var = fallout_player_commit_generation
					value = global.fallout_transition_generation
					compare = equals
				}
				var:fallout_player_final_target = { exists = yes }
			}
			var:fallout_player_final_target = {
				if = {
					limit = { has_country_flag = fallout_player_assignment_claimed }
					set_country_flag = fallout_player_assignment_collision_target
					set_temp_variable = { fallout_player_assignment_collision_found = constant:fallout_runtime.one }
				}
				else = { set_country_flag = fallout_player_assignment_claimed }
			}
		}
	}

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = { has_country_flag = fallout_player_continuation_committed }
			if = {
				limit = { fallout_player_continuation_commit_is_valid = yes }
				if = {
					limit = {
						has_country_flag = fallout_player_continuation_error
						check_variable = {
							var = fallout_player_error_code
							value = constant:fallout_transition_error_code.player_commit_incomplete
							compare = equals
						}
					}
					clr_country_flag = fallout_player_continuation_error
					clear_variable = fallout_player_error_code
					set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.committed }
				}
				var:fallout_player_final_target = { add_to_array = { array = global.fallout_player_committed_targets value = THIS } }
				add_to_variable = { global.fallout_player_committed_count = constant:fallout_runtime.one }
			}
			else = {
				set_temp_variable = { fallout_player_invalid_commit_found = constant:fallout_runtime.one }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
			}
		}
	}
	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				has_country_flag = fallout_player_continuation_committed
				has_variable = fallout_player_final_target
				var:fallout_player_final_target = { has_country_flag = fallout_player_assignment_collision_target }
			}
			set_country_flag = fallout_player_continuation_error
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
			set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
	}
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}

	if = {
		limit = { check_variable = { fallout_player_assignment_collision_found > constant:fallout_runtime.zero } }
		if = {
			limit = {
				NOT = { has_global_flag = fallout_transition_error }
				check_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
			}
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
	}
	else_if = {
		limit = { check_variable = { fallout_player_invalid_commit_found > constant:fallout_runtime.zero } }
		if = {
			limit = {
				NOT = { has_global_flag = fallout_transition_error }
				check_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
			}
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.fallout_player_snapshot_count
				value = global.fallout_player_committed_count
				compare = equals
			}
			all_of_scopes = {
				array = global.fallout_pretransition_player_countries
				fallout_player_continuation_commit_is_valid = yes
			}
		}
		set_global_flag = fallout_player_continuation_commit_complete
		set_global_flag = fallout_player_assignment_uniqueness_verified
	}
}

# Supported scope: COUNTRY
fallout_resolve_player_continuation_ledger = {
	if = {
		limit = { NOT = { has_global_flag = fallout_player_continuation_planning_initialized } }
		clear_array = global.fallout_player_primary_targets
		clear_array = global.fallout_player_committed_targets
		set_variable = { global.fallout_player_planned_count = constant:fallout_runtime.zero }
		set_variable = { global.fallout_player_committed_count = constant:fallout_runtime.zero }
		clr_global_flag = fallout_player_continuation_ledger_complete
		clr_global_flag = fallout_player_continuation_planning_blocked
		clr_global_flag = fallout_player_materialization_required
		every_state = {
			clr_state_flag = fallout_player_continuation_reserved
			clr_state_flag = fallout_player_primary_capital_reserved
			clear_variable = fallout_player_reservation_origin
			clear_variable = fallout_player_reservation_generation
		}
		for_each_scope_loop = {
			array = global.fallout_pretransition_player_countries
			if = {
				limit = { fallout_player_persisted_assignment_can_rebuild_reservation = yes }
				set_country_flag = fallout_player_primary_capital_reserved
				set_variable = { fallout_player_primary_target = fallout_player_final_target }
				var:fallout_player_final_target = {
					capital_scope = {
						set_state_flag = fallout_player_continuation_reserved
						set_state_flag = fallout_player_primary_capital_reserved
						set_variable = { fallout_player_reservation_origin = PREV.PREV }
						set_variable = { fallout_player_reservation_generation = global.fallout_transition_generation }
						set_variable = { PREV.PREV.fallout_player_primary_capital_state = THIS }
					}
				}
			}
			if = {
				limit = {
					has_country_flag = fallout_player_continuation_committed
					fallout_player_continuation_commit_is_valid = yes
				}
				set_country_flag = fallout_player_continuation_planned
				var:fallout_player_final_target = {
					add_to_array = { array = global.fallout_player_primary_targets value = THIS }
					add_to_array = { array = global.fallout_player_committed_targets value = THIS }
					capital_scope = {
						set_state_flag = fallout_player_continuation_reserved
						set_state_flag = fallout_player_primary_capital_reserved
						set_variable = { fallout_player_reservation_origin = PREV.PREV }
						set_variable = { fallout_player_reservation_generation = global.fallout_transition_generation }
					}
				}
				add_to_variable = { global.fallout_player_planned_count = constant:fallout_runtime.one }
				add_to_variable = { global.fallout_player_committed_count = constant:fallout_runtime.one }
			}
			else_if = {
				limit = { has_country_flag = fallout_player_continuation_committed }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
				clr_global_flag = fallout_player_continuation_commit_complete
				clr_global_flag = fallout_player_assignment_uniqueness_verified
			}
			else = {
				clr_country_flag = fallout_player_primary_capital_reserved
				clr_country_flag = fallout_player_continuation_preserved
				clr_country_flag = fallout_player_continues_old_country
				clr_country_flag = fallout_player_successor_choice_required
				clr_country_flag = fallout_player_continuation_planned
				clear_variable = fallout_player_primary_capital_state
				clear_variable = fallout_player_primary_target
				clear_variable = fallout_player_continuation_branch
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.snapshotted }
			}
		}
		set_global_flag = fallout_player_continuation_planning_initialized
	}
	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				NOT = { has_country_flag = fallout_player_continuation_planned }
				NOT = { has_country_flag = fallout_player_continuation_committed }
				fallout_country_has_survivable_state = yes
				has_country_flag = fallout_player_source_anchor_preserved
			}
			fallout_select_player_primary_capital = yes
		}
		if = {
			limit = {
				NOT = { has_country_flag = fallout_player_continuation_planned }
				NOT = { has_country_flag = fallout_player_continuation_committed }
				has_country_flag = fallout_player_primary_capital_reserved
			}
			set_country_flag = fallout_player_continuation_preserved
			set_country_flag = fallout_player_continues_old_country
			set_country_flag = fallout_player_continuation_planned
			set_variable = { fallout_player_primary_target = THIS }
			set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.surviving_tag }
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.primary_reserved }
			add_to_array = { array = global.fallout_player_primary_targets value = THIS }
			add_to_variable = { global.fallout_player_planned_count = constant:fallout_runtime.one }
		}
		else_if = {
			limit = {
				NOT = { has_country_flag = fallout_player_continuation_planned }
				NOT = { has_country_flag = fallout_player_continuation_committed }
			}
			set_country_flag = fallout_player_successor_choice_required
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.choice_pending }
			if = {
				limit = {
					any_owned_state = {
						has_variable = fallout_state_subtype
						check_variable = { fallout_state_subtype = constant:fallout_state_subtype.altered_biosphere }
					}
				}
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.altered_transformation }
			}
			else_if = {
				limit = {
					any_owned_state = {
						is_coastal = yes
						check_variable = { building_level@naval_base > constant:fallout_runtime.zero }
					}
				}
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.refuge_government }
			}
			else_if = {
				limit = { any_owned_state = { has_state_flag = fallout_player_origin_state } }
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.fragmented_tag }
			}
			else = {
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.emergency_council }
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = global.fallout_player_snapshot_count
				value = global.fallout_player_planned_count
				compare = equals
			}
		}
		set_global_flag = fallout_player_continuation_ledger_complete
		clr_global_flag = fallout_player_continuation_planning_blocked
		clr_global_flag = fallout_player_materialization_required
	}
	else = {
		set_global_flag = fallout_player_continuation_planning_blocked
		set_global_flag = fallout_player_materialization_required
	}
}

# Supported scope: COUNTRY
fallout_apply_transition_phase_map_return = {
	if = {
		limit = {
			NOT = { has_global_flag = fallout_transition_map_return_applied }
			fallout_map_return_postconditions_are_met = yes
		}
		set_global_flag = fallout_transition_map_return_applied
		clr_global_flag = fallout_transition_map_return_blocked
		set_global_flag = fallout_active
		set_global_flag = fallout_transition_complete
		set_global_flag = fallout_scheduler_initialization_pending
		set_variable = { global.fallout_cause_memory_source = global.fallout_request_source }
		set_variable = { global.fallout_cause_memory_intensity = global.fallout_request_intensity }
		set_variable = { global.fallout_cause_memory_date = global.fallout_request_date }
		if = {
			limit = { has_event_target = fallout_request_actor }
			event_target:fallout_request_actor = { save_global_event_target_as = fallout_cause_actor }
		}
		if = {
			limit = { has_event_target = fallout_request_state }
			event_target:fallout_request_state = { save_global_event_target_as = fallout_cause_state }
		}
		clear_global_event_target = fallout_request_actor
		clear_global_event_target = fallout_request_state
		clear_variable = global.fallout_request_source
		clear_variable = global.fallout_request_intensity
		clear_variable = global.fallout_request_date
		clr_global_flag = fallout_request_locked
		set_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.complete }
		add_to_variable = { global.fallout_transition_dirty = constant:fallout_runtime.one }
		fallout_mark_blackout_gui_dirty = yes
		clr_global_flag = fallout_transition_phase_event_scheduled
		clr_global_flag = fallout_transition_active
		clear_global_event_target = fallout_transition_coordinator
	}
	else = {
		set_global_flag = fallout_transition_map_return_blocked
		add_to_variable = { global.fallout_transition_dirty = constant:fallout_runtime.one }
		fallout_mark_blackout_gui_dirty = yes
	}
}
