# ============================================================================
# CHAOS REDUX - FALLOUT WORLD END EFFECTS
# ============================================================================
# One idempotent request coordinator owns Fallout entry, blackout sequencing,
# save reconciliation, generation-bound state receipts, and the rewrite pass.
# Public callers provide temporary request source and intensity variables.
# ============================================================================

# Timed flag duration fields do not accept script-constant tokens reliably.
# Keep this file-scoped literal aligned with
# constant:fallout_transition_timing.recovery_watch_days.
@FALLOUT_TRANSITION_RECOVERY_WATCH_DAYS = 1

# Request intake and validation
# ---------------------------------------------------------------------------

# Supported scope: COUNTRY
# Inputs:
# - fallout_request_source_input
# - fallout_request_intensity_input
# - optional COUNTRY event target fallout_request_actor_input
# - optional event target fallout_request_state_input
fallout_request_aftermath = {
	set_temp_variable = { fallout_request_dispatch_source = fallout_request_source_input }
	set_temp_variable = { fallout_request_dispatch_intensity = fallout_request_intensity_input }
	if = {
		limit = { NOT = { check_variable = { fallout_request_dispatch_source > constant:fallout_request_source.none } } }
		set_temp_variable = { fallout_request_dispatch_source = constant:fallout_request_source.gradual_air_collapse }
	}
	if = {
		limit = { NOT = { check_variable = { fallout_request_dispatch_intensity > constant:fallout_runtime.zero } } }
		set_temp_variable = { fallout_request_dispatch_intensity = constant:fallout_request_intensity.standard }
	}

	if = {
		limit = { any_country = { has_country_flag = is_global_host } }
		every_country = {
			limit = {
				has_country_flag = is_global_host
				fallout_request_ledger_is_free = yes
			}
			set_variable = { global.fallout_request_source = fallout_request_dispatch_source }
			set_variable = { global.fallout_request_intensity = fallout_request_dispatch_intensity }
			fallout_accept_request_envelope = yes
		}
	}
	else_if = {
		limit = {
			exists = yes
			fallout_request_ledger_is_free = yes
		}
		set_variable = { global.fallout_request_source = fallout_request_dispatch_source }
		set_variable = { global.fallout_request_intensity = fallout_request_dispatch_intensity }
		fallout_queue_request_envelope = yes
	}
}

# Supported scope: COUNTRY
fallout_accept_request_envelope = {
	fallout_queue_request_envelope = yes
	save_global_event_target_as = fallout_transition_coordinator
	fallout_validate_pending_request = yes
}

# Supported scope: COUNTRY. This records one request without appointing the
# caller as host. The existing host pulse claims and validates the envelope.
fallout_queue_request_envelope = {
	# Pause Air Cleanliness at the moment a valid request enters the shared
	# coordinator. This closes the host-reconciliation gap before the durable
	# transition lock is written. The temporary ownership flag is cleared again
	# if validation rejects the pending envelope.
	if = {
		limit = { NOT = { has_global_flag = fallout_air_cleanliness_disabled } }
		set_global_flag = fallout_air_cleanliness_disabled
		set_global_flag = fallout_air_cleanliness_request_paused
		air_contamination_apply_state_modifier = yes
	}
	clear_global_event_target = fallout_request_actor
	clear_global_event_target = fallout_request_state
	# A new request is admitted only while the ledger is free. Clear any stale
	# coordinator pointer before the host pulse claims this envelope.
	clear_global_event_target = fallout_transition_coordinator
	set_global_flag = fallout_request_pending
	set_variable = { global.fallout_request_date = global.date }
	if = {
		limit = { has_event_target = fallout_request_actor_input }
		event_target:fallout_request_actor_input = { save_global_event_target_as = fallout_request_actor }
	}
	if = {
		limit = { has_event_target = fallout_request_state_input }
		event_target:fallout_request_state_input = { save_global_event_target_as = fallout_request_state }
	}
}

# Supported scope: COUNTRY. This clears an uncommitted request without touching
# a completed cause-memory ledger or any active Fallout transition.
fallout_clear_pending_request_envelope = {
	if = {
		limit = {
			has_global_flag = fallout_air_cleanliness_request_paused
			NOT = { has_global_flag = fallout_request_locked }
			NOT = { has_global_flag = world_end_fallout }
		}
		clr_global_flag = fallout_air_cleanliness_request_paused
		clr_global_flag = fallout_air_cleanliness_disabled
		air_contamination_apply_state_modifier = yes
	}
	if = {
		limit = {
			fallout_request_source_is_strategic_singularity = yes
			has_event_target = fallout_request_actor
		}
		event_target:fallout_request_actor = {
			brilliant_scientist_recover_rejected_singularity_fallout_request = yes
		}
	}
	clr_global_flag = fallout_request_pending
	clear_variable = global.fallout_request_source
	clear_variable = global.fallout_request_intensity
	clear_variable = global.fallout_request_date
	clear_global_event_target = fallout_request_actor
	clear_global_event_target = fallout_request_state
	clear_global_event_target = fallout_transition_coordinator
}

# Supported scope: COUNTRY
fallout_validate_pending_request = {
	if = {
		limit = {
			has_global_flag = fallout_request_pending
			fallout_current_country_is_coordinator = yes
			fallout_pending_request_is_valid = yes
		}
		fallout_lock_transition = yes
	}
	else_if = {
		limit = { has_global_flag = fallout_request_pending }
		fallout_clear_pending_request_envelope = yes
	}
}

# Supported scope: COUNTRY
fallout_evaluate_air_contamination_request = {
	if = {
		limit = {
			NOT = { has_global_flag = world_end }
			NOT = { has_global_flag = world_end_disabled }
			NOT = { has_global_flag = fallout_air_cleanliness_disabled }
			world_end_fallout_scenario_enabled = yes
			check_variable = {
				var = global.air_contamination_bp
				value = constant:air_contamination_threshold_bp.irreversible_100
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { fallout_request_source_input = constant:fallout_request_source.gradual_air_collapse }
		set_temp_variable = { fallout_request_intensity_input = constant:fallout_request_intensity.terminal }
		fallout_request_aftermath = yes
	}
}

# Supported scope: COUNTRY. Dedicated system-owned settings control. The
# shared disabled-scenario array remains the single authoritative toggle ledger.
fallout_toggle_system_scenario_disabled = {
	if = {
		limit = { is_in_array = { global.disabled_world_end_scenarios = constant:fallout_consequence_id.settings_ledger } }
		remove_from_array = { global.disabled_world_end_scenarios = constant:fallout_consequence_id.settings_ledger }
		fallout_evaluate_air_contamination_request = yes
	}
	else = {
		add_to_array = { global.disabled_world_end_scenarios = constant:fallout_consequence_id.settings_ledger }
	}
}

# Supported scope: COUNTRY
fallout_lock_transition = {
	set_variable = { global.fallout_transition_generation = global.date }
	set_variable = { global.fallout_transition_phase = constant:fallout_transition_phase.snapshot }
	set_variable = { global.fallout_transition_schema_version = constant:fallout_world_end_schema.version }
	clear_variable = global.fallout_supply_network_collapse_schema_version
	clear_variable = global.fallout_supply_network_collapse_generation
	clr_global_flag = fallout_transition_legacy_completed_without_air_winter_provenance
	clr_global_flag = fallout_transition_legacy_completed_without_current_schema_receipts
	clr_global_flag = fallout_transition_legacy_completed_without_supply_network_receipts
	clr_global_flag = fallout_transition_legacy_completed_without_survival_inputs
	clr_global_flag = fallout_snapshot_prelock_capture_authorized
	fallout_clear_recoverable_player_snapshot_error = yes
	fallout_clear_recoverable_snapshot_error = yes
	if = {
		limit = { NOT = { has_global_flag = fallout_transition_error } }
		set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
		set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.none }
	}
	set_global_flag = fallout_snapshot_prelock_capture_authorized
	fallout_rebuild_complete_snapshot_epoch = yes
	clr_global_flag = fallout_snapshot_prelock_capture_authorized
	if = {
		limit = { fallout_snapshot_epoch_is_ready_to_lock = yes }
		clr_global_flag = fallout_request_pending
		set_global_flag = fallout_request_locked
		set_global_flag = fallout_transition_active
		set_global_flag = fallout_air_cleanliness_disabled
		clr_global_flag = fallout_air_cleanliness_request_paused
		# Remove already-applied Air Cleanliness pressure immediately. The
		# monthly pass is paused by the same durable flag, so waiting for expiry
		# would leave stale modifiers during the blackout transition.
		air_contamination_apply_state_modifier = yes
		# The monthly Air Winter pass pauses at this lock. Clear its country air
		# burden, state modifiers, and ordinary-map entities before blackout
		# scheduling while preserving the pretransition survival ledger.
		air_winter_suspend_all_states_for_fallout = yes
		air_cleanliness_treaty_pause_for_fallout = yes
		set_global_flag = world_end
		set_global_flag = world_end_fallout
		set_variable = { global.fallout_transition_dirty = constant:fallout_runtime.one }
		fallout_mark_blackout_gui_dirty = yes
		fallout_dispatch_blackout_audio = yes
		fallout_schedule_current_transition_phase = yes
		if = {
			limit = {
				fallout_request_source_is_strategic_singularity = yes
				has_event_target = fallout_request_actor
			}
			event_target:fallout_request_actor = {
				brilliant_scientist_finalize_singularity_after_fallout_lock = yes
			}
		}
	}
	else_if = {
		limit = { fallout_request_source_is_manual = yes }
		fallout_clear_pending_request_envelope = yes
	}
}

# Supported scope: COUNTRY
fallout_dispatch_blackout_audio = {
	every_country = {
		limit = { is_ai = no }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = fallout_blackout_audio_generation }
					check_variable = {
						var = fallout_blackout_audio_generation
						value = constant:fallout_world_end_schema.audio_generation
						compare = less_than
					}
				}
			}
			set_variable = { fallout_blackout_audio_generation = constant:fallout_world_end_schema.audio_generation }
			fallout_play_blackout_audio = yes
		}
	}
}

# Supported scope: COUNTRY
fallout_mark_blackout_gui_dirty = {
	every_country = {
		limit = { is_ai = no }
		add_to_variable = { fallout_blackout_gui_dirty = constant:fallout_runtime.one }
	}
}

	if = {
		limit = {
			has_variable = settings_super_event_audio_volume
			check_variable = { settings_super_event_audio_volume = constant:settings_super_event_audio_volume.x0 }
		}
	}
	else_if = {
		limit = {
			has_variable = settings_super_event_audio_mode
			check_variable = { settings_super_event_audio_mode = constant:settings_super_event_audio_mode.sound }
		}
		meta_effect = {
			text = { scoped_sound_effect = "fallout_world_end_blackout_sound_[VOLUME_SUFFIX]" }
			VOLUME_SUFFIX = "[GetFalloutWorldEndAudioVolumeToken]"
		}
	}
	else = {
		meta_effect = {
			text = { scoped_play_song = "fallout_world_end_blackout_[VOLUME_SUFFIX]" }
			VOLUME_SUFFIX = "[GetFalloutWorldEndAudioVolumeToken]"
		}
	}
}

}

# Supported scope: COUNTRY. Schema 10 physical rows predate the province
# supply-network receipt. Before successor allocation starts, a later active
# phase may safely return to physical collapse and derive the missing receipt.
# Unrelated owned errors and any begun allocation remain untouched.
fallout_reconcile_supply_network_schema_extension = {
	if = {
		limit = {
			has_global_flag = fallout_transition_active
			NOT = { has_global_flag = fallout_transition_error }
			NOT = { has_global_flag = fallout_successor_allocation_initialized }
			has_variable = global.fallout_transition_generation
			check_variable = {
				var = global.fallout_transition_schema_version
				value = constant:fallout_world_end_schema.version
				compare = equals
			}
			check_variable = { global.fallout_transition_phase > constant:fallout_transition_phase.physical_collapse }
			check_variable = { global.fallout_transition_phase < constant:fallout_transition_phase.player_continuation }
			OR = {
				NOT = { has_variable = global.fallout_supply_network_collapse_schema_version }
				NOT = { has_variable = global.fallout_supply_network_collapse_generation }
				NOT = {
					check_variable = {
						var = global.fallout_supply_network_collapse_schema_version
						value = constant:fallout_supply_network_collapse_schema.version
						compare = equals
					}
				}
				NOT = {
					check_variable = {
						var = global.fallout_supply_network_collapse_generation
						value = global.fallout_transition_generation
						compare = equals
					}
				}
			}
		}
		clr_global_flag = fallout_transition_phase_event_scheduled
		clr_global_flag = fallout_transition_physical_collapse_applied
		clr_global_flag = fallout_physical_collapse_complete
		clr_global_flag = fallout_transition_government_sorting_applied
		clr_global_flag = fallout_government_sorting_complete

					check_variable = { global.fallout_pretransition_request_source = constant:fallout_request_source.mixed_terminal }
				}
				check_variable = {
					var = fallout_grade_score
					value = constant:fallout_grade_threshold.altered_biosphere
					compare = greater_than_or_equals
				}
			}
			set_variable = { fallout_state_subtype = constant:fallout_state_subtype.altered_biosphere }
			set_state_flag = fallout_altered_biosphere
		}
		fallout_calculate_state_survival_value = yes
		set_state_flag = fallout_state_graded
		set_variable = { fallout_state_grade_date = global.date }

	}

# Supported scope: STATE. This never applies population loss. It reconciles the
# one issued mutation from its live baseline, then registers only the exact
# observed delta through Deaths with state population application disabled.
fallout_reconcile_population_loss_receipt = {
	clr_state_flag = fallout_population_loss_reconciled
	if = {
		limit = { fallout_manual_population_baseline_is_current = yes }
		fallout_manual_calculate_population_loss_intent = yes
	}
	else = { fallout_calculate_population_loss_intent = yes }
	set_temp_variable = { fallout_expected_population_available_from_live_baseline = fallout_population_before_loss_people }
	subtract_from_temp_variable = { fallout_expected_population_available_from_live_baseline = constant:fallout_runtime.minimum_remaining_population }
	clamp_temp_variable = { var = fallout_expected_population_available_from_live_baseline min = constant:fallout_runtime.zero }
	round_temp_variable = fallout_expected_population_available_from_live_baseline
	set_temp_variable = { fallout_expected_population_commit_from_live_baseline = fallout_expected_population_requested_loss }
	clamp_temp_variable = {
		var = fallout_expected_population_commit_from_live_baseline
		min = constant:fallout_runtime.zero
		max = fallout_expected_population_available_from_live_baseline
	}
	round_temp_variable = fallout_expected_population_commit_from_live_baseline
	set_variable = { fallout_population_after_loss_people = state_population_k }
	set_variable = { fallout_population_after_loss_k = state_population_k }
	multiply_variable = { fallout_population_after_loss_people = constant:fallout_runtime.population_scale }
	round_variable = fallout_population_after_loss_people
	set_variable = { fallout_population_before_loss_people_replay = fallout_population_before_loss_k }
	multiply_variable = { fallout_population_before_loss_people_replay = constant:fallout_runtime.population_scale }
	round_variable = fallout_population_before_loss_people_replay
	set_variable = { fallout_population_reconciled_loss = fallout_population_before_loss_people }
	subtract_from_variable = { fallout_population_reconciled_loss = fallout_population_after_loss_people }
	clamp_variable = { var = fallout_population_reconciled_loss min = constant:fallout_runtime.zero }
	round_variable = fallout_population_reconciled_loss
	set_variable = { fallout_population_reconciled_total_people = fallout_population_after_loss_people }
	add_to_variable = { fallout_population_reconciled_total_people = fallout_population_reconciled_loss }
	if = {
		limit = {
			has_state_flag = fallout_population_loss_mutation_issued
			check_variable = { var = fallout_population_loss_percent_applied value = fallout_state_loss_percent compare = equals }
			check_variable = { var = fallout_population_loss_requested_memory value = fallout_expected_population_requested_loss compare = equals }
			check_variable = { var = fallout_population_available_before_loss value = fallout_expected_population_available_from_live_baseline compare = equals }
			check_variable = { var = fallout_population_expected_loss value = fallout_expected_population_commit_from_live_baseline compare = equals }
			check_variable = { var = fallout_population_reconciled_loss value = fallout_expected_population_commit_from_live_baseline compare = equals }
		}
		if = {
			limit = { NOT = { has_state_flag = fallout_population_loss_deaths_evaluated } }
			set_variable = { fallout_population_loss_memory = fallout_population_reconciled_loss }
			if = {
				limit = { NOT = { has_variable = chaos_state_civilian_deaths_total } }
				set_variable = { chaos_state_civilian_deaths_total = constant:fallout_runtime.zero }
			}
			set_variable = { fallout_population_deaths_total_before = global.chaos_meter_deaths_total }
			set_variable = { fallout_population_deaths_sequence_before = global.chaos_meter_deaths_sequence }
			set_variable = { fallout_population_state_deaths_before = chaos_state_civilian_deaths_total }
			set_variable = { fallout_population_deaths_accounting_status = constant:fallout_deaths_accounting_status.zero_loss }
			if = {
				limit = { check_variable = { fallout_population_loss_memory > constant:fallout_runtime.zero } }
				set_variable = { fallout_population_deaths_accounting_status = constant:fallout_deaths_accounting_status.registered }
				set_temp_variable = { chaos_deaths_change = fallout_population_loss_memory }
				set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.fallout_aftermath }
				set_temp_variable = { chaos_deaths_is_civilian = constant:fallout_runtime.one }
				set_temp_variable = { chaos_deaths_apply_state_pop = constant:fallout_runtime.zero }
				set_temp_variable = { chaos_deaths_record_state_ledger = constant:fallout_runtime.one }
				set_temp_variable = { chaos_deaths_target_country = var:fallout_original_owner_scope }
				set_temp_variable = { chaos_deaths_has_target_country = constant:fallout_runtime.one }
				chaos_meter_register_deaths = yes
			}
			set_variable = { fallout_population_deaths_total_after = global.chaos_meter_deaths_total }
			set_variable = { fallout_population_deaths_total_delta = fallout_population_deaths_total_after }
			subtract_from_variable = { fallout_population_deaths_total_delta = fallout_population_deaths_total_before }
			set_variable = { fallout_population_deaths_sequence_after = global.chaos_meter_deaths_sequence }
			set_variable = { fallout_population_deaths_sequence_delta = fallout_population_deaths_sequence_after }
			subtract_from_variable = { fallout_population_deaths_sequence_delta = fallout_population_deaths_sequence_before }
			set_variable = { fallout_population_state_deaths_after = chaos_state_civilian_deaths_total }
			set_variable = { fallout_population_state_deaths_delta = fallout_population_state_deaths_after }
			subtract_from_variable = { fallout_population_state_deaths_delta = fallout_population_state_deaths_before }
			set_state_flag = fallout_population_loss_deaths_evaluated
		}
		if = {
			limit = { fallout_population_deaths_accounting_receipt_is_durable = yes }
			if = {
				limit = { check_variable = { fallout_population_loss_memory > constant:fallout_runtime.zero } }
				set_state_flag = fallout_population_loss_applied
				clr_state_flag = fallout_population_loss_zero
			}
			else = {
				clr_state_flag = fallout_population_loss_applied
				set_state_flag = fallout_population_loss_zero
			}
			set_state_flag = fallout_population_loss_evaluated
			set_state_flag = fallout_population_loss_reconciled
		}
	}
	set_temp_variable = { chaos_deaths_change = constant:fallout_runtime.zero }
	set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.unknown }
	set_temp_variable = { chaos_deaths_is_civilian = constant:fallout_runtime.zero }
	set_temp_variable = { chaos_deaths_apply_state_pop = constant:fallout_runtime.zero }
	set_temp_variable = { chaos_deaths_record_state_ledger = constant:fallout_runtime.zero }
	set_temp_variable = { chaos_deaths_target_country = OWNER }
	set_temp_variable = { chaos_deaths_has_target_country = constant:fallout_runtime.zero }
}

# Supported scope: STATE. Dead-city and higher grades remove every province
# supply node and railway through the documented all-provinces selector. The
# row intent is immutable. Each family settles only after its exact zero target
# is visible. Retrying set-to-zero after an interrupted attempt is idempotent.
fallout_apply_supply_network_collapse = {
	if = {
		limit = {
			fallout_state_grading_row_is_current = yes
			NOT = { has_state_flag = fallout_supply_network_collapse_intent_committed }
		}
		set_variable = { fallout_supply_network_before_supply_node = building_level@supply_node }
		set_variable = { fallout_supply_network_before_rail_way = building_level@rail_way }
		set_variable = { fallout_supply_network_snapshot_supply_node = fallout_pretransition_supply_node }
		set_variable = { fallout_supply_network_snapshot_rail_way = fallout_pretransition_rail_way }

		clr_state_flag = fallout_rail_way_collapse_settled
		clr_state_flag = fallout_supply_network_collapse_evaluated
		clr_state_flag = fallout_supply_network_collapse_reconciled
		clr_state_flag = fallout_supply_network_collapse_applied

		set_variable = { fallout_supply_network_requested_rail_way = constant:fallout_runtime.zero }
		if = {
			limit = {
				check_variable = {
					var = fallout_state_grade
					value = constant:fallout_state_grade.dead_city
					compare = greater_than_or_equals
				}
			}
			set_variable = { fallout_supply_network_requested_supply_node = fallout_supply_network_before_supply_node }
			set_variable = { fallout_supply_network_requested_rail_way = fallout_supply_network_before_rail_way }

		clear_variable = fallout_supply_network_after_rail_way
		clear_variable = fallout_supply_network_observed_supply_node
		clear_variable = fallout_supply_network_observed_rail_way
		set_state_flag = fallout_supply_network_collapse_intent_committed
	}
	if = {
		limit = {
			has_state_flag = fallout_supply_network_collapse_intent_committed

				var = fallout_state_grade
				value = constant:fallout_state_grade.dead_city

				var = fallout_state_grade
				value = constant:fallout_state_grade.dead_city
				compare = greater_than_or_equals
			}
			NOT = { fallout_state_rail_way_surface_is_zero = yes }
		}
		clr_state_flag = fallout_rail_way_collapse_settled
	}
	if = {
		limit = {
			has_state_flag = fallout_supply_network_collapse_intent_committed
			NOT = { has_state_flag = fallout_supply_node_collapse_settled }
		}
		if = {
			limit = {
				fallout_state_grading_row_is_current = yes
				check_variable = { fallout_state_grade < constant:fallout_state_grade.dead_city }
			}

					value = constant:fallout_state_grade.dead_city

				set_global_flag = fallout_transition_destructive_started
				set_building_level = {
					type = supply_node
					level = 0
					province = {
						all_provinces = yes
						limit_to_border = no
						level > 0
					}
				}
			}
			if = {
				limit = { fallout_state_supply_node_surface_is_zero = yes }
				set_state_flag = fallout_supply_node_collapse_settled
			}
		}
	}
	if = {
		limit = {
			has_state_flag = fallout_supply_network_collapse_intent_committed
			NOT = { has_state_flag = fallout_rail_way_collapse_settled }
		}
		if = {

			}
		}
	}
	if = {

			has_state_flag = fallout_supply_node_collapse_settled
			has_state_flag = fallout_rail_way_collapse_settled

			limit = {
				check_variable = {

					compare = equals
				}
			}

		}

				compare = greater_than_or_equals
			}

	}

	set_state_flag = fallout_category_conversion_evaluated
}

		if = {
			limit = {
				NOT = {
					any_country = {
						any_other_country = { is_lend_leasing = PREV }
					}
				}
			}
			set_global_flag = fallout_transition_lend_lease_reset_verified
			set_variable = { global.fallout_transition_lend_lease_verification_generation = global.fallout_transition_generation }
			clr_global_flag = fallout_transition_lend_lease_reset_blocked
		}
		else = {
			clr_global_flag = fallout_transition_lend_lease_reset_verified
			clear_variable = global.fallout_transition_lend_lease_verification_generation
			set_global_flag = fallout_transition_lend_lease_reset_blocked
		}

		if = {
			limit = {
				NOT = {
					any_country = {
						any_other_country = {
							received_expeditionary_forces = {
								sender = PREV
								value > constant:fallout_runtime.zero
							}
						}
					}
				}
			}
			set_global_flag = fallout_transition_expeditionaries_reset_verified
			set_variable = { global.fallout_transition_expeditionaries_verification_generation = global.fallout_transition_generation }
			clr_global_flag = fallout_transition_expeditionaries_reset_blocked
		}
		else = {
			clr_global_flag = fallout_transition_expeditionaries_reset_verified
			clear_variable = global.fallout_transition_expeditionaries_verification_generation
			set_global_flag = fallout_transition_expeditionaries_reset_blocked
		}

		set_global_flag = fallout_transition_trade_routes_reset_blocked
		set_global_flag = fallout_transition_intelligence_reset_blocked
		set_global_flag = fallout_transition_unprovable_diplomacy_surface
		set_temp_variable = { fallout_transition_error_to_register = constant:fallout_transition_error_code.unprovable_diplomacy_surface }
		fallout_register_transition_error_if_unowned = yes
	}
	set_temp_variable = { fallout_transition_error_to_register = constant:fallout_transition_error_code.none }

# Supported scope: COUNTRY. The former capital wins when it is still hostable.
# Otherwise the highest survival value wins with the lowest state id as the
# deterministic final tie breaker.
fallout_select_player_primary_capital = {
	set_temp_variable = { fallout_player_capital_found = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_best_survival = constant:fallout_runtime.negative_one }
	set_temp_variable = { fallout_player_best_state_id = constant:fallout_snapshot_limit.maximum_database_id }
	if = {
		limit = { has_variable = fallout_pretransition_capital_state }
		var:fallout_pretransition_capital_state = {
			if = {
				limit = {
					has_state_flag = fallout_player_origin_state
					has_variable = fallout_player_origin_scope
					var:fallout_player_origin_scope = { tag = PREV.PREV }
					fallout_state_can_host_government = yes
				}
				set_temp_variable = { fallout_player_primary_capital_scope = THIS }
				set_temp_variable = { fallout_player_capital_found = constant:fallout_runtime.one }
			}
		}
	}
	if = {
		limit = { check_variable = { fallout_player_capital_found = constant:fallout_runtime.zero } }
		every_state = {
			limit = {
				has_state_flag = fallout_player_origin_state
				has_variable = fallout_player_origin_scope
				has_variable = fallout_player_origin_generation
				check_variable = {
					var = fallout_player_origin_generation
					value = global.fallout_transition_generation
					compare = equals
				}
				var:fallout_player_origin_scope = { tag = PREV.PREV }
				fallout_state_can_host_government = yes
			}
			set_temp_variable = { fallout_player_candidate_survival = fallout_survival_value }
			set_temp_variable = { fallout_player_candidate_state_id = THIS.id }
			if = {
				limit = {
					OR = {
						check_variable = {
							var = fallout_player_candidate_survival
							value = fallout_player_best_survival
							compare = greater_than
						}
						AND = {
							check_variable = { fallout_player_candidate_survival = fallout_player_best_survival }
							check_variable = {
								var = fallout_player_candidate_state_id
								value = fallout_player_best_state_id
								compare = less_than
							}
						}
					}
				}
				set_temp_variable = { fallout_player_best_survival = fallout_player_candidate_survival }
				set_temp_variable = { fallout_player_best_state_id = fallout_player_candidate_state_id }
				set_temp_variable = { fallout_player_primary_capital_scope = THIS }
				set_temp_variable = { fallout_player_capital_found = constant:fallout_runtime.one }
			}
		}
	}
	if = {
		limit = { check_variable = { fallout_player_capital_found = constant:fallout_runtime.one } }
		set_country_flag = fallout_player_primary_capital_reserved
		set_variable = { fallout_player_primary_capital_state = fallout_player_primary_capital_scope }
		var:fallout_player_primary_capital_scope = {
			set_state_flag = fallout_player_continuation_reserved
			set_state_flag = fallout_player_primary_capital_reserved
			set_variable = { fallout_player_reservation_origin = PREV }
			set_variable = { fallout_player_reservation_generation = global.fallout_transition_generation }
		}
	}
}

# Supported scope: COUNTRY. Pass one claims every structurally addressable
# existing and proposed target before strict validation. No player switch
# occurs unless every snapshotted human has one ready and unique target.
fallout_preflight_player_commit_targets = {
	clr_global_flag = fallout_player_commit_preflight_complete
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}
	set_temp_variable = { fallout_player_preflight_existing_count = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_preflight_ready_count = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.zero }
	set_temp_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.zero }

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = { has_country_flag = fallout_player_continuation_committed }
			if = {
				limit = {
					has_variable = fallout_player_final_target
					has_variable = fallout_player_commit_generation
					check_variable = {
						var = fallout_player_commit_generation
						value = global.fallout_transition_generation
						compare = equals
					}
					var:fallout_player_final_target = { exists = yes }
				}
				var:fallout_player_final_target = {
					if = {
						limit = { has_country_flag = fallout_player_assignment_claimed }
						set_country_flag = fallout_player_assignment_collision_target
						set_temp_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.one }
					}
					else = { set_country_flag = fallout_player_assignment_claimed }
				}
			}
			if = {
				limit = { fallout_player_continuation_commit_is_valid = yes }
				add_to_variable = { fallout_player_preflight_existing_count = constant:fallout_runtime.one }
			}
			else = {
				set_temp_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.one }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
			}
		}
	}

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = { NOT = { has_country_flag = fallout_player_continuation_committed } }
			if = {
				limit = {
					has_variable = fallout_player_primary_target
					check_variable = {
						var = fallout_player_snapshot_generation
						value = global.fallout_transition_generation
						compare = equals
					}
					var:fallout_player_primary_target = { exists = yes }
				}
				var:fallout_player_primary_target = {
					if = {
						limit = { has_country_flag = fallout_player_assignment_claimed }
						set_country_flag = fallout_player_assignment_collision_target
						set_temp_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.one }
					}
					else = { set_country_flag = fallout_player_assignment_claimed }
				}
			}
			if = {
				limit = { fallout_player_primary_target_is_commit_ready = yes }
				add_to_variable = { fallout_player_preflight_ready_count = constant:fallout_runtime.one }
			}
			else = {
				set_temp_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.one }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
			}
		}
	}
	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				OR = {
					AND = {
						has_country_flag = fallout_player_continuation_committed
						has_variable = fallout_player_final_target
						var:fallout_player_final_target = { has_country_flag = fallout_player_assignment_collision_target }
					}
					AND = {
						NOT = { has_country_flag = fallout_player_continuation_committed }
						has_variable = fallout_player_primary_target
						var:fallout_player_primary_target = { has_country_flag = fallout_player_assignment_collision_target }
					}
				}
			}
			set_country_flag = fallout_player_continuation_error
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
			set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
	}

	set_temp_variable = { fallout_player_preflight_total_count = fallout_player_preflight_existing_count }
	add_to_temp_variable = { fallout_player_preflight_total_count = fallout_player_preflight_ready_count }

		limit = {
			check_variable = { fallout_player_preflight_invalid_found = constant:fallout_runtime.zero }
			check_variable = { fallout_player_preflight_collision_found = constant:fallout_runtime.zero }
			check_variable = {
				var = fallout_player_preflight_total_count
				value = global.fallout_player_snapshot_count
				compare = equals
			}
		}
		set_global_flag = fallout_player_commit_preflight_complete
	}
	else = {
		set_global_flag = fallout_transition_error
		set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
		if = {
			limit = { check_variable = { fallout_player_preflight_collision_found > constant:fallout_runtime.zero } }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
		else = {
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
		}
	}
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}
}

				else = { set_country_flag = fallout_player_assignment_claimed }
			}
		}
	}

	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = { has_country_flag = fallout_player_continuation_committed }
			if = {
				limit = { fallout_player_continuation_commit_is_valid = yes }
				if = {
					limit = {
						has_country_flag = fallout_player_continuation_error
						check_variable = {
							var = fallout_player_error_code
							value = constant:fallout_transition_error_code.player_commit_incomplete
							compare = equals
						}
					}
					clr_country_flag = fallout_player_continuation_error
					clear_variable = fallout_player_error_code
					set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.committed }
				}
				var:fallout_player_final_target = { add_to_array = { array = global.fallout_player_committed_targets value = THIS } }
				add_to_variable = { global.fallout_player_committed_count = constant:fallout_runtime.one }
			}
			else = {
				set_temp_variable = { fallout_player_invalid_commit_found = constant:fallout_runtime.one }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
			}
		}
	}
	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				has_country_flag = fallout_player_continuation_committed
				has_variable = fallout_player_final_target
				var:fallout_player_final_target = { has_country_flag = fallout_player_assignment_collision_target }
			}
			set_country_flag = fallout_player_continuation_error
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
			set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
	}
	every_country = {
		clr_country_flag = fallout_player_assignment_claimed
		clr_country_flag = fallout_player_assignment_collision_target
	}

	if = {
		limit = { check_variable = { fallout_player_assignment_collision_found > constant:fallout_runtime.zero } }
		if = {
			limit = {
				NOT = { has_global_flag = fallout_transition_error }
				check_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
			}
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_assignment_collision }
		}
	}
	else_if = {
		limit = { check_variable = { fallout_player_invalid_commit_found > constant:fallout_runtime.zero } }
		if = {
			limit = {
				NOT = { has_global_flag = fallout_transition_error }
				check_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
			}
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.fallout_player_snapshot_count
				value = global.fallout_player_committed_count
				compare = equals
			}
			all_of_scopes = {
				array = global.fallout_pretransition_player_countries
				fallout_player_continuation_commit_is_valid = yes
			}
		}
		set_global_flag = fallout_player_continuation_commit_complete
		set_global_flag = fallout_player_assignment_uniqueness_verified
	}
}

						set_variable = { fallout_player_reservation_generation = global.fallout_transition_generation }
						set_variable = { PREV.PREV.fallout_player_primary_capital_state = THIS }
					}
				}
			}
			if = {
				limit = {
					has_country_flag = fallout_player_continuation_committed
					fallout_player_continuation_commit_is_valid = yes
				}
				set_country_flag = fallout_player_continuation_planned
				var:fallout_player_final_target = {
					add_to_array = { array = global.fallout_player_committed_targets value = THIS }
					capital_scope = {
						set_state_flag = fallout_player_continuation_reserved
						set_state_flag = fallout_player_primary_capital_reserved
						set_variable = { fallout_player_reservation_origin = PREV.PREV }
						set_variable = { fallout_player_reservation_generation = global.fallout_transition_generation }
					}
				}
				add_to_variable = { global.fallout_player_committed_count = constant:fallout_runtime.one }
			}
			else_if = {
				limit = { has_country_flag = fallout_player_continuation_committed }
				set_country_flag = fallout_player_continuation_error
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.error }
				set_variable = { fallout_player_error_code = constant:fallout_transition_error_code.player_commit_incomplete }
				clr_global_flag = fallout_player_continuation_commit_complete
				clr_global_flag = fallout_player_assignment_uniqueness_verified
			}
			else = {
				clr_country_flag = fallout_player_primary_capital_reserved
				clr_country_flag = fallout_player_continuation_preserved
				clr_country_flag = fallout_player_continues_old_country
				clr_country_flag = fallout_player_successor_choice_required
				clr_country_flag = fallout_player_continuation_planned
				clear_variable = fallout_player_primary_capital_state
				clear_variable = fallout_player_primary_target
				clear_variable = fallout_player_continuation_branch
				set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.snapshotted }
			}
		}
		set_global_flag = fallout_player_continuation_planning_initialized
	}
	for_each_scope_loop = {
		array = global.fallout_pretransition_player_countries
		if = {
			limit = {
				NOT = { has_country_flag = fallout_player_continuation_planned }
				NOT = { has_country_flag = fallout_player_continuation_committed }
				fallout_player_source_has_survivable_origin_state = yes
				has_country_flag = fallout_player_source_anchor_preserved
			}
			fallout_select_player_primary_capital = yes
		}

			limit = {
				NOT = { has_country_flag = fallout_player_continuation_planned }
				NOT = { has_country_flag = fallout_player_continuation_committed }
				has_country_flag = fallout_player_primary_capital_reserved

			set_country_flag = fallout_player_continues_old_country

				NOT = { has_country_flag = fallout_player_continuation_planned }
				NOT = { has_country_flag = fallout_player_continuation_committed }
			}
			set_country_flag = fallout_player_successor_choice_required
			set_variable = { fallout_player_continuation_status = constant:fallout_player_continuation_status.choice_pending }
			if = {
				limit = {
					any_state = {
						has_state_flag = fallout_player_origin_state
						has_variable = fallout_player_origin_scope
						var:fallout_player_origin_scope = { tag = PREV.PREV }
						has_variable = fallout_state_subtype
						check_variable = { fallout_state_subtype = constant:fallout_state_subtype.altered_biosphere }
					}
				}
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.altered_transformation }
			}
			else_if = {
				limit = {
					any_state = {
						has_state_flag = fallout_player_origin_state
						has_variable = fallout_player_origin_scope
						var:fallout_player_origin_scope = { tag = PREV.PREV }
						has_state_flag = fallout_pretransition_coastal
						check_variable = { fallout_pretransition_dockyard > constant:fallout_runtime.zero }
					}
				}
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.refuge_government }
			}
			else_if = {
				limit = {
					any_state = {
						has_state_flag = fallout_player_origin_state
						has_variable = fallout_player_origin_scope
						var:fallout_player_origin_scope = { tag = PREV.PREV }
					}
				}
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.fragmented_tag }
			}
			else = {
				set_variable = { fallout_player_continuation_branch = constant:fallout_player_continuation_branch.emergency_council }
			}
		}
	}
	if = {
		limit = { fallout_player_allocation_reservation_ledger_is_current = yes }
		set_global_flag = fallout_player_allocation_reservation_ledger_complete
	}
	else = {
		clr_global_flag = fallout_player_allocation_reservation_ledger_complete
		set_temp_variable = { fallout_transition_error_to_register = constant:fallout_transition_error_code.player_reservation_incomplete }
		fallout_register_transition_error_if_unowned = yes
		set_temp_variable = { fallout_transition_error_to_register = constant:fallout_transition_error_code.none }
	}
	fallout_rebuild_player_planning_ledger = yes
}
