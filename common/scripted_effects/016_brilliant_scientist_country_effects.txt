# COUNTRY scope on the sovereign carrier. Kruger keeps his scientist role. The
# theorist role is retired before the leader role is added, so his +100% research
# benefit is present exactly once through the sovereign trait.
brilliant_scientist_promote_kruger_as_sovereign = {
	if = {
		limit = { KRG_warren_kruger = { is_hired_as_advisor = yes } }
		deactivate_advisor = KRG_warren_kruger
	}
	KRG_warren_kruger = {
		remove_advisor_role = { slot = theorist }
		set_character_flag = brilliant_scientist_personal_sovereign
	}
	if = {
		limit = {
			NOT = {
				has_country_leader = {
					character = KRG_warren_kruger
				}
			}
		}
		add_country_leader_role = {
			character = KRG_warren_kruger
			promote_leader = yes
			country_leader = {
				ideology = despotism
				traits = { brilliant_scientist_sovereign_director }
			}
		}
	}
}

		set_country_flag = brilliant_scientist_character_transaction_lock

		# The dormant fixed tag has no starting cores. Seed only the verified
		# capital, release KRG, and let the existing route transfer below recheck
		# every selected state before it changes ownership or cores.
		event_target:brilliant_scientist_formation_capital_state = {
			add_core_of = KRG
		}
		release = KRG

		}

		brilliant_scientist_reconcile_old_host_after_transfer = yes

		set_country_flag = brilliant_scientist_kruger_departed
		if = {
			limit = { has_event_target = brilliant_scientist_current_host }
			clear_global_event_target = brilliant_scientist_current_host
		}
		KRG_warren_kruger = { set_nationality = KRG }

			set_country_flag = brilliant_scientist_character_transaction_lock
			save_event_target_as = brilliant_scientist_kruger_state_carrier
			set_country_flag = brilliant_scientist_host_history

			}

			}

				limit = { event_target:brilliant_scientist_kruger_state_former_host = { has_country_flag = brilliant_scientist_enclave_settlement } }
				set_country_flag = brilliant_scientist_formation_enclave
			}
			if = {
				limit = {
					has_event_target = brilliant_scientist_formation_capital_state
					event_target:brilliant_scientist_formation_capital_state = {
						is_owned_by = event_target:brilliant_scientist_kruger_state_carrier
						has_state_flag = brilliant_scientist_primary_facility
					}
				}

			brilliant_scientist_bind_formation_capital_as_primary_facility = yes
			brilliant_scientist_bind_transferred_secondary_facility = yes
			brilliant_scientist_add_kruger_roles = yes
			brilliant_scientist_initialize_current_kruger_state = yes
			clr_country_flag = brilliant_scientist_character_transaction_lock
		}

		brilliant_scientist_clear_formation_territory_plan = yes
		clr_country_flag = brilliant_scientist_character_transaction_lock
		clr_global_flag = brilliant_scientist_character_transaction_lock

	}
