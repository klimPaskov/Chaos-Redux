fallout_event_401_apply_failure_deaths = {
	var:fallout_event_401_target_state_id = {
		if = {
			limit = { check_variable = { state_population_k value = constant:fallout_event_runtime.zero compare = greater_than } }
			set_temp_variable = { fallout_event_401_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_401_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_401_death_request = constant:fallout_event_401_threshold.result_failure_death_percent }
			round_temp_variable = fallout_event_401_death_request
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_requested = fallout_event_401_death_request }
			set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:fallout_event_401_threshold.minimum_remaining_people }
			set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.fallout_aftermath }
			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }
			apply_exact_state_civilian_population_loss = yes
			clear_temp_variable = fallout_event_401_death_request
		}
	}
}

# Supported scope: COUNTRY. Advance through conflict and choice after the
# major opening envelope has been consumed.

	set_temp_variable = { fallout_event_401_arc_stage_transition_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = { has_variable = fallout_event_401_major_arc_ticket }
		set_temp_variable = { fallout_event_requested_ticket = fallout_event_401_major_arc_ticket }
		set_temp_variable = { fallout_event_requested_arc_stage = constant:fallout_event_arc_stage.conflict }
		set_temp_variable = { fallout_event_requested_outcome = constant:fallout_event_outcome_type.none }
		fallout_event_advance_major_arc = yes
		if = {
			limit = { check_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one } }
			set_temp_variable = { fallout_event_requested_arc_stage = constant:fallout_event_arc_stage.choice }
			fallout_event_advance_major_arc = yes
			if = {
				limit = { check_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one } }
				set_temp_variable = { fallout_event_401_arc_stage_transition_accepted = constant:fallout_event_runtime.one }
			}
		}
	}
	else = {
		set_temp_variable = { fallout_event_401_arc_stage_transition_accepted = constant:fallout_event_runtime.one }
	}
}

fallout_event_401_schedule_result = {
	set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			OR = {
				fallout_event_issued_ordinary_receipt_is_current = yes
				fallout_event_issued_major_arc_stage_is_current = yes
			}
			NOT = { has_country_flag = fallout_event_401_chain_committed }
			check_variable = { fallout_event_401_requested_branch > constant:fallout_event_401_branch.none }
			check_variable = { fallout_event_401_requested_branch < constant:fallout_event_401_branch.upper_bound }
			fallout_event_401_opening_target_is_current = yes
