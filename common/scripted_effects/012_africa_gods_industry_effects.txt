# HOST. Choose a physical proposal without reserving a program slot.
gods_of_africa_select_industry_site = {
	clr_country_flag = gods_of_africa_industry_candidate_available
	clear_variable = gods_of_africa_industry_selected_site
	clear_variable = gods_of_africa_industry_selected_damage
	if = {
		limit = { gods_of_africa_industry_candidate_is_available = yes }
		capital_scope = {
			if = {
				limit = { gods_of_africa_industry_site_is_eligible = yes }
				PREV = {
					set_variable = { gods_of_africa_industry_selected_site = PREV }
					set_variable = { gods_of_africa_industry_selected_damage = PREV.damaged_building_level@industrial_complex }
					set_country_flag = gods_of_africa_industry_candidate_available
				}
			}
		}
		if = {
			limit = { NOT = { has_country_flag = gods_of_africa_industry_candidate_available } }
			every_owned_state = {
				limit = { gods_of_africa_industry_site_is_eligible = yes }
				if = {
					limit = {
						OR = {
							PREV = { NOT = { has_country_flag = gods_of_africa_industry_candidate_available } }
							check_variable = { var = damaged_building_level@industrial_complex value = PREV.gods_of_africa_industry_selected_damage compare = greater_than }
						}
					}
					PREV = {
						set_variable = { gods_of_africa_industry_selected_site = PREV }
						set_variable = { gods_of_africa_industry_selected_damage = PREV.damaged_building_level@industrial_complex }
						set_country_flag = gods_of_africa_industry_candidate_available
					}
				}
			}
		}
	}
}

# ACTOR after demand epoch + ledger open + final CPP quote. No repricing.
gods_of_africa_bind_industry_terms = {

			NOT = { has_country_flag = gods_of_africa_industry_terms_bound }
			var:gods_of_africa_demand_host = {
				gods_of_africa_industry_candidate_is_available = yes

			}
		}
		# Reclaim only the exact closed, expired record identified by this owner.

			if = {
				limit = { gods_of_africa_industry_program_can_reclaim = yes }

			set_variable = { gods_of_africa_industry_contract_id = gods_of_africa_demand_id }
			set_variable = { gods_of_africa_industry_quote = gods_of_africa_demand_amount }
			var:gods_of_africa_industry_host = {
				PREV = { set_variable = { gods_of_africa_industry_site = PREV.gods_of_africa_industry_selected_site } }
			}
			set_temp_variable = { gods_of_africa_industry_days = constant:gods_of_africa_industry.base_service_days }
			multiply_temp_variable = { gods_of_africa_industry_days = gods_of_africa_industry_quote }

			set_variable = { gods_of_africa_industry_service_days = gods_of_africa_industry_days }
			set_country_flag = gods_of_africa_industry_terms_bound
			clr_country_flag = gods_of_africa_industry_terms_retired
			clr_country_flag = gods_of_africa_industry_contract_closed
			var:gods_of_africa_industry_host = {

				set_variable = { gods_of_africa_industry_program_host_id = PREV.gods_of_africa_industry_host_id }

				set_variable = { gods_of_africa_industry_program_contract_id = PREV.gods_of_africa_industry_contract_id }

# HOST private. Remove only the recorded recipient's state effect.
# Does not require current authority, participant existence or global host pointer.
gods_of_africa_industry_remove_native = {
	if = {
		limit = { has_variable = gods_of_africa_industry_program_site }
		if = {
			limit = { var:gods_of_africa_industry_program_site = { has_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV } } }
			var:gods_of_africa_industry_program_site = { remove_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV } }
		}
		if = {
			limit = { var:gods_of_africa_industry_program_site = { has_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV } } }
			set_country_flag = gods_of_africa_industry_program_unresolved
		}
		else = { clr_country_flag = gods_of_africa_industry_program_native_applied }
	}
	else = { set_country_flag = gods_of_africa_industry_program_unresolved }
}

# HOST. No donor dereference and no reinstall on reconciliation.
gods_of_africa_review_industry_program = {
	if = {
		limit = { has_country_flag = gods_of_africa_industry_program_reserved NOT = { has_country_flag = gods_of_africa_industry_program_inflight } }
		if = {
			limit = {
				OR = {
					NOT = { gods_of_africa_industry_program_owner_is_current = yes }
					NOT = { var:gods_of_africa_industry_program_site = { gods_of_africa_industry_program_site_is_secured = yes } }
				}
			}
			set_temp_variable = { gods_of_africa_industry_retire_host_id = gods_of_africa_industry_program_host_id }
			set_temp_variable = { gods_of_africa_industry_retire_generation = gods_of_africa_industry_program_generation }
			set_temp_variable = { gods_of_africa_industry_retire_origin_host_id = gods_of_africa_industry_program_origin_host_id }
			set_temp_variable = { gods_of_africa_industry_retire_epoch = gods_of_africa_industry_program_epoch }
			set_temp_variable = { gods_of_africa_industry_retire_contract_id = gods_of_africa_industry_program_contract_id }
			gods_of_africa_retire_industry_program = yes
		}
		else = {
			if = {
				limit = { check_variable = { var = gods_of_africa_industry_program_due_day value = global.num_days compare = less_than_or_equals } }
				# No late zero-damage observation can manufacture paid completion.
				gods_of_africa_industry_remove_native = yes
			}
			else_if = {
				limit = { has_country_flag = gods_of_africa_industry_program_native_applied }
				if = {
					limit = { var:gods_of_africa_industry_program_site = { has_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV } } }
					set_temp_variable = { gods_of_africa_industry_observed_damage = constant:gods_of_africa_value.zero }
					var:gods_of_africa_industry_program_site = { set_temp_variable = { gods_of_africa_industry_observed_damage = damaged_building_level@industrial_complex } }
					if = {
						limit = {
							check_variable = { gods_of_africa_industry_observed_damage = constant:gods_of_africa_value.zero }
							check_variable = { gods_of_africa_industry_program_last_damage > constant:gods_of_africa_value.zero }
							check_variable = { gods_of_africa_industry_program_funded > constant:gods_of_africa_value.zero }
							check_variable = { gods_of_africa_industry_program_granted_days > constant:gods_of_africa_value.zero }
						}
						set_country_flag = gods_of_africa_industry_program_site_completed
						gods_of_africa_industry_remove_native = yes
					}
					set_variable = { gods_of_africa_industry_program_last_damage = gods_of_africa_industry_observed_damage }
					set_variable = { gods_of_africa_industry_program_observation_day = global.num_days }
				}
				else = { set_country_flag = gods_of_africa_industry_program_unresolved }
			}
			if = {
				limit = { gods_of_africa_industry_program_can_reclaim = yes }
				set_temp_variable = { gods_of_africa_industry_retire_host_id = gods_of_africa_industry_program_host_id }
				set_temp_variable = { gods_of_africa_industry_retire_generation = gods_of_africa_industry_program_generation }
				set_temp_variable = { gods_of_africa_industry_retire_origin_host_id = gods_of_africa_industry_program_origin_host_id }
				set_temp_variable = { gods_of_africa_industry_retire_epoch = gods_of_africa_industry_program_epoch }
				set_temp_variable = { gods_of_africa_industry_retire_contract_id = gods_of_africa_industry_program_contract_id }
				gods_of_africa_retire_industry_program = yes
			}
		}
	}
}

# ACTOR. Ledger/history owner chooses outcome. Close financial bindings only.

}

# ACTOR. Donor/substitute/default cleanup preserves all host-owned paid time.

	}
}

# ACTOR private. Durable rollback uses measured current deltas, never prior funding.
gods_of_africa_industry_rollback_transaction = {
	if = {
		limit = { gods_of_africa_industry_transaction_is_owned = yes }
		set_temp_variable = { gods_of_africa_industry_refund = var:gods_of_africa_industry_host.gods_of_africa_industry_program_txn_debit }
		if = { limit = { check_variable = { gods_of_africa_industry_refund > constant:gods_of_africa_value.zero } } add_political_power = gods_of_africa_industry_refund }
		var:gods_of_africa_industry_host = {
			set_temp_variable = { gods_of_africa_industry_reverse_credit = gods_of_africa_industry_program_txn_credit }
			multiply_temp_variable = { gods_of_africa_industry_reverse_credit = constant:gods_of_africa_industry.negation_factor }
			if = { limit = { check_variable = { gods_of_africa_industry_program_txn_credit > constant:gods_of_africa_value.zero } } add_political_power = gods_of_africa_industry_reverse_credit }
			if = {
				limit = { has_country_flag = gods_of_africa_industry_program_txn_native_written }
				gods_of_africa_industry_remove_native = yes
				if = {
					limit = {
						has_country_flag = gods_of_africa_industry_program_txn_had_native
						check_variable = { var = gods_of_africa_industry_program_txn_due_day value = global.num_days compare = greater_than }
					}
					set_variable = { gods_of_africa_industry_program_native_days = gods_of_africa_industry_program_txn_due_day }
					subtract_from_variable = { gods_of_africa_industry_program_native_days = global.num_days }
					var:gods_of_africa_industry_program_site = { add_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV days = PREV.gods_of_africa_industry_program_native_days } }
					if = {
						limit = { var:gods_of_africa_industry_program_site = { has_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV } } }
						set_country_flag = gods_of_africa_industry_program_native_applied
					}
					else = { set_country_flag = gods_of_africa_industry_program_unresolved }
				}
				else_if = {
					limit = { var:gods_of_africa_industry_program_site = { has_dynamic_modifier = { modifier = gods_of_africa_industry_repair_program scope = PREV } } }
					set_country_flag = gods_of_africa_industry_program_unresolved
				}
			}
			if = {
				limit = {
					OR = {
						NOT = { check_variable = { var = has_political_power value = gods_of_africa_industry_program_txn_host_pp compare = equals } }
						NOT = { check_variable = { var = PREV.has_political_power value = gods_of_africa_industry_program_txn_donor_pp compare = equals } }
					}
				}
				set_country_flag = gods_of_africa_industry_program_unresolved
			}
			clr_country_flag = gods_of_africa_industry_program_inflight
			clr_country_flag = gods_of_africa_industry_program_txn_native_written
			clr_country_flag = gods_of_africa_industry_program_txn_had_native
			clear_variable = gods_of_africa_industry_program_native_days
		}
	}
	else = {
		# A foreign/stale transaction cannot refund or touch a replacement record.
		set_country_flag = gods_of_africa_industry_transaction_unresolved
	}
}

# ACTOR. Caller initializes normal payment_family/amount. Outputs default zero.
gods_of_africa_pay_industrial_tribute = {
	# A fresh actor click can observe an earned completion between owner ticks.
	# Verify the existing bound quote before reconciling its own host record.
	if = {
		limit = { gods_of_africa_industry_quote_is_current = yes }
		var:gods_of_africa_industry_host = { gods_of_africa_review_industry_program = yes }

	set_variable = { gods_of_africa_industry_payment_new_days = constant:gods_of_africa_value.zero }
	# Forwarded trigger outputs have caller-owned temporary storage.

	set_temp_variable = { gods_of_africa_industry_increment = constant:gods_of_africa_value.zero }
	if = {
		limit = { gods_of_africa_industry_payment_can_execute = yes }
		# Persist both authority and exact money/service snapshot before debit.
		var:gods_of_africa_industry_host = {
			set_variable = { gods_of_africa_industry_program_txn_participant = gods_of_africa_industry_program_participant }
			set_variable = { gods_of_africa_industry_program_txn_host_id = gods_of_africa_industry_program_host_id }
			set_variable = { gods_of_africa_industry_program_txn_generation = gods_of_africa_industry_program_generation }

			set_variable = { gods_of_africa_industry_program_txn_epoch = gods_of_africa_industry_program_epoch }
			set_variable = { gods_of_africa_industry_program_txn_contract_id = gods_of_africa_industry_program_contract_id }
			set_variable = { gods_of_africa_industry_program_txn_site = gods_of_africa_industry_program_site }
			set_variable = { gods_of_africa_industry_program_txn_quote = gods_of_africa_industry_program_quote }
			set_variable = { gods_of_africa_industry_program_txn_service_days = gods_of_africa_industry_program_service_days }
			set_variable = { gods_of_africa_industry_program_txn_funded = gods_of_africa_industry_program_funded }
			set_variable = { gods_of_africa_industry_program_txn_granted_days = gods_of_africa_industry_program_granted_days }
			set_variable = { gods_of_africa_industry_program_txn_due_day = gods_of_africa_industry_program_due_day }
			set_variable = { gods_of_africa_industry_program_txn_amount = PREV.gods_of_africa_payment_amount }
			set_variable = { gods_of_africa_industry_program_txn_donor_pp = PREV.has_political_power }
			set_variable = { gods_of_africa_industry_program_txn_host_pp = has_political_power }
			set_variable = { gods_of_africa_industry_program_txn_debit = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_industry_program_txn_credit = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_industry_program_txn_entitlement = gods_of_africa_industry_entitlement }

			if = {

				set_country_flag = gods_of_africa_industry_program_txn_had_native
			}
			set_country_flag = gods_of_africa_industry_program_inflight
		}
		set_temp_variable = { gods_of_africa_industry_debit_amount = gods_of_africa_payment_amount }
		multiply_temp_variable = { gods_of_africa_industry_debit_amount = constant:gods_of_africa_industry.negation_factor }
		add_political_power = gods_of_africa_industry_debit_amount
		var:gods_of_africa_industry_host = {
			add_political_power = gods_of_africa_industry_program_txn_amount
			set_variable = { gods_of_africa_industry_program_txn_debit = gods_of_africa_industry_program_txn_donor_pp }

					check_variable = { gods_of_africa_industry_program_txn_credit = gods_of_africa_industry_program_txn_amount }
				}
			}
			if = {
				limit = { NOT = { gods_of_africa_industry_completion_is_proven = yes } }

					}
					PREV = {
						set_variable = { gods_of_africa_payment_delivered = PREV.gods_of_africa_industry_program_txn_debit }
						set_variable = { gods_of_africa_payment_received = PREV.gods_of_africa_industry_program_txn_credit }
					}
					clr_country_flag = gods_of_africa_industry_program_inflight
					clr_country_flag = gods_of_africa_industry_program_txn_native_written
					clr_country_flag = gods_of_africa_industry_program_txn_had_native
					clear_variable = gods_of_africa_industry_program_native_days
				}
			}
			else = { gods_of_africa_industry_rollback_transaction = yes }
		}
		else = { gods_of_africa_industry_rollback_transaction = yes }
	}
