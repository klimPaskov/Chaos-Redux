		set_temp_variable = { brilliant_scientist_project_capacity_release = constant:brilliant_scientist_project_capacity.prototype }

		set_temp_variable = { brilliant_scientist_project_capacity_release = constant:brilliant_scientist_project_capacity.deployment }

		set_temp_variable = { brilliant_scientist_project_capacity_release = constant:brilliant_scientist_project_capacity.weaponization }

		limit = { brilliant_scientist_project_is_suspended = yes }

	clamp_variable = { var = brilliant_scientist_project_capacity min = constant:brilliant_scientist_measure.minimum max = constant:brilliant_scientist_measure.maximum }
