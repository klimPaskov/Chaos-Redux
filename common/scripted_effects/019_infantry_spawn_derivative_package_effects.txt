# district projects, ghost population decline, defeat handling, and annex

# and lot totals after derivative materialization.

infantry_spawn_derivative_find_registry_row = {
	set_temp_variable = { infantry_spawn_derivative_registry_index = constant:infantry_spawn_event.invalid_index }
	set_temp_variable = { chaos_unit_family_registry_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_derivative_registry_index < constant:infantry_spawn_event.first_index }
			check_variable = { var = chaos_unit_family_registry_index value = global.chaos_unit_family_id_entries^num compare = less_than }
		}
		if = {
			limit = {
				check_variable = { var = global.chaos_unit_family_id_entries^chaos_unit_family_registry_index value = infantry_spawn_derivative_family_id compare = equals }
				chaos_unit_family_current_row_is_enabled = yes
				chaos_unit_family_current_row_allows_event19_family_lot = yes
				chaos_unit_family_current_row_can_spawn = yes
			}
			set_temp_variable = { infantry_spawn_derivative_registry_index = chaos_unit_family_registry_index }
		}
		add_to_temp_variable = { chaos_unit_family_registry_index = constant:infantry_spawn_derivative_package_value.one }
	}
	if = {
		limit = { check_variable = { infantry_spawn_derivative_registry_index > constant:infantry_spawn_event.invalid_index } }
		set_temp_variable = { chaos_unit_family_registry_index = infantry_spawn_derivative_registry_index }
		set_temp_variable = { infantry_spawn_current_family_id = global.chaos_unit_family_id_entries^infantry_spawn_derivative_registry_index }
		set_temp_variable = { infantry_spawn_current_registry_provider_id = global.chaos_unit_family_provider_id_entries^infantry_spawn_derivative_registry_index }
	}
}

infantry_spawn_derivative_select_reinforcement_origin = {
	save_event_target_as = infantry_spawn_generation_country
	if = {
		limit = {
			any_controlled_state = {
				infantry_spawn_state_is_eligible = yes
				has_state_flag = infantry_spawn_derivative_sustainment_site
			}
		}
		random_controlled_state = {
			limit = {
				infantry_spawn_state_is_eligible = yes
				has_state_flag = infantry_spawn_derivative_sustainment_site
			}
			save_event_target_as = infantry_spawn_current_origin_state
		}
	}
	else_if = {
		limit = { capital_scope = { infantry_spawn_state_is_eligible = yes } }
		capital_scope = { save_event_target_as = infantry_spawn_current_origin_state }
	}
	else = {
		random_controlled_state = {
			limit = { infantry_spawn_state_is_eligible = yes }
			save_event_target_as = infantry_spawn_current_origin_state
		}
	}
}

infantry_spawn_derivative_materialize_one_family_formation = {
	set_temp_variable = { infantry_spawn_derivative_materialization_succeeded = constant:infantry_spawn_derivative_package_value.zero }
	infantry_spawn_initialize_global_system = yes
	infantry_spawn_derivative_initialize_private_ledger = yes
	infantry_spawn_derivative_find_registry_row = yes
	if = {
		limit = {
			check_variable = { infantry_spawn_derivative_registry_index > constant:infantry_spawn_event.invalid_index }
			infantry_spawn_country_has_eligible_state = yes
			infantry_spawn_all_ledgers_are_aligned = yes
		}
		infantry_spawn_derivative_select_reinforcement_origin = yes
		if = {
			limit = { has_event_target = infantry_spawn_current_origin_state }
			set_temp_variable = { infantry_spawn_derivative_global_lots_before = global.infantry_spawn_total_lots }
			set_temp_variable = { infantry_spawn_derivative_global_formations_before = global.infantry_spawn_total_formations }
			add_to_variable = { infantry_spawn_generation = constant:infantry_spawn_derivative_package_value.one }
			infantry_spawn_allocate_generation_uid = yes
			set_temp_variable = { infantry_spawn_current_evolution_stage = constant:infantry_spawn_stage.anomalous_muster }
			set_temp_variable = { infantry_spawn_current_generation_profile = constant:infantry_spawn_generation_profile.anomalous }
			set_temp_variable = { infantry_spawn_current_eligible_state_count = constant:infantry_spawn_derivative_package_value.one }
			set_temp_variable = { infantry_spawn_current_selected_state_count = constant:infantry_spawn_derivative_package_value.one }
			set_temp_variable = { infantry_spawn_current_generation_lot_count = constant:infantry_spawn_derivative_package_value.zero }
			set_temp_variable = { infantry_spawn_current_generation_unit_count = constant:infantry_spawn_derivative_package_value.zero }
			set_temp_variable = { infantry_spawn_current_generation_status = constant:infantry_spawn_generation_status.open }
			infantry_spawn_append_current_generation_row = yes
			infantry_spawn_prepare_current_state_context = yes
			set_temp_variable = { infantry_spawn_current_lot_profile = constant:infantry_spawn_lot_family.registered_anomalous }
			infantry_spawn_create_current_registered_lot = yes
			if = {
				limit = { NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure } }
				infantry_spawn_append_selected_state_row = yes
				meta_effect = {
					text = { chaos_unit_family_provider_[PROVIDER]_event19_spawn_unit = yes }
					PROVIDER = "[?infantry_spawn_current_registry_provider_id|.0]"
				}
			}
			if = {
				limit = { NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure } }
				set_temp_variable = { infantry_spawn_current_generation_status = constant:infantry_spawn_generation_status.resolved }
				infantry_spawn_update_current_generation_totals = yes
				set_temp_variable = { infantry_spawn_derivative_materialization_succeeded = constant:infantry_spawn_derivative_package_value.one }
				add_to_variable = { infantry_spawn_derivative_reinforcement_count = constant:infantry_spawn_derivative_package_value.one }
			}
			set_variable = { global.infantry_spawn_total_lots = infantry_spawn_derivative_global_lots_before }
			set_variable = { global.infantry_spawn_total_formations = infantry_spawn_derivative_global_formations_before }
		}
	}
}

infantry_spawn_derivative_count_live_family_divisions = {
	set_temp_variable = { infantry_spawn_family_live_division_count = constant:infantry_spawn_derivative_package_value.zero }
	every_country_division = {
		limit = {
			has_variable = infantry_spawn_event19_family_id
			check_variable = { var = infantry_spawn_event19_family_id value = ROOT.infantry_spawn_derivative_family_id compare = equals }
		}
		add_to_temp_variable = { infantry_spawn_family_live_division_count = constant:infantry_spawn_derivative_package_value.one }
	}
}

infantry_spawn_derivative_evaluate_current_family_management = {

infantry_spawn_derivative_unlock_zombie_templates = {
	if = {
		limit = { NOT = { is_in_array = { array = infantry_spawn_trainable_family_ids value = constant:event19_zombie_family.family_id } } }
		add_to_array = { array = infantry_spawn_trainable_family_ids value = constant:event19_zombie_family.family_id }
	}
	every_country_division = {
		limit = {
			has_variable = infantry_spawn_event19_family_id
			has_variable = infantry_spawn_event19_template_uid
			check_variable = { var = infantry_spawn_event19_family_id value = constant:event19_zombie_family.family_id compare = equals }
		}
		set_temp_variable = { infantry_spawn_derivative_zombie_template_uid = infantry_spawn_event19_template_uid }
		ROOT = {
			meta_effect = {
				text = {
					set_division_force_allow_recruiting = { division_template = "Unbidden Muster [TEMPLATE_UID]" force_allow_recruiting = yes }
					set_division_template_lock = { division_template = "Unbidden Muster [TEMPLATE_UID]" is_locked = yes }
				}
				TEMPLATE_UID = "[?infantry_spawn_derivative_zombie_template_uid|.0]"
			}
		}
	}
	set_country_flag = infantry_spawn_derivative_zombie_training_authorized
}

infantry_spawn_derivative_authorize_zombie_training = {
	infantry_spawn_derivative_evaluate_current_family_management = yes
	if = {
		limit = { check_variable = { infantry_spawn_family_provider_can_train > constant:infantry_spawn_derivative_package_value.zero } }
		set_temp_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
		infantry_spawn_derivative_pay_current_family_management = yes
		if = {
			limit = { check_variable = { infantry_spawn_family_provider_payment_succeeded > constant:infantry_spawn_derivative_package_value.zero } }
			infantry_spawn_derivative_unlock_zombie_templates = yes
		}
	}
}

infantry_spawn_derivative_rally_zombie_band = {
	if = {
		limit = {
			infantry_spawn_derivative_is_zombie = yes
			NOT = { has_country_flag = infantry_spawn_derivative_reinforcement_cooldown }
			check_variable = { var = manpower value = constant:infantry_spawn_derivative_reinforcement.zombie_manpower_cost compare = greater_than_or_equals }
			check_variable = { var = num_equipment@infantry_equipment value = constant:infantry_spawn_derivative_reinforcement.zombie_infantry_equipment_cost compare = greater_than_or_equals }
		}
		set_temp_variable = { infantry_spawn_derivative_negative_cost = constant:infantry_spawn_derivative_package_value.zero }
		subtract_from_temp_variable = { infantry_spawn_derivative_negative_cost = constant:infantry_spawn_derivative_reinforcement.zombie_manpower_cost }
		add_manpower = infantry_spawn_derivative_negative_cost
		set_temp_variable = { infantry_spawn_derivative_negative_cost = constant:infantry_spawn_derivative_package_value.zero }
		subtract_from_temp_variable = { infantry_spawn_derivative_negative_cost = constant:infantry_spawn_derivative_reinforcement.zombie_infantry_equipment_cost }
		add_equipment_to_stockpile = { type = infantry_equipment amount = infantry_spawn_derivative_negative_cost }
		infantry_spawn_derivative_materialize_one_family_formation = yes
		if = {
			limit = { check_variable = { infantry_spawn_derivative_materialization_succeeded > constant:infantry_spawn_derivative_package_value.zero } }
			if = {
				limit = { has_country_flag = infantry_spawn_derivative_zombie_training_authorized }
				infantry_spawn_derivative_unlock_zombie_templates = yes
			}
			set_temp_variable = { infantry_spawn_derivative_cooldown_days = constant:infantry_spawn_derivative_runtime.reinforcement_cooldown_days }
			set_country_flag = { flag = infantry_spawn_derivative_reinforcement_cooldown days = infantry_spawn_derivative_cooldown_days }
		}
		else = {
			add_manpower = constant:infantry_spawn_derivative_reinforcement.zombie_manpower_cost
			add_equipment_to_stockpile = { type = infantry_equipment amount = constant:infantry_spawn_derivative_reinforcement.zombie_infantry_equipment_cost }
		}
	}
}

infantry_spawn_derivative_manifest_paid_family_formation = {
	infantry_spawn_derivative_evaluate_current_family_management = yes
	if = {
		limit = {
			OR = {
				check_variable = { infantry_spawn_family_provider_can_spawn > constant:infantry_spawn_derivative_package_value.zero }
				check_variable = { infantry_spawn_family_provider_can_train > constant:infantry_spawn_derivative_package_value.zero }
			}
		}
		set_temp_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
		infantry_spawn_derivative_pay_current_family_management = yes
		if = {
			limit = { check_variable = { infantry_spawn_family_provider_payment_succeeded > constant:infantry_spawn_derivative_package_value.zero } }
			infantry_spawn_derivative_materialize_one_family_formation = yes
			if = {
				limit = { check_variable = { infantry_spawn_derivative_materialization_succeeded > constant:infantry_spawn_derivative_package_value.zero } }
				set_temp_variable = { infantry_spawn_derivative_cooldown_days = constant:infantry_spawn_derivative_runtime.reinforcement_cooldown_days }
				set_country_flag = { flag = infantry_spawn_derivative_reinforcement_cooldown days = infantry_spawn_derivative_cooldown_days }
			}
			else = { infantry_spawn_derivative_refund_current_family_management = yes }
		}
	}
}

	save_event_target_as = infantry_spawn_derivative_current_marker_state

				array = infantry_spawn_derivative_owned_marker_states
