fallout_event_691_record_history = {
	set_temp_variable = { fallout_event_691_history_should_record = constant:fallout_event_runtime.zero }
	if = { limit = { AND = { check_variable = { fallout_event_691_history_stage = constant:fallout_event_runtime.zero } NOT = { has_country_flag = fallout_event_691_choice_history_recorded } } } set_temp_variable = { fallout_event_691_history_should_record = constant:fallout_event_runtime.one } }
	else_if = { limit = { AND = { check_variable = { fallout_event_691_history_stage = constant:fallout_event_runtime.one } NOT = { has_country_flag = fallout_event_691_result_history_recorded } } } set_temp_variable = { fallout_event_691_history_should_record = constant:fallout_event_runtime.one } }
	else_if = { limit = { AND = { check_variable = { fallout_event_691_history_stage = constant:fallout_event_runtime.two } NOT = { has_country_flag = fallout_event_691_callback_history_recorded } } } set_temp_variable = { fallout_event_691_history_should_record = constant:fallout_event_runtime.one } }
	if = {
		limit = { check_variable = { fallout_event_691_history_should_record = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.cancelled }
		if = {
			limit = { check_variable = { fallout_event_691_history_stage = constant:fallout_event_runtime.zero } }
			if = { limit = { check_variable = { fallout_event_691_selected_branch value = constant:fallout_event_691_branch.release_compensate compare = equals } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.choice_release } }
			else_if = { limit = { check_variable = { fallout_event_691_selected_branch value = constant:fallout_event_691_branch.maintain_cordon compare = equals } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.choice_cordon } }
			else_if = { limit = { check_variable = { fallout_event_691_selected_branch value = constant:fallout_event_691_branch.ward_tribunal compare = equals } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.choice_tribunal } }
			else = { set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.choice_rotate } }
		}
		else_if = {
			limit = { check_variable = { fallout_event_691_history_stage = constant:fallout_event_runtime.one } }
			if = {
				limit = { check_variable = { fallout_event_691_selected_branch value = constant:fallout_event_691_branch.release_compensate compare = equals } }
				if = { limit = { check_variable = { fallout_event_691_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.release_success } }
				else_if = { limit = { check_variable = { fallout_event_691_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.release_partial } }
				else = { set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.release_failure } }
			}
			else_if = {
				limit = { check_variable = { fallout_event_691_selected_branch value = constant:fallout_event_691_branch.maintain_cordon compare = equals } }
				if = { limit = { check_variable = { fallout_event_691_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.cordon_success } }
				else_if = { limit = { check_variable = { fallout_event_691_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.cordon_partial } }
				else = { set_temp_variable = { fallout_event_691_history_payload = constant:fallout_event_691_log.cordon_failure } }
