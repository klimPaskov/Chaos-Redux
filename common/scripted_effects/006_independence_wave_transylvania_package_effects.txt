independence_wave_refresh_tra_settlement_lifecycle = {
	if = {
		limit = { is_independence_wave_tra_package = yes }
		remove_ideas = { tra_divided_border_authority tra_danube_settlement }
		if = {
			limit = { has_stable_independence_wave_tra_settlement = yes }
			add_ideas = tra_danube_settlement
			set_country_flag = independence_wave_tra_settlement_stabilized
		}
		else = {
			add_ideas = tra_divided_border_authority
			clr_country_flag = independence_wave_tra_settlement_stabilized
		}
	}
}

		add_to_variable = { independence_wave_tra_federal_legitimacy = independence_wave_tra_federal_delta }

			var = independence_wave_tra_frontier_cohesion
			min = constant:independence_wave_transylvania_pressure.minimum
			max = constant:independence_wave_transylvania_pressure.maximum
		}

			max = constant:independence_wave_transylvania_pressure.maximum

	}
}
