# ============================================================================
# CHAOS REDUX - CBRN SPECIAL-PROJECT SELECTION EFFECTS
# ============================================================================
# Overview:
# - Owns CBRN-specific project-pool selection used by explicitly integrated
#   event-family rewards.
# - Keeps biological, chemical, and weaponized-zombie project grants outside
#   the shared cross-system dynamic-effect registry.
# - The stable public identifier is retained without a wrapper or alias.
# ============================================================================

grant_random_chaos_special_project_available_tech = {
	clr_country_flag = chaos_random_special_project_granted
	if = {
		limit = {
			OR = {
				NOT = { has_tech = anthrax_bomb_delivery_systems }
				NOT = { has_tech = plague_bomb_delivery_systems }
				NOT = { has_tech = tularemia_bomb_delivery_systems }
				NOT = { has_tech = smallpox_bomb_delivery_systems }
				NOT = { has_tech = zombie_disease_bomb_delivery_systems }
				NOT = { has_tech = sarin }
				NOT = { has_tech = soman }
			}
		}
		random_list = {
			10 = {
				modifier = { factor = 0 has_tech = anthrax_bomb_delivery_systems }
				complete_special_project = { project = sp:anthrax_bomb show_modifiers = no }
				set_technology = { anthrax_bomb_delivery_systems = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
			10 = {
				modifier = { factor = 0 has_tech = plague_bomb_delivery_systems }
				complete_special_project = { project = sp:plague_bomb show_modifiers = no }
				set_technology = { anthrax_bomb_delivery_systems = 1 plague_bomb_delivery_systems = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
			8 = {
				modifier = { factor = 0 has_tech = tularemia_bomb_delivery_systems }
				complete_special_project = { project = sp:tularemia_bomb show_modifiers = no }
				set_technology = { tularemia_bomb_delivery_systems = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
			6 = {
				modifier = { factor = 0 has_tech = smallpox_bomb_delivery_systems }
				complete_special_project = { project = sp:smallpox_bomb show_modifiers = no }
				set_technology = { smallpox_bomb_delivery_systems = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
			8 = {
				modifier = { factor = 0 has_tech = zombie_disease_bomb_delivery_systems }
				randomize_and_complete_weaponized_zombie_project = yes
				set_technology = { zombie_disease_bomb_delivery_systems = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
			8 = {
				modifier = { factor = 0 has_tech = sarin }
				complete_special_project = { project = sp:sp_cw_sarin_program show_modifiers = no }
				set_technology = { sarin = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
			6 = {
				modifier = { factor = 0 has_tech = soman }
				complete_special_project = { project = sp:sp_cw_sarin_program show_modifiers = no }
				complete_special_project = { project = sp:sp_cw_soman_program show_modifiers = no }
				set_technology = { sarin = 1 soman = 1 }
				set_country_flag = chaos_random_special_project_granted
			}
		}
	}
}
