# ============================================================================
# CHAOS REDUX - CBRN SPECIAL-PROJECT SELECTION EFFECTS
# ============================================================================
# Overview:
# - Owns CBRN-specific project-pool selection used by explicitly integrated
#   event-family rewards.
# - Keeps biological, chemical, and weaponized-zombie project grants outside
#   the shared cross-system dynamic-effect registry.
# - A missing delivery technology can still be restored after its project is
#   complete; completion outputs and zombie profiles are not awarded twice.
# - The stable public identifier is retained without a wrapper or alias.

					if = { limit = { tag = UWR is_special_project_completed = sp:anthrax_bomb soviet_collapse_uwr_random_bio_site_is_valid = yes } soviet_collapse_uwr_register_random_biological_arsenal = yes }
