# COUNTRY scope. Administrative and submission scores remain route-owned
# progress measures. Terminal territorial proof is deliberately separate and
# is computed only by the explicit audit below, never from these receipts.

	clamp_variable = { var = brilliant_scientist_global_administration_score min = constant:brilliant_scientist_krg_value.zero max = constant:brilliant_scientist_krg_value.percent_scale }
	clamp_variable = { var = brilliant_scientist_global_submission_score min = constant:brilliant_scientist_krg_value.zero max = constant:brilliant_scientist_krg_value.percent_scale }
	clamp_variable = { var = brilliant_scientist_world_control_ratio min = constant:brilliant_scientist_krg_value.zero max = constant:brilliant_scientist_krg_value.one }
	clamp_variable = { var = brilliant_scientist_integrated_state_ratio min = constant:brilliant_scientist_krg_value.zero max = constant:brilliant_scientist_krg_value.one }

	brilliant_scientist_refresh_world_threat_source = yes

# COUNTRY scope. This is an on-demand terminal/map audit, invoked only by
# deliberate administration or terminal-resolution decisions. It counts real
# control of currently owned world states, direct core integration, and living
# major non-subject powers; it is never called from a recurring on_action.
brilliant_scientist_krg_audit_terminal_world_state = {
	set_variable = { brilliant_scientist_krg_terminal_world_state_count = constant:brilliant_scientist_krg_value.zero }
	set_variable = { brilliant_scientist_krg_terminal_controlled_state_count = constant:brilliant_scientist_krg_value.zero }
	set_variable = { brilliant_scientist_krg_terminal_integrated_state_count = constant:brilliant_scientist_krg_value.zero }
	set_variable = { brilliant_scientist_major_opposition_count = constant:brilliant_scientist_krg_value.zero }
	every_state = {
		limit = { OWNER = { exists = yes } }
		ROOT = { add_to_variable = { brilliant_scientist_krg_terminal_world_state_count = constant:brilliant_scientist_krg_value.one } }
		if = {
			limit = {
				OR = {
					is_controlled_by = ROOT
					CONTROLLER = { is_subject_of = ROOT }
				}
			}
			ROOT = { add_to_variable = { brilliant_scientist_krg_terminal_controlled_state_count = constant:brilliant_scientist_krg_value.one } }
		}
		if = {
			limit = {
				is_controlled_by = ROOT
				is_core_of = ROOT
			}
			ROOT = { add_to_variable = { brilliant_scientist_krg_terminal_integrated_state_count = constant:brilliant_scientist_krg_value.one } }
		}
	}
	every_country = {
		limit = {
			exists = yes
			is_major = yes
			NOT = { tag = ROOT }
			NOT = { has_capitulated = yes }
			NOT = { is_subject_of = ROOT }
		}
		ROOT = { add_to_variable = { brilliant_scientist_major_opposition_count = constant:brilliant_scientist_krg_value.one } }
	}
	set_temp_variable = { brilliant_scientist_krg_terminal_control_ratio = brilliant_scientist_krg_terminal_controlled_state_count }
	set_temp_variable = { brilliant_scientist_krg_terminal_integration_ratio = brilliant_scientist_krg_terminal_integrated_state_count }
	if = {
		limit = { check_variable = { brilliant_scientist_krg_terminal_world_state_count > constant:brilliant_scientist_krg_value.zero } }
		divide_temp_variable = { brilliant_scientist_krg_terminal_control_ratio = brilliant_scientist_krg_terminal_world_state_count }
		divide_temp_variable = { brilliant_scientist_krg_terminal_integration_ratio = brilliant_scientist_krg_terminal_world_state_count }
	}
	else = {
		set_temp_variable = { brilliant_scientist_krg_terminal_control_ratio = constant:brilliant_scientist_krg_value.zero }
		set_temp_variable = { brilliant_scientist_krg_terminal_integration_ratio = constant:brilliant_scientist_krg_value.zero }
	}
	set_variable = { brilliant_scientist_world_control_ratio = brilliant_scientist_krg_terminal_control_ratio }
	set_variable = { brilliant_scientist_integrated_state_ratio = brilliant_scientist_krg_terminal_integration_ratio }
	set_country_flag = brilliant_scientist_krg_terminal_map_audit_completed
	if = {
		limit = { brilliant_scientist_krg_terminal_map_control_is_sufficient = yes }
		set_country_flag = brilliant_scientist_krg_terminal_map_audit_passed
	}
	else = { clr_country_flag = brilliant_scientist_krg_terminal_map_audit_passed }
	brilliant_scientist_refresh_world_threat_source = yes
}
