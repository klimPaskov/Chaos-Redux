# ============================================================================
# CHAOS REDUX - CBRN DIPLOMACY SCRIPTED EFFECTS
# ============================================================================
# Overview:
# - Implements exact-target inspection demands, exact-state forensic
#   publication, and exact-state international decontamination missions.
# - Uses model-aware CBRN family debit helpers and concrete support, truck,
#   and convoy debits.
# - Forensic publication advances only one recorded state action or pathogen
#   episode at a time. It never invents an actor or clears responsibility.
# - International cleanup reduces current contamination and medical pressure;
#   it never reverses recorded deaths, evidence, attribution, or use history.
# ============================================================================

# Country scope. Inputs:
# - event_target:cbrn_retaliation_target_country: exact proposed victim
# - cbrn_retaliation_action_proportionate_proof: supplied only for a verified
#   military-target response
# Output: cbrn_action_retaliation_status.
cbrn_diplomacy_classify_retaliation_action = {
	set_temp_variable = { cbrn_action_retaliation_status = constant:cbrn_retaliation_status.none }
	if = {
		limit = {
			has_event_target = cbrn_retaliation_target_country
			event_target:cbrn_retaliation_target_country = { exists = yes }
			has_war_with = event_target:cbrn_retaliation_target_country
			NOT = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.nerve_suppression compare = equals } }
			NOT = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.doomsday_release compare = equals } }
			NOT = { check_variable = { var = cbrn_action_severity value = constant:cbrn_operation_severity.doomsday compare = equals } }
		}
		event_target:cbrn_retaliation_target_country = {
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}
		if = {
			limit = { cbrn_country_has_active_retaliation_right_against_target_id = yes }
			set_temp_variable = { cbrn_action_retaliation_status = constant:cbrn_retaliation_status.authorized }
			if = {
				limit = {
					check_variable = {
						var = cbrn_retaliation_action_proportionate_proof
						value = constant:cbrn_proof.supplied
						compare = equals
					}
				}
				set_temp_variable = { cbrn_action_retaliation_status = constant:cbrn_retaliation_status.proportionate }
			}
			set_variable = { cbrn_last_retaliation_target = event_target:cbrn_retaliation_target_country }
			set_variable = { cbrn_last_retaliation_action_date = global.num_days }
			set_variable = { cbrn_last_retaliation_action_status = cbrn_action_retaliation_status }
		}
	}
}

# State scope. Classifies an exact deliberate biological seed without changing
# agent potency or native raid outcome odds.
cbrn_diplomacy_classify_biological_seed_retaliation = {
	set_temp_variable = { bio_seed_retaliation_status = constant:cbrn_retaliation_status.none }
	if = {
		limit = {
			check_variable = { var = bio_seed_actor_proof value = constant:bio_lifecycle_proof.supplied compare = equals }
			check_variable = { var = bio_seed_victim_proof value = constant:bio_lifecycle_proof.supplied compare = equals }
			check_variable = { var = bio_seed_source value = constant:bio_lifecycle_source.deliberate compare = equals }
			has_event_target = bio_seed_actor
			has_event_target = bio_seed_victim
			event_target:bio_seed_actor = { exists = yes }
			event_target:bio_seed_victim = { exists = yes }
		}
		event_target:bio_seed_victim = {
			save_event_target_as = cbrn_retaliation_target_country
		}
		set_temp_variable = { cbrn_retaliation_action_proportionate_proof = constant:cbrn_proof.missing }
		if = {
			limit = {
				check_variable = {
					var = bio_seed_route
					value = constant:bio_lifecycle_route.battlefield_dissemination
					compare = equals
				}
				CONTROLLER = {
					tag = event_target:bio_seed_victim
					divisions_in_state = {
						state = PREV
						size > constant:cbrn_math.zero
					}
				}
			}
			set_temp_variable = { cbrn_retaliation_action_proportionate_proof = constant:cbrn_proof.supplied }
		}
		event_target:bio_seed_actor = {
			cbrn_diplomacy_classify_retaliation_action = yes
			set_temp_variable = { bio_seed_retaliation_status = cbrn_action_retaliation_status }
		}
	}
}

# Country scope. The current country is the exact confirmed offender and the
# caller has supplied cbrn_retaliation_target_id.
cbrn_diplomacy_revoke_retaliation_right_against_target_id = {
	meta_effect = {
		text = {
			clear_variable = cbrn_retaliation_[TARGET_ID]_expiry_day
			set_variable = { cbrn_retaliation_[TARGET_ID]_revoked_day = global.num_days }
		}
		TARGET_ID = "[?cbrn_retaliation_target_id|.0]"
	}
}

# Country scope. The current country is the exact victim. Inputs are the
# confirmed offender id, action date, and CBRN source.
cbrn_diplomacy_grant_retaliation_right_against_offender = {
	set_temp_variable = { cbrn_retaliation_expiry_day = global.num_days }
	add_to_temp_variable = { cbrn_retaliation_expiry_day = constant:cbrn_diplomacy_duration.retaliation_window_days }
	meta_effect = {
		text = {
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_expiry_day = cbrn_retaliation_expiry_day }
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_first_use_date = cbrn_confirmed_use_action_date }
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_confirmed_day = global.num_days }
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_source = cbrn_retaliation_source_type }
		}
		OFFENDER_ID = "[?cbrn_retaliation_offender_id|.0]"
	}
	set_temp_variable = { cbrn_retaliation_flag_days = constant:cbrn_diplomacy_duration.retaliation_window_days }
	set_country_flag = {
		flag = cbrn_retaliation_authorized
		days = cbrn_retaliation_flag_days
	}
	set_country_flag = cbrn_confirmed_cbrn_attack_victim_history
}

# Country scope. Records the earliest confirmed action date in both directions
# and reconciles the bilateral first-user result. Same-day actions are treated
# as contested because the engine exposes only day precision; neither side
# receives retaliation mitigation from an unresolvable tie.
cbrn_diplomacy_record_confirmed_cbrn_use = {
	if = {
		limit = {
			has_event_target = cbrn_retaliation_target_country
			event_target:cbrn_retaliation_target_country = { exists = yes }
			NOT = { tag = event_target:cbrn_retaliation_target_country }
			has_variable = cbrn_confirmed_use_action_date
		}
		save_event_target_as = cbrn_retaliation_confirmed_offender
		set_temp_variable = { cbrn_retaliation_offender_id = id }
		event_target:cbrn_retaliation_target_country = {
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}

		meta_effect = {
			text = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable = cbrn_confirmed_use_against_[TARGET_ID]_first_action_date }
							check_variable = {
								var = cbrn_confirmed_use_action_date
								value = cbrn_confirmed_use_against_[TARGET_ID]_first_action_date
								compare = less_than
							}
						}
					}
					set_variable = { cbrn_confirmed_use_against_[TARGET_ID]_first_action_date = cbrn_confirmed_use_action_date }
				}
				set_temp_variable = { cbrn_pair_offender_first_action_date = cbrn_confirmed_use_against_[TARGET_ID]_first_action_date }
			}
			TARGET_ID = "[?cbrn_retaliation_target_id|.0]"
		}

		set_temp_variable = { cbrn_pair_victim_has_confirmed_use = constant:cbrn_proof.missing }
		event_target:cbrn_retaliation_target_country = {
			meta_effect = {
				text = {
					if = {
						limit = { has_variable = cbrn_confirmed_use_against_[OFFENDER_ID]_first_action_date }
						set_temp_variable = { cbrn_pair_victim_has_confirmed_use = constant:cbrn_proof.supplied }
						set_temp_variable = { cbrn_pair_victim_first_action_date = cbrn_confirmed_use_against_[OFFENDER_ID]_first_action_date }
					}
				}
				OFFENDER_ID = "[?cbrn_retaliation_offender_id|.0]"
			}
		}

		if = {
			limit = {
				OR = {
					check_variable = {
						var = cbrn_pair_victim_has_confirmed_use
						value = constant:cbrn_proof.missing
						compare = equals
					}
					check_variable = {
						var = cbrn_pair_offender_first_action_date
						value = cbrn_pair_victim_first_action_date
						compare = less_than
					}
				}
			}
			event_target:cbrn_retaliation_target_country = {
				cbrn_diplomacy_grant_retaliation_right_against_offender = yes
			}
			cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
			set_country_flag = cbrn_confirmed_cbrn_first_user_history
		}
		else_if = {
			limit = {
				check_variable = {
					var = cbrn_pair_victim_first_action_date
					value = cbrn_pair_offender_first_action_date
					compare = less_than
				}
			}
			event_target:cbrn_retaliation_target_country = {
				set_temp_variable = { cbrn_retaliation_target_id = cbrn_retaliation_offender_id }
				cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
			}
		}
		else = {
			cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
			event_target:cbrn_retaliation_target_country = {
				set_temp_variable = { cbrn_retaliation_target_id = cbrn_retaliation_offender_id }
				cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
				set_country_flag = cbrn_contested_same_day_first_use_history
			}
			set_country_flag = cbrn_contested_same_day_first_use_history
		}

		set_variable = { cbrn_last_confirmed_cbrn_use_date = cbrn_confirmed_use_action_date }
		set_variable = { cbrn_last_confirmed_cbrn_use_target = event_target:cbrn_retaliation_target_country }
		set_variable = { cbrn_last_confirmed_cbrn_use_source = cbrn_retaliation_source_type }
		if = {
			limit = { has_variable = cbrn_confirmed_use_retaliation_status }
			set_variable = { cbrn_last_confirmed_cbrn_use_retaliation_status = cbrn_confirmed_use_retaliation_status }
		}
	}
}

cbrn_diplomacy_pay_inspection_demand_cost = {
	set_temp_variable = { cbrn_hq_family_debit_requested = constant:cbrn_diplomacy_cost.inspection_instruments }
	cbrn_hq_debit_instrument_stock_oldest_first = yes
	add_to_variable = { cbrn_diplomacy_instruments_spent = cbrn_hq_family_debit_completed }

	set_temp_variable = { cbrn_diplomacy_equipment_debit = constant:cbrn_diplomacy_cost.inspection_support }
	multiply_temp_variable = { cbrn_diplomacy_equipment_debit = constant:cbrn_payload_math.negation_multiplier }
	add_equipment_to_stockpile = {
		type = support_equipment_1
		amount = cbrn_diplomacy_equipment_debit
	}
}

cbrn_diplomacy_send_inspection_demand = {
	if = {
		limit = {
			has_event_target = cbrn_diplomacy_target_country
			cbrn_country_can_pay_inspection_demand_cost = yes
			event_target:cbrn_diplomacy_target_country = {
				cbrn_target_can_receive_inspection_demand_from_root = yes
			}
		}
		cbrn_diplomacy_pay_inspection_demand_cost = yes
		add_to_variable = { cbrn_inspection_demands_issued = constant:cbrn_math.one }
		event_target:cbrn_diplomacy_target_country = {
			set_country_flag = cbrn_inspection_demand_active
			set_variable = { cbrn_inspection_demand_source = ROOT }
			set_variable = { cbrn_inspection_demand_date = global.num_days }
			add_opinion_modifier = {
				target = ROOT
				modifier = cbrn_inspection_demanded
			}
			activate_mission = cbrn_answer_inspection_demand_mission
		}
	}
}

# Country scope. Called whenever an accepted or refused inspection demand has
# been resolved. The timeout path sets a guard so it does not remove itself.
cbrn_diplomacy_clear_active_inspection_demand = {
	if = {
		limit = {
			has_active_mission = cbrn_answer_inspection_demand_mission
			NOT = { has_country_flag = cbrn_inspection_demand_timeout_resolving }
		}
		remove_mission = cbrn_answer_inspection_demand_mission
	}
	clr_country_flag = cbrn_inspection_demand_active
	clear_variable = cbrn_inspection_demand_source
	clear_variable = cbrn_inspection_demand_date
}

cbrn_diplomacy_resolve_inspection_demand_timeout = {
	set_country_flag = cbrn_inspection_demand_timeout_resolving
	if = {
		limit = {
			has_variable = cbrn_inspection_demand_source
			var:cbrn_inspection_demand_source = { exists = yes }
		}
		condemnation_refuse_inspections = yes
	}
	else = {
		cbrn_diplomacy_clear_active_inspection_demand = yes
	}
	clr_country_flag = cbrn_inspection_demand_timeout_resolving
}

# State scope. Chooses the strongest eligible exact state record. Ties keep the
# already selected record; weapon potency never affects this choice.
cbrn_diplomacy_select_forensic_record = {
	set_temp_variable = { cbrn_diplomacy_selected_source = constant:cbrn_diplomacy_source.none }
	set_temp_variable = { cbrn_diplomacy_selected_agent = constant:bio_lifecycle_agent.none }
	set_temp_variable = { cbrn_diplomacy_selected_evidence = constant:cbrn_math.zero }

	if = {
		limit = { cbrn_state_has_shareable_chemical_evidence = yes }
		set_temp_variable = { cbrn_diplomacy_selected_source = constant:cbrn_diplomacy_source.chemical }
		set_temp_variable = { cbrn_diplomacy_selected_evidence = cbrn_evidence_quality }
	}
	if = {
		limit = {
			cbrn_state_has_shareable_anthrax_evidence = yes
			check_variable = {
				var = bio_agent_1_evidence
				value = cbrn_diplomacy_selected_evidence
				compare = greater_than
			}
		}
		set_temp_variable = { cbrn_diplomacy_selected_source = constant:cbrn_diplomacy_source.biological }
		set_temp_variable = { cbrn_diplomacy_selected_agent = constant:bio_lifecycle_agent.anthrax }
		set_temp_variable = { cbrn_diplomacy_selected_evidence = bio_agent_1_evidence }
	}
	if = {
		limit = {
			cbrn_state_has_shareable_plague_evidence = yes
			check_variable = {
				var = bio_agent_2_evidence
				value = cbrn_diplomacy_selected_evidence
				compare = greater_than
			}
		}
		set_temp_variable = { cbrn_diplomacy_selected_source = constant:cbrn_diplomacy_source.biological }
		set_temp_variable = { cbrn_diplomacy_selected_agent = constant:bio_lifecycle_agent.plague }
		set_temp_variable = { cbrn_diplomacy_selected_evidence = bio_agent_2_evidence }
	}
	if = {
		limit = {
			cbrn_state_has_shareable_tularemia_evidence = yes
			check_variable = {
				var = bio_agent_3_evidence
				value = cbrn_diplomacy_selected_evidence
				compare = greater_than
			}
		}
		set_temp_variable = { cbrn_diplomacy_selected_source = constant:cbrn_diplomacy_source.biological }
		set_temp_variable = { cbrn_diplomacy_selected_agent = constant:bio_lifecycle_agent.tularemia }
		set_temp_variable = { cbrn_diplomacy_selected_evidence = bio_agent_3_evidence }
	}
	if = {
		limit = {
			cbrn_state_has_shareable_smallpox_evidence = yes
			check_variable = {
				var = bio_agent_4_evidence
				value = cbrn_diplomacy_selected_evidence
				compare = greater_than
			}
		}
		set_temp_variable = { cbrn_diplomacy_selected_source = constant:cbrn_diplomacy_source.biological }
		set_temp_variable = { cbrn_diplomacy_selected_agent = constant:bio_lifecycle_agent.smallpox }
		set_temp_variable = { cbrn_diplomacy_selected_evidence = bio_agent_4_evidence }
	}
}

# State scope. Exposes only the unpaid liability stored by this exact chemical
# action. The actor's aggregate hidden bucket is only the settlement account;
# it never determines the amount released.
cbrn_diplomacy_expose_recorded_chemical_responsibility = {
	set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_math.zero }
	if = {
		limit = {
			check_variable = {
				var = cbrn_diplomacy_published_attribution
				value = constant:cbrn_attribution_state.confirmed
				compare = equals
			}
		}
		set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_diplomacy_evidence.confirmed_hidden_exposure }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cbrn_diplomacy_published_attribution
				value = constant:cbrn_attribution_state.probable
				compare = equals
			}
		}
		set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_diplomacy_evidence.probable_hidden_exposure }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cbrn_diplomacy_published_attribution
				value = constant:cbrn_attribution_state.suspected
				compare = equals
			}
		}
		set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_diplomacy_evidence.suspected_hidden_exposure }
	}

	if = {
		limit = {
			has_variable = cbrn_last_chemical_actor
			var:cbrn_last_chemical_actor = { exists = yes }
			has_variable = cbrn_last_chemical_condemnation_total
			has_variable = cbrn_last_chemical_condemnation_paid
			has_variable = cbrn_last_chemical_condemnation_unpaid
			check_variable = { cbrn_last_chemical_condemnation_unpaid > constant:cbrn_math.zero }
		}
		set_temp_variable = { cbrn_diplomacy_desired_paid_liability = cbrn_last_chemical_condemnation_total }
		multiply_temp_variable = { cbrn_diplomacy_desired_paid_liability = cbrn_diplomacy_target_paid_share }
		set_temp_variable = { cbrn_diplomacy_exact_liability_delta = cbrn_diplomacy_desired_paid_liability }
		subtract_from_temp_variable = { cbrn_diplomacy_exact_liability_delta = cbrn_last_chemical_condemnation_paid }
		clamp_temp_variable = {
			var = cbrn_diplomacy_exact_liability_delta
			min = constant:cbrn_math.zero
			max = cbrn_last_chemical_condemnation_unpaid
		}
		set_temp_variable = { cbrn_diplomacy_exact_relief_delta = constant:cbrn_math.zero }
		if = {
			limit = {
				has_variable = cbrn_last_chemical_retaliation_relief_total
				has_variable = cbrn_last_chemical_retaliation_relief_paid
				has_variable = cbrn_last_chemical_retaliation_relief_unpaid
			}
			set_temp_variable = { cbrn_diplomacy_desired_paid_relief = cbrn_last_chemical_retaliation_relief_total }
			multiply_temp_variable = { cbrn_diplomacy_desired_paid_relief = cbrn_diplomacy_target_paid_share }
			set_temp_variable = { cbrn_diplomacy_exact_relief_delta = cbrn_diplomacy_desired_paid_relief }
			subtract_from_temp_variable = { cbrn_diplomacy_exact_relief_delta = cbrn_last_chemical_retaliation_relief_paid }
			clamp_temp_variable = {
				var = cbrn_diplomacy_exact_relief_delta
				min = constant:cbrn_math.zero
				max = cbrn_last_chemical_retaliation_relief_unpaid
			}
		}
		var:cbrn_last_chemical_actor = {
			set_temp_variable = { condemnation_source_type = constant:condemnation_source.chemical }
			set_temp_variable = { condemnation_context_type = constant:condemnation_context.forensic_disclosure }
			set_temp_variable = { condemnation_exact_hidden_exposure_amount = cbrn_diplomacy_exact_liability_delta }
			set_temp_variable = { condemnation_exact_hidden_retaliation_relief_amount = cbrn_diplomacy_exact_relief_delta }
			condemnation_expose_exact_hidden_source_amount = yes
			condemnation_recalculate_hidden_total = yes
			condemnation_recalculate_total = yes
			clamp_variable = {
				var = condemnation_recent_gain
				min = constant:cbrn_math.zero
				max = constant:condemnation_threshold.recent_gain_cap
			}
			if = {
				limit = { check_variable = { condemnation_exact_hidden_exposed_gain > constant:cbrn_math.zero } }
				set_temp_variable = { condemnation_recent_gain_days = constant:condemnation_duration_days.recent_gain }
				set_country_flag = {
					flag = condemnation_recent_gain_visible
					days = condemnation_recent_gain_days
				}
			}
			condemnation_refresh_tier = yes
			condemnation_start_targeted_pulse = yes
		}
		if = {
			limit = { check_variable = { condemnation_exact_hidden_exposed_gain > constant:cbrn_math.zero } }
			add_to_variable = { cbrn_last_chemical_condemnation_paid = condemnation_exact_hidden_exposed_gain }
			subtract_from_variable = { cbrn_last_chemical_condemnation_unpaid = condemnation_exact_hidden_exposed_gain }
			clamp_variable = {
				var = cbrn_last_chemical_condemnation_unpaid
				min = constant:cbrn_math.zero
				max = cbrn_last_chemical_condemnation_total
			}
			set_variable = { cbrn_last_chemical_condemnation_paid_attribution = cbrn_diplomacy_published_attribution }
			set_variable = { cbrn_last_chemical_condemnation_last_exposure_date = global.num_days }
			if = {
				limit = { check_variable = { condemnation_exact_hidden_retaliation_relief_exposed > constant:cbrn_math.zero } }
				add_to_variable = { cbrn_last_chemical_retaliation_relief_paid = condemnation_exact_hidden_retaliation_relief_exposed }
				subtract_from_variable = { cbrn_last_chemical_retaliation_relief_unpaid = condemnation_exact_hidden_retaliation_relief_exposed }
				clamp_variable = {
					var = cbrn_last_chemical_retaliation_relief_unpaid
					min = constant:cbrn_math.zero
					max = cbrn_last_chemical_retaliation_relief_total
				}
			}
		}
	}
}

# State scope. Registers an exact chemical action when later evidence, rather
# than the initial dispatch, first reaches confirmed attribution.
cbrn_diplomacy_register_newly_confirmed_chemical_record = {
	if = {
		limit = {
			check_variable = {
				var = cbrn_state_attribution_output
				value = constant:cbrn_attribution_state.confirmed
				compare = equals
			}
			NOT = { has_state_flag = cbrn_last_chemical_confirmed_use_registered }
			has_variable = cbrn_last_chemical_actor
			var:cbrn_last_chemical_actor = { exists = yes }
			has_variable = cbrn_last_chemical_victim_country
			var:cbrn_last_chemical_victim_country = { exists = yes }
			has_variable = cbrn_last_chemical_action_date
			has_variable = cbrn_last_chemical_retaliation_status
		}
		var:cbrn_last_chemical_victim_country = {
			save_event_target_as = cbrn_retaliation_target_country
			set_country_flag = cbrn_confirmed_chemical_attack_victim_history
		}
		set_temp_variable = { cbrn_confirmed_use_action_date = cbrn_last_chemical_action_date }
		set_temp_variable = { cbrn_confirmed_use_retaliation_status = cbrn_last_chemical_retaliation_status }
		set_temp_variable = { cbrn_retaliation_source_type = constant:cbrn_diplomacy_source.chemical }
		var:cbrn_last_chemical_actor = {
			set_country_flag = cbrn_confirmed_chemical_use_history
			set_country_flag = used_unconventional_weapon
			air_cleanliness_treaty_on_unconventional_use = yes
			cbrn_diplomacy_record_confirmed_cbrn_use = yes
		}
		set_state_flag = cbrn_confirmed_chemical_use_history
		set_state_flag = cbrn_last_chemical_confirmed_use_registered
		set_global_flag = cbrn_world_confirmed_chemical_use
	}
}

# State scope. Advances exactly one chosen action/episode and marks its action

				var = cbrn_diplomacy_selected_source

		cbrn_apply_state_evidence_delta_internal = yes

				compare = equals

# State scope. The exact provider is read from the state ledger. Cleanup uses
# current severity and applies only actual contamination still present.
cbrn_diplomacy_complete_international_decon_mission = {
	if = {
		limit = {
			has_state_flag = cbrn_international_decon_mission_active
			has_variable = cbrn_international_decon_provider
			var:cbrn_international_decon_provider = { exists = yes }
			has_variable = cbrn_international_decon_recipient
			var:cbrn_international_decon_recipient = { exists = yes }
			CONTROLLER = { tag = var:PREV.cbrn_international_decon_recipient }
			cbrn_state_has_chemical_contamination = yes
		}
		var:cbrn_international_decon_provider = {
			save_event_target_as = cbrn_diplomacy_decon_provider
		}
		cbrn_refresh_state_contamination_class = yes
		set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.trace_or_local_cleanup }
		if = {
			limit = {
				check_variable = {
					var = cbrn_state_contamination_class_output
					value = constant:cbrn_contamination_class.catastrophic
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.catastrophic_cleanup }
		}
		else_if = {
			limit = {
				check_variable = {
					var = cbrn_state_contamination_class_output
					value = constant:cbrn_contamination_class.severe
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.severe_cleanup }
		}
		else_if = {
			limit = {
				check_variable = {
					var = cbrn_state_contamination_class_output
					value = constant:cbrn_contamination_class.serious
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.serious_cleanup }
		}
		if = {
			limit = {
				event_target:cbrn_diplomacy_decon_provider = {
					cbrn_designer_has_international_medical_mission = yes
				}
			}
			multiply_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.international_medical_mission_mult }
		}

		set_temp_variable = { cbrn_state_contamination_delta = cbrn_diplomacy_cleanup_points }
		multiply_temp_variable = { cbrn_state_contamination_delta = constant:cbrn_payload_math.negation_multiplier }
		cbrn_apply_state_contamination_delta_internal = yes
		set_temp_variable = { cbrn_diplomacy_cleanup_completed = cbrn_state_previous_contamination_value }
		subtract_from_temp_variable = { cbrn_diplomacy_cleanup_completed = cbrn_state_new_contamination_value }

		if = {
			limit = { has_variable = cbrn_medical_saturation }
			subtract_from_variable = {
				cbrn_medical_saturation = constant:cbrn_diplomacy_decon.medical_saturation_reduction
			}
			clamp_variable = {
				var = cbrn_medical_saturation
				min = constant:cbrn_math.zero
				max = constant:cbrn_chemical_dispatch.state_medical_maximum
			}
		}

		if = {
			limit = { cbrn_state_has_observable_chemical_evidence = yes }
			set_temp_variable = { cbrn_state_evidence_delta = constant:cbrn_diplomacy_evidence.decon_observer_add }
			cbrn_apply_state_evidence_delta_internal = yes
			cbrn_diplomacy_register_newly_confirmed_chemical_record = yes
			set_temp_variable = { cbrn_diplomacy_published_attribution = cbrn_state_attribution_output }
			cbrn_diplomacy_expose_recorded_chemical_responsibility = yes
		}

		event_target:cbrn_diplomacy_decon_provider = {
			add_to_variable = { cbrn_international_decon_missions_completed = constant:cbrn_math.one }
			add_to_variable = { cbrn_international_decon_cleanup_total = cbrn_diplomacy_cleanup_completed }
		}
		CONTROLLER = {
			add_to_variable = { cbrn_international_decon_aid_received = cbrn_diplomacy_cleanup_completed }
			add_opinion_modifier = {
				target = event_target:cbrn_diplomacy_decon_provider
				modifier = cbrn_international_decontamination_aid
			}
		}
		set_state_flag = cbrn_international_decon_mission_history
	}

	clr_state_flag = cbrn_international_decon_mission_active
	set_temp_variable = { cbrn_diplomacy_state_cooldown_days = constant:cbrn_diplomacy_duration.decon_state_cooldown_days }
	set_state_flag = {
		flag = cbrn_international_decon_mission_cooldown
		days = cbrn_diplomacy_state_cooldown_days
	}
	clear_variable = cbrn_international_decon_provider
	clear_variable = cbrn_international_decon_recipient
	clear_variable = cbrn_international_decon_start_date
}

cbrn_diplomacy_cancel_international_decon_mission = {
	clr_state_flag = cbrn_international_decon_mission_active
	clear_variable = cbrn_international_decon_provider
	clear_variable = cbrn_international_decon_recipient
	clear_variable = cbrn_international_decon_start_date
}
