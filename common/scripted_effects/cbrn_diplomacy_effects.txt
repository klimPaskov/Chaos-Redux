# Some fields in this file reject script-constant tokens.
# These file-scoped values mirror the authoritative shared constants.
@CR_SC_CBRN_MATH_ZERO = 0
# ============================================================================
# CHAOS REDUX - CBRN DIPLOMACY SCRIPTED EFFECTS
# ============================================================================
# Overview:
# - Implements exact-target inspection demands, exact-state forensic
#   publication, and exact-state international decontamination missions.
# - Uses model-aware CBRN family debit helpers and concrete support, truck,
#   and convoy debits.
# - Forensic publication advances only one recorded state action or pathogen
#   episode at a time. It never invents an actor or clears responsibility.
# - International cleanup reduces current contamination and medical pressure;
#   it never reverses recorded deaths, evidence, attribution, or use history.
# ============================================================================

# Country scope. Inputs:
# - event_target:cbrn_retaliation_target_country: exact proposed victim
# - cbrn_retaliation_action_proportionate_proof: supplied only for a verified
#   military-target response
# Output: cbrn_action_retaliation_status.
cbrn_diplomacy_classify_retaliation_action = {
	set_temp_variable = { cbrn_action_retaliation_status = constant:cbrn_retaliation_status.none }
	if = {
		limit = {
			has_event_target = cbrn_retaliation_target_country
			event_target:cbrn_retaliation_target_country = { exists = yes }
			has_war_with = event_target:cbrn_retaliation_target_country
			NOT = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.nerve_suppression compare = equals } }
			NOT = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.doomsday_release compare = equals } }
			NOT = { check_variable = { var = cbrn_action_severity value = constant:cbrn_operation_severity.doomsday compare = equals } }
		}
		event_target:cbrn_retaliation_target_country = {
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}
		if = {
			limit = { cbrn_country_has_active_retaliation_right_against_target_id = yes }
			set_temp_variable = { cbrn_action_retaliation_status = constant:cbrn_retaliation_status.authorized }
			if = {
				limit = {
					check_variable = {
						var = cbrn_retaliation_action_proportionate_proof
						value = constant:cbrn_proof.supplied
						compare = equals
					}
				}
				set_temp_variable = { cbrn_action_retaliation_status = constant:cbrn_retaliation_status.proportionate }
			}
			set_variable = { cbrn_last_retaliation_target = event_target:cbrn_retaliation_target_country }
			set_variable = { cbrn_last_retaliation_action_date = global.num_days }
			set_variable = { cbrn_last_retaliation_action_status = cbrn_action_retaliation_status }
		}
	}
}

# State scope. Classifies an exact deliberate biological seed without changing
# agent potency or native raid outcome odds.
cbrn_diplomacy_classify_biological_seed_retaliation = {
	set_temp_variable = { bio_seed_retaliation_status = constant:cbrn_retaliation_status.none }
	if = {
		limit = {
			check_variable = { var = bio_seed_actor_proof value = constant:bio_lifecycle_proof.supplied compare = equals }
			check_variable = { var = bio_seed_victim_proof value = constant:bio_lifecycle_proof.supplied compare = equals }
			check_variable = { var = bio_seed_source value = constant:bio_lifecycle_source.deliberate compare = equals }
			has_event_target = bio_seed_actor
			has_event_target = bio_seed_victim
			event_target:bio_seed_actor = { exists = yes }
			event_target:bio_seed_victim = { exists = yes }
		}
		event_target:bio_seed_victim = {
			save_event_target_as = cbrn_retaliation_target_country
		}
		set_temp_variable = { cbrn_retaliation_action_proportionate_proof = constant:cbrn_proof.missing }
		if = {
			limit = {
				check_variable = {
					var = bio_seed_route
					value = constant:bio_lifecycle_route.battlefield_dissemination
					compare = equals
				}
				CONTROLLER = {
					tag = event_target:bio_seed_victim
					divisions_in_state = {
						state = PREV
						size > @CR_SC_CBRN_MATH_ZERO
					}
				}
			}
			set_temp_variable = { cbrn_retaliation_action_proportionate_proof = constant:cbrn_proof.supplied }
		}
		event_target:bio_seed_actor = {
			cbrn_diplomacy_classify_retaliation_action = yes
			set_temp_variable = { bio_seed_retaliation_status = cbrn_action_retaliation_status }
		}
	}
}

# Country scope. The current country is the exact confirmed offender and the
# caller has supplied cbrn_retaliation_target_id.
cbrn_diplomacy_revoke_retaliation_right_against_target_id = {
	meta_effect = {
		text = {
			clear_variable = cbrn_retaliation_[TARGET_ID]_expiry_day
			set_variable = { cbrn_retaliation_[TARGET_ID]_revoked_day = global.num_days }
		}
		TARGET_ID = "[?cbrn_retaliation_target_id|.0]"
	}
}

# Country scope. The current country is the exact victim. Inputs are the
# confirmed offender id, action date, and CBRN source.
cbrn_diplomacy_grant_retaliation_right_against_offender = {
	set_temp_variable = { cbrn_retaliation_expiry_day = global.num_days }
	add_to_temp_variable = { cbrn_retaliation_expiry_day = constant:cbrn_diplomacy_duration.retaliation_window_days }
	meta_effect = {
		text = {
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_expiry_day = cbrn_retaliation_expiry_day }
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_first_use_date = cbrn_confirmed_use_action_date }
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_confirmed_day = global.num_days }
			set_variable = { cbrn_retaliation_[OFFENDER_ID]_source = cbrn_retaliation_source_type }
		}
		OFFENDER_ID = "[?cbrn_retaliation_offender_id|.0]"
	}
	set_temp_variable = { cbrn_retaliation_flag_days = constant:cbrn_diplomacy_duration.retaliation_window_days }
	set_country_flag = {
		flag = cbrn_retaliation_authorized
		days = var:cbrn_retaliation_flag_days
	}
	set_country_flag = cbrn_confirmed_cbrn_attack_victim_history
}

# Country scope. Records the earliest confirmed action date in both directions
# and reconciles the bilateral first-user result. Same-day actions are treated
# as contested because the engine exposes only day precision; neither side
# receives retaliation mitigation from an unresolvable tie.
cbrn_diplomacy_record_confirmed_cbrn_use = {
	if = {
		limit = {
			has_event_target = cbrn_retaliation_target_country
			event_target:cbrn_retaliation_target_country = { exists = yes }
			NOT = { tag = event_target:cbrn_retaliation_target_country }
			has_variable = cbrn_confirmed_use_action_date
		}
		save_event_target_as = cbrn_retaliation_confirmed_offender
		set_temp_variable = { cbrn_retaliation_offender_id = id }
		event_target:cbrn_retaliation_target_country = {
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}

		meta_effect = {
			text = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable = cbrn_confirmed_use_against_[TARGET_ID]_first_action_date }
							check_variable = {
								var = cbrn_confirmed_use_action_date
								value = cbrn_confirmed_use_against_[TARGET_ID]_first_action_date
								compare = less_than
							}
						}
					}
					set_variable = { cbrn_confirmed_use_against_[TARGET_ID]_first_action_date = cbrn_confirmed_use_action_date }
				}
				set_temp_variable = { cbrn_pair_offender_first_action_date = cbrn_confirmed_use_against_[TARGET_ID]_first_action_date }
			}
			TARGET_ID = "[?cbrn_retaliation_target_id|.0]"
		}

		set_temp_variable = { cbrn_pair_victim_has_confirmed_use = constant:cbrn_proof.missing }
		event_target:cbrn_retaliation_target_country = {
			meta_effect = {
				text = {
					if = {
						limit = { has_variable = cbrn_confirmed_use_against_[OFFENDER_ID]_first_action_date }
						set_temp_variable = { cbrn_pair_victim_has_confirmed_use = constant:cbrn_proof.supplied }
						set_temp_variable = { cbrn_pair_victim_first_action_date = cbrn_confirmed_use_against_[OFFENDER_ID]_first_action_date }
					}
				}
				OFFENDER_ID = "[?cbrn_retaliation_offender_id|.0]"
			}
		}

		if = {
			limit = {
				OR = {
					check_variable = {
						var = cbrn_pair_victim_has_confirmed_use
						value = constant:cbrn_proof.missing
						compare = equals
					}
					check_variable = {
						var = cbrn_pair_offender_first_action_date
						value = cbrn_pair_victim_first_action_date
						compare = less_than
					}
				}
			}
			event_target:cbrn_retaliation_target_country = {
				cbrn_diplomacy_grant_retaliation_right_against_offender = yes
			}
			cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
			set_country_flag = cbrn_confirmed_cbrn_first_user_history
		}
		else_if = {
			limit = {
				check_variable = {
					var = cbrn_pair_victim_first_action_date
					value = cbrn_pair_offender_first_action_date
					compare = less_than
				}
			}
			event_target:cbrn_retaliation_target_country = {
				set_temp_variable = { cbrn_retaliation_target_id = cbrn_retaliation_offender_id }
				cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
			}
		}
		else = {
			cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
			event_target:cbrn_retaliation_target_country = {
				set_temp_variable = { cbrn_retaliation_target_id = cbrn_retaliation_offender_id }
				cbrn_diplomacy_revoke_retaliation_right_against_target_id = yes
				set_country_flag = cbrn_contested_same_day_first_use_history
			}
			set_country_flag = cbrn_contested_same_day_first_use_history
		}

		set_variable = { cbrn_last_confirmed_cbrn_use_date = cbrn_confirmed_use_action_date }
		set_variable = { cbrn_last_confirmed_cbrn_use_target = event_target:cbrn_retaliation_target_country }
		set_variable = { cbrn_last_confirmed_cbrn_use_source = cbrn_retaliation_source_type }
		if = {
			limit = { has_variable = cbrn_confirmed_use_retaliation_status }
			set_variable = { cbrn_last_confirmed_cbrn_use_retaliation_status = cbrn_confirmed_use_retaliation_status }
		}
	}
}

cbrn_diplomacy_pay_inspection_demand_cost = {
	set_temp_variable = { cbrn_hq_family_debit_requested = constant:cbrn_diplomacy_cost.inspection_instruments }
	cbrn_hq_debit_instrument_stock_oldest_first = yes
	add_to_variable = { cbrn_diplomacy_instruments_spent = cbrn_hq_family_debit_completed }

	set_temp_variable = { cbrn_diplomacy_equipment_debit = constant:cbrn_diplomacy_cost.inspection_support }
	multiply_temp_variable = { cbrn_diplomacy_equipment_debit = constant:cbrn_payload_math.negation_multiplier }
	add_equipment_to_stockpile = {
		type = support_equipment_1
		amount = cbrn_diplomacy_equipment_debit
	}
}

cbrn_diplomacy_send_inspection_demand = {
	if = {
		limit = {
			has_event_target = cbrn_diplomacy_target_country
			cbrn_country_can_pay_inspection_demand_cost = yes
			event_target:cbrn_diplomacy_target_country = {
				cbrn_target_can_receive_inspection_demand_from_root = yes
			}
		}
		cbrn_diplomacy_pay_inspection_demand_cost = yes
		add_to_variable = { cbrn_inspection_demands_issued = constant:cbrn_math.one }
		set_temp_variable = { cbrn_inspection_sender_cooldown_days = constant:cbrn_diplomacy_duration.inspection_sender_cooldown_days }
		set_country_flag = {
			flag = cbrn_inspection_demand_sender_cooldown
			days = var:cbrn_inspection_sender_cooldown_days
		}
		event_target:cbrn_diplomacy_target_country = {
			set_country_flag = cbrn_inspection_demand_active
			set_variable = { cbrn_inspection_demand_source = ROOT }
			set_variable = { cbrn_inspection_demand_date = global.num_days }
			add_opinion_modifier = {
				target = ROOT
				modifier = cbrn_inspection_demanded
			}
			activate_mission = cbrn_answer_inspection_demand_mission
		}
	}
}

# Country scope. Called whenever an accepted or refused inspection demand has
# been resolved. The timeout path sets a guard so it does not remove itself.
cbrn_diplomacy_clear_active_inspection_demand = {
	if = {
		limit = {
			has_active_mission = cbrn_answer_inspection_demand_mission
			NOT = { has_country_flag = cbrn_inspection_demand_timeout_resolving }
		}
		remove_mission = cbrn_answer_inspection_demand_mission
	}
	clr_country_flag = cbrn_inspection_demand_active
	clear_variable = cbrn_inspection_demand_source
	clear_variable = cbrn_inspection_demand_date
}

cbrn_diplomacy_resolve_inspection_demand_timeout = {
	set_country_flag = cbrn_inspection_demand_timeout_resolving
	if = {
		limit = {
			has_variable = cbrn_inspection_demand_source
			var:cbrn_inspection_demand_source = { exists = yes }
		}
		condemnation_refuse_inspections = yes
	}
	else = {
		cbrn_diplomacy_clear_active_inspection_demand = yes
	}
	clr_country_flag = cbrn_inspection_demand_timeout_resolving
}

# State scope. Appends one exact chemical-action row to parallel persistent
# arrays. Rows are never overwritten or deleted; only their evidence,
# attribution, settlement, confirmation, and status fields advance.
cbrn_diplomacy_append_chemical_action_record = {
	if = {
		limit = {
			has_event_target = cbrn_action_actor
			has_event_target = cbrn_action_victim_country
		}
		if = {
			limit = { NOT = { has_variable = global.cbrn_chemical_action_uid_next } }
			set_variable = { global.cbrn_chemical_action_uid_next = @CR_SC_CBRN_MATH_ZERO }
		}
		add_to_variable = { global.cbrn_chemical_action_uid_next = constant:cbrn_math.one }
		set_temp_variable = { cbrn_chemical_record_uid = global.cbrn_chemical_action_uid_next }
		save_event_target_as = cbrn_diplomacy_action_record_state
		event_target:cbrn_action_actor = {
			set_variable = { cbrn_last_chemical_record_uid = cbrn_chemical_record_uid }
			add_to_array = {
				array = cbrn_chemical_actor_record_state_entries
				value = event_target:cbrn_diplomacy_action_record_state
			}
			add_to_array = {
				array = cbrn_chemical_actor_record_uid_entries
				value = cbrn_chemical_record_uid
			}
		}
		set_temp_variable = { cbrn_chemical_record_evidence = cbrn_action_evidence_points }
		if = {
			limit = {
				check_variable = {
					var = cbrn_chemical_record_evidence
					value = cbrn_action_evidence_floor
					compare = less_than
				}
			}
			set_temp_variable = { cbrn_chemical_record_evidence = cbrn_action_evidence_floor }
		}
		clamp_temp_variable = {
			var = cbrn_chemical_record_evidence
			min = @CR_SC_CBRN_MATH_ZERO
			max = constant:cbrn_evidence_threshold.maximum
		}
		set_temp_variable = { cbrn_chemical_record_status_value = constant:cbrn_chemical_record_status.settled }
		if = {
			limit = {
				check_variable = { cbrn_chemical_hidden_liability_delta > @CR_SC_CBRN_MATH_ZERO }
			}
			set_temp_variable = { cbrn_chemical_record_status_value = constant:cbrn_chemical_record_status.unavailable }
			if = {
				limit = {
					check_variable = {
						var = cbrn_chemical_record_evidence
						value = constant:cbrn_diplomacy_evidence.chemical_sample_minimum
						compare = greater_than_or_equals
					}
				}
				set_temp_variable = { cbrn_chemical_record_status_value = constant:cbrn_chemical_record_status.open }
				if = {
					limit = { NOT = { has_variable = cbrn_chemical_open_forensic_record_count } }
					set_variable = { cbrn_chemical_open_forensic_record_count = @CR_SC_CBRN_MATH_ZERO }
				}
				add_to_variable = { cbrn_chemical_open_forensic_record_count = constant:cbrn_math.one }
			}
		}
		set_temp_variable = { cbrn_chemical_record_confirmed_registered = @CR_SC_CBRN_MATH_ZERO }
		if = {
			limit = {
				check_variable = {
					var = cbrn_action_release_status
					value = constant:cbrn_chemical_release_status.released
					compare = equals
				}
				check_variable = {
					var = cbrn_action_attribution_state
					value = constant:cbrn_attribution_state.confirmed
					compare = equals
				}
			}
			set_temp_variable = { cbrn_chemical_record_confirmed_registered = constant:cbrn_math.one }
		}

		add_to_array = { array = cbrn_chemical_record_uid_entries value = cbrn_chemical_record_uid }
		add_to_array = { array = cbrn_chemical_record_actor_entries value = event_target:cbrn_action_actor }
		add_to_array = { array = cbrn_chemical_record_victim_entries value = event_target:cbrn_action_victim_country }
		add_to_array = { array = cbrn_chemical_record_action_date_entries value = cbrn_dispatch_current_day }
		add_to_array = { array = cbrn_chemical_record_agent_entries value = cbrn_action_agent }
		add_to_array = { array = cbrn_chemical_record_agent_class_entries value = cbrn_action_agent_class }
		add_to_array = { array = cbrn_chemical_record_delivery_route_entries value = cbrn_action_delivery_route }
		add_to_array = { array = cbrn_chemical_record_severity_entries value = cbrn_action_severity }
		add_to_array = { array = cbrn_chemical_record_source_label_entries value = cbrn_action_source_label }
		add_to_array = { array = cbrn_chemical_record_release_status_entries value = cbrn_action_release_status }
		add_to_array = { array = cbrn_chemical_record_raid_result_entries value = cbrn_action_raid_result }
		add_to_array = { array = cbrn_chemical_record_civilian_death_entries value = cbrn_action_civilian_deaths }
		add_to_array = { array = cbrn_chemical_record_contamination_entries value = cbrn_action_contamination_points }
		add_to_array = { array = cbrn_chemical_record_medical_entries value = cbrn_action_medical_saturation_points }
		add_to_array = { array = cbrn_chemical_record_evidence_entries value = cbrn_chemical_record_evidence }
		add_to_array = { array = cbrn_chemical_record_attribution_entries value = cbrn_action_attribution_state }
		add_to_array = { array = cbrn_chemical_record_retaliation_status_entries value = cbrn_action_retaliation_status }
		add_to_array = { array = cbrn_chemical_record_condemnation_total_entries value = cbrn_chemical_total_liability_delta }
		add_to_array = { array = cbrn_chemical_record_condemnation_paid_entries value = cbrn_chemical_public_liability_delta }
		add_to_array = { array = cbrn_chemical_record_condemnation_unpaid_entries value = cbrn_chemical_hidden_liability_delta }
		add_to_array = { array = cbrn_chemical_record_relief_total_entries value = cbrn_chemical_total_relief_delta }
		add_to_array = { array = cbrn_chemical_record_relief_paid_entries value = cbrn_chemical_public_relief_delta }
		add_to_array = { array = cbrn_chemical_record_relief_unpaid_entries value = cbrn_chemical_hidden_relief_delta }
		add_to_array = { array = cbrn_chemical_record_status_entries value = cbrn_chemical_record_status_value }
		add_to_array = { array = cbrn_chemical_record_confirmed_registered_entries value = cbrn_chemical_record_confirmed_registered }
	}
}

# State scope. Applies one actor-level inspection or observer disclosure
# fraction to the exact UID row. It changes only that row's paid/unpaid
# settlement fields and returns the actual reconciled amounts in temporary
# accumulator variables owned by the calling country chain.
cbrn_diplomacy_reconcile_exact_row_generic_exposure = {
	set_temp_variable = { cbrn_diplomacy_reconcile_row = @CR_SC_CBRN_MATH_ZERO }
	set_temp_variable = { cbrn_diplomacy_reconcile_row_found = @CR_SC_CBRN_MATH_ZERO }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = cbrn_diplomacy_reconcile_row
				value = cbrn_chemical_record_uid_entries^num
				compare = less_than
			}
			check_variable = {
				var = cbrn_diplomacy_reconcile_row_found
				value = @CR_SC_CBRN_MATH_ZERO
				compare = equals
			}
		}
		if = {
			limit = {
				check_variable = {
					var = cbrn_chemical_record_uid_entries^cbrn_diplomacy_reconcile_row
					value = cbrn_diplomacy_reconcile_uid
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_reconcile_row_found = constant:cbrn_math.one }
			set_temp_variable = {
				cbrn_diplomacy_reconcile_liability = cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_reconcile_row
			}
			multiply_temp_variable = {
				cbrn_diplomacy_reconcile_liability = cbrn_diplomacy_generic_exposure_fraction
			}
			clamp_temp_variable = {
				var = cbrn_diplomacy_reconcile_liability
				min = @CR_SC_CBRN_MATH_ZERO
				max = cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_reconcile_row
			}
			clamp_temp_variable = {
				var = cbrn_diplomacy_reconcile_liability
				min = @CR_SC_CBRN_MATH_ZERO
				max = cbrn_diplomacy_generic_liability_remaining
			}
			add_to_variable = {
				cbrn_chemical_record_condemnation_paid_entries^cbrn_diplomacy_reconcile_row = cbrn_diplomacy_reconcile_liability
			}
			subtract_from_variable = {
				cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_reconcile_row = cbrn_diplomacy_reconcile_liability
			}
			add_to_temp_variable = {
				cbrn_diplomacy_generic_exposed_liability = cbrn_diplomacy_reconcile_liability
			}
			subtract_from_temp_variable = {
				cbrn_diplomacy_generic_liability_remaining = cbrn_diplomacy_reconcile_liability
			}

			set_temp_variable = {
				cbrn_diplomacy_reconcile_relief = cbrn_chemical_record_relief_unpaid_entries^cbrn_diplomacy_reconcile_row
			}
			multiply_temp_variable = {
				cbrn_diplomacy_reconcile_relief = cbrn_diplomacy_generic_exposure_fraction
			}
			clamp_temp_variable = {
				var = cbrn_diplomacy_reconcile_relief
				min = @CR_SC_CBRN_MATH_ZERO
				max = cbrn_chemical_record_relief_unpaid_entries^cbrn_diplomacy_reconcile_row
			}
			clamp_temp_variable = {
				var = cbrn_diplomacy_reconcile_relief
				min = @CR_SC_CBRN_MATH_ZERO
				max = cbrn_diplomacy_generic_relief_remaining
			}
			add_to_variable = {
				cbrn_chemical_record_relief_paid_entries^cbrn_diplomacy_reconcile_row = cbrn_diplomacy_reconcile_relief
			}
			subtract_from_variable = {
				cbrn_chemical_record_relief_unpaid_entries^cbrn_diplomacy_reconcile_row = cbrn_diplomacy_reconcile_relief
			}
			add_to_temp_variable = {
				cbrn_diplomacy_generic_exposed_relief = cbrn_diplomacy_reconcile_relief
			}
			subtract_from_temp_variable = {
				cbrn_diplomacy_generic_relief_remaining = cbrn_diplomacy_reconcile_relief
			}

			if = {
				limit = {
					check_variable = {
						var = cbrn_chemical_record_status_entries^cbrn_diplomacy_reconcile_row
						value = constant:cbrn_chemical_record_status.open
						compare = equals
					}
					check_variable = {
						var = cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_reconcile_row
						value = @CR_SC_CBRN_MATH_ZERO
						compare = less_than_or_equals
					}
				}
				set_variable = {
					cbrn_chemical_record_status_entries^cbrn_diplomacy_reconcile_row = constant:cbrn_chemical_record_status.settled
				}
				subtract_from_variable = { cbrn_chemical_open_forensic_record_count = constant:cbrn_math.one }
				clamp_variable = {
					var = cbrn_chemical_open_forensic_record_count
					min = @CR_SC_CBRN_MATH_ZERO
					max = cbrn_chemical_record_status_entries^num
				}
			}
		}
		add_to_temp_variable = { cbrn_diplomacy_reconcile_row = constant:cbrn_math.one }
	}

# Country scope. Reconciles an existing targeted inspection/observer disclosure
# against every exact chemical action registered to this actor. The same
# fraction is applied independently to each row; unmatched aggregate liability
# remains hidden instead of being assigned to an inferred action.
cbrn_diplomacy_reconcile_actor_generic_chemical_exposure = {
	set_temp_variable = { cbrn_diplomacy_generic_exposed_liability = @CR_SC_CBRN_MATH_ZERO }
	set_temp_variable = { cbrn_diplomacy_generic_exposed_relief = @CR_SC_CBRN_MATH_ZERO }
	set_temp_variable = { cbrn_diplomacy_generic_liability_remaining = condemnation_hidden_chemical }
	multiply_temp_variable = {
		cbrn_diplomacy_generic_liability_remaining = cbrn_diplomacy_generic_exposure_fraction
	}
	set_temp_variable = {
		cbrn_diplomacy_generic_relief_remaining = condemnation_hidden_chemical_retaliation_relief
	}
	multiply_temp_variable = {
		cbrn_diplomacy_generic_relief_remaining = cbrn_diplomacy_generic_exposure_fraction
	}
	set_temp_variable = { cbrn_diplomacy_reconcile_actor_row = @CR_SC_CBRN_MATH_ZERO }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = cbrn_diplomacy_reconcile_actor_row
				value = cbrn_chemical_actor_record_state_entries^num
				compare = less_than
			}
			check_variable = {
				var = cbrn_diplomacy_reconcile_actor_row
				value = cbrn_chemical_actor_record_uid_entries^num
				compare = less_than
			}
			OR = {
				check_variable = { cbrn_diplomacy_generic_liability_remaining > @CR_SC_CBRN_MATH_ZERO }
				check_variable = { cbrn_diplomacy_generic_relief_remaining > @CR_SC_CBRN_MATH_ZERO }
			}
		}
		set_temp_variable = {
			cbrn_diplomacy_reconcile_uid = cbrn_chemical_actor_record_uid_entries^cbrn_diplomacy_reconcile_actor_row
		}
		var:cbrn_chemical_actor_record_state_entries^cbrn_diplomacy_reconcile_actor_row = {
			cbrn_diplomacy_reconcile_exact_row_generic_exposure = yes
		}

	}

					var = cbrn_diplomacy_selected_source

cbrn_diplomacy_classify_loaded_chemical_record = {

		limit = {

				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { cbrn_diplomacy_published_attribution = constant:cbrn_attribution_state.probable }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cbrn_diplomacy_record_evidence
				value = constant:cbrn_evidence_threshold.suspected
				compare = greater_than_or_equals
			}
		}

		cbrn_chemical_record_attribution_entries^cbrn_diplomacy_selected_chemical_row = cbrn_diplomacy_published_attribution
	}
}

# State scope. Exposes only the unpaid liability of the loaded exact row.
# A dead perpetrator remains historically identified, but no country effect is
# redirected to a successor or inferred proxy.
cbrn_diplomacy_expose_recorded_chemical_responsibility = {
	set_temp_variable = { cbrn_diplomacy_target_paid_share = @CR_SC_CBRN_MATH_ZERO }
	if = {
		limit = { check_variable = { var = cbrn_diplomacy_published_attribution value = constant:cbrn_attribution_state.confirmed compare = equals } }
		set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_diplomacy_evidence.confirmed_hidden_exposure }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_diplomacy_published_attribution value = constant:cbrn_attribution_state.probable compare = equals } }
		set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_diplomacy_evidence.probable_hidden_exposure }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_diplomacy_published_attribution value = constant:cbrn_attribution_state.suspected compare = equals } }
		set_temp_variable = { cbrn_diplomacy_target_paid_share = constant:cbrn_diplomacy_evidence.suspected_hidden_exposure }
	}

	set_temp_variable = { condemnation_exact_hidden_exposed_gain = @CR_SC_CBRN_MATH_ZERO }
	set_temp_variable = { condemnation_exact_hidden_retaliation_relief_exposed = @CR_SC_CBRN_MATH_ZERO }
	if = {
		limit = {
			check_variable = { cbrn_diplomacy_record_loaded > @CR_SC_CBRN_MATH_ZERO }
			check_variable = { cbrn_diplomacy_record_condemnation_unpaid > @CR_SC_CBRN_MATH_ZERO }
			has_event_target = cbrn_diplomacy_record_actor
			event_target:cbrn_diplomacy_record_actor = { exists = yes }
		}
		set_temp_variable = { cbrn_diplomacy_desired_paid_liability = cbrn_diplomacy_record_condemnation_total }
		multiply_temp_variable = { cbrn_diplomacy_desired_paid_liability = cbrn_diplomacy_target_paid_share }
		set_temp_variable = { cbrn_diplomacy_exact_liability_delta = cbrn_diplomacy_desired_paid_liability }
		subtract_from_temp_variable = { cbrn_diplomacy_exact_liability_delta = cbrn_diplomacy_record_condemnation_paid }
		clamp_temp_variable = { var = cbrn_diplomacy_exact_liability_delta min = @CR_SC_CBRN_MATH_ZERO max = cbrn_diplomacy_record_condemnation_unpaid }
		set_temp_variable = { cbrn_diplomacy_desired_paid_relief = cbrn_diplomacy_record_relief_total }
		multiply_temp_variable = { cbrn_diplomacy_desired_paid_relief = cbrn_diplomacy_target_paid_share }
		set_temp_variable = { cbrn_diplomacy_exact_relief_delta = cbrn_diplomacy_desired_paid_relief }
		subtract_from_temp_variable = { cbrn_diplomacy_exact_relief_delta = cbrn_diplomacy_record_relief_paid }
		clamp_temp_variable = { var = cbrn_diplomacy_exact_relief_delta min = @CR_SC_CBRN_MATH_ZERO max = cbrn_diplomacy_record_relief_unpaid }
		event_target:cbrn_diplomacy_record_actor = {
			set_temp_variable = { condemnation_source_type = constant:condemnation_source.chemical }
			set_temp_variable = { condemnation_context_type = constant:condemnation_context.forensic_disclosure }
			set_temp_variable = { condemnation_exact_hidden_exposure_amount = cbrn_diplomacy_exact_liability_delta }
			set_temp_variable = { condemnation_exact_hidden_retaliation_relief_amount = cbrn_diplomacy_exact_relief_delta }
			condemnation_expose_exact_hidden_source_amount = yes
			condemnation_recalculate_hidden_total = yes
			condemnation_recalculate_total = yes
			clamp_variable = { var = condemnation_recent_gain min = @CR_SC_CBRN_MATH_ZERO max = constant:condemnation_threshold.recent_gain_cap }
			if = {
				limit = { check_variable = { condemnation_exact_hidden_exposed_gain > @CR_SC_CBRN_MATH_ZERO } }
				set_temp_variable = { condemnation_recent_gain_days = constant:condemnation_duration_days.recent_gain }
				set_country_flag = { flag = condemnation_recent_gain_visible days = var:condemnation_recent_gain_days }
			}
			condemnation_refresh_tier = yes
			condemnation_start_targeted_pulse = yes
		}
		add_to_variable = {
			cbrn_chemical_record_condemnation_paid_entries^cbrn_diplomacy_selected_chemical_row = condemnation_exact_hidden_exposed_gain
		}
		subtract_from_variable = {
			cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_selected_chemical_row = condemnation_exact_hidden_exposed_gain
		}
		clamp_variable = {
			var = cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_selected_chemical_row
			min = @CR_SC_CBRN_MATH_ZERO
			max = cbrn_diplomacy_record_condemnation_total
		}
		add_to_variable = {
			cbrn_chemical_record_relief_paid_entries^cbrn_diplomacy_selected_chemical_row = condemnation_exact_hidden_retaliation_relief_exposed
		}
		subtract_from_variable = {
			cbrn_chemical_record_relief_unpaid_entries^cbrn_diplomacy_selected_chemical_row = condemnation_exact_hidden_retaliation_relief_exposed
		}
		clamp_variable = {
			var = cbrn_chemical_record_relief_unpaid_entries^cbrn_diplomacy_selected_chemical_row
			min = @CR_SC_CBRN_MATH_ZERO
			max = cbrn_diplomacy_record_relief_total
		}
	}

	if = {
		limit = {
			OR = {
				check_variable = {
					var = cbrn_chemical_record_condemnation_unpaid_entries^cbrn_diplomacy_selected_chemical_row
					value = @CR_SC_CBRN_MATH_ZERO
					compare = less_than_or_equals
				}
				AND = {
					has_event_target = cbrn_diplomacy_record_actor
					event_target:cbrn_diplomacy_record_actor = { exists = no }
					check_variable = {
						var = cbrn_diplomacy_published_attribution
						value = constant:cbrn_attribution_state.confirmed
						compare = equals
					}
				}
			}
		}
		if = {
			limit = {
				has_event_target = cbrn_diplomacy_record_actor
				event_target:cbrn_diplomacy_record_actor = { exists = no }
			}
			set_variable = {
				cbrn_chemical_record_status_entries^cbrn_diplomacy_selected_chemical_row = constant:cbrn_chemical_record_status.historical_closed
			}
		}
		else = {
			set_variable = {
				cbrn_chemical_record_status_entries^cbrn_diplomacy_selected_chemical_row = constant:cbrn_chemical_record_status.settled
			}
		}
		subtract_from_variable = { cbrn_chemical_open_forensic_record_count = constant:cbrn_math.one }
		clamp_variable = {
			var = cbrn_chemical_open_forensic_record_count
			min = @CR_SC_CBRN_MATH_ZERO
			max = cbrn_chemical_record_status_entries^num
		}
	}
}

# State scope. Registers confirmation against the exact selected row. No
# successor country inherits responsibility when either historical scope is
# no longer active.
cbrn_diplomacy_register_newly_confirmed_chemical_record = {
	if = {
		limit = {
			check_variable = { var = cbrn_diplomacy_published_attribution value = constant:cbrn_attribution_state.confirmed compare = equals }
			check_variable = {
				var = cbrn_diplomacy_record_release_status
				value = constant:cbrn_chemical_release_status.released
				compare = equals
			}
			check_variable = {
				var = cbrn_chemical_record_confirmed_registered_entries^cbrn_diplomacy_selected_chemical_row
				value = @CR_SC_CBRN_MATH_ZERO
				compare = equals
			}
		}
		set_variable = {
			cbrn_chemical_record_confirmed_registered_entries^cbrn_diplomacy_selected_chemical_row = constant:cbrn_math.one
		}
		set_state_flag = cbrn_confirmed_chemical_use_history
		set_global_flag = cbrn_world_confirmed_chemical_use
		if = {
			limit = {
				has_event_target = cbrn_diplomacy_record_actor
				event_target:cbrn_diplomacy_record_actor = { exists = yes }
			}
			event_target:cbrn_diplomacy_record_actor = {
				set_country_flag = cbrn_confirmed_chemical_use_history
				set_country_flag = used_unconventional_weapon
				air_cleanliness_treaty_on_unconventional_use = yes
			}
			if = {
				limit = {
					has_event_target = cbrn_diplomacy_record_victim
					event_target:cbrn_diplomacy_record_victim = { exists = yes }
				}
				event_target:cbrn_diplomacy_record_victim = {
					save_event_target_as = cbrn_retaliation_target_country
					set_country_flag = cbrn_confirmed_chemical_attack_victim_history
				}
				set_temp_variable = { cbrn_confirmed_use_action_date = cbrn_diplomacy_record_action_date }
				set_temp_variable = { cbrn_confirmed_use_retaliation_status = cbrn_diplomacy_record_retaliation_status }
				set_temp_variable = { cbrn_retaliation_source_type = constant:cbrn_diplomacy_source.chemical }
				event_target:cbrn_diplomacy_record_actor = {
					cbrn_diplomacy_record_confirmed_cbrn_use = yes
				}
			}
		}
		else = {
			set_state_flag = cbrn_confirmed_chemical_historical_actor_inactive
		}
	}
}

# State scope. The exact provider is read from the state ledger. Cleanup uses
# current severity and applies only actual contamination still present.
cbrn_diplomacy_complete_international_decon_mission = {
	if = {
		limit = {
			has_state_flag = cbrn_international_decon_mission_active
			has_variable = cbrn_international_decon_provider
			var:cbrn_international_decon_provider = { exists = yes }
			has_variable = cbrn_international_decon_recipient
			var:cbrn_international_decon_recipient = { exists = yes }
			CONTROLLER = { tag = var:PREV.cbrn_international_decon_recipient }
			cbrn_state_retains_international_decontamination_access = yes
			cbrn_state_has_chemical_contamination = yes
		}
		var:cbrn_international_decon_provider = {
			save_event_target_as = cbrn_diplomacy_decon_provider
		}
		cbrn_refresh_state_contamination_class = yes
		set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.trace_or_local_cleanup }
		if = {
			limit = {
				check_variable = {
					var = cbrn_state_contamination_class_output
					value = constant:cbrn_contamination_class.catastrophic
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.catastrophic_cleanup }
		}
		else_if = {
			limit = {
				check_variable = {
					var = cbrn_state_contamination_class_output
					value = constant:cbrn_contamination_class.severe
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.severe_cleanup }
		}
		else_if = {
			limit = {
				check_variable = {
					var = cbrn_state_contamination_class_output
					value = constant:cbrn_contamination_class.serious
					compare = equals
				}
			}
			set_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.serious_cleanup }
		}
		if = {
			limit = {
				event_target:cbrn_diplomacy_decon_provider = {
					cbrn_designer_has_international_medical_mission = yes
				}
			}
			multiply_temp_variable = { cbrn_diplomacy_cleanup_points = constant:cbrn_diplomacy_decon.international_medical_mission_mult }
		}

		set_temp_variable = { cbrn_state_contamination_delta = cbrn_diplomacy_cleanup_points }
		multiply_temp_variable = { cbrn_state_contamination_delta = constant:cbrn_payload_math.negation_multiplier }
		cbrn_apply_state_contamination_delta_internal = yes
		set_temp_variable = { cbrn_diplomacy_cleanup_completed = cbrn_state_previous_contamination_value }
		subtract_from_temp_variable = { cbrn_diplomacy_cleanup_completed = cbrn_state_new_contamination_value }

		if = {
			limit = { has_variable = cbrn_medical_saturation }
			subtract_from_variable = {
				cbrn_medical_saturation = constant:cbrn_diplomacy_decon.medical_saturation_reduction
			}
			clamp_variable = {
				var = cbrn_medical_saturation
				min = @CR_SC_CBRN_MATH_ZERO
				max = constant:cbrn_chemical_dispatch.state_medical_maximum
			}
		}

		if = {
			limit = { cbrn_state_has_observable_chemical_evidence = yes }
			cbrn_diplomacy_observe_best_chemical_record = yes
		}

		event_target:cbrn_diplomacy_decon_provider = {
			add_to_variable = { cbrn_international_decon_missions_completed = constant:cbrn_math.one }
			add_to_variable = { cbrn_international_decon_cleanup_total = cbrn_diplomacy_cleanup_completed }
		}
		CONTROLLER = {
			add_to_variable = { cbrn_international_decon_aid_received = cbrn_diplomacy_cleanup_completed }
			add_opinion_modifier = {
				target = event_target:cbrn_diplomacy_decon_provider
				modifier = cbrn_international_decontamination_aid
			}
		}
		set_state_flag = cbrn_international_decon_mission_history
	}

	clr_state_flag = cbrn_international_decon_mission_active
	set_temp_variable = { cbrn_diplomacy_state_cooldown_days = constant:cbrn_diplomacy_duration.decon_state_cooldown_days }
	set_state_flag = {
		flag = cbrn_international_decon_mission_cooldown
		days = var:cbrn_diplomacy_state_cooldown_days
	}
	clear_variable = cbrn_international_decon_provider
	clear_variable = cbrn_international_decon_recipient
	clear_variable = cbrn_international_decon_start_date
}

	clear_variable = cbrn_international_decon_provider
	clear_variable = cbrn_international_decon_recipient
	clear_variable = cbrn_international_decon_start_date
