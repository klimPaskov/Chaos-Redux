# ============================================================================
# CHAOS REDUX - EVENT 025 THE WRECK IS BREAKING APART (EVOLUTION IV)
# ============================================================================
# Overview:
# - Wreck Integrity is one global value, global.chaosx_nr25_wreck_integrity,
#   seeded when Evolution IV activates (lower when the race opens with it).
# - The first expedition pulse of each pulse period decays it unless a
#   stabilisation is holding; strong signal pulses, armed clashes, seizures,
#   cutting and failed final attempts damage it. Stabilisation raises it only
#   up to a cap below the original hull.
# - Below the fragment-field threshold, each pulse period may bury or destroy
#   one unrecovered fragment site in the custody ledger. Every expedition's
#   board shows the site as lost or buried; those that had surveyed it get a
#   report. The hidden core sector is never lost this way.
# - Cutting fragments at low integrity can bring out a contaminated fragment
#   of reduced value.
# - At zero integrity the hull breaks. The command core stays the winning
#   objective, so the race can still be won.
# - Only the expedition's own country, the global race values, the stored
#   participant registry and the six ledger slots are read. Nothing scans the
#   world, and nothing runs while the evolution is disabled.
# ============================================================================

# GLOBAL value set from any scope. Seeds Wreck Integrity when Evolution IV
# activates: lower when the race opened with the wreck already unstable.
chaosx_nr25_seed_wreck_integrity = {
	if = {
		limit = { has_global_flag = chaosx_nr25_opening_wreck_breaking_apart }
		set_variable = { global.chaosx_nr25_wreck_integrity = constant:chaosx_nr25_opening.wreck_start_integrity }
	}
	else = {
		set_variable = { global.chaosx_nr25_wreck_integrity = constant:chaosx_nr25_wreck.full_integrity }
	}
	set_variable = { global.chaosx_nr25_wreck_decay_date = global.date }
}

# Any scope. Applies the temporary chaosx_nr25_wreck_damage (positive) while
# Evolution IV is in force, and breaks the hull at zero.
chaosx_nr25_damage_wreck = {
	if = {
		limit = { chaosx_nr25_wreck_breaking_apart_evolution_in_force = yes }
		if = {
			limit = { NOT = { has_variable = global.chaosx_nr25_wreck_integrity } }
			chaosx_nr25_seed_wreck_integrity = yes
		}
		subtract_from_variable = { global.chaosx_nr25_wreck_integrity = chaosx_nr25_wreck_damage }
		clamp_variable = {
			var = global.chaosx_nr25_wreck_integrity
			min = constant:chaosx_nr25_value.zero
			max = constant:chaosx_nr25_wreck.full_integrity
		}
		chaosx_nr25_check_hull_break = yes
	}
}

# Any scope. The hull breaks once, when Wreck Integrity reaches zero; every
# expedition still in the race hears that the command core is now the prize.
chaosx_nr25_check_hull_break = {
	if = {
		limit = {
			NOT = { has_global_flag = chaosx_nr25_hull_broken }
			NOT = { check_variable = { var = global.chaosx_nr25_wreck_integrity value = constant:chaosx_nr25_value.zero compare = greater_than } }
		}
		set_global_flag = chaosx_nr25_hull_broken
		for_each_scope_loop = {
			array = global.chaosx_nr25_participants
			if = {
				limit = {
					exists = yes
					has_country_flag = chaosx_nr25_participant
					NOT = { has_country_flag = chaosx_nr25_terminal }
				}
				country_event = { id = chaosx.nr25.345 days = 1 }
			}
		}
	}
}

# Any scope. A stabilisation raises Wreck Integrity by the temporary
# chaosx_nr25_wreck_gain, never above the stabilisation cap, and holds the
# hull against decay for a while.
chaosx_nr25_stabilize_wreck_integrity = {
	if = {
		limit = { NOT = { has_variable = global.chaosx_nr25_wreck_integrity } }
		chaosx_nr25_seed_wreck_integrity = yes
	}
	if = {
		limit = { check_variable = { var = global.chaosx_nr25_wreck_integrity value = constant:chaosx_nr25_wreck.stabilize_cap compare = less_than } }
		add_to_variable = { global.chaosx_nr25_wreck_integrity = chaosx_nr25_wreck_gain }
		if = {
			limit = { check_variable = { var = global.chaosx_nr25_wreck_integrity value = constant:chaosx_nr25_wreck.stabilize_cap compare = greater_than } }
			set_variable = { global.chaosx_nr25_wreck_integrity = constant:chaosx_nr25_wreck.stabilize_cap }
		}
	}
	set_temp_variable = { chaosx_nr25_wreck_days = constant:chaosx_nr25_wreck.held_days }
	set_global_flag = { flag = chaosx_nr25_wreck_held days = chaosx_nr25_wreck_days value = 1 }
}

# COUNTRY scope (the expedition, from its pulse while Evolution IV is in force).
chaosx_nr25_wreck_pulse = {
	if = {
		limit = { NOT = { has_variable = global.chaosx_nr25_wreck_integrity } }
		chaosx_nr25_seed_wreck_integrity = yes
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.chaosx_nr25_wreck_decay_date }
				check_variable = { var = global.date value = global.chaosx_nr25_wreck_decay_date compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { chaosx_date_days = constant:chaosx_nr25_timing.pulse_days }
		set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
		chaosx_convert_days_to_date_offset = yes
		set_variable = { global.chaosx_nr25_wreck_decay_date = global.date }
		add_to_variable = { global.chaosx_nr25_wreck_decay_date = chaosx_date_offset }
		if = {
			limit = { NOT = { has_global_flag = chaosx_nr25_wreck_held } }
			set_temp_variable = { chaosx_nr25_wreck_damage = constant:chaosx_nr25_wreck.pulse_decay }
			chaosx_nr25_damage_wreck = yes
		}
		if = {
			limit = { check_variable = { var = global.chaosx_nr25_wreck_integrity value = constant:chaosx_nr25_wreck.field_threshold compare = less_than } }
			set_temp_variable = { chaosx_nr25_site_loss_chance = constant:chaosx_nr25_wreck.site_loss_chance }
			if = {
				limit = { check_variable = { var = global.chaosx_nr25_wreck_integrity value = constant:chaosx_nr25_wreck.critical_threshold compare = less_than } }
				set_temp_variable = { chaosx_nr25_site_loss_chance = constant:chaosx_nr25_wreck.site_loss_critical_chance }
			}
			if = {
				limit = { has_global_flag = chaosx_nr25_hull_broken }
				set_temp_variable = { chaosx_nr25_site_loss_chance = constant:chaosx_nr25_wreck.site_loss_broken_chance }
			}
			set_temp_variable_to_random = chaosx_nr25_site_loss_roll
			multiply_temp_variable = { chaosx_nr25_site_loss_roll = constant:chaosx_nr25_wreck.percent }
			if = {
				limit = { check_variable = { var = chaosx_nr25_site_loss_roll value = chaosx_nr25_site_loss_chance compare = less_than } }
				chaosx_nr25_lose_fragment_site = yes
			}
		}
	}
}

# Any scope. Buries or destroys one random unrecovered fragment site other
# than the hidden core sector, through the ledger's slot flags, and marks it
# lost on every expedition's charts. Expeditions that had surveyed it get a
# report.
chaosx_nr25_lose_fragment_site = {
	set_temp_variable = { chaosx_nr25_site_count = constant:chaosx_nr25_value.zero }
	for_loop_effect = {
		start = constant:chaosx_nr25_value.one
		end = constant:chaosx_nr25_cost.fragments_total
		compare = less_than_or_equals
		value = chaosx_nr25_site_index
		if = {
			limit = { chaosx_nr25_fragment_site_can_be_lost = yes }
			add_to_temp_variable = { chaosx_nr25_site_count = constant:chaosx_nr25_value.one }
		}
	}
	if = {
		limit = { check_variable = { var = chaosx_nr25_site_count value = constant:chaosx_nr25_value.zero compare = greater_than } }
		set_temp_variable_to_random = chaosx_nr25_site_pick
		multiply_temp_variable = { chaosx_nr25_site_pick = chaosx_nr25_site_count }
		set_temp_variable = { chaosx_nr25_lost_sector = constant:chaosx_nr25_value.zero }
		for_loop_effect = {
			start = constant:chaosx_nr25_value.one
			end = constant:chaosx_nr25_cost.fragments_total
			compare = less_than_or_equals
			value = chaosx_nr25_site_index
			if = {
				limit = {
					check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.zero compare = equals }
					chaosx_nr25_fragment_site_can_be_lost = yes
				}
				if = {
					limit = { check_variable = { var = chaosx_nr25_site_pick value = constant:chaosx_nr25_value.one compare = less_than } }
					set_temp_variable = { chaosx_nr25_lost_sector = chaosx_nr25_site_index }
				}
				subtract_from_temp_variable = { chaosx_nr25_site_pick = constant:chaosx_nr25_value.one }
			}
		}
		if = {
			limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.zero compare = greater_than } }
			set_variable = { global.chaosx_nr25_last_lost_site = chaosx_nr25_lost_sector }
			set_temp_variable_to_random = chaosx_nr25_site_fate_roll
			multiply_temp_variable = { chaosx_nr25_site_fate_roll = constant:chaosx_nr25_wreck.percent }
			if = {
				limit = { check_variable = { var = chaosx_nr25_site_fate_roll value = constant:chaosx_nr25_wreck.destroyed_share compare = less_than } }
				set_global_flag = chaosx_nr25_last_lost_site_destroyed
				meta_effect = {
					text = { set_global_flag = chaosx_alien_recovery_fragment_[SLOT]_destroyed }
					SLOT = "[?chaosx_nr25_lost_sector|.0]"
				}
			}
			else = {
				clr_global_flag = chaosx_nr25_last_lost_site_destroyed
				meta_effect = {
					text = { set_global_flag = chaosx_alien_recovery_fragment_[SLOT]_buried }
					SLOT = "[?chaosx_nr25_lost_sector|.0]"
				}
			}
			for_each_scope_loop = {
				array = global.chaosx_nr25_participants
				if = {
					limit = {
						exists = yes
						has_country_flag = chaosx_nr25_participant
					}
					set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.zero }
					chaosx_nr25_mark_sector_lost = yes
					chaosx_nr25_refresh_sector_frames = yes
					if = {
						limit = {
							check_variable = { var = chaosx_nr25_sector_was_known value = constant:chaosx_nr25_value.one compare = equals }
							NOT = { has_country_flag = chaosx_nr25_terminal }
						}
						country_event = { id = chaosx.nr25.344 days = 1 }
					}
				}
			}
		}
	}
}

# COUNTRY scope. Marks the sector in the temporary chaosx_nr25_lost_sector as
# lost on this expedition's charts. Sets the temporary
# chaosx_nr25_sector_was_known to one when the expedition had surveyed it.
chaosx_nr25_mark_sector_lost = {
	if = {
		limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.one compare = equals } }
		if = { limit = { NOT = { check_variable = { var = chaosx_nr25_sector_ice_shelf_state value = constant:chaosx_nr25_sector.unresolved compare = equals } } } set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.one } }
		set_variable = { chaosx_nr25_sector_ice_shelf_state = constant:chaosx_nr25_sector.lost }
	}
	else_if = {
		limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.two compare = equals } }
		if = { limit = { NOT = { check_variable = { var = chaosx_nr25_sector_mountain_shadow_state value = constant:chaosx_nr25_sector.unresolved compare = equals } } } set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.one } }
		set_variable = { chaosx_nr25_sector_mountain_shadow_state = constant:chaosx_nr25_sector.lost }
	}
	else_if = {
		limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.three compare = equals } }
		if = { limit = { NOT = { check_variable = { var = chaosx_nr25_sector_signal_basin_state value = constant:chaosx_nr25_sector.unresolved compare = equals } } } set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.one } }
		set_variable = { chaosx_nr25_sector_signal_basin_state = constant:chaosx_nr25_sector.lost }
	}
	else_if = {
		limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.four compare = equals } }
		if = { limit = { NOT = { check_variable = { var = chaosx_nr25_sector_coastal_fracture_state value = constant:chaosx_nr25_sector.unresolved compare = equals } } } set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.one } }
		set_variable = { chaosx_nr25_sector_coastal_fracture_state = constant:chaosx_nr25_sector.lost }
	}
	else_if = {
		limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.five compare = equals } }
		if = { limit = { NOT = { check_variable = { var = chaosx_nr25_sector_magnetic_silence_state value = constant:chaosx_nr25_sector.unresolved compare = equals } } } set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.one } }
		set_variable = { chaosx_nr25_sector_magnetic_silence_state = constant:chaosx_nr25_sector.lost }
	}
	else_if = {
		limit = { check_variable = { var = chaosx_nr25_lost_sector value = constant:chaosx_nr25_value.six compare = equals } }
		if = { limit = { NOT = { check_variable = { var = chaosx_nr25_sector_false_map_state value = constant:chaosx_nr25_sector.unresolved compare = equals } } } set_temp_variable = { chaosx_nr25_sector_was_known = constant:chaosx_nr25_value.one } }
		set_variable = { chaosx_nr25_sector_false_map_state = constant:chaosx_nr25_sector.lost }
	}
}

# COUNTRY scope. Cut Fragments Free: one field team cuts at the nearest
# debris. It damages the hull; below the contamination threshold the piece
# may come out contaminated, bringing half the progress and more exposure.
chaosx_nr25_wreck_response_cut = {
	set_temp_variable = { chaosx_nr25_wreck_damage = constant:chaosx_nr25_wreck.cut_damage }
	chaosx_nr25_damage_wreck = yes
	set_temp_variable = { chaosx_nr25_cut_progress = constant:chaosx_nr25_wreck.cut_progress }
	clr_country_flag = chaosx_nr25_last_cut_contaminated
	if = {
		limit = {
			has_variable = global.chaosx_nr25_wreck_integrity
			check_variable = { var = global.chaosx_nr25_wreck_integrity value = constant:chaosx_nr25_wreck.contaminated_threshold compare = less_than }
		}
		set_temp_variable_to_random = chaosx_nr25_cut_roll
		multiply_temp_variable = { chaosx_nr25_cut_roll = constant:chaosx_nr25_wreck.percent }
		if = {
			limit = { check_variable = { var = chaosx_nr25_cut_roll value = constant:chaosx_nr25_wreck.contaminated_chance compare = less_than } }
			set_country_flag = chaosx_nr25_last_cut_contaminated
			multiply_temp_variable = { chaosx_nr25_cut_progress = constant:chaosx_nr25_wreck.contaminated_progress_factor }
			add_to_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_wreck.contaminated_exposure }
		}
	}
	add_to_variable = { chaosx_nr25_progress = chaosx_nr25_cut_progress }
	add_to_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_cost.exposure_wreck_cut }
	chaosx_nr25_mark_survey_sector = yes
	set_temp_variable = { chaosx_alien_recovery_fragment_id = chaosx_nr25_current_sector }
	set_temp_variable = { chaosx_alien_recovery_fragment_role = chaosx_nr25_fragment_role }
	set_temp_variable = { chaosx_alien_recovery_fragment_assigned = constant:chaosx_nr25_value.zero }
	chaosx_alien_recovery_ledger_assign_fragment = yes
	if = {
		limit = { check_variable = { var = chaosx_alien_recovery_fragment_assigned value = constant:chaosx_nr25_value.one compare = greater_than_or_equals } }
		set_country_flag = chaosx_nr25_fragment_recovered
		chaosx_nr25_record_fragment_role = yes
		if = {
			limit = { has_country_flag = chaosx_nr25_last_cut_contaminated }
			set_country_flag = chaosx_nr25_holds_contaminated_fragment
			country_event = { id = chaosx.nr25.346 days = 1 }
		}
		else = { chaosx_nr25_report_fragment_recovered = yes }
		chaosx_nr25_confirm_current_sector = yes
	}
	set_country_flag = chaosx_nr25_wreck_cut_route
}
