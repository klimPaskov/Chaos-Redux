@cannibalism_super_event_visible_days = 14
@cannibalism_state_pressure_days = 180
@cannibalism_short_state_pressure_days = 90
@cannibalism_direct_delay_days = 1
@cannibalism_direct_random_days = 4

cannibalism_prepare_random_event_fire = {
	set_temp_variable = { cannibalism_context_ready = 0 }
	cannibalism_select_origin_country = yes
	if = {
		limit = { check_variable = { cannibalism_origin_selected > 0 } }
		event_target:cannibalism_origin_country = {
			save_event_target_as = cannibalism_latest_actor
		}
		set_temp_variable = { cannibalism_context_ready = 1 }
	}
}

cannibalism_call_direct_fire_once_if_possible = {
	if = {
		limit = { NOT = { is_in_array = { global.fired_events = constant:cannibalism_event_log.event_id } } }
		if = {
			limit = { cannibalism_can_be_origin = yes }
			country_event = { id = chaosx.nr14.2 }
		}
		else = {
			cannibalism_prepare_random_event_fire = yes
			if = {
				limit = { check_variable = { cannibalism_context_ready > 0 } }
				event_target:cannibalism_origin_country = {
					country_event = { id = chaosx.nr14.2 }
				}
			}
		}
	}
}

cannibalism_select_origin_country = {
	clear_temp_array = cannibalism_origin_pool
	set_temp_variable = { cannibalism_origin_pool_count = 0 }
	set_temp_variable = { cannibalism_origin_selected = 0 }
	if = {
		limit = { cannibalism_has_valid_origin_available = yes }
		every_country = {
			limit = { cannibalism_can_be_origin = yes }
			cannibalism_add_current_country_to_origin_pool = yes
		}
	}
	if = {
		limit = { check_variable = { cannibalism_origin_pool_count > 0 } }
		random_scope_in_array = {
			array = cannibalism_origin_pool
			save_event_target_as = cannibalism_origin_country
			set_temp_variable = { cannibalism_origin_selected = 1 }
		}
	}
	clear_temp_array = cannibalism_origin_pool
	set_temp_variable = { cannibalism_origin_pool_count = 0 }
	set_temp_variable = { cannibalism_origin_pool_weight_remaining = 0 }
	set_temp_variable = { cannibalism_origin_weight = 0 }
}

cannibalism_prepare_origin_weight = {
	set_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.base }
	if = {
		limit = { is_ai = no }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.player_bonus }
	}
	if = {
		limit = { any_enemy_country = { is_major = yes } }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.major_war_bonus }
	}
	if = {
		limit = { has_stability < 0.45 }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.low_stability_bonus }
	}
	if = {
		limit = { has_war_support < 0.45 }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.low_war_support_bonus }
	}
	if = {
		limit = { num_divisions > constant:cannibalism_target_threshold.large_army_divisions }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.large_army_bonus }
	}
	if = {
		limit = { any_controlled_state = { is_coastal = yes } }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.coastal_bonus }
	}
	if = {
		limit = { capital_scope = { is_island_state = yes } }
		add_to_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.island_bonus }
	}
	if = {
		limit = { has_country_flag = cannibalism_country_defeated_outbreak }
		subtract_from_temp_variable = { cannibalism_origin_weight = constant:cannibalism_target_weight.already_contained_penalty }
	}
	round_temp_variable = cannibalism_origin_weight
	clamp_temp_variable = {
		var = cannibalism_origin_weight
		min = constant:cannibalism_target_weight.min
		max = constant:cannibalism_target_weight.max
	}
}

cannibalism_add_current_country_to_origin_pool = {
	cannibalism_prepare_origin_weight = yes
	set_temp_variable = { cannibalism_origin_pool_weight_remaining = cannibalism_origin_weight }
	while_loop_effect = {
		limit = { check_variable = { cannibalism_origin_pool_weight_remaining > 0 } }
		add_to_temp_array = { cannibalism_origin_pool = THIS }
		add_to_temp_variable = { cannibalism_origin_pool_count = 1 }
		subtract_from_temp_variable = { cannibalism_origin_pool_weight_remaining = 1 }
	}
	set_temp_variable = { cannibalism_origin_pool_weight_remaining = 0 }
	set_temp_variable = { cannibalism_origin_weight = 0 }
}

cannibalism_register_direct_fire_once_if_needed = {
	if = {
		limit = {
			NOT = { has_country_flag = cannibalism_event_dispatch_context }
			NOT = { is_in_array = { global.fired_events = constant:cannibalism_event_log.event_id } }

		set_temp_variable = { event_id = constant:cannibalism_event_log.event_id }
		on_fire_once_event_fired = yes
		update_last_fired_event_display = yes
	}
	clr_country_flag = cannibalism_event_dispatch_context
}

cannibalism_try_country_resolution = {
	if = {
		limit = {
			cannibalism_country_containment_success_ready = yes
			NOT = { has_country_flag = cannibalism_country_contained }
		}
		country_event = { id = chaosx.nr14.6 days = 1 }
	}
	if = {
		limit = {
			cannibalism_country_has_outbreak = yes
			OR = {
				check_variable = {
					var = cannibalism_hunger
					value = constant:cannibalism_meter.crisis
					compare = greater_than_or_equals
				}
				check_variable = {
					var = cannibalism_cult
					value = constant:cannibalism_meter.crisis
					compare = greater_than_or_equals
				}
				check_variable = {
					var = cannibalism_spread
					value = constant:cannibalism_meter.crisis
					compare = greater_than_or_equals
				}
			}
			NOT = { has_country_flag = cannibalism_country_containment_failed }
		}
		country_event = { id = chaosx.nr14.5 days = 1 }
	}
}

cannibalism_country_containment_success = {
	if = {
		limit = { has_country_flag = cannibalism_spread_outbreak }
		set_country_flag = cannibalism_country_was_spread_outbreak
	}
	set_country_flag = cannibalism_country_contained
	set_country_flag = cannibalism_country_defeated_outbreak
	clr_country_flag = cannibalism_containment_mission_active
	clr_country_flag = cannibalism_outbreak_active
	clr_country_flag = cannibalism_spread_outbreak
	set_variable = { cannibalism_stage = constant:cannibalism_stage.contained }
	remove_ideas = cannibalism_field_collapse
	remove_ideas = cannibalism_ritual_hunger
	if = { limit = { has_active_mission = cannibalism_containment_deadline_mission } remove_mission = cannibalism_containment_deadline_mission }
	if = {
		limit = {
			has_country_flag = cannibalism_original_outbreak_country
			NOT = { has_global_flag = cannibalism_any_spread_country_reported }
			NOT = { has_country_flag = cannibalism_posture_secrecy }
			NOT = { has_country_flag = cannibalism_exploitation_was_used }
		}
		set_country_flag = cannibalism_clean_mess_achieved
		set_global_flag = cannibalism_clean_mess_achieved
	}
	if = {
		limit = {
			is_major = yes
			has_war = yes
			has_country_flag = cannibalism_posture_logistics
			NOT = { has_country_flag = cannibalism_used_harsh_military_police }
			NOT = { has_country_flag = cannibalism_exploitation_was_used }
		}
		set_country_flag = cannibalism_logistics_major_war_eligible
		set_country_flag = cannibalism_do_not_feed_the_front_achieved
	}
	if = {
		limit = {
			has_country_flag = cannibalism_public_truth_route
			has_stability > constant:cannibalism_achievement_threshold.trial_stability
			NOT = { has_country_flag = cannibalism_posture_secrecy }
		}
		set_country_flag = cannibalism_trial_without_panic_achieved
	}
	add_to_variable = { global.cannibalism_contained_countries = 1 }
	set_global_flag = cannibalism_any_country_contained
	add_stability = 0.02
	add_war_support = 0.01
	cannibalism_cleanup_country_state_pressure = yes
	cannibalism_refresh_global_defeat_state = yes
	cannibalism_refresh_world_threat_source = yes
}

			add_equipment_to_stockpile = { type = train_equipment_1 amount = constant:cannibalism_country_package.starting_trains }

		add_to_variable = { global.cannibalism_network_strength = constant:cannibalism_delta.medium }
		CBL = {
			if = {
				limit = { NOT = { has_focus_tree = cannibalism_commune_focus_tree } }

cannibalism_refresh_global_defeat_state = {
	if = {
		limit = {
			NOT = { any_country = { cannibalism_country_has_outbreak = yes } }
			OR = {
				NOT = { country_exists = CBL }
				CBL = { has_capitulated = yes }
			}
		}
		set_global_flag = cannibalism_globally_defeated
		if = {
			limit = {
				has_global_flag = cannibalism_any_country_contained
				country_exists = CBL
				CBL = { has_capitulated = yes }
			}
			set_global_flag = cannibalism_no_second_table_achieved
			every_country = {
				limit = {
					exists = yes
					has_country_flag = cannibalism_country_defeated_outbreak
					NOT = { has_country_flag = cannibalism_lost_capital_to_cannibals }
					NOT = { is_cannibal_commune_country = yes }
					NOT = { is_special_chaos_country = yes }
					NOT = { is_actual_nonhuman_country = yes }
					capital_scope = { NOT = { is_controlled_by = CBL } }
				}
				set_country_flag = cannibalism_no_second_table_achieved
			}
		}
		if = {
			limit = { has_global_flag = cannibalism_evolution_global_table_recorded }
			set_global_flag = cannibalism_living_are_not_cattle_achieved
		}
		cannibalism_cleanup_commune_country_pressure = yes
		clr_global_flag = cannibalism_system_active
		cannibalism_emit_defeat_super_event = yes
		country_event = { id = chaosx.nr14.14 days = 1 }
	}
}

cannibalism_refresh_world_threat_source = {
	if = {
		limit = {
			OR = {
				has_global_flag = cannibalism_evolution_global_table_recorded
				country_exists = CBL
				has_global_flag = cannibalism_world_end_started
			}
			NOT = { has_global_flag = cannibalism_globally_defeated }
		}
		set_global_flag = world_threat_source_cannibalism
	}
	else = {
		clr_global_flag = world_threat_source_cannibalism
	}
	refresh_world_threat_state = yes
	cannibalism_try_hunger_of_hannibal_achievement = yes
}
