# ============================================================================
# Country-scoped view builders and action dispatch for the post-Evolution III
# Muster Board. The GUI is a player surface only: all gameplay transactions are
# public effects also called by decisions and the country AI pulse. Family-
# specific train/spawn/sustainment costs are provider-owned and dispatched by
# the generic Chaos unit family registry provider id.

# ---------------------------------------------------------------------------
# Timed, family-specific management ledger
# ---------------------------------------------------------------------------

infantry_spawn_family_management_ledger_is_aligned = {
	set_temp_variable = { infantry_spawn_family_management_ledger_aligned = 1 }
	if = {
		limit = {
			OR = {
				NOT = { check_variable = { var = infantry_spawn_family_management_action_entries^num value = infantry_spawn_family_management_family_id_entries^num compare = equals } }
				NOT = { check_variable = { var = infantry_spawn_family_management_until_date_entries^num value = infantry_spawn_family_management_family_id_entries^num compare = equals } }
			}
		}
		set_temp_variable = { infantry_spawn_family_management_ledger_aligned = 0 }
		infantry_spawn_mark_ledger_invariant_failure = yes
	}
}

infantry_spawn_find_family_management_row = {
	set_temp_variable = { infantry_spawn_family_management_row_index = constant:infantry_spawn_muster_gui_value.invalid_index }
	set_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.zero }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_family_management_row_index < constant:infantry_spawn_muster_gui_value.zero }
			check_variable = { var = infantry_spawn_family_management_scan_index value = infantry_spawn_family_management_family_id_entries^num compare = less_than }
		}
		if = {
			limit = {
				check_variable = { var = infantry_spawn_family_management_family_id_entries^infantry_spawn_family_management_scan_index value = infantry_spawn_family_management_target_family_id compare = equals }
				check_variable = { var = infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index value = infantry_spawn_family_management_target_action compare = equals }
			}
			set_temp_variable = { infantry_spawn_family_management_row_index = infantry_spawn_family_management_scan_index }
		}
		add_to_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.one }
	}
}

infantry_spawn_refresh_family_management_summary_flags = {
	clr_country_flag = infantry_spawn_anomalous_cantonment_active
	clr_country_flag = infantry_spawn_anomalous_liaison_active
	clr_country_flag = infantry_spawn_anomalous_restricted_deployment
	clr_country_flag = infantry_spawn_anomalous_sustainment_active
	clr_country_flag = infantry_spawn_anomalous_containment_success
	set_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.zero }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_family_management_scan_index value = infantry_spawn_family_management_action_entries^num compare = less_than } }
		if = {
			limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.cantonment } }
			set_country_flag = infantry_spawn_anomalous_cantonment_active
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.liaison } }
			set_country_flag = infantry_spawn_anomalous_liaison_active
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.restricted_deployment } }
			set_country_flag = infantry_spawn_anomalous_restricted_deployment
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.sustainment } }
			set_country_flag = infantry_spawn_anomalous_sustainment_active
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.containment } }
			set_country_flag = infantry_spawn_anomalous_containment_success
		}
		add_to_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.one }
	}
}

infantry_spawn_cleanup_expired_family_management = {
	infantry_spawn_family_management_ledger_is_aligned = yes
	if = {
		limit = { check_variable = { infantry_spawn_family_management_ledger_aligned > 0 } }
		set_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.zero }
		while_loop_effect = {
			limit = { check_variable = { var = infantry_spawn_family_management_scan_index value = infantry_spawn_family_management_family_id_entries^num compare = less_than } }
			if = {
				limit = {
					check_variable = { var = infantry_spawn_family_management_until_date_entries^infantry_spawn_family_management_scan_index value = global.date compare = less_than_or_equals }
				}
				remove_from_array = { array = infantry_spawn_family_management_family_id_entries index = infantry_spawn_family_management_scan_index }
				remove_from_array = { array = infantry_spawn_family_management_action_entries index = infantry_spawn_family_management_scan_index }
				remove_from_array = { array = infantry_spawn_family_management_until_date_entries index = infantry_spawn_family_management_scan_index }
			}
			else = { add_to_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.one } }
		}
	}
	infantry_spawn_refresh_family_management_summary_flags = yes
}

infantry_spawn_record_timed_family_management_action = {
	infantry_spawn_cleanup_expired_family_management = yes
	set_temp_variable = { infantry_spawn_family_management_target_family_id = infantry_spawn_selected_family_id }
	set_temp_variable = { infantry_spawn_family_management_target_action = infantry_spawn_family_management_action }
	infantry_spawn_find_family_management_row = yes
	set_temp_variable = { infantry_spawn_family_management_until_date = global.date }
	add_to_temp_variable = { infantry_spawn_family_management_until_date = infantry_spawn_family_management_duration_days }
	if = {
		limit = { check_variable = { infantry_spawn_family_management_row_index > constant:infantry_spawn_muster_gui_value.invalid_index } }
		set_variable = { infantry_spawn_family_management_until_date_entries^infantry_spawn_family_management_row_index = infantry_spawn_family_management_until_date }
	}
	else = {
		add_to_array = { array = infantry_spawn_family_management_family_id_entries value = infantry_spawn_selected_family_id }
		add_to_array = { array = infantry_spawn_family_management_action_entries value = infantry_spawn_family_management_action }
		add_to_array = { array = infantry_spawn_family_management_until_date_entries value = infantry_spawn_family_management_until_date }
	}
	infantry_spawn_refresh_family_management_summary_flags = yes
}

# Shared family-state helpers. They live beside the Muster Board builder because
# the board was their first consumer, but natural derivative selection and AI
# use the same temporary result contract.
infantry_spawn_load_current_family_action_state = {
	set_temp_variable = { infantry_spawn_family_has_cantonment = 0 }
	set_temp_variable = { infantry_spawn_family_has_liaison = 0 }
	set_temp_variable = { infantry_spawn_family_has_restricted_deployment = 0 }
	set_temp_variable = { infantry_spawn_family_has_sustainment = 0 }
	set_temp_variable = { infantry_spawn_family_has_containment = 0 }
	set_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.zero }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_family_management_scan_index value = infantry_spawn_family_management_family_id_entries^num compare = less_than } }
		if = {
			limit = { check_variable = { var = infantry_spawn_family_management_family_id_entries^infantry_spawn_family_management_scan_index value = infantry_spawn_current_family_id compare = equals } }
			if = {
				limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.cantonment } }
				set_temp_variable = { infantry_spawn_family_has_cantonment = 1 }
			}
			else_if = {
				limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.liaison } }
				set_temp_variable = { infantry_spawn_family_has_liaison = 1 }
			}
			else_if = {
				limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.restricted_deployment } }
				set_temp_variable = { infantry_spawn_family_has_restricted_deployment = 1 }
			}
			else_if = {
				limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.sustainment } }
				set_temp_variable = { infantry_spawn_family_has_sustainment = 1 }
			}
			else_if = {
				limit = { check_variable = { infantry_spawn_family_management_action_entries^infantry_spawn_family_management_scan_index = constant:infantry_spawn_family_management_action.containment } }
				set_temp_variable = { infantry_spawn_family_has_containment = 1 }
			}
		}
		add_to_temp_variable = { infantry_spawn_family_management_scan_index = constant:infantry_spawn_muster_gui_value.one }
	}
}

infantry_spawn_calculate_current_family_state = {
	set_temp_variable = { infantry_spawn_family_live_division_count = 0 }
	set_temp_variable = { infantry_spawn_family_claimant_loyal_division_count = 0 }
	set_temp_variable = { infantry_spawn_family_live_lot_count = 0 }
	set_temp_variable = { infantry_spawn_family_dispersible_lot_count = 0 }
	set_temp_variable = { infantry_spawn_family_claimant_lot_count = 0 }
	set_temp_variable = { infantry_spawn_family_primary_claimant_uid = 0 }
	set_temp_variable = { infantry_spawn_family_primary_claimant_influence = 0 }
	set_temp_variable = { infantry_spawn_family_primary_claimant_archetype = 0 }
	set_temp_variable = { infantry_spawn_family_latest_generation_uid = 0 }
	set_temp_variable = { infantry_spawn_muster_gui_scan_index = constant:infantry_spawn_muster_gui_value.zero }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_muster_gui_scan_index value = infantry_spawn_unit_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_unit_family_id_entries^infantry_spawn_muster_gui_scan_index value = infantry_spawn_current_family_id compare = equals }
				check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_muster_gui_scan_index value = constant:infantry_spawn_unit_status.untracked compare = greater_than }
				check_variable = { var = infantry_spawn_unit_status_entries^infantry_spawn_muster_gui_scan_index value = constant:infantry_spawn_unit_status.transferred_out compare = less_than }
			}
			add_to_temp_variable = { infantry_spawn_family_live_division_count = 1 }
			if = {
				limit = { check_variable = { infantry_spawn_unit_claimant_uid_entries^infantry_spawn_muster_gui_scan_index > 0 } }
				add_to_temp_variable = { infantry_spawn_family_claimant_loyal_division_count = 1 }
			}
			if = {
				limit = { check_variable = { infantry_spawn_unit_generation_uid_entries^infantry_spawn_muster_gui_scan_index > infantry_spawn_family_latest_generation_uid } }
				set_temp_variable = { infantry_spawn_family_latest_generation_uid = infantry_spawn_unit_generation_uid_entries^infantry_spawn_muster_gui_scan_index }
			}
		}
		add_to_temp_variable = { infantry_spawn_muster_gui_scan_index = constant:infantry_spawn_muster_gui_value.one }
	}
	set_temp_variable = { infantry_spawn_muster_gui_scan_index = constant:infantry_spawn_muster_gui_value.zero }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_muster_gui_scan_index value = infantry_spawn_lot_uid_entries^num compare = less_than } }
		if = {
			limit = {
				check_variable = { var = infantry_spawn_lot_family_id_entries^infantry_spawn_muster_gui_scan_index value = infantry_spawn_current_family_id compare = equals }
				check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_muster_gui_scan_index value = constant:infantry_spawn_lot_state.demobilized compare = less_than }
			}
			add_to_temp_variable = { infantry_spawn_family_live_lot_count = 1 }
			if = {
				limit = { check_variable = { infantry_spawn_lot_claimant_uid_entries^infantry_spawn_muster_gui_scan_index > 0 } }
				add_to_temp_variable = { infantry_spawn_family_claimant_lot_count = 1 }
				set_temp_variable = { infantry_spawn_lookup_claimant_uid = infantry_spawn_lot_claimant_uid_entries^infantry_spawn_muster_gui_scan_index }
				infantry_spawn_find_claimant_row = yes
				if = {
					limit = {
						check_variable = { infantry_spawn_claimant_row_index > constant:infantry_spawn_event.invalid_index }
						check_variable = { var = infantry_spawn_claimant_status_entries^infantry_spawn_claimant_row_index value = constant:infantry_spawn_claimant_status.retired compare = less_than }
						check_variable = { infantry_spawn_claimant_influence_entries^infantry_spawn_claimant_row_index > infantry_spawn_family_primary_claimant_influence }
					}
					set_temp_variable = { infantry_spawn_family_primary_claimant_uid = infantry_spawn_claimant_uid_entries^infantry_spawn_claimant_row_index }
					set_temp_variable = { infantry_spawn_family_primary_claimant_influence = infantry_spawn_claimant_influence_entries^infantry_spawn_claimant_row_index }
					set_temp_variable = { infantry_spawn_family_primary_claimant_archetype = infantry_spawn_claimant_archetype_entries^infantry_spawn_claimant_row_index }
				}
			}
			else = { add_to_temp_variable = { infantry_spawn_family_dispersible_lot_count = 1 } }
		}
		add_to_temp_variable = { infantry_spawn_muster_gui_scan_index = constant:infantry_spawn_muster_gui_value.one }
	}
	infantry_spawn_load_current_family_action_state = yes
	set_temp_variable = { infantry_spawn_family_current_pressure = infantry_spawn_family_live_division_count }
	multiply_temp_variable = { infantry_spawn_family_current_pressure = constant:infantry_spawn_family_pressure.per_live_division }
	set_temp_variable = { infantry_spawn_family_pressure_component = infantry_spawn_family_live_lot_count }
	multiply_temp_variable = { infantry_spawn_family_pressure_component = constant:infantry_spawn_family_pressure.per_unresolved_lot }
	add_to_temp_variable = { infantry_spawn_family_current_pressure = infantry_spawn_family_pressure_component }
	if = {
		limit = { check_variable = { infantry_spawn_family_claimant_lot_count > 0 } }
		multiply_temp_variable = { infantry_spawn_family_current_pressure = constant:infantry_spawn_family_pressure.claimant_multiplier }
	}
	if = { limit = { check_variable = { infantry_spawn_family_has_cantonment > 0 } } subtract_from_temp_variable = { infantry_spawn_family_current_pressure = constant:infantry_spawn_family_pressure.cantonment_relief } }
	if = { limit = { check_variable = { infantry_spawn_family_has_restricted_deployment > 0 } } subtract_from_temp_variable = { infantry_spawn_family_current_pressure = constant:infantry_spawn_family_pressure.restricted_relief } }
	if = { limit = { check_variable = { infantry_spawn_family_has_sustainment > 0 } } subtract_from_temp_variable = { infantry_spawn_family_current_pressure = constant:infantry_spawn_family_pressure.sustainment_relief } }
	if = { limit = { check_variable = { infantry_spawn_family_has_containment > 0 } } subtract_from_temp_variable = { infantry_spawn_family_current_pressure = constant:infantry_spawn_family_pressure.containment_relief } }
	clamp_temp_variable = { var = infantry_spawn_family_current_pressure min = constant:infantry_spawn_saturation.minimum max = constant:infantry_spawn_saturation.maximum }
	set_temp_variable = { infantry_spawn_family_current_sustainment = constant:infantry_spawn_family_sustainment_state.dormant }
	if = {
		limit = { check_variable = { infantry_spawn_family_live_division_count > 0 } }
		set_temp_variable = { infantry_spawn_family_current_sustainment = constant:infantry_spawn_family_sustainment_state.unsupported }
		if = {
			limit = { check_variable = { infantry_spawn_equipment_debt > constant:infantry_spawn_debt_threshold.modest } }
			set_temp_variable = { infantry_spawn_family_current_sustainment = constant:infantry_spawn_family_sustainment_state.strained }
		}
		if = {
			limit = { check_variable = { infantry_spawn_family_has_sustainment > 0 } }
			set_temp_variable = { infantry_spawn_family_current_sustainment = constant:infantry_spawn_family_sustainment_state.sustained }
		}
	}
}

					infantry_spawn_ai_dispersal_weight = { infantry_spawn_disperse_selected_anomalous_lot = yes }

				set_temp_variable = { infantry_spawn_ai_restricted_weight = constant:infantry_spawn_anomalous_ai_weight.ordinary_restricted_deployment }
