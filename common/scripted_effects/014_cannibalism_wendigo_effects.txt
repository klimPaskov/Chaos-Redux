# Selects the exact live Event 2 Wendigo country deterministically, preserves
# that original-ZZZ country in place, transfers one eligible human donor before

cannibalism_calculate_current_wendigo_merge_host_score = {
	set_temp_variable = { cannibalism_current_wendigo_merge_host_score = constant:cannibalism_wendigo_merge.base_host_score }
	if = {
		limit = { is_ai = no }
		add_to_temp_variable = { cannibalism_current_wendigo_merge_host_score = constant:cannibalism_wendigo_merge.human_host_score }
	}
	set_temp_variable = { cannibalism_wendigo_merge_score_component = num_divisions }
	multiply_temp_variable = { cannibalism_wendigo_merge_score_component = constant:cannibalism_wendigo_merge.division_score }
	add_to_temp_variable = { cannibalism_current_wendigo_merge_host_score = cannibalism_wendigo_merge_score_component }
	set_temp_variable = { cannibalism_wendigo_merge_score_component = num_of_controlled_states }
	multiply_temp_variable = { cannibalism_wendigo_merge_score_component = constant:cannibalism_wendigo_merge.controlled_state_score }
	add_to_temp_variable = { cannibalism_current_wendigo_merge_host_score = cannibalism_wendigo_merge_score_component }
	set_temp_variable = { cannibalism_wendigo_merge_controlled_population_k = 0 }
	every_controlled_state = {
		add_to_temp_variable = { cannibalism_wendigo_merge_controlled_population_k = state_population_k }
	}
	set_temp_variable = { cannibalism_wendigo_merge_score_component = cannibalism_wendigo_merge_controlled_population_k }
	divide_temp_variable = { cannibalism_wendigo_merge_score_component = constant:cannibalism_wendigo_merge.population_k_per_million }
	multiply_temp_variable = { cannibalism_wendigo_merge_score_component = constant:cannibalism_wendigo_merge.controlled_population_score_per_million }
	add_to_temp_variable = { cannibalism_current_wendigo_merge_host_score = cannibalism_wendigo_merge_score_component }
	clamp_temp_variable = {
		var = cannibalism_current_wendigo_merge_host_score
		min = constant:cannibalism_wendigo_merge.base_host_score
		max = constant:cannibalism_wendigo_merge.maximum_host_score
	}
}

cannibalism_select_wendigo_merge_host = {
	clear_global_event_target = cannibalism_wendigo_merge_host
	set_temp_variable = { cannibalism_best_wendigo_merge_host_score = 0 }
	every_country = {
		limit = { cannibalism_is_valid_wendigo_merge_host = yes }
		cannibalism_calculate_current_wendigo_merge_host_score = yes
		if = {
			limit = {
				check_variable = {
					var = cannibalism_current_wendigo_merge_host_score
					value = cannibalism_best_wendigo_merge_host_score
					compare = greater_than
				}
			}
			set_temp_variable = { cannibalism_best_wendigo_merge_host_score = cannibalism_current_wendigo_merge_host_score }
			save_global_event_target_as = cannibalism_wendigo_merge_host
		}
	}
	set_variable = { global.cannibalism_wendigo_merge_host_score = cannibalism_best_wendigo_merge_host_score }
}

cannibalism_calculate_current_wendigo_anchor_score = {
	set_temp_variable = { cannibalism_current_wendigo_anchor_score = constant:cannibalism_wendigo_anchor.base_score }
	if = {
		limit = { is_capital = yes }
		add_to_temp_variable = { cannibalism_current_wendigo_anchor_score = constant:cannibalism_wendigo_anchor.capital_score }
	}
	set_temp_variable = { cannibalism_wendigo_anchor_score_component = state_population_k }
	divide_temp_variable = { cannibalism_wendigo_anchor_score_component = constant:cannibalism_wendigo_anchor.population_score_divisor }
	add_to_temp_variable = { cannibalism_current_wendigo_anchor_score = cannibalism_wendigo_anchor_score_component }
	if = { limit = { is_coastal = yes } add_to_temp_variable = { cannibalism_current_wendigo_anchor_score = constant:cannibalism_wendigo_anchor.coastal_score } }
	if = { limit = { naval_base > 0 } add_to_temp_variable = { cannibalism_current_wendigo_anchor_score = constant:cannibalism_wendigo_anchor.naval_base_score } }
	if = { limit = { supply_node > 0 } add_to_temp_variable = { cannibalism_current_wendigo_anchor_score = constant:cannibalism_wendigo_anchor.supply_node_score } }
	if = { limit = { rail_way > 0 } add_to_temp_variable = { cannibalism_current_wendigo_anchor_score = constant:cannibalism_wendigo_anchor.rail_score } }
}

cannibalism_register_current_wendigo_anchor = {
	set_temp_variable = { cannibalism_wendigo_anchor_registration_result = 0 }
	if = {
		limit = {
			check_variable = {
				var = global.cannibalism_wendigo_active_anchor_count
				value = constant:cannibalism_wendigo_anchor.maximum
				compare = less_than
			}
			NOT = { has_state_flag = cannibalism_wendigo_transformation_anchor }
			NOT = { has_state_flag = cannibalism_wendigo_anchor_population_committed }
			NOT = { has_state_flag = cannibalism_recovery_active }
			NOT = { is_in_array = { array = global.cannibalism_transformation_anchor_states value = THIS } }
			OWNER = { tag = event_target:cannibalism_wendigo_merge_host }
			CONTROLLER = { tag = event_target:cannibalism_wendigo_merge_host }
			NOT = { cannibalism_state_is_unusable_larder = yes }
			NOT = { has_state_flag = cannibalism_consumption_cooldown }
			check_variable = {
				var = state_population_k
				value = constant:cannibalism_wendigo_anchor.minimum_creation_population_k
				compare = greater_than_or_equals
			}
		}
		save_event_target_as = cannibalism_consumption_target_state
		event_target:cannibalism_wendigo_merge_host = {
			set_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_wendigo_anchor.feeding_population_k }
			multiply_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_consumption.population_people_per_k }
			set_temp_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.silent_larder }
			cannibalism_prepare_consumption_context = yes
		}
		if = {
			limit = {
				check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied }
				check_variable = {
					var = cannibalism_population_loss_applied
					value = cannibalism_consumption_exact_request_people
					compare = equals
				}
			}
			set_state_flag = cannibalism_wendigo_transformation_anchor
			set_state_flag = cannibalism_wendigo_anchor_population_committed
			clr_state_flag = cannibalism_wendigo_anchor_broken
			set_variable = { cannibalism_wendigo_anchor_strength = constant:cannibalism_wendigo_anchor.initial_strength }
			set_variable = { cannibalism_wendigo_anchor_source_country = event_target:cannibalism_wendigo_merge_host }
			set_variable = { cannibalism_wendigo_anchor_source_generation = event_target:cannibalism_wendigo_merge_host.cannibalism_actor_generation }
			add_dynamic_modifier = { modifier = cannibalism_wendigo_transformation_anchor }
			add_to_array = { array = global.cannibalism_transformation_anchor_states value = THIS }
			set_temp_variable = { cannibalism_wendigo_anchor_registration_result = 1 }
		}
		set_temp_variable = { cannibalism_consumption_exact_request_people = 0 }
	}
}

cannibalism_select_next_initial_wendigo_anchor = {
	set_temp_variable = { cannibalism_wendigo_anchor_candidate_found = 0 }
	set_temp_variable = { cannibalism_best_wendigo_anchor_score = 0 }
	event_target:cannibalism_wendigo_merge_host = {
		every_controlled_state = {
			limit = {
				NOT = { has_state_flag = cannibalism_wendigo_transformation_anchor }
				NOT = { has_state_flag = cannibalism_wendigo_anchor_population_committed }
				NOT = { has_state_flag = cannibalism_recovery_active }
				NOT = { is_in_array = { array = global.cannibalism_transformation_anchor_states value = THIS } }
				NOT = { cannibalism_state_is_unusable_larder = yes }
				NOT = { has_state_flag = cannibalism_consumption_cooldown }
				check_variable = {
					var = state_population_k
					value = constant:cannibalism_wendigo_anchor.minimum_creation_population_k
					compare = greater_than_or_equals
				}
			}
			cannibalism_calculate_current_wendigo_anchor_score = yes
			if = {
				limit = {
					check_variable = {
						var = cannibalism_current_wendigo_anchor_score
						value = cannibalism_best_wendigo_anchor_score
						compare = greater_than
					}
				}
				set_temp_variable = { cannibalism_best_wendigo_anchor_score = cannibalism_current_wendigo_anchor_score }
				set_temp_variable = { cannibalism_wendigo_anchor_candidate_found = 1 }
				save_event_target_as = cannibalism_selected_wendigo_anchor_state
			}
		}
	}
	if = {
		limit = { check_variable = { cannibalism_wendigo_anchor_candidate_found > 0 } }
		event_target:cannibalism_selected_wendigo_anchor_state = {
			cannibalism_register_current_wendigo_anchor = yes
		}
	}
}

cannibalism_create_initial_wendigo_anchors = {
	resize_array = { global.cannibalism_transformation_anchor_states = 0 }
	set_temp_variable = { cannibalism_wendigo_anchor_candidate_found = 1 }
	while_loop_effect = {
		limit = {
			check_variable = { cannibalism_wendigo_anchor_candidate_found > 0 }
			check_variable = {
				var = global.cannibalism_transformation_anchor_states^num
				value = constant:cannibalism_wendigo_anchor.initial_target
				compare = less_than
			}
		}
		cannibalism_select_next_initial_wendigo_anchor = yes
	}
	set_variable = { global.cannibalism_wendigo_active_anchor_count = global.cannibalism_transformation_anchor_states^num }
	if = {
		limit = { has_event_target = cannibalism_wendigo_merge_host }
		event_target:cannibalism_wendigo_merge_host = { cannibalism_rebuild_wendigo_anchor_runtime = yes }
	}
}

			}
		}

		cannibalism_absorb_primary_warlord_donor_into_wendigo = yes
		event_target:cannibalism_wendigo_merge_host = {
			set_variable = { global.cannibalism_unification_host_generation = cannibalism_actor_generation }

			cannibalism_apply_chaos_change = yes
			cannibalism_emit_reveal_super_event = yes
		}

		event_target:cannibalism_wendigo_merge_host = { cannibalism_load_wendigo_focus_overlay = yes }

cannibalism_try_execute_wendigo_unification = {
	set_temp_variable = { cannibalism_wendigo_unification_result = 0 }
	if = {
		limit = {
			has_global_flag = cannibalism_evolution_iii_transition_due
			NOT = { has_global_flag = cannibalism_reveal_complete }
			NOT = { has_global_flag = world_end }
			cannibalism_valid_wendigo_merge_host_exists = yes
		}
		# This is the first route mutation. It blocks the legacy Event 2
		# terminal while the exact survivor is selected and transformed.
		set_global_flag = cannibalism_wendigo_transformation_pending
		cannibalism_select_wendigo_merge_host = yes
		cannibalism_select_unification_host = yes
		if = {
			limit = {
				cannibalism_selected_wendigo_merge_host_is_valid = yes
				cannibalism_selected_unification_host_is_valid = yes
			}
			cannibalism_create_wendigo_unification_from_selected_hosts = yes
		}
		else = {
			clr_global_flag = cannibalism_wendigo_transformation_pending
			clear_global_event_target = cannibalism_wendigo_merge_host
		}
	}
}

cannibalism_break_loaded_wendigo_anchor = {
	if = {
		limit = { has_state_flag = cannibalism_wendigo_transformation_anchor }
		clr_state_flag = cannibalism_wendigo_transformation_anchor
		set_state_flag = cannibalism_wendigo_anchor_broken
		remove_dynamic_modifier = { modifier = cannibalism_wendigo_transformation_anchor }
		set_temp_variable = { cannibalism_wendigo_broken_anchor_recovery_days = constant:cannibalism_wendigo_anchor.broken_anchor_recovery_days }
		add_dynamic_modifier = {
			modifier = cannibalism_wendigo_broken_anchor_recovery
			days = cannibalism_wendigo_broken_anchor_recovery_days
		}
		clear_variable = cannibalism_wendigo_anchor_strength
		clear_variable = cannibalism_wendigo_anchor_source_country
		clear_variable = cannibalism_wendigo_anchor_source_generation
		if = {
			limit = { NOT = { is_in_array = { array = global.cannibalism_recovery_states value = THIS } } }
			add_to_array = { array = global.cannibalism_recovery_states value = THIS }
		}
		set_state_flag = cannibalism_recovery_active
		set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.liberated_emergency }
		cannibalism_set_current_state_stage = yes
		set_variable = { cannibalism_recovery_next_date = global.date }
		add_to_variable = { cannibalism_recovery_next_date = constant:cannibalism_timing.liberated_emergency_days }
		if = {
			limit = { has_event_target = cannibalism_wendigo_merge_host }
			event_target:cannibalism_wendigo_merge_host = {
				add_to_variable = { cannibalism_wendigo_broken_anchor_count = 1 }
				subtract_from_variable = { cannibalism_wendigo_transformation_progress = constant:cannibalism_wendigo_anchor.loss_progress_penalty }
				clamp_variable = {
					var = cannibalism_wendigo_transformation_progress
					min = constant:cannibalism_wendigo_countdown.progress_min
					max = constant:cannibalism_wendigo_countdown.progress_max
				}
			}
		}
		CONTROLLER = {
			if = {
				limit = {
					exists = yes
					NOT = { is_cannibalism_wendigo_hannibal_country = yes }
				}
				country_event = { id = chaosx.nr14.73 hours = @CANNIBALISM_WENDIGO_STAGE_EVENT_HOURS }
			}
		}
	}
}

cannibalism_rebuild_wendigo_anchor_runtime = {
	set_temp_variable = { cannibalism_wendigo_anchor_rebuild_index = global.cannibalism_transformation_anchor_states^num }
	set_temp_variable = { cannibalism_wendigo_live_anchor_count = 0 }
	while_loop_effect = {
		limit = { check_variable = { cannibalism_wendigo_anchor_rebuild_index > 0 } }
		subtract_from_temp_variable = { cannibalism_wendigo_anchor_rebuild_index = 1 }
		set_temp_variable = { cannibalism_wendigo_loaded_anchor_is_live = 0 }
		var:global.cannibalism_transformation_anchor_states^cannibalism_wendigo_anchor_rebuild_index = {
			if = {
				limit = { cannibalism_current_wendigo_anchor_is_live = yes }
				set_temp_variable = { cannibalism_wendigo_loaded_anchor_is_live = 1 }
				add_to_temp_variable = { cannibalism_wendigo_live_anchor_count = 1 }
			}
			else = {
				cannibalism_break_loaded_wendigo_anchor = yes
			}
		}
		if = {
			limit = { check_variable = { cannibalism_wendigo_loaded_anchor_is_live < 1 } }
			remove_from_array = {
				array = global.cannibalism_transformation_anchor_states
				index = cannibalism_wendigo_anchor_rebuild_index
			}
		}
	}
	set_variable = { global.cannibalism_wendigo_active_anchor_count = cannibalism_wendigo_live_anchor_count }
	set_variable = { cannibalism_wendigo_pack_capacity = constant:cannibalism_wendigo_decision.train_pack_base_cap }
	set_temp_variable = { cannibalism_wendigo_pack_capacity_component = cannibalism_wendigo_live_anchor_count }
	multiply_temp_variable = { cannibalism_wendigo_pack_capacity_component = constant:cannibalism_wendigo_decision.train_pack_cap_per_anchor }
	add_to_variable = { cannibalism_wendigo_pack_capacity = cannibalism_wendigo_pack_capacity_component }
	if = {
		limit = { has_variable = cannibalism_wendigo_focus_pack_capacity_bonus }
		add_to_variable = { cannibalism_wendigo_pack_capacity = cannibalism_wendigo_focus_pack_capacity_bonus }
	}
}

cannibalism_calculate_wendigo_countdown_progress_change = {
	set_temp_variable = { cannibalism_wendigo_countdown_progress_change = constant:cannibalism_wendigo_countdown.weekly_base }
	set_temp_variable = { cannibalism_wendigo_countdown_component = global.cannibalism_wendigo_active_anchor_count }
	multiply_temp_variable = { cannibalism_wendigo_countdown_component = constant:cannibalism_wendigo_countdown.progress_per_anchor }
	add_to_temp_variable = { cannibalism_wendigo_countdown_progress_change = cannibalism_wendigo_countdown_component }
	set_temp_variable = { cannibalism_wendigo_countdown_component = cannibalism_wendigo_winter_victories }
	multiply_temp_variable = { cannibalism_wendigo_countdown_component = constant:cannibalism_wendigo_countdown.progress_per_winter_victory }
	clamp_temp_variable = {
		var = cannibalism_wendigo_countdown_component
		min = constant:cannibalism_wendigo_countdown.progress_min
		max = constant:cannibalism_wendigo_countdown.maximum_victory_progress
	}
	add_to_temp_variable = { cannibalism_wendigo_countdown_progress_change = cannibalism_wendigo_countdown_component }
	if = {
		limit = { check_variable = { cannibalism_larder > constant:cannibalism_wendigo_countdown.larder_high } }
		add_to_temp_variable = { cannibalism_wendigo_countdown_progress_change = constant:cannibalism_wendigo_countdown.larder_bonus }
	}
	if = {
		limit = { check_variable = { global.cannibalism_consumed_population_k > constant:cannibalism_wendigo_countdown.consumed_population_k } }
		add_to_temp_variable = { cannibalism_wendigo_countdown_progress_change = constant:cannibalism_wendigo_countdown.consumption_bonus }
	}
	if = {
		limit = { capital_scope = { is_controlled_by = PREV } }
		add_to_temp_variable = { cannibalism_wendigo_countdown_progress_change = constant:cannibalism_wendigo_countdown.progress_per_controlled_capital }
	}
	set_temp_variable = { cannibalism_wendigo_countdown_component = cannibalism_wendigo_broken_anchor_count }
	multiply_temp_variable = { cannibalism_wendigo_countdown_component = constant:cannibalism_wendigo_countdown.broken_anchor_penalty }
	clamp_temp_variable = {
		var = cannibalism_wendigo_countdown_component
		min = constant:cannibalism_wendigo_countdown.progress_min
		max = constant:cannibalism_wendigo_countdown.maximum_broken_penalty
	}
	subtract_from_temp_variable = { cannibalism_wendigo_countdown_progress_change = cannibalism_wendigo_countdown_component }
}

cannibalism_process_wendigo_transformation_pulse = {
	if = {
		limit = {
			is_cannibalism_wendigo_hannibal_country = yes
			NOT = { has_country_flag = cannibalism_wendigo_transformation_locked }
			NOT = { has_global_flag = world_end }

		cannibalism_rebuild_wendigo_anchor_runtime = yes
		cannibalism_break_wendigo_transformation_route = yes
		cannibalism_try_start_wendigo_countdown = yes
		if = {
			limit = { has_country_flag = cannibalism_wendigo_countdown_active }

				limit = { cannibalism_wendigo_can_start_countdown = yes }
				cannibalism_calculate_wendigo_countdown_progress_change = yes
				add_to_variable = { cannibalism_wendigo_transformation_progress = cannibalism_wendigo_countdown_progress_change }

				subtract_from_variable = { cannibalism_wendigo_transformation_progress = constant:cannibalism_wendigo_countdown.broken_anchor_penalty }
			}
			clamp_variable = {
				var = cannibalism_wendigo_transformation_progress
				min = constant:cannibalism_wendigo_countdown.progress_min
				max = constant:cannibalism_wendigo_countdown.progress_max
			}
			cannibalism_complete_wendigo_terminal_lock = yes
		}
	}
