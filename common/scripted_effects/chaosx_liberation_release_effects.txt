# ============================================================================
# CHAOS REDUX - LIBERATIONS RELEASE COORDINATOR EFFECTS
# ============================================================================
# Overview:
# - Owns the shared frozen-plan lifecycle for Event 005 and Event 006.
# - Stores exact country, state, host, protected-state, original-capital, and
#   rejection ledgers.
# - Performs mutation-free lock validation and a final live revalidation before
#   either event may change ownership.
# - Does not select Event 006 packages or execute event-specific country setup;
#   those remain in the owning event's files.
# - Never performs broad periodic world iteration.
# ============================================================================

# Plan cleanup and lifecycle

liberation_release_clear_plan_scope_marks = {
	for_each_scope_loop = {
		array = global.liberation_plan_states
		clr_state_flag = liberation_release_reserved_state
		clear_variable = liberation_release_plan_id
		clear_variable = liberation_release_package_id
		clear_variable = liberation_release_plan_owner
		clear_variable = liberation_release_state_role
		clear_variable = liberation_release_original_owner
		clear_variable = liberation_release_original_controller
		clear_variable = liberation_release_target_country
		clr_state_flag = liberation_release_core_snapshot_recorded
		clr_state_flag = liberation_release_original_owner_had_core
		clr_state_flag = liberation_release_target_had_core
	}
	for_each_scope_loop = {
		array = global.liberation_plan_host_protected_states
		clr_state_flag = liberation_release_protected_host_state
		clear_variable = liberation_release_plan_id
		clear_variable = liberation_release_protected_host
	}
	for_each_scope_loop = {
		array = global.liberation_plan_hosts
		clr_country_flag = liberation_release_plan_host
		clear_variable = liberation_release_plan_id
		clear_variable = liberation_release_planned_loss_count
		clear_variable = liberation_release_protected_state
		clear_variable = liberation_release_original_capital
	}
	for_each_scope_loop = {
		array = global.liberation_plan_countries
		clr_country_flag = liberation_release_reserved_country
		clear_variable = liberation_release_plan_id
		clear_variable = liberation_release_plan_owner
		clear_variable = liberation_release_package_id
	}
}

# Inputs (temporary variables):
# - liberation_call_mode
# - liberation_call_expected_country_count
# - liberation_call_plan_owner (Soviet Collapse, Independence Wave, or joint)
liberation_release_begin_plan = {
	if = {
		limit = { can_liberation_release_begin_plan = yes }
		liberation_release_clear_plan_scope_marks = yes
		liberation_release_clear_plan_arrays = yes
		clr_global_flag = liberation_release_plan_valid
		clr_global_flag = liberation_release_plan_includes_event5
		clr_global_flag = liberation_release_plan_includes_event6
		clr_global_flag = liberation_release_joint_plan_executed
		clr_global_flag = liberation_release_execution_started
		clr_global_flag = liberation_release_capital_restore_failed
		clr_global_flag = liberation_release_plan_rollback_completed
		clr_global_flag = liberation_release_plan_rollback_failed
		clr_global_flag = liberation_release_event5_rollback_cleanup_validated
		clr_global_flag = liberation_release_event6_rollback_cleanup_validated
		clr_global_flag = liberation_release_event5_finalizer_started
		clr_global_flag = liberation_release_finalizer_started
		clr_global_flag = liberation_release_plan_finalization_failed
		clr_global_flag = liberation_release_event5_finalization_validated
		clr_global_flag = liberation_release_event6_finalization_validated
		clr_global_flag = liberation_release_event6_origin_commit_started
		clear_variable = global.liberation_plan_rollback_failure
		clear_variable = global.liberation_plan_finalization_failure
		clear_variable = global.liberation_plan_last_failure
		if = {
			limit = { NOT = { has_variable = global.liberation_next_plan_id } }
			set_variable = { global.liberation_next_plan_id = constant:liberation_release_limit.minimum }
		}
		add_to_variable = { global.liberation_next_plan_id = constant:liberation_release_limit.one }
		set_variable = { global.liberation_plan_id = global.liberation_next_plan_id }
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.collecting }
		set_variable = { global.liberation_plan_mode = liberation_call_mode }
		set_variable = { global.liberation_plan_owner = liberation_call_plan_owner }
		set_variable = { global.liberation_plan_expected_country_count = liberation_call_expected_country_count }
		set_variable = { global.liberation_plan_started_date = global.date }
		set_variable = { global.liberation_plan_source_country = THIS }
		if = {
			limit = {
				OR = {
					check_variable = { liberation_call_plan_owner = constant:liberation_plan_owner.soviet_collapse }
					check_variable = { liberation_call_plan_owner = constant:liberation_plan_owner.joint }
				}
			}
			set_global_flag = liberation_release_plan_includes_event5
		}
		if = {
			limit = {
				OR = {
					check_variable = { liberation_call_plan_owner = constant:liberation_plan_owner.independence_wave }
					check_variable = { liberation_call_plan_owner = constant:liberation_plan_owner.joint }
				}
			}
			set_global_flag = liberation_release_plan_includes_event6
		}
	}
	else_if = {
		limit = { can_liberation_release_reset_plan = yes }
		set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.invalid_scope }
	}
	else = {
		set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.stale_plan }
	}
}

liberation_release_enter_allocation_phase = {
	if = {
		limit = {
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.collecting }
			OR = {
				has_global_flag = liberation_release_plan_includes_event5
				has_global_flag = liberation_release_plan_includes_event6
			}
		}
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.allocating }
	}
}

}

# Reads event target `liberation_candidate_state_host` and returns the
# temporary boolean `liberation_candidate_host_has_loss_capacity`. The host's
# protected remnant consumes the final owned-state slot.
liberation_release_measure_candidate_host_loss_capacity = {
	set_temp_variable = { liberation_candidate_host_has_loss_capacity = constant:liberation_release_limit.minimum }
	if = {
		limit = { has_event_target = liberation_candidate_state_host }
		event_target:liberation_candidate_state_host = {
			set_temp_variable = { liberation_candidate_host_loss_ceiling = num_owned_states }
			subtract_from_temp_variable = { liberation_candidate_host_loss_ceiling = constant:liberation_release_limit.one }
			if = {
				limit = {
					check_variable = {
						var = liberation_release_planned_loss_count
						value = liberation_candidate_host_loss_ceiling
						compare = less_than
					}
				}
				set_temp_variable = { liberation_candidate_host_has_loss_capacity = constant:liberation_release_limit.one }
			}
		}
	}
}

# COUNTRY scope. Selects the accepted host-survival priority (existing row,
# owned/controlled capital, controlled core, controlled state, uncontrolled
# core, then any owned state) and reserves it.
# Output: liberation_candidate_reject_reason.
liberation_release_select_first_host_state_candidate = {
	set_temp_variable = { liberation_host_state_selection_complete = constant:liberation_release_limit.minimum }
	for_each_scope_loop = {
		array = liberation_host_state_candidates
		break = liberation_host_state_selection_complete
		save_event_target_as = liberation_candidate_protected_state
		set_temp_variable = { liberation_host_state_selection_complete = constant:liberation_release_limit.one }
	}
}

liberation_release_select_and_reserve_host_state = {
	save_event_target_as = liberation_candidate_host
	set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.none }
	if = {
		limit = {
			is_in_array = { array = global.liberation_plan_hosts value = THIS }
			has_variable = liberation_release_protected_state
			check_variable = { var = liberation_release_plan_id value = global.liberation_plan_id compare = equals }
			var:liberation_release_protected_state = { is_liberation_release_host_protection_candidate = yes }
		}
		var:liberation_release_protected_state = { save_event_target_as = liberation_candidate_protected_state }
		set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.existing }
	}
	else_if = {
		limit = {
			capital_scope = {
				is_owned_by = PREV
				is_controlled_by = PREV
				is_liberation_release_host_protection_candidate = yes
			}
		}
		capital_scope = { save_event_target_as = liberation_candidate_protected_state }
		set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.capital }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_core_of = PREV
				is_controlled_by = PREV
				is_liberation_release_host_protection_candidate = yes
			}
		}
		clear_temp_array = liberation_host_state_candidates
		every_owned_state = {
			limit = {
				is_core_of = PREV
				is_controlled_by = PREV
				is_liberation_release_host_protection_candidate = yes
			}
			add_to_temp_array = { liberation_host_state_candidates = THIS }
		}
		liberation_release_select_first_host_state_candidate = yes
		clear_temp_array = liberation_host_state_candidates
		set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.owned_core }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_controlled_by = PREV
				is_liberation_release_host_protection_candidate = yes
			}
		}
		clear_temp_array = liberation_host_state_candidates
		every_owned_state = {
			limit = {
				is_controlled_by = PREV
				is_liberation_release_host_protection_candidate = yes
			}
			add_to_temp_array = { liberation_host_state_candidates = THIS }
		}
		liberation_release_select_first_host_state_candidate = yes
		clear_temp_array = liberation_host_state_candidates
		set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.owned_state }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_core_of = PREV
				is_liberation_release_host_protection_candidate = yes
			}
		}
		clear_temp_array = liberation_host_state_candidates
		every_owned_state = {
			limit = {
				is_core_of = PREV
				is_liberation_release_host_protection_candidate = yes
			}
			add_to_temp_array = { liberation_host_state_candidates = THIS }
		}
		liberation_release_select_first_host_state_candidate = yes
		clear_temp_array = liberation_host_state_candidates
		set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.owned_core }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_liberation_release_host_protection_candidate = yes
			}
		}
		clear_temp_array = liberation_host_state_candidates
		every_owned_state = {
			limit = {
				is_liberation_release_host_protection_candidate = yes
			}
			add_to_temp_array = { liberation_host_state_candidates = THIS }
		}
		liberation_release_select_first_host_state_candidate = yes
		clear_temp_array = liberation_host_state_candidates
		set_temp_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.owned_state }
	}
	if = {
		limit = { NOT = { check_variable = { liberation_host_protection_source = constant:liberation_host_protection_source.none } } }
		liberation_release_reserve_host_state = yes
	}
	else = {
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.host_survival }
	}
}

# Inputs (event targets): liberation_candidate_host and
# liberation_candidate_protected_state. Output: liberation_candidate_reject_reason.
liberation_release_reserve_host_state = {
	set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.invalid_scope }
	if = {
		limit = {
			is_liberation_release_plan_open = yes
			has_event_target = liberation_candidate_host
			has_event_target = liberation_candidate_protected_state
			event_target:liberation_candidate_host = {
				exists = yes
				is_in_array = { array = global.liberation_plan_hosts value = THIS }
				has_country_flag = liberation_release_plan_host
				check_variable = { var = liberation_release_plan_id value = global.liberation_plan_id compare = equals }
				check_variable = { var = liberation_release_protected_state value = event_target:liberation_candidate_protected_state compare = equals }
			}
			event_target:liberation_candidate_protected_state = {
				is_owned_by = event_target:liberation_candidate_host
				is_liberation_release_current_protected_state = yes
				OR = {
					NOT = { has_event_target = liberation_candidate_anchor }
					NOT = { state = event_target:liberation_candidate_anchor }
				}
			}
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none }
	}
	else_if = {
		limit = {
			is_liberation_release_plan_open = yes
			has_event_target = liberation_candidate_host
			has_event_target = liberation_candidate_protected_state
			event_target:liberation_candidate_host = {
				exists = yes
				NOT = { is_in_array = { array = global.liberation_plan_hosts value = THIS } }
			}
			event_target:liberation_candidate_protected_state = {
				is_owned_by = event_target:liberation_candidate_host
				is_liberation_release_host_protection_candidate = yes
			}
		}
		event_target:liberation_candidate_host = {
			set_temp_variable = { liberation_host_owned_state_snapshot = constant:liberation_release_limit.minimum }
			every_owned_state = {
				add_to_temp_variable = { liberation_host_owned_state_snapshot = constant:liberation_release_limit.one }
			}
			add_to_array = { array = global.liberation_plan_hosts value = THIS }
			add_to_array = { array = global.liberation_plan_host_owned_state_snapshots value = liberation_host_owned_state_snapshot }
			add_to_array = { array = global.liberation_plan_host_protected_states value = event_target:liberation_candidate_protected_state }
			capital_scope = {
				add_to_array = { array = global.liberation_plan_host_original_capitals value = THIS }
				save_event_target_as = liberation_candidate_original_capital
			}
			set_variable = { liberation_release_original_capital = event_target:liberation_candidate_original_capital }
			set_country_flag = liberation_release_plan_host
			set_variable = { liberation_release_plan_id = global.liberation_plan_id }
			set_variable = { liberation_release_planned_loss_count = constant:liberation_release_limit.minimum }
			set_variable = { liberation_release_protected_state = event_target:liberation_candidate_protected_state }
		}
		event_target:liberation_candidate_protected_state = {
			set_state_flag = liberation_release_protected_host_state
			set_variable = { liberation_release_plan_id = global.liberation_plan_id }
			set_variable = { liberation_release_protected_host = event_target:liberation_candidate_host }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none }
	}
	else_if = {
		limit = { NOT = { is_liberation_release_plan_open = yes } }
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.stale_plan }
	}
	else_if = {
		limit = {
			has_event_target = liberation_candidate_host
			event_target:liberation_candidate_host = {
				is_in_array = { array = global.liberation_plan_hosts value = THIS }
			}
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.protected_state_mismatch }
	}
}

# Inputs:
# - event targets liberation_candidate_country, liberation_candidate_anchor,
#   and liberation_candidate_primary_host
# - temporary variables liberation_candidate_package_id,
#   liberation_candidate_plan_owner, liberation_candidate_reservation_group,
#   liberation_candidate_territory_level, liberation_candidate_force_level
# Output: liberation_candidate_reject_reason.
liberation_release_add_country_reservation = {
	set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.invalid_scope }
	if = {
		limit = {
			is_liberation_release_plan_open = yes
			is_liberation_release_candidate_owner_allowed = yes
			has_event_target = liberation_candidate_country
			has_event_target = liberation_candidate_anchor
			has_event_target = liberation_candidate_primary_host
			event_target:liberation_candidate_country = {
				NOT = { exists = yes }
				NOT = { is_in_array = { array = global.liberation_plan_countries value = THIS } }
			}
			NOT = { is_in_array = { array = global.liberation_plan_package_ids value = liberation_candidate_package_id } }
			event_target:liberation_candidate_anchor = {
				is_owned_by = event_target:liberation_candidate_primary_host
				NOT = { is_liberation_release_current_reserved_state = yes }
				NOT = { is_liberation_release_current_protected_state = yes }
			}
			event_target:liberation_candidate_primary_host = {
				has_country_flag = liberation_release_plan_host
				check_variable = { var = liberation_release_plan_id value = global.liberation_plan_id compare = equals }
			}
			OR = {
				check_variable = { liberation_candidate_reservation_group = constant:liberation_release_limit.minimum }
				NOT = { is_in_array = { array = global.liberation_plan_reservation_groups value = liberation_candidate_reservation_group } }
				# The accepted Event 006 Rhineland/Saar pair shares a coarse
				# reservation group but has distinct anchors and hosts. Allow the
				# counterpart only after its sibling package is already reserved;
				# all other groups retain the one-package rule.
				AND = {
					check_variable = { liberation_candidate_reservation_group = constant:independence_wave_reservation_group_id.rg_rhine_saar }
					OR = {
						AND = {
							check_variable = { liberation_candidate_package_id = constant:independence_wave_package_id.iw_008 }
							is_in_array = { array = global.liberation_plan_package_ids value = constant:independence_wave_package_id.iw_010 }
						}
						AND = {
							check_variable = { liberation_candidate_package_id = constant:independence_wave_package_id.iw_010 }
							is_in_array = { array = global.liberation_plan_package_ids value = constant:independence_wave_package_id.iw_008 }
						}
					}
				}
			}
		}
		add_to_array = { array = global.liberation_plan_package_ids value = liberation_candidate_package_id }
		add_to_array = { array = global.liberation_plan_country_owners value = liberation_candidate_plan_owner }
		add_to_array = { array = global.liberation_plan_countries value = event_target:liberation_candidate_country }
		add_to_array = { array = global.liberation_plan_anchor_states value = event_target:liberation_candidate_anchor }
		add_to_array = { array = global.liberation_plan_primary_hosts value = event_target:liberation_candidate_primary_host }
		add_to_array = { array = global.liberation_plan_reservation_groups value = liberation_candidate_reservation_group }
		add_to_array = { array = global.liberation_plan_territory_levels value = liberation_candidate_territory_level }
		add_to_array = { array = global.liberation_plan_force_levels value = liberation_candidate_force_level }
		event_target:liberation_candidate_country = {
			set_country_flag = liberation_release_reserved_country
			set_variable = { liberation_release_plan_id = global.liberation_plan_id }
			set_variable = { liberation_release_plan_owner = liberation_candidate_plan_owner }
			set_variable = { liberation_release_package_id = liberation_candidate_package_id }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none }
	}
	else_if = {
		limit = { NOT = { is_liberation_release_plan_open = yes } }
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.stale_plan }
	}
	else_if = {
		limit = { is_in_array = { array = global.liberation_plan_package_ids value = liberation_candidate_package_id } }
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.package_already_selected }
	}
	else_if = {
		limit = {
			has_event_target = liberation_candidate_country
			event_target:liberation_candidate_country = { exists = yes }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.living_tag }
	}
	else_if = {
		limit = {
			has_event_target = liberation_candidate_anchor
			event_target:liberation_candidate_anchor = { is_liberation_release_current_protected_state = yes }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.host_survival }
	}
	else_if = {
		limit = {
			has_event_target = liberation_candidate_anchor
			event_target:liberation_candidate_anchor = { is_liberation_release_current_reserved_state = yes }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.state_reserved }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { liberation_candidate_reservation_group = constant:liberation_release_limit.minimum } }
			is_in_array = { array = global.liberation_plan_reservation_groups value = liberation_candidate_reservation_group }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.reservation_group_taken }
	}

# STATE scope. Inputs are event targets liberation_candidate_country and
# liberation_candidate_state_host plus liberation_candidate_package_id,
# liberation_candidate_plan_owner, and liberation_candidate_state_role.
# Output: liberation_candidate_reject_reason.
liberation_release_add_state_reservation = {
	set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.invalid_scope }
	if = {
		limit = {
			is_liberation_release_plan_open = yes
			is_liberation_release_candidate_owner_allowed = yes
			has_event_target = liberation_candidate_country
			has_event_target = liberation_candidate_state_host
			is_liberation_release_state_available_for_reservation = yes
			is_owned_by = event_target:liberation_candidate_state_host
			event_target:liberation_candidate_state_host = {
				has_country_flag = liberation_release_plan_host
				has_variable = liberation_release_planned_loss_count
				check_variable = { var = liberation_release_plan_id value = global.liberation_plan_id compare = equals }
			}
			event_target:liberation_candidate_country = {
				is_in_array = { array = global.liberation_plan_countries value = THIS }
				has_country_flag = liberation_release_reserved_country
				check_variable = { var = liberation_release_plan_id value = global.liberation_plan_id compare = equals }
				check_variable = { var = liberation_release_plan_owner value = liberation_candidate_plan_owner compare = equals }
				check_variable = { var = liberation_release_package_id value = liberation_candidate_package_id compare = equals }
			}
		}
		add_to_array = { array = global.liberation_plan_states value = THIS }
		add_to_array = { array = global.liberation_plan_state_package_ids value = liberation_candidate_package_id }
		add_to_array = { array = global.liberation_plan_state_country_owners value = liberation_candidate_plan_owner }
		add_to_array = { array = global.liberation_plan_state_target_countries value = event_target:liberation_candidate_country }
		add_to_array = { array = global.liberation_plan_state_hosts value = event_target:liberation_candidate_state_host }
		add_to_array = { array = global.liberation_plan_state_roles value = liberation_candidate_state_role }
		set_state_flag = liberation_release_reserved_state
		set_variable = { liberation_release_plan_id = global.liberation_plan_id }
		set_variable = { liberation_release_package_id = liberation_candidate_package_id }
		set_variable = { liberation_release_plan_owner = liberation_candidate_plan_owner }
		set_variable = { liberation_release_state_role = liberation_candidate_state_role }
		set_variable = { liberation_release_original_owner = event_target:liberation_candidate_state_host }
		set_variable = { liberation_release_original_controller = CONTROLLER }
		set_variable = { liberation_release_target_country = event_target:liberation_candidate_country }
		set_state_flag = liberation_release_core_snapshot_recorded
		clr_state_flag = liberation_release_original_owner_had_core
		clr_state_flag = liberation_release_target_had_core
		if = {
			limit = { is_core_of = event_target:liberation_candidate_state_host }
			set_state_flag = liberation_release_original_owner_had_core
		}
		if = {
			limit = { is_core_of = event_target:liberation_candidate_country }
			set_state_flag = liberation_release_target_had_core
		}
		event_target:liberation_candidate_state_host = {
			add_to_variable = { liberation_release_planned_loss_count = constant:liberation_release_limit.one }
		}
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none }
	}
	else_if = {
		limit = { NOT = { is_liberation_release_plan_open = yes } }
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.stale_plan }
	}
	else_if = {
		limit = { is_liberation_release_current_protected_state = yes }
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.host_survival }
	}
	else_if = {
		limit = { is_liberation_release_current_reserved_state = yes }
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.state_reserved }
	}
}

# Removes the most recently appended state row. Candidate planners call this
# only while their own provisional state rows remain at the tail of every
# aligned state array.
liberation_release_rollback_last_state_reservation = {
	if = {
		limit = {
			check_variable = {
				var = global.liberation_plan_states^num
				value = constant:liberation_release_limit.minimum
				compare = greater_than
			}
		}
		set_temp_variable = { liberation_rollback_state_index = global.liberation_plan_states^num }
		subtract_from_temp_variable = { liberation_rollback_state_index = constant:liberation_release_limit.one }
		var:global.liberation_plan_states^liberation_rollback_state_index = {
			save_event_target_as = liberation_rollback_state
		}
		var:global.liberation_plan_state_hosts^liberation_rollback_state_index = {
			save_event_target_as = liberation_rollback_state_host
		}
		if = {
			limit = { has_event_target = liberation_rollback_state }
			event_target:liberation_rollback_state = {
				clr_state_flag = liberation_release_reserved_state
				clear_variable = liberation_release_plan_id
				clear_variable = liberation_release_package_id
				clear_variable = liberation_release_plan_owner
				clear_variable = liberation_release_state_role
				clear_variable = liberation_release_original_owner
				clear_variable = liberation_release_original_controller
				clear_variable = liberation_release_target_country
				clr_state_flag = liberation_release_core_snapshot_recorded
				clr_state_flag = liberation_release_original_owner_had_core
				clr_state_flag = liberation_release_target_had_core
			}
		}
		if = {
			limit = { has_event_target = liberation_rollback_state_host }
			event_target:liberation_rollback_state_host = {
				subtract_from_variable = { liberation_release_planned_loss_count = constant:liberation_release_limit.one }
				if = {
					limit = {
						check_variable = {
							var = liberation_release_planned_loss_count
							value = constant:liberation_release_limit.minimum
							compare = less_than
						}
					}
					set_variable = { liberation_release_planned_loss_count = constant:liberation_release_limit.minimum }
				}
			}
		}
		remove_from_array = { array = global.liberation_plan_states }
		remove_from_array = { array = global.liberation_plan_state_package_ids }
		remove_from_array = { array = global.liberation_plan_state_country_owners }
		remove_from_array = { array = global.liberation_plan_state_target_countries }
		remove_from_array = { array = global.liberation_plan_state_hosts }
		remove_from_array = { array = global.liberation_plan_state_roles }
	}
}

# Input: liberation_candidate_state_row_start, captured before the candidate's
# first state is appended.
liberation_release_rollback_candidate_state_reservations = {
	if = {
		limit = {
			check_variable = { var = liberation_candidate_state_row_start value = constant:liberation_release_limit.minimum compare = greater_than_or_equals }
			check_variable = { var = liberation_candidate_state_row_start value = global.liberation_plan_states^num compare = less_than_or_equals }
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					var = global.liberation_plan_states^num
					value = liberation_candidate_state_row_start
					compare = greater_than
				}
			}
			liberation_release_rollback_last_state_reservation = yes
		}
	}
	else = {
		set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.invalid_scope }
	}
}

# Removes host-protection rows that no accepted country row references and that
# have no remaining provisional state losses. The reverse index walk preserves
# alignment while removing arbitrary unused rows.
liberation_release_cleanup_unused_host_reservations = {
	set_temp_variable = { liberation_rollback_host_index = global.liberation_plan_hosts^num }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = liberation_rollback_host_index
				value = constant:liberation_release_limit.minimum
				compare = greater_than
			}
		}
		subtract_from_temp_variable = { liberation_rollback_host_index = constant:liberation_release_limit.one }
		var:global.liberation_plan_hosts^liberation_rollback_host_index = {
			save_event_target_as = liberation_rollback_host
		}
		var:global.liberation_plan_host_protected_states^liberation_rollback_host_index = {
			save_event_target_as = liberation_rollback_protected_state
		}
		if = {
			limit = {
				has_event_target = liberation_rollback_host
				NOT = { is_in_array = { array = global.liberation_plan_primary_hosts value = event_target:liberation_rollback_host } }
				event_target:liberation_rollback_host = {
					check_variable = {
						var = liberation_release_planned_loss_count
						value = constant:liberation_release_limit.minimum
						compare = equals
					}
				}
			}
			event_target:liberation_rollback_host = {
				clr_country_flag = liberation_release_plan_host
				clear_variable = liberation_release_plan_id
				clear_variable = liberation_release_planned_loss_count
				clear_variable = liberation_release_protected_state
				clear_variable = liberation_release_original_capital
			}
			if = {
				limit = { has_event_target = liberation_rollback_protected_state }
				event_target:liberation_rollback_protected_state = {
					clr_state_flag = liberation_release_protected_host_state
					clear_variable = liberation_release_plan_id
					clear_variable = liberation_release_protected_host
				}
			}
			remove_from_array = {
				array = global.liberation_plan_hosts
				index = liberation_rollback_host_index
			}
			remove_from_array = {
				array = global.liberation_plan_host_owned_state_snapshots
				index = liberation_rollback_host_index
			}
			remove_from_array = {
				array = global.liberation_plan_host_protected_states
				index = liberation_rollback_host_index
			}
			remove_from_array = {
				array = global.liberation_plan_host_original_capitals
				index = liberation_rollback_host_index
			}
		}
	}
}

	if = {
		limit = {

							has_global_flag = liberation_release_plan_includes_event5
							NOT = { has_global_flag = liberation_release_plan_includes_event6 }

							check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.independence_wave }
							has_global_flag = liberation_release_plan_includes_event6
							NOT = { has_global_flag = liberation_release_plan_includes_event5 }

							has_global_flag = liberation_release_plan_includes_event5

			}
		}

	}
}

# Restores the capital snapshot for every host if a locked plan is cancelled
# before execution begins. This makes capital preparation reversible even when
# a later host fails relocation or final live validation.
liberation_release_restore_host_capitals_before_execution = {
	clr_global_flag = liberation_release_capital_restore_failed
	if = {
		limit = {
			NOT = { has_global_flag = liberation_release_execution_started }
			liberation_release_host_arrays_are_aligned = yes
		}
		set_temp_variable = { liberation_host_capital_restore_index = constant:liberation_release_limit.minimum }
		while_loop_effect = {
			limit = {
				check_variable = { var = liberation_host_capital_restore_index value = global.liberation_plan_hosts^num compare = less_than }
			}
			var:global.liberation_plan_hosts^liberation_host_capital_restore_index = { save_event_target_as = liberation_host_capital_restore_country }
			var:global.liberation_plan_host_original_capitals^liberation_host_capital_restore_index = { save_event_target_as = liberation_host_capital_restore_state }
			if = {
				limit = {
					has_event_target = liberation_host_capital_restore_country
					has_event_target = liberation_host_capital_restore_state
					event_target:liberation_host_capital_restore_country = { exists = yes }
					event_target:liberation_host_capital_restore_state = { is_owned_by = event_target:liberation_host_capital_restore_country }
				}
				event_target:liberation_host_capital_restore_country = {
					if = {
						limit = { NOT = { capital_scope = { state = event_target:liberation_host_capital_restore_state } } }
						set_capital = {
							state = event_target:liberation_host_capital_restore_state
							remember_old_capital = no
						}
						if = {
							limit = { NOT = { capital_scope = { state = event_target:liberation_host_capital_restore_state } } }
							set_global_flag = liberation_release_capital_restore_failed
						}
						else = {
							clear_variable = liberation_release_last_protected_capital
							clear_variable = liberation_release_last_capital_relocation_date
						}
					}
				}
			}
			else = {
				set_global_flag = liberation_release_capital_restore_failed
			}
			add_to_temp_variable = { liberation_host_capital_restore_index = constant:liberation_release_limit.one }
		}
	}
	else = {
		set_global_flag = liberation_release_capital_restore_failed
	}
}

# Moves a host capital only when its current capital is one of the frozen
# release rows. The protected state was selected before lock and is guaranteed
# to survive the incident. This runs while the plan is locked, before any state
# ownership mutation, so a failed relocation can stop execution cleanly.
liberation_release_prepare_host_capitals_for_execution = {
	set_temp_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.none }
	if = {
		limit = {
			is_liberation_release_plan_locked = yes
			liberation_release_host_arrays_are_aligned = yes
		}
		set_temp_variable = { liberation_host_capital_index = constant:liberation_release_limit.minimum }
		while_loop_effect = {
			limit = {
				check_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.none }
				check_variable = { var = liberation_host_capital_index value = global.liberation_plan_hosts^num compare = less_than }
			}
			var:global.liberation_plan_hosts^liberation_host_capital_index = { save_event_target_as = liberation_host_capital_country }
			var:global.liberation_plan_host_protected_states^liberation_host_capital_index = { save_event_target_as = liberation_host_capital_protected_state }
			if = {
				limit = {
					OR = {
						NOT = { has_event_target = liberation_host_capital_country }
						NOT = { has_event_target = liberation_host_capital_protected_state }
						NOT = { event_target:liberation_host_capital_country = { exists = yes } }
						NOT = { event_target:liberation_host_capital_protected_state = { is_owned_by = event_target:liberation_host_capital_country } }
					}
				}
				set_temp_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.protected_state_mismatch }
			}
			if = {
				limit = {
					check_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.none }
					event_target:liberation_host_capital_country = {
						capital_scope = { is_in_array = { array = global.liberation_plan_states value = THIS } }
					}
				}
				event_target:liberation_host_capital_country = {
					set_capital = {
						state = event_target:liberation_host_capital_protected_state
						remember_old_capital = no
					}
					if = {
						limit = { NOT = { capital_scope = { state = event_target:liberation_host_capital_protected_state } } }
						set_temp_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.protected_state_mismatch }
					}
					else = {
						set_variable = { liberation_release_last_protected_capital = event_target:liberation_host_capital_protected_state }
						set_variable = { liberation_release_last_capital_relocation_date = global.date }
					}
				}
			}
			add_to_temp_variable = { liberation_host_capital_index = constant:liberation_release_limit.one }
		}
	}
	else = {
		set_temp_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.stale_plan }
	}
	if = {
		limit = { NOT = { check_variable = { liberation_host_capital_relocation_failure = constant:liberation_plan_reject_reason.none } } }
		set_variable = { global.liberation_plan_last_failure = liberation_host_capital_relocation_failure }
	}
}

# Crosses the point of no return only after every ownership/controller row has
# matched the frozen ledger. Event-specific finalizers may create forces,
# inherit technology, transfer host assets, and publish durable origin data, so
# a plan in this phase is never eligible for compensating rollback.
liberation_release_begin_finalization = {
	set_temp_variable = { liberation_release_finalization_begin_success = constant:liberation_release_limit.minimum }
	if = {
		limit = {
			is_liberation_release_plan_executing = yes
			liberation_release_execution_matches_frozen_ownership = yes
		}
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.finalizing }
		set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.none }
		set_global_flag = liberation_release_finalizer_started
		clr_global_flag = liberation_release_plan_finalization_failed
		clr_global_flag = liberation_release_event5_finalization_validated
		clr_global_flag = liberation_release_event6_finalization_validated
		if = {
			limit = { has_global_flag = liberation_release_plan_includes_event5 }
			set_global_flag = liberation_release_event5_finalizer_started
		}
		set_temp_variable = { liberation_release_finalization_begin_success = constant:liberation_release_limit.one }
	}
	else = {
		set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.unsafe_instantiation }
	}
}

# Terminal diagnostic state for an unexpected failure after the first
# irreversible finalizer mutation. The frozen ledger and all scope marks remain
# intact; callers must not present the incident as committed and must not route
# this state into the compensating rollback.
liberation_release_mark_finalization_failed = {
	if = {
		limit = {
			is_liberation_release_plan_finalizing = yes
			has_global_flag = liberation_release_finalizer_started
		}
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.finalization_failed }
		clr_global_flag = liberation_release_plan_valid
		set_global_flag = liberation_release_plan_finalization_failed
		if = {
			limit = { check_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.none } }
			set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.invalid_phase }
		}
	}
}

# Starts or retries compensation without discarding any frozen row. A malformed
# ledger is retained in rollback_failed so a later repair can use the same
# owner, controller, core, country, and capital snapshots.
liberation_release_begin_plan_rollback = {
	clr_global_flag = liberation_release_plan_rollback_completed
	clr_global_flag = liberation_release_plan_rollback_failed
	clr_global_flag = liberation_release_event5_rollback_cleanup_validated
	clr_global_flag = liberation_release_event6_rollback_cleanup_validated
	set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
	if = {
		limit = {
			has_global_flag = liberation_release_execution_started
			OR = {
				check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.executing }
				check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.rolling_back }
				check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.rollback_failed }
			}
		}
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.rolling_back }
		clr_global_flag = liberation_release_plan_valid
	}
	else = {
		set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.invalid_ledger }
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.rollback_failed }
		set_global_flag = liberation_release_plan_rollback_failed
	}
}

# Restores the two country-core facts whose values were snapshotted for every
# reserved state. This runs before ownership restoration while all newly
# instantiated country scopes are still guaranteed to be addressable.
liberation_release_restore_reserved_state_core_facts = {
	for_each_scope_loop = {
		array = global.liberation_plan_states
		save_event_target_as = liberation_rollback_core_state
		if = {
			limit = {
				has_state_flag = liberation_release_core_snapshot_recorded
				has_variable = liberation_release_original_owner
				has_variable = liberation_release_target_country
			}
			var:liberation_release_original_owner = { save_event_target_as = liberation_rollback_original_owner }
			var:liberation_release_target_country = { save_event_target_as = liberation_rollback_target_country }
			if = {
				limit = { has_state_flag = liberation_release_original_owner_had_core }
				add_core_of = event_target:liberation_rollback_original_owner
			}
			else = { remove_core_of = event_target:liberation_rollback_original_owner }
			if = {
				limit = { has_state_flag = liberation_release_target_had_core }
				add_core_of = event_target:liberation_rollback_target_country
			}
			else = { remove_core_of = event_target:liberation_rollback_target_country }
			if = {
				limit = {
					check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
					OR = {
						AND = {
							has_state_flag = liberation_release_original_owner_had_core
							NOT = { is_core_of = event_target:liberation_rollback_original_owner }
						}
						AND = {
							NOT = { has_state_flag = liberation_release_original_owner_had_core }
							is_core_of = event_target:liberation_rollback_original_owner
						}
						AND = {
							has_state_flag = liberation_release_target_had_core
							NOT = { is_core_of = event_target:liberation_rollback_target_country }
						}
						AND = {
							NOT = { has_state_flag = liberation_release_target_had_core }
							is_core_of = event_target:liberation_rollback_target_country
						}
					}
				}
				set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.core_restore }
			}
		}
		else_if = {
			limit = { check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none } }
			set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.state_scope }
		}
	}
}

# Restores each state to its exact pre-plan owner and controller. Ownership and
# occupation are separate engine facts, so both are written and verified.
liberation_release_restore_reserved_state_ownership = {
	for_each_scope_loop = {
		array = global.liberation_plan_states
		if = {
			limit = {
				has_variable = liberation_release_original_owner
				has_variable = liberation_release_original_controller
				var:liberation_release_original_owner = { exists = yes }
				var:liberation_release_original_controller = { exists = yes }
			}
			var:liberation_release_original_owner = { save_event_target_as = liberation_rollback_original_owner }
			var:liberation_release_original_controller = { save_event_target_as = liberation_rollback_original_controller }
			set_state_owner_to = event_target:liberation_rollback_original_owner
			set_state_controller_to = event_target:liberation_rollback_original_controller
			if = {
				limit = {
					check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
					NOT = { is_owned_by = event_target:liberation_rollback_original_owner }
				}
				set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.owner_restore }
			}
			if = {
				limit = {
					check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
					NOT = { is_controlled_by = event_target:liberation_rollback_original_controller }
				}
				set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.controller_restore }
			}
		}
		else_if = {
			limit = { check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none } }
			set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.state_scope }
		}
	}
}

# All plan countries were proven absent before lock. After their states and
# package-local data are restored, annexing a state-less temporary country with
# troop transfer disabled removes its transient units and living-country state.
liberation_release_deinstantiate_created_countries = {
	set_temp_variable = { liberation_rollback_country_index = constant:liberation_release_limit.minimum }
	while_loop_effect = {
		limit = {
			check_variable = { var = liberation_rollback_country_index value = global.liberation_plan_countries^num compare = less_than }
		}
		var:global.liberation_plan_countries^liberation_rollback_country_index = { save_event_target_as = liberation_rollback_country }
		var:global.liberation_plan_primary_hosts^liberation_rollback_country_index = { save_event_target_as = liberation_rollback_primary_host }
		if = {
			limit = {
				event_target:liberation_rollback_country = { exists = yes }
				event_target:liberation_rollback_primary_host = { exists = yes }
				event_target:liberation_rollback_country = { NOT = { any_owned_state = { always = yes } } }
			}
			event_target:liberation_rollback_primary_host = {
				annex_country = {
					target = event_target:liberation_rollback_country
					transfer_troops = no
				}
			}
		}
		if = {
			limit = {
				check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
				event_target:liberation_rollback_country = { exists = yes }
			}
			set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.country_deinstantiation }
		}
		add_to_temp_variable = { liberation_rollback_country_index = constant:liberation_release_limit.one }
	}
}

# Ownership restoration makes every original capital eligible again. The
# aligned host snapshot is verified before any ledger row may be cleared.
liberation_release_restore_host_capitals_after_execution = {
	set_temp_variable = { liberation_rollback_host_index = constant:liberation_release_limit.minimum }
	while_loop_effect = {
		limit = {
			check_variable = { var = liberation_rollback_host_index value = global.liberation_plan_hosts^num compare = less_than }
		}
		var:global.liberation_plan_hosts^liberation_rollback_host_index = { save_event_target_as = liberation_rollback_host }
		var:global.liberation_plan_host_original_capitals^liberation_rollback_host_index = { save_event_target_as = liberation_rollback_original_capital }
		if = {
			limit = {
				event_target:liberation_rollback_host = { exists = yes }
				event_target:liberation_rollback_original_capital = { is_owned_by = event_target:liberation_rollback_host }
			}
			event_target:liberation_rollback_host = {
				set_capital = {
					state = event_target:liberation_rollback_original_capital
					remember_old_capital = no
				}
				if = {
					limit = { capital_scope = { state = event_target:liberation_rollback_original_capital } }
					clear_variable = liberation_release_last_protected_capital
					clear_variable = liberation_release_last_capital_relocation_date
				}
			}
		}
		if = {
			limit = {
				check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
				OR = {
					NOT = { event_target:liberation_rollback_host = { exists = yes } }
					NOT = { event_target:liberation_rollback_original_capital = { is_owned_by = event_target:liberation_rollback_host } }
					NOT = { event_target:liberation_rollback_host = { capital_scope = { state = event_target:liberation_rollback_original_capital } } }
				}
			}
			set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.host_capital_restore }
		}
		add_to_temp_variable = { liberation_rollback_host_index = constant:liberation_release_limit.one }
	}
}

liberation_release_finish_verified_plan_rollback = {
	if = {
		limit = { liberation_release_rollback_matches_frozen_ledger = yes }
		if = {
			limit = { has_global_flag = liberation_release_plan_includes_event6 }
			independence_wave_clear_plan_contribution = yes
		}
		if = {
			limit = { has_global_flag = liberation_release_plan_includes_event5 }
			SOV = { soviet_collapse_joint_clear_plan_contribution = yes }
		}
		liberation_release_clear_plan_scope_marks = yes
		liberation_release_clear_plan_arrays = yes
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.aborted }
		set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
		clr_global_flag = liberation_release_execution_started
		clr_global_flag = liberation_release_plan_valid
		clr_global_flag = liberation_release_plan_rollback_failed
		clr_global_flag = liberation_release_finalizer_started
		clr_global_flag = liberation_release_plan_finalization_failed
		clr_global_flag = liberation_release_event5_finalization_validated
		clr_global_flag = liberation_release_event6_finalization_validated
		clr_global_flag = liberation_release_event5_finalizer_started
		clr_global_flag = liberation_release_event6_origin_commit_started
		clear_variable = global.liberation_plan_finalization_failure
		set_global_flag = liberation_release_plan_rollback_completed
	}
}

liberation_release_mark_plan_rollback_failed = {
	set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.rollback_failed }
	clr_global_flag = liberation_release_plan_valid
	set_global_flag = liberation_release_plan_rollback_failed
	clr_global_flag = liberation_release_plan_rollback_completed
	if = {
		limit = { check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none } }
		set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.invalid_ledger }
	}
}

# Shared post-mutation failure path. Event-specific reversible state is cleaned
# first; then cores, ownership, countries, and capitals are restored. Any failed
# assertion leaves the complete ledger available for an idempotent retry.
liberation_release_execute_compensating_rollback = {
	if = {
		limit = {
			NOT = { has_global_flag = liberation_release_finalizer_started }
			NOT = { is_liberation_release_plan_finalizing = yes }
			NOT = { has_liberation_release_plan_finalization_failed = yes }
		}
		liberation_release_begin_plan_rollback = yes
	}
	if = {
		limit = { is_liberation_release_plan_rolling_back = yes }
		if = {
			limit = { liberation_release_rollback_ledger_is_valid = yes }
			if = {
				limit = { has_global_flag = liberation_release_plan_includes_event6 }
				independence_wave_rollback_reversible_frozen_country_packages = yes
			}
			else = { set_global_flag = liberation_release_event6_rollback_cleanup_validated }
			if = {
				limit = {
					has_global_flag = liberation_release_plan_includes_event5
					NOT = { has_global_flag = liberation_release_event5_finalizer_started }
				}
				set_global_flag = liberation_release_event5_rollback_cleanup_validated
			}
			else_if = {
				limit = {
					has_global_flag = liberation_release_plan_includes_event5
					check_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.none }
				}
				set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.event5_package_cleanup }
			}
			else = { set_global_flag = liberation_release_event5_rollback_cleanup_validated }
			liberation_release_restore_reserved_state_core_facts = yes
			liberation_release_restore_reserved_state_ownership = yes
			liberation_release_deinstantiate_created_countries = yes
			liberation_release_restore_host_capitals_after_execution = yes
			if = {
				limit = { liberation_release_rollback_matches_frozen_ledger = yes }
				liberation_release_finish_verified_plan_rollback = yes
			}
			else = { liberation_release_mark_plan_rollback_failed = yes }
		}
		else = {
			set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.invalid_ledger }
			liberation_release_mark_plan_rollback_failed = yes
		}
	}
}

liberation_release_commit_plan = {
	if = {
		limit = {
			is_liberation_release_plan_finalizing = yes
			OR = {
				NOT = { has_global_flag = liberation_release_plan_includes_event5 }
				has_global_flag = liberation_release_event5_finalization_validated
			}
			OR = {
				NOT = { has_global_flag = liberation_release_plan_includes_event6 }
				has_global_flag = liberation_release_event6_finalization_validated
			}
		}
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.committed }
		set_variable = { global.liberation_plan_committed_date = global.date }
		set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.none }
		set_global_flag = liberation_release_joint_plan_executed
		clr_global_flag = liberation_release_execution_started
		clr_global_flag = liberation_release_finalizer_started
		clr_global_flag = liberation_release_plan_finalization_failed
		clr_global_flag = liberation_release_event5_finalization_validated
		clr_global_flag = liberation_release_event6_finalization_validated
		clr_global_flag = liberation_release_event5_finalizer_started
		clr_global_flag = liberation_release_event6_origin_commit_started
		liberation_release_clear_plan_scope_marks = yes
	}
}

liberation_release_abort_plan = {
	if = {
		limit = {
			has_global_flag = liberation_release_finalizer_started
			is_liberation_release_plan_finalizing = yes
		}
		liberation_release_mark_finalization_failed = yes
	}
	else_if = {
		limit = { has_liberation_release_plan_finalization_failed = yes }
	}
	else_if = {
		limit = { has_global_flag = liberation_release_execution_started }
		liberation_release_execute_compensating_rollback = yes
	}
	else_if = {
		limit = { has_global_flag = liberation_release_capital_restore_failed }
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.rollback_failed }
		set_variable = { global.liberation_plan_rollback_failure = constant:liberation_rollback_failure.host_capital_restore }
		clr_global_flag = liberation_release_plan_valid
		set_global_flag = liberation_release_plan_rollback_failed
	}
	else = {
		set_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.aborted }
		clr_global_flag = liberation_release_plan_valid
		liberation_release_clear_plan_scope_marks = yes
	}
}
