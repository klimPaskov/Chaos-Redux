# ============================================================================
# CHAOS REDUX - MIGRATION PERSECUTION PROJECTION
# ============================================================================
# Overview:
# - Maintains a state-scoped projection of exact current persecution owners.
# - Records only accepted state/action seams that provide a responsible actor,
#   a cause class, and a monotonic state-local generation.
# - Keeps actor, cause, and generation metadata separate from the canonical
#   boolean flag so consumers cannot treat the flag as an identity container.
# - Aggregates overlapping owner classes and clears only the class whose exact
#   terminal path supplied a matching generation proof.
# - Performs no population, Deaths, pressure, migration, or world-scan work.
# ============================================================================

# State scope. Allocate the next durable generation for one accepted owner
# action. Call only after the caller has passed its exact action/state guards.
migration_persecution_issue_generation = {
	if = {
		limit = { NOT = { has_variable = migration_persecution_generation_counter } }
		set_variable = {
			migration_persecution_generation_counter = constant:migration_persecution_owner_class.none
		}
	}
	add_to_variable = {
		migration_persecution_generation_counter = constant:humanitarian_runtime.one
	}
}

# State scope. Record one accepted request and mark its owner class. The actor
# is supplied as a scope pointer by the caller and must resolve to ROOT, the
# country that accepted the operation. A mismatched actor fails closed.
migration_persecution_record_state = {
	if = {
		limit = {
			has_variable = migration_persecution_pending_owner
			check_variable = {
				var = migration_persecution_pending_owner
				value = constant:migration_persecution_owner_class.none
				compare = greater_than
			}
			has_variable = migration_persecution_pending_actor
			var:migration_persecution_pending_actor = {
				exists = yes
				tag = ROOT
			}
		}
		if = {
			limit = { NOT = { has_variable = migration_persecution_pending_generation } }
			migration_persecution_issue_generation = yes
			set_variable = {
				migration_persecution_pending_generation = migration_persecution_generation_counter
			}
		}
		migration_persecution_mark_state = yes
	}
	clear_variable = migration_persecution_pending_owner
	clear_variable = migration_persecution_pending_actor
	clear_variable = migration_persecution_pending_generation
}

# State scope. Rebuild the canonical projection from exact owner classes. The
# priority only selects metadata; it never suppresses another owner marker.
migration_persecution_recompute_state = {
	if = {
		limit = {
			has_state_flag = migration_persecution_cbrn_active
			has_variable = migration_persecution_cbrn_generation
			has_variable = migration_persecution_cbrn_actor_id
			has_variable = migration_persecution_cbrn_cause_class
		}
		set_state_flag = migration_persecution_active
		set_variable = {
			migration_persecution_active_generation = migration_persecution_cbrn_generation
		}
		set_variable = {
			migration_persecution_active_actor_id = migration_persecution_cbrn_actor_id
		}
		set_variable = {
			migration_persecution_active_cause_class = migration_persecution_cbrn_cause_class
		}
	}
	else_if = {
		limit = {
			has_state_flag = migration_persecution_extermination_active
			has_variable = migration_persecution_extermination_generation
			has_variable = migration_persecution_extermination_actor_id
			has_variable = migration_persecution_extermination_cause_class
		}
		set_state_flag = migration_persecution_active
		set_variable = {
			migration_persecution_active_generation = migration_persecution_extermination_generation
		}
		set_variable = {

			has_variable = migration_persecution_gulag_generation
			has_variable = migration_persecution_gulag_actor_id

		}

			migration_persecution_active_cause_class = migration_persecution_gulag_cause_class
		}
	}
	else_if = {
		limit = {
			has_state_flag = migration_persecution_expanded_labor_active
			has_variable = migration_persecution_expanded_labor_generation
			has_variable = migration_persecution_expanded_labor_actor_id
			has_variable = migration_persecution_expanded_labor_cause_class
		}
		set_state_flag = migration_persecution_active
		set_variable = {
			migration_persecution_active_generation = migration_persecution_expanded_labor_generation

		set_variable = {

		}

			migration_persecution_active_cause_class = migration_persecution_expanded_labor_cause_class
		}
	}
	else_if = {
		limit = {
			has_state_flag = migration_persecution_labor_project_active
			has_variable = migration_persecution_labor_project_generation
			has_variable = migration_persecution_labor_project_actor_id
			has_variable = migration_persecution_labor_project_cause_class
		}
		set_state_flag = migration_persecution_active
		set_variable = {
			migration_persecution_active_generation = migration_persecution_labor_project_generation

		set_variable = {
			migration_persecution_active_actor_id = migration_persecution_labor_project_actor_id

		clear_variable = migration_persecution_active_cause_class

# State scope. Capture the current camp owner class and generation before an
# exact evidence/inactive-site terminal path mutates the site profile.
migration_persecution_prepare_camp_terminal = {
	camp_rework_refresh_state_site_profile = yes
	clear_variable = migration_persecution_pending_owner
	clear_variable = migration_persecution_pending_clear_generation
	if = {
		limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.detention } }
		set_variable = {
			migration_persecution_pending_owner = constant:migration_persecution_owner_class.detention
		}
		if = {
			limit = { has_variable = migration_persecution_detention_generation }
			set_variable = {
				migration_persecution_pending_clear_generation = migration_persecution_detention_generation
			}
		}
	}
	else_if = {
		limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.expanded_labor } }
		set_variable = {
			migration_persecution_pending_owner = constant:migration_persecution_owner_class.expanded_labor
		}
		if = {
			limit = { has_variable = migration_persecution_expanded_labor_generation }
			set_variable = {
				migration_persecution_pending_clear_generation = migration_persecution_expanded_labor_generation
			}
		}
	}
	else_if = {
		limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.gulag } }
		set_variable = {
			migration_persecution_pending_owner = constant:migration_persecution_owner_class.gulag
		}
		if = {
			limit = { has_variable = migration_persecution_gulag_generation }
			set_variable = {
				migration_persecution_pending_clear_generation = migration_persecution_gulag_generation
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized }
				check_variable = { camp_state_site_type = constant:camp_rework_site_type.experiment }
			}
		}
		set_variable = {
			migration_persecution_pending_owner = constant:migration_persecution_owner_class.extermination
		}
		if = {
			limit = { has_variable = migration_persecution_extermination_generation }
			set_variable = {
				migration_persecution_pending_clear_generation = migration_persecution_extermination_generation
			}
		}
	}
}

# State scope. Capture every camp owner generation before dismantlement clears
# the underlying site flags. The later clear helper preserves any newer owner
# that appeared before the terminal callback completes.
migration_persecution_prepare_camp_dismantlement = {
	clear_variable = migration_persecution_terminal_detention_generation
	clear_variable = migration_persecution_terminal_expanded_labor_generation
	clear_variable = migration_persecution_terminal_labor_project_generation
	clear_variable = migration_persecution_terminal_gulag_generation
	clear_variable = migration_persecution_terminal_extermination_generation
	if = {
		limit = { has_variable = migration_persecution_detention_generation }
		set_variable = {
			migration_persecution_terminal_detention_generation = migration_persecution_detention_generation
		}
	}
	if = {
		limit = { has_variable = migration_persecution_expanded_labor_generation }
		set_variable = {
			migration_persecution_terminal_expanded_labor_generation = migration_persecution_expanded_labor_generation
		}
	}
	if = {
		limit = { has_variable = migration_persecution_labor_project_generation }
		set_variable = {
			migration_persecution_terminal_labor_project_generation = migration_persecution_labor_project_generation
		}
	}
	if = {
		limit = { has_variable = migration_persecution_gulag_generation }
		set_variable = {
			migration_persecution_terminal_gulag_generation = migration_persecution_gulag_generation
		}
	}
	if = {
		limit = { has_variable = migration_persecution_extermination_generation }
		set_variable = {
			migration_persecution_terminal_extermination_generation = migration_persecution_extermination_generation
		}
	}
}
