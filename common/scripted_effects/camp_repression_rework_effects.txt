# ============================================================================
# CHAOS REDUX - CAMP AND REPRESSION REWORK EFFECTS
# ============================================================================
# Overview:
# - Provides one duplicate-safe active-site and active-country registry.
# - Recalculates the connected national pressure model from registered states.
# - Applies one exclusive monthly Deaths pulse per active state, including
#   abstract chemical/biological factors without operational instructions.
# - Routes one-time restricted chemical escalation through the CBRN shared
#   exposure dispatcher; recurring camp processing remains camp-owned.
# - Owns cleanup, display caches, bounded GUI arrays, and shared reform hooks.
# ============================================================================

# Country scope. This is the only public initializer for the repression
# system. Special Chaos countries are actively scrubbed instead of receiving
# dormant variables that could later expose ordinary-country UI or missions.
camp_rework_initialize_country_variables = {
	if = {
		limit = { camp_rework_country_is_eligible = yes }
		camp_rework_initialize_eligible_country_variables = yes
	}
	else = {
		camp_rework_exclude_special_country = yes
	}
}

# Cancels only coercive expansion and exploitation missions. Evidence,
# containment, review, redress, and dismantlement missions remain live.
camp_rework_cancel_active_expansion_projects = {
	if = {
		limit = { has_active_mission = generic_labor_project_cycle }
		if = {
			limit = { has_variable = generic_labor_project_state_id }
			var:generic_labor_project_state_id = { clr_state_flag = camp_rework_labor_project_active }
		}
		clr_country_flag = generic_labor_project_active
		clear_variable = generic_labor_project_state_id
		clear_variable = generic_labor_project_type
		remove_mission = generic_labor_project_cycle
	}

	if = {
		limit = { has_active_mission = germany_occupied_poland_camp_expansion_mission }
		camp_rework_germany_cancel_occupied_poland_expansion_mission = yes
		remove_mission = germany_occupied_poland_camp_expansion_mission
	}
	if = {
		limit = { has_active_mission = germany_eastern_fortifications_labor_mission }
		camp_rework_germany_cancel_eastern_fortifications = yes
		remove_mission = germany_eastern_fortifications_labor_mission
	}
	if = {
		limit = { has_active_mission = germany_ss_laboratory_annex_mission }
		camp_rework_germany_cancel_laboratory_annex = yes
		remove_mission = germany_ss_laboratory_annex_mission
	}

	if = {
		limit = { has_active_mission = japan_pingfang_research_bureau_mission }
		camp_rework_japan_cancel_pingfang_bureau = yes
		remove_mission = japan_pingfang_research_bureau_mission
	}

	if = {
		limit = { has_active_mission = sov_gulag_quota_cycle }
		camp_rework_soviet_cancel_gulag_quota_cycle = yes
		remove_mission = sov_gulag_quota_cycle
	}

	if = {
		limit = { has_active_mission = uk_hold_raj_security_line }
		clear_variable = uk_raj_security_line_state_id
		remove_mission = uk_hold_raj_security_line
	}
	if = {
		limit = { has_active_mission = uk_complete_raj_military_works }
		clr_country_flag = camp_rework_uk_military_works_active
		clear_variable = uk_raj_military_works_state_id
		remove_mission = uk_complete_raj_military_works
	}

	if = {
		limit = { has_active_mission = fr_north_africa_rail_labor_project }
		clear_variable = fr_north_africa_project_state_id
		remove_mission = fr_north_africa_rail_labor_project
	}

	if = {
		limit = { has_active_mission = ita_desert_road_labor_project }
		clear_variable = ita_desert_road_project_state_id
		remove_mission = ita_desert_road_labor_project
	}
	if = {
		limit = { has_active_mission = ita_colonial_security_sweep }
		clr_country_flag = camp_rework_italy_security_battalions_active
		clear_variable = ita_security_sweep_state_id
		remove_mission = ita_colonial_security_sweep
	}

	if = {
		limit = { has_active_mission = bel_congo_resource_quota_cycle }
		clear_variable = bel_congo_quota_state_id
		remove_mission = bel_congo_resource_quota_cycle
	}
	if = {
		limit = { has_active_mission = bel_congo_transport_corridor_project }
		clear_variable = bel_congo_corridor_state_id
		remove_mission = bel_congo_transport_corridor_project
	}
}

camp_rework_register_country_for_monthly_pulse = {
	if = {
		limit = { camp_rework_country_is_eligible = yes }
		camp_rework_initialize_eligible_country_variables = yes
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.camp_repression_active_countries
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.camp_repression_active_countries
				value = THIS
			}
		}
	}
	else = {
		camp_rework_exclude_special_country = yes
	}
}

camp_rework_classify_state_pool_for_root = {
	set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none }
	if = {
		limit = { camp_rework_state_is_occupied_noncore_pool = yes }
		set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.occupied_noncore }
	}
	else_if = {
		limit = { camp_rework_state_is_colonial_subject_pool = yes }
		set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.colonial_subject }
	}
	else_if = {
		limit = { camp_rework_state_is_noncore_pool = yes }
		set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.noncore }
	}
	else_if = {
		limit = { camp_rework_state_is_periphery_pool = yes }
		set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.periphery }
	}
	else_if = {
		limit = { camp_rework_state_is_core_fallback_pool = yes }
		set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.core_fallback }
	}
}

camp_rework_refresh_state_site_profile = {
	if = { limit = { NOT = { has_variable = camp_state_evidence_depth } } set_variable = { camp_state_evidence_depth = 0 } }
	if = { limit = { NOT = { has_variable = camp_state_resistance_accumulation } } set_variable = { camp_state_resistance_accumulation = 0 } }
	if = { limit = { NOT = { has_variable = camp_state_observer_exposure } } set_variable = { camp_state_observer_exposure = 0 } }
	if = { limit = { NOT = { has_variable = camp_state_pool_type } } set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none } }
	if = { limit = { NOT = { has_variable = camp_state_evidence_site_type } } set_variable = { camp_state_evidence_site_type = constant:camp_rework_site_type.none } }

	set_variable = { camp_state_site_type = constant:camp_rework_site_type.none }
	if = {
		limit = {
			OR = {
				has_state_flag = genocide_restricted_chemical_site_active
				has_state_flag = camp_rework_chemical_method_active
				has_state_flag = camp_rework_biological_method_active
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.contaminated }
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = camp_rework_radicalized_site
				has_state_flag = genocide_extermination_site
				extermination_camp > 0
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized }
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = camp_rework_experiment_site
				has_state_flag = genocide_auschwitz_experiment_site
				has_state_flag = genocide_ss_laboratory_site
				has_state_flag = genocide_japanese_biowarfare_atrocity_site
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.experiment }
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = genocide_gulag_site
				has_state_flag = genocide_soviet_mass_repression_site
				has_state_flag = genocide_soviet_famine_pressure_site
				gulag_labor_camp_network > 0
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.gulag }
	}
	else_if = {
		limit = { has_state_flag = camp_rework_expanded_labor_site }
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.expanded_labor }
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = genocide_concentration_site
				concentration_camp > 0
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.detention }
	}

	set_variable = { camp_state_reach = 0 }
	set_variable = { camp_state_labor_output = 0 }
	set_variable = { camp_state_coercive_control = 0 }
	set_variable = { camp_state_population_loss_pressure = 0 }
	set_variable = { camp_state_resistance_base = 0 }
	set_variable = { camp_state_stability_damage = 0 }
	set_variable = { camp_state_evidence_base = 0 }
	set_variable = { camp_state_overstretch = 0 }
	set_variable = { camp_state_visibility_base = 0 }
	set_variable = { camp_state_tribunal_base = 0 }
	set_variable = { camp_state_hardliner_pressure = 0 }
	set_variable = { camp_state_legitimacy_damage = 0 }

	if = {
		limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.detention } }
		set_variable = { camp_state_reach = constant:camp_rework_site_contribution.detention_reach }
		set_variable = { camp_state_labor_output = constant:camp_rework_site_contribution.detention_labor }
		set_variable = { camp_state_coercive_control = constant:camp_rework_site_contribution.detention_control }
		set_variable = { camp_state_population_loss_pressure = constant:camp_rework_site_contribution.detention_population_loss }
		set_variable = { camp_state_resistance_base = constant:camp_rework_site_contribution.detention_resistance }
		set_variable = { camp_state_stability_damage = constant:camp_rework_site_contribution.detention_stability }

camp_rework_register_active_site = {
	if = {

	}
	if = {
		limit = {
			var:genocide_responsible_country = {
				camp_rework_country_is_eligible = yes
			}
			OR = {
				var:genocide_responsible_country = { tag = ROOT }
				has_global_flag = camp_rework_migration_in_progress
			}
		}
		set_state_flag = genocide_site_has_responsible_country
		set_state_flag = genocide_atrocity_site_known_to_regime
		set_state_flag = camp_rework_site_active

		clr_state_flag = camp_rework_site_evidence_resolved
		clr_state_flag = camp_rework_annexed_evidence_site
		clr_state_flag = genocide_historical_quiet_camp

		if = {
			limit = {
				check_variable = { camp_register_site_type > constant:camp_rework_site_type.none }
			}
			if = {
				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.expanded_labor } }
				set_state_flag = camp_rework_expanded_labor_site
			}
			else_if = {
				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.experiment } }
				set_state_flag = camp_rework_experiment_site
			}
			else_if = {
				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.radicalized } }
				set_state_flag = camp_rework_radicalized_site
			}
			else_if = {
				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.contaminated } }
				set_state_flag = camp_rework_contaminated_evidence_site
			}
		}

		if = {
			limit = {
				OR = {
					NOT = { has_variable = camp_state_pool_type }
					check_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none }
				}
				var:genocide_responsible_country = { tag = ROOT }
			}
			camp_rework_classify_state_pool_for_root = yes
		}
		if = {
			limit = {
				NOT = { has_global_flag = camp_rework_migration_in_progress }
				NOT = { has_state_flag = camp_rework_historical_baseline_registration }
			}
			camp_rework_apply_core_fallback_penalty = yes
		}

		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.genocide_active_camp_states
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.genocide_active_camp_states
				value = THIS
			}
		}

		camp_rework_refresh_state_site_profile = yes
		if = {
			limit = {
				OR = {
					check_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized }
					check_variable = { camp_state_site_type = constant:camp_rework_site_type.contaminated }
				}
			}
			var:genocide_responsible_country = { set_country_flag = camp_rework_achievement_operated_radicalized_site }
		}
		if = {
			limit = { check_variable = { camp_state_site_type > constant:camp_rework_site_type.none } }
			set_variable = { camp_state_evidence_site_type = camp_state_site_type }
		}
		camp_rework_register_evidence_state = yes
		var:genocide_responsible_country = {
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = camp_active_site_states
							value = PREV
						}
					}
				}
				add_to_array = {
					array = camp_active_site_states
					value = PREV
				}
			}
			camp_rework_register_country_for_monthly_pulse = yes
			if = {
				limit = { NOT = { has_global_flag = camp_rework_migration_in_progress } }
				camp_rework_recalculate_country_values = yes
			}
		}
	}
	set_temp_variable = { camp_register_site_type = 0 }

				PREV = { add_to_variable = { camp_soviet_dismantlement_site_count = 1 } }
				if = {
					limit = { camp_rework_state_is_actor_or_subject_controlled = yes }
					PREV = { add_to_variable = { camp_soviet_dismantlement_closable_site_count = 1 } }
				}
			}
			if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.detention } }
				PREV = { add_to_variable = { camp_detention_site_count = 1 } }
			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.expanded_labor } }
				PREV = { add_to_variable = { camp_expanded_site_count = 1 } }
			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.gulag } }
				PREV = { add_to_variable = { camp_gulag_site_count = 1 } }
			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.experiment } }
				PREV = { add_to_variable = { camp_experiment_site_count = 1 } }
			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized } }
				PREV = { add_to_variable = { camp_radicalized_site_count = 1 } }
			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.contaminated } }
				PREV = { add_to_variable = { camp_contaminated_site_count = 1 } }
			}
		}
	}

	# Inactive evidence remains attributable after a site is dismantled or lost.
	# Active entries are excluded because their contributions were added above.
	for_each_scope_loop = {
		array = camp_evidence_states
		if = {
			limit = {
				has_variable = genocide_responsible_country
				PREV = {
					NOT = {
						is_in_array = {
							array = camp_active_site_states
							value = PREV

				}

			add_to_temp_variable = { camp_inactive_visibility_value = camp_state_observer_exposure }

			multiply_temp_variable = { camp_inactive_tribunal_value = constant:camp_rework_evidence_carry.tribunal_factor }
			set_temp_variable = { camp_inactive_reform_value = camp_state_evidence_depth }
			multiply_temp_variable = { camp_inactive_reform_value = constant:camp_rework_evidence_carry.reform_factor }
			PREV = {
				add_to_variable = { camp_evidence_level = camp_inactive_evidence_value }
				add_to_variable = { camp_foreign_visibility = camp_inactive_visibility_value }
				add_to_variable = { camp_tribunal_severity = camp_inactive_tribunal_value }
				add_to_variable = { camp_reform_pressure = camp_inactive_reform_value }
			}
		}
	}

	add_to_variable = { camp_labor_output = camp_policy_labor_output }
	add_to_variable = { camp_coercive_control = camp_policy_coercive_control }
	add_to_variable = { camp_population_loss_index = camp_policy_population_loss }

	clamp_variable = { var = camp_labor_output min = 0 max = constant:camp_rework_threshold.max_value }

	if = {
		limit = {
			check_variable = {
				var = camp_tribunal_severity

				compare = greater_than_or_equals
			}
		}
		set_country_flag = camp_rework_achievement_severe_tribunal_reached
	}

	set_variable = { camp_current_phase = constant:camp_rework_phase.dormant }
	if = {
		limit = {
			OR = {
				has_country_flag = camp_rework_dismantlement_in_progress
				has_country_flag = camp_rework_reform_route_open
				check_variable = { camp_reform_pressure > 0 }

		}
		set_variable = { camp_current_phase = constant:camp_rework_phase.reform }
	}
	else_if = {
		limit = { genocide_country_has_discovered_sites = yes }

	}
	else_if = {

	}
	else_if = {
		limit = { check_variable = { camp_radicalized_site_count > 0 } }
		set_variable = { camp_current_phase = constant:camp_rework_phase.radicalized }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { camp_expanded_site_count > 0 }
				check_variable = { camp_network_reach > constant:camp_rework_threshold.expanded_reach }
			}
		}
		set_variable = { camp_current_phase = constant:camp_rework_phase.expanded }
	}
	else_if = {
		limit = { check_variable = { camp_active_site_count > 0 } }
		set_variable = { camp_current_phase = constant:camp_rework_phase.active }
	}
	else_if = {
		limit = { has_country_flag = camp_rework_reformed_legacy }
		set_variable = { camp_current_phase = constant:camp_rework_phase.dismantled }
	}

	camp_rework_recalculate_burdens = yes
	camp_rework_sync_legacy_compatibility_values = yes
	camp_rework_apply_ai_cap = yes
	camp_rework_copy_display_values = yes
}

	if = {

					has_country_flag = camp_rework_reform_route_open
				}
			}
		}
		set_country_flag = camp_ai_can_expand
	}

# validation and country-kit checks before applying costs or effects.

		camp_rework_gui_apply_selected_chemical_method = yes
	}

		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.biological_method } }
		camp_rework_gui_apply_selected_biological_method = yes
	}
	else_if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.camp_repression_close_dormant_legacy_site }
			has_variable = camp_rework_action_state_id
			check_variable = { var = manpower value = constant:camp_rework_cost.dismantle_manpower compare = greater_than }
			has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
			var:camp_rework_action_state_id = { is_valid_camp_dormant_reform_state = yes }
		}
		camp_rework_consume_dismantlement_resources = yes
		set_country_flag = camp_rework_dormant_legacy_closure_active
	}
	else = {
		if = {
			limit = {
				OR = {
					check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_activate_detention_network }
					check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_redirect_labor_to_construction }
					check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_redirect_labor_to_resource_extraction }
				}
			}
			camp_rework_dispatch_generic_action = yes
		}

		else_if = { limit = { original_tag = JAP } camp_rework_dispatch_japan_action = yes }
		else_if = { limit = { original_tag = SOV } camp_rework_dispatch_soviet_action = yes }
		else_if = { limit = { original_tag = ENG } camp_rework_dispatch_uk_action = yes }
		else_if = { limit = { original_tag = USA } camp_rework_dispatch_usa_action = yes }
		else_if = { limit = { OR = { original_tag = FRA tag = VIC } } camp_rework_dispatch_france_action = yes }
		else_if = { limit = { original_tag = ITA } camp_rework_dispatch_italy_action = yes }
		else_if = { limit = { original_tag = BEL } camp_rework_dispatch_belgium_action = yes }
		else = { camp_rework_dispatch_generic_action = yes }

camp_rework_gui_expand_selected_pool = {
	if = {
		limit = {
			camp_rework_country_uses_generic_kit = yes
			NOT = { has_country_flag = camp_rework_expansion_frozen }
			has_variable = camp_selected_state_id
			var:camp_selected_state_id = {
				OR = {
					is_controlled_by = ROOT
					CONTROLLER = { is_subject_of = ROOT }
				}
				camp_rework_state_can_accept_responsibility_from_root = yes
			}
			OR = {
				AND = {
					camp_rework_country_can_expand_generic_labor = yes
					var:camp_selected_state_id = {
						is_valid_camp_active_site_state = yes
						camp_rework_state_is_actor_responsible_controlled = yes
						NOT = { has_state_flag = camp_rework_expanded_labor_site }
						NOT = { extermination_camp > 0 }
					}
					NOT = { has_political_power < constant:camp_rework_cost.expansion_pp }
					check_variable = { var = manpower value = constant:camp_rework_cost.expansion_manpower compare = greater_than }
					has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
					has_equipment = { train_equipment_1 > constant:camp_rework_equipment_cost.trains }
				}
				AND = {
					camp_rework_country_can_activate_generic = yes
					var:camp_selected_state_id = {
						is_generic_player_selectable_camp_pool_state = yes
						NOT = { is_valid_camp_active_site_state = yes }
						NOT = { concentration_camp > 0 }
					}
					NOT = { has_political_power < constant:camp_rework_cost.activation_pp }
					check_variable = { var = manpower value = constant:camp_rework_cost.activation_manpower compare = greater_than }
					command_power > constant:camp_rework_cost.activation_command
					has_equipment = { support_equipment_1 > constant:camp_rework_equipment_cost.support_equipment }
				}
			}
		}
		set_temp_variable = { camp_gui_political_cost = constant:camp_rework_cost.activation_pp }
		if = {
			limit = { var:camp_selected_state_id = { is_valid_camp_active_site_state = yes } }
			set_temp_variable = { camp_gui_political_cost = constant:camp_rework_cost.expansion_pp }
		}
		camp_rework_gui_pay_political_cost = yes
		set_variable = { camp_rework_action_state_id = camp_selected_state_id }
		if = {
			limit = { var:camp_rework_action_state_id = { is_valid_camp_active_site_state = yes } }
			camp_rework_expand_labor_in_action_state = yes
		}
		else = {
			camp_rework_activate_detention_in_action_state = yes
		}
		camp_rework_rebuild_display_values = yes
	}
}

# Versioned one-time migration. This is the only broad reconstruction pass.
# ---------------------------------------------------------------------------

camp_rework_run_versioned_migration = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = global.camp_rework_schema_version
					value = constant:camp_rework_schema.current
					compare = greater_than_or_equals
				}
			}
		}
		set_global_flag = camp_rework_migration_in_progress
		clear_array = global.genocide_active_camp_states
		clear_array = global.camp_repression_active_countries
		# Pass one clears every destination before any controller can append a
		# state to a different responsible country's arrays.
		every_country = {
			limit = { exists = yes }
			clear_array = camp_active_site_states
			clear_array = camp_evidence_states
			camp_rework_initialize_country_variables = yes
		}
		# Pass two rebuilds state membership and preserves the stored perpetrator
		# even when a different country currently controls the evidence site.
		every_country = {
			limit = { exists = yes }
			every_controlled_state = {
				limit = {
					OR = {
						concentration_camp > 0
						extermination_camp > 0
						gulag_labor_camp_network > 0
						has_state_flag = genocide_concentration_site
						has_state_flag = genocide_extermination_site
						has_state_flag = genocide_gulag_site
						has_state_flag = genocide_auschwitz_experiment_site
						has_state_flag = genocide_ss_laboratory_site
						has_state_flag = genocide_japanese_biowarfare_atrocity_site
						has_state_flag = genocide_restricted_chemical_site
						has_state_flag = genocide_site_discovered
						has_state_flag = genocide_evidence_destroyed
						has_state_flag = genocide_evidence_destroy_failed
					}
				}
				if = {
					limit = { NOT = { has_variable = genocide_responsible_country } }
					set_variable = { genocide_responsible_country = PREV }
				}
				if = {
					limit = {
						OR = {
							NOT = { has_variable = camp_state_pool_type }
							check_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none }
						}
					}
					set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.core_fallback }
					if = {
						limit = {
							is_controlled_by = PREV
							NOT = { is_owned_by = PREV }
							NOT = { is_core_of = PREV }
						}
						set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.occupied_noncore }
					}
					else_if = {
						limit = {
							is_owned_and_controlled_by = PREV
							NOT = { is_core_of = PREV }
						}
						set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.noncore }
					}
					else_if = {
						limit = {
							OR = {
								has_state_flag = camp_rework_periphery_pool
								has_state_flag = camp_rework_political_opposition_pool
								has_state_flag = camp_rework_prison_labor_pool
							}
						}
						set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.periphery }
					}
				}
				if = {
					limit = {
						NOT = { has_state_flag = genocide_historical_quiet_camp }
						NOT = { has_state_flag = camp_rework_annexed_evidence_site }
						OR = {
							concentration_camp > 0
							extermination_camp > 0
							gulag_labor_camp_network > 0
							has_state_flag = genocide_auschwitz_experiment_site
							has_state_flag = genocide_ss_laboratory_site
							has_state_flag = genocide_japanese_biowarfare_atrocity_site
							has_state_flag = genocide_restricted_chemical_site_active
						}
					}
					set_temp_variable = { camp_register_site_type = constant:camp_rework_site_type.none }
					camp_rework_register_active_site = yes
				}
				if = {
					limit = {
						OR = {
							has_state_flag = genocide_site_discovered
							has_state_flag = genocide_evidence_destroyed
							has_state_flag = genocide_evidence_destroy_failed
							has_state_flag = genocide_restricted_chemical_site
						}
					}
					camp_rework_register_evidence_state = yes
				}
			}
		}
		# Pass three recalculates only after cross-country array writes are done.
		every_country = {
			limit = {
				exists = yes
				OR = {
					check_variable = { camp_active_site_states^num > 0 }
					check_variable = { camp_evidence_states^num > 0 }
					has_visible_camp_reform_work = yes
				}
			}
			camp_rework_register_country_for_monthly_pulse = yes
			camp_rework_recalculate_country_values = yes
		}
		set_variable = { global.camp_rework_schema_version = constant:camp_rework_schema.current }
		clr_global_flag = camp_rework_migration_in_progress
	}
}

# Country-specific idea and monthly bridge dispatchers are expanded by the
# country packages. Keeping a stable shared call site avoids duplicate loops.
camp_rework_refresh_country_specific_ideas = {
	if = {
		limit = { camp_rework_country_uses_generic_kit = yes }
		if = {
			limit = { has_country_flag = camp_rework_reformed_legacy }
			remove_ideas = generic_detention_network_administration
			remove_ideas = generic_expanded_labor_network
			remove_ideas = generic_overextended_repression_network
			remove_ideas = generic_radicalized_atrocity_policy
			remove_ideas = generic_reform_pressure
			if = {
				limit = { NOT = { has_idea = generic_reformed_legacy } }
				add_ideas = generic_reformed_legacy
			}
		}
		else = {
			remove_ideas = generic_detention_network_administration
			remove_ideas = generic_expanded_labor_network
			remove_ideas = generic_overextended_repression_network
			if = {
				limit = {
					check_variable = {
						var = camp_overstretch
						value = constant:camp_rework_threshold.overstretched
						compare = greater_than_or_equals
					}
				}
				add_ideas = generic_overextended_repression_network
			}
			else_if = {
				limit = {
					OR = {
						check_variable = { camp_expanded_site_count > 0 }
						check_variable = { camp_labor_output > constant:camp_rework_threshold.band_medium }
					}
				}
				add_ideas = generic_expanded_labor_network
			}
			else_if = {
				limit = { check_variable = { camp_active_site_count > 0 } }
				add_ideas = generic_detention_network_administration
			}

			if = {
				limit = {
					check_variable = { camp_radicalized_site_count > 0 }
					NOT = { has_idea = generic_radicalized_atrocity_policy }
				}
				add_ideas = generic_radicalized_atrocity_policy
			}
			else_if = {
				limit = {
					has_idea = generic_radicalized_atrocity_policy
					check_variable = { camp_radicalized_site_count = 0 }
				}
				remove_ideas = generic_radicalized_atrocity_policy
			}

			if = {
				limit = {
					has_visible_camp_reform_work = yes
					NOT = { has_idea = generic_reform_pressure }
				}
				add_ideas = generic_reform_pressure
			}
			else_if = {
				limit = {
					has_idea = generic_reform_pressure
					NOT = { has_visible_camp_reform_work = yes }
				}
				remove_ideas = generic_reform_pressure
			}
		}
	}
	camp_rework_refresh_major_country_ideas = yes
	camp_rework_refresh_colonial_country_ideas = yes
}

			camp_rework_register_evidence_state = yes
			if = {
				limit = {
					has_variable = genocide_responsible_country
					var:genocide_responsible_country = { exists = yes }
				}
				var:genocide_responsible_country = {
					genocide_apply_discovery_to_responsible_from_prev_state = yes
				}
			}
			set_temp_variable = { camp_retreat_discovery_processed = 1 }

		camp_rework_expose_network_crisis = yes
	}
