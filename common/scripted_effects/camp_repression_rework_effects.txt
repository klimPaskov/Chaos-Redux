# Country scope. This is the only public initializer for the repression
# system. Special Chaos countries are actively scrubbed instead of receiving
# dormant variables that could later expose ordinary-country UI or missions.
camp_rework_initialize_country_variables = {
	if = {
		limit = { camp_rework_country_is_eligible = yes }
		camp_rework_initialize_eligible_country_variables = yes
	}
	else = {
		camp_rework_exclude_special_country = yes
	}
}

# Terminal country removal stops the selected-site closure without a timeout payout.
camp_rework_cancel_selected_dismantlement_on_terminal = {
	if = {
		limit = { has_active_mission = camp_gui_selected_dismantlement_mission }
		camp_rework_gui_cancel_selected_dismantlement = yes
		remove_mission = camp_gui_selected_dismantlement_mission
	}
}

camp_rework_register_country_for_monthly_pulse = {
	if = {
		limit = { camp_rework_country_is_eligible = yes }
		camp_rework_initialize_eligible_country_variables = yes
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.camp_repression_active_countries
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.camp_repression_active_countries
				value = THIS
			}
		}
		camp_rework_ensure_country_timer = yes
	}
	else = {
		camp_rework_exclude_special_country = yes
	}
}

# One delayed country event is outstanding until it either reschedules or clears its flag.
camp_rework_ensure_country_timer = {
	if = {
		limit = { NOT = { hidden_trigger = { has_country_flag = camp_admin_dispatch_timer_active } } }
		set_country_flag = camp_admin_dispatch_timer_active
		camp_rework_schedule_country_timer = yes
	}
}

# Country scope. Schedules the chain's next dispatch event and records the
# earliest date it may run. An event left over from an earlier chain (before
# annexation and re-release) arrives before that date and stops without
# rescheduling, so only one chain stays outstanding.
camp_rework_schedule_country_timer = {
	set_temp_variable = { chaosx_date_days = constant:camp_admin_dispatch.period_days }
	subtract_from_temp_variable = { chaosx_date_days = constant:camp_admin_dispatch.due_tolerance_days }
	set_temp_variable = { chaosx_date_offset = constant:camp_admin_value.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { camp_admin_dispatch_timer_earliest = global.date }
	add_to_variable = { camp_admin_dispatch_timer_earliest = chaosx_date_offset }
	country_event = {
		id = camp_administration_country.2
		days = constant:camp_admin_dispatch.period_days
	}
}

camp_rework_refresh_state_site_profile = {
	if = { limit = { NOT = { has_variable = camp_state_evidence_depth } } set_variable = { camp_state_evidence_depth = 0 } }
	if = { limit = { NOT = { has_variable = camp_state_resistance_accumulation } } set_variable = { camp_state_resistance_accumulation = 0 } }

	if = { limit = { NOT = { has_variable = camp_state_pool_type } } set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none } }

	set_variable = { camp_state_site_type = constant:camp_rework_site_type.none }
	if = {
		limit = {
			OR = {
				has_state_flag = genocide_restricted_chemical_site_active
				has_state_flag = camp_rework_chemical_method_active
				has_state_flag = camp_rework_biological_method_active
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.contaminated }
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = camp_rework_radicalized_site
				hidden_trigger = { has_state_flag = genocide_extermination_site }
				extermination_camp > 0
			}
		}
		set_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized }
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = camp_rework_experiment_site
				has_state_flag = genocide_auschwitz_experiment_site
				has_state_flag = genocide_ss_laboratory_site
				has_state_flag = genocide_japanese_biowarfare_atrocity_site

			OR = {

					country_exists = ROOT

			}

				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.expanded_labor } }
				set_state_flag = camp_rework_expanded_labor_site

			}

				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.radicalized } }
				set_state_flag = camp_rework_radicalized_site
			}
			else_if = {
				limit = { check_variable = { camp_register_site_type = constant:camp_rework_site_type.contaminated } }
				set_state_flag = camp_rework_contaminated_evidence_site
			}
		}

		if = {
			limit = {
				OR = {
					NOT = { has_variable = camp_state_pool_type }
					check_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none }
				}
				country_exists = ROOT

		}

				NOT = { hidden_trigger = { has_global_flag = camp_rework_migration_in_progress } }
				NOT = { hidden_trigger = { has_state_flag = camp_rework_historical_baseline_registration } }
			}
			camp_rework_apply_core_fallback_penalty = yes
		}

		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.genocide_active_camp_states
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.genocide_active_camp_states
				value = THIS
			}
		}

		camp_rework_refresh_state_site_profile = yes
		camp_admin_initialize_site = yes
		camp_admin_reopen_site = yes
		if = {
			limit = {
				OR = {
					check_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized }
					check_variable = { camp_state_site_type = constant:camp_rework_site_type.contaminated }
				}

			limit = { check_variable = { camp_state_site_type > constant:camp_rework_site_type.none } }
			set_variable = { camp_state_evidence_site_type = camp_state_site_type }
		}

					NOT = {

							value = PREV
						}
					}
				}
				add_to_array = {
					array = camp_active_site_states
					value = PREV
				}
			}

			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.radicalized } }
				PREV = { add_to_variable = { camp_radicalized_site_count = 1 } }
			}
			else_if = {
				limit = { check_variable = { camp_state_site_type = constant:camp_rework_site_type.contaminated } }
				PREV = { add_to_variable = { camp_contaminated_site_count = 1 } }
			}
		}
	}

	# Inactive evidence remains attributable after a site is dismantled or lost.
	# Active entries are excluded because their contributions were added above.
	for_each_scope_loop = {
		array = camp_evidence_states
		if = {
			limit = {
				has_variable = genocide_responsible_country
				PREV = {
					NOT = {
						is_in_array = {
							array = camp_active_site_states
							value = PREV
						}
					}
				}
			}
			set_temp_variable = { camp_inactive_evidence_value = camp_state_evidence_depth }
			set_temp_variable = { camp_inactive_visibility_value = camp_state_evidence_depth }
			multiply_temp_variable = { camp_inactive_visibility_value = constant:camp_rework_evidence_carry.visibility_factor }
			add_to_temp_variable = { camp_inactive_visibility_value = camp_state_observer_exposure }
			set_temp_variable = { camp_inactive_tribunal_value = camp_state_evidence_depth }
			multiply_temp_variable = { camp_inactive_tribunal_value = constant:camp_rework_evidence_carry.tribunal_factor }
			set_temp_variable = { camp_inactive_reform_value = camp_state_evidence_depth }
			multiply_temp_variable = { camp_inactive_reform_value = constant:camp_rework_evidence_carry.reform_factor }
			PREV = {
				add_to_variable = { camp_evidence_level = camp_inactive_evidence_value }
				add_to_variable = { camp_foreign_visibility = camp_inactive_visibility_value }
				add_to_variable = { camp_tribunal_severity = camp_inactive_tribunal_value }
				add_to_variable = { camp_reform_pressure = camp_inactive_reform_value }
			}
		}
	}

	add_to_variable = { camp_labor_output = camp_policy_labor_output }
	add_to_variable = { camp_coercive_control = camp_policy_coercive_control }
	add_to_variable = { camp_population_loss_index = camp_policy_population_loss }
	add_to_variable = { camp_resistance_pressure = camp_policy_resistance_pressure }
	add_to_variable = { camp_stability_damage = camp_policy_stability_damage }
	add_to_variable = { camp_evidence_level = camp_policy_evidence }
	add_to_variable = { camp_overstretch = camp_policy_overstretch }
	add_to_variable = { camp_foreign_visibility = camp_policy_visibility }
	add_to_variable = { camp_tribunal_severity = camp_policy_tribunal }
	add_to_variable = { camp_hardliner_pressure = camp_policy_hardliner }
	add_to_variable = { camp_democratic_legitimacy_damage = camp_policy_legitimacy }
	add_to_variable = { camp_reform_pressure = camp_policy_reform }

	if = {
		limit = {
			OR = {
				has_government = democratic
				check_variable = { party_popularity@democratic > constant:camp_rework_threshold.democratic_support }
			}
			check_variable = { camp_active_site_count > 0 }
		}
		add_to_variable = { camp_democratic_legitimacy_damage = camp_active_site_count }
		add_to_variable = { camp_reform_pressure = camp_active_site_count }
	}

	clamp_variable = { var = camp_network_reach min = 0 max = constant:camp_rework_threshold.max_site_count }
	clamp_variable = { var = camp_labor_output min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_coercive_control min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_population_loss_index min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_resistance_pressure min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_stability_damage min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_evidence_level min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_overstretch min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_foreign_visibility min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_tribunal_severity min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_hardliner_pressure min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_democratic_legitimacy_damage min = 0 max = constant:camp_rework_threshold.max_value }
	clamp_variable = { var = camp_reform_pressure min = 0 max = constant:camp_rework_threshold.max_value }
	if = {
		limit = {
			check_variable = {
				var = camp_tribunal_severity
				value = constant:camp_rework_threshold.band_severe
				compare = greater_than_or_equals
			}
		}
		set_country_flag = camp_rework_achievement_severe_tribunal_reached
	}

	set_variable = { camp_current_phase = constant:camp_rework_phase.dormant }
	if = {
		limit = {
			OR = {
				has_country_flag = camp_rework_dismantlement_in_progress
				has_country_flag = camp_rework_reform_route_open
				check_variable = { camp_reform_pressure > 0 }
			}
		}
		set_variable = { camp_current_phase = constant:camp_rework_phase.reform }
	}
	else_if = {
		limit = { genocide_country_has_discovered_sites = yes }
		set_variable = { camp_current_phase = constant:camp_rework_phase.discovered }
	}
	else_if = {
		limit = { check_variable = { camp_contaminated_site_count > 0 } }
		set_variable = { camp_current_phase = constant:camp_rework_phase.contaminated }
	}
	else_if = {
		limit = { check_variable = { camp_radicalized_site_count > 0 } }
		set_variable = { camp_current_phase = constant:camp_rework_phase.radicalized }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { camp_expanded_site_count > 0 }
				check_variable = { camp_network_reach > constant:camp_rework_threshold.expanded_reach }
			}
		}
		set_variable = { camp_current_phase = constant:camp_rework_phase.expanded }
	}
	else_if = {
		limit = { check_variable = { camp_active_site_count > 0 } }
		set_variable = { camp_current_phase = constant:camp_rework_phase.active }
	}
	else_if = {
		limit = { has_country_flag = camp_rework_reformed_legacy }
		set_variable = { camp_current_phase = constant:camp_rework_phase.dismantled }
	}

	camp_rework_recalculate_burdens = yes
	camp_rework_sync_legacy_compatibility_values = yes
	camp_rework_apply_ai_cap = yes
	camp_rework_copy_display_values = yes
}

					camp_rework_state_is_actor_or_subject_controlled = yes

							is_in_array = { array = camp_active_site_states value = PREV }

				}

		else_if = { limit = { original_tag = BEL } camp_rework_dispatch_belgium_action = yes }

	}

		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.sov_release_prisoners_for_military_service } }

	}

# ---------------------------------------------------------------------------
# Versioned one-time migration. This is the only broad reconstruction pass.
# ---------------------------------------------------------------------------

camp_rework_run_versioned_migration = {
	camp_admin_integration_bootstrap_known_sites = yes
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = global.camp_rework_schema_version
					value = constant:camp_rework_schema.current
					compare = greater_than_or_equals
				}
			}
		}
		set_global_flag = camp_rework_migration_in_progress
		clear_array = global.genocide_active_camp_states
		clear_array = global.camp_repression_active_countries
		# Pass one clears every destination before any controller can append a
		# state to a different responsible country's arrays.
		every_country = {
			limit = { exists = yes }
			clear_array = camp_active_site_states
			clear_array = camp_evidence_states
			camp_rework_initialize_country_variables = yes
		}
		# Pass two rebuilds state membership and preserves the stored perpetrator
		# even when a different country currently controls the evidence site.
		every_country = {
			limit = { exists = yes }
			every_controlled_state = {
				limit = {
					OR = {
						concentration_camp > 0
						extermination_camp > 0
						gulag_labor_camp_network > 0

					}
				}
				if = {
					limit = { NOT = { has_variable = genocide_responsible_country } }
					set_variable = { genocide_responsible_country = PREV }
				}
				if = {
					limit = {
						OR = {
							NOT = { has_variable = camp_state_pool_type }
							check_variable = { camp_state_pool_type = constant:camp_rework_pool_type.none }
						}
					}
					set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.core_fallback }
					if = {
						limit = {
							is_controlled_by = PREV
							NOT = { is_owned_by = PREV }
							NOT = { is_core_of = PREV }
						}
						set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.occupied_noncore }
					}
					else_if = {
						limit = {
							is_owned_and_controlled_by = PREV
							NOT = { is_core_of = PREV }
						}
						set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.noncore }
					}
					else_if = {
						limit = {
							OR = {
								has_state_flag = camp_rework_periphery_pool
								has_state_flag = camp_rework_political_opposition_pool
								has_state_flag = camp_rework_prison_labor_pool
							}
						}
						set_variable = { camp_state_pool_type = constant:camp_rework_pool_type.periphery }
					}
				}
				if = {
					limit = {
						NOT = { has_state_flag = genocide_historical_quiet_camp }
						NOT = { has_state_flag = camp_rework_annexed_evidence_site }
						OR = {
							concentration_camp > 0
							extermination_camp > 0
							gulag_labor_camp_network > 0
							has_state_flag = genocide_auschwitz_experiment_site
							has_state_flag = genocide_ss_laboratory_site
							has_state_flag = genocide_japanese_biowarfare_atrocity_site
							has_state_flag = genocide_restricted_chemical_site_active
						}
					}
					set_temp_variable = { camp_register_site_type = constant:camp_rework_site_type.none }
					camp_rework_register_active_site = yes
				}
				if = {
					limit = {
						OR = {
							has_state_flag = genocide_site_discovered
							has_state_flag = genocide_evidence_destroyed
							has_state_flag = genocide_evidence_destroy_failed
							has_state_flag = genocide_restricted_chemical_site
						}
					}
					camp_rework_register_evidence_state = yes
				}
			}
		}
		# Pass three recalculates only after cross-country array writes are done.
		every_country = {
			limit = {
				exists = yes
				OR = {
					check_variable = { camp_active_site_states^num > 0 }
					check_variable = { camp_evidence_states^num > 0 }
					has_visible_camp_reform_work = yes
				}
			}
			camp_rework_register_country_for_monthly_pulse = yes
			camp_rework_recalculate_country_values = yes
		}
		set_variable = { global.camp_rework_schema_version = constant:camp_rework_schema.current }
		clr_global_flag = camp_rework_migration_in_progress
	}
}

				}

			}
			else_if = {
				limit = { check_variable = { camp_active_site_count > 0 } }
				add_ideas = generic_detention_network_administration
			}

			if = {
				limit = {
					check_variable = { camp_radicalized_site_count > 0 }
					NOT = { has_idea = generic_radicalized_atrocity_policy }
				}
				add_ideas = generic_radicalized_atrocity_policy

			else_if = {
				limit = {
					has_idea = generic_radicalized_atrocity_policy
					check_variable = { camp_radicalized_site_count = 0 }
				}
				remove_ideas = generic_radicalized_atrocity_policy

				limit = {
					has_visible_camp_reform_work = yes
					NOT = { has_idea = generic_reform_pressure }
				}
				add_ideas = generic_reform_pressure
			}

				limit = {
					has_idea = generic_reform_pressure
					NOT = { has_visible_camp_reform_work = yes }
				}
				remove_ideas = generic_reform_pressure

# State scope. Complete the already-staged generic owner transaction after
# the concrete site or labor assignment exists, then issue the strict receipt.
# The adapter resolves the explicit staged ID again here so the accepted amount
# is the current whole live row after owner mutations, never a quota or death.
camp_rework_commit_exact_cohort_custody = {
	if = {
		limit = {
			check_variable = { camp_rework_exact_cohort_staged = constant:humanitarian_runtime.one }
			check_variable = { camp_rework_exact_cohort_id > constant:humanitarian_runtime.zero }
			check_variable = { camp_rework_exact_cohort_action > constant:migration_custody_action.none }
			check_variable = { camp_rework_exact_cohort_amount > constant:humanitarian_population.zero }
			OR = {
				is_valid_camp_active_site_state = yes
				has_state_flag = camp_rework_labor_project_active
			}
			has_variable = camp_rework_exact_cohort_actor_token
			check_variable = { camp_rework_exact_cohort_actor_token > constant:humanitarian_runtime.zero }
		}
		migration_initialize_runtime = yes
		set_temp_variable = { migration_cohort_custody_id_request = camp_rework_exact_cohort_id }
		set_temp_variable = { migration_cohort_custody_action_request = camp_rework_exact_cohort_action }
		set_temp_variable = { migration_cohort_custody_transaction_proven = global.date }
		set_temp_variable = { migration_cohort_custody_actor_proven = camp_rework_exact_cohort_actor_token }
		set_temp_variable = { migration_cohort_custody_site_proven = ROOT.camp_rework_action_state_id }
		set_temp_variable = { migration_cohort_custody_route_proven = constant:humanitarian_runtime.zero }
		set_temp_variable = { migration_cohort_custody_generation = global.civilian_transfer_generation }
		set_temp_variable = { migration_cohort_custody_revision = camp_rework_exact_cohort_amount }
		set_temp_variable = { migration_cohort_custody_request_id = global.date }
		ROOT = { save_event_target_as = migration_cohort_custody_actor }
		# Re-resolve the staged row before the adapter so route proof represents
		# the current host and whole-row amount, not a stale camp quota.
		set_temp_variable = { migration_cohort_id_request = camp_rework_exact_cohort_id }
		migration_resolve_cohort_origin = yes
		if = {
			limit = {
				check_variable = { migration_cohort_origin_resolution_result = constant:civilian_transfer_route_result.valid }
				check_variable = { migration_cohort_resolved_amount > constant:humanitarian_population.zero }
				has_event_target = migration_cohort_host
				state = event_target:migration_cohort_host
			}
			set_temp_variable = { migration_cohort_custody_route_proven = migration_cohort_resolved_amount }
		}
		set_temp_variable = { migration_cohort_id_request = constant:humanitarian_runtime.zero }
		set_temp_variable = { migration_current_state_custody_result = constant:civilian_transfer_route_result.invalid }
		set_temp_variable = { migration_cohort_custody_already_recorded = constant:humanitarian_runtime.zero }
		set_temp_variable = { migration_cohort_custody_resolved_amount = constant:humanitarian_population.zero }
		migration_record_current_state_cohort_custody_exact = yes
		if = {
			limit = {
				OR = {
					check_variable = { migration_current_state_custody_result = constant:civilian_transfer_route_result.valid }
					check_variable = { migration_cohort_custody_already_recorded = constant:humanitarian_runtime.one }
				}
			}
			set_state_flag = camp_rework_exact_cohort_custody_receipted
			set_variable = { camp_rework_exact_cohort_custody_id = camp_rework_exact_cohort_id }
			set_variable = { camp_rework_exact_cohort_custody_amount = migration_cohort_custody_resolved_amount }
			set_variable = { camp_rework_exact_cohort_custody_action = camp_rework_exact_cohort_action }
		}
		set_temp_variable = { migration_cohort_id_request = constant:humanitarian_runtime.zero }
		clear_variable = camp_rework_exact_cohort_staged
		clear_variable = camp_rework_exact_cohort_id
		clear_variable = camp_rework_exact_cohort_amount
		clear_variable = camp_rework_exact_cohort_action
		clear_variable = camp_rework_exact_cohort_actor_token
	}
}
