# ============================================================================
# CHAOS REDUX - EVENT 006 TRIGGERABLE SCENARIO EFFECTS
# ============================================================================
# Overview:
# - Implements SCN-008 as one Event 006 frozen-plan transaction.
# - Attempts the same 138 selectable current-map-bound packages at every intensity.
# - Ranks anchor uniqueness first, then research readiness and package depth.
# - Preserves content-readiness, living-tag, host-remnant, unique-anchor, and
#   Event 005-origin safety gates.
# - Applies type behavior only after the synchronized release commits.
# - Stores an inspectable package/rejection ledger for the scenario summary.
# ============================================================================

independence_wave_scenario_initialize_settings = {
	if = {
		limit = { NOT = { has_variable = triggerable_scenarios_independence_wave_type } }
		set_variable = { triggerable_scenarios_independence_wave_type = constant:independence_wave_scenario_type.sovereign_scatter }
	}
	if = {
		limit = { NOT = { has_variable = triggerable_scenarios_independence_wave_belligerence_rule } }
		set_variable = { triggerable_scenarios_independence_wave_belligerence_rule = constant:independence_wave_scenario_belligerence_rule.former_hosts }
	}
}

# The one generic Type control exposes six type families. While Universal
# Belligerence is selected, the same control walks its three bounded rules
# before advancing to the next family.
independence_wave_scenario_type_previous = {
	independence_wave_scenario_initialize_settings = yes
	if = {
		limit = {
			check_variable = { triggerable_scenarios_independence_wave_type = constant:independence_wave_scenario_type.universal_belligerence }
			check_variable = {
				var = triggerable_scenarios_independence_wave_belligerence_rule
				value = constant:independence_wave_scenario_belligerence_rule.minimum
				compare = greater_than
			}
		}
		subtract_from_variable = { triggerable_scenarios_independence_wave_belligerence_rule = constant:independence_wave_scenario_registry.one }
	}
	else = {
		subtract_from_variable = { triggerable_scenarios_independence_wave_type = constant:independence_wave_scenario_registry.one }
		if = {
			limit = {
				check_variable = {
					var = triggerable_scenarios_independence_wave_type
					value = constant:independence_wave_scenario_type.minimum
					compare = less_than
				}
			}
			set_variable = { triggerable_scenarios_independence_wave_type = constant:independence_wave_scenario_type.maximum }
		}
		if = {
			limit = { check_variable = { triggerable_scenarios_independence_wave_type = constant:independence_wave_scenario_type.universal_belligerence } }
			set_variable = { triggerable_scenarios_independence_wave_belligerence_rule = constant:independence_wave_scenario_belligerence_rule.maximum }
		}
	}
}

# Outputs are temporary wave and country-value values consumed by the scenario

	}
	else_if = {
		limit = { check_variable = { global.independence_wave_scenario_last_intensity = constant:triggerable_scenario_intensity.high } }
		set_temp_variable = { independence_wave_wave_chaos_band = constant:independence_wave_chaos_band.totalen_chaos }
		set_temp_variable = { independence_wave_wave_territory_level = constant:liberation_territory_level.extended }
		set_temp_variable = { independence_wave_wave_force_level = constant:liberation_force_level.armed }
		set_temp_variable = { independence_wave_scenario_legitimacy_delta = constant:independence_wave_scenario_country_value.high_legitimacy }
		set_temp_variable = { independence_wave_scenario_recognition_delta = constant:independence_wave_scenario_country_value.high_recognition }
		set_temp_variable = { independence_wave_scenario_capacity_delta = constant:independence_wave_scenario_country_value.high_capacity }
		set_temp_variable = { independence_wave_scenario_security_delta = constant:independence_wave_scenario_country_value.high_security }
		set_temp_variable = { independence_wave_scenario_instability_delta = constant:independence_wave_scenario_country_value.high_instability }
	}
	else_if = {
		limit = { check_variable = { global.independence_wave_scenario_last_intensity = constant:triggerable_scenario_intensity.maximum } }
		set_temp_variable = { independence_wave_wave_chaos_band = constant:independence_wave_chaos_band.world_collapse }
		set_temp_variable = { independence_wave_wave_territory_level = constant:liberation_territory_level.extended }
		set_temp_variable = { independence_wave_wave_force_level = constant:liberation_force_level.high_chaos }
		set_temp_variable = { independence_wave_scenario_legitimacy_delta = constant:independence_wave_scenario_country_value.maximum_legitimacy }
		set_temp_variable = { independence_wave_scenario_recognition_delta = constant:independence_wave_scenario_country_value.maximum_recognition }
		set_temp_variable = { independence_wave_scenario_capacity_delta = constant:independence_wave_scenario_country_value.maximum_capacity }
		set_temp_variable = { independence_wave_scenario_security_delta = constant:independence_wave_scenario_country_value.maximum_security }
		set_temp_variable = { independence_wave_scenario_instability_delta = constant:independence_wave_scenario_country_value.maximum_instability }
	}

	# Great Partition advances the territorial package by one tier before any
	# state is reserved. The ordinary intensity ladder remains authoritative,
	# while the partition type consumes more optional territory where the same
	# unique-anchor and host-remnant proofs permit it.
	if = {
		limit = {
			check_variable = { global.independence_wave_scenario_last_type = constant:independence_wave_scenario_type.great_partition }
			check_variable = { independence_wave_wave_territory_level = constant:liberation_territory_level.anchor }
		}
		set_temp_variable = { independence_wave_wave_territory_level = constant:liberation_territory_level.compact }
	}
	else_if = {
		limit = {
			check_variable = { global.independence_wave_scenario_last_type = constant:independence_wave_scenario_type.great_partition }
			check_variable = { independence_wave_wave_territory_level = constant:liberation_territory_level.compact }
		}
		set_temp_variable = { independence_wave_wave_territory_level = constant:liberation_territory_level.extended }
	}
}

# Attempts every bound registry row. Unready content is recorded rather than
# reserved; living tags and territorial failures pass through the shared
# reservation API so their precise rejection reason is preserved.
independence_wave_scenario_attempt_ranked_packages = {
	for_each_loop = {
		array = global.independence_wave_scenario_ranked_package_ids
		value = independence_wave_scenario_dispatch_package_id
		# Scenario preflight shares the normal execution attestation trigger.
		# Temporary variables are unscoped, so this bridge remains visible while
		# the candidate-country scope checks the immutable content authority.
		set_temp_variable = { independence_wave_execution_package_id = independence_wave_scenario_dispatch_package_id }
		independence_wave_scenario_load_dispatch_package = yes
		if = {
			limit = {
				has_event_target = liberation_candidate_country
				event_target:liberation_candidate_country = { is_independence_wave_scenario_package_preflight_ready = yes }
			}
			set_temp_variable = { independence_wave_scenario_rejection_count_before = global.liberation_plan_rejected_package_ids^num }
			independence_wave_scenario_reserve_dispatch_package = yes
			if = {
				limit = {
					check_variable = {
						var = global.liberation_plan_rejected_package_ids^num
						value = independence_wave_scenario_rejection_count_before
						compare = greater_than
					}
				}
				add_to_array = { array = global.independence_wave_scenario_blocked_country_entries value = event_target:liberation_candidate_country }
			}
		}
		else = {
			add_to_variable = { global.independence_wave_plan_attempt_count = constant:independence_wave_scenario_registry.one }
			set_temp_variable = { liberation_rejected_package_id = independence_wave_scenario_dispatch_package_id }
			set_temp_variable = { liberation_reject_reason = constant:liberation_plan_reject_reason.package_unready }
			liberation_release_record_rejection = yes
			if = {
				limit = { has_event_target = liberation_candidate_country }
				add_to_array = { array = global.independence_wave_scenario_blocked_country_entries value = event_target:liberation_candidate_country }
			}
		}
	}
}

independence_wave_allocate_scenario_packages = {
	clr_global_flag = independence_wave_plan_contribution_ready
	clr_global_flag = independence_wave_plan_exact_count_failed
	if = {
		limit = {
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.allocating }
			check_variable = { global.liberation_plan_mode = constant:liberation_plan_mode.triggerable_scenario }
			check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.independence_wave }
			has_global_flag = liberation_release_plan_includes_event6
		}
		independence_wave_begin_plan_contribution = yes
		independence_wave_scenario_set_intensity_tuning = yes
		set_variable = { global.independence_wave_plan_target_count = constant:independence_wave_scenario_registry.bound_package_count }
		set_variable = { global.independence_wave_plan_chaos_band = independence_wave_wave_chaos_band }
		independence_wave_scenario_rebuild_ranked_registry = yes
		independence_wave_scenario_attempt_ranked_packages = yes
		independence_wave_expand_selected_optional_territory = yes

		set_variable = { global.independence_wave_plan_target_count = global.independence_wave_plan_selected_count }
		set_variable = { global.liberation_plan_expected_country_count = global.independence_wave_plan_selected_count }
		if = {
			limit = {
				check_variable = {
					var = global.independence_wave_plan_selected_count
					value = constant:independence_wave_scenario_registry.no_packages
					compare = greater_than
				}
				NOT = { has_global_flag = independence_wave_plan_optional_expansion_failed }
				independence_wave_plan_metadata_arrays_are_aligned = yes
			}
			set_global_flag = independence_wave_plan_contribution_ready
		}
		else = {
			set_global_flag = independence_wave_plan_exact_count_failed
			if = {
				limit = { has_global_flag = independence_wave_plan_optional_expansion_failed }
				set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.aligned_array_failure }
			}
			else = {
				set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.insufficient_pool }
			}
		}
	}
	else = {
		set_global_flag = independence_wave_plan_exact_count_failed
		set_variable = { global.liberation_plan_last_failure = constant:liberation_plan_reject_reason.stale_plan }
	}
}

# Current scope: one released country. The Event 6 country opens the accepted
# separation war against its surviving former host. Failed or unlawful
# declarations become regional threat missions and never count as witnesses.
# Universal Former Hosts enables the same per-launch belligerence reservation
# used by the neighboring-release and nearby-nonleague rules, so two releases
# cannot open duplicate wars against one surviving host. Wars of Separation
# leaves this policy disabled and retains one viable host war per release.
independence_wave_scenario_start_host_war = {
	save_event_target_as = independence_wave_scenario_war_actor
	set_temp_variable = { independence_wave_scenario_war_started = constant:independence_wave_scenario_registry.no_packages }
	set_temp_variable = { independence_wave_scenario_host_target_saved = constant:independence_wave_scenario_registry.no_packages }
	clr_country_flag = independence_wave_danger_former_host_war_actor
	clear_variable = independence_wave_danger_former_host_war_batch_id
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_variable = independence_wave_former_host
			var:independence_wave_former_host = {
				exists = yes
				NOT = { tag = event_target:independence_wave_scenario_war_actor }
				OR = {
					NOT = { check_variable = { global.independence_wave_scenario_former_host_unique_policy = constant:independence_wave_scenario_registry.one } }
					NOT = { has_country_flag = independence_wave_scenario_belligerence_targeted }
				}
			}
		}
		var:independence_wave_former_host = { save_event_target_as = independence_wave_scenario_war_host }
		set_temp_variable = { independence_wave_scenario_host_target_saved = constant:independence_wave_scenario_registry.one }
		if = {
			limit = {
				has_event_target = independence_wave_scenario_war_host
				NOT = { has_war_with = event_target:independence_wave_scenario_war_host }
				can_declare_war_on = event_target:independence_wave_scenario_war_host
			}
			if = {
				limit = { check_variable = { global.independence_wave_scenario_former_host_unique_policy = constant:independence_wave_scenario_registry.one } }
				event_target:independence_wave_scenario_war_host = {
					set_country_flag = independence_wave_scenario_belligerence_targeted
					add_to_array = { array = global.independence_wave_scenario_belligerence_targets value = THIS }
				}
			}
			event_target:independence_wave_scenario_war_actor = {
				declare_war_on = {
					target = event_target:independence_wave_scenario_war_host
					type = annex_everything
				}
				if = {
					limit = { has_war_with = event_target:independence_wave_scenario_war_host }
					set_temp_variable = { independence_wave_scenario_war_started = constant:independence_wave_scenario_registry.one }
				}
			}
		}
		if = {
			limit = {
				check_variable = { global.independence_wave_scenario_former_host_unique_policy = constant:independence_wave_scenario_registry.one }
				check_variable = { independence_wave_scenario_host_target_saved = constant:independence_wave_scenario_registry.one }
				NOT = { check_variable = { independence_wave_scenario_war_started = constant:independence_wave_scenario_registry.one } }
			}
			event_target:independence_wave_scenario_war_host = {
				clr_country_flag = independence_wave_scenario_belligerence_targeted
				remove_from_array = { array = global.independence_wave_scenario_belligerence_targets value = THIS }
			}
		}
	}
	if = {
		limit = { check_variable = { independence_wave_scenario_war_started = constant:independence_wave_scenario_registry.one } }
		add_to_variable = { global.independence_wave_scenario_active_war_count = constant:independence_wave_scenario_registry.one }
		add_to_variable = { global.independence_wave_danger_former_host_war_count = constant:independence_wave_scenario_registry.one }
		set_country_flag = independence_wave_scenario_host_war_active
		set_country_flag = independence_wave_danger_former_host_war_actor
		set_variable = { independence_wave_danger_former_host_war_batch_id = global.independence_wave_danger_former_host_war_batch_id }
		if = {
			limit = { NOT = { has_event_target = independence_wave_first_former_host_war_actor } }
			save_global_event_target_as = independence_wave_first_former_host_war_actor
		}
	}
	else = {
		set_country_flag = independence_wave_scenario_regional_threat_mission
	}
}

independence_wave_scenario_freeze_summary = {
	clear_array = global.independence_wave_scenario_released_package_ids
	clear_array = global.independence_wave_scenario_blocked_package_ids
	clear_array = global.independence_wave_scenario_blocked_reasons
	set_variable = { global.independence_wave_scenario_host_count = constant:independence_wave_scenario_registry.no_packages }
	if = {
		limit = {
			has_variable = global.independence_wave_scenario_last_plan_id
			check_variable = {
				var = global.independence_wave_scenario_last_plan_id
				value = global.liberation_plan_id
				compare = equals
			}
			check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.independence_wave }
		}
		for_each_loop = {
			array = global.independence_wave_plan_selected_package_ids
			value = independence_wave_scenario_summary_package_id
			add_to_array = { array = global.independence_wave_scenario_released_package_ids value = independence_wave_scenario_summary_package_id }
		}
		for_each_loop = {
			array = global.liberation_plan_rejected_package_ids
			value = independence_wave_scenario_summary_package_id
			add_to_array = { array = global.independence_wave_scenario_blocked_package_ids value = independence_wave_scenario_summary_package_id }
		}
		for_each_loop = {
			array = global.liberation_plan_rejection_reasons
			value = independence_wave_scenario_summary_reason
			add_to_array = { array = global.independence_wave_scenario_blocked_reasons value = independence_wave_scenario_summary_reason }
		}
		set_variable = { global.independence_wave_scenario_host_count = global.liberation_plan_hosts^num }
	}
	set_variable = { global.independence_wave_scenario_released_count = global.independence_wave_scenario_released_package_ids^num }
	set_variable = { global.independence_wave_scenario_bound_blocked_count = global.independence_wave_scenario_blocked_package_ids^num }
	set_variable = { global.independence_wave_scenario_disabled_unbound_count = constant:independence_wave_scenario_registry.disabled_unbound_package_count }
	independence_wave_scenario_append_unbound_registry_rows = yes
	set_variable = { global.independence_wave_scenario_blocked_count = global.independence_wave_scenario_blocked_package_ids^num }
	set_variable = { global.independence_wave_scenario_territory_level = independence_wave_wave_territory_level }
	set_variable = { global.independence_wave_scenario_force_level = independence_wave_wave_force_level }
	set_variable = { global.independence_wave_scenario_summary_date = global.date }
}

independence_wave_trigger_scenario = {
	independence_wave_scenario_initialize_settings = yes
	independence_wave_scenario_reset_summary = yes
	clr_global_flag = independence_wave_scenario_committed
	clr_global_flag = independence_wave_scenario_failed
	clr_global_flag = independence_wave_scenario_finalization_failed
	clear_variable = global.independence_wave_scenario_last_plan_id
	clear_variable = global.independence_wave_scenario_last_failure
	clear_variable = global.independence_wave_execution_last_failure
	independence_wave_scenario_set_intensity_tuning = yes

	set_temp_variable = { liberation_call_mode = constant:liberation_plan_mode.triggerable_scenario }
	set_temp_variable = { liberation_call_plan_owner = constant:liberation_plan_owner.independence_wave }
	set_temp_variable = { liberation_call_expected_country_count = constant:independence_wave_scenario_registry.bound_package_count }
	liberation_release_begin_plan = yes
	if = {
		limit = {
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.collecting }
			check_variable = { global.liberation_plan_mode = constant:liberation_plan_mode.triggerable_scenario }
			check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.independence_wave }
		}
		set_variable = { global.independence_wave_scenario_last_plan_id = global.liberation_plan_id }
		liberation_release_enter_allocation_phase = yes
	}
	if = {
		limit = { check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.allocating } }
		independence_wave_allocate_scenario_packages = yes
	}
	if = {
		limit = { has_global_flag = independence_wave_plan_contribution_ready }
		independence_wave_execute_standalone_frozen_plan = yes
	}
	if = {
		limit = {
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.committed }
			NOT = { has_global_flag = independence_wave_execution_failed }
		}
		set_global_flag = independence_wave_scenario_committed
		clear_variable = global.independence_wave_scenario_last_failure
		independence_wave_scenario_set_intensity_tuning = yes
		independence_wave_scenario_prepare_all_releases = yes
		set_global_flag = independence_wave_scenario_application_open
		independence_wave_scenario_apply_type = yes
		clr_global_flag = independence_wave_scenario_application_open
		independence_wave_scenario_freeze_summary = yes
		independence_wave_achievement_begin_scenario_survival_challenge = yes
		country_event = { id = chaosx.nr6.2 }
		country_event = { id = chaosx.triggerable_scenarios.80 days = 1 }
	}
	else = {
		set_global_flag = independence_wave_scenario_failed
		if = {
			limit = { has_variable = global.independence_wave_execution_last_failure }
			set_variable = { global.independence_wave_scenario_last_failure = global.independence_wave_execution_last_failure }
		}
		else_if = {
			limit = { has_variable = global.liberation_plan_last_failure }
			set_variable = { global.independence_wave_scenario_last_failure = global.liberation_plan_last_failure }
		}
		else = {
			set_variable = { global.independence_wave_scenario_last_failure = constant:liberation_plan_reject_reason.unsafe_instantiation }
		}
		independence_wave_scenario_set_intensity_tuning = yes
		independence_wave_scenario_freeze_summary = yes
		if = {
			limit = { has_liberation_release_plan_finalization_failed = yes }
			set_global_flag = independence_wave_scenario_finalization_failed
		}
		else_if = {
			limit = { has_global_flag = liberation_release_execution_started }
			liberation_release_execute_compensating_rollback = yes
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.collecting }
					check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.allocating }
					check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.locked }
					check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.aborted }
				}
				NOT = { has_global_flag = liberation_release_execution_started }
			}
			liberation_release_restore_host_capitals_before_execution = yes
			if = {
				limit = { NOT = { has_global_flag = liberation_release_capital_restore_failed } }
				independence_wave_clear_plan_contribution = yes
				liberation_release_abort_plan = yes
			}
			else = { liberation_release_abort_plan = yes }
		}
		country_event = { id = chaosx.triggerable_scenarios.80 days = 1 }
	}
}
