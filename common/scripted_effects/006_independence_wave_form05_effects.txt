# Candidate-country scope with ROOT as the charter carrier.
independence_wave_form05_store_invitation_from_root = {
	independence_wave_form05_clear_country_invitation = yes
	set_country_flag = independence_wave_form05_charter_invited
	if = {
		limit = { ROOT = { original_tag = COR } }
		set_country_flag = independence_wave_form05_invited_by_cor
	}
	else_if = {
		limit = { ROOT = { original_tag = ARX } }
		set_country_flag = independence_wave_form05_invited_by_arx
	}
	else_if = {
		limit = { ROOT = { original_tag = ASX } }
		set_country_flag = independence_wave_form05_invited_by_asx
	}
}

# Carrier-country scope. This is an explicit action-time pass over the Event
# 006 active-country registry and never a periodic all-country iteration.
independence_wave_form05_invite_connected_members = {
	independence_wave_reconcile_active_registry = yes
	if = {
		limit = { independence_wave_active_country_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_form05_invitation_index = constant:independence_wave_formable_tuning.first_array_index }
		while_loop_effect = {
			limit = {
				check_variable = { var = independence_wave_form05_invitation_index value = global.independence_wave_active_countries^num compare = less_than }
			}
			var:global.independence_wave_active_countries^independence_wave_form05_invitation_index = {
				if = {
					limit = {
						NOT = { tag = ROOT }
						has_independence_wave_form05_maritime_connection_to_root = yes
					}
					independence_wave_form05_store_invitation_from_root = yes
				}
			}
			add_to_temp_variable = { independence_wave_form05_invitation_index = constant:independence_wave_formable_tuning.loop_increment }
		}
	}
}

# Carrier-country scope. Clears FORM-05 response state only from active Event
# 006 entries, including any compatible island packages added later.
independence_wave_form05_clear_all_invitations = {
	independence_wave_reconcile_active_registry = yes
	if = {
		limit = { independence_wave_active_country_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_form05_cleanup_index = constant:independence_wave_formable_tuning.first_array_index }
		while_loop_effect = {
			limit = {
				check_variable = { var = independence_wave_form05_cleanup_index value = global.independence_wave_active_countries^num compare = less_than }
			}
			var:global.independence_wave_active_countries^independence_wave_form05_cleanup_index = {
				if = {
					limit = {
						has_country_flag = independence_wave_formable_family_selected
						check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.mediterranean_island_league }
					}
					independence_wave_form05_clear_country_invitation = yes
				}
			}
			add_to_temp_variable = { independence_wave_form05_cleanup_index = constant:independence_wave_formable_tuning.loop_increment }
		}
	}
}

# Candidate-country scope. The response markers are mutually exclusive and the
# carrier count can increase only once because the decision then disappears.
independence_wave_form05_accept_invitation = {
	if = {
		limit = { can_independence_wave_form05_answer_invitation = yes }
		set_country_flag = independence_wave_form05_response_accepted
		clr_country_flag = independence_wave_form05_response_withheld
		clr_country_flag = independence_wave_form05_charter_invited
		if = {
			limit = { has_country_flag = independence_wave_form05_invited_by_cor }
			COR = {
				if = {
					limit = { is_independence_wave_form05_active_charter_carrier = yes }
					add_to_variable = { independence_wave_form05_consent_count = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
		}
		else_if = {
			limit = { has_country_flag = independence_wave_form05_invited_by_arx }
			ARX = {
				if = {
					limit = { is_independence_wave_form05_active_charter_carrier = yes }
					add_to_variable = { independence_wave_form05_consent_count = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
		}
		else_if = {
			limit = { has_country_flag = independence_wave_form05_invited_by_asx }
			ASX = {
				if = {
					limit = { is_independence_wave_form05_active_charter_carrier = yes }
					add_to_variable = { independence_wave_form05_consent_count = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
		}
	}
}
