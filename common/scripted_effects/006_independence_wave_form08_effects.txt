			limit = { check_variable = { var = independence_wave_form08_member_index value = independence_wave_formable_member_country_entries^num compare = less_than } }

					limit = {
						NOT = { tag = ROOT }
						is_independence_wave_form08_member_candidate = yes

						limit = { has_country_flag = independence_wave_form0124_full_integration_authorized }

					}
					add_to_temp_variable = { independence_wave_form08_additional_members_processed = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
			add_to_temp_variable = { independence_wave_form08_member_index = constant:independence_wave_formable_tuning.loop_increment }

			limit = {
