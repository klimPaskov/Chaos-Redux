# ============================================================================
# CHAOS REDUX - STARTUP HISTORY COMPATIBILITY EFFECTS
# ============================================================================
# Additive startup grants that replace copied vanilla history overrides.
# Existing-country technologies, stockpiles, static scientists, commander traits,
# breakthrough progress, special projects, delayed events, and starting facility
# placement belong here unless they define a new Chaos Redux country package.
# Static scientist identities are recruited by history/general; this file marks
# only the identity receipts and date-specific country setup. Scientist skill is
# independent from offensive authorization supplied by advisor cards.
# Every `date` check in this file is structural: it detects the bookmark the
# campaign began on (1936 or 1939) for one-time setup, or places a real person's
# documented career dates. No check here gates a player action or mechanic.
# ============================================================================

@eng_1939_respirator_factories = 1
@eng_1939_respirator_efficiency = 100
@fra_1939_respirator_factories = 1
@fra_1939_respirator_efficiency = 100

chaosx_apply_startup_history_grants = {
	if = {
		limit = {
			NOT = { has_global_flag = chaosx_startup_history_grants_applied }
		}
		set_global_flag = chaosx_startup_history_grants_applied

		chaosx_startup_grant_ast = yes
		chaosx_startup_grant_bel = yes
		chaosx_startup_grant_bul = yes
		chaosx_startup_grant_can = yes
		chaosx_startup_grant_cze = yes
		chaosx_startup_grant_eng = yes
		chaosx_startup_grant_fra = yes
		chaosx_startup_grant_ger = yes
		chaosx_startup_grant_gre = yes
		chaosx_startup_grant_hol = yes
		chaosx_startup_grant_hun = yes
		chaosx_startup_grant_ita = yes
		chaosx_startup_grant_jap = yes
		chaosx_startup_grant_lib = yes
		chaosx_startup_grant_man = yes
		chaosx_startup_grant_nzl = yes
		chaosx_startup_grant_pol = yes
		chaosx_startup_grant_raj = yes
		chaosx_startup_grant_rom = yes
		chaosx_startup_grant_saf = yes
		chaosx_startup_grant_sov = yes
		chaosx_startup_grant_spr = yes
		chaosx_startup_grant_tur = yes
		chaosx_startup_grant_usa = yes
		chaosx_startup_grant_yug = yes
		# Both historical bookmarks use the same military coverage bands, sized
		# against the deployed force at the selected start date.
		every_country = {
			chaosx_startup_mark_1936_cbrn_force_scaled_profile = yes
		}
		every_country = {
			limit = {
				has_country_flag = cbrn_historical_1936_force_scaled_protection
				NOT = { tag = ENG }
			}
			# Preserve every pre-existing gas-mask model before the common
			# profile's fixed receipt is reconciled into force-scaled real stock.
			set_variable = { cbrn_historical_1936_preprofile_basic = num_equipment@gas_mask_equipment_1 }
			set_variable = { cbrn_historical_1936_preprofile_reconditioned = num_equipment@gas_mask_equipment_reconditioned }
			set_variable = { cbrn_historical_1936_preprofile_improved = num_equipment@gas_mask_equipment_2 }
			set_variable = { cbrn_historical_1936_preprofile_advanced = num_equipment@gas_mask_equipment_3 }
			set_variable = { cbrn_historical_1936_preprofile_sealed = num_equipment@gas_mask_equipment_4 }
			set_country_flag = cbrn_historical_1936_profile_snapshot_ready
		}
		chaosx_apply_starting_cbrn_mask_profiles = yes
		ENG = {
			if = {
				limit = { has_country_flag = cbrn_historical_1936_force_scaled_protection }
				chaosx_apply_historical_eng_1936_force_scaled_profile = yes
			}
		}
		every_country = {
			limit = {
				has_country_flag = cbrn_historical_1936_force_scaled_protection
				NOT = { tag = ENG }
			}
			chaosx_apply_historical_1936_force_scaled_profile = yes
		}
		# The 1939 British respirator receipt must run in an ENG-root event so
		# the shared state distributor debits British stock and core states.
		if = {
			limit = { date > 1938.12.31 }
			ENG = {
				if = {
					limit = { has_country_flag = cbrn_uk_1939_mass_respirator_issue }
					country_event = { id = cbrn_protection.3 days = 0 }
				}
			}
		}
		# One-time achievement eligibility receipts are written after the
		# accepted starting profiles. They prevent country or profile switching
		# from manufacturing a starting condition later in the campaign.
		every_country = {
			limit = { is_ai = no }
			set_country_flag = cbrn_achievement_start_country_eligible
			if = {
				limit = { is_major = yes }
				set_country_flag = cbrn_achievement_starting_major_power
			}
			if = {
				limit = {
					OR = {
						cbrn_profile_is_mass_civil_defence = yes
						cbrn_profile_is_civil_defence_network = yes
					}
				}
				set_country_flag = cbrn_achievement_starting_civil_defence_profile
			}
		}
		every_country = {
			limit = { exists = yes }
		}
	}
}

chaosx_startup_sync_chemical_technology_unlocks = {
	chem_sync_chemical_tactic_preferred_weight_ideas = yes

	if = {
		limit = {
			has_tech = basic_gas_masks
			NOT = { has_tactic = tactic_gas_mask_defense }
		}
		unlock_tactic = tactic_gas_mask_defense
	}

	if = {
		limit = {
			NOT = { has_tactic = tactic_chemical_shelling }
			OR = {
				has_tech = chlorine_gas
				has_tech = phosgene
				has_tech = mustard_gas
				has_tech = lewisite
				has_tech = sarin
				has_tech = soman
			}
		}
		unlock_tactic = tactic_chemical_shelling
	}
}

chaosx_startup_mark_historical_1939_cbrn_setup = {
	if = {
		limit = { date > 1938.12.31 }
		# This receipt distinguishes a historical 1939 setup from a 1936
		# bookmark. It is intentionally country-scoped so monthly protection
		# logic can require this setup or a verified live threat.
		set_country_flag = cbrn_historical_1939_start
	}
}

# Historical starts use a force-scaled military receipt instead of the fixed
# warehouse matrix. The shared matrix still owns the actual profile
# lifecycle, registration, readiness, and maintenance setup; this marker only
# supplies the historical band that the startup reconciliation consumes.
chaosx_startup_mark_1936_cbrn_force_scaled_profile = {
	if = {
		limit = {
			OR = {
				tag = ENG

				tag = ITA
				tag = JAP

				tag = AST
				tag = NZL
				tag = SAF
				tag = ROM
				tag = YUG
				tag = TUR
				tag = SPR
				tag = CHI
				tag = PRC
				tag = SHX
				tag = GXC
				tag = YUN
				tag = XSM
				tag = SIK
				tag = MAN
				tag = BUL
				tag = HUN
				tag = GRE
				tag = RAJ
				tag = AUS
				tag = POR
				tag = SWE
				tag = NOR
				tag = DEN
				tag = FIN
				tag = BRA
				tag = ARG
				tag = MEX
				tag = CHL
			}
		}
		set_country_flag = cbrn_historical_1936_force_scaled_protection
		set_variable = {
			cbrn_historical_1936_replacement_reserve_ratio = constant:startup_history_protection_targets.replacement_reserve_ratio_1936
		}
		if = {
			limit = {
				OR = {
					tag = ENG
					tag = FRA
					tag = GER
					tag = SOV
					tag = USA
				}
			}
			set_variable = {
				cbrn_historical_1936_military_issue_target = constant:startup_history_protection_targets.major_military_issue_target_1936
			}
		}
		else_if = {
			limit = { tag = ITA }
			set_variable = {
				cbrn_historical_1936_military_issue_target = constant:startup_history_protection_targets.italy_active_war_issue_target_1936
			}
		}
		else_if = {
			limit = { tag = JAP }
			set_variable = {
				cbrn_historical_1936_military_issue_target = constant:startup_history_protection_targets.japan_limited_program_issue_target_1936
			}
		}
		else = {
			set_variable = {
				cbrn_historical_1936_military_issue_target = constant:startup_history_protection_targets.smaller_defensive_issue_target_1936
			}
		}
	}
}

# Reconcile the common fixed profile into the historical force-scaled
# receipt for every marked country except ENG, whose shared core helper owns the
# same transaction. This preserves pre-existing stock, removes only the common
# startup profile's issued/civilian accounting, issues the correct active share
# from real stock, and leaves the separate replacement reserve in the warehouse.
chaosx_apply_historical_1936_force_scaled_profile = {
	if = {
		limit = {
			NOT = { tag = ENG }
			has_country_flag = cbrn_historical_1936_force_scaled_protection
			has_country_flag = cbrn_historical_1936_profile_snapshot_ready
			NOT = { has_country_flag = cbrn_historical_1936_force_scaled_profile_applied }
			has_variable = cbrn_historical_1936_military_issue_target
			has_variable = cbrn_historical_1936_replacement_reserve_ratio
		}
		cbrn_refresh_country_mask_snapshot = yes
		set_temp_variable = { cbrn_historical_1936_replacement_reserve = cbrn_military_mask_requirement }
		multiply_temp_variable = {
			cbrn_historical_1936_replacement_reserve = cbrn_historical_1936_replacement_reserve_ratio
		}
		round_temp_variable = cbrn_historical_1936_replacement_reserve

		set_temp_variable = { cbrn_historical_1936_current_basic = num_equipment@gas_mask_equipment_1 }
		multiply_temp_variable = { cbrn_historical_1936_current_basic = -1 }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_1
			amount = cbrn_historical_1936_current_basic
		}
		set_temp_variable = { cbrn_historical_1936_current_reconditioned = num_equipment@gas_mask_equipment_reconditioned }
		multiply_temp_variable = { cbrn_historical_1936_current_reconditioned = -1 }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_reconditioned
			amount = cbrn_historical_1936_current_reconditioned
		}
		set_temp_variable = { cbrn_historical_1936_current_improved = num_equipment@gas_mask_equipment_2 }
		multiply_temp_variable = { cbrn_historical_1936_current_improved = -1 }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_2
			amount = cbrn_historical_1936_current_improved
		}
		set_temp_variable = { cbrn_historical_1936_current_advanced = num_equipment@gas_mask_equipment_3 }
		multiply_temp_variable = { cbrn_historical_1936_current_advanced = -1 }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_3
			amount = cbrn_historical_1936_current_advanced
		}
		set_temp_variable = { cbrn_historical_1936_current_sealed = num_equipment@gas_mask_equipment_4 }
		multiply_temp_variable = { cbrn_historical_1936_current_sealed = -1 }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_4
			amount = cbrn_historical_1936_current_sealed
		}

		set_temp_variable = { cbrn_historical_1936_basic_target = cbrn_historical_1936_preprofile_basic }
		add_to_temp_variable = {
			cbrn_historical_1936_basic_target = cbrn_historical_1936_replacement_reserve
		}
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_1
			amount = cbrn_historical_1936_basic_target
		}
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_reconditioned
			amount = cbrn_historical_1936_preprofile_reconditioned
		}
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_2
			amount = cbrn_historical_1936_preprofile_improved
		}
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_3
			amount = cbrn_historical_1936_preprofile_advanced
		}
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_4
			amount = cbrn_historical_1936_preprofile_sealed
		}

		set_variable = { cbrn_military_issued_mask_basic = constant:cbrn_math.zero }
		set_variable = { cbrn_military_issued_mask_reconditioned = constant:cbrn_math.zero }
		set_variable = { cbrn_military_issued_mask_improved = constant:cbrn_math.zero }
		set_variable = { cbrn_military_issued_mask_advanced = constant:cbrn_math.zero }
		set_variable = { cbrn_military_issued_mask_sealed = constant:cbrn_math.zero }
		set_variable = { cbrn_military_mask_filter_condition = constant:cbrn_mask_filters.condition_full }
		set_variable = { cbrn_mask_replacement_demand = constant:cbrn_math.zero }
		set_variable = { cbrn_civilian_mask_total_crates = constant:cbrn_math.zero }
		set_variable = { cbrn_civilian_mask_total_basic = constant:cbrn_math.zero }
		set_variable = { cbrn_civilian_mask_total_reconditioned = constant:cbrn_math.zero }
		set_variable = { cbrn_civilian_mask_total_improved = constant:cbrn_math.zero }
		set_variable = { cbrn_civilian_mask_total_advanced = constant:cbrn_math.zero }
		set_variable = { cbrn_civilian_mask_total_sealed = constant:cbrn_math.zero }
		every_owned_state = {
			limit = {
				is_controlled_by = PREV
				is_core_of = PREV
			}
			set_variable = { cbrn_civilian_mask_basic_crates = constant:cbrn_math.zero }
			set_variable = { cbrn_civilian_mask_reconditioned_crates = constant:cbrn_math.zero }
			set_variable = { cbrn_civilian_mask_improved_crates = constant:cbrn_math.zero }
			set_variable = { cbrn_civilian_mask_advanced_crates = constant:cbrn_math.zero }
			set_variable = { cbrn_civilian_mask_sealed_crates = constant:cbrn_math.zero }
			set_variable = { cbrn_civilian_mask_fit_points = constant:cbrn_math.zero }
			set_variable = { cbrn_civilian_mask_filter_condition = constant:cbrn_mask_filters.condition_full }
			set_variable = { cbrn_civilian_mask_replacement_demand = constant:cbrn_math.zero }
			cbrn_refresh_state_civilian_mask_snapshot = yes
		}

		cbrn_refresh_country_mask_snapshot = yes
		set_temp_variable = { cbrn_historical_1936_issue_shortfall = cbrn_military_mask_requirement }
		multiply_temp_variable = {
			cbrn_historical_1936_issue_shortfall = cbrn_historical_1936_military_issue_target
		}
		subtract_from_temp_variable = { cbrn_historical_1936_issue_shortfall = cbrn_military_mask_crates }
		clamp_temp_variable = { var = cbrn_historical_1936_issue_shortfall min = constant:cbrn_math.zero }
		round_temp_variable = cbrn_historical_1936_issue_shortfall
		set_temp_variable = { cbrn_historical_1936_stock_grant = cbrn_historical_1936_issue_shortfall }
		add_to_temp_variable = {
			cbrn_historical_1936_stock_grant = cbrn_historical_1936_replacement_reserve
		}
		subtract_from_temp_variable = { cbrn_historical_1936_stock_grant = cbrn_mask_reserve_crates }
		clamp_temp_variable = { var = cbrn_historical_1936_stock_grant min = constant:cbrn_math.zero }
		round_temp_variable = cbrn_historical_1936_stock_grant
		if = {
			limit = { check_variable = { cbrn_historical_1936_stock_grant > constant:cbrn_math.zero } }
			add_equipment_to_stockpile = {
				type = gas_mask_equipment_1
				amount = cbrn_historical_1936_stock_grant
			}
		}
		set_temp_variable = { cbrn_mask_issue_requested_crates = cbrn_historical_1936_issue_shortfall }
		cbrn_issue_requested_masks_to_military = yes
		set_country_flag = cbrn_historical_1936_force_scaled_profile_applied
		cbrn_refresh_country_mask_snapshot = yes
	}
}
