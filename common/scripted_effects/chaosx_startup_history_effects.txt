# ============================================================================
# CHAOS REDUX - STARTUP HISTORY COMPATIBILITY EFFECTS
# ============================================================================
# Additive startup grants that replace copied vanilla history overrides.
# Existing-country technologies, stockpiles, generated scientists, commander traits,
# breakthrough progress, special projects, delayed events, and starting facility
# placement belong here unless they define a new Chaos Redux country package.
# ============================================================================

chaosx_apply_startup_history_grants = {
	if = {
		limit = {
			NOT = { has_global_flag = chaosx_startup_history_grants_applied }
		}
		set_global_flag = chaosx_startup_history_grants_applied

		chaosx_startup_grant_ast = yes
		chaosx_startup_grant_bel = yes
		chaosx_startup_grant_bul = yes
		chaosx_startup_grant_can = yes
		chaosx_startup_grant_cze = yes

		chaosx_startup_grant_tur = yes
		chaosx_startup_grant_usa = yes
		chaosx_startup_grant_yug = yes
		chaosx_apply_starting_cbrn_mask_profiles = yes
		# One-time achievement eligibility receipts are written after the
		# accepted starting profiles. They prevent country or profile switching
		# from manufacturing a starting condition later in the campaign.
		every_country = {
			limit = { is_ai = no }
			set_country_flag = cbrn_achievement_start_country_eligible
			if = {
				limit = { is_major = yes }
				set_country_flag = cbrn_achievement_starting_major_power
			}
			if = {
				limit = {
					OR = {
						cbrn_profile_is_mass_civil_defence = yes
						cbrn_profile_is_civil_defence_network = yes
					}
				}
				set_country_flag = cbrn_achievement_starting_civil_defence_profile
			}
		}
		cbrn_initialize_battlefield_operation_selection = yes
		cbrn_occupation_initialize_action_conditions = yes
	}
}

chaosx_startup_sync_chemical_technology_unlocks = {
	chem_sync_chemical_tactic_preferred_weight_ideas = yes

	if = {
		limit = {
			has_tech = basic_gas_masks
			NOT = { has_tactic = tactic_gas_mask_defense }
		}
		unlock_tactic = tactic_gas_mask_defense
	}

	if = {
		limit = {
			NOT = { has_tactic = tactic_chemical_shelling }
			OR = {
				has_tech = chlorine_gas
				has_tech = phosgene
				has_tech = mustard_gas
				has_tech = lewisite
				has_tech = sarin
				has_tech = soman
			}
		}
		unlock_tactic = tactic_chemical_shelling
	}
}

}

		clr_character_flag = chaosx_startup_scientist_preexisting
		clr_character_flag = chaosx_startup_scientist_named
	}
}

chaosx_startup_grant_ast = {
	AST = {
		if = {
			limit = { exists = yes }
			set_technology = {
				basic_gas_masks = 1
				portable_anemometer = 1
				popup = no
			}
			chaosx_startup_sync_chemical_technology_unlocks = yes
			chaosx_startup_mark_existing_scientists = yes
			generate_scientist_character = {
				portrait = GFX_portrait_AST_howard_florey
				gender = male
				traits = { scientist_trait_fast_learner }
				skills = {
					specialization_biowarfare = 3
				}
			}
			random_scientist = {
				limit = {
					NOT = { has_character_flag = chaosx_startup_scientist_preexisting }
					NOT = { has_character_flag = chaosx_startup_scientist_named }
				}
				set_character_name = AST_howard_florey
				set_portraits = {
					army = {
						large = "GFX_portrait_AST_howard_florey"
					}
				}
				set_character_flag = chaosx_startup_scientist_named
				set_character_flag = chaosx_scientist_ast_howard_florey
			}
			chaosx_startup_clear_generated_scientist_helper_flags = yes
			add_breakthrough_progress = {
				specialization = specialization_biowarfare
				value = constant:startup_history_breakthrough.australia_biowarfare
			}
			set_variable = { var = AST_citizen_army_spirit_army_non_core_attack_factor value = constant:startup_history_australia.citizen_army_non_core_attack_factor }
			set_variable = { var = AST_citizen_army_spirit_army_non_core_defence_factor value = constant:startup_history_australia.citizen_army_non_core_defence_factor }
