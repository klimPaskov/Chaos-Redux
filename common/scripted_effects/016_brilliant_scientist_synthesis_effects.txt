# COUNTRY scope. Public qualification keeps the flight tables with the host's
# institution and never writes a portable Kruger result.
brilliant_scientist_resolve_high_speed_materials_trial_national_board = {
	if = {
		limit = { brilliant_scientist_high_speed_materials_trial_resolution_is_valid = yes }
		set_country_flag = brilliant_scientist_high_speed_materials_trial_completed
		set_country_flag = brilliant_scientist_high_speed_materials_national_board
		clr_country_flag = brilliant_scientist_high_speed_materials_kruger_tables
		remove_dynamic_modifier = { modifier = brilliant_scientist_high_speed_kruger_envelope }
		add_dynamic_modifier = { modifier = brilliant_scientist_high_speed_qualification_board }
		KRG_warren_kruger = {
			set_character_flag = brilliant_scientist_personal_high_speed_materials_trial_completed
		}
		set_temp_variable = { brilliant_scientist_value_delta = constant:brilliant_scientist_high_speed_materials_trial.national_mandate }
		brilliant_scientist_change_mandate = yes
		set_temp_variable = { brilliant_scientist_value_delta = constant:brilliant_scientist_high_speed_materials_trial.national_dependence }
		brilliant_scientist_change_dependence = yes
		set_temp_variable = { brilliant_scientist_value_delta = constant:brilliant_scientist_high_speed_materials_trial.national_exposure }
		brilliant_scientist_change_exposure = yes
		set_temp_variable = { brilliant_scientist_value_delta = constant:brilliant_scientist_high_speed_materials_trial.national_capacity }
		brilliant_scientist_change_project_capacity = yes
		set_temp_variable = { brilliant_scientist_value_delta = constant:brilliant_scientist_high_speed_materials_trial.national_independent_capacity }
		brilliant_scientist_change_independent_capacity = yes
		set_temp_variable = { brilliant_scientist_value_delta = constant:brilliant_scientist_high_speed_materials_trial.national_grievance }
		brilliant_scientist_change_grievance = yes
		if = {
			limit = { has_event_target = brilliant_scientist_high_speed_test_corridor }
			event_target:brilliant_scientist_high_speed_test_corridor = {
				clr_state_flag = brilliant_scientist_high_speed_test_corridor_active
				set_state_flag = brilliant_scientist_high_speed_test_corridor_certified
			}
			clear_global_event_target = brilliant_scientist_high_speed_test_corridor
		}
		clr_country_flag = brilliant_scientist_high_speed_materials_trial_pending
		clr_country_flag = brilliant_scientist_high_speed_materials_trial_in_progress
	}
	else = {
		brilliant_scientist_fail_high_speed_materials_trial = yes
	}
}
