# A qualifying action must set chaosx_032_achievement_required_evolution_enabled
# to one before this effect. Missing input fails closed and disqualifies the
# named achievement, preventing disabled evolutions from being retroactively
# credited. Callers must invoke the matching wrapper for the action.
chaosx_032_achievement_latch_required_evolution = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = chaosx_032_achievement_required_evolution_enabled
					value = constant:missiles_program_value.one
					compare = equals
				}
			}
		}
		set_country_flag = chaosx_032_achievement_required_evolution_disabled
	}
}

chaosx_032_achievement_record_precision_operation = {
	# Input: profile, target_profile, payload, outcome, target_state_id,
	# target_country_id, target_is_enemy, civilian_deaths, and
	# neutral_or_self are required chaosx_032_input_* variables.
	if = {
		limit = {
			has_variable = chaosx_032_input_target_state_id
			has_variable = chaosx_032_input_target_country_id
			has_variable = chaosx_032_input_target_is_enemy
			check_variable = { var = chaosx_032_input_target_is_enemy value = constant:missiles_program_value.one compare = equals }
			check_variable = { var = chaosx_032_input_profile value = constant:missiles_strike_profile.precision compare = equals }
			check_variable = { var = chaosx_032_input_payload value = constant:missiles_payload.conventional compare = equals }
			OR = {
				check_variable = { var = chaosx_032_input_outcome value = constant:missiles_guidance_outcome.on_target compare = equals }
				check_variable = { var = chaosx_032_input_outcome value = constant:missiles_guidance_outcome.degraded compare = equals }
			}
			NOT = { check_variable = { var = chaosx_032_input_target_profile value = constant:missiles_target_profile.command compare = equals } }
			NOT = { check_variable = { var = chaosx_032_input_target_profile value = constant:missiles_target_profile.broad_strategic compare = equals } }
			NOT = { check_variable = { var = chaosx_032_input_capital_centered value = constant:missiles_program_value.one compare = equals } }
		}
		if = {
			limit = { NOT = { has_variable = chaosx_032_exact_distance_precision_strike_count } }
			set_variable = { chaosx_032_exact_distance_precision_strike_count = constant:missiles_program_value.zero }
		}
		add_to_variable = { chaosx_032_exact_distance_precision_strike_count = constant:missiles_program_value.one }
		if = {
			limit = { NOT = { is_in_array = { array = chaosx_032_exact_distance_states value = chaosx_032_input_target_state_id } } }
			add_to_array = { array = chaosx_032_exact_distance_states value = chaosx_032_input_target_state_id }
			add_to_variable = { chaosx_032_exact_distance_state_count = constant:missiles_program_value.one }
		}
		if = {
			limit = { NOT = { is_in_array = { array = chaosx_032_exact_distance_countries value = chaosx_032_input_target_country_id } } }
			add_to_array = { array = chaosx_032_exact_distance_countries value = chaosx_032_input_target_country_id }
			add_to_variable = { chaosx_032_exact_distance_country_count = constant:missiles_program_value.one }
		}
		set_country_flag = chaosx_032_exact_distance_precision_only
		set_country_flag = chaosx_032_exact_distance_valid_receipts
		set_country_flag = chaosx_032_exact_distance_no_special_payload
		if = {
			limit = { has_variable = chaosx_032_input_civilian_deaths }
			add_to_variable = { chaosx_032_exact_distance_civilian_deaths = chaosx_032_input_civilian_deaths }
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = chaosx_032_input_neutral_or_self value = constant:missiles_program_value.one compare = equals }
					check_variable = { var = chaosx_032_input_outcome value = constant:missiles_guidance_outcome.neutral_accident compare = equals }
					check_variable = { var = chaosx_032_input_outcome value = constant:missiles_guidance_outcome.self_strike compare = equals }
				}
			}
			set_country_flag = chaosx_032_exact_distance_disqualified_strike
		}
		if = {
			limit = { NOT = { has_country_flag = chaosx_032_exact_distance_disqualified_strike } }
			set_country_flag = chaosx_032_exact_distance_no_neutral_or_self_strike
		}
	}
	else = {
		set_country_flag = chaosx_032_exact_distance_disqualified_strike
	}
	chaosx_032_achievement_finalize_exact_distance = yes
}

		limit = { check_variable = { var = chaosx_032_input_scuttled value = constant:missiles_program_value.one compare = equals } }
		set_country_flag = chaosx_032_still_on_the_line_scuttled
	}

		limit = { check_variable = { var = chaosx_032_input_site_setup_invalid value = constant:missiles_program_value.one compare = equals } }
		set_country_flag = chaosx_032_still_on_the_line_lost_all_capacity

		set_country_flag = chaosx_032_still_on_the_line_two_sites_retained

		limit = { check_variable = { var = chaosx_032_input_retained_operational_site_count value = constant:missiles_program_value.zero compare = equals } }

chaosx_032_achievement_record_retaliation_closure = {
	# Input: incident_root_id, warning_resolved, incident_closed, and
	# later_launch. A closed incident starts the independent 90-day clock.
	if = {
		limit = {
			check_variable = { var = chaosx_032_input_incident_root_id value = chaosx_032_break_the_chain_incident_root_id compare = equals }
			check_variable = { var = chaosx_032_input_warning_resolved value = constant:missiles_program_value.one compare = equals }
			check_variable = { var = chaosx_032_input_incident_closed value = constant:missiles_program_value.one compare = equals }
		}
		set_country_flag = chaosx_032_break_the_chain_warning_resolved
		set_country_flag = chaosx_032_break_the_chain_incident_closed
		set_variable = { chaosx_032_break_the_chain_survival_deadline = global.date }
		add_to_variable = { chaosx_032_break_the_chain_survival_deadline = @missiles_achievement_survival_days }
		if = {
			limit = { NOT = { has_country_flag = chaosx_032_break_the_chain_survival_timer_queued } }
			set_country_flag = chaosx_032_break_the_chain_survival_timer_queued
			country_event = { id = chaosx.nr32.90 days = @missiles_achievement_survival_days }
		}
		if = {
			limit = { check_variable = { var = chaosx_032_input_later_launch value = constant:missiles_program_value.one compare = equals } }
			set_country_flag = chaosx_032_break_the_chain_later_launch
		}
	}
	chaosx_032_achievement_finalize_break_the_chain = yes
}

				if = { limit = { has_country_flag = chaosx_032_empty_silos_enemy_rebuilt } ROOT = { set_variable = { chaosx_032_input_enemy_rebuilt = constant:missiles_program_value.one } } }
				remove_from_array = { array = chaosx_032_empty_silos_tracking_attackers value = ROOT }
				if = {
					limit = { has_variable = chaosx_032_empty_silos_tracking_count check_variable = { var = chaosx_032_empty_silos_tracking_count value = constant:missiles_program_value.zero compare = greater_than } }
					subtract_from_variable = { chaosx_032_empty_silos_tracking_count = constant:missiles_program_value.one }
				}
				if = {
					limit = { has_variable = chaosx_032_empty_silos_tracking_count check_variable = { var = chaosx_032_empty_silos_tracking_count value = constant:missiles_program_value.zero compare = less_than_or_equals } }
					clear_array = chaosx_032_empty_silos_tracking_attackers
					clear_variable = chaosx_032_empty_silos_tracking_count
					clr_country_flag = chaosx_032_empty_silos_tracked_enemy
				}
			}
			clear_variable = chaosx_032_empty_the_silos_target_country
		}
		if = { limit = { has_variable = chaosx_032_empty_the_silos_war_id } chaosx_032_achievement_record_empty_silos_war_resolution = yes }
		if = { limit = { has_variable = chaosx_032_sky_full_of_steel_war_id } chaosx_032_achievement_record_saturation_war_resolution = yes }
		chaosx_032_achievement_prepare_long_reach_snapshot = yes
		set_variable = { missiles_achievement_last_war_id = missiles_achievement_current_war_id }
		set_variable = { missiles_achievement_current_war_id = constant:missiles_program_value.zero }
		set_variable = { missiles_achievement_war_won_pending = constant:missiles_program_value.zero }
		clr_country_flag = chaosx_032_achievement_at_war_with_major
	}
}

chaosx_032_achievement_record_rogue_resolution = {
	# The caller is the affected country and invokes this while the incident
	# site pointer still exists. This receipt is written once at the actual
	# resolution boundary, not when a crisis window merely opens.
	set_variable = { chaosx_032_input_reserve_or_prepared = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_recovery_complete = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_unauthorized_launch = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_scuttled = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_foreign_seizure = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_destroyed = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_secure_restored = constant:missiles_program_value.zero }
	set_variable = { chaosx_032_input_control = missiles_command_control }
	if = { limit = { has_country_flag = chaosx_032_keys_returned_initial_proof } set_variable = { chaosx_032_input_reserve_or_prepared = constant:missiles_program_value.one } }
	if = {
		limit = {
			OR = {
				has_country_flag = missiles_site_recovery_complete
				has_country_flag = missiles_command_negotiation_complete
			}
		}
		set_variable = { chaosx_032_input_recovery_complete = constant:missiles_program_value.one }
	}
	if = { limit = { has_country_flag = missiles_rogue_launch_committed } set_variable = { chaosx_032_input_unauthorized_launch = constant:missiles_program_value.one } }
	if = {
		limit = {
			OR = {
				check_variable = { var = missiles_rogue_incident_family value = constant:missiles_rogue_incident_family.regional_seizure compare = equals }
				check_variable = { var = missiles_rogue_incident_family value = constant:missiles_rogue_incident_family.captured_arsenal compare = equals }
			}
		}
		set_variable = { chaosx_032_input_foreign_seizure = constant:missiles_program_value.one }
	}
	if = {
		limit = {
			has_variable = missiles_incident_site
			var:missiles_incident_site = {
				OR = {
					has_state_flag = missiles_site_scuttled
					has_state_flag = missiles_site_damaged_beyond_use
				}
			}
		}
		set_variable = { chaosx_032_input_scuttled = constant:missiles_program_value.one }
		set_variable = { chaosx_032_input_destroyed = constant:missiles_program_value.one }
	}
	if = {
		limit = {
			has_variable = missiles_incident_site
			var:missiles_incident_site = { has_state_flag = missiles_site_captured }
		}
		set_variable = { chaosx_032_input_foreign_seizure = constant:missiles_program_value.one }
	}
	if = {
		limit = {
			has_variable = missiles_incident_site
			var:missiles_incident_site = { missiles_site_is_secure = yes }
		}
		set_variable = { chaosx_032_input_secure_restored = constant:missiles_program_value.one }
	}
	if = {
		limit = {
			OR = {
				check_variable = { var = chaosx_032_input_destroyed value = constant:missiles_program_value.one compare = equals }
				check_variable = { var = chaosx_032_input_foreign_seizure value = constant:missiles_program_value.one compare = equals }
			}
		}
		chaosx_032_achievement_record_long_reach_rogue_site_loss = yes
	}
	chaosx_032_achievement_record_rogue_recovery = yes
}
