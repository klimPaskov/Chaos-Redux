# Supported scope: COUNTRY. FROM must be the source STATE. The reception array
# contains exactly one state because designation clears it before insertion.
air_winter_response_complete_controlled_evacuation = {
	FROM = { save_event_target_as = air_winter_response_state }
	event_target:air_winter_response_state = {
		air_winter_normalize_state_ledger = yes
		set_temp_variable = { air_winter_response_population_transfer = state_population }
		multiply_temp_variable = { air_winter_response_population_transfer = constant:air_winter_response_result.controlled_evacuation_population_share }
		round_temp_variable = air_winter_response_population_transfer
		set_temp_variable = { air_winter_response_population_loss = air_winter_response_population_transfer }
		multiply_temp_variable = { air_winter_response_population_loss = constant:air_winter_runtime.negative_one }
		add_manpower = air_winter_response_population_loss
		subtract_from_variable = { air_winter_refugee_pressure = constant:air_winter_response_result.controlled_evacuation_source_pressure_reduction }
		set_state_flag = air_winter_controlled_evacuation_completed
		set_temp_variable = { air_winter_response_project_cooldown = constant:air_winter_response_timing.severe_state_cooldown_months }
		air_winter_response_finish_state_project = yes
	}
	for_each_scope_loop = {
		array = air_winter_reception_states
		air_winter_normalize_state_ledger = yes
		add_manpower = air_winter_response_population_transfer
		add_to_variable = { air_winter_refugee_pressure = constant:air_winter_response_result.controlled_evacuation_receiver_pressure }
		air_winter_normalize_state_ledger = yes
		air_winter_calculate_survival_value = yes
	}
}
