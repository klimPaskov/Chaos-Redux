# Supported scope: COUNTRY. Clears the current population-scaled evacuation
# quote. A quote is rebuilt only for the country's selected priority state and
# its lowest-id valid reception state.
air_winter_response_clear_evacuation_cache = {
	clr_country_flag = air_winter_evacuation_cache_valid
	clear_variable = air_winter_evacuation_cache_source_state
	clear_variable = air_winter_evacuation_cache_receiver_state
	clear_variable = air_winter_evacuation_cache_source_population_k
	clear_variable = air_winter_evacuation_cache_receiver_population_k
	clear_variable = air_winter_evacuation_cache_receiver_pressure
	clear_variable = air_winter_evacuation_cache_cycle_id
	clear_variable = air_winter_controlled_evacuation_population_transfer_k
	clear_variable = air_winter_controlled_evacuation_population_transfer_people
	clear_variable = air_winter_controlled_evacuation_train_cost
	clear_variable = air_winter_controlled_evacuation_convoy_cost
	clear_variable = air_winter_controlled_evacuation_manpower_cost
	clear_variable = air_winter_controlled_evacuation_stability_cost
	clear_variable = air_winter_controlled_evacuation_receiver_pressure_gain
	clear_variable = air_winter_controlled_evacuation_receiver_pressure_after
	clear_variable = air_winter_controlled_evacuation_source_pressure_relief
	clear_variable = air_winter_final_evacuation_population_transfer_k
	clear_variable = air_winter_final_evacuation_population_transfer_people
	clear_variable = air_winter_final_evacuation_train_cost
	clear_variable = air_winter_final_evacuation_convoy_cost
	clear_variable = air_winter_final_evacuation_support_cost
	clear_variable = air_winter_final_evacuation_manpower_cost
	clear_variable = air_winter_final_evacuation_receiver_pressure_gain
	clear_variable = air_winter_final_evacuation_receiver_pressure_after
	clear_variable = air_winter_final_evacuation_source_pressure_relief
}

# Supported scope: STATE. Rebuilds one country quote from the selected source
# population and the designated receiver's current population and pressure.
# Temporary variables deliberately carry arithmetic across the local scopes.
air_winter_response_refresh_state_evacuation_cache = {
	save_event_target_as = air_winter_evacuation_cache_source
	set_temp_variable = { air_winter_evacuation_source_state_id = THIS.id }
	OWNER = {
		save_event_target_as = air_winter_evacuation_cache_owner
		air_winter_response_clear_evacuation_cache = yes
	}

	if = {
		limit = {
			air_winter_state_is_valid = yes
			has_state_flag = air_winter_response_priority_target
			has_variable = air_winter_response_priority_owner
			check_variable = {
				var = air_winter_response_priority_owner
				value = event_target:air_winter_evacuation_cache_owner
				compare = equals
			}
			is_owned_by = event_target:air_winter_evacuation_cache_owner
			is_controlled_by = event_target:air_winter_evacuation_cache_owner
			check_variable = { state_population_k > constant:air_winter_runtime.zero }
			event_target:air_winter_evacuation_cache_owner = {
				has_variable = air_winter_response_priority_state
				check_variable = {
					var = air_winter_response_priority_state
					value = air_winter_evacuation_source_state_id
					compare = equals
				}
			}
		}

		set_temp_variable = {
			air_winter_evacuation_selected_receiver_state = constant:air_winter_response_evacuation.state_id_selection_sentinel
		}
		event_target:air_winter_evacuation_cache_owner = {
			for_each_scope_loop = {
				array = air_winter_reception_states
				if = {
					limit = {
						has_state_flag = air_winter_response_reception_target
						has_variable = air_winter_response_reception_owner
						check_variable = {
							var = air_winter_response_reception_owner
							value = event_target:air_winter_evacuation_cache_owner
							compare = equals
						}
						is_owned_by = event_target:air_winter_evacuation_cache_owner
						is_controlled_by = event_target:air_winter_evacuation_cache_owner
						air_winter_response_is_reception_state = yes
						NOT = { state = event_target:air_winter_evacuation_cache_source }
					}
					set_temp_variable = { air_winter_evacuation_receiver_candidate_state = THIS.id }
					if = {
						limit = {
							check_variable = {
								var = air_winter_evacuation_receiver_candidate_state
								value = air_winter_evacuation_selected_receiver_state
								compare = less_than
							}
						}
						set_temp_variable = {
							air_winter_evacuation_selected_receiver_state = air_winter_evacuation_receiver_candidate_state
						}
					}
				}
			}
		}

		if = {
			limit = {
				check_variable = {
					var = air_winter_evacuation_selected_receiver_state
					value = constant:air_winter_response_evacuation.state_id_selection_sentinel
					compare = less_than
				}
			}

			set_temp_variable = { air_winter_evacuation_source_population_k = state_population_k }

			set_temp_variable = { air_winter_controlled_evacuation_transfer_k = air_winter_evacuation_source_population_k }
			multiply_temp_variable = {
				air_winter_controlled_evacuation_transfer_k = constant:air_winter_response_result.controlled_evacuation_population_share
			}
			set_temp_variable = { air_winter_controlled_evacuation_transfer_people = air_winter_controlled_evacuation_transfer_k }
			multiply_temp_variable = {
				air_winter_controlled_evacuation_transfer_people = constant:air_winter_response_evacuation.population_people_per_k
			}
			round_temp_variable = air_winter_controlled_evacuation_transfer_people

			set_temp_variable = { air_winter_final_evacuation_transfer_k = air_winter_evacuation_source_population_k }
			multiply_temp_variable = {
				air_winter_final_evacuation_transfer_k = constant:air_winter_response_result.final_evacuation_population_share
			}
			set_temp_variable = { air_winter_final_evacuation_transfer_people = air_winter_final_evacuation_transfer_k }
			multiply_temp_variable = {
				air_winter_final_evacuation_transfer_people = constant:air_winter_response_evacuation.population_people_per_k
			}
			round_temp_variable = air_winter_final_evacuation_transfer_people

			set_temp_variable = { air_winter_controlled_evacuation_trains = air_winter_controlled_evacuation_transfer_k }
			divide_temp_variable = {
				air_winter_controlled_evacuation_trains = constant:air_winter_response_evacuation.population_k_per_train
			}
			add_to_temp_variable = {
				air_winter_controlled_evacuation_trains = constant:air_winter_response_evacuation.ceil_rounding_offset
			}
			round_temp_variable = air_winter_controlled_evacuation_trains
			if = {
				limit = { check_variable = { air_winter_controlled_evacuation_trains < constant:air_winter_runtime.one } }
				set_temp_variable = { air_winter_controlled_evacuation_trains = constant:air_winter_runtime.one }
			}

			set_temp_variable = { air_winter_controlled_evacuation_stability = air_winter_controlled_evacuation_transfer_k }
			divide_temp_variable = {
				air_winter_controlled_evacuation_stability = constant:air_winter_response_evacuation.controlled_stability_population_k_per_percent
			}
			multiply_temp_variable = {
				air_winter_controlled_evacuation_stability = constant:air_winter_response_evacuation.controlled_stability_percent_unit
			}
			clamp_temp_variable = {
				var = air_winter_controlled_evacuation_stability
				min = constant:air_winter_response_evacuation.controlled_stability_minimum
				max = constant:air_winter_response_evacuation.controlled_stability_maximum
			}

			set_temp_variable = { air_winter_controlled_evacuation_convoys = air_winter_controlled_evacuation_transfer_k }
			divide_temp_variable = {
				air_winter_controlled_evacuation_convoys = constant:air_winter_response_evacuation.population_k_per_convoy
			}
			add_to_temp_variable = {
				air_winter_controlled_evacuation_convoys = constant:air_winter_response_evacuation.ceil_rounding_offset
			}
			round_temp_variable = air_winter_controlled_evacuation_convoys
			if = {
				limit = { check_variable = { air_winter_controlled_evacuation_convoys < constant:air_winter_runtime.one } }
				set_temp_variable = { air_winter_controlled_evacuation_convoys = constant:air_winter_runtime.one }
			}

			set_temp_variable = { air_winter_controlled_evacuation_staff = air_winter_controlled_evacuation_transfer_k }
			multiply_temp_variable = {
				air_winter_controlled_evacuation_staff = constant:air_winter_response_evacuation.administrative_staff_per_population_k
			}
			round_temp_variable = air_winter_controlled_evacuation_staff
			if = {
				limit = {
					check_variable = {
						var = air_winter_controlled_evacuation_staff
						value = constant:air_winter_response_evacuation.minimum_administrative_staff
						compare = less_than
					}
				}
				set_temp_variable = {
					air_winter_controlled_evacuation_staff = constant:air_winter_response_evacuation.minimum_administrative_staff
				}
			}

			set_temp_variable = { air_winter_final_evacuation_trains = air_winter_final_evacuation_transfer_k }
			divide_temp_variable = {
				air_winter_final_evacuation_trains = constant:air_winter_response_evacuation.population_k_per_train
			}
			add_to_temp_variable = {
				air_winter_final_evacuation_trains = constant:air_winter_response_evacuation.ceil_rounding_offset
			}
			round_temp_variable = air_winter_final_evacuation_trains
			if = {
				limit = { check_variable = { air_winter_final_evacuation_trains < constant:air_winter_runtime.one } }
				set_temp_variable = { air_winter_final_evacuation_trains = constant:air_winter_runtime.one }
			}

			set_temp_variable = { air_winter_final_evacuation_convoys = air_winter_final_evacuation_transfer_k }
			divide_temp_variable = {
				air_winter_final_evacuation_convoys = constant:air_winter_response_evacuation.population_k_per_convoy
			}
			add_to_temp_variable = {
				air_winter_final_evacuation_convoys = constant:air_winter_response_evacuation.ceil_rounding_offset
			}
			round_temp_variable = air_winter_final_evacuation_convoys
			if = {
				limit = { check_variable = { air_winter_final_evacuation_convoys < constant:air_winter_runtime.one } }
				set_temp_variable = { air_winter_final_evacuation_convoys = constant:air_winter_runtime.one }
			}

			set_temp_variable = { air_winter_final_evacuation_support = air_winter_final_evacuation_transfer_k }
			divide_temp_variable = {
				air_winter_final_evacuation_support = constant:air_winter_response_evacuation.population_k_per_support_equipment
			}
			add_to_temp_variable = {
				air_winter_final_evacuation_support = constant:air_winter_response_evacuation.ceil_rounding_offset
			}
			round_temp_variable = air_winter_final_evacuation_support
			if = {

# Supported scope: COUNTRY. FROM must be the source STATE. The project snapshot
# preserves the paid population transfer and receiver even if the monthly quote
# changes while the timed decision is active.
air_winter_response_complete_controlled_evacuation = {
	if = {
		limit = { air_winter_response_controlled_evacuation_project_is_valid = yes }
		FROM = {
			save_event_target_as = air_winter_response_state
			set_temp_variable = { air_winter_response_population_transfer_people = air_winter_evacuation_project_population_transfer_people }
			set_temp_variable = { air_winter_response_receiver_pressure_gain = air_winter_evacuation_project_receiver_pressure_gain }
			air_winter_normalize_state_ledger = yes
			set_temp_variable = { air_winter_response_population_loss = air_winter_response_population_transfer_people }
			multiply_temp_variable = { air_winter_response_population_loss = constant:air_winter_runtime.negative_one }
			add_manpower = air_winter_response_population_loss
			subtract_from_variable = { air_winter_refugee_pressure = air_winter_evacuation_project_source_pressure_relief }
			set_state_flag = air_winter_controlled_evacuation_completed
			air_winter_normalize_state_ledger = yes
			air_winter_calculate_survival_value = yes
			var:air_winter_evacuation_project_receiver_state = {
				air_winter_normalize_state_ledger = yes
				add_manpower = air_winter_response_population_transfer_people
				add_to_variable = { air_winter_refugee_pressure = air_winter_response_receiver_pressure_gain }
				air_winter_normalize_state_ledger = yes
				air_winter_calculate_survival_value = yes
			}
			# The offline effect reference records that negative state add_manpower
			# credits the removed population to recruitable manpower. Remove that
			# engine credit at country scope after the balanced state transfer.
			ROOT = { add_manpower = air_winter_response_population_loss }
			set_temp_variable = { air_winter_response_project_cooldown = constant:air_winter_response_timing.severe_state_cooldown_months }
			air_winter_response_finish_state_project = yes
		}
	}
	else = { FROM = { air_winter_response_cancel_state_project = yes } }
}
