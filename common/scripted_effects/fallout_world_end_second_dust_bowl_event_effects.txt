fallout_event_656_apply_failure_deaths = {
	var:fallout_event_656_target_state_id = {
		if = {
			limit = { check_variable = { state_population_k > constant:fallout_event_runtime.zero } }
			set_temp_variable = { fallout_event_656_state_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_656_state_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_656_state_death_request = constant:fallout_event_656_threshold.result_failure_death_percent }
			round_temp_variable = fallout_event_656_state_death_request
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_requested = fallout_event_656_state_death_request }
			set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:fallout_event_656_threshold.minimum_remaining_people }
			set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.fallout_aftermath }
			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }
			apply_exact_state_civilian_population_loss = yes
			clear_temp_variable = fallout_event_656_state_death_request
		}
	}
}

	set_variable = { fallout_event_656_target_state_id = fallout_event_dispatch_issued_target }
	fallout_event_cancel_issued_ordinary_receipt = yes
	if = {
		limit = { check_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_656_history_stage = constant:fallout_event_656_log.cancelled_stage }
		fallout_event_656_record_history = yes
		clear_variable = fallout_event_656_target_state_id

	clear_temp_variable = fallout_event_requested_event_token

fallout_event_656_record_history = {
	set_temp_variable = { fallout_event_656_history_should_record = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.zero }
			NOT = { has_country_flag = fallout_event_656_choice_history_recorded }
		}
		set_temp_variable = { fallout_event_656_history_should_record = constant:fallout_event_runtime.one }
	}
	else_if = {
		limit = {
			check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.one }
			NOT = { has_country_flag = fallout_event_656_result_history_recorded }
		}
		set_temp_variable = { fallout_event_656_history_should_record = constant:fallout_event_runtime.one }
	}
	else_if = {
		limit = {
			check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.two }
			NOT = { has_country_flag = fallout_event_656_callback_history_recorded }
		}
		set_temp_variable = { fallout_event_656_history_should_record = constant:fallout_event_runtime.one }
	}
	else_if = {
		limit = {
			check_variable = { fallout_event_656_history_stage = constant:fallout_event_656_log.cancelled_stage }
			NOT = { has_country_flag = fallout_event_656_cancellation_history_recorded }
		}
		set_temp_variable = { fallout_event_656_history_should_record = constant:fallout_event_runtime.one }
	}
	if = {
		limit = { check_variable = { fallout_event_656_history_should_record = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.callback_failure }
		if = {
			limit = { check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.zero } }
			if = { limit = { check_variable = { fallout_event_656_selected_branch = constant:fallout_event_656_branch.shelter_fields } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.choice_shelter } }
			else_if = { limit = { check_variable = { fallout_event_656_selected_branch = constant:fallout_event_656_branch.move_farms } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.choice_move } }
			else_if = { limit = { check_variable = { fallout_event_656_selected_branch = constant:fallout_event_656_branch.seed_cold_crops } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.choice_seed } }
			else = { set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.choice_abandon } }
		}
		else_if = {
			limit = { check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.one } }
			if = {
				limit = { check_variable = { fallout_event_656_selected_branch = constant:fallout_event_656_branch.shelter_fields } }
				if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.shelter_success } }
				else_if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.shelter_partial } }
				else = { set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.shelter_failure } }
			}
			else_if = {
				limit = { check_variable = { fallout_event_656_selected_branch = constant:fallout_event_656_branch.move_farms } }
				if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.move_success } }
				else_if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.move_partial } }
				else = { set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.move_failure } }
			}
			else_if = {
				limit = { check_variable = { fallout_event_656_selected_branch = constant:fallout_event_656_branch.seed_cold_crops } }
				if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.seed_success } }
				else_if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.seed_partial } }
				else = { set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.seed_failure } }
			}
			else = {
				if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.abandon_success } }
				else_if = { limit = { check_variable = { fallout_event_656_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.abandon_partial } }
				else = { set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.abandon_failure } }
			}
		}
		else_if = { limit = { check_variable = { fallout_event_656_history_stage = constant:fallout_event_656_log.cancelled_stage } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.cancelled } }
		else_if = { limit = { check_variable = { fallout_event_656_callback_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.callback_success } }
		else_if = { limit = { check_variable = { fallout_event_656_callback_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_656_history_payload = constant:fallout_event_656_log.callback_partial } }
		set_temp_variable = { events_log_system_event_id = constant:fallout_event_656_log.history_id }
		set_temp_variable = { events_log_system_event_type = constant:event_system_event_type.fallout_country_memory }
		set_temp_variable = { events_log_system_payload = fallout_event_656_history_payload }
		set_temp_variable = { events_log_system_actor = ROOT.id }
		set_temp_variable = { events_log_system_has_actor = constant:fallout_event_runtime.one }
		set_temp_variable = { events_log_system_secondary_actor = fallout_event_656_target_state_id }
		set_temp_variable = { events_log_system_has_secondary_actor = constant:fallout_event_runtime.one }
		record_events_log_system_history_entry = yes
		refresh_events_log_system_history_views = yes
		if = { limit = { check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.zero } } set_country_flag = fallout_event_656_choice_history_recorded }
		else_if = { limit = { check_variable = { fallout_event_656_history_stage = constant:fallout_event_runtime.one } } set_country_flag = fallout_event_656_result_history_recorded }
		else_if = { limit = { check_variable = { fallout_event_656_history_stage = constant:fallout_event_656_log.cancelled_stage } } set_country_flag = fallout_event_656_cancellation_history_recorded }
		else = { set_country_flag = fallout_event_656_callback_history_recorded }
	}
	clear_temp_variable = fallout_event_656_history_should_record
	clear_temp_variable = fallout_event_656_history_payload
	clear_temp_variable = events_log_system_event_id
	clear_temp_variable = events_log_system_event_type
	clear_temp_variable = events_log_system_payload
	clear_temp_variable = events_log_system_actor
	clear_temp_variable = events_log_system_has_actor
	clear_temp_variable = events_log_system_secondary_actor
	clear_temp_variable = events_log_system_has_secondary_actor
	clear_temp_variable = fallout_event_656_history_stage
}

# generation-bound stale path for the Second Dust Bowl row. It cancels an
# unconsumed opening, refunds a branch that never settled, terminalizes any
