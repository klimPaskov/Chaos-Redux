# Input: temporary variable chaosx_test_country_registration_frontline_subunit
# containing token:<subunit_id>. The owning package's registered CXT setup
# effect should call this helper.
chaosx_test_country_register_frontline_subunit = {
	if = {
		limit = { has_variable = chaosx_test_country_registration_frontline_subunit }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.chaosx_test_country_registered_frontline_subunits
						value = chaosx_test_country_registration_frontline_subunit
					}
				}
			}
			add_to_array = {
				array = global.chaosx_test_country_registered_frontline_subunits
				value = chaosx_test_country_registration_frontline_subunit
			}
		}
	}
}

# Inputs: temporary variables chaosx_test_country_registration_support_subunit
# and chaosx_test_country_registration_support_anchor, each containing
# token:<subunit_id>. The owning package's registered CXT setup effect should
# call this helper.
# A support company needs a compatible line anchor for its generated CXT
# template; the caller owns that compatibility decision.
chaosx_test_country_register_support_subunit = {
	if = {
		limit = {
			has_variable = chaosx_test_country_registration_support_subunit
			has_variable = chaosx_test_country_registration_support_anchor
		}
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.chaosx_test_country_registered_support_subunits
						value = chaosx_test_country_registration_support_subunit
					}
				}
			}
			add_to_array = {
				array = global.chaosx_test_country_registered_support_subunits
				value = chaosx_test_country_registration_support_subunit
			}
			add_to_array = {
				array = global.chaosx_test_country_registered_support_anchors
				value = chaosx_test_country_registration_support_anchor
			}
		}
	}
}

chaosx_test_country_create_registered_unit_roster = {
	for_each_loop = {
		array = global.chaosx_test_country_registered_frontline_subunits
		value = chaosx_test_country_current_frontline_subunit
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = chaosx_test_country_processed_frontline_subunits
						value = chaosx_test_country_current_frontline_subunit
					}
				}
			}
			chaosx_test_country_create_registered_frontline_subunit = yes
			add_to_array = {
				array = chaosx_test_country_processed_frontline_subunits
				value = chaosx_test_country_current_frontline_subunit
			}
		}
	}
	for_each_loop = {
		array = global.chaosx_test_country_registered_support_subunits
		value = chaosx_test_country_current_support_subunit
		index = chaosx_test_country_current_support_index
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = chaosx_test_country_processed_support_subunits
						value = chaosx_test_country_current_support_subunit
					}
				}
			}
			set_temp_variable = {
				var = chaosx_test_country_current_support_anchor
