# Called by the shared government-route lock after the route variable and lock
# flag are written. This keeps the vanilla ICE roster while making each Event
# 006 route visibly change the emergency republic's government and route idea.
independence_wave_ice_apply_government_route_politics = {
	if = {
		limit = { is_independence_wave_ice_package = yes has_country_flag = independence_wave_government_route_locked }
		independence_wave_ice_remove_route_ideas = yes
		if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.constitutional_republic } }
			set_politics = { ruling_party = democratic elections_allowed = yes }
			set_popularities = {
				democratic = constant:independence_wave_ice_route_politics.constitutional_democratic
				communism = constant:independence_wave_ice_route_politics.constitutional_communism
				neutrality = constant:independence_wave_ice_route_politics.constitutional_neutrality
				fascism = constant:independence_wave_ice_route_politics.constitutional_fascism
			}
			add_ideas = ice_municipal_shipping_council
			promote_character = { character = ICE_sveinn_bjornsson ideology = centrism }
		}
		else_if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.traditional_restoration } }
			set_politics = { ruling_party = neutrality elections_allowed = no }
			set_popularities = {
				democratic = constant:independence_wave_ice_route_politics.traditional_democratic
				communism = constant:independence_wave_ice_route_politics.traditional_communism
				neutrality = constant:independence_wave_ice_route_politics.traditional_neutrality
				fascism = constant:independence_wave_ice_route_politics.traditional_fascism
			}
			add_ideas = ice_north_atlantic_service_compact
			promote_character = { character = ICE_sveinn_bjornsson ideology = despotism }
		}
		else_if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.emergency_military } }
			set_politics = { ruling_party = neutrality elections_allowed = no }
			set_popularities = {
				democratic = constant:independence_wave_ice_route_politics.emergency_democratic
				communism = constant:independence_wave_ice_route_politics.emergency_communism
				neutrality = constant:independence_wave_ice_route_politics.emergency_neutrality
				fascism = constant:independence_wave_ice_route_politics.emergency_fascism
			}
			add_ideas = ice_coastwatch_emergency_command
			promote_character = { character = ICE_bjorn_sveinsson_bjornsson ideology = despotism }
		}
		else_if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.patron_client } }
			set_politics = { ruling_party = neutrality elections_allowed = no }
			set_popularities = {
				democratic = constant:independence_wave_ice_route_politics.patron_democratic
				communism = constant:independence_wave_ice_route_politics.patron_communism
				neutrality = constant:independence_wave_ice_route_politics.patron_neutrality
				fascism = constant:independence_wave_ice_route_politics.patron_fascism
			}
			add_ideas = ice_north_atlantic_patron_mandate
			set_country_flag = independence_wave_ice_patron_mandate_active
			promote_character = { character = ICE_sveinn_bjornsson ideology = oligarchism }
		}
	}
}
