fallout_event_164_record_history = {

	if = {
		limit = {
			check_variable = { fallout_event_164_history_stage = constant:fallout_event_runtime.one }
			NOT = { has_country_flag = fallout_event_164_result_history_recorded }
		}
		set_temp_variable = { fallout_event_164_history_should_record = constant:fallout_event_runtime.one }
	}
	else_if = {
		limit = {
			check_variable = { fallout_event_164_history_stage = constant:fallout_event_runtime.two }
			NOT = { has_country_flag = fallout_event_164_callback_history_recorded }
		}
		set_temp_variable = { fallout_event_164_history_should_record = constant:fallout_event_runtime.one }
	}
	if = {
		limit = { check_variable = { fallout_event_164_history_should_record = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_164_history_payload = constant:fallout_event_164_log.callback_failure }

			limit = { check_variable = { fallout_event_164_history_stage = constant:fallout_event_runtime.one } }

				limit = { check_variable = { fallout_event_164_selected_branch = constant:fallout_event_164_branch.feed_people } }

				else = { set_temp_variable = { fallout_event_164_history_payload = constant:fallout_event_164_log.division_failure } }
			}
		}
		else_if = { limit = { check_variable = { fallout_event_164_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_164_history_payload = constant:fallout_event_164_log.callback_success } }

		set_temp_variable = { events_log_system_actor = ROOT.id }
		set_temp_variable = { events_log_system_has_actor = constant:fallout_event_runtime.one }
		set_temp_variable = { events_log_system_secondary_actor = constant:fallout_event_runtime.zero }
		set_temp_variable = { events_log_system_has_secondary_actor = constant:fallout_event_runtime.zero }

fallout_event_164_calculate_outcome = {
	set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.failure }
	var:fallout_event_dispatch_issued_target = {
		set_temp_variable = { fallout_event_164_target_feed = fallout_pretransition_air_winter_food_reserve }
		set_temp_variable = { fallout_event_164_target_adaptation = fallout_pretransition_air_winter_adaptation }
		set_temp_variable = { fallout_event_164_target_reclamation = fallout_pretransition_air_winter_reclamation }
	}
	if = {
		limit = { check_variable = { fallout_event_164_requested_branch = constant:fallout_event_164_branch.feed_people } }
		if = {
			limit = {
				check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_164_threshold.people_success_food compare = greater_than_or_equals }
				check_variable = { fallout_event_164_target_feed value = constant:fallout_event_164_threshold.people_partial_feed compare = greater_than_or_equals }
			}
			set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.success }
		}
		else_if = { limit = { check_variable = { fallout_event_164_target_feed value = constant:fallout_event_164_threshold.people_partial_feed compare = greater_than_or_equals } } set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	else_if = {
		limit = { check_variable = { fallout_event_164_requested_branch = constant:fallout_event_164_branch.preserve_breeding } }
		if = {
			limit = {
				check_variable = { fallout_event_164_target_feed value = constant:fallout_event_164_threshold.breeding_success_feed compare = greater_than_or_equals }
				check_variable = { fallout_event_164_target_adaptation value = constant:fallout_event_164_threshold.breeding_success_adaptation compare = greater_than_or_equals }
			}
			set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.success }
		}
		else_if = { limit = { check_variable = { fallout_event_164_target_feed value = constant:fallout_event_164_threshold.breeding_partial_feed compare = greater_than_or_equals } } set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	else = {
		if = {
			limit = {
				check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_164_threshold.division_success_food compare = greater_than_or_equals }
				check_variable = { fallout_event_164_target_reclamation value = constant:fallout_event_164_threshold.division_success_reclamation compare = greater_than_or_equals }
			}
			set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.success }
		}
		else_if = { limit = { check_variable = { fallout_event_164_target_feed value = constant:fallout_event_164_threshold.division_partial_feed compare = greater_than_or_equals } } set_temp_variable = { fallout_event_164_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
}
