fallout_event_803_pay_branch_cost = {
	if = {
		limit = { NOT = { has_country_flag = fallout_event_803_cost_paid } }
		if = {
			limit = { check_variable = { fallout_event_803_selected_branch value = constant:fallout_event_803_branch.hospital_priority compare = equals } }
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_803_cost.hospital_priority_food }
			subtract_from_variable = { fallout_survival_current_entries^3 = constant:fallout_event_803_cost.hospital_priority_medicine }
			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_803_cost.hospital_priority_recognition }
		}
		else_if = {
			limit = { check_variable = { fallout_event_803_selected_branch value = constant:fallout_event_803_branch.food_priority compare = equals } }
			subtract_from_variable = { fallout_survival_current_entries^5 = constant:fallout_event_803_cost.food_priority_fuel }
			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_803_cost.food_priority_recognition }
			set_temp_variable = { fallout_survival_cohesion_delta = constant:fallout_event_803_cost.food_priority_cohesion }
			multiply_temp_variable = { fallout_survival_cohesion_delta = -1 }
			fallout_survival_apply_cohesion_delta = yes
		}
		else_if = {
			limit = { check_variable = { fallout_event_803_selected_branch value = constant:fallout_event_803_branch.industry_priority compare = equals } }
			subtract_from_variable = { fallout_survival_current_entries^4 = constant:fallout_event_803_cost.industry_priority_scrap }
			subtract_from_variable = { fallout_survival_current_entries^6 = constant:fallout_event_803_cost.industry_priority_power }
			subtract_from_variable = { fallout_survival_current_entries^3 = constant:fallout_event_803_cost.industry_priority_medicine }
		}
		else = {
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_803_cost.defense_priority_food }
			subtract_from_variable = { fallout_survival_current_entries^5 = constant:fallout_event_803_cost.defense_priority_fuel }
			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_803_cost.defense_priority_recognition }
		}
		set_country_flag = fallout_event_803_cost_paid
		fallout_orientation_clamp_resources = yes
		set_temp_variable = { fallout_survival_cohesion_delta = constant:fallout_event_runtime.zero }
	}
}
