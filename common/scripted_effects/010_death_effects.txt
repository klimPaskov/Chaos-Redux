death_consume_current_state = {
	if = {
		limit = { NOT = { death_is_consumed_state = yes } }
		set_temp_variable = { death_last_consumed_population_k = state_population_k }
		set_variable = { death_consumed_population_k = death_last_consumed_population_k }
		set_variable = { death_consumed_date = global.date }
		death_record_last_ferry_consumption_credit = yes
		if = {
			limit = {
				NOT = { has_global_flag = settings_chaos_deaths_disabled }
				NOT = { has_global_flag = death_triggerable_scenario_started }
			}
			death_register_consumed_population_deaths = yes
		}
		else = {
			death_remove_current_state_population_without_ledger = yes
		}
		death_strip_current_state_buildings = yes
		death_wither_divisions_in_current_state = yes
		if = {
			limit = { NOT = { has_state_category = wasteland } }
			set_state_category = wasteland
		}
		if = {
			limit = {
				OR = {
					has_global_flag = death_publicly_revealed
					death_should_reveal_from_current_state = yes
				}
			}
			death_declare_war_on_current_state_owner = yes
		}
			transfer_state_to = DTH
			add_core_of = DTH
			set_state_controller_to = DTH
			death_apply_active_wasteland_state = yes
			add_to_variable = { global.death_consumed_states = 1 }
			add_to_variable = { global.death_consumed_population = death_last_consumed_population_k }
			death_refresh_consumed_population_display_variables = yes
			death_refresh_soul_power_display_variables = yes
			DTH = {
				death_setup_country = yes
				set_capital = {
					state = PREV
					remember_old_capital = no
				}
			}
			if = {
				limit = { is_island_state = yes }
				add_to_variable = { global.death_consumed_island_states = 1 }
			}
			if = {
				limit = { is_island_state = no }
				add_to_variable = { global.death_mainland_consumed_states = 1 }
				if = {
					limit = { is_on_continent = europe }
					add_to_variable = { global.death_mainland_consumed_states_europe = 1 }
				}
				if = {
					limit = { is_on_continent = middle_east }
					add_to_variable = { global.death_mainland_consumed_states_middle_east = 1 }
				}
				if = {
					limit = { is_on_continent = asia }
					add_to_variable = { global.death_mainland_consumed_states_asia = 1 }
				}
				if = {
					limit = { is_on_continent = africa }
					add_to_variable = { global.death_mainland_consumed_states_africa = 1 }
				}
				if = {
					limit = { is_on_continent = australia }
					add_to_variable = { global.death_mainland_consumed_states_australia = 1 }
				}
				if = {
					limit = { is_on_continent = north_america }
					add_to_variable = { global.death_mainland_consumed_states_north_america = 1 }
				}
				if = {
					limit = { is_on_continent = south_america }
					add_to_variable = { global.death_mainland_consumed_states_south_america = 1 }
				}
		}
		if = {
			limit = { death_should_reveal_from_current_state = yes }
			death_reveal_from_current_state = yes
		}
		if = {
			limit = { has_global_flag = death_publicly_revealed }
			DTH = {
				death_declare_war_on_neighbors = yes
			}
			death_refresh_world_threat_source = yes
		}
		death_try_start_world_end = yes
		death_try_fire_whole_world_consumed = yes
	}
	else = {
		if = {
			limit = { is_controlled_by = DTH }
			death_apply_active_wasteland_state = yes
			death_wither_divisions_in_current_state = yes
		}
	}
}

	clr_country_flag = death_pulse_scheduled
	if = {
		limit = { has_global_flag = death_defeated }
		death_refresh_world_threat_source = yes
	}
	else = {
		death_check_defeat = yes
		if = {
			limit = { NOT = { has_global_flag = death_defeated } }
			death_process_herald_debts = yes
			death_record_weak_ghost_hosts_evolution = yes
			death_record_hollow_ghost_hosts_evolution = yes
			if = {
				limit = { has_global_flag = death_world_end_started }
				death_record_world_end_evolution = yes
				death_spawn_ghost_hosts_for_world_end = yes
				death_attempt_coastal_jump = yes
			}
			else_if = {
				limit = { has_global_flag = death_publicly_revealed }
				death_spawn_ghost_hosts_for_current_tier = yes
				death_absorb_unopposed_neighbors_from_active_wasteland = yes
				death_try_wither = yes
				death_attempt_coastal_jump = yes
			}
			else = {
				death_record_second_shore_evolution = yes
				if = {
					limit = { death_can_spread_to_islands = yes }
					death_try_island_spread = yes
					}
					death_record_mainland_hunger_evolution = yes
					death_record_mainland_hunger_evolution_from_war_bypass = yes
					if = {
						limit = { death_can_push_mainland_reveal = yes }

					}
