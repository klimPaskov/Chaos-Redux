death_consume_current_state = {
	if = {
		limit = { NOT = { death_is_consumed_state = yes } }
		set_temp_variable = { death_last_consumed_population_k = state_population_k }
		set_variable = { death_consumed_population_k = death_last_consumed_population_k }
		set_variable = { death_consumed_date = global.date }
		if = {
			limit = { NOT = { has_global_flag = settings_chaos_deaths_disabled } }
			death_register_consumed_population_deaths = yes
		}
		else = {
			death_remove_current_state_population_without_ledger = yes
		}
		death_strip_current_state_buildings = yes
		death_wither_divisions_in_current_state = yes
		if = {
			limit = { NOT = { has_state_category = wasteland } }
			set_state_category = wasteland
		}
		transfer_state_to = DTH
		add_core_of = DTH
		set_state_controller_to = DTH
		death_apply_active_wasteland_state = yes
		add_to_variable = { global.death_consumed_states = 1 }
		add_to_variable = { global.death_consumed_population = death_last_consumed_population_k }
		DTH = {
			death_setup_country = yes
			set_capital = {
				state = PREV
				remember_old_capital = no
			}
		}
		if = {
			limit = { is_island_state = no }
			add_to_variable = { global.death_mainland_consumed_states = 1 }
		}
		if = {
			limit = { death_should_reveal_from_current_state = yes }
			death_reveal_from_current_state = yes
		}
		death_try_start_world_end = yes
		death_try_fire_whole_world_consumed = yes
	}
	else = {
		if = {
			limit = { is_controlled_by = DTH }
			death_apply_active_wasteland_state = yes
			death_wither_divisions_in_current_state = yes
		}
	}
}

				compare = greater_than_or_equals
			}
			any_controlled_state = { NOT = { death_is_consumed_state = yes } }

				death_clamp_herald_variables = yes
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = death_name_debt
				value = constant:death_decision_tuning.name_debt_high
				compare = greater_than_or_equals
			}
			any_controlled_state = {
				NOT = { is_capital = yes }
				NOT = { death_is_consumed_state = yes }
				OR = {
					death_state_borders_active_wasteland = yes
					AND = {
						has_global_flag = death_world_end_started
						is_coastal = yes
					}
				}
			}
		}
		set_country_flag = { flag = death_herald_collection_cooldown days = @death_herald_collection_cooldown_days }
		random_controlled_state = {
			limit = {
				NOT = { is_capital = yes }
				NOT = { death_is_consumed_state = yes }
				OR = {
					death_state_borders_active_wasteland = yes
					AND = {
						has_global_flag = death_world_end_started
						is_coastal = yes
					}
				}
			}
			death_consume_current_state = yes
			PREV = {
				add_to_variable = { death_name_debt = constant:death_decision_tuning.name_debt_feed_reduction }
				death_clamp_herald_variables = yes
			}
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = death_name_debt
				value = constant:death_decision_tuning.name_debt_high
				compare = greater_than_or_equals
			}
			NOT = { has_war_with = DTH }
			DTH = { can_declare_war_on = ROOT }
		}
		DTH = { declare_war_on = { target = ROOT type = annex_everything } }
	}
}

death_process_herald_debts = {
	every_country = {
		limit = {
			has_country_flag = death_herald_of_zol
			NOT = { is_death_country = yes }
			NOT = { has_global_flag = death_defeated }
		}
		death_initialize_herald_variables = yes
		add_to_variable = { death_name_debt = constant:death_decision_tuning.name_debt_pulse_gain }
		death_clamp_herald_variables = yes
		if = {
			limit = {
				check_variable = {
					var = death_name_debt
					value = constant:death_decision_tuning.name_debt_high
					compare = greater_than_or_equals
				}
				check_variable = {
					var = death_black_favor
					value = 0
					compare = greater_than
				}
			}
			add_to_variable = { death_name_debt = constant:death_decision_tuning.name_debt_favor_payment }
			add_to_variable = { death_black_favor = -1 }
			death_clamp_herald_variables = yes
		}
		else = {
			death_try_collect_herald_name_debt = yes
		}
	}
}

		limit = { has_global_flag = death_world_end_started }
		set_global_flag = { flag = death_coastal_jump_cooldown days = @death_world_end_coastal_jump_cooldown_days }
	}
	else_if = {
		limit = {
			has_global_flag = death_publicly_revealed
			DTH = {
				OR = {
					has_country_flag = death_focus_another_shoreline
					has_country_flag = death_focus_no_ferry_returns
				}
			}
		}
		set_global_flag = { flag = death_coastal_jump_cooldown days = @death_focused_coastal_jump_cooldown_days }
	}
	else_if = {
		limit = { has_global_flag = death_publicly_revealed }
		set_global_flag = { flag = death_coastal_jump_cooldown days = @death_revealed_coastal_jump_cooldown_days }
	}
	else = {
		set_global_flag = { flag = death_coastal_jump_cooldown days = @death_coastal_jump_cooldown_days }
	}
}

death_spawn_ghost_hosts_for_current_tier = {
	if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:death_ghosts.stronger_threshold
				compare = greater_than_or_equals
			}
		}
		random_controlled_state = {
			limit = { has_state_flag = death_active_wasteland }
			death_spawn_stronger_hosts_in_current_state = yes
		}
		random_controlled_state = {
			limit = { has_state_flag = death_active_wasteland }
			death_spawn_stronger_hosts_in_current_state = yes
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:death_ghosts.passive_threshold
				compare = greater_than_or_equals
			}
		}
		random_controlled_state = {
			limit = { has_state_flag = death_active_wasteland }
			death_spawn_passive_hosts_in_current_state = yes
		}
	}
}

death_check_defeat = {
	if = {
		limit = {
			death_country_exists = yes
			DTH = { num_of_controlled_states < 1 }
			NOT = { has_global_flag = death_defeated }
		}
		death_finish_defeat = yes
	}
}

death_finish_defeat = {
	if = {
		limit = { has_global_flag = death_publicly_revealed }
		death_mark_public_defeat_achievements = yes
	}
	else = {
		set_global_flag = death_defeated_before_public_name
		death_mark_quiet_defeat_achievements = yes
	}
	set_global_flag = death_defeated
	clr_global_flag = death_active
	clr_global_flag = death_world_end_started
	clr_global_flag = world_end_death
	DTH = {
		remove_ideas = death_first_silence
		remove_ideas = death_public_death
		remove_ideas = death_world_end_death
		remove_ideas = death_black_census
		delete_unit_template_and_units = { division_template = "Death Passive Host" disband = no }
		delete_unit_template_and_units = { division_template = "Death Hollow Host" disband = no }
		delete_unit_template_and_units = { division_template = "Death Last Shore Host" disband = no }
		every_controlled_state = {
			death_apply_recaptured_wasteland_state = yes
		}
	}
	death_refresh_world_threat_source = yes
	if = {
		limit = { death_should_emit_defeat_super_event = yes }
		death_emit_defeat_super_event = yes
	}
	clr_global_flag = death_publicly_revealed
	clr_global_flag = death_living_compact_formed
	clear_global_event_target = death_living_compact_leader
	clear_global_event_target = death_current_reveal_state
	every_country = {
		limit = { NOT = { is_death_country = yes } }
		remove_ideas = death_black_book_offices
		remove_ideas = death_black_book_scandal
		delete_unit_template_and_units = { division_template = "Black Book Bound Shade" disband = no }
		death_cleanup_herald_status = yes
		clr_country_flag = death_living_compact_member
		clr_country_flag = death_living_compact_leader
		clr_country_flag = death_living_compact_invited
		clr_country_flag = death_quarantine_hold_mission_active
	}
}

death_on_state_control_changed = {
	if = {
		limit = { has_state_flag = death_consumed_state }
		if = {
			limit = { is_controlled_by = DTH }
			death_apply_active_wasteland_state = yes
		}
		else = {
			death_apply_recaptured_wasteland_state = yes
			if = {
				limit = {
					has_state_flag = death_world_end_foothold
					NOT = { has_state_flag = death_world_end_foothold_recapture_counted }
				}
				set_state_flag = death_world_end_foothold_recapture_counted
				add_to_variable = { global.death_world_end_footholds_recaptured = 1 }
			}
			death_check_defeat = yes
		}
	}
}
