death_consume_current_state = {
	if = {
		limit = { NOT = { death_is_consumed_state = yes } }
		set_temp_variable = { death_last_consumed_population_k = state_population_k }
		set_variable = { death_consumed_population_k = death_last_consumed_population_k }
		set_variable = { death_consumed_date = global.date }
		if = {
			limit = { NOT = { has_global_flag = settings_chaos_deaths_disabled } }
			death_register_consumed_population_deaths = yes
		}
		else = {
			death_remove_current_state_population_without_ledger = yes
		}
		death_strip_current_state_buildings = yes
		death_wither_divisions_in_current_state = yes
		if = {
			limit = { NOT = { has_state_category = wasteland } }
			set_state_category = wasteland
		}
		transfer_state_to = DTH
		add_core_of = DTH
		set_state_controller_to = DTH
		death_apply_active_wasteland_state = yes
		add_to_variable = { global.death_consumed_states = 1 }
		add_to_variable = { global.death_consumed_population = death_last_consumed_population_k }
		DTH = {
			death_setup_country = yes
			set_capital = {
				state = PREV
				remember_old_capital = no
			}
		}
		if = {
			limit = { is_island_state = no }
			add_to_variable = { global.death_mainland_consumed_states = 1 }
		}
		if = {
			limit = { death_should_reveal_from_current_state = yes }
			death_reveal_from_current_state = yes
		}
		death_try_start_world_end = yes
		death_try_fire_whole_world_consumed = yes
	}
	else = {
		if = {
			limit = { is_controlled_by = DTH }
			death_apply_active_wasteland_state = yes
			death_wither_divisions_in_current_state = yes
		}
	}
}
