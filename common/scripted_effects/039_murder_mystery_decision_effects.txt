# ============================================================================
# CHAOS REDUX - EVENT 039 MURDER MYSTERY DECISION EFFECTS
# ============================================================================
# Overview:
# - This file is the bounded decision and mission adapter for Event 039.
# - It translates ordinary decision clicks and mission outcomes into the
#   existing runtime transaction, evolution, cell, split, derivative, and
#   terminal helpers without duplicating payment or lifecycle state.
# - Local wrappers exist only where the existing owner contract needs a
#   decision-facing sequence, such as mission failure pressure, foreign-cell
#   target binding, or registration of a later terminal sector.
# - No recurring on_action, GUI, character deletion, or ordinary civil-war
#   effect belongs in this file.
# ============================================================================

# Shared action and mission adapters

murder_mystery_decision_prepare_actor_phase = {
	if = {
		limit = {
			has_variable = murder_mystery_investigation_phase
			check_variable = { var = murder_mystery_investigation_phase value = constant:murder_mystery_phase.none compare = equals }
			has_country_flag = murder_mystery_investigation_country
			NOT = { has_country_flag = murder_mystery_assassin_state }
			NOT = { has_country_flag = murder_mystery_derivative_country }
		}
		set_variable = { murder_mystery_investigation_phase = constant:murder_mystery_phase.map_network }
	}
	if = {
		limit = {
			has_variable = murder_mystery_investigation_phase
			check_variable = { var = murder_mystery_investigation_phase value = constant:murder_mystery_phase.none compare = equals }
			OR = {
				has_country_flag = murder_mystery_assassin_state
				has_country_flag = murder_mystery_derivative_country
			}
		}
		set_variable = { murder_mystery_investigation_phase = constant:murder_mystery_phase.final_operation }
	}
}

murder_mystery_decision_clear_uncommitted_action = {
	if = {
		limit = { NOT = { has_country_flag = murder_mystery_action_completed } }
		clear_variable = murder_mystery_pending_action
		clear_variable = murder_mystery_pending_phase
	}
}

murder_mystery_decision_advance_after_action = {
	if = {
		limit = {
			has_country_flag = murder_mystery_action_completed
			OR = {
				has_country_flag = murder_mystery_action_succeeded
				has_country_flag = murder_mystery_action_partial
			}
		}
		murder_mystery_advance_investigation_phase = yes
	}
}

# Binds one persistent country and state scope for the real objective missions.
# Country variables retain the scope ids across save/reload without introducing
# a shared global event target that another actor could overwrite.
murder_mystery_decision_bind_objective_context = {
	clear_variable = murder_mystery_objective_country
	clear_variable = murder_mystery_objective_state
	capital_scope = {
		ROOT = { set_variable = { murder_mystery_objective_state = PREV } }
	}
	if = {
		limit = {
			has_event_target = murder_mystery_selected_cell
			event_target:murder_mystery_selected_cell = { exists = yes }
			NOT = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.survive_opening_war compare = equals } }
			NOT = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.complete_terminal_sector compare = equals } }
		}
		event_target:murder_mystery_selected_cell = {
			ROOT = { set_variable = { murder_mystery_objective_country = PREV } }
			capital_scope = {
				ROOT = { set_variable = { murder_mystery_objective_state = PREV } }
			}
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.secure_evidence_convoy compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.guard_named_route compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.place_supplied_divisions compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.secure_depots_rail_hubs compare = equals }
			}
		}
		random_owned_controlled_state = {
			limit = {
				impassable = no
				NOT = { is_capital = yes }
			}
			ROOT = { set_variable = { murder_mystery_objective_state = PREV } }
		}
	}
}

murder_mystery_decision_resolve_objective_success = {
	if = {
		limit = { has_country_flag = murder_mystery_objective_active }
		murder_mystery_decision_record_objective_history = yes
		set_country_flag = murder_mystery_last_objective_succeeded
		clr_country_flag = murder_mystery_last_objective_partial
		clr_country_flag = murder_mystery_last_objective_failed
		if = {
			limit = {
				NOT = { has_country_flag = murder_mystery_assassin_state }
				NOT = { has_country_flag = murder_mystery_derivative_country }
			}
			set_variable = { murder_mystery_case_progress_delta = constant:murder_mystery_value.ten }
			set_variable = { murder_mystery_network_reach_delta = constant:murder_mystery_value.negative_seven }
			set_variable = { murder_mystery_evidence_integrity_delta = constant:murder_mystery_value.five }
			murder_mystery_adjust_case_progress = yes
			murder_mystery_adjust_network_reach = yes
			murder_mystery_adjust_evidence_integrity = yes
		}
		else = {
			set_variable = { murder_mystery_network_reach_delta = constant:murder_mystery_value.five }
			set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_value.five }
			murder_mystery_adjust_network_reach = yes
			murder_mystery_adjust_brotherhood_cohesion = yes
		}
		set_variable = { murder_mystery_pending_action = constant:murder_mystery_action.none }
		set_variable = { murder_mystery_action_result_input = constant:murder_mystery_action_result.success }
		murder_mystery_record_incident = yes
		clear_variable = murder_mystery_case_progress_delta
		clear_variable = murder_mystery_network_reach_delta
		clear_variable = murder_mystery_evidence_integrity_delta
		clear_variable = murder_mystery_brotherhood_cohesion_delta
		clear_variable = murder_mystery_pending_action
		clear_variable = murder_mystery_action_result_input
	}
	murder_mystery_decision_clear_objective_context = yes
}

			NOT = { has_country_flag = murder_mystery_terminal_exempt }

		}
		save_global_event_target_as = murder_mystery_terminal_target
		set_country_flag = murder_mystery_terminal_target_classified

	set_variable = { murder_mystery_terminal_registration_war_state = constant:murder_mystery_terminal.status_engaged }

		murder_mystery_decision_mark_terminal_sector_engaged = yes

	clear_variable = murder_mystery_terminal_registration_war_state
	clear_variable = murder_mystery_terminal_registration_cell
