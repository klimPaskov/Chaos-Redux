# ============================================================================
# CHAOS REDUX - EVENT 039 MURDER MYSTERY DECISION EFFECTS
# ============================================================================
# Overview:
# - This file is the bounded decision and mission adapter for Event 039.
# - It translates ordinary decision clicks and mission outcomes into the
#   existing runtime transaction, evolution, cell, split, derivative, and
#   terminal helpers without duplicating payment or lifecycle state.
# - Local wrappers exist only where the existing owner contract needs a
#   decision-facing sequence, such as mission failure pressure, foreign-cell
#   target binding, or registration of a later terminal sector.
# - No recurring on_action, GUI, character removal, or ordinary civil-war
#   effect belongs in this file.
# ============================================================================

# add_ai_strategy takes literal values. Objective targets the central state
# must conquer or protect while their objective runs.
@murder_mystery_objective_ai_conquer = 150
@murder_mystery_objective_ai_protect = 100

# Shared action and mission adapters

murder_mystery_decision_prepare_actor_phase = {
	if = {
		limit = {
			has_variable = murder_mystery_investigation_phase
			check_variable = { var = murder_mystery_investigation_phase value = constant:murder_mystery_phase.none compare = equals }
			has_country_flag = murder_mystery_investigation_country
			NOT = { has_country_flag = murder_mystery_assassin_state }
			NOT = { has_country_flag = murder_mystery_derivative_country }
		}
		set_variable = { murder_mystery_investigation_phase = constant:murder_mystery_phase.map_network }
	}
	if = {
		limit = {
			has_variable = murder_mystery_investigation_phase
			check_variable = { var = murder_mystery_investigation_phase value = constant:murder_mystery_phase.none compare = equals }
			OR = {
				has_country_flag = murder_mystery_assassin_state
				has_country_flag = murder_mystery_derivative_country
			}
		}
		set_variable = { murder_mystery_investigation_phase = constant:murder_mystery_phase.final_operation }
	}
}

murder_mystery_decision_clear_uncommitted_action = {
	if = {
		limit = { NOT = { has_country_flag = murder_mystery_action_completed } }
		clear_variable = murder_mystery_pending_action
		clear_variable = murder_mystery_pending_phase
	}
	clear_variable = murder_mystery_pending_cost_profile
}

murder_mystery_decision_advance_after_action = {
	if = {
		limit = {
			has_country_flag = murder_mystery_action_completed
			OR = {
				has_country_flag = murder_mystery_action_succeeded
				has_country_flag = murder_mystery_action_partial
			}
		}
		murder_mystery_advance_investigation_phase = yes
	}
}

murder_mystery_decision_run_investigation_action = {
	clr_country_flag = murder_mystery_action_completed
	clr_country_flag = murder_mystery_action_succeeded
	clr_country_flag = murder_mystery_action_partial
	clr_country_flag = murder_mystery_action_failed
	murder_mystery_decision_prepare_actor_phase = yes
	murder_mystery_execute_investigation_action = yes
	murder_mystery_decision_advance_after_action = yes
	murder_mystery_decision_clear_uncommitted_action = yes
	if = {
		limit = { NOT = { has_country_flag = murder_mystery_objective_active } }
		clr_country_flag = murder_mystery_phase_mission_active
		clear_variable = murder_mystery_phase_mission_expiry_days
	}
}

	set_variable = { murder_mystery_pending_phase = constant:murder_mystery_phase.protect_likely_targets }

		limit = {
			has_country_flag = murder_mystery_action_completed

# Binds one persistent country and state scope for the real objective missions.
# Country variables retain the scope ids across save/reload without introducing
# a shared global event target that another actor could overwrite.
murder_mystery_decision_bind_objective_context = {
	# ROOT can still be the original host during Assassin State activation.
	# The regular pointer names this helper's recipient for the entire chain.
	save_event_target_as = murder_mystery_objective_actor
	clear_variable = murder_mystery_objective_country
	clear_variable = murder_mystery_objective_state
	clear_variable = murder_mystery_objective_origin_state
	clear_variable = murder_mystery_objective_sector
	clear_array = murder_mystery_objective_sector_countries
	set_variable = { murder_mystery_objective_owner = THIS }
	every_owned_state = {
		limit = { is_capital = yes is_owned_and_controlled_by = event_target:murder_mystery_objective_actor }
		event_target:murder_mystery_objective_actor = {
			set_variable = { murder_mystery_objective_origin_state = PREV }
			set_variable = { murder_mystery_objective_state = PREV }
		}
	}
	if = {
		limit = {
			has_event_target = murder_mystery_selected_cell
			event_target:murder_mystery_selected_cell = { exists = yes }
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.prevent_revolt_preparation compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.dismantle_local_recruitment compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_local_government compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.intercept_support_shipment compare = equals }
			}
		}
		event_target:murder_mystery_selected_cell = {
			event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_country = PREV } }
		}
	}
	else_if = {
		limit = {
			murder_mystery_country_has_registered_cell = yes
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.prevent_revolt_preparation compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.dismantle_local_recruitment compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_local_government compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.intercept_support_shipment compare = equals }
			}
		}
		set_variable = { murder_mystery_objective_country = THIS }
	}
	if = {
		limit = {
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.secure_evidence_convoy compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.guard_named_route compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.place_supplied_divisions compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.secure_depots_rail_hubs compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.keep_capital_connected compare = equals }
			}
		}
		random_owned_controlled_state = {
			limit = {
				impassable = no
				NOT = { is_capital = yes }
			}
			event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
		}
	}
	murder_mystery_decision_bind_movement_objective = yes
}

# COUNTRY scope: a foreign government after its Dismantle the Local Cell
# mission. Success dismantles the registered cell; a partial result costs the
# cell maturity and one stage.

		limit = {
			has_country_flag = murder_mystery_action_succeeded

		}
		murder_mystery_record_captured_cell = yes
		set_temp_variable = { murder_mystery_cell_retire_reason = constant:murder_mystery_defeat.retire_reason_dismantled }

	}
	else_if = {
		limit = {
			has_country_flag = murder_mystery_action_partial
			murder_mystery_country_has_registered_cell = yes
		}
		murder_mystery_damage_cell_row = yes
	}
}

murder_mystery_decision_resolve_objective_success = {
	if = {
		limit = { murder_mystery_objective_owner_is_valid = yes }
		murder_mystery_decision_record_objective_history = yes
		murder_mystery_achievement_record_objective_success = yes
		set_temp_variable = { murder_mystery_objective_outcome = constant:murder_mystery_action_result.success }
		murder_mystery_decision_apply_objective_outcome = yes
		set_country_flag = murder_mystery_last_objective_succeeded
		clr_country_flag = murder_mystery_last_objective_partial
		clr_country_flag = murder_mystery_last_objective_failed
		if = {
			limit = {
				NOT = { has_country_flag = murder_mystery_assassin_state }
				NOT = { has_country_flag = murder_mystery_derivative_country }
			}

# COUNTRY scope: the objective owner. Input: temp murder_mystery_offered_objective.
# Starts that objective only when the owner has no objective running, so a
# later action never overwrites the running objective's context.
murder_mystery_decision_offer_objective = {
	if = {
		limit = {
			murder_mystery_runtime_is_active = yes
			NOT = { has_country_flag = murder_mystery_objective_active }
			check_variable = { var = murder_mystery_offered_objective value = constant:murder_mystery_objective.none compare = greater_than }
		}
		set_variable = { murder_mystery_objective_type = murder_mystery_offered_objective }
		murder_mystery_decision_start_bound_objective = yes
		if = {
			limit = { NOT = { has_country_flag = murder_mystery_objective_active } }
			clear_variable = murder_mystery_objective_type
		}
	}
	set_temp_variable = { murder_mystery_offered_objective = constant:murder_mystery_objective.none }
}

# COUNTRY scope: the objective owner. Output: temp
# murder_mystery_objective_row_index, the cell registry row of the bound
# country, or -1 when it has none.
murder_mystery_decision_read_objective_cell_row = {
	set_temp_variable = { murder_mystery_objective_row_index = constant:murder_mystery_value.negative_one }
	if = {
		limit = {
			has_variable = murder_mystery_objective_country
			murder_mystery_cell_registry_is_aligned = yes
		}
		var:murder_mystery_objective_country = { save_event_target_as = murder_mystery_objective_cell_target }
		set_temp_variable = { murder_mystery_objective_row_scan = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_objective_row_scan value = global.murder_mystery_cell_country_entries^num compare = less_than } }
			var:global.murder_mystery_cell_country_entries^murder_mystery_objective_row_scan = {
				if = {
					limit = { tag = event_target:murder_mystery_objective_cell_target }
					set_temp_variable = { murder_mystery_objective_row_index = murder_mystery_objective_row_scan }
				}
			}
			add_to_temp_variable = { murder_mystery_objective_row_scan = constant:murder_mystery_identity.loop_increment }
		}
	}
}

# COUNTRY scope: the objective owner. Input: temps
# murder_mystery_objective_cell_maturity_delta and
# murder_mystery_objective_cell_readiness_delta and
# murder_mystery_objective_cell_exposure_delta. Applies them to the bound
# cell's registry row and resyncs its mature flag.
murder_mystery_decision_adjust_objective_cell_row = {
	murder_mystery_decision_read_objective_cell_row = yes
	if = {
		limit = { check_variable = { var = murder_mystery_objective_row_index value = constant:murder_mystery_value.zero compare = greater_than_or_equals } }
		set_temp_variable = { murder_mystery_objective_row_value = global.murder_mystery_cell_maturity_entries^murder_mystery_objective_row_index }
		add_to_temp_variable = { murder_mystery_objective_row_value = murder_mystery_objective_cell_maturity_delta }
		clamp_temp_variable = { var = murder_mystery_objective_row_value min = constant:murder_mystery_value.minimum max = constant:murder_mystery_value.maximum }
		set_variable = { global.murder_mystery_cell_maturity_entries^murder_mystery_objective_row_index = murder_mystery_objective_row_value }
		set_temp_variable = { murder_mystery_objective_row_value = global.murder_mystery_cell_revolt_readiness_entries^murder_mystery_objective_row_index }
		add_to_temp_variable = { murder_mystery_objective_row_value = murder_mystery_objective_cell_readiness_delta }
		clamp_temp_variable = { var = murder_mystery_objective_row_value min = constant:murder_mystery_cell.minimum_revolt_readiness max = constant:murder_mystery_cell.maximum_revolt_readiness }
		set_variable = { global.murder_mystery_cell_revolt_readiness_entries^murder_mystery_objective_row_index = murder_mystery_objective_row_value }
		set_temp_variable = { murder_mystery_objective_row_value = global.murder_mystery_cell_exposure_entries^murder_mystery_objective_row_index }
		add_to_temp_variable = { murder_mystery_objective_row_value = murder_mystery_objective_cell_exposure_delta }
		clamp_temp_variable = { var = murder_mystery_objective_row_value min = constant:murder_mystery_value.minimum max = constant:murder_mystery_value.maximum }
		set_variable = { global.murder_mystery_cell_exposure_entries^murder_mystery_objective_row_index = murder_mystery_objective_row_value }
		set_temp_variable = { murder_mystery_objective_row_value = global.murder_mystery_cell_maturity_entries^murder_mystery_objective_row_index }
		var:murder_mystery_objective_country = {
			if = {
				limit = { check_variable = { var = murder_mystery_objective_row_value value = constant:murder_mystery_action_tuning.high_value_maturity compare = greater_than_or_equals } }
				set_country_flag = murder_mystery_cell_mature
			}
		}
	}
	set_temp_variable = { murder_mystery_objective_cell_maturity_delta = constant:murder_mystery_value.zero }
	set_temp_variable = { murder_mystery_objective_cell_readiness_delta = constant:murder_mystery_value.zero }
	set_temp_variable = { murder_mystery_objective_cell_exposure_delta = constant:murder_mystery_value.zero }
}

# COUNTRY scope: the objective owner. Output:
# murder_mystery_objective_progress_value, the number of governments in the
# bound sector snapshot that capitulated, accepted subject status or are gone.
murder_mystery_decision_count_sector_progress = {
	save_event_target_as = murder_mystery_objective_progress_owner
	set_temp_variable = { murder_mystery_objective_progress_count = constant:murder_mystery_value.zero }
	if = {
		limit = {
			has_variable = murder_mystery_objective_sector
			check_variable = { var = murder_mystery_objective_sector_countries^num value = constant:murder_mystery_terminal.maximum_registry_rows compare = less_than_or_equals }
		}
		for_each_scope_loop = {
			array = murder_mystery_objective_sector_countries
			if = {
				limit = {
					OR = {
						NOT = { exists = yes }
						has_capitulated = yes
						is_subject_of = event_target:murder_mystery_objective_progress_owner
					}
				}
				add_to_temp_variable = { murder_mystery_objective_progress_count = constant:murder_mystery_value.one }
			}
		}
	}
	set_variable = { murder_mystery_objective_progress_value = murder_mystery_objective_progress_count }
}

# COUNTRY scope: the objective owner, called at the end of
# murder_mystery_decision_bind_objective_context with
# event_target:murder_mystery_objective_actor saved. Binds the targets of the
# agency, movement and terminal objectives; a target that cannot be found
# stays unset and murder_mystery_objective_binding_is_valid rejects the start.
murder_mystery_decision_bind_movement_objective = {
	clear_variable = murder_mystery_objective_target_value
	clear_variable = murder_mystery_objective_limit_value
	clear_variable = murder_mystery_objective_start_value
	clear_variable = murder_mystery_objective_progress_value
	clear_variable = murder_mystery_objective_floor_value
	clear_array = murder_mystery_objective_subject_countries

	# Thresholds shown in the mission text.
	if = {
		limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.synchronized_operation compare = equals } }
		set_variable = { murder_mystery_objective_target_value = constant:murder_mystery_objective_tuning.synchronized_cell_count }
	}
	if = {
		limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.reduce_sector_governments compare = equals } }
		set_variable = { murder_mystery_objective_target_value = constant:murder_mystery_objective_tuning.reduce_governments_minimum }
	}
	if = {
		limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_central_command compare = equals } }
		set_variable = { murder_mystery_objective_floor_value = constant:murder_mystery_objective_tuning.central_command_cohesion_floor }
	}

	# Agency operation: the case must climb before the network moves.
	if = {
		limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.agency_operation compare = equals } }
		set_variable = { murder_mystery_objective_floor_value = constant:murder_mystery_objective_tuning.agency_evidence_floor }
		set_variable = { murder_mystery_objective_target_value = murder_mystery_case_progress }
		add_to_variable = { murder_mystery_objective_target_value = constant:murder_mystery_objective_tuning.agency_case_progress_gain }
		set_variable = { murder_mystery_objective_limit_value = murder_mystery_network_reach }
		add_to_variable = { murder_mystery_objective_limit_value = constant:murder_mystery_objective_tuning.agency_reach_tolerance }
		if = {
			limit = { murder_mystery_country_has_registered_cell = yes }
			set_variable = { murder_mystery_objective_country = THIS }
		}
	}

	# Supply route and host archive: a host state on the front line.
	if = {
		limit = {
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.contest_supply_route compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.capture_host_archive compare = equals }
			}
			has_event_target = murder_mystery_original_host
			event_target:murder_mystery_original_host = {
				exists = yes
				NOT = { tag = event_target:murder_mystery_objective_actor }
			}
		}
		clear_variable = murder_mystery_objective_state
		event_target:murder_mystery_original_host = {
			event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_country = PREV } }
		}
		if = {
			limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.capture_host_archive compare = equals } }
			event_target:murder_mystery_original_host = {
				capital_scope = {
					if = {
						limit = {
							is_controlled_by = PREV
							any_neighbor_state = { is_controlled_by = event_target:murder_mystery_objective_actor }
						}
						event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
					}
				}
			}
		}
		if = {
			limit = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.contest_supply_route compare = equals }
			}
			event_target:murder_mystery_original_host = {
				random_owned_controlled_state = {
					limit = {
						impassable = no
						supply_node > 0
						any_neighbor_state = { is_controlled_by = event_target:murder_mystery_objective_actor }
					}
					event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
				}
			}
		}
		if = {
			limit = { NOT = { has_variable = murder_mystery_objective_state } }
			event_target:murder_mystery_original_host = {
				random_owned_controlled_state = {
					limit = {
						impassable = no
						any_neighbor_state = { is_controlled_by = event_target:murder_mystery_objective_actor }
					}
					event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
				}
			}
		}
	}

	# Cell objectives: the cell the central state selected.
	if = {
		limit = {
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.keep_cell_supplied compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.raise_cell_quietly compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.prepare_revolt_cluster compare = equals }
			}
			has_event_target = murder_mystery_selected_cell
			event_target:murder_mystery_selected_cell = {
				exists = yes
				murder_mystery_country_has_registered_cell = yes
			}
		}
		event_target:murder_mystery_selected_cell = {
			event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_country = PREV } }
			capital_scope = {
				event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
			}
		}
		if = {
			limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.prepare_revolt_cluster compare = equals } }
			event_target:murder_mystery_selected_cell = {
				random_owned_controlled_state = {
					limit = {
						impassable = no
						is_capital = no
					}
					event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
				}
			}
		}
		if = {
			limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.raise_cell_quietly compare = equals } }
			murder_mystery_decision_read_objective_cell_row = yes
			if = {
				limit = { check_variable = { var = murder_mystery_objective_row_index value = constant:murder_mystery_value.zero compare = greater_than_or_equals } }
				set_variable = { murder_mystery_objective_start_value = global.murder_mystery_cell_maturity_entries^murder_mystery_objective_row_index }
				set_variable = { murder_mystery_objective_target_value = murder_mystery_objective_start_value }
				add_to_variable = { murder_mystery_objective_target_value = constant:murder_mystery_objective_tuning.quiet_maturity_gain }
				set_variable = { murder_mystery_objective_limit_value = constant:murder_mystery_objective_tuning.quiet_exposure_ceiling }
			}
		}
	}

	# Derivative objectives. Subject loyalty watches every bound derivative and
	# highlights the first; Protect a Derivative takes the smallest one.
	if = {
		limit = {
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.hold_subject_loyalty compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_derivative compare = equals }
			}
			murder_mystery_derivative_registry_is_aligned = yes
		}
		set_temp_variable = { murder_mystery_objective_best_states = constant:murder_mystery_value.negative_one }
		set_temp_variable = { murder_mystery_objective_derivative_scan = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_objective_derivative_scan value = global.murder_mystery_derivative_country_entries^num compare = less_than } }
			if = {
				limit = { check_variable = { var = global.murder_mystery_derivative_status_entries^murder_mystery_objective_derivative_scan value = constant:murder_mystery_derivative.status_active compare = equals } }
				var:global.murder_mystery_derivative_country_entries^murder_mystery_objective_derivative_scan = {
					if = {
						limit = {
							exists = yes
							has_capitulated = no
							is_subject_of = event_target:murder_mystery_objective_actor
						}
						event_target:murder_mystery_objective_actor = { add_to_array = { array = murder_mystery_objective_subject_countries value = PREV } }
					}
					if = {
						limit = {
							exists = yes
							has_capitulated = no
							is_subject_of = event_target:murder_mystery_objective_actor
							OR = {
								check_variable = { var = murder_mystery_objective_best_states value = constant:murder_mystery_value.zero compare = less_than }
								check_variable = { var = num_owned_states value = murder_mystery_objective_best_states compare = less_than }
							}
						}
						set_temp_variable = { murder_mystery_objective_best_states = num_owned_states }
						event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_country = PREV } }
						capital_scope = {
							event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
						}
					}
				}
			}
			add_to_temp_variable = { murder_mystery_objective_derivative_scan = constant:murder_mystery_identity.loop_increment }
		}
		if = {
			limit = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_derivative compare = equals }
				has_variable = murder_mystery_objective_country
			}
			var:murder_mystery_objective_country = {
				event_target:murder_mystery_objective_actor = { add_ai_strategy = { type = protect id = PREV value = @murder_mystery_objective_ai_protect } }
			}
		}
	}

	# Terminal war targets: the current campaign target government.
	if = {
		limit = {
			OR = {
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.dismantle_command_network compare = equals }
				check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.force_capitulation compare = equals }
			}
			has_event_target = murder_mystery_terminal_target
			event_target:murder_mystery_terminal_target = {
				exists = yes
				NOT = { tag = event_target:murder_mystery_objective_actor }
			}
		}
		event_target:murder_mystery_terminal_target = {
			event_target:murder_mystery_objective_actor = {
				set_variable = { murder_mystery_objective_country = PREV }
				add_ai_strategy = { type = conquer id = PREV value = @murder_mystery_objective_ai_conquer }
			}
			capital_scope = {
				event_target:murder_mystery_objective_actor = { set_variable = { murder_mystery_objective_state = PREV } }
			}
		}
	}
}

# COUNTRY scope: the objective owner, inside an owner-valid resolution.
# Input: temp murder_mystery_objective_outcome (success, partial or failure
# from murder_mystery_action_result). Applies the outcome of the agency,
# movement and terminal objectives beyond the shared value changes.
murder_mystery_decision_apply_objective_outcome = {
	set_temp_variable = { murder_mystery_objective_cell_maturity_delta = constant:murder_mystery_value.zero }
	set_temp_variable = { murder_mystery_objective_cell_readiness_delta = constant:murder_mystery_value.zero }
	set_temp_variable = { murder_mystery_objective_cell_exposure_delta = constant:murder_mystery_value.zero }

	# Agency operation: a completed operation breaks the cell it was watching.
	# The host's own cell is its national cell; a registered foreign cell row
	# loses maturity and a stage.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.agency_operation compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
		}
		if = {
			limit = { murder_mystery_objective_cell_is_registered = yes }
			var:murder_mystery_objective_country = { murder_mystery_damage_cell_row = yes }
		}
		else_if = {
			limit = {
				has_country_flag = murder_mystery_original_host
				has_variable = murder_mystery_cell_maturity
			}
			set_variable = { murder_mystery_cell_maturity_delta = constant:murder_mystery_objective_tuning.agency_cell_maturity_loss }
			murder_mystery_adjust_cell_maturity = yes
			clear_variable = murder_mystery_cell_maturity_delta
		}
	}

	# Supply route: the host's security corridor breaks and the movement
	# captures fuel stocks.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.contest_supply_route compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
		}
		add_fuel = constant:murder_mystery_objective_tuning.supply_route_fuel
		if = {
			limit = {
				has_variable = murder_mystery_objective_country
				var:murder_mystery_objective_country = {
					exists = yes
					has_variable = murder_mystery_route_security
				}
			}
			var:murder_mystery_objective_country = {
				set_variable = { murder_mystery_route_security_delta = constant:murder_mystery_objective_tuning.supply_route_host_security_loss }
				murder_mystery_adjust_route_security = yes
				clear_variable = murder_mystery_route_security_delta
			}
		}
	}

	# Host archive: the case files burn with the archive.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.capture_host_archive compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
		}
		set_variable = { murder_mystery_network_reach_delta = constant:murder_mystery_objective_tuning.archive_reach_gain }
		murder_mystery_adjust_network_reach = yes
		clear_variable = murder_mystery_network_reach_delta
		if = {
			limit = {
				has_variable = murder_mystery_objective_country
				var:murder_mystery_objective_country = {
					exists = yes
					has_variable = murder_mystery_case_progress
				}
			}
			var:murder_mystery_objective_country = {
				set_variable = { murder_mystery_case_progress_delta = constant:murder_mystery_objective_tuning.archive_host_case_loss }
				set_variable = { murder_mystery_evidence_integrity_delta = constant:murder_mystery_objective_tuning.archive_host_evidence_loss }
				murder_mystery_adjust_case_progress = yes
				murder_mystery_adjust_evidence_integrity = yes
				clear_variable = murder_mystery_case_progress_delta
				clear_variable = murder_mystery_evidence_integrity_delta
			}
		}
	}

	# Keep a cell supplied: the cell matures and readies.
	if = {
		limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.keep_cell_supplied compare = equals } }
		if = {
			limit = { check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals } }
			set_temp_variable = { murder_mystery_objective_cell_maturity_delta = constant:murder_mystery_objective_tuning.cell_supplied_maturity_success }
			set_temp_variable = { murder_mystery_objective_cell_readiness_delta = constant:murder_mystery_objective_tuning.cell_supplied_readiness_success }
			murder_mystery_decision_adjust_objective_cell_row = yes
		}
		if = {
			limit = { check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.partial compare = equals } }
			set_temp_variable = { murder_mystery_objective_cell_maturity_delta = constant:murder_mystery_objective_tuning.cell_supplied_maturity_partial }
			murder_mystery_decision_adjust_objective_cell_row = yes
		}
	}

	# Quiet maturity: a cell that grew unseen sheds exposure.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.raise_cell_quietly compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
		}
		set_temp_variable = { murder_mystery_objective_cell_exposure_delta = constant:murder_mystery_objective_tuning.quiet_exposure_relief }
		murder_mystery_decision_adjust_objective_cell_row = yes
	}

	# Revolt cluster: a prepared cell is first in line for the next revolt.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.prepare_revolt_cluster compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
		}
		set_temp_variable = { murder_mystery_objective_cell_readiness_delta = constant:murder_mystery_cell.maximum_revolt_readiness }
		murder_mystery_decision_adjust_objective_cell_row = yes
	}

	# Subject loyalty: the settlement holds, or Cohesion breaks with it.
	if = {
		limit = { check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.hold_subject_loyalty compare = equals } }
		if = {
			limit = { check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals } }
			set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_objective_tuning.subject_loyalty_cohesion_success }
		}
		else_if = {
			limit = { check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.failure compare = equals } }
			set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_objective_tuning.subject_loyalty_cohesion_failure }
		}
		if = {
			limit = { has_variable = murder_mystery_brotherhood_cohesion_delta }
			murder_mystery_adjust_brotherhood_cohesion = yes
			clear_variable = murder_mystery_brotherhood_cohesion_delta
		}
	}

	# Synchronized operation: every insurrection cell moves closer to revolt.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.synchronized_operation compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
			murder_mystery_cell_registry_is_aligned = yes
		}
		set_variable = { murder_mystery_network_reach_delta = constant:murder_mystery_objective_tuning.synchronized_reach_gain }
		murder_mystery_adjust_network_reach = yes
		clear_variable = murder_mystery_network_reach_delta
		set_temp_variable = { murder_mystery_objective_sync_scan = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_objective_sync_scan value = global.murder_mystery_cell_country_entries^num compare = less_than } }
			if = {
				limit = { check_variable = { var = global.murder_mystery_cell_stage_entries^murder_mystery_objective_sync_scan value = constant:murder_mystery_cell_stage.insurrection compare = greater_than_or_equals } }
				set_temp_variable = { murder_mystery_objective_row_value = global.murder_mystery_cell_revolt_readiness_entries^murder_mystery_objective_sync_scan }
				add_to_temp_variable = { murder_mystery_objective_row_value = constant:murder_mystery_objective_tuning.synchronized_readiness_gain }
				clamp_temp_variable = { var = murder_mystery_objective_row_value min = constant:murder_mystery_cell.minimum_revolt_readiness max = constant:murder_mystery_cell.maximum_revolt_readiness }
				set_variable = { global.murder_mystery_cell_revolt_readiness_entries^murder_mystery_objective_sync_scan = murder_mystery_objective_row_value }
			}
			add_to_temp_variable = { murder_mystery_objective_sync_scan = constant:murder_mystery_identity.loop_increment }
		}
	}

	# Force a capitulation: a partial result keeps the campaign council
	# together.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.force_capitulation compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.partial compare = equals }
		}
		set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_objective_tuning.capitulation_partial_cohesion }
		murder_mystery_adjust_brotherhood_cohesion = yes
		clear_variable = murder_mystery_brotherhood_cohesion_delta
	}

	# Protect the central command: a command that held through the delegation
	# regains its authority.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_central_command compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
		}
		set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_objective_tuning.central_command_cohesion_success }
		murder_mystery_adjust_brotherhood_cohesion = yes
		clear_variable = murder_mystery_brotherhood_cohesion_delta
	}

	# Protect a derivative: the rescued government's Cohesion recovers.
	if = {
		limit = {
			check_variable = { var = murder_mystery_objective_type value = constant:murder_mystery_objective.protect_derivative compare = equals }
			check_variable = { var = murder_mystery_objective_outcome value = constant:murder_mystery_action_result.success compare = equals }
			has_variable = murder_mystery_objective_country
			var:murder_mystery_objective_country = {
				exists = yes
				has_variable = murder_mystery_brotherhood_cohesion
			}
		}
		var:murder_mystery_objective_country = {
			set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_objective_tuning.protect_derivative_cohesion_success }
			murder_mystery_adjust_brotherhood_cohesion = yes
			clear_variable = murder_mystery_brotherhood_cohesion_delta
		}
	}
}

# COUNTRY scope: the central state, holding murder_mystery_selected_terminal_sector.

		}

		set_temp_variable = { murder_mystery_revolt_sector_filter = murder_mystery_selected_terminal_sector }

			if = {
				limit = {
					NOT = { has_event_target = murder_mystery_selected_cell }

					murder_mystery_derivative_registry_has_capacity = yes

				murder_mystery_select_revolt_cell = yes
				if = {
					limit = { has_event_target = murder_mystery_selected_cell }
					murder_mystery_start_derivative_creation = yes
					murder_mystery_clear_selected_cell_context = yes
				}
				else = {
					set_temp_variable = { murder_mystery_terminal_uprisings_left = constant:murder_mystery_value.zero }
				}
			}
			else = {
				set_temp_variable = { murder_mystery_terminal_uprisings_left = constant:murder_mystery_value.zero }
			}
		}
		set_temp_variable = { murder_mystery_revolt_sector_filter = constant:murder_mystery_value.zero }

# COUNTRY scope: a country with a registered cell. Input: temp
# murder_mystery_cell_local_case_change. Adds it to the cell row's local case.
murder_mystery_adjust_cell_local_case = {
	if = {
		limit = { murder_mystery_cell_registry_is_aligned = yes }
		save_event_target_as = murder_mystery_local_case_country
		set_temp_variable = { murder_mystery_local_case_index = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_local_case_index value = global.murder_mystery_cell_country_entries^num compare = less_than } }
			set_temp_variable = { murder_mystery_local_case_match = constant:murder_mystery_value.zero }
			var:global.murder_mystery_cell_country_entries^murder_mystery_local_case_index = {
				if = {
					limit = { tag = event_target:murder_mystery_local_case_country }
					set_temp_variable = { murder_mystery_local_case_match = constant:murder_mystery_value.one }
				}
			}
			if = {
				limit = { check_variable = { var = murder_mystery_local_case_match value = constant:murder_mystery_value.one compare = equals } }
				set_temp_variable = { murder_mystery_local_case_value = global.murder_mystery_cell_local_case_entries^murder_mystery_local_case_index }
				add_to_temp_variable = { murder_mystery_local_case_value = murder_mystery_cell_local_case_change }
				clamp_temp_variable = { var = murder_mystery_local_case_value min = constant:murder_mystery_value.minimum max = constant:murder_mystery_value.maximum }
				set_variable = { global.murder_mystery_cell_local_case_entries^murder_mystery_local_case_index = murder_mystery_local_case_value }
				set_temp_variable = { murder_mystery_local_case_index = global.murder_mystery_cell_country_entries^num }
			}
			add_to_temp_variable = { murder_mystery_local_case_index = constant:murder_mystery_identity.loop_increment }
		}
	}
}

# COUNTRY scope: the supporting government; FROM is the partner chosen in
# Support a Threatened Partner. The support costs the route watch family.
# While the movement's Network Reach is above intercept_reach it can intercept
# the convoy: the partner gains nothing, the movement learns from the
# shipment and the sender's investigators are exposed. Otherwise the partner's
# local case and the protection of its likely targets improve.
murder_mystery_decision_support_threatened_partner = {
	set_variable = { murder_mystery_pending_cost_profile = constant:murder_mystery_cost_profile.route_watch }
	murder_mystery_pay_action_cost = yes
	murder_mystery_clear_action_cost_receipt = yes
	murder_mystery_start_action_cooldown = yes
	FROM = { save_event_target_as = murder_mystery_supported_partner }
	murder_mystery_read_movement_reach = yes
	set_temp_variable = { murder_mystery_partner_intercepted = constant:murder_mystery_value.zero }
	if = {
		limit = { check_variable = { var = murder_mystery_pulse_reach value = constant:murder_mystery_partner_support.intercept_reach compare = greater_than } }
		random = {
			chance = @murder_mystery_partner_intercept_chance
			set_temp_variable = { murder_mystery_partner_intercepted = constant:murder_mystery_value.one }
		}
	}
	if = {
		limit = { check_variable = { var = murder_mystery_partner_intercepted value = constant:murder_mystery_value.one compare = equals } }
		murder_mystery_initialize_country_runtime = yes
		add_to_variable = { murder_mystery_investigator_exposure = constant:murder_mystery_partner_support.intercept_exposure }
		murder_mystery_clamp_country_values = yes
		if = {
			limit = {
				has_global_flag = murder_mystery_assassin_state_exists
				has_event_target = murder_mystery_assassin_state
				event_target:murder_mystery_assassin_state = { exists = yes }
			}
			event_target:murder_mystery_assassin_state = {
				add_to_variable = { murder_mystery_network_reach = constant:murder_mystery_partner_support.intercept_reach_gain }
				murder_mystery_clamp_country_values = yes
			}
		}
		else_if = {
			limit = {
				has_event_target = murder_mystery_original_host
				event_target:murder_mystery_original_host = { exists = yes }
			}
			event_target:murder_mystery_original_host = {
				add_to_variable = { murder_mystery_network_reach = constant:murder_mystery_partner_support.intercept_reach_gain }
				murder_mystery_clamp_country_values = yes
			}
		}
		country_event = { id = chaosx.nr39.67 }
	}
	else = {
		FROM = {
			murder_mystery_initialize_country_runtime = yes
			add_to_variable = { murder_mystery_witness_safety = constant:murder_mystery_partner_support.witness_safety_gain }
			subtract_from_variable = { murder_mystery_target_vulnerability = constant:murder_mystery_partner_support.vulnerability_loss }
			murder_mystery_clamp_country_values = yes
			set_temp_variable = { murder_mystery_cell_local_case_change = constant:murder_mystery_partner_support.local_case_gain }
			murder_mystery_adjust_cell_local_case = yes
			set_country_flag = murder_mystery_partner_supported
		}
	}
}

	}
	send_equipment = {
		equipment = support_equipment
		amount = constant:murder_mystery_derivative_support.support_shipment
		target = FROM
	}
	set_temp_variable = { murder_mystery_reinforce_manpower = constant:murder_mystery_value.zero }
	subtract_from_temp_variable = { murder_mystery_reinforce_manpower = constant:murder_mystery_derivative_support.manpower }
	add_manpower = murder_mystery_reinforce_manpower
	FROM = {
		add_manpower = constant:murder_mystery_derivative_support.manpower
		set_temp_variable = { murder_mystery_timed_flag_days = constant:murder_mystery_derivative_support.reinforcement_window_days }
		meta_effect = {
			text = {
				set_country_flag = { flag = murder_mystery_derivative_reinforced days = [TIMED_FLAG_DAYS] }
			}
			TIMED_FLAG_DAYS = "[?murder_mystery_timed_flag_days|.0]"
		}
	}
	set_temp_variable = { murder_mystery_reinforced_derivatives = constant:murder_mystery_value.zero }
	for_each_scope_loop = {
		array = global.murder_mystery_derivative_country_entries
		if = {
			limit = {
				exists = yes
				has_country_flag = murder_mystery_derivative_reinforced
			}
			add_to_temp_variable = { murder_mystery_reinforced_derivatives = constant:murder_mystery_value.one }
		}
	}
	if = {
		limit = { check_variable = { var = murder_mystery_reinforced_derivatives value = constant:murder_mystery_derivative_support.stretched_count compare = greater_than_or_equals } }
		set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_derivative_support.stretched_cohesion }
		murder_mystery_adjust_brotherhood_cohesion = yes
		clear_variable = murder_mystery_brotherhood_cohesion_delta
	}
}

# COUNTRY scope: a derivative. Input: temp murder_mystery_priority_choice (a
# murder_mystery_derivative_priority constant). Sets its priority toward the
# central state, swaps the matching spirit and starts the cooldown.
murder_mystery_decision_set_derivative_priority = {
	set_variable = { murder_mystery_pending_cost_profile = constant:murder_mystery_cost_profile.local_administration }
	murder_mystery_pay_action_cost = yes
	murder_mystery_clear_action_cost_receipt = yes
	remove_ideas = murder_mystery_priority_spirit_obedience
	remove_ideas = murder_mystery_priority_spirit_autonomy
	remove_ideas = murder_mystery_priority_spirit_local_campaign
	set_variable = { murder_mystery_derivative_priority = murder_mystery_priority_choice }
	if = {
		limit = { check_variable = { var = murder_mystery_priority_choice value = constant:murder_mystery_derivative_priority.obedience compare = equals } }
		add_ideas = murder_mystery_priority_spirit_obedience
		if = {
			limit = { is_subject = yes }
			add_autonomy_ratio = { value = @murder_mystery_priority_autonomy_obedience localization = murder_mystery_priority_obedience_autonomy }
		}
	}
	else_if = {
		limit = { check_variable = { var = murder_mystery_priority_choice value = constant:murder_mystery_derivative_priority.autonomy compare = equals } }
		add_ideas = murder_mystery_priority_spirit_autonomy
		if = {
			limit = { is_subject = yes }
			add_autonomy_ratio = { value = @murder_mystery_priority_autonomy_autonomy localization = murder_mystery_priority_autonomy_autonomy }
		}
	}
	else = {
		add_ideas = murder_mystery_priority_spirit_local_campaign
		if = {
			limit = {
				has_event_target = murder_mystery_assassin_state
				event_target:murder_mystery_assassin_state = { exists = yes }
			}
			event_target:murder_mystery_assassin_state = {
				set_variable = { murder_mystery_brotherhood_cohesion_delta = constant:murder_mystery_derivative_priority.local_campaign_cohesion }
				murder_mystery_adjust_brotherhood_cohesion = yes
				clear_variable = murder_mystery_brotherhood_cohesion_delta
			}
		}
	}
	set_temp_variable = { murder_mystery_timed_flag_days = constant:murder_mystery_derivative_priority.cooldown_days }
	meta_effect = {
		text = {
			set_country_flag = { flag = murder_mystery_derivative_priority_cooldown days = [TIMED_FLAG_DAYS] }
		}
		TIMED_FLAG_DAYS = "[?murder_mystery_timed_flag_days|.0]"
	}
}
