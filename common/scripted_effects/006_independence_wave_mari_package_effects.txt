independence_wave_refresh_mel_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_mari_package = yes }
		remove_ideas = { mel_fragmented_forest_mandate mel_mari_forest_compact }
		if = {
			limit = { has_stable_independence_wave_mel_compact = yes }

# Inputs are temporary `independence_wave_mel_congress_delta` and
# `independence_wave_mel_forest_delta`. Values are clamped to the package
# pressure ladder so repeated focus/decision calls cannot overflow the ledger.
independence_wave_change_mel_compact_values = {
	if = {
		limit = { is_independence_wave_mari_package = yes }
		add_to_variable = { independence_wave_mel_congress_cohesion = independence_wave_mel_congress_delta }
		add_to_variable = { independence_wave_mel_forest_readiness = independence_wave_mel_forest_delta }
		clamp_variable = {
			var = independence_wave_mel_congress_cohesion
			min = constant:independence_wave_mari_pressure.minimum
			max = constant:independence_wave_mari_pressure.maximum
		}
		clamp_variable = {
			var = independence_wave_mel_forest_readiness
			min = constant:independence_wave_mari_pressure.minimum
			max = constant:independence_wave_mari_pressure.maximum
		}
		independence_wave_refresh_mel_compact_lifecycle = yes
		independence_wave_refresh_country_state = yes
	}
}
