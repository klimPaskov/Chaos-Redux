# ============================================================================
# EVENT 016 DIRECTORATE STRATEGIC SINGULARITY CONSTRUCTION TRANSACTIONS
# ============================================================================
# COUNTRY scope. The state-target decisions provide a regular selected-state
# event target plus literal role and quantity. The private site ledger owns
# physical rows; these effects own only exact payment and callback settlement.
# Native civilian_factory_use owns its temporary factory reservation.

brilliant_scientist_mengele_singularity_construction_clear_receipt = {
	clr_country_flag = mengele_event016_singularity_construction_receipt_active
	clear_variable = mengele_event016_singularity_construction_receipt_state
	clear_variable = mengele_event016_singularity_construction_receipt_role
	clear_variable = mengele_event016_singularity_construction_receipt_quantity
	clear_variable = mengele_event016_singularity_construction_receipt_political_power
	clear_variable = mengele_event016_singularity_construction_receipt_support_equipment
	clear_variable = mengele_event016_singularity_construction_receipt_trucks
	clear_variable = mengele_event016_singularity_construction_receipt_fuel
	clear_variable = mengele_event016_singularity_construction_receipt_manpower
}

# Snapshot only an exact paid receipt; malformed receipts are cleared without
# reconstructing a refund from today's constants.
brilliant_scientist_mengele_singularity_construction_snapshot_receipt = {
	set_temp_variable = { mengele_event016_singularity_construction_snapshot_complete = constant:brilliant_scientist_value.one }
	if = { limit = { NOT = { has_variable = mengele_event016_singularity_construction_receipt_political_power } } set_temp_variable = { mengele_event016_singularity_construction_snapshot_complete = constant:brilliant_scientist_value.zero } }
	if = { limit = { NOT = { has_variable = mengele_event016_singularity_construction_receipt_support_equipment } } set_temp_variable = { mengele_event016_singularity_construction_snapshot_complete = constant:brilliant_scientist_value.zero } }
	if = { limit = { NOT = { has_variable = mengele_event016_singularity_construction_receipt_trucks } } set_temp_variable = { mengele_event016_singularity_construction_snapshot_complete = constant:brilliant_scientist_value.zero } }
	if = { limit = { NOT = { has_variable = mengele_event016_singularity_construction_receipt_fuel } } set_temp_variable = { mengele_event016_singularity_construction_snapshot_complete = constant:brilliant_scientist_value.zero } }
	if = { limit = { NOT = { has_variable = mengele_event016_singularity_construction_receipt_manpower } } set_temp_variable = { mengele_event016_singularity_construction_snapshot_complete = constant:brilliant_scientist_value.zero } }
	set_temp_variable = { mengele_event016_singularity_construction_refund_political_power = mengele_event016_singularity_construction_receipt_political_power }
	set_temp_variable = { mengele_event016_singularity_construction_refund_support_equipment = mengele_event016_singularity_construction_receipt_support_equipment }
	set_temp_variable = { mengele_event016_singularity_construction_refund_trucks = mengele_event016_singularity_construction_receipt_trucks }
	set_temp_variable = { mengele_event016_singularity_construction_refund_fuel = mengele_event016_singularity_construction_receipt_fuel }
	set_temp_variable = { mengele_event016_singularity_construction_refund_manpower = mengele_event016_singularity_construction_receipt_manpower }
}

brilliant_scientist_mengele_singularity_construction_refund_snapshot = {
	if = {
		limit = { check_variable = { var = mengele_event016_singularity_construction_snapshot_complete value = constant:brilliant_scientist_value.one compare = equals } }
		add_political_power = mengele_event016_singularity_construction_refund_political_power
		add_equipment_to_stockpile = { type = support_equipment amount = mengele_event016_singularity_construction_refund_support_equipment }
		add_equipment_to_stockpile = { type = motorized_equipment amount = mengele_event016_singularity_construction_refund_trucks }
		add_fuel = mengele_event016_singularity_construction_refund_fuel
		add_manpower = mengele_event016_singularity_construction_refund_manpower
	}
}

# Called only after the physical reservation succeeds and the exact generic
# affordability trigger still passes. The two role branches share the price
# and debit source with Kruger's construction but write private receipts only.
brilliant_scientist_mengele_singularity_construction_record_payment = {
	if = {
		limit = { check_variable = { var = mengele_event016_singularity_construction_expected_role value = constant:brilliant_scientist_singularity_component.command_core compare = equals } }
		set_temp_variable = { mengele_event016_singularity_construction_quote_political_power = constant:brilliant_scientist_project_board.singularity_node_political_power }
		set_temp_variable = { mengele_event016_singularity_construction_quote_support_equipment = constant:brilliant_scientist_project_board.singularity_node_support_equipment }
		set_temp_variable = { mengele_event016_singularity_construction_quote_trucks = constant:brilliant_scientist_project_board.singularity_node_trucks }
		set_temp_variable = { mengele_event016_singularity_construction_quote_fuel = constant:brilliant_scientist_project_board.singularity_node_fuel }
		set_temp_variable = { mengele_event016_singularity_construction_quote_manpower = constant:brilliant_scientist_project_board.singularity_node_manpower }
	}
	else = {
		set_temp_variable = { mengele_event016_singularity_construction_quote_political_power = constant:brilliant_scientist_project_board.singularity_link_political_power }
		set_temp_variable = { mengele_event016_singularity_construction_quote_support_equipment = constant:brilliant_scientist_project_board.singularity_link_support_equipment }
		set_temp_variable = { mengele_event016_singularity_construction_quote_trucks = constant:brilliant_scientist_project_board.singularity_link_trucks }
		set_temp_variable = { mengele_event016_singularity_construction_quote_fuel = constant:brilliant_scientist_project_board.singularity_link_fuel }
		set_temp_variable = { mengele_event016_singularity_construction_quote_manpower = constant:brilliant_scientist_project_board.singularity_link_manpower }
	}
	set_variable = { mengele_event016_singularity_construction_receipt_state = event_target:mengele_event016_singularity_site_selected_state }
	set_variable = { mengele_event016_singularity_construction_receipt_role = mengele_event016_singularity_site_selected_role }
	set_variable = { mengele_event016_singularity_construction_receipt_quantity = mengele_event016_singularity_site_selected_quantity }
	set_variable = { mengele_event016_singularity_construction_receipt_political_power = mengele_event016_singularity_construction_quote_political_power }
	set_variable = { mengele_event016_singularity_construction_receipt_support_equipment = mengele_event016_singularity_construction_quote_support_equipment }
	set_variable = { mengele_event016_singularity_construction_receipt_trucks = mengele_event016_singularity_construction_quote_trucks }
	set_variable = { mengele_event016_singularity_construction_receipt_fuel = mengele_event016_singularity_construction_quote_fuel }
	set_variable = { mengele_event016_singularity_construction_receipt_manpower = mengele_event016_singularity_construction_quote_manpower }
	set_country_flag = mengele_event016_singularity_construction_receipt_active
	if = { limit = { check_variable = { var = mengele_event016_singularity_construction_expected_role value = constant:brilliant_scientist_singularity_component.command_core compare = equals } } brilliant_scientist_pay_singularity_command_node = yes }
	else = { brilliant_scientist_pay_singularity_power_link = yes }
	set_temp_variable = { mengele_event016_singularity_construction_payment_political_power = mengele_event016_singularity_construction_quote_political_power }
	multiply_temp_variable = { mengele_event016_singularity_construction_payment_political_power = constant:brilliant_scientist_directorate_value.negative_one }
	add_political_power = mengele_event016_singularity_construction_payment_political_power
}

brilliant_scientist_mengele_singularity_construction_begin_core = {
	set_temp_variable = { mengele_event016_singularity_construction_begin_applied = constant:brilliant_scientist_value.zero }
	if = {
		limit = {
			check_variable = { var = mengele_event016_singularity_site_selected_role value = mengele_event016_singularity_construction_expected_role compare = equals }
			check_variable = { var = mengele_event016_singularity_site_selected_quantity value = constant:brilliant_scientist_value.one compare = equals }
			OR = {
				AND = { check_variable = { var = mengele_event016_singularity_construction_expected_role value = constant:brilliant_scientist_singularity_component.command_core compare = equals } brilliant_scientist_mengele_singularity_construction_can_begin_command_node_after_native_reservation = yes }
				AND = { check_variable = { var = mengele_event016_singularity_construction_expected_role value = constant:brilliant_scientist_singularity_component.power_link compare = equals } brilliant_scientist_mengele_singularity_construction_can_begin_power_link_after_native_reservation = yes }
			}
		}
		brilliant_scientist_mengele_singularity_site_reserve_construction = yes
		if = {
			limit = { check_variable = { var = mengele_event016_singularity_site_reservation_applied value = constant:brilliant_scientist_value.one compare = equals } }
			if = {
				limit = {
					OR = {
						AND = { check_variable = { var = mengele_event016_singularity_construction_expected_role value = constant:brilliant_scientist_singularity_component.command_core compare = equals } brilliant_scientist_mengele_singularity_construction_can_pay_direct_command_node = yes }
						AND = { check_variable = { var = mengele_event016_singularity_construction_expected_role value = constant:brilliant_scientist_singularity_component.power_link compare = equals } brilliant_scientist_mengele_singularity_construction_can_pay_direct_power_link = yes }
					}
				}
				brilliant_scientist_mengele_singularity_construction_record_payment = yes
				set_temp_variable = { mengele_event016_singularity_construction_begin_applied = constant:brilliant_scientist_value.one }
			}
			else = { brilliant_scientist_mengele_singularity_site_cancel_pending_construction = yes }
		}
	}
}

brilliant_scientist_mengele_singularity_construction_begin_command_node = {
	set_temp_variable = { mengele_event016_singularity_construction_expected_role = constant:brilliant_scientist_singularity_component.command_core }
	brilliant_scientist_mengele_singularity_construction_begin_core = yes
}

brilliant_scientist_mengele_singularity_construction_begin_power_link = {
	set_temp_variable = { mengele_event016_singularity_construction_expected_role = constant:brilliant_scientist_singularity_component.power_link }
	brilliant_scientist_mengele_singularity_construction_begin_core = yes
}

# Clear the direct-payment receipt before touching the physical pending row.
# A matching but invalidated callback refunds exactly once. A mismatched
# state/role callback is a strict no-op and cannot settle another timer.
brilliant_scientist_mengele_singularity_construction_finish_core = {
	set_temp_variable = { mengele_event016_singularity_construction_finish_applied = constant:brilliant_scientist_value.zero }
	if = {
		limit = { brilliant_scientist_mengele_singularity_construction_callback_matches_receipt = yes }
		if = {
			limit = { brilliant_scientist_mengele_singularity_site_pending_callback_is_valid = yes }
			brilliant_scientist_mengele_singularity_construction_snapshot_receipt = yes
			brilliant_scientist_mengele_singularity_construction_clear_receipt = yes
			brilliant_scientist_mengele_singularity_site_complete_pending_construction = yes
			if = { limit = { check_variable = { var = mengele_event016_singularity_site_completion_applied value = constant:brilliant_scientist_value.one compare = equals } } set_temp_variable = { mengele_event016_singularity_construction_finish_applied = constant:brilliant_scientist_value.one } }
			else = { brilliant_scientist_mengele_singularity_site_cancel_pending_construction = yes brilliant_scientist_mengele_singularity_construction_refund_snapshot = yes }
		}
		else = {
			brilliant_scientist_mengele_singularity_construction_snapshot_receipt = yes
			brilliant_scientist_mengele_singularity_construction_clear_receipt = yes
			brilliant_scientist_mengele_singularity_site_cancel_pending_construction = yes
			brilliant_scientist_mengele_singularity_construction_refund_snapshot = yes
		}
	}
}

brilliant_scientist_mengele_singularity_construction_finish_command_node = {
	set_temp_variable = { mengele_event016_singularity_construction_expected_role = constant:brilliant_scientist_singularity_component.command_core }
	brilliant_scientist_mengele_singularity_construction_finish_core = yes
}

brilliant_scientist_mengele_singularity_construction_finish_power_link = {
	set_temp_variable = { mengele_event016_singularity_construction_expected_role = constant:brilliant_scientist_singularity_component.power_link }
	brilliant_scientist_mengele_singularity_construction_finish_core = yes
}
