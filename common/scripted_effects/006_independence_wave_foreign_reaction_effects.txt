# ============================================================================
# CHAOS REDUX - EVENT 006 FOREIGN REACTION EFFECTS
# ============================================================================
# Overview:
# - Build one bounded foreign reaction queue from the committed Event 006
#   cohort after every country opening and host response has resolved. The
#   final host-response callback calls the builder before it clears its own
#   plan receipts, so the report/reset barrier passes to this queue.
# - Actor sources: each target's registered package contacts, its neighbors,
#   its recorded patrons and earlier recognizers, the allies of its living
#   former host, and the active Event 006 countries of earlier waves. Contacts
#   are gathered first so the actor cap never drops them. No world scan and no
#   periodic hook.
# - Score every actor/target pair with the accepted 4/3/2/1 families, one -3
#   conflict penalty and a floor of 0, keep each actor's highest target with
#   ties going to the first committed report row, and reserve one actor-wave
#   receipt for every actor at or above the cutoff of 6.
# - Dispatch hidden router chaosx.nr6.64 one day apart. Humans receive the
#   neighbor .65, regional .66, great-power .67 or earlier-wave .68 profile;
#   AI actors resolve the same stances through deterministic policy.
# - Patron-style offers wait for the released country's consent (.71 for
#   humans, the hidden router for AI) before any influence is booked.
# - Every receipt is consumed exactly once: on an answer, the 13-day timeout,
#   an invalid or re-released target, annexation of the actor, or the queue
#   deadline. The last receipt releases the report/reset barrier.
# Recognition and guarantees write the shared generation-matched bilateral
# receipts, so a foreign act and a country-initiated decision never pay out
# twice for the same pair.
# ============================================================================

# COUNTRY scope (any). Called by `independence_wave_host_response_finish_plan`
# while the committed report barrier is still raised. Sets the global queue
# flag only when at least one actor receipt was reserved.
independence_wave_foreign_reaction_build_queue = {
	if = {
		limit = {
			has_global_flag = independence_wave_country_opening_report_pending
			has_global_flag = liberation_release_plan_includes_event6
			NOT = { has_global_flag = independence_wave_foreign_reaction_queue_pending }
			check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.committed }
			has_variable = global.liberation_plan_id
			has_variable = global.independence_wave_country_opening_report_plan_id
			check_variable = { var = global.independence_wave_country_opening_report_plan_id value = global.liberation_plan_id compare = equals }
			NOT = { check_variable = { var = global.independence_wave_foreign_reaction_built_plan_id value = global.liberation_plan_id compare = equals } }
			NOT = {
				AND = {
					has_global_flag = independence_wave_join_conversion_active
					check_variable = { var = global.independence_wave_join_active_plan_id value = global.liberation_plan_id compare = equals }
				}
			}
			has_variable = global.independence_wave_plan_selected_count
			check_variable = { global.independence_wave_plan_selected_count > constant:independence_wave_value.minimum }
			liberation_release_country_arrays_are_aligned = yes
			independence_wave_plan_metadata_arrays_are_aligned = yes
		}
		set_variable = { global.independence_wave_foreign_reaction_built_plan_id = global.liberation_plan_id }
		clear_temp_array = independence_wave_foreign_cohort_targets
		clear_temp_array = independence_wave_foreign_candidates

		# 1. Freeze the committed Event 006 cohort in public report row order.
		set_temp_variable = { independence_wave_foreign_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_foreign_index value = global.independence_wave_plan_selected_count compare = less_than } }
			set_temp_variable = { independence_wave_foreign_row = global.independence_wave_plan_country_row_indices^independence_wave_foreign_index }
			if = {
				limit = {
					check_variable = { var = independence_wave_foreign_row value = constant:independence_wave_plan.first_array_index compare = greater_than_or_equals }
					check_variable = { var = independence_wave_foreign_row value = global.liberation_plan_countries^num compare = less_than }
					check_variable = { var = global.liberation_plan_country_owners^independence_wave_foreign_row value = constant:liberation_plan_owner.independence_wave compare = equals }
				}
				var:global.liberation_plan_countries^independence_wave_foreign_row = {
					if = {
						limit = {
							is_valid_independence_wave_foreign_reaction_target = yes
							NOT = { is_in_array = { array = independence_wave_foreign_cohort_targets value = THIS } }
						}
						add_to_temp_array = { independence_wave_foreign_cohort_targets = THIS }
					}
				}
			}
			add_to_temp_variable = { independence_wave_foreign_index = constant:independence_wave_plan.loop_increment }
		}

		# 2. Gather the bounded actor sources once.
		set_temp_variable = { independence_wave_foreign_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_foreign_index value = independence_wave_foreign_cohort_targets^num compare = less_than } }
			var:independence_wave_foreign_cohort_targets^independence_wave_foreign_index = {
				for_each_scope_loop = {
					array = independence_wave_package_contact_entries
					independence_wave_foreign_reaction_add_candidate = yes
				}
				every_neighbor_country = { independence_wave_foreign_reaction_add_candidate = yes }
				for_each_scope_loop = {
					array = independence_wave_patron_country_entries
					independence_wave_foreign_reaction_add_candidate = yes
				}
				for_each_scope_loop = {
					array = independence_wave_recognition_receipt_country_entries
					independence_wave_foreign_reaction_add_candidate = yes
				}
				if = {
					limit = { has_independence_wave_living_former_host = yes }
					var:independence_wave_former_host = {
						every_allied_country = { independence_wave_foreign_reaction_add_candidate = yes }
					}
				}
			}
			add_to_temp_variable = { independence_wave_foreign_index = constant:independence_wave_plan.loop_increment }
		}
		for_each_scope_loop = {
			array = global.independence_wave_active_countries
			if = {
				limit = { is_independence_wave_active_country = yes }
				independence_wave_foreign_reaction_add_candidate = yes
			}
		}

		# 3. Score every pair and reserve each qualifying actor's top target.
		set_variable = { global.independence_wave_foreign_reaction_plan_id = global.liberation_plan_id }
		clear_array = global.independence_wave_foreign_reaction_actors
		set_variable = { global.independence_wave_foreign_reactions_remaining = constant:independence_wave_value.minimum }
		set_temp_variable = { independence_wave_foreign_dispatch_delay = constant:independence_wave_foreign_reaction_timing.dispatch_after_host_days }
		set_temp_variable = { independence_wave_foreign_actor_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_foreign_actor_index value = independence_wave_foreign_candidates^num compare = less_than } }
			var:independence_wave_foreign_candidates^independence_wave_foreign_actor_index = {
				save_event_target_as = independence_wave_foreign_actor
				set_temp_variable = { independence_wave_foreign_best_score = constant:independence_wave_foreign_interest.no_score }
				set_temp_variable = { independence_wave_foreign_best_row = constant:independence_wave_plan.invalid_index }
				set_temp_variable = { independence_wave_foreign_best_tier = constant:independence_wave_foreign_tier.none }
				set_temp_variable = { independence_wave_foreign_target_index = constant:independence_wave_plan.first_array_index }
				while_loop_effect = {
					limit = { check_variable = { var = independence_wave_foreign_target_index value = independence_wave_foreign_cohort_targets^num compare = less_than } }
					independence_wave_foreign_reaction_load_cohort_target = yes
					if = {
						limit = { is_valid_independence_wave_foreign_actor_for_target = yes }
						independence_wave_foreign_reaction_score_pair = yes
						# Strictly greater keeps the first committed report row on a tie.
						if = {
							limit = { check_variable = { var = independence_wave_foreign_score value = independence_wave_foreign_best_score compare = greater_than } }
							set_temp_variable = { independence_wave_foreign_best_score = independence_wave_foreign_score }
							set_temp_variable = { independence_wave_foreign_best_row = independence_wave_foreign_target_index }
							set_temp_variable = { independence_wave_foreign_best_tier = independence_wave_foreign_tier_result }
						}
					}
					add_to_temp_variable = { independence_wave_foreign_target_index = constant:independence_wave_plan.loop_increment }
				}
				if = {
					limit = {
						check_variable = { var = independence_wave_foreign_best_score value = constant:independence_wave_foreign_interest.detailed_cutoff compare = greater_than_or_equals }
						check_variable = { var = independence_wave_foreign_best_row value = constant:independence_wave_plan.first_array_index compare = greater_than_or_equals }
					}
					set_temp_variable = { independence_wave_foreign_target_index = independence_wave_foreign_best_row }
					independence_wave_foreign_reaction_load_cohort_target = yes
					independence_wave_foreign_reaction_reserve_actor = yes
				}
			}
			add_to_temp_variable = { independence_wave_foreign_actor_index = constant:independence_wave_plan.loop_increment }
		}

		# 4. Raise the queue and schedule its deadline on every target, so a
		# receipt whose actor vanished without an annexation still closes.
		if = {
			limit = { check_variable = { global.independence_wave_foreign_reactions_remaining > constant:independence_wave_value.minimum } }
			set_global_flag = independence_wave_foreign_reaction_queue_pending
			set_temp_variable = { independence_wave_foreign_deadline_delay = independence_wave_foreign_dispatch_delay }
			add_to_temp_variable = { independence_wave_foreign_deadline_delay = constant:independence_wave_foreign_reaction_timing.response_timeout_days }
			add_to_temp_variable = { independence_wave_foreign_deadline_delay = constant:independence_wave_foreign_reaction_timing.consent_timeout_days }
			# Each target stores the date from which this queue may be closed: the
			# last dispatch plus the response and consent timeouts. The margin days
			# stay between that date and the event, so the event never arrives
			# before the date it checks.
			set_temp_variable = { chaosx_date_days = independence_wave_foreign_deadline_delay }
			set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
			chaosx_convert_days_to_date_offset = yes
			add_to_temp_variable = { independence_wave_foreign_deadline_delay = constant:independence_wave_foreign_reaction_timing.deadline_margin_days }
			set_temp_variable = { independence_wave_foreign_index = constant:independence_wave_plan.first_array_index }
			while_loop_effect = {
				limit = { check_variable = { var = independence_wave_foreign_index value = independence_wave_foreign_cohort_targets^num compare = less_than } }
				var:independence_wave_foreign_cohort_targets^independence_wave_foreign_index = {
					set_variable = { independence_wave_foreign_deadline_plan_id = global.independence_wave_foreign_reaction_plan_id }
					set_variable = { independence_wave_foreign_deadline_date = global.date }
					add_to_variable = { independence_wave_foreign_deadline_date = chaosx_date_offset }
					country_event = { id = chaosx.nr6.72 days = independence_wave_foreign_deadline_delay }
				}
				add_to_temp_variable = { independence_wave_foreign_index = constant:independence_wave_plan.loop_increment }
			}
		}
		else = {
			clear_variable = global.independence_wave_foreign_reaction_plan_id
			clear_variable = global.independence_wave_foreign_reactions_remaining
			clear_array = global.independence_wave_foreign_reaction_actors
		}
	}
}

# COUNTRY scope: a candidate from one of the bounded sources. Living and
# sovereign, outside the committed cohort, not a host answering this wave
# through its own host response, and added once.
independence_wave_foreign_reaction_add_candidate = {
	if = {
		limit = {
			exists = yes
			is_subject = no
			NOT = { is_in_array = { array = independence_wave_foreign_candidates value = THIS } }
			NOT = { is_in_array = { array = independence_wave_foreign_cohort_targets value = THIS } }
			NOT = { is_in_array = { array = global.independence_wave_host_response_hosts value = THIS } }
			check_variable = { var = independence_wave_foreign_candidates^num value = constant:independence_wave_foreign_capacity.actor_cap compare = less_than }
		}
		add_to_temp_array = { independence_wave_foreign_candidates = THIS }
	}
}

# Any scope. Input: temporary `independence_wave_foreign_target_index`.
# Loads the cohort target and its living former host as event targets.
independence_wave_foreign_reaction_load_cohort_target = {
	set_temp_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_value.minimum }
	var:independence_wave_foreign_cohort_targets^independence_wave_foreign_target_index = {
		save_event_target_as = independence_wave_foreign_target
		if = {
			limit = { has_independence_wave_living_former_host = yes }
			var:independence_wave_former_host = { save_event_target_as = independence_wave_foreign_target_host }
			set_temp_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_plan.loop_increment }
		}
	}
}

# COUNTRY scope: the actor. Loads the receipt's target, the stored former
# host, and the actor itself as event targets for scoring and stances.
independence_wave_foreign_reaction_load_receipt_targets = {
	save_event_target_as = independence_wave_foreign_actor
	set_temp_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_value.minimum }
	var:independence_wave_foreign_reaction_target = { save_event_target_as = independence_wave_foreign_target }
	if = {
		limit = {
			has_variable = independence_wave_foreign_reaction_host
			var:independence_wave_foreign_reaction_host = { exists = yes }
		}
		var:independence_wave_foreign_reaction_host = { save_event_target_as = independence_wave_foreign_target_host }
		set_temp_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_plan.loop_increment }
	}
}

# COUNTRY scope: the actor. Inputs: event targets `independence_wave_foreign_actor`
# (this country), `independence_wave_foreign_target`, and the optional host.
# Outputs: temporary `independence_wave_foreign_score` (floor 0) and
# `independence_wave_foreign_tier_result`. Each family contributes once.
independence_wave_foreign_reaction_score_pair = {
	set_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.floor }
	set_temp_variable = { independence_wave_foreign_strategic_hit = constant:independence_wave_value.minimum }
	if = {
		limit = { has_independence_wave_foreign_direct_stake = yes }
		add_to_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.direct_stake }
	}
	if = {
		limit = { has_independence_wave_foreign_recorded_relationship = yes }
		add_to_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.recorded_relationship }
	}
	if = {
		limit = { has_independence_wave_foreign_strategic_fit = yes }
		add_to_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.strategic_fit }
		set_temp_variable = { independence_wave_foreign_strategic_hit = constant:independence_wave_plan.loop_increment }
	}
	if = {
		limit = { has_independence_wave_foreign_political_fit = yes }
		add_to_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.political_fit }
	}
	if = {
		limit = { has_independence_wave_foreign_conflict = yes }
		add_to_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.conflict_penalty }
	}
	if = {
		limit = { check_variable = { var = independence_wave_foreign_score value = constant:independence_wave_foreign_interest.floor compare = less_than } }
		set_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.floor }
	}
	# Profile priority: earlier-wave country, great power with an actual
	# contact, registered regional stakeholder, then local neighbor. Major
	# status alone supplies neither a contact nor the great-power profile.
	if = {
		limit = { is_independence_wave_active_country = yes }
		set_temp_variable = { independence_wave_foreign_tier_result = constant:independence_wave_foreign_tier.earlier_wave }
	}
	else_if = {
		limit = {
			is_major = yes
			has_independence_wave_foreign_recorded_relationship = yes
		}
		set_temp_variable = { independence_wave_foreign_tier_result = constant:independence_wave_foreign_tier.great_power }
	}
	else_if = {
		limit = {
			OR = {
				has_independence_wave_foreign_package_contact = yes
				check_variable = { independence_wave_foreign_strategic_hit = constant:independence_wave_plan.loop_increment }
				NOT = { is_neighbor_of = event_target:independence_wave_foreign_target }
			}
		}
		set_temp_variable = { independence_wave_foreign_tier_result = constant:independence_wave_foreign_tier.regional }
	}
	else = {
		set_temp_variable = { independence_wave_foreign_tier_result = constant:independence_wave_foreign_tier.neighbor }
	}
}

# COUNTRY scope: the actor, with the event targets loaded. Records which
# facts and families produced the live score so the event can name them.
independence_wave_foreign_reaction_record_reasons = {
	independence_wave_foreign_reaction_clear_reasons = yes
	if = {
		limit = { is_neighbor_of = event_target:independence_wave_foreign_target }
		set_country_flag = independence_wave_foreign_fact_frontier
	}
	if = {
		limit = {
			event_target:independence_wave_foreign_target = {
				any_owned_state = { is_claimed_by = event_target:independence_wave_foreign_actor }
			}
		}
		set_country_flag = independence_wave_foreign_fact_claim
	}
	if = {
		limit = { has_guaranteed = event_target:independence_wave_foreign_target }
		set_country_flag = independence_wave_foreign_fact_guarantee
	}
	if = {
		limit = { has_independence_wave_foreign_direct_stake = yes }
		set_country_flag = independence_wave_foreign_family_direct
	}
	if = {
		limit = {
			event_target:independence_wave_foreign_target = {
				is_in_array = { array = independence_wave_patron_country_entries value = PREV }
			}
		}
		set_country_flag = independence_wave_foreign_fact_patron
	}
	if = {
		limit = { event_target:independence_wave_foreign_target = { has_independence_wave_recognition_receipt_from_prev = yes } }
		set_country_flag = independence_wave_foreign_fact_prior_recognition
	}
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_country_flag = independence_wave_network_member
		}
		set_country_flag = independence_wave_foreign_fact_network
	}
	if = {
		limit = { is_ally_with = event_target:independence_wave_foreign_target }
		set_country_flag = independence_wave_foreign_fact_alliance
	}
	if = {
		limit = { has_independence_wave_foreign_package_contact = yes }
		set_country_flag = independence_wave_foreign_fact_contact
	}
	if = {
		limit = {
			check_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_plan.loop_increment }
			is_ally_with = event_target:independence_wave_foreign_target_host
		}
		set_country_flag = independence_wave_foreign_fact_host_ally
	}
	if = {
		limit = { has_independence_wave_foreign_recorded_relationship = yes }
		set_country_flag = independence_wave_foreign_family_relationship
	}
	if = {
		limit = {
			OR = {
				has_military_access_to = event_target:independence_wave_foreign_target
				event_target:independence_wave_foreign_target = { has_military_access_to = PREV }
			}
		}
		set_country_flag = independence_wave_foreign_fact_access
	}
	if = {
		limit = {
			check_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_plan.loop_increment }
			is_neighbor_of = event_target:independence_wave_foreign_target_host
			NOT = { is_neighbor_of = event_target:independence_wave_foreign_target }
		}
		set_country_flag = independence_wave_foreign_fact_balance
	}
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_variable = independence_wave_region
			event_target:independence_wave_foreign_target = {
				has_variable = independence_wave_region
				check_variable = { var = independence_wave_region value = PREV.independence_wave_region compare = equals }
			}
		}
		set_country_flag = independence_wave_foreign_fact_region
	}
	if = {
		limit = { has_independence_wave_foreign_strategic_fit = yes }
		set_country_flag = independence_wave_foreign_family_strategic
	}
	if = {
		limit = { has_independence_wave_foreign_political_fit = yes }
		set_country_flag = independence_wave_foreign_fact_ideology
		set_country_flag = independence_wave_foreign_family_political
	}
	if = {
		limit = { has_war = yes }
		set_country_flag = independence_wave_foreign_fact_war_burden
	}
	else_if = {
		limit = { has_independence_wave_foreign_conflict = yes }
		set_country_flag = independence_wave_foreign_fact_settled
	}
}

# COUNTRY scope: the actor, with the chosen target loaded. Reserves the one
# actor-wave receipt and schedules its router one day after the previous one.
independence_wave_foreign_reaction_reserve_actor = {
	independence_wave_foreign_reaction_clear_actor_receipt = yes
	set_country_flag = independence_wave_foreign_reaction_pending
	set_variable = { independence_wave_foreign_reaction_plan_id = global.independence_wave_foreign_reaction_plan_id }
	set_variable = { independence_wave_foreign_reaction_target = event_target:independence_wave_foreign_target }
	event_target:independence_wave_foreign_target = {
		set_temp_variable = { independence_wave_foreign_target_generation = independence_wave_generation_id }
	}
	set_variable = { independence_wave_foreign_reaction_target_generation = independence_wave_foreign_target_generation }
	set_variable = { independence_wave_foreign_reaction_tier = independence_wave_foreign_best_tier }
	set_variable = { independence_wave_foreign_reaction_score = independence_wave_foreign_best_score }
	if = {
		limit = { check_variable = { independence_wave_foreign_target_has_host = constant:independence_wave_plan.loop_increment } }
		set_variable = { independence_wave_foreign_reaction_host = event_target:independence_wave_foreign_target_host }
	}
	add_to_array = { array = global.independence_wave_foreign_reaction_actors value = THIS }
	add_to_variable = { global.independence_wave_foreign_reactions_remaining = constant:independence_wave_plan.loop_increment }
	country_event = { id = chaosx.nr6.64 days = independence_wave_foreign_dispatch_delay }
	add_to_temp_variable = { independence_wave_foreign_dispatch_delay = constant:independence_wave_foreign_reaction_timing.dispatch_spacing_days }
}

# COUNTRY scope: the actor in the hidden router (ROOT is the actor). Re-checks
# the receipt and the live pair score, then sends a human the profile event or
# resolves an AI actor through the same stances.
independence_wave_foreign_reaction_route_actor = {
	set_country_flag = independence_wave_foreign_reaction_routed
	set_temp_variable = { independence_wave_foreign_score = constant:independence_wave_foreign_interest.no_score }
	if = {
		limit = { has_current_independence_wave_foreign_reaction_receipt = yes }
		independence_wave_foreign_reaction_load_receipt_targets = yes
		if = {
			limit = { is_valid_independence_wave_foreign_actor_for_target = yes }
			independence_wave_foreign_reaction_score_pair = yes
		}
	}
	if = {
		limit = { check_variable = { var = independence_wave_foreign_score value = constant:independence_wave_foreign_interest.detailed_cutoff compare = greater_than_or_equals } }
		set_variable = { independence_wave_foreign_reaction_tier = independence_wave_foreign_tier_result }
		set_variable = { independence_wave_foreign_reaction_score = independence_wave_foreign_score }
		independence_wave_foreign_reaction_record_reasons = yes
		if = {
			limit = { is_ai = yes }
			independence_wave_foreign_reaction_resolve_ai_action = yes
			independence_wave_foreign_reaction_resolve_choice = yes
		}
		else_if = {
			limit = { check_variable = { independence_wave_foreign_reaction_tier = constant:independence_wave_foreign_tier.neighbor } }
			country_event = { id = chaosx.nr6.65 }
		}
		else_if = {
			limit = { check_variable = { independence_wave_foreign_reaction_tier = constant:independence_wave_foreign_tier.regional } }
			country_event = { id = chaosx.nr6.66 }
		}
		else_if = {
			limit = { check_variable = { independence_wave_foreign_reaction_tier = constant:independence_wave_foreign_tier.great_power } }
			country_event = { id = chaosx.nr6.67 }
		}
		else_if = {
			limit = { check_variable = { independence_wave_foreign_reaction_tier = constant:independence_wave_foreign_tier.earlier_wave } }
			country_event = { id = chaosx.nr6.68 }
		}
		else = { independence_wave_foreign_reaction_complete_current_actor = yes }
	}
	else = { independence_wave_foreign_reaction_complete_current_actor = yes }
}

# COUNTRY scope: the actor. Consumes the actor-wave receipt exactly once. A
# receipt waiting for a consent reply is consumed by that reply instead.

# COUNTRY scope: a released country. Closes its open offer once and lets the
# waiting actor's receipt complete, whether the offer was answered,
# invalidated, or timed out.
independence_wave_foreign_reaction_close_offer = {
	if = {
		limit = {
			has_variable = independence_wave_foreign_offer_country
			var:independence_wave_foreign_offer_country = {
				has_country_flag = independence_wave_foreign_reaction_awaiting_consent

			independence_wave_foreign_reaction_complete_current_actor = yes
		}

	independence_wave_foreign_reaction_clear_offer = yes
}

# COUNTRY scope (any). Clears every receipt of the current queue, then
# releases the report/reset barrier that the host phase handed to the queue.
independence_wave_foreign_reaction_finish_queue = {
	if = {
		limit = { has_global_flag = independence_wave_foreign_reaction_queue_pending }
		set_temp_variable = { independence_wave_foreign_finishing_plan_id = global.independence_wave_foreign_reaction_plan_id }
		for_each_scope_loop = {
			array = global.independence_wave_foreign_reaction_actors
			if = {
				limit = {
					has_variable = independence_wave_foreign_reaction_plan_id
					check_variable = { var = independence_wave_foreign_reaction_plan_id value = independence_wave_foreign_finishing_plan_id compare = equals }
				}
				if = {
					limit = { has_variable = independence_wave_foreign_reaction_offer_target }
					var:independence_wave_foreign_reaction_offer_target = {
						if = {
							limit = {
								has_variable = independence_wave_foreign_offer_plan_id
								check_variable = { var = independence_wave_foreign_offer_plan_id value = independence_wave_foreign_finishing_plan_id compare = equals }
							}
							independence_wave_foreign_reaction_clear_offer = yes
						}
					}
				}
				independence_wave_foreign_reaction_clear_actor_receipt = yes
			}
		}
		clear_array = global.independence_wave_foreign_reaction_actors
		clear_variable = global.independence_wave_foreign_reactions_remaining
		clear_variable = global.independence_wave_foreign_reaction_plan_id
		clr_global_flag = independence_wave_foreign_reaction_queue_pending
		if = {
			limit = {
				has_variable = global.independence_wave_country_opening_report_plan_id
				check_variable = { var = global.independence_wave_country_opening_report_plan_id value = independence_wave_foreign_finishing_plan_id compare = equals }
			}
			independence_wave_host_response_release_report_barrier = yes
		}
	}
}

# COUNTRY scope: a target holding a scheduled deadline. Closes the queue that
# scheduled it if any receipt is still open; later queues are untouched.
independence_wave_foreign_reaction_check_deadline = {
	# A deadline event from an earlier queue can arrive after a newer queue
	# rescheduled this target. It acts only once the stored date of the current
	# schedule has passed, so it never closes the newer queue early and never
	# clears the newer schedule's receipt.
	if = {
		limit = {
			has_variable = independence_wave_foreign_deadline_plan_id
			has_variable = independence_wave_foreign_deadline_date
			check_variable = { var = global.date value = independence_wave_foreign_deadline_date compare = greater_than_or_equals }
		}
		if = {
			limit = {
				has_global_flag = independence_wave_foreign_reaction_queue_pending
				has_variable = global.independence_wave_foreign_reaction_plan_id
				check_variable = { var = independence_wave_foreign_deadline_plan_id value = global.independence_wave_foreign_reaction_plan_id compare = equals }
			}
			independence_wave_foreign_reaction_finish_queue = yes
		}
		clear_variable = independence_wave_foreign_deadline_plan_id
		clear_variable = independence_wave_foreign_deadline_date
	}
	# A schedule without a stored date keeps the plan-id check alone.
	else_if = {
		limit = {
			has_variable = independence_wave_foreign_deadline_plan_id
			NOT = { has_variable = independence_wave_foreign_deadline_date }
		}
		if = {
			limit = {
				has_global_flag = independence_wave_foreign_reaction_queue_pending
				has_variable = global.independence_wave_foreign_reaction_plan_id
				check_variable = { var = independence_wave_foreign_deadline_plan_id value = global.independence_wave_foreign_reaction_plan_id compare = equals }
			}
			independence_wave_foreign_reaction_finish_queue = yes
		}
		clear_variable = independence_wave_foreign_deadline_plan_id
	}
}

# COUNTRY scope: the annexed country in `on_annex`. An actor's receipt is
# consumed and a target's open offer is closed before the country disappears.
independence_wave_foreign_reaction_release_annexed_country = {

		limit = { has_country_flag = independence_wave_foreign_offer_pending }
		independence_wave_foreign_reaction_close_offer = yes
	}
	if = {

			limit = { has_variable = independence_wave_foreign_reaction_offer_target }

		}
		clr_country_flag = independence_wave_foreign_reaction_awaiting_consent
		independence_wave_foreign_reaction_complete_current_actor = yes
	}
}

			}

			limit = {
				OR = {
					has_country_flag = independence_wave_foreign_fact_ideology
					has_country_flag = independence_wave_foreign_fact_alliance
				}
				can_independence_wave_foreign_recognize = yes
			}
			set_temp_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.recognize }
		}

			limit = {
				has_war = no

				can_independence_wave_foreign_mediate = yes
			}
			set_temp_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.mediate }

		}

			set_temp_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.mediate }

# Input: temporary `independence_wave_foreign_action_input`. A stance whose
# requirements no longer hold when it is applied resolves as noncommittal.
independence_wave_foreign_reaction_apply_action = {
	if = {
		limit = { has_current_independence_wave_foreign_reaction_receipt = yes }
		independence_wave_foreign_reaction_load_receipt_targets = yes

			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.recognize }
				can_independence_wave_foreign_recognize = yes
			}
			independence_wave_foreign_reaction_apply_recognition = yes
		}
		else_if = {
			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.recognize_with_aid }
				can_independence_wave_foreign_recognize_with_aid = yes
			}
			event_target:independence_wave_foreign_target = {
				# The actor (ROOT) ships from its own stockpile to this country.
				ROOT = {
					send_equipment = {
						equipment = infantry_equipment
						amount = constant:independence_wave_decision_cost.infantry_light
						target = PREV
					}
				}
				set_temp_variable = { independence_wave_value_delta = constant:independence_wave_decision_effect.value_minor }
				independence_wave_change_security = yes
			}
			independence_wave_foreign_reaction_apply_recognition = yes
		}
		else_if = {
			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.share_institutions }
				can_independence_wave_foreign_share_institutions = yes
			}
			event_target:independence_wave_foreign_target = {
				set_temp_variable = { independence_wave_value_delta = constant:independence_wave_decision_effect.value_minor }
				independence_wave_change_capacity = yes
				set_temp_variable = { independence_wave_network_delta = constant:independence_wave_decision_effect.network_standard }
				independence_wave_change_network_standing = yes
			}
			set_temp_variable = { independence_wave_network_delta = constant:independence_wave_decision_effect.network_standard }
			independence_wave_change_network_standing = yes

			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.mediate }
				can_independence_wave_foreign_mediate = yes
			}
			independence_wave_foreign_reaction_apply_mediation = yes
		}
		else_if = {
			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.contain }
				OR = {
					AND = {
						check_variable = { independence_wave_foreign_reaction_tier = constant:independence_wave_foreign_tier.great_power }
						can_independence_wave_foreign_contain_standard = yes
					}
					AND = {
						NOT = { check_variable = { independence_wave_foreign_reaction_tier = constant:independence_wave_foreign_tier.great_power } }

					}
				}
			}
			independence_wave_foreign_reaction_apply_containment = yes
		}
		else_if = {
			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.guarantee }
				can_independence_wave_foreign_guarantee = yes
			}
			independence_wave_foreign_reaction_apply_guarantee = yes
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.invest }
						can_independence_wave_foreign_invest = yes
					}
					AND = {
						check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.arms_offer }
						can_independence_wave_foreign_offer_arms = yes
					}
				}
			}
			independence_wave_foreign_reaction_make_offer = yes
		}
		else_if = {
			limit = {
				check_variable = { independence_wave_foreign_action_input = constant:independence_wave_foreign_action.network_aid }
				can_independence_wave_foreign_network_aid = yes
			}
			independence_wave_foreign_reaction_apply_network_aid = yes

				can_independence_wave_foreign_contest = yes
			}
			independence_wave_foreign_reaction_apply_contest = yes
		}
	}

					add_opinion_modifier = { target = event_target:independence_wave_foreign_actor modifier = independence_wave_foreign_recognized_breakaway }
				}
			}
			set_temp_variable = { independence_wave_foreign_history_payload = constant:independence_wave_foreign_history.recognition }
			event_target:independence_wave_foreign_actor = { independence_wave_foreign_reaction_record_history = yes }

		id = event_target:independence_wave_foreign_target
		value = @CR_SC_INDEPENDENCE_WAVE_FOREIGN_AI_STRATEGY_STANDARD
	}

# A remnant former host that chose legal continuity or a regional charter may,
# within 30 days, ask the highest-interest government around it for a
# guarantee. The petition is paid once per country identity, reaches one
# recipient, and never protects a released country.

# COUNTRY scope: the host, from its remnant host response (ROOT is the host).
# Consumes the outside-guarantee opportunity and opens the window when the
# host has not used its petition yet and a qualifying recipient exists.
independence_wave_remnant_petition_open_window = {
	clr_country_flag = independence_wave_host_response_outside_guarantee_opportunity_open
	clr_country_flag = independence_wave_remnant_petition_window
	clear_variable = independence_wave_remnant_petition_candidate
	if = {
		limit = {
			exists = yes
			NOT = { has_country_flag = independence_wave_remnant_petition_used }
			has_variable = independence_wave_host_response_plan_id
		}
		set_variable = { independence_wave_remnant_petition_plan_id = independence_wave_host_response_plan_id }
		independence_wave_remnant_petition_select_candidate = yes
		if = {
			limit = { has_variable = independence_wave_remnant_petition_candidate }
			set_temp_variable = { independence_wave_remnant_petition_window_days = constant:independence_wave_remnant_petition.window_days }
			set_country_flag = {
				flag = independence_wave_remnant_petition_window
				days = independence_wave_remnant_petition_window_days
			}
		}
	}
}

# COUNTRY scope: the host. Ranks the neighbors and allies of the host with the
# accepted interest score and stores the first highest-scoring recipient at or
# above the detailed cutoff.
independence_wave_remnant_petition_select_candidate = {
	save_event_target_as = independence_wave_petition_host
	set_temp_variable = { independence_wave_petition_wave_input = independence_wave_remnant_petition_plan_id }
	set_temp_variable = { independence_wave_petition_best_score = constant:independence_wave_foreign_interest.no_score }
	every_neighbor_country = { independence_wave_remnant_petition_consider_recipient = yes }
	every_allied_country = { independence_wave_remnant_petition_consider_recipient = yes }
	if = {
		limit = { check_variable = { var = independence_wave_petition_best_score value = constant:independence_wave_foreign_interest.detailed_cutoff compare = greater_than_or_equals } }
		set_variable = { independence_wave_remnant_petition_candidate = event_target:independence_wave_petition_best_recipient }
	}
}

# COUNTRY scope: a candidate recipient. Strictly greater keeps the first
# candidate on a tie (neighbors before allies).
independence_wave_remnant_petition_consider_recipient = {
	if = {
		limit = { is_valid_independence_wave_remnant_petition_recipient = yes }
		save_event_target_as = independence_wave_petition_recipient
		set_temp_variable = { independence_wave_petition_score = constant:independence_wave_foreign_interest.floor }
		if = {
			limit = { is_neighbor_of = event_target:independence_wave_petition_host }
			add_to_temp_variable = { independence_wave_petition_score = constant:independence_wave_foreign_interest.direct_stake }
		}
		if = {
			limit = { is_ally_with = event_target:independence_wave_petition_host }
			add_to_temp_variable = { independence_wave_petition_score = constant:independence_wave_foreign_interest.recorded_relationship }
		}
		if = {
			limit = { has_independence_wave_petition_strategic_fit = yes }
			add_to_temp_variable = { independence_wave_petition_score = constant:independence_wave_foreign_interest.strategic_fit }
		}
		if = {
			limit = { has_independence_wave_petition_political_fit = yes }
			add_to_temp_variable = { independence_wave_petition_score = constant:independence_wave_foreign_interest.political_fit }
		}
		if = {
			limit = { has_independence_wave_petition_conflict = yes }
			add_to_temp_variable = { independence_wave_petition_score = constant:independence_wave_foreign_interest.conflict_penalty }
		}
		if = {
			limit = { check_variable = { var = independence_wave_petition_score value = independence_wave_petition_best_score compare = greater_than } }
			set_temp_variable = { independence_wave_petition_best_score = independence_wave_petition_score }
			save_event_target_as = independence_wave_petition_best_recipient
		}
	}
}

# COUNTRY scope: the host taking the petition decision (ROOT is the host).
# The decision pays the standard diplomatic cost. Uses the host's one
# petition and opens the recipient's answer window.
independence_wave_remnant_petition_submit = {
	if = {
		limit = {
			has_open_independence_wave_remnant_petition_window = yes
			is_valid_independence_wave_remnant_petition_candidate_for_root = yes
		}
		set_country_flag = independence_wave_remnant_petition_used
		clr_country_flag = independence_wave_remnant_petition_window
		set_country_flag = independence_wave_remnant_petition_pending
		clear_variable = independence_wave_remnant_petition_outcome
		set_variable = { independence_wave_remnant_petition_recipient = independence_wave_remnant_petition_candidate }
		set_temp_variable = { independence_wave_petition_wave_input = independence_wave_remnant_petition_plan_id }
		var:independence_wave_remnant_petition_recipient = {
			set_country_flag = independence_wave_remnant_petition_received
			set_variable = { independence_wave_remnant_petition_from = ROOT }
			set_variable = { independence_wave_remnant_petition_received_plan_id = independence_wave_petition_wave_input }
			if = {
				limit = { is_ai = yes }
				country_event = { id = chaosx.nr6.64 }
			}
			else = { country_event = { id = chaosx.nr6.69 } }
		}
	}
}

# COUNTRY scope: the recipient (ROOT is the recipient). Input: temporary
# `independence_wave_remnant_petition_reply_input`. A guarantee needs the
# acceptance requirements to still hold; anything else gives no protection.
# The host is told the result once.
independence_wave_remnant_petition_resolve = {
	set_temp_variable = { independence_wave_remnant_petition_result = constant:independence_wave_remnant_petition_outcome.lapsed }
	if = {
		limit = { has_current_independence_wave_remnant_petition = yes }
		var:independence_wave_remnant_petition_from = { save_event_target_as = independence_wave_petition_host }
		if = {
			limit = {
				check_variable = { independence_wave_remnant_petition_reply_input = constant:independence_wave_remnant_petition_reply.accept }
				can_independence_wave_remnant_petition_accept = yes
			}
			independence_wave_decision_pay_diplomatic_standard = yes
			event_target:independence_wave_petition_host = {
				ROOT = { give_guarantee = PREV }
				# The host's transit concession takes effect only with the guarantee.
				give_military_access = ROOT
				add_opinion_modifier = { target = ROOT modifier = independence_wave_remnant_petition_guarantor }
			}
			add_ai_strategy = {
				type = protect
				id = event_target:independence_wave_petition_host
				value = @CR_SC_INDEPENDENCE_WAVE_FOREIGN_AI_STRATEGY_STANDARD
			}
			set_temp_variable = { independence_wave_remnant_petition_result = constant:independence_wave_remnant_petition_outcome.accepted }
			independence_wave_remnant_petition_record_history = yes
		}
		else_if = {
			limit = { check_variable = { independence_wave_remnant_petition_reply_input = constant:independence_wave_remnant_petition_reply.refuse } }
			event_target:independence_wave_petition_host = {
				add_opinion_modifier = { target = ROOT modifier = independence_wave_remnant_petition_refused }
			}
			set_temp_variable = { independence_wave_remnant_petition_result = constant:independence_wave_remnant_petition_outcome.refused }
		}
	}
	independence_wave_remnant_petition_notify_host = yes
}

# COUNTRY scope: the recipient. Closes its side of the petition and tells a
# living host that is still waiting for this recipient. Input: temporary
# `independence_wave_remnant_petition_result`.
independence_wave_remnant_petition_notify_host = {
	if = {
		limit = {
			has_variable = independence_wave_remnant_petition_from
			var:independence_wave_remnant_petition_from = {
				exists = yes
				has_country_flag = independence_wave_remnant_petition_pending
				has_variable = independence_wave_remnant_petition_recipient
				check_variable = { var = independence_wave_remnant_petition_recipient value = PREV.id compare = equals }
			}
		}
		var:independence_wave_remnant_petition_from = {
			clr_country_flag = independence_wave_remnant_petition_pending
			set_variable = { independence_wave_remnant_petition_outcome = independence_wave_remnant_petition_result }
			country_event = { id = chaosx.nr6.70 }
		}
	}
	clr_country_flag = independence_wave_remnant_petition_received
	clear_variable = independence_wave_remnant_petition_from
}

# COUNTRY scope: an AI recipient answering through the hidden router (ROOT is
# the recipient). Guarantees at peace for an ally or a government of the same
# outlook when it can; otherwise refuses.
independence_wave_remnant_petition_resolve_ai = {
	set_temp_variable = { independence_wave_remnant_petition_reply_input = constant:independence_wave_remnant_petition_reply.refuse }
	if = {
		limit = { has_current_independence_wave_remnant_petition = yes }
		save_event_target_as = independence_wave_petition_recipient
		var:independence_wave_remnant_petition_from = { save_event_target_as = independence_wave_petition_host }
		if = {
			limit = {
				has_war = no
				can_independence_wave_remnant_petition_accept = yes
				OR = {
					is_ally_with = event_target:independence_wave_petition_host
					has_independence_wave_petition_political_fit = yes
				}
			}
			set_temp_variable = { independence_wave_remnant_petition_reply_input = constant:independence_wave_remnant_petition_reply.accept }
		}
	}
	independence_wave_remnant_petition_resolve = yes
}
