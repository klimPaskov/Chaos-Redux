# COUNTRY scope on the current host. Uses the same weighted recipient model as
# the opening referral, but permits AI and persists the recipient across the
# visible decision window. The global target is always cleared at resolution.
brilliant_scientist_select_containment_exile_recipient = {
	if = {
		limit = { has_event_target = brilliant_scientist_containment_exile_recipient }
		clear_global_event_target = brilliant_scientist_containment_exile_recipient
	}
	clear_temp_array = brilliant_scientist_send_away_selection_pool
	set_temp_variable = { brilliant_scientist_send_away_pool_count = constant:brilliant_scientist_value.zero }
	save_event_target_as = brilliant_scientist_send_away_rejector
	every_country = {
		limit = {
			brilliant_scientist_is_valid_transfer_recipient = yes
			NOT = { has_war_with = event_target:brilliant_scientist_send_away_rejector }
		}
		brilliant_scientist_add_current_country_to_send_away_pool = yes
	}
	if = {
		limit = { check_variable = { brilliant_scientist_send_away_pool_count > constant:brilliant_scientist_value.zero } }
		random_scope_in_array = {
			array = brilliant_scientist_send_away_selection_pool
			save_global_event_target_as = brilliant_scientist_containment_exile_recipient
		}
	}
	clear_temp_array = brilliant_scientist_send_away_selection_pool
	set_temp_variable = { brilliant_scientist_send_away_pool_count = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_send_away_weight_remaining = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_send_away_selection_weight = constant:brilliant_scientist_value.zero }
}

		}

		}

			limit = { check_variable = { brilliant_scientist_containment_action = constant:brilliant_scientist_containment_action.exile } }
			brilliant_scientist_resolve_clean_exile = yes
		}

			brilliant_scientist_resolve_institutional_takeover = yes
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { brilliant_scientist_containment_action = constant:brilliant_scientist_containment_action.arrest }
					check_variable = { brilliant_scientist_containment_action = constant:brilliant_scientist_containment_action.shutdown }

			brilliant_scientist_resolve_coercive_containment = yes
		}
		else = { brilliant_scientist_reopen_invalidated_sovereignty_board = yes }
	}
}
