# ============================================================================
# CHAOS REDUX - EVENT 040 LAWRENCE OF ARABIA EVOLUTION EFFECTS
# ============================================================================
# Evolution pacing and activation (active entry through the pulse, evolved
# opening before the first target), Evolution I (the Arab Revolt Network:
# cells, incidents, breaking the cells, funding, government replacement) and
# Evolution II (the British Arabian System: regional nodes, six institutions,
# the favoured partner, client autonomy and the counter-bloc).
#
# Evolution state is global: global.lawrence_evolution (0 to 3). An
# evolution gives no Chaos itself. Each stage records one Event Log entry
# through record_events_log_evolution_entry and only after
# is_current_evolution_enabled passes, so a disabled stage never sets a
# recorded flag or unlocks content. Stage 3 is added by the federation
# phase.
#
# Scope: COUNTRY. The pulse hooks run in the live target; institutions,
# funding and replacement are British actions in ENG.
# ============================================================================

# ---------------------------------------------------------------------------
# Evolution pacing and activation
# ---------------------------------------------------------------------------

# COUNTRY, the live target. Called by the weekly pulse. MTTH pacing: a stage
# that is ready and enabled waits about 90 days, then activates.
lawrence_try_evolution = {
	if = {
		limit = {
			lawrence_is_live_target = yes
			check_variable = { var = global.lawrence_evolution value = constant:lawrence_evolution.highest_stage compare = less_than }
		}
		set_temp_variable = { lawrence_next_stage = global.lawrence_evolution }
		add_to_temp_variable = { lawrence_next_stage = constant:lawrence_event.one }
		set_temp_variable = { events_log_evolution_event_id = constant:lawrence_event.id }
		set_temp_variable = { events_log_evolution_type = constant:lawrence_event.evolution_type }
		set_temp_variable = { events_log_evolution_stage = lawrence_next_stage }
		set_temp_variable = { events_log_evolution_tier = lawrence_next_stage }
		if = {
			limit = {
				is_current_evolution_enabled = yes
				lawrence_evolution_stage_ready = yes
			}
			if = {
				limit = { NOT = { has_variable = global.lawrence_evolution_due_day } }
				set_temp_variable = { lawrence_evolution_delay = mtth:lawrence_evolution_interval }
				set_variable = { global.lawrence_evolution_due_day = global.num_days }
				add_to_variable = { global.lawrence_evolution_due_day = lawrence_evolution_delay }
			}
			else_if = {
				limit = { check_variable = { var = global.num_days value = global.lawrence_evolution_due_day compare = greater_than_or_equals } }
				clear_variable = global.lawrence_evolution_due_day
				lawrence_activate_evolution = yes
			}
		}
		else = {
			clear_variable = global.lawrence_evolution_due_day
		}
	}
}

# COUNTRY, the live target. Inputs: temporary lawrence_next_stage and the
# shared evolution context (event id, type, stage, tier).
lawrence_activate_evolution = {
	set_variable = { global.lawrence_evolution = lawrence_next_stage }
	set_temp_variable = { events_log_evolution_has_actor = constant:lawrence_event.one }
	save_event_target_as = events_log_evolution_actor
	record_events_log_evolution_entry = yes
	lawrence_apply_evolution_entry = yes
	set_temp_variable = { lawrence_report_delay = constant:lawrence_event.report_delay_days }
	if = {
		limit = { check_variable = { var = global.lawrence_evolution value = constant:lawrence_evolution.stage_1 compare = equals } }
		country_event = { id = chaosx.nr40.96 days = lawrence_report_delay }
		ENG = { country_event = { id = chaosx.nr40.96 days = lawrence_report_delay } }
	}
	else_if = {
		limit = { check_variable = { var = global.lawrence_evolution value = constant:lawrence_evolution.stage_2 compare = equals } }
		ENG = { country_event = { id = chaosx.nr40.97 days = lawrence_report_delay } }
	}
}

# COUNTRY, the live target. What a freshly active stage unlocks in the
# running campaign.
lawrence_apply_evolution_entry = {
	if = {
		limit = { check_variable = { var = global.lawrence_evolution value = constant:lawrence_evolution.stage_1 compare = equals } }
		lawrence_seed_cells = yes
		set_temp_variable = { lawrence_cell_delay = constant:lawrence_cells.first_incident_days }
		set_variable = { lawrence_next_cell_day = global.num_days }
		add_to_variable = { lawrence_next_cell_day = lawrence_cell_delay }
	}
	else_if = {
		limit = { check_variable = { var = global.lawrence_evolution value = constant:lawrence_evolution.stage_2 compare = equals } }
		ENG = { lawrence_collect_nodes = yes }
	}
}

# COUNTRY, scoped to the first target. Evolved opening: every enabled stage
# whose Chaos tier is reached is already active at the first intervention.
# Each stage records its own log entry. The opening stays contestable: it
# raises starting reach and adds one immediate response, never a coup or a
# civil war.
lawrence_resolve_opening_evolution = {
	set_variable = { global.lawrence_evolution = constant:lawrence_event.zero }
	set_temp_variable = { events_log_evolution_event_id = constant:lawrence_event.id }
	set_temp_variable = { events_log_evolution_type = constant:lawrence_event.evolution_type }
	set_temp_variable = { events_log_evolution_stage = constant:lawrence_evolution.stage_1 }
	set_temp_variable = { events_log_evolution_tier = constant:lawrence_evolution.stage_1 }
	if = {
		limit = { is_current_evolution_enabled = yes }
		set_variable = { global.lawrence_evolution = constant:lawrence_evolution.stage_1 }
		set_temp_variable = { events_log_evolution_has_actor = constant:lawrence_event.one }
		save_event_target_as = events_log_evolution_actor
		record_events_log_evolution_entry = yes
		lawrence_apply_opening_evolution_1 = yes
		# Evolution II recognises only relationships that already exist.
		ENG = { lawrence_collect_nodes = yes }
		set_temp_variable = { events_log_evolution_event_id = constant:lawrence_event.id }
		set_temp_variable = { events_log_evolution_type = constant:lawrence_event.evolution_type }
		set_temp_variable = { events_log_evolution_stage = constant:lawrence_evolution.stage_2 }
		set_temp_variable = { events_log_evolution_tier = constant:lawrence_evolution.stage_2 }
		if = {
			limit = {
				is_current_evolution_enabled = yes
				check_variable = { var = global.lawrence_node_count value = constant:lawrence_system.minimum_nodes compare = greater_than_or_equals }
			}
			set_variable = { global.lawrence_evolution = constant:lawrence_evolution.stage_2 }
			set_temp_variable = { events_log_evolution_has_actor = constant:lawrence_event.one }
			save_event_target_as = events_log_evolution_actor
			record_events_log_evolution_entry = yes
			lawrence_apply_opening_evolution_2 = yes
		}
	}
	lawrence_resolve_opening_evolution_extra = yes
}

# COUNTRY, the live target. Starts the hidden cell network from the access
# that already exists.
lawrence_seed_cells = {
	if = {
		limit = { NOT = { has_variable = lawrence_cells } }
		set_variable = { lawrence_cells = constant:lawrence_cells.start }
		if = {
			limit = { has_country_flag = lawrence_delivery_pending }
			add_to_variable = { lawrence_cells = constant:lawrence_cells.start_shipment }
		}
		if = {
			limit = { check_variable = { var = lawrence_officers value = constant:lawrence_officers.capture_threshold compare = greater_than_or_equals } }
			add_to_variable = { lawrence_cells = constant:lawrence_cells.start_officers }
		}
		if = {
			limit = { check_variable = { var = lawrence_security value = constant:lawrence_cells.security_dampening compare = greater_than_or_equals } }
			subtract_from_variable = { lawrence_cells = constant:lawrence_cells.start_security }
		}
		clamp_variable = { var = lawrence_cells min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
	}
}

# COUNTRY, the live target. Weekly: cells grow with Influence, British
# funding and officers and shrink with security, secured nodes and exposure.
lawrence_track_cells = {
	if = {
		limit = {
			lawrence_cells_are_active = yes
			lawrence_status_is_free = yes
		}
		set_temp_variable = { lawrence_cells_delta = lawrence_influence }
		multiply_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.influence_factor }
		if = {
			limit = { has_country_flag = lawrence_cells_funded }
			add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.funded }
		}
		if = {
			limit = { check_variable = { var = lawrence_officers value = constant:lawrence_officers.capture_threshold compare = greater_than_or_equals } }
			add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.officers }
		}
		set_temp_variable = { lawrence_cells_security = lawrence_security }
		multiply_temp_variable = { lawrence_cells_security = constant:lawrence_cells.security_factor }
		subtract_from_temp_variable = { lawrence_cells_delta = lawrence_cells_security }
		if = {
			limit = { has_country_flag = lawrence_nodes_secured }
			add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.nodes_secured }
		}
		if = {
			limit = { has_country_flag = lawrence_exposed }
			add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.exposed }
		}
		if = {
			limit = { has_country_flag = lawrence_file }
			add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.file }
		}
		if = {
			limit = { has_active_mission = lawrence_break_cells_mission lawrence_all_nodes_garrisoned = yes }
			add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.mission_rate }
			if = {
				limit = { has_country_flag = lawrence_file }
				add_to_temp_variable = { lawrence_cells_delta = constant:lawrence_cells.mission_file_bonus }
			}
		}
		clamp_temp_variable = {
			var = lawrence_cells_delta
			min = constant:lawrence_cells.weekly_minimum
			max = constant:lawrence_cells.weekly_maximum
		}
		add_to_variable = { lawrence_cells = lawrence_cells_delta }
		clamp_variable = { var = lawrence_cells min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
		lawrence_track_cell_incidents = yes
	}
}

# COUNTRY, the live target. One incident per MTTH interval once the cells
# are strong enough. The family follows geography and current access.
lawrence_track_cell_incidents = {
	if = {
		limit = {
			NOT = { has_country_flag = lawrence_cell_incident_open }
			NOT = { has_country_flag = lawrence_final_pending }
			lawrence_is_live_target_contest = yes
			check_variable = { var = lawrence_cells value = constant:lawrence_cells.incident_floor compare = greater_than_or_equals }
			check_variable = { var = global.num_days value = lawrence_next_cell_day compare = greater_than_or_equals }
		}
		set_temp_variable = { lawrence_cell_delay = mtth:lawrence_cell_incident_interval }
		set_variable = { lawrence_next_cell_day = global.num_days }
		add_to_variable = { lawrence_next_cell_day = lawrence_cell_delay }
		set_country_flag = lawrence_cell_incident_open
		lawrence_record_revolt_chaos = yes
		random_list = {
			20 = {
				modifier = { add = 25 lawrence_target_is_coastal = yes }
				modifier = { add = 15 has_country_flag = lawrence_delivery_pending }
				modifier = { factor = 0.4 has_country_flag = lawrence_route_sea has_country_flag = lawrence_nodes_secured }
				country_event = { id = chaosx.nr40.90 }
			}
			20 = {
				modifier = { add = 30 has_variable = lawrence_node_route }
				modifier = { factor = 0.4 NOT = { has_variable = lawrence_node_route } }
				modifier = { factor = 0.5 has_country_flag = lawrence_nodes_secured }
				country_event = { id = chaosx.nr40.91 }
			}
			20 = {
				modifier = { add = 30 check_variable = { var = lawrence_officers value = constant:lawrence_cells.officer_incident_floor compare = greater_than_or_equals } }
				modifier = { factor = 0.3 check_variable = { var = lawrence_officers value = constant:lawrence_officers.incident_floor compare = less_than } }
				country_event = { id = chaosx.nr40.92 }
			}
			20 = {
				modifier = { add = 15 NOT = { has_country_flag = lawrence_exposed } }
				country_event = { id = chaosx.nr40.93 }
			}
			20 = {
				modifier = { add = 25 has_country_flag = lawrence_press_active }
				modifier = { add = 10 check_variable = { var = lawrence_influence value = constant:lawrence_influence.contested_edge compare = greater_than_or_equals } }
				country_event = { id = chaosx.nr40.94 }
			}
		}
	}
}
