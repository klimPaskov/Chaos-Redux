	clr_country_flag = brilliant_scientist_context_briefing_pending
	clr_country_flag = brilliant_scientist_context_assistant_conflict_pending
	clr_country_flag = brilliant_scientist_impossible_lecture_pending

# COUNTRY scope. Event 025 can resolve before or after Kruger is appointed. The
# permanent Antarctic idea is the evidence input; this helper only creates the
# one-time Event 016 warning receipt for an active carrier.
brilliant_scientist_try_schedule_alien_artifact_contact = {
	if = {
		limit = {
			brilliant_scientist_is_current_host = yes
			has_idea = antarctica_success
			NOT = { has_global_flag = brilliant_scientist_event_resolved }
			NOT = { has_country_flag = brilliant_scientist_alien_artifact_contact }
			NOT = { has_country_flag = brilliant_scientist_alien_artifact_contact_report_seen }
		}
		set_country_flag = brilliant_scientist_alien_artifact_contact
		set_country_flag = brilliant_scientist_alien_artifact_contact_pending
		country_event = {
			id = chaosx.nr16.17
			days = constant:brilliant_scientist_directorate_timing.alien_artifact_contact_report_days
		}
	}
}

# COUNTRY scope. The Evolution I report is scheduled only after both context
# choices have resolved. The pending flag gives transfer and sovereignty code a
# durable obligation to carry instead of creating a second report path.
brilliant_scientist_try_schedule_impossible_lecture = {
	if = {
		limit = {
			OR = {
				brilliant_scientist_is_current_host = yes
				brilliant_scientist_is_active_kruger_state = yes
			}
			brilliant_scientist_is_evolution_i_enabled = yes
			has_global_flag = brilliant_scientist_evolution_i_chronology_recorded
			NOT = { has_global_flag = world_end }
			has_country_flag = brilliant_scientist_context_briefing_resolved
			has_country_flag = brilliant_scientist_assistant_conflict_resolved
			NOT = { has_country_flag = brilliant_scientist_impossible_lecture_pending }
			NOT = { has_country_flag = brilliant_scientist_impossible_lecture_resolved }
		}
		set_country_flag = brilliant_scientist_impossible_lecture_pending
		country_event = { id = chaosx.nr16.12 days = constant:brilliant_scientist_directorate_timing.impossible_lecture_days }
	}
}

# COUNTRY scope. The report opens the shared computation family at Theory at
# most once. Its four presentation variants move the existing hidden causal
# values through a centralized policy vector and leave a permanent character
# receipt for future continuity checks.
brilliant_scientist_resolve_impossible_lecture = {
	if = {
		limit = {
			OR = {
				brilliant_scientist_is_current_host = yes
				brilliant_scientist_is_active_kruger_state = yes
			}
			brilliant_scientist_is_evolution_i_enabled = yes
			has_global_flag = brilliant_scientist_evolution_i_chronology_recorded
			NOT = { has_global_flag = world_end }
			has_country_flag = brilliant_scientist_impossible_lecture_pending
			NOT = { has_country_flag = brilliant_scientist_impossible_lecture_resolved }
		}
		set_country_flag = brilliant_scientist_impossible_lecture_resolved
		clr_country_flag = brilliant_scientist_impossible_lecture_pending
		KRG_warren_kruger = { set_character_flag = brilliant_scientist_personal_impossible_lecture_witnessed }

		set_temp_variable = { brilliant_scientist_policy_mandate = constant:brilliant_scientist_value.zero }
		set_temp_variable = { brilliant_scientist_policy_dependence = constant:brilliant_scientist_value.zero }

			limit = { has_country_flag = brilliant_scientist_impossible_lecture_public }
			set_temp_variable = { brilliant_scientist_policy_mandate = constant:brilliant_scientist_impossible_lecture.public_mandate }
			set_temp_variable = { brilliant_scientist_policy_exposure = constant:brilliant_scientist_impossible_lecture.public_exposure }
			set_temp_variable = { brilliant_scientist_policy_capacity = constant:brilliant_scientist_impossible_lecture.public_capacity }

			set_temp_variable = { brilliant_scientist_policy_grievance = constant:brilliant_scientist_impossible_lecture.public_grievance }
		}
		else_if = {
			limit = { has_country_flag = brilliant_scientist_impossible_lecture_military }
			set_temp_variable = { brilliant_scientist_policy_mandate = constant:brilliant_scientist_impossible_lecture.military_mandate }
			set_temp_variable = { brilliant_scientist_policy_dependence = constant:brilliant_scientist_impossible_lecture.military_dependence }
			set_temp_variable = { brilliant_scientist_policy_exposure = constant:brilliant_scientist_impossible_lecture.military_exposure }
			set_temp_variable = { brilliant_scientist_policy_capacity = constant:brilliant_scientist_impossible_lecture.military_capacity }
			set_temp_variable = { brilliant_scientist_policy_grievance = constant:brilliant_scientist_impossible_lecture.military_grievance }
		}
		else_if = {
			limit = { has_country_flag = brilliant_scientist_impossible_lecture_industrial }
			set_temp_variable = { brilliant_scientist_policy_mandate = constant:brilliant_scientist_impossible_lecture.industrial_mandate }
			set_temp_variable = { brilliant_scientist_policy_dependence = constant:brilliant_scientist_impossible_lecture.industrial_dependence }
			set_temp_variable = { brilliant_scientist_policy_exposure = constant:brilliant_scientist_impossible_lecture.industrial_exposure }
			set_temp_variable = { brilliant_scientist_policy_capacity = constant:brilliant_scientist_impossible_lecture.industrial_capacity }
			set_temp_variable = { brilliant_scientist_policy_independent_capacity = constant:brilliant_scientist_impossible_lecture.industrial_independent_capacity }
			set_temp_variable = { brilliant_scientist_policy_grievance = constant:brilliant_scientist_impossible_lecture.industrial_grievance }
		}
		else = {
			set_temp_variable = { brilliant_scientist_policy_mandate = constant:brilliant_scientist_impossible_lecture.interrupted_mandate }
			set_temp_variable = { brilliant_scientist_policy_dependence = constant:brilliant_scientist_impossible_lecture.interrupted_dependence }
			set_temp_variable = { brilliant_scientist_policy_exposure = constant:brilliant_scientist_impossible_lecture.interrupted_exposure }
			set_temp_variable = { brilliant_scientist_policy_capacity = constant:brilliant_scientist_impossible_lecture.interrupted_capacity }
			set_temp_variable = { brilliant_scientist_policy_independent_capacity = constant:brilliant_scientist_impossible_lecture.interrupted_independent_capacity }
			set_temp_variable = { brilliant_scientist_policy_grievance = constant:brilliant_scientist_impossible_lecture.interrupted_grievance }
		}
		brilliant_scientist_apply_evolution_policy_vector = yes

		set_temp_variable = { brilliant_scientist_project_family = constant:brilliant_scientist_project_family.computation }
		set_temp_variable = { brilliant_scientist_requested_project_stage = constant:brilliant_scientist_project_stage.theory }
