# - Initializes Chaos Warfare as a conditional national CBRN institution.
# - Records track mastery without periodic polling and grants doctrine-only
#   technology only after every mapped prerequisite is present.

		limit = {

cbrn_apply_theater_decontamination_assignment_paid = {
	cbrn_refresh_state_contamination_class = yes
	set_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_doctrine_cleanup.trace_or_local_points }
	if = {
		limit = {
			check_variable = {
				var = cbrn_state_contamination_class_output
				value = constant:cbrn_contamination_class.catastrophic
				compare = equals
			}
		}
		set_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_doctrine_cleanup.catastrophic_points }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cbrn_state_contamination_class_output
				value = constant:cbrn_contamination_class.severe
				compare = equals
			}
		}
		set_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_doctrine_cleanup.severe_points }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cbrn_state_contamination_class_output
				value = constant:cbrn_contamination_class.serious
				compare = equals
			}
		}
		set_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_doctrine_cleanup.serious_points }
	}

	if = {
		limit = { CONTROLLER = { has_idea = cbrn_theater_contamination_doctrine_spirit } }
		multiply_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_doctrine_spirit.theater_contamination.cleanup_output_mult }
	}
	if = {
		limit = {
			CONTROLLER = { cbrn_designer_has_low_water_process = yes }
			OR = {
				infrastructure < 4
				is_desert_state = yes
			}
		}
		multiply_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_decontamination_designer_effect.low_water_cleanup_mult }
	}
	if = {
		limit = { CONTROLLER = { cbrn_designer_has_rapid_route_clearance = yes } }
		multiply_temp_variable = { cbrn_doctrine_cleanup_points = constant:cbrn_decontamination_designer_effect.rapid_route_cleanup_mult }
	}

	set_temp_variable = { cbrn_state_contamination_delta = cbrn_doctrine_cleanup_points }
	multiply_temp_variable = { cbrn_state_contamination_delta = constant:cbrn_doctrine_value.negation_multiplier }
	cbrn_apply_state_contamination_delta_internal = yes

	set_temp_variable = { cbrn_doctrine_cleanup_completed = cbrn_state_previous_contamination_value }
	subtract_from_temp_variable = { cbrn_doctrine_cleanup_completed = cbrn_state_new_contamination_value }
	CONTROLLER = {
		add_to_variable = { cbrn_doctrine_cleanup_points_total = cbrn_doctrine_cleanup_completed }
	}

	set_state_flag = {
		flag = cbrn_hq_decontamination_corridor_assigned
		days = @CBRN_DOCTRINE_THEATER_CLEANUP_ASSIGNMENT_DAYS
	}
}
