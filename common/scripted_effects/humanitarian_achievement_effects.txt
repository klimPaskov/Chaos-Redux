# ============================================================================
# CHAOS REDUX - HUMANITARIAN ACHIEVEMENT EVIDENCE
# ============================================================================
# Persistent evidence hooks for the eight cross-mechanic achievements. These
# helpers observe completed food, route, transfer, reception, and policy
# transactions; they never move population, apply mortality, or create a
# second crisis meter.
# ============================================================================

# Country scope. The startup registration calls this once for every existing
# country before either mechanic's runtime can mutate state population.
# It is intentionally write-once: later achievement interactions never
# refresh a baseline that could have been shifted by deaths or movement.
humanitarian_achievement_seed_initial_core_baseline = {
	if = {
		limit = {
			humanitarian_country_is_valid = yes
			NOT = { has_country_flag = humanitarian_achievement_tracking_initialized }
		}
		set_country_flag = humanitarian_achievement_tracking_initialized
		set_variable = { humanitarian_achievement_initial_core_population = constant:humanitarian_achievement_value.zero }
		every_core_state = {
			set_temp_variable = { humanitarian_achievement_state_population_people = state_population_k }
			multiply_temp_variable = { humanitarian_achievement_state_population_people = constant:humanitarian_achievement_value.people_per_k }
			add_to_variable = { PREV.humanitarian_achievement_initial_core_population = humanitarian_achievement_state_population_people }
		}
		set_variable = { humanitarian_achievement_integration_population_target = humanitarian_achievement_initial_core_population }
		multiply_variable = { humanitarian_achievement_integration_population_target = constant:humanitarian_achievement_threshold.integration_host_population_share }
		set_variable = { humanitarian_achievement_national_threatened_target = humanitarian_achievement_initial_core_population }
		multiply_variable = { humanitarian_achievement_national_threatened_target = constant:humanitarian_achievement_threshold.national_threatened_share }
		set_variable = { humanitarian_achievement_distinct_origin_count = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_severe_core_state_count = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_recovered_core_state_count = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_threatened_population = constant:humanitarian_achievement_value.zero }
		clear_array = humanitarian_achievement_origin_country_ids
		clear_array = humanitarian_achievement_gate_protected_cohort_ids
	}
}

humanitarian_achievement_initialize_country = {
	if = {
		limit = { NOT = { has_country_flag = humanitarian_achievement_protected_cohort_evidence_initialized } }
		set_country_flag = humanitarian_achievement_protected_cohort_evidence_initialized
		clear_array = humanitarian_achievement_protected_internment_cohort_ids
		clear_array = humanitarian_achievement_protected_forced_labor_cohort_ids
		clear_array = humanitarian_achievement_integration_forced_return_cohort_ids
		clear_array = humanitarian_achievement_integration_violent_pushback_cohort_ids
	}
	if = {
		limit = { NOT = { has_country_flag = humanitarian_achievement_identity_generation_contract_initialized } }
		set_country_flag = humanitarian_achievement_identity_generation_contract_initialized
		set_variable = { humanitarian_achievement_generation_counter_seen = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_gate_generation = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_gate_valid_people = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_gate_protected_people = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_gate_resolution_count = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_blockade_generation = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_extraction_generation = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_receiving_state_count = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_receiving_state_safe_count = constant:humanitarian_achievement_value.zero }
		set_variable = { humanitarian_achievement_threat_generation = constant:humanitarian_achievement_value.zero }
		clear_array = humanitarian_achievement_gate_qualifying_cohort_ids
		clear_array = humanitarian_achievement_gate_qualifying_cohort_amounts
		clear_array = humanitarian_achievement_gate_protected_cohort_ids
		clear_array = humanitarian_achievement_gate_protected_cohort_amounts
		clear_array = humanitarian_achievement_gate_resolved_cohort_ids
		clear_array = humanitarian_achievement_gate_terminal_qualifying_cohort_ids
		clear_array = humanitarian_achievement_gate_terminal_qualifying_cohort_amounts
		clear_array = humanitarian_achievement_gate_terminal_protected_cohort_ids
		clear_array = humanitarian_achievement_gate_terminal_protected_cohort_amounts
		clear_array = humanitarian_achievement_gate_terminal_resolved_cohort_ids
		clear_array = humanitarian_achievement_severe_core_state_ids
		clear_array = humanitarian_achievement_recovered_core_state_ids
		clear_array = humanitarian_achievement_receiving_state_ids
		clear_array = humanitarian_achievement_receiving_state_generations
		clear_array = humanitarian_achievement_receiving_integration_state_ids
		clear_array = humanitarian_achievement_receiving_integration_cohort_ids
		clear_array = humanitarian_achievement_receiving_state_safe_ids
		clr_country_flag = humanitarian_achievement_gate_major_crisis
		clr_country_flag = humanitarian_achievement_gate_all_protected
		clr_country_flag = humanitarian_achievement_gate_durable_outcome
		clr_country_flag = humanitarian_achievement_gate_reception_safe
		clr_country_flag = humanitarian_achievement_blockade_relief_restored
		clr_country_flag = humanitarian_achievement_blockade_recovered
		clr_country_flag = humanitarian_achievement_blockade_survival_met
		clr_country_flag = humanitarian_achievement_extraction_shock_proven
		clr_country_flag = humanitarian_achievement_extraction_suspended_before_catastrophic
		clr_country_flag = humanitarian_achievement_extraction_state_recovered
		clr_country_flag = humanitarian_achievement_alternative_logistics_proven
		clr_country_flag = humanitarian_achievement_receiving_states_food_safe
		clr_country_flag = humanitarian_achievement_all_famine_states_recovered
		clr_country_flag = humanitarian_achievement_all_major_cohorts_resolved
		clr_country_flag = humanitarian_achievement_national_threatened_share_met
	}
}

# GLOBAL scope through the normal country/state caller. One monotonic nonce is
# shared by every achievement identity producer; the exact state/country/cohort
# fields remain the authority for matching a transaction to that nonce.
humanitarian_achievement_advance_generation = {
	if = {
		limit = { NOT = { has_variable = global.humanitarian_achievement_generation_counter } }
		set_variable = { global.humanitarian_achievement_generation_counter = constant:humanitarian_achievement_value.zero }
	}
	add_to_variable = { global.humanitarian_achievement_generation_counter = constant:humanitarian_achievement_value.one }
}

# COUNTRY scope entered from the exact blocked state. This binds the earning
# ledger to the eligible human owner/controller before relief or recovery can
# be credited; a later unrelated state cannot reuse the receipt.
humanitarian_achievement_record_blockade_start_actor = {
	if = {
		limit = {
			humanitarian_achievement_possible_campaign_is_valid = yes
			PREV = {
				has_state_flag = humanitarian_achievement_blockade_state
				check_variable = { var = humanitarian_achievement_blockade_generation value = constant:humanitarian_achievement_value.zero compare = greater_than }
			}
		}
		humanitarian_achievement_initialize_country = yes
		clr_country_flag = humanitarian_achievement_blockade_relief_restored
		clr_country_flag = humanitarian_achievement_blockade_relief_donor_proven
		clr_country_flag = humanitarian_achievement_blockade_recovered
		clr_country_flag = humanitarian_achievement_blockade_recovery_proven
		clr_country_flag = humanitarian_achievement_blockade_survival_met
		clr_country_flag = humanitarian_achievement_blockade_wasteland
		clr_country_flag = humanitarian_achievement_blockade_forced_removal
		clr_country_flag = humanitarian_achievement_blockade_predatory_requisition
		clr_country_flag = humanitarian_achievement_blockade_self_route_exploit
		set_variable = { humanitarian_achievement_blockade_state_id = PREV.id }
		set_variable = { humanitarian_achievement_blockade_generation = PREV.humanitarian_achievement_blockade_generation }
		set_variable = { PREV.humanitarian_achievement_blockade_earning_country_id = id }
		set_country_flag = humanitarian_achievement_blockade_famine_proven
	}
}

# STATE scope. Called after the food stage transition and mortality transaction.
# The evidence pass is state-local and keeps famine floors, blockade identity,
# and threat generations separate from migration route ledgers.
humanitarian_achievement_process_state = {
	if = {
		limit = { OWNER = { exists = yes } }
		OWNER = { humanitarian_achievement_initialize_country = yes }
	}
	if = {
		limit = {
			OWNER = {
				has_country_flag = humanitarian_achievement_threat_window_active
				NOT = { has_war = yes }
			}
		}
		OWNER = { set_country_flag = humanitarian_achievement_threat_window_closed }
	}
	if = {
		limit = { OWNER = { exists = yes } }
		OWNER = { humanitarian_achievement_refresh_receiving_state_food_safety = yes }
	}
	if = {
		limit = {
			has_state_flag = humanitarian_achievement_blockade_state
			has_state_category = wasteland
			OWNER = { exists = yes }
		}
		OWNER = { humanitarian_achievement_record_blockade_wasteland = yes }
	}
	if = {
		limit = {
			has_state_flag = famine_food_security_active
			has_state_flag = famine_extraction_active
			OR = {
				has_state_flag = famine_food_stage_acute_shortage
				has_state_flag = famine_food_stage_famine
			}
			check_variable = { famine_component_extraction > constant:famine_food_component.zero }
			OR = {
				check_variable = { famine_component_production > constant:famine_food_component.zero }
				check_variable = { famine_component_transport > constant:famine_food_component.zero }
			}
			check_variable = { famine_extraction_generation > constant:humanitarian_achievement_value.zero }
			NOT = { has_state_flag = humanitarian_achievement_extraction_shock_recorded }
		}
		set_state_flag = humanitarian_achievement_extraction_shock_recorded
		OWNER = { humanitarian_achievement_record_extraction_shock = yes }
	}
	if = {
		limit = {
			has_state_flag = humanitarian_achievement_extraction_shock_recorded
			has_state_flag = famine_extraction_suspended
			check_variable = { famine_food_stage = constant:famine_food_stage.stable }
			OWNER = { has_country_flag = humanitarian_achievement_extraction_suspended_before_catastrophic }
		}
		OWNER = { humanitarian_achievement_record_extraction_recovery = yes }
	}
	if = {
		limit = {
			OR = {
				has_state_flag = famine_food_stage_famine
				has_state_flag = famine_food_stage_catastrophic
			}
			NOT = { has_state_flag = humanitarian_achievement_severe_recorded }
		}
		set_state_flag = humanitarian_achievement_severe_recorded
		set_variable = { humanitarian_achievement_crisis_start_population = state_population_k }
		multiply_variable = { humanitarian_achievement_crisis_start_population = constant:humanitarian_achievement_value.people_per_k }
		set_variable = { humanitarian_achievement_protected_floor = humanitarian_achievement_crisis_start_population }
		multiply_variable = { humanitarian_achievement_protected_floor = constant:humanitarian_achievement_threshold.protected_recovery_floor_share }
		set_temp_variable = { humanitarian_achievement_threatened_amount = migration_trapped_population }
		if = {
			limit = { NOT = { check_variable = { humanitarian_achievement_threatened_amount > constant:humanitarian_achievement_value.zero } } }
			set_temp_variable = { humanitarian_achievement_threatened_amount = migration_flight_pressure }
		}
		if = {
			limit = {
				is_core_of = OWNER
				OWNER = { has_war = yes }
				has_state_flag = migration_displacement_active
				check_variable = { humanitarian_achievement_threatened_amount > constant:humanitarian_achievement_value.zero }
			}
			set_state_flag = humanitarian_achievement_severe_qualifying
			OWNER = {
				if = {
					limit = { has_country_flag = humanitarian_achievement_all_famine_states_recovered }
					clear_array = humanitarian_achievement_severe_core_state_ids
					clear_array = humanitarian_achievement_recovered_core_state_ids
					set_variable = { humanitarian_achievement_severe_core_state_count = constant:humanitarian_achievement_value.zero }
					set_variable = { humanitarian_achievement_recovered_core_state_count = constant:humanitarian_achievement_value.zero }
					set_variable = { humanitarian_achievement_threatened_population = constant:humanitarian_achievement_value.zero }
					clr_country_flag = humanitarian_achievement_all_famine_states_recovered
					clr_country_flag = humanitarian_achievement_national_threatened_share_met
					clr_country_flag = humanitarian_achievement_threatened_war_proven
					clr_country_flag = humanitarian_achievement_threatened_displacement_proven
					humanitarian_achievement_advance_generation = yes
					set_variable = { humanitarian_achievement_threat_generation = global.humanitarian_achievement_generation_counter }
				}
				if = {
					limit = { has_country_flag = humanitarian_achievement_threat_window_closed }
					clear_array = humanitarian_achievement_severe_core_state_ids
					clear_array = humanitarian_achievement_recovered_core_state_ids
					set_variable = { humanitarian_achievement_severe_core_state_count = constant:humanitarian_achievement_value.zero }
					set_variable = { humanitarian_achievement_recovered_core_state_count = constant:humanitarian_achievement_value.zero }
					set_variable = { humanitarian_achievement_threatened_population = constant:humanitarian_achievement_value.zero }
					clr_country_flag = humanitarian_achievement_all_famine_states_recovered
					clr_country_flag = humanitarian_achievement_national_threatened_share_met
					clr_country_flag = humanitarian_achievement_threat_window_closed
					humanitarian_achievement_advance_generation = yes
					set_variable = { humanitarian_achievement_threat_generation = global.humanitarian_achievement_generation_counter }
				}
				if = {
					limit = { NOT = { check_variable = { humanitarian_achievement_threat_generation > constant:humanitarian_achievement_value.zero } } }
					humanitarian_achievement_advance_generation = yes
					set_variable = { humanitarian_achievement_threat_generation = global.humanitarian_achievement_generation_counter }
				}
				set_country_flag = humanitarian_achievement_threat_window_active
				set_variable = { PREV.humanitarian_achievement_threat_generation = humanitarian_achievement_threat_generation }
				if = {
					limit = {
						NOT = {
							is_in_array = {
								array = humanitarian_achievement_severe_core_state_ids
								value = PREV.id
							}
						}
					}
					add_to_array = { array = humanitarian_achievement_severe_core_state_ids value = PREV.id }
					set_variable = { humanitarian_achievement_severe_core_state_count = humanitarian_achievement_severe_core_state_ids^num }
					add_to_variable = { humanitarian_achievement_threatened_population = humanitarian_achievement_threatened_amount }
				}
				clr_country_flag = humanitarian_achievement_all_famine_states_recovered
				set_country_flag = humanitarian_achievement_threatened_war_proven
				set_country_flag = humanitarian_achievement_threatened_displacement_proven
				if = {
					limit = {
						check_variable = {
							var = humanitarian_achievement_threatened_population
							value = humanitarian_achievement_national_threatened_target
							compare = greater_than_or_equals
						}
					}
					set_country_flag = humanitarian_achievement_national_threatened_share_met
				}
			}
		}
	}
	# Break the Blockade begins in the exact famine/catastrophic blockade state.
	# The earning actor is the eligible human owner/controller at start; all
	# later relief and recovery receipts must match this state generation.
	if = {
		limit = {
			famine_blockade_proof = yes
			OR = {
				has_state_flag = famine_food_stage_famine
				has_state_flag = famine_food_stage_catastrophic
			}
			NOT = { has_state_flag = humanitarian_achievement_blockade_state }
			OR = {
				CONTROLLER = { humanitarian_achievement_possible_campaign_is_valid = yes }
				OWNER = { humanitarian_achievement_possible_campaign_is_valid = yes }
			}
		}
		set_state_flag = humanitarian_achievement_blockade_state
		humanitarian_achievement_advance_generation = yes
		set_variable = { humanitarian_achievement_blockade_generation = global.humanitarian_achievement_generation_counter }
		set_variable = { humanitarian_achievement_blockade_start_population = state_population_k }
		multiply_variable = { humanitarian_achievement_blockade_start_population = constant:humanitarian_achievement_value.people_per_k }
		clear_variable = humanitarian_achievement_blockade_earning_country_id
		if = {
			limit = { CONTROLLER = { humanitarian_achievement_possible_campaign_is_valid = yes } }
			CONTROLLER = { humanitarian_achievement_record_blockade_start_actor = yes }
		}
		if = {
			limit = {
				NOT = { has_variable = humanitarian_achievement_blockade_earning_country_id }
				OWNER = { humanitarian_achievement_possible_campaign_is_valid = yes }
			}
			OWNER = { humanitarian_achievement_record_blockade_start_actor = yes }
		}
		if = {
			limit = { NOT = { has_variable = humanitarian_achievement_blockade_earning_country_id } }
			clr_state_flag = humanitarian_achievement_blockade_state
			clear_variable = humanitarian_achievement_blockade_generation
			clear_variable = humanitarian_achievement_blockade_start_population
		}
	}
	if = {
		limit = {
			has_state_flag = humanitarian_achievement_severe_recorded
			check_variable = { famine_food_stage = constant:famine_food_stage.stable }
			NOT = { has_state_flag = humanitarian_achievement_recovery_recorded }
		}
		set_state_flag = humanitarian_achievement_recovery_recorded
		if = {
			limit = {
				is_core_of = OWNER
				has_state_flag = humanitarian_achievement_severe_qualifying
				check_variable = { humanitarian_achievement_threat_generation > constant:humanitarian_achievement_value.zero }
			}
			OWNER = {
				if = {
					limit = { check_variable = { var = PREV.humanitarian_achievement_threat_generation value = humanitarian_achievement_threat_generation compare = equals } }
					if = {
						limit = {
							NOT = {
								is_in_array = {
									array = humanitarian_achievement_recovered_core_state_ids
									value = PREV.id
								}
							}
						}
						add_to_array = { array = humanitarian_achievement_recovered_core_state_ids value = PREV.id }
						set_variable = { humanitarian_achievement_recovered_core_state_count = humanitarian_achievement_recovered_core_state_ids^num }
					}
					if = {
						limit = {
							check_variable = { var = humanitarian_achievement_recovered_core_state_count value = humanitarian_achievement_severe_core_state_count compare = equals }
							check_variable = { var = humanitarian_achievement_severe_core_state_ids^num value = humanitarian_achievement_severe_core_state_count compare = equals }
							check_variable = { var = humanitarian_achievement_recovered_core_state_ids^num value = humanitarian_achievement_recovered_core_state_count compare = equals }
							check_variable = { var = humanitarian_achievement_severe_core_state_count value = constant:humanitarian_achievement_value.zero compare = greater_than }
						}
						set_country_flag = humanitarian_achievement_all_famine_states_recovered
					}
				}
			}
		}
	}
	if = {
		limit = {
			has_state_flag = humanitarian_achievement_blockade_state
			OWNER = { exists = yes }
		}
		set_temp_variable = { humanitarian_achievement_blockade_survivors = state_population_k }
		multiply_temp_variable = { humanitarian_achievement_blockade_survivors = constant:humanitarian_achievement_value.people_per_k }
		set_temp_variable = { humanitarian_achievement_blockade_survival_target = humanitarian_achievement_blockade_start_population }
		multiply_temp_variable = { humanitarian_achievement_blockade_survival_target = constant:humanitarian_achievement_threshold.blockade_survival_share }
		if = {
			limit = {
				CONTROLLER = {
					check_variable = { var = id value = PREV.humanitarian_achievement_blockade_earning_country_id compare = equals }
				}
			}
			CONTROLLER = { humanitarian_achievement_record_blockade_recovery = yes }
		}
		else_if = {
			limit = {
				OWNER = {
					check_variable = { var = id value = PREV.humanitarian_achievement_blockade_earning_country_id compare = equals }
				}
			}
			OWNER = { humanitarian_achievement_record_blockade_recovery = yes }
		}
	}
	# Sample the protected floor on every active severe-generation pulse, before
	# any recovery cleanup can clear the state-local severe markers.
	if = {
		limit = {
			has_state_flag = humanitarian_achievement_severe_recorded
			check_variable = { humanitarian_achievement_protected_floor > constant:humanitarian_achievement_value.zero }
		}
		set_temp_variable = { humanitarian_achievement_current_population_people = state_population_k }
		multiply_temp_variable = { humanitarian_achievement_current_population_people = constant:humanitarian_achievement_value.people_per_k }
		if = {
			limit = {
				check_variable = {
					var = humanitarian_achievement_current_population_people
					value = humanitarian_achievement_protected_floor
					compare = less_than
				}
			}
			OWNER = { set_country_flag = humanitarian_achievement_core_floor_breached }
		}
	}
	if = {
		limit = {
			has_state_flag = humanitarian_achievement_severe_recorded
			has_state_flag = humanitarian_achievement_recovery_recorded
			check_variable = { var = famine_food_stage value = constant:famine_food_stage.stable compare = equals }
		}
		clr_state_flag = humanitarian_achievement_severe_recorded
		clr_state_flag = humanitarian_achievement_severe_qualifying
		clr_state_flag = humanitarian_achievement_recovery_recorded
	}
}

# COUNTRY scope. The completed exact-transfer transaction supplies the origin
# country id and the actual debit. Cycling the same origin never increments the
# distinct-origin counter twice.
humanitarian_achievement_record_arrival = {
	humanitarian_achievement_initialize_country = yes
	if = {
		limit = {
			check_variable = { civilian_transfer_actual_origin_debit > constant:humanitarian_achievement_value.zero }
			check_variable = { civilian_transfer_origin_country_id > constant:humanitarian_achievement_value.zero }
			NOT = {
				is_in_array = {
					array = humanitarian_achievement_origin_country_ids
					value = civilian_transfer_origin_country_id
				}
			}
		}
		add_to_array = { array = humanitarian_achievement_origin_country_ids value = civilian_transfer_origin_country_id }
		set_variable = { humanitarian_achievement_distinct_origin_count = humanitarian_achievement_origin_country_ids^num }
	}
}

# COUNTRY scope. The caller supplies the currently tracked cohort population and
# one unambiguous major cohort row. Aggregate country load is never allowed to
# make one cohort mature another cohort.
humanitarian_achievement_record_major_cohort_duration = {
	humanitarian_achievement_initialize_country = yes
	set_temp_variable = { humanitarian_achievement_major_cohort_target = humanitarian_achievement_initial_core_population }
	multiply_temp_variable = { humanitarian_achievement_major_cohort_target = constant:humanitarian_achievement_threshold.major_cohort_country_share }
	clamp_temp_variable = {
		var = humanitarian_achievement_major_cohort_target
		min = constant:humanitarian_achievement_threshold.major_cohort_minimum_people
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = humanitarian_achievement_major_cohort_candidate_id }
				NOT = { check_variable = { var = humanitarian_achievement_major_cohort_candidate_id value = constant:humanitarian_achievement_value.zero compare = greater_than } }
			}
		}
		clr_country_flag = humanitarian_achievement_major_cohort_displaced_long_enough
		clr_country_flag = humanitarian_achievement_all_major_cohorts_resolved
		clear_variable = humanitarian_achievement_major_cohort_start_date
		clear_variable = humanitarian_achievement_major_cohort_maturity_date
		clear_variable = humanitarian_achievement_major_cohort_origin_state_id

# COUNTRY scope. Called only after a valid integration, return, or resettlement
# transaction. The ID must belong to the exact active/terminal gate set and is
# recorded once for that generation.
humanitarian_achievement_record_gate_cohort_resolution = {
	set_temp_variable = { humanitarian_achievement_gate_resolution_id = humanitarian_achievement_resolution_cohort_id }
	if = {
		limit = { NOT = { check_variable = { var = humanitarian_achievement_gate_resolution_id value = constant:humanitarian_achievement_value.zero compare = greater_than } } }
		set_temp_variable = { humanitarian_achievement_gate_resolution_id = migration_cohort_id_to_move }
	}
	set_temp_variable = { humanitarian_achievement_gate_resolution_valid = constant:humanitarian_achievement_value.zero }
	if = {
		limit = {
			check_variable = { var = humanitarian_achievement_gate_resolution_id value = constant:humanitarian_achievement_value.zero compare = greater_than }
			OR = {
				AND = {
					check_variable = { var = migration_integration_result value = constant:humanitarian_runtime.one compare = equals }
					is_in_array = { array = humanitarian_achievement_gate_qualifying_cohort_ids value = humanitarian_achievement_gate_resolution_id }
				}
				AND = {
					check_variable = { var = migration_integration_result value = constant:humanitarian_runtime.one compare = equals }
					is_in_array = { array = humanitarian_achievement_gate_terminal_qualifying_cohort_ids value = humanitarian_achievement_gate_resolution_id }
				}
				AND = {
					check_variable = { var = civilian_transfer_result value = constant:civilian_transfer_route_result.valid compare = equals }
					check_variable = { civilian_transfer_survivor_credit > constant:humanitarian_achievement_value.zero }
					is_in_array = { array = humanitarian_achievement_gate_qualifying_cohort_ids value = humanitarian_achievement_gate_resolution_id }
				}
				AND = {
					check_variable = { var = civilian_transfer_result value = constant:civilian_transfer_route_result.valid compare = equals }
					check_variable = { civilian_transfer_survivor_credit > constant:humanitarian_achievement_value.zero }
					is_in_array = { array = humanitarian_achievement_gate_terminal_qualifying_cohort_ids value = humanitarian_achievement_gate_resolution_id }
				}
			}
		}
		set_temp_variable = { humanitarian_achievement_gate_resolution_valid = constant:humanitarian_achievement_value.one }
	}
	if = {
		limit = { check_variable = { var = humanitarian_achievement_gate_resolution_valid value = constant:humanitarian_achievement_value.one compare = equals } }
		if = {
			limit = {
				is_in_array = { array = humanitarian_achievement_gate_qualifying_cohort_ids value = humanitarian_achievement_gate_resolution_id }
				NOT = { is_in_array = { array = humanitarian_achievement_gate_resolved_cohort_ids value = humanitarian_achievement_gate_resolution_id } }
			}
			add_to_array = { array = humanitarian_achievement_gate_resolved_cohort_ids value = humanitarian_achievement_gate_resolution_id }
		}
		if = {
			limit = {
				is_in_array = { array = humanitarian_achievement_gate_terminal_qualifying_cohort_ids value = humanitarian_achievement_gate_resolution_id }
				NOT = { is_in_array = { array = humanitarian_achievement_gate_terminal_resolved_cohort_ids value = humanitarian_achievement_gate_resolution_id } }
			}
			add_to_array = { array = humanitarian_achievement_gate_terminal_resolved_cohort_ids value = humanitarian_achievement_gate_resolution_id }
		}
		set_variable = { humanitarian_achievement_gate_resolution_count = humanitarian_achievement_gate_resolved_cohort_ids^num }
		if = {
			limit = {
				check_variable = { var = humanitarian_achievement_gate_resolved_cohort_ids^num value = humanitarian_achievement_gate_qualifying_cohort_ids^num compare = equals }
				check_variable = { var = humanitarian_achievement_gate_qualifying_cohort_ids^num value = constant:humanitarian_achievement_value.zero compare = greater_than }
			}
			set_country_flag = humanitarian_achievement_gate_all_resolved
		}
		if = {
			limit = {
				check_variable = { var = humanitarian_achievement_gate_terminal_resolved_cohort_ids^num value = humanitarian_achievement_gate_terminal_qualifying_cohort_ids^num compare = equals }
				check_variable = { var = humanitarian_achievement_gate_terminal_qualifying_cohort_ids^num value = constant:humanitarian_achievement_value.zero compare = greater_than }
			}
			set_country_flag = humanitarian_achievement_gate_terminal_all_resolved
		}
		humanitarian_achievement_record_durable_outcome = yes
	}
}

# COUNTRY scope entered from a state callback. Only a previously bound state
# can update the safe set; no unrelated owned state is snapshotted.
humanitarian_achievement_refresh_receiving_state_food_safety = {
	set_temp_variable = { humanitarian_achievement_receiving_state_bound = constant:humanitarian_achievement_value.zero }
	set_temp_variable = { humanitarian_achievement_receiving_state_generation_match = constant:humanitarian_achievement_value.zero }
	set_temp_variable = { humanitarian_achievement_receiving_state_index = humanitarian_achievement_receiving_state_ids^num }
	while_loop_effect = {
			limit = { check_variable = { var = humanitarian_achievement_receiving_state_index value = constant:humanitarian_achievement_value.zero compare = greater_than } }
		subtract_from_temp_variable = { humanitarian_achievement_receiving_state_index = constant:humanitarian_runtime.array_index_increment }
		if = {
			limit = {
				PREV = { has_variable = humanitarian_achievement_receiving_state_generation }
				check_variable = { var = humanitarian_achievement_receiving_state_ids^humanitarian_achievement_receiving_state_index value = PREV.id compare = equals }
				check_variable = { var = humanitarian_achievement_receiving_state_generations^humanitarian_achievement_receiving_state_index value = PREV.humanitarian_achievement_receiving_state_generation compare = equals }
			}
			set_temp_variable = { humanitarian_achievement_receiving_state_generation_match = constant:humanitarian_achievement_value.one }
		}
	}
	if = {
		limit = {
			check_variable = { var = humanitarian_achievement_receiving_state_generation_match value = constant:humanitarian_achievement_value.one compare = equals }
		}
		set_temp_variable = { humanitarian_achievement_receiving_state_bound = constant:humanitarian_achievement_value.one }
	}
	if = {
		limit = { check_variable = { var = humanitarian_achievement_receiving_state_bound value = constant:humanitarian_achievement_value.one compare = equals } }
		if = {
			limit = {
				PREV = {
					OR = {
						has_state_flag = famine_food_stage_acute_shortage
						has_state_flag = famine_food_stage_famine
						has_state_flag = famine_food_stage_catastrophic
					}
				}
			}
			remove_from_array = { array = humanitarian_achievement_receiving_state_safe_ids value = PREV.id }
			set_variable = { humanitarian_achievement_receiving_state_safe_count = humanitarian_achievement_receiving_state_safe_ids^num }
			clr_country_flag = humanitarian_achievement_receiving_states_food_safe
		}
		else = {
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = humanitarian_achievement_receiving_state_safe_ids
							value = PREV.id
						}
					}
				}
				add_to_array = { array = humanitarian_achievement_receiving_state_safe_ids value = PREV.id }
			}
			set_variable = { humanitarian_achievement_receiving_state_safe_count = humanitarian_achievement_receiving_state_safe_ids^num }
			if = {
				limit = {
					check_variable = { var = humanitarian_achievement_receiving_state_count value = humanitarian_achievement_receiving_state_safe_count compare = equals }
					check_variable = { var = humanitarian_achievement_receiving_state_count value = constant:humanitarian_achievement_value.zero compare = greater_than }
				}
				set_country_flag = humanitarian_achievement_receiving_states_food_safe
			}
		}
		if = {
			limit = {
				PREV = {
					NOT = {
						OR = {
							has_state_flag = famine_food_stage_acute_shortage
							has_state_flag = famine_food_stage_famine
							has_state_flag = famine_food_stage_catastrophic
						}
					}
				}
				OR = {
					check_variable = { var = humanitarian_achievement_gate_front_state_id value = PREV.id compare = equals }
					check_variable = { var = humanitarian_achievement_gate_terminal_front_state_id value = PREV.id compare = equals }
				}
			}
			set_country_flag = humanitarian_achievement_gate_reception_safe
		}
	}
}
