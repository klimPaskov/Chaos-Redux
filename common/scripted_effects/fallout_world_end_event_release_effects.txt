# ============================================================================
# CHAOS REDUX - FALLOUT LIVING-WORLD RELEASE AUDIT EFFECTS
# ============================================================================
# Overview:
# - Records one generation-bound human review receipt for the 660-block floor.
# - Records separate engine-sensitive handoff flags without claiming runtime
#   proof that has not been supplied.
# - Never sets either scheduler activation flag.
# - Never runs automatically from a periodic on_action.
# ============================================================================

# Supported scope: COUNTRY coordinator. Input:
# fallout_event_release_rejection_reason_to_record.
fallout_event_record_release_audit_rejection = {
	if = {
		limit = {
			has_variable = global.fallout_transition_generation
		}
		set_variable = { global.fallout_event_release_audit_generation = global.fallout_transition_generation }
	}
	else = {
		set_variable = { global.fallout_event_release_audit_generation = constant:fallout_event_release_floor.zero }
	}
	set_variable = { global.fallout_event_release_audit_schema_version = constant:fallout_event_release_audit_schema.version }
	set_variable = { global.fallout_event_release_audit_status = constant:fallout_event_release_audit_status.rejected }
	set_variable = { global.fallout_event_release_audit_reason = fallout_event_release_rejection_reason_to_record }
	set_variable = { global.fallout_event_release_audit_day = global.num_days }
	clr_global_flag = fallout_event_release_audit_accepted
}

# Supported scope: COUNTRY coordinator. Inputs are temporary review receipts:
# fallout_event_release_reviewed_block_count, fallout_event_release_manual_sweep_reviewed,
# fallout_event_release_blackout_reviewed, fallout_event_release_host_authority_reviewed,
# fallout_event_release_save_recovery_reviewed, and
# fallout_event_release_multiplayer_reviewed. This effect stores evidence only.
fallout_event_record_release_floor_review = {
	set_temp_variable = { fallout_event_release_review_submission_valid = constant:fallout_event_release_floor.zero }
	set_temp_variable = { fallout_event_release_review_submission_skip = constant:fallout_event_release_floor.zero }
	if = {
		limit = {
			fallout_current_country_is_coordinator = yes
			has_global_flag = fallout_active
			has_global_flag = fallout_transition_complete
			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_event_scheduler_activation_approved }
			NOT = { has_global_flag = fallout_event_scheduler_active }
			has_variable = fallout_event_release_reviewed_block_count
			check_variable = {
				var = fallout_event_release_reviewed_block_count
				value = constant:fallout_event_release_floor.minimum_reviewed_blocks
				compare = greater_than_or_equals
			}
			has_variable = fallout_event_release_manual_sweep_reviewed
			has_variable = fallout_event_release_blackout_reviewed
			has_variable = fallout_event_release_host_authority_reviewed
			has_variable = fallout_event_release_save_recovery_reviewed
			has_variable = fallout_event_release_multiplayer_reviewed
			OR = {
				check_variable = { fallout_event_release_manual_sweep_reviewed = constant:fallout_event_release_floor.one }
				check_variable = { fallout_event_release_manual_sweep_reviewed = constant:fallout_event_release_floor.zero }
			}
			OR = {
				check_variable = { fallout_event_release_blackout_reviewed = constant:fallout_event_release_floor.one }
				check_variable = { fallout_event_release_blackout_reviewed = constant:fallout_event_release_floor.zero }
			}
			OR = {
				check_variable = { fallout_event_release_host_authority_reviewed = constant:fallout_event_release_floor.one }
				check_variable = { fallout_event_release_host_authority_reviewed = constant:fallout_event_release_floor.zero }
			}
			OR = {
				check_variable = { fallout_event_release_save_recovery_reviewed = constant:fallout_event_release_floor.one }
				check_variable = { fallout_event_release_save_recovery_reviewed = constant:fallout_event_release_floor.zero }
			}
			OR = {
				check_variable = { fallout_event_release_multiplayer_reviewed = constant:fallout_event_release_floor.one }
				check_variable = { fallout_event_release_multiplayer_reviewed = constant:fallout_event_release_floor.zero }
			}
		}
		set_temp_variable = { fallout_event_release_review_submission_valid = constant:fallout_event_release_floor.one }
	}
	if = {
		limit = {
			check_variable = { fallout_event_release_review_submission_valid = constant:fallout_event_release_floor.one }
			check_variable = { fallout_event_release_review_submission_skip = constant:fallout_event_release_floor.zero }
		}
		set_variable = { global.fallout_event_release_audit_schema_version = constant:fallout_event_release_audit_schema.version }
		set_variable = { global.fallout_event_release_reviewed_block_count = fallout_event_release_reviewed_block_count }
		set_variable = { global.fallout_event_release_review_generation = global.fallout_transition_generation }
		set_variable = { global.fallout_event_release_review_date = global.date }
		set_variable = { global.fallout_event_release_review_day = global.num_days }
		set_variable = { global.fallout_event_release_engine_review_generation = global.fallout_transition_generation }
		set_global_flag = fallout_event_release_floor_review_complete
		if = {
			limit = { check_variable = { fallout_event_release_manual_sweep_reviewed = constant:fallout_event_release_floor.one } }
			set_global_flag = fallout_event_release_manual_sweep_review_complete
		}
		else = { clr_global_flag = fallout_event_release_manual_sweep_review_complete }
		if = {
			limit = { check_variable = { fallout_event_release_blackout_reviewed = constant:fallout_event_release_floor.one } }
			set_global_flag = fallout_event_release_blackout_review_complete
		}
		else = { clr_global_flag = fallout_event_release_blackout_review_complete }
		if = {
			limit = { check_variable = { fallout_event_release_host_authority_reviewed = constant:fallout_event_release_floor.one } }
			set_global_flag = fallout_event_release_host_authority_review_complete
		}
		else = { clr_global_flag = fallout_event_release_host_authority_review_complete }
		if = {
			limit = { check_variable = { fallout_event_release_save_recovery_reviewed = constant:fallout_event_release_floor.one } }
			set_global_flag = fallout_event_release_save_recovery_review_complete
		}
		else = { clr_global_flag = fallout_event_release_save_recovery_review_complete }
		if = {
			limit = { check_variable = { fallout_event_release_multiplayer_reviewed = constant:fallout_event_release_floor.one } }
			set_global_flag = fallout_event_release_multiplayer_review_complete
		}
		else = { clr_global_flag = fallout_event_release_multiplayer_review_complete }
		set_variable = { global.fallout_event_release_audit_status = constant:fallout_event_release_audit_status.none }
		clear_variable = global.fallout_event_release_audit_reason
		clr_global_flag = fallout_event_release_audit_accepted
		if = {
			limit = {
				fallout_event_release_floor_receipt_is_current = yes
				fallout_event_release_engine_surface_receipts_are_current = yes
				fallout_event_release_world_barrier_is_current = yes
			}
			set_global_flag = fallout_event_release_audit_accepted
			set_variable = { global.fallout_event_release_audit_status = constant:fallout_event_release_audit_status.accepted }
			set_variable = { global.fallout_event_release_audit_generation = global.fallout_transition_generation }
			set_variable = { global.fallout_event_release_audit_day = global.num_days }
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					has_global_flag = fallout_event_scheduler_activation_approved
					has_global_flag = fallout_event_scheduler_active
				}
			}
			set_temp_variable = { fallout_event_release_review_submission_skip = constant:fallout_event_release_floor.one }
		}
		else = {
		set_temp_variable = { fallout_event_release_rejection_reason_to_record = constant:fallout_event_release_rejection_reason.invalid_scope }
		if = {
			limit = {
				fallout_current_country_is_coordinator = yes
				OR = {
					NOT = { has_variable = fallout_event_release_reviewed_block_count }
					check_variable = { fallout_event_release_reviewed_block_count = constant:fallout_event_release_floor.zero }
					check_variable = {
						fallout_event_release_reviewed_block_count
						value = constant:fallout_event_release_floor.minimum_reviewed_blocks
						compare = less_than
					}
				}
			}
			set_temp_variable = { fallout_event_release_rejection_reason_to_record = constant:fallout_event_release_rejection_reason.reviewed_floor_missing }
		}
		fallout_event_record_release_audit_rejection = yes
		}
	}
}
