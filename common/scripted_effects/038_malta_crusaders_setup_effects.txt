# ============================================================================
# CHAOS REDUX - EVENT 038 MALTA CRUSADERS OPENING TRANSACTION AND PACKAGE
# ============================================================================
# The release transaction in three steps, then the Malta country package.
#
#   1. malta_prepare_opening_transaction (ANY scope)
#      Freeze. Reads the live map and writes the proposed state set, the
#      owners those states have now and the displaced top owners to private
#      global arrays. No ownership changes. Sets global.malta_opening_ready
#      to 1 or leaves a reject code in global.malta_opening_reject:
#        1 the actor tag is a subject, capitulated, special or already marked
#        2 the heartland (Malta, state 116) cannot be taken or released
#        3 no Holy Land command state can be taken or is already held
#        4 neither Cyprus nor an island foothold can serve as the sea bridge
#        5 more displaced owners than the package allows
#   2. malta_validate_opening_transaction (ANY scope)
#      Re-checks every frozen state against the map as it is at commit.
#   3. malta_commit_opening_transaction (ANY scope)
#      Transfers the frozen states, proves ownership, applies the package,
#      declares the bounded wars and records the result in
#      global.malta_opening_result (1 success, 2 rolled back or rejected).
#      A failed proof restores every state to the owner recorded at freeze
#      and leaves no actor package, no war and no flag behind.
#
# The actor is the vanilla tag MLT. A live Malta is transformed. A dead Malta
# comes back through the first state transfer and receives the package.
# Package level is the highest active evolution stage (0 to 3). A stage that
# is disabled in the Event Details list contributes nothing, and the opening
# builds the cumulative package in one pass.
#
# Interim stand-in: the army uses the ordinary infantry, cavalry, artillery
# and engineer sub-units under templates named after the medieval formations.
# The custom unit families wait for the model pipeline. Swapping a family in
# later means changing the template tables in malta_create_templates only.
# ============================================================================

# ---------------------------------------------------------------------------
# Stages
# ---------------------------------------------------------------------------

# ANY scope. Sets global flags malta_stage_1_active to malta_stage_3_active for
# every evolution stage that is enabled and reached by the Chaos value, and
# global.malta_package_level to the highest of them.
malta_resolve_stages = {
	clr_global_flag = malta_stage_1_active
	clr_global_flag = malta_stage_2_active
	clr_global_flag = malta_stage_3_active
	set_variable = { global.malta_package_level = 0 }
	set_temp_variable = { events_log_evolution_event_id = constant:malta_event.id }
	set_temp_variable = { events_log_evolution_type = constant:malta_event.evolution_type }

	set_temp_variable = { events_log_evolution_stage = constant:malta_evolution.stage_1 }
	set_temp_variable = { events_log_evolution_tier = constant:malta_evolution.stage_1 }
	if = {
		limit = { is_current_evolution_enabled = yes }
		set_global_flag = malta_stage_1_active
		set_variable = { global.malta_package_level = constant:malta_evolution.stage_1 }
	}
	set_temp_variable = { events_log_evolution_stage = constant:malta_evolution.stage_2 }
	set_temp_variable = { events_log_evolution_tier = constant:malta_evolution.stage_2 }
	if = {
		limit = { is_current_evolution_enabled = yes }
		set_global_flag = malta_stage_2_active
		set_variable = { global.malta_package_level = constant:malta_evolution.stage_2 }
	}
	set_temp_variable = { events_log_evolution_stage = constant:malta_evolution.stage_3 }
	set_temp_variable = { events_log_evolution_tier = constant:malta_evolution.stage_3 }
	if = {
		limit = { is_current_evolution_enabled = yes }
		set_global_flag = malta_stage_3_active
		set_variable = { global.malta_package_level = constant:malta_evolution.stage_3 }
	}
}

# ---------------------------------------------------------------------------
# Freeze
# ---------------------------------------------------------------------------

# STATE scope. Saves the top owner of the state in event target
# malta_top_owner: the owner, or the overlord when the owner is a subject, up
# to two links.
malta_find_top_owner = {
	owner = {
		save_event_target_as = malta_top_owner
		if = {
			limit = { is_subject = yes }
			overlord = {
				save_event_target_as = malta_top_owner
				if = {
					limit = { is_subject = yes }
					overlord = { save_event_target_as = malta_top_owner }
				}
			}
		}
	}
}

# STATE scope. Appends the state and its present owner to the frozen ledger
# and records the top owner as displaced.
malta_freeze_state = {
	malta_freeze_state_peaceful = yes
	malta_find_top_owner = yes
	event_target:malta_top_owner = {
		if = {
			limit = { NOT = { is_in_array = { array = global.malta_displaced_owners value = THIS } } }
			add_to_array = { global.malta_displaced_owners = THIS }
		}
	}
}

# STATE scope. The same without recording a displaced owner. The heartland is
# released this way: it is the one state whose owner does not take part in the
# war set unless it loses another frozen state.
malta_freeze_state_peaceful = {
	add_to_array = { global.malta_frozen_states = THIS }
	owner = { add_to_array = { global.malta_frozen_owners = THIS } }
}

# STATE scope. Takes one island foothold when fewer than the wanted number
# were taken and the top owner fits the displaced-owner cap.
malta_try_pick_foothold = {
	if = {
		limit = {
			check_variable = { var = malta_foothold_picks value = malta_foothold_wanted compare = less_than }
			malta_state_can_be_opening_target = yes
		}
		malta_find_top_owner = yes
		set_temp_variable = { malta_cap_ok = 1 }
		event_target:malta_top_owner = {
			if = {
				limit = {
					NOT = { is_in_array = { array = global.malta_displaced_owners value = THIS } }
					check_variable = { var = global.malta_displaced_owners^num value = constant:malta_opening.owner_cap compare = greater_than_or_equals }
				}
				set_temp_variable = { malta_cap_ok = 0 }
			}
		}
		if = {
			limit = { check_variable = { var = malta_cap_ok value = 1 compare = equals } }
			malta_freeze_state = yes
			add_to_temp_variable = { malta_foothold_picks = 1 }
		}
	}
}

# ANY scope. Freeze. See the file header for the reject codes.
malta_prepare_opening_transaction = {
	clear_array = global.malta_frozen_states
	clear_array = global.malta_frozen_owners
	clear_array = global.malta_displaced_owners
	set_variable = { global.malta_opening_ready = 0 }
	set_variable = { global.malta_opening_reject = 0 }
	set_temp_variable = { malta_reject = 0 }
	set_temp_variable = { malta_holy_land_ok = 0 }
	set_temp_variable = { malta_bridge_ok = 0 }
	set_temp_variable = { malta_foothold_picks = 0 }
	set_temp_variable = { malta_foothold_wanted = constant:malta_opening.footholds_baseline }
	malta_resolve_stages = yes

	clr_global_flag = malta_transformed_existing
	MLT = {
		if = {
			limit = { exists = yes }
			set_global_flag = malta_transformed_existing
		}
		if = {
			limit = {
				OR = {
					AND = { exists = yes is_subject = yes }
					AND = { exists = yes has_capitulated = yes }
					AND = { exists = yes is_special_chaos_country = yes }
					AND = { exists = yes is_actual_nonhuman_country = yes }
					has_country_flag = malta_origin
				}
			}
			set_temp_variable = { malta_reject = 1 }
		}
	}

	116 = {
		if = {
			limit = { NOT = { is_owned_by = MLT } }
			if = {
				limit = { malta_state_can_be_opening_target = yes }
				malta_freeze_state_peaceful = yes
			}
			else_if = {
				limit = { check_variable = { var = malta_reject value = 0 compare = equals } }
				set_temp_variable = { malta_reject = 2 }
			}
		}
	}

	454 = {
		if = {
			limit = { is_owned_by = MLT }
			set_temp_variable = { malta_holy_land_ok = 1 }
		}
		else_if = {
			limit = { malta_state_can_be_opening_target = yes }
			malta_freeze_state = yes
			set_temp_variable = { malta_holy_land_ok = 1 }
		}
	}
	455 = {
		if = {
			limit = {
				check_variable = { var = malta_holy_land_ok value = 1 compare = equals }
				malta_state_can_be_opening_target = yes
			}
			malta_freeze_state = yes
		}
	}
	183 = {
		if = {
			limit = { is_owned_by = MLT }
			set_temp_variable = { malta_bridge_ok = 1 }
		}
		else_if = {
			limit = { malta_state_can_be_opening_target = yes }
			malta_freeze_state = yes
			set_temp_variable = { malta_bridge_ok = 1 }
		}
	}

	if = {
		limit = { has_global_flag = malta_stage_2_active }
		add_to_temp_variable = { malta_foothold_wanted = constant:malta_opening.footholds_stage_2_extra }
	}
	164 = { malta_try_pick_foothold = yes }
	187 = { malta_try_pick_foothold = yes }
	182 = { malta_try_pick_foothold = yes }
	if = {
		limit = { check_variable = { var = malta_foothold_picks value = 0 compare = greater_than } }
		set_temp_variable = { malta_bridge_ok = 1 }
	}
	# An owned island foothold also serves as the bridge.
	if = {
		limit = { MLT = { any_owned_state = { malta_state_is_island_foothold = yes } } }
		set_temp_variable = { malta_bridge_ok = 1 }
	}

	if = {
		limit = {
			check_variable = { var = malta_reject value = 0 compare = equals }
			check_variable = { var = malta_holy_land_ok value = 1 compare = less_than }
		}
		set_temp_variable = { malta_reject = 3 }
	}
	if = {
		limit = {
			check_variable = { var = malta_reject value = 0 compare = equals }
			check_variable = { var = malta_bridge_ok value = 1 compare = less_than }
		}
		set_temp_variable = { malta_reject = 4 }
	}
	if = {
		limit = {
			check_variable = { var = malta_reject value = 0 compare = equals }
			check_variable = { var = global.malta_displaced_owners^num value = constant:malta_opening.owner_cap compare = greater_than }
		}
		set_temp_variable = { malta_reject = 5 }
	}
	set_variable = { global.malta_opening_reject = malta_reject }
	if = {
		limit = { check_variable = { var = malta_reject value = 0 compare = equals } }
		set_variable = { global.malta_opening_ready = 1 }
	}
}

# ---------------------------------------------------------------------------
# Validate
# ---------------------------------------------------------------------------

# ANY scope. Output: temporary malta_validate_ok (1 when every frozen state
# can still be taken).
malta_validate_opening_transaction = {
	set_temp_variable = { malta_validate_ok = 1 }
	if = {
		limit = {
			OR = {
				check_variable = { var = global.malta_opening_ready value = 1 compare = less_than }
				check_variable = { var = global.malta_frozen_states^num value = constant:malta_opening.minimum_states compare = less_than }
			}
		}
		set_temp_variable = { malta_validate_ok = 0 }
	}
	for_each_scope_loop = {
		array = global.malta_frozen_states
		if = {
			limit = { NOT = { malta_state_can_be_opening_target = yes } }
			set_temp_variable = { malta_validate_ok = 0 }
		}
	}
}

# ---------------------------------------------------------------------------
# Commit and rollback
# ---------------------------------------------------------------------------

# ANY scope. Restores every frozen state to the owner recorded at freeze.
malta_rollback_opening_transaction = {
	set_temp_variable = { malta_rollback_count = global.malta_frozen_states^num }
	for_loop_effect = {
		start = 0
		end = malta_rollback_count
		value = malta_rollback_index
		set_temp_variable = { malta_rollback_state = global.malta_frozen_states^malta_rollback_index }
		set_temp_variable = { malta_rollback_owner = global.malta_frozen_owners^malta_rollback_index }
		var:malta_rollback_state = {
			transfer_state_to = var:malta_rollback_owner
		}
	}
}

# ANY scope. Commit. Reads the frozen ledger and records the result in
# global.malta_opening_result.
malta_commit_opening_transaction = {
	set_variable = { global.malta_opening_result = 0 }
	malta_validate_opening_transaction = yes
	if = {
		limit = { check_variable = { var = malta_validate_ok value = 1 compare = equals } }
		set_global_flag = malta_opening_in_progress
		set_variable = { global.malta_transaction_id = global.num_days }
		add_to_variable = { global.malta_transaction_id = constant:malta_opening.transaction_offset }
		for_each_scope_loop = {
			array = global.malta_frozen_states
			owner = { save_event_target_as = malta_freeze_owner }
			set_variable = { malta_former_owner = event_target:malta_freeze_owner }
			transfer_state_to = MLT
		}
		set_temp_variable = { malta_commit_ok = 1 }
		if = {
			limit = { NOT = { country_exists = MLT } }
			set_temp_variable = { malta_commit_ok = 0 }
		}
		for_each_scope_loop = {
			array = global.malta_frozen_states
			if = {
				limit = { NOT = { is_owned_by = MLT } }
				set_temp_variable = { malta_commit_ok = 0 }
			}
		}
		if = {
			limit = { check_variable = { var = malta_commit_ok value = 1 compare = equals } }
			MLT = {
				malta_initialize_country_package = yes
				malta_declare_opening_wars = yes
			}
			set_global_flag = malta_event_fired
			set_variable = { global.malta_opening_result = 1 }
		}
		else = {
			malta_rollback_opening_transaction = yes
			set_variable = { global.malta_opening_result = 2 }
		}
		clr_global_flag = malta_opening_in_progress
	}
	else = {
		set_variable = { global.malta_opening_result = 2 }
	}
	# The ledger keeps its owners for settlement decisions and is cleared on
	# the next freeze. Per-package stage flags are cleared by the package.
}

# ---------------------------------------------------------------------------
# Wars
# ---------------------------------------------------------------------------

# COUNTRY (MLT). One war per displaced top owner that is not already at war
# with Malta. A claim on every registry state the owner still holds gives the
# war a real goal.
malta_declare_opening_wars = {
	save_event_target_as = malta_actor
	for_each_scope_loop = {
		array = global.malta_displaced_owners
		if = {
			limit = {
				exists = yes
				NOT = { has_war_with = event_target:malta_actor }
			}
			malta_add_registry_claims = yes
			event_target:malta_actor = {
				declare_war_on = {
					target = PREV
					type = take_claimed_state
				}
			}
		}
	}
}

# COUNTRY (the owner). Claims by the actor on every campaign-pool state the
# scoped country owns. Used before a war so take_claimed_state has a goal.
malta_add_registry_claims = {
	every_owned_state = {
		limit = {
			malta_state_is_campaign_pool = yes
			malta_state_is_open_for_claims = yes
			NOT = { is_claimed_by = event_target:malta_actor }
		}
		add_claim_by = event_target:malta_actor
	}
}

# STATE scope. Spawns malta_spawn_count divisions of template id
# malta_spawn_template here. The names come from the scripted localisation
# functions GetMaltaTemplateName and GetMaltaDivisionName.
malta_spawn_loop = {
	for_loop_effect = {
		start = 0
		end = malta_spawn_count
		value = malta_spawn_n
		meta_effect = {
			text = {
				create_unit = {
					division = "name = \"[DIV_NAME]\" division_template = \"[TEMPLATE_NAME]\" start_experience_factor = [XP_FACTOR] start_equipment_factor = [EQ_FACTOR] start_manpower_factor = [MP_FACTOR]"
					owner = event_target:malta_spawn_owner
					allow_spawning_on_enemy_provs = no
				}
			}
			DIV_NAME = "[This.GetMaltaDivisionName]"
			TEMPLATE_NAME = "[This.GetMaltaTemplateName]"
			XP_FACTOR = "[?global.malta_spawn_xp|.2]"
			EQ_FACTOR = "[?global.malta_spawn_equipment|.2]"
			MP_FACTOR = "[?global.malta_spawn_manpower|.2]"
		}
	}
}

# COUNTRY (MLT). Input: temp malta_spawn_in_state (state id). Spawns the division count in
# malta_spawn_count of template malta_spawn_template in that state when the actor owns it, in
# Malta otherwise.
malta_spawn_in = {
	meta_effect = {
		text = {
			if = {
				limit = { owns_state = [STATE] }
				[STATE] = { malta_spawn_loop = yes }
			}
			else = {
				116 = { malta_spawn_loop = yes }
			}
		}
		STATE = "[?malta_spawn_in_state|.0]"
	}
}

# COUNTRY (MLT). Spawns the division count in malta_spawn_count of template
# malta_spawn_template in an owned island foothold, in Malta when there is
# none.
malta_spawn_in_foothold = {
	if = {
		limit = { any_owned_state = { malta_state_is_island_foothold = yes } }
		random_owned_state = {
			limit = { malta_state_is_island_foothold = yes }
			malta_spawn_loop = yes
		}
	}
	else = {
		116 = { malta_spawn_loop = yes }
	}
}
