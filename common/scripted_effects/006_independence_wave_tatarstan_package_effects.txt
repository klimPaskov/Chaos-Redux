independence_wave_refresh_tat_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_tat_package = yes }
		remove_ideas = { tat_fragmented_river_mandate tat_tatarstan_host_compact }
		if = {
			limit = { has_stable_independence_wave_tat_compact = yes }
			add_ideas = tat_tatarstan_host_compact
			set_country_flag = independence_wave_tat_compact_stabilized
		}
		else = {
			add_ideas = tat_fragmented_river_mandate
			clr_country_flag = independence_wave_tat_compact_stabilized
		}
	}
}
