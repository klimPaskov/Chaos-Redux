# ============================================================================
# CHAOS REDUX - EVENT CLUSTER EFFECTS
# ============================================================================
# Event clusters sit above the individual random-event selection layer. When an
# automatically selected event belongs to a cluster, the cluster can fire first,
# record a cluster log row, and then queue valid member events through the same
# event handlers used by single events. Member events run with a cluster context
# guard so the cluster updates timer and major-event pacing once for the group.
#
# To add a future cluster, register it in initialize_event_cluster_definitions,
# map member events in event_belongs_to_cluster, and add ordered member rows in
# load_event_cluster_members.
# ============================================================================


initialize_event_cluster_system = {
    # Definitions and successful history are save data. Only bootstrap an
    # absent registry; do not erase them when an existing save re-enters the
    # event-system initializer.
    if = {
        limit = { check_variable = { global.event_clusters^num < 1 } }
        clear_array = global.event_clusters
        clear_array = global.event_cluster_type_entries
        clear_array = global.event_cluster_unlock_tier_entries
        initialize_event_cluster_definitions = yes
    }
    clear_array = global.events_log_cluster_view_sequence_entries
    clear_array = global.events_log_cluster_view_date_entries
    clear_array = global.events_log_cluster_view_id_entries
    clear_array = global.events_log_cluster_view_type_entries
    clear_array = global.events_log_cluster_view_tier_entries
    clear_array = global.events_log_cluster_view_trigger_event_id_entries
    clear_array = global.events_log_cluster_view_actor_entries
    clear_array = global.events_log_cluster_view_has_actor_entries
    clear_array = global.events_log_cluster_view_member_count_entries
    clear_array = global.events_log_cluster_view_fired_count_entries
    clear_array = global.events_log_cluster_view_skipped_count_entries
    clear_array = global.events_log_cluster_view_unlock_tier_entries
    clear_array = global.events_log_cluster_view_roll_chance_entries
    clear_array = global.events_log_cluster_view_available_entries
    clear_array = global.events_log_cluster_view_enabled_entries
	clear_array = global.events_log_cluster_view_unavailability_reason_entries
	# These aligned arrays are append-only history/probability memory. Preserve
	# them across reloads; migration is performed by the versioned helper below.
	clear_array = global.events_log_cluster_detail_member_event_id_entries
    clear_array = global.events_log_cluster_detail_member_status_entries
    clear_array = global.events_log_cluster_detail_member_danger_entries
	clear_array = global.events_log_cluster_detail_member_unavailability_reason_entries
	clear_array = global.events_log_cluster_detail_member_row_id_entries
	clear_array = global.events_log_cluster_detail_member_role_entries
	clear_array = global.events_log_cluster_detail_member_severity_entries
	clear_array = global.events_log_cluster_detail_member_effective_min_tier_entries
	clear_array = global.events_log_cluster_detail_member_base_chance_entries
	clear_array = global.events_log_cluster_detail_member_final_chance_entries
	clear_array = global.events_log_cluster_detail_member_roll_entries
	clear_array = global.events_log_cluster_detail_member_trigger_entries
	if = {
		limit = { NOT = { has_country_flag = event_cluster_pending_members_active } }
		clear_array = event_cluster_pending_member_event_id_entries
		clear_array = event_cluster_pending_member_batch_index_entries
		clear_array = event_cluster_pending_member_event013_state_entries
		clear_array = event_cluster_pending_member_event013_target_country_entries
		clear_array = event_cluster_pending_member_event013_family_entries
		clear_array = event_cluster_pending_member_event013_variant_entries
		clear_array = event_cluster_pending_member_batch_id_entries
		clear_array = event_cluster_pending_member_cluster_id_entries
		clear_array = event_cluster_pending_member_row_id_entries
		clear_array = event_cluster_pending_member_trigger_entries
		clear_array = event_cluster_pending_member_history_sequence_entries
		clear_array = event_cluster_pending_member_role_entries
		clear_array = event_cluster_pending_member_manual_entries
		clear_array = event_cluster_pending_member_optional_fired_entries
		clear_array = event_cluster_pending_member_optional_eligible_entries
		clear_array = event_cluster_pending_member_fired_entries
		clear_array = event_cluster_pending_member_skipped_entries
	}

    if = {
		limit = { NOT = { has_variable = global.events_log_cluster_sequence } }
		set_variable = { global.events_log_cluster_sequence = 0 }
	}
	if = {
		limit = { NOT = { has_variable = global.total_event_clusters_fired } }
		set_variable = { global.total_event_clusters_fired = 0 }
	}
	if = {
		limit = { NOT = { has_variable = global.last_fired_cluster_id } }
		set_variable = { global.last_fired_cluster_id = 0 }
	}
	if = {
		limit = { NOT = { has_variable = global.event_cluster_next_batch_id } }
		set_variable = { global.event_cluster_next_batch_id = 0 }
	}
    if = {
		limit = { NOT = { has_country_flag = event_cluster_pending_members_active } }
		set_variable = { event_cluster_pending_member_index = 0 }
		set_variable = { event_cluster_pending_batch_id = 0 }
	}
	else = {
		if = {
			limit = { NOT = { has_variable = event_cluster_pending_member_index } }
			set_variable = { event_cluster_pending_member_index = 0 }
		}
		if = {
			limit = { NOT = { has_variable = event_cluster_pending_batch_id } }
			set_variable = { event_cluster_pending_batch_id = 0 }
		}
	}
	if = {
		limit = { has_country_flag = event_cluster_pending_members_active }
		event_cluster_migrate_pending_queue_metadata = yes
	}
	# Append the dynamic, memberless cluster non-destructively for saves whose
	# registry predates it. Appending preserves every existing aligned index.
	if = {
		limit = { NOT = { is_in_array = { global.event_clusters = constant:event_cluster_id.random_stuff } } }
		add_to_array = { global.event_clusters = constant:event_cluster_id.random_stuff }
		add_to_array = { array = global.event_cluster_type_entries value = constant:event_cluster_type.repeatable }
		add_to_array = { array = global.event_cluster_unlock_tier_entries value = constant:event_cluster_random_stuff.unlock_tier }
	}
	clr_country_flag = event_cluster_member_fire_context

	event_cluster_ensure_probability_state = yes
	random_stuff_cluster_ensure_state = yes
}

# Add metadata introduced by the dynamic cluster queue without clearing an
# active queue in an older save. Legacy rows retain their event ID and context.
# A missing row ID is resolved from the event ID at dispatch time, with the
# primary-row rule providing a deterministic duplicate fallback.
event_cluster_migrate_pending_queue_metadata = {
	set_temp_variable = { event_cluster_pending_legacy_count = event_cluster_pending_member_event_id_entries^num }
	if = {
		limit = { check_variable = { event_cluster_pending_legacy_count > 0 } }
		for_loop_effect = {
			start = 0
			end = event_cluster_pending_legacy_count
			value = i
			if = {
				limit = { check_variable = { var = event_cluster_pending_member_batch_id_entries^num value = i compare = less_than_or_equals } }
				add_to_array = { array = event_cluster_pending_member_batch_id_entries value = event_cluster_pending_batch_id }
			}
			if = {
				limit = { check_variable = { var = event_cluster_pending_member_cluster_id_entries^num value = i compare = less_than_or_equals } }
				add_to_array = { array = event_cluster_pending_member_cluster_id_entries value = global.last_fired_cluster_id }
			}
			if = {
				limit = { check_variable = { var = event_cluster_pending_member_row_id_entries^num value = i compare = less_than_or_equals } }
				add_to_array = { array = event_cluster_pending_member_row_id_entries value = 0 }
			}
			if = {
				limit = { check_variable = { var = event_cluster_pending_member_trigger_entries^num value = i compare = less_than_or_equals } }
				add_to_array = { array = event_cluster_pending_member_trigger_entries value = 0 }

				add_to_array = { array = event_cluster_pending_member_skipped_entries value = 0 }

get_current_chaos_tier_value = {
    set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_0 }
    if = {
        limit = { has_global_flag = { flag = chaos_tier value = 5 } }
        set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_final }
    }
    else_if = {
        limit = { has_global_flag = { flag = chaos_tier value = 4 } }
        set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_4 }
    }
    else_if = {
        limit = { has_global_flag = { flag = chaos_tier value = 3 } }
        set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_3 }
    }
    else_if = {
        limit = { has_global_flag = { flag = chaos_tier value = 2 } }
        set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_2 }
    }
    else_if = {
        limit = { has_global_flag = { flag = chaos_tier value = 1 } }
        set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_1 }
    }
}

get_event_cluster_definition_index = {
    set_temp_variable = { event_cluster_definition_index = -1 }
    for_loop_effect = {
        start = 0
        end = global.event_clusters^num
        value = i
        if = {
            limit = {
                check_variable = {
                    var = global.event_clusters^i
                    value = event_cluster_id
                    compare = equals
                }
            }
            set_temp_variable = { event_cluster_definition_index = i }
        }
    }
}

# Idempotently align probability memory with the append-only cluster registry.
# This helper is deliberately separate from initialize_event_cluster_system:
# the latter clears history and is only safe for a fresh event-system setup.
# Existing saves reach this helper lazily from cluster checks and UI refreshes.
event_cluster_ensure_probability_state = {
	set_temp_variable = { event_cluster_probability_state_current_version = constant:event_cluster_probability_state.version }
	set_temp_variable = { event_cluster_probability_state_mismatch = 0 }
	if = {
		limit = { NOT = { has_variable = global.event_cluster_probability_state_version } }
		# A missing version is a bootstrap state. Do not write the current
		# version until every aligned array has been created and verified.
		set_temp_variable = { event_cluster_probability_state_needs_bootstrap = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.event_cluster_probability_state_version
				value = event_cluster_probability_state_current_version
				compare = greater_than
			}
		}
		set_temp_variable = { event_cluster_probability_state_mismatch = 1 }
	}
	else = {
		# Older versions are migrated by append-missing below. Preserve the saved

		set_temp_variable = { event_cluster_trigger_severity = temp_event_cluster_member_severity_entries^event_cluster_trigger_member_row_index }
	}
}

event_cluster_member_check_availability = {
	set_temp_variable = { event_cluster_member_can_fire = 0 }
	set_temp_variable = { event_cluster_member_status = constant:event_cluster_member_status.skipped_unavailable }
	set_temp_variable = { event_cluster_member_unavailability_reason = constant:event_system_event_unavailability_reason.none }
	if = {
		limit = { NOT = { has_variable = event_cluster_member_severity } }
		set_temp_variable = { event_cluster_member_severity = event_cluster_member_danger }
	}
	if = {
		limit = { NOT = { has_variable = event_cluster_member_min_tier } }
		set_temp_variable = { event_cluster_member_min_tier = constant:chaos_meter_tier_id.tier_0 }
	}
	set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_0 }
	get_current_chaos_tier_value = yes
	event_cluster_get_member_effective_min_tier = yes
	if = {
		limit = { check_variable = { event_cluster_force_fire > 0 } }
		# Settings force mode sends the equivalent single event directly to its
		# normal dispatcher, bypassing automatic pool state such as weight, fired
		# status, Event Chaos Level, and the disabled list. Match that force context
		# here; event-specific runtime preparation and dispatch can still reject the
		# row, so manual cluster force never fabricates a successful event firing.
		set_temp_variable = { event_cluster_member_can_fire = 1 }
		set_temp_variable = { event_cluster_member_status = constant:event_cluster_member_status.available }
	}
	else = {
		set_temp_variable = { random_event_selection_candidate_id = event_cluster_member_event_id }
		set_temp_variable = { random_event_selection_filter = constant:settings_event_filter.all }
		set_temp_variable = { random_event_selection_candidate_is_valid = 0 }
		set_temp_variable = { event_cluster_liberations_member_availability_context = 0 }
		if = {
			limit = {
				check_variable = { event_cluster_id = constant:event_cluster_id.liberations }
				check_variable = { event_cluster_member_event_id = constant:independence_wave_event.id }
			}
			set_temp_variable = { event_cluster_liberations_member_availability_context = 1 }
		}
		evaluate_random_event_selection_candidate = yes
		set_temp_variable = { event_cluster_member_unavailability_reason = event_pool_candidate_unavailability_reason }
		set_temp_variable = { event_cluster_liberations_member_availability_context = 0 }
		if = {
			limit = { check_variable = { random_event_selection_candidate_is_valid < 1 } }
			set_temp_variable = { event_cluster_member_can_fire = 0 }
			if = {
				limit = { check_variable = { event_cluster_member_unavailability_reason = constant:event_system_event_unavailability_reason.disabled } }
				set_temp_variable = { event_cluster_member_status = constant:event_cluster_member_status.skipped_disabled }
			}
			else = {
				set_temp_variable = { event_cluster_member_status = constant:event_cluster_member_status.skipped_unavailable }
			}
			if = {
				limit = { check_variable = { event_cluster_member_unavailability_reason = constant:event_system_event_unavailability_reason.none } }
				set_temp_variable = { event_cluster_member_unavailability_reason = constant:event_system_event_unavailability_reason.unknown }
			}
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_variable = event_cluster_member_event_system_only }
					check_variable = { event_cluster_member_event_system_only < 1 }
				}
				check_variable = {
					var = current_chaos_tier_value
					value = event_cluster_member_effective_min_tier
					compare = less_than
				}
			}
			set_temp_variable = { event_cluster_member_can_fire = 0 }
			set_temp_variable = { event_cluster_member_status = constant:event_cluster_member_status.skipped_chaos_tier }
			# The event system already accepted the event's own Chaos Level. Keep the
			# canonical event-system reason empty when the stricter cluster member floor
			# is what blocks participation.
			set_temp_variable = { event_cluster_member_unavailability_reason = constant:event_system_event_unavailability_reason.none }
		}
		else = {
			set_temp_variable = { event_cluster_member_can_fire = 1 }
			set_temp_variable = { event_cluster_member_status = constant:event_cluster_member_status.available }
			set_temp_variable = { event_cluster_member_unavailability_reason = constant:event_system_event_unavailability_reason.none }
		}
	}
}

event_cluster_scan_members_for_activation = {
	load_event_cluster_members = yes
	clear_array = temp_event_cluster_member_base_eligible_entries
	clear_array = temp_event_cluster_member_effective_min_tier_entries
	clear_array = temp_event_cluster_member_status_entries
	clear_array = temp_event_cluster_member_unavailability_reason_entries
	clear_array = temp_event_cluster_member_trigger_entries
	clear_array = temp_event_cluster_member_base_chance_entries
	clear_array = temp_event_cluster_member_final_chance_entries
	clear_array = temp_event_cluster_member_roll_entries
	set_temp_variable = { event_cluster_eligible_member_count = 0 }
	set_temp_variable = { event_cluster_optional_eligible_count = 0 }
	set_temp_variable = { event_cluster_trigger_member_row_is_eligible = 0 }
	set_temp_variable = { event_cluster_any_member_can_fire = 0 }
	set_temp_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_0 }
	get_current_chaos_tier_value = yes
	event_cluster_resolve_trigger_member_row = yes
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		set_temp_variable = { event_cluster_member_event_id = temp_event_cluster_member_event_id_entries^i }
		set_temp_variable = { event_cluster_member_role = temp_event_cluster_member_role_entries^i }
		set_temp_variable = { event_cluster_member_min_tier = temp_event_cluster_member_min_tier_entries^i }
		set_temp_variable = { event_cluster_member_danger = temp_event_cluster_member_danger_entries^i }
		set_temp_variable = { event_cluster_member_severity = temp_event_cluster_member_severity_entries^i }
		set_temp_variable = { event_cluster_member_row_id = temp_event_cluster_member_row_id_entries^i }
		set_temp_variable = { event_cluster_member_primary_trigger = temp_event_cluster_member_primary_trigger_entries^i }
		set_temp_variable = { event_cluster_member_trigger = 0 }
		set_temp_variable = { event_cluster_member_event_system_only = 0 }
		if = {
			limit = { check_variable = { i = event_cluster_trigger_member_row_index } }
			set_temp_variable = { event_cluster_member_trigger = 1 }
		}
		event_cluster_get_member_effective_min_tier = yes
		set_temp_variable = { event_cluster_member_tier = current_chaos_tier_value }
		event_cluster_get_member_base_participation_chance = yes
		event_cluster_member_check_availability = yes
		add_to_array = { array = temp_event_cluster_member_base_eligible_entries value = event_cluster_member_can_fire }
		add_to_array = { array = temp_event_cluster_member_effective_min_tier_entries value = event_cluster_member_effective_min_tier }
		add_to_array = { array = temp_event_cluster_member_status_entries value = event_cluster_member_status }
		add_to_array = { array = temp_event_cluster_member_unavailability_reason_entries value = event_cluster_member_unavailability_reason }
		add_to_array = { array = temp_event_cluster_member_trigger_entries value = event_cluster_member_trigger }
		add_to_array = { array = temp_event_cluster_member_base_chance_entries value = event_cluster_member_base_chance }
		add_to_array = { array = temp_event_cluster_member_final_chance_entries value = 0 }
		add_to_array = { array = temp_event_cluster_member_roll_entries value = 0 }
		if = {
			limit = { check_variable = { event_cluster_member_can_fire > 0 } }
			add_to_temp_variable = { event_cluster_eligible_member_count = 1 }
			set_temp_variable = { event_cluster_any_member_can_fire = 1 }
			if = {
				limit = {
					check_variable = { event_cluster_member_role = constant:event_cluster_member_role.optional }
					check_variable = { event_cluster_member_trigger < 1 }
				}
				add_to_temp_variable = { event_cluster_optional_eligible_count = 1 }
			}
		}
	}
	# High and Severe non-trigger members need another base-eligible row. The
	# second pass observes the complete first-pass set, so paired rows support
	# one another without circular evaluation.
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		if = {
			limit = {
				check_variable = { event_cluster_force_fire < 1 }
				check_variable = { temp_event_cluster_member_base_eligible_entries^i > 0 }
				check_variable = { temp_event_cluster_member_trigger_entries^i < 1 }
				check_variable = { temp_event_cluster_member_severity_entries^i > constant:event_cluster_member_danger.medium }
				check_variable = { temp_event_cluster_member_event_id_entries^num > 1 }
				check_variable = { event_cluster_eligible_member_count < 2 }
			}
			set_variable = { temp_event_cluster_member_base_eligible_entries^i = 0 }
			set_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.skipped_escalation_support }
			set_variable = { temp_event_cluster_member_unavailability_reason_entries^i = constant:event_system_event_unavailability_reason.none }
		}
	}
	set_temp_variable = { event_cluster_eligible_member_count = 0 }
	set_temp_variable = { event_cluster_optional_eligible_count = 0 }
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		if = {
			limit = { check_variable = { temp_event_cluster_member_base_eligible_entries^i > 0 } }
			add_to_temp_variable = { event_cluster_eligible_member_count = 1 }
			if = {
				limit = {
					check_variable = { temp_event_cluster_member_role_entries^i = constant:event_cluster_member_role.optional }
					check_variable = { temp_event_cluster_member_trigger_entries^i < 1 }
				}
				add_to_temp_variable = { event_cluster_optional_eligible_count = 1 }
			}
			if = {
				limit = { check_variable = { temp_event_cluster_member_trigger_entries^i > 0 } }
				set_temp_variable = { event_cluster_trigger_member_row_is_eligible = 1 }
			}
		}
	}
	if = {
		limit = { check_variable = { event_cluster_trigger_member_row_found < 1 } }
		set_temp_variable = { event_cluster_trigger_member_row_is_eligible = 0 }
	}
}

# Read-only cluster availability/chance breakdown for event-details callers.
# Inputs: event_cluster_id and either event_cluster_trigger_member_row_index or
# event_cluster_trigger_member_row_id. Outputs are temporary values only.
# No random roll, history write, queue write, or probability-state mutation is
# performed here.
event_cluster_get_current_activation_breakdown = {
	set_temp_variable = { event_cluster_activation_chance = 0 }
	set_temp_variable = { event_cluster_trigger_row_is_eligible = 0 }
	set_temp_variable = { event_cluster_activation_breakdown_eligible_count = 0 }
	set_temp_variable = { event_cluster_activation_breakdown_row_id = 0 }
	set_temp_variable = { event_cluster_activation_breakdown_available = 0 }
	set_temp_variable = { event_cluster_cluster_gates_passed = 0 }
	set_temp_variable = { event_cluster_breakdown_requested_row_id = 0 }
	set_temp_variable = { event_cluster_breakdown_requested_row_index = -1 }
	if = {
		limit = { has_variable = event_cluster_trigger_member_row_id }
		set_temp_variable = { event_cluster_breakdown_requested_row_id = event_cluster_trigger_member_row_id }
	}
	if = {
		limit = { has_variable = event_cluster_trigger_member_row_index }
		set_temp_variable = { event_cluster_breakdown_requested_row_index = event_cluster_trigger_member_row_index }
	}
	# This helper deliberately does not call event_cluster_ensure_probability_state
	# or any persistent initializer. The caller receives current registry data;
	# save migration is owned by normal cluster processing.
	load_event_cluster_members = yes
	if = {
		limit = {
			check_variable = { event_cluster_breakdown_requested_row_id > 0 }
		}
		set_temp_variable = { event_cluster_trigger_member_row_id = event_cluster_breakdown_requested_row_id }
	}
	else_if = {
		limit = {
			check_variable = { event_cluster_breakdown_requested_row_index > -1 }
			check_variable = {
				var = event_cluster_breakdown_requested_row_index
				value = 0
				compare = greater_than_or_equals
			}
			check_variable = {
				var = event_cluster_breakdown_requested_row_index
				value = temp_event_cluster_member_row_id_entries^num
				compare = less_than
			}
		}
		set_temp_variable = { event_cluster_trigger_member_row_id = temp_event_cluster_member_row_id_entries^event_cluster_breakdown_requested_row_index }
	}
	event_cluster_scan_members_for_activation = yes
	set_temp_variable = { event_cluster_activation_breakdown_eligible_count = event_cluster_eligible_member_count }
	set_temp_variable = { event_cluster_trigger_row_is_eligible = event_cluster_trigger_member_row_is_eligible }
	set_temp_variable = { event_cluster_activation_breakdown_row_id = event_cluster_trigger_member_row_id }
	event_cluster_check_cluster_gates = yes
	if = {
		limit = {
			check_variable = { global.event_cluster_probability_state_ready > 0 }
			check_variable = { event_cluster_cluster_gates_passed > 0 }
			check_variable = { event_cluster_trigger_member_row_is_eligible > 0 }
		}
		set_temp_variable = { event_cluster_activation_breakdown_available = 1 }
	}
	if = {
		limit = {
			check_variable = { global.event_cluster_probability_state_ready > 0 }
			check_variable = { event_cluster_cluster_gates_passed > 0 }
			check_variable = { event_cluster_trigger_member_row_is_eligible > 0 }
		}
		set_temp_variable = { event_cluster_activation_base_chance = constant:event_cluster_roll_chance_default.tier_0 }
		if = {
			limit = { check_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_final } }
			set_temp_variable = { event_cluster_activation_base_chance = constant:event_cluster_roll_chance_default.tier_final }
		}
		else_if = {
			limit = { check_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_4 } }
			set_temp_variable = { event_cluster_activation_base_chance = constant:event_cluster_roll_chance_default.tier_4 }
		}
		else_if = {
			limit = { check_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_3 } }
			set_temp_variable = { event_cluster_activation_base_chance = constant:event_cluster_roll_chance_default.tier_3 }
		}
		else_if = {
			limit = { check_variable = { current_chaos_tier_value = constant:chaos_meter_tier_id.tier_2 } }
			set_temp_variable = { event_cluster_activation_base_chance = constant:event_cluster_roll_chance_default.tier_2 }
		}
		else_if = {

# Compute the live catalogue By Roll metric. Each distinct primary trigger row
# is tested through the same two-pass event-system evaluator used by firing.
# Positive automatic-pool event weights are the weighting basis, while the
# trigger-specific activation chance is the value being averaged. Outputs are
# temporary and the helper performs no roll, history write, or queue mutation.
event_cluster_get_live_trigger_weighted_roll = {
	set_temp_variable = { event_cluster_live_trigger_weight_sum = 0 }
	set_temp_variable = { event_cluster_live_trigger_weighted_chance_sum = 0 }
	set_temp_variable = { event_cluster_live_trigger_weighted_roll = 0 }
	clear_array = temp_event_cluster_live_trigger_event_id_entries
	clear_array = temp_event_cluster_live_trigger_row_id_entries
	clear_array = temp_event_cluster_live_trigger_seen_event_id_entries
	set_temp_variable = { event_cluster_force_fire = 0 }
	load_event_cluster_members = yes
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		if = {
			limit = {
				check_variable = { temp_event_cluster_member_primary_trigger_entries^i = constant:event_cluster_member_primary_trigger.yes }
			}
			add_to_array = { array = temp_event_cluster_live_trigger_event_id_entries value = temp_event_cluster_member_event_id_entries^i }
			add_to_array = { array = temp_event_cluster_live_trigger_row_id_entries value = temp_event_cluster_member_row_id_entries^i }
		}
	}
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_live_trigger_event_id_entries^num
		value = trigger_index
		set_temp_variable = { event_cluster_live_trigger_event_id = temp_event_cluster_live_trigger_event_id_entries^trigger_index }
		set_temp_variable = { event_cluster_live_trigger_row_id = temp_event_cluster_live_trigger_row_id_entries^trigger_index }
		set_temp_variable = { event_cluster_trigger_event_id = event_cluster_live_trigger_event_id }
		set_temp_variable = { event_cluster_trigger_member_row_id = event_cluster_live_trigger_row_id }
		event_cluster_scan_members_for_activation = yes
		event_cluster_get_current_activation_breakdown = yes
		if = {
			limit = { check_variable = { event_cluster_activation_breakdown_available > 0 } }
			set_temp_variable = { random_event_selection_candidate_id = event_cluster_live_trigger_event_id }
			set_temp_variable = { random_event_selection_filter = constant:settings_event_filter.all }
			evaluate_random_event_selection_candidate = yes
			if = {
				limit = {
					check_variable = { random_event_selection_candidate_is_valid > 0 }
					check_variable = { random_event_selection_candidate_weight_scaled > 0 }
				}
				set_temp_variable = { event_cluster_live_trigger_seen = 0 }
				for_loop_effect = {
					start = 0
					end = temp_event_cluster_live_trigger_seen_event_id_entries^num
					value = seen_index
					if = {
						limit = {
							check_variable = {
								var = temp_event_cluster_live_trigger_seen_event_id_entries^seen_index
								value = event_cluster_live_trigger_event_id
								compare = equals
							}
						}
						set_temp_variable = { event_cluster_live_trigger_seen = 1 }
					}
				}
				if = {
					limit = { check_variable = { event_cluster_live_trigger_seen < 1 } }
					add_to_array = { array = temp_event_cluster_live_trigger_seen_event_id_entries value = event_cluster_live_trigger_event_id }
					set_temp_variable = { event_cluster_live_trigger_activation_chance = event_cluster_activation_chance }
					set_temp_variable = { event_cluster_live_trigger_weighted_value = random_event_selection_candidate_weight_scaled }
					add_to_temp_variable = { event_cluster_live_trigger_weight_sum = random_event_selection_candidate_weight_scaled }
					multiply_temp_variable = { event_cluster_live_trigger_weighted_value = event_cluster_live_trigger_activation_chance }
					add_to_temp_variable = { event_cluster_live_trigger_weighted_chance_sum = event_cluster_live_trigger_weighted_value }
				}
			}
		}
	}
	if = {
		limit = { check_variable = { event_cluster_live_trigger_weight_sum > 0 } }
		set_temp_variable = { event_cluster_live_trigger_weighted_roll = event_cluster_live_trigger_weighted_chance_sum }
		divide_temp_variable = { event_cluster_live_trigger_weighted_roll = event_cluster_live_trigger_weight_sum }
		round_temp_variable = event_cluster_live_trigger_weighted_roll
	}
}

evaluate_event_cluster_unavailability_reason = {
	set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.none }
	if = {
		limit = { check_variable = { event_cluster_can_fire < 1 } }
		set_temp_variable = { event_cluster_definition_index = -1 }
		get_event_cluster_definition_index = yes
		if = {
			limit = { check_variable = { event_cluster_definition_index < 0 } }
			set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.definition_missing }
		}
		else_if = {
			limit = { is_in_array = { global.disabled_event_clusters = event_cluster_id } }
			set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.disabled }
		}
		else = {
			get_event_cluster_type = yes
			get_event_cluster_unlock_tier = yes
			get_current_chaos_tier_value = yes
			if = {
				limit = { check_variable = { current_chaos_tier_value < event_cluster_unlock_tier } }
				set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.chaos_level }
			}
			else_if = {
				limit = {
					check_variable = { event_cluster_type_value = constant:event_cluster_type.one_time }
					is_in_array = { global.fired_event_clusters = event_cluster_id }
				}
				set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.already_fired }
			}
			else_if = {
				limit = {
					OR = {
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.wars } has_global_flag = event_cluster_wars_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.liberations } has_global_flag = event_cluster_liberations_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.diplomatic_panic } has_global_flag = event_cluster_diplomatic_panic_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.peace } has_global_flag = event_cluster_peace_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.natural_disasters } has_global_flag = event_cluster_natural_disasters_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.formables } has_global_flag = event_cluster_formables_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.economy_positive } has_global_flag = event_cluster_economy_positive_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.diseases } has_global_flag = event_cluster_diseases_cooldown }
						AND = { check_variable = { event_cluster_id = constant:event_cluster_id.random_stuff } has_global_flag = event_cluster_random_stuff_cooldown }
					}
				}
				set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.cooldown }
			}
			else = {
				set_temp_variable = { event_cluster_unavailability_reason = constant:event_cluster_unavailability_reason.no_available_members }
			}
		}
	}
}

# Optional members are ordered by severity band first, then by an independent
# random score inside that band. This keeps the accepted Low -> Severe order
# while retaining randomization among peers.
event_cluster_clear_optional_order_candidates = {

	clear_array = temp_event_cluster_optional_order_candidate_score_entries

	set_temp_variable = { event_cluster_optional_order_random_ceiling = constant:event_cluster_member_order.score_maximum }
	add_to_temp_variable = { event_cluster_optional_order_random_ceiling = 1 }

		var = event_cluster_optional_order_candidate_score
		min = constant:event_cluster_member_order.score_minimum
		max = event_cluster_optional_order_random_ceiling

	}
	add_to_array = { array = temp_event_cluster_optional_order_candidate_index_entries value = event_cluster_optional_order_candidate_index }

event_cluster_append_optional_order_candidates_by_score = {
	set_temp_variable = { event_cluster_optional_order_source_count = temp_event_cluster_optional_order_candidate_index_entries^num }
	set_temp_variable = { event_cluster_optional_order_sorted_count = 0 }
	while_loop_effect = {
		limit = { check_variable = { event_cluster_optional_order_sorted_count < event_cluster_optional_order_source_count } }
		set_temp_variable = { event_cluster_optional_order_best_index = -1 }
		set_temp_variable = { event_cluster_optional_order_scan_index = 0 }
		while_loop_effect = {
			limit = { check_variable = { event_cluster_optional_order_scan_index < event_cluster_optional_order_source_count } }
			if = {
				limit = { check_variable = { temp_event_cluster_optional_order_candidate_index_entries^event_cluster_optional_order_scan_index > -1 } }
				set_temp_variable = { event_cluster_optional_order_candidate_score = temp_event_cluster_optional_order_candidate_score_entries^event_cluster_optional_order_scan_index }
				set_temp_variable = { event_cluster_optional_order_candidate_index = temp_event_cluster_optional_order_candidate_index_entries^event_cluster_optional_order_scan_index }
				if = {
					limit = { check_variable = { event_cluster_optional_order_best_index < 0 } }
					set_temp_variable = { event_cluster_optional_order_best_index = event_cluster_optional_order_scan_index }
					set_temp_variable = { event_cluster_optional_order_best_score = event_cluster_optional_order_candidate_score }
					set_temp_variable = { event_cluster_optional_order_best_member_index = event_cluster_optional_order_candidate_index }
				}
				else_if = {
					limit = { check_variable = { event_cluster_optional_order_candidate_score < event_cluster_optional_order_best_score } }
					set_temp_variable = { event_cluster_optional_order_best_index = event_cluster_optional_order_scan_index }
					set_temp_variable = { event_cluster_optional_order_best_score = event_cluster_optional_order_candidate_score }
					set_temp_variable = { event_cluster_optional_order_best_member_index = event_cluster_optional_order_candidate_index }
				}
				else_if = {
					limit = {
						check_variable = { event_cluster_optional_order_candidate_score = event_cluster_optional_order_best_score }
						check_variable = {
							var = event_cluster_optional_order_candidate_index
							value = event_cluster_optional_order_best_member_index
							compare = less_than
						}
					}
					set_temp_variable = { event_cluster_optional_order_best_index = event_cluster_optional_order_scan_index }
					set_temp_variable = { event_cluster_optional_order_best_member_index = event_cluster_optional_order_candidate_index }
				}
			}
			add_to_temp_variable = { event_cluster_optional_order_scan_index = 1 }
		}
		if = {
			limit = { check_variable = { event_cluster_optional_order_best_index > -1 } }
			add_to_array = { array = temp_event_cluster_optional_order_index_entries value = event_cluster_optional_order_best_member_index }
			set_variable = { temp_event_cluster_optional_order_candidate_index_entries^event_cluster_optional_order_best_index = -1 }
			add_to_temp_variable = { event_cluster_optional_order_sorted_count = 1 }
		}
		else = {
			set_temp_variable = { event_cluster_optional_order_sorted_count = event_cluster_optional_order_source_count }
		}
	}
}

prepare_event_cluster_firing = {
	if = {
		limit = { NOT = { has_variable = event_cluster_force_fire } }
		set_temp_variable = { event_cluster_force_fire = 0 }
	}
	event_cluster_scan_members_for_activation = yes
	set_temp_variable = { event_cluster_member_count = temp_event_cluster_member_event_id_entries^num }
	set_temp_variable = { event_cluster_fired_member_count = 0 }
	set_temp_variable = { event_cluster_skipped_member_count = 0 }
	set_temp_variable = { event_cluster_optional_decay_count = 0 }
	set_temp_variable = { event_cluster_accepted_optional_count = 0 }

	# Trigger and required rows are guaranteed only after the shared evaluator
	# marks them base-eligible. Manual force changes cluster participation, not
	# event-system fireability.
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		set_temp_variable = { event_cluster_member_trigger = temp_event_cluster_member_trigger_entries^i }
		if = {
			limit = { check_variable = { temp_event_cluster_member_base_eligible_entries^i > 0 } }
			if = {
				limit = {
					OR = {
						check_variable = { temp_event_cluster_member_trigger_entries^i > 0 }
						check_variable = { temp_event_cluster_member_role_entries^i = constant:event_cluster_member_role.required }
					}
				}
				set_variable = { temp_event_cluster_member_final_chance_entries^i = constant:event_cluster_probability_bounds.member_maximum }
				set_variable = { temp_event_cluster_member_roll_entries^i = 0 }
				if = {
					limit = { check_variable = { event_cluster_force_fire > 0 } }
					set_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.manual }
				}
				else = {
					set_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.guaranteed }
				}
			}
		}
	}
	event_cluster_build_optional_participation_order = yes
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_optional_order_index_entries^num
		value = order_index
		set_temp_variable = { i = temp_event_cluster_optional_order_index_entries^order_index }
		set_temp_variable = { event_cluster_member_base_chance = temp_event_cluster_member_base_chance_entries^i }
		set_temp_variable = { event_cluster_member_role = temp_event_cluster_member_role_entries^i }
		set_temp_variable = { event_cluster_member_trigger = temp_event_cluster_member_trigger_entries^i }
		event_cluster_get_member_final_participation_chance = yes
		set_variable = { temp_event_cluster_member_final_chance_entries^i = event_cluster_member_final_chance }
		if = {
			limit = { check_variable = { event_cluster_force_fire > 0 } }
			set_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.manual }
			set_variable = { temp_event_cluster_member_roll_entries^i = 0 }
			add_to_temp_variable = { event_cluster_accepted_optional_count = 1 }
		}
		else = {
			set_temp_variable = { event_cluster_random_ceiling = constant:event_cluster_roll.maximum }
			add_to_temp_variable = { event_cluster_random_ceiling = 1 }
			set_temp_variable_to_random = {
				var = event_cluster_member_roll
				min = constant:event_cluster_roll.minimum
				max = event_cluster_random_ceiling
				integer = yes
			}
			set_variable = { temp_event_cluster_member_roll_entries^i = event_cluster_member_roll }
			if = {
				limit = {
					check_variable = {
						var = event_cluster_member_roll
						value = event_cluster_member_final_chance
						compare = less_than_or_equals
					}
				}
				set_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.fired }
				add_to_temp_variable = { event_cluster_accepted_optional_count = 1 }
				add_to_temp_variable = { event_cluster_optional_decay_count = 1 }
			}
			else = {
				set_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.skipped_participation }
			}
		}
	}
	set_temp_variable = { event_cluster_fired_member_count = 0 }
	set_temp_variable = { event_cluster_skipped_member_count = 0 }
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		set_variable = { temp_event_cluster_member_chance_entries^i = temp_event_cluster_member_final_chance_entries^i }
		if = {
			limit = {
				OR = {
					check_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.fired }
					check_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.guaranteed }
					check_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.manual }
				}
			}
			add_to_temp_variable = { event_cluster_fired_member_count = 1 }
		}
		else = {
			add_to_temp_variable = { event_cluster_skipped_member_count = 1 }
		}
	}
	build_event_cluster_firing_order = yes
}

event_cluster_append_order_candidates_by_score = {
    set_temp_variable = { event_cluster_order_source_count = temp_event_cluster_order_candidate_event_id_entries^num }
    set_temp_variable = { event_cluster_order_sorted_count = 0 }
    while_loop_effect = {
        limit = { check_variable = { event_cluster_order_sorted_count < event_cluster_order_source_count } }

        set_temp_variable = { event_cluster_order_best_index = -1 }
        set_temp_variable = { event_cluster_order_scan_index = 0 }
        while_loop_effect = {
            limit = { check_variable = { event_cluster_order_scan_index < event_cluster_order_source_count } }
            if = {
                limit = {
                    check_variable = {
                        var = temp_event_cluster_order_candidate_event_id_entries^event_cluster_order_scan_index
                        value = 0
                        compare = greater_than
                    }
                }
                set_temp_variable = { event_cluster_order_candidate_score = temp_event_cluster_order_candidate_score_entries^event_cluster_order_scan_index }
                set_temp_variable = { event_cluster_order_candidate_event_id = temp_event_cluster_order_candidate_event_id_entries^event_cluster_order_scan_index }

                if = {
                    limit = { check_variable = { event_cluster_order_best_index < 0 } }
                    set_temp_variable = { event_cluster_order_best_index = event_cluster_order_scan_index }
                    set_temp_variable = { event_cluster_order_best_score = event_cluster_order_candidate_score }
                    set_temp_variable = { event_cluster_order_best_event_id = event_cluster_order_candidate_event_id }
                }
                else_if = {
                    limit = { check_variable = { event_cluster_order_candidate_score < event_cluster_order_best_score } }
                    set_temp_variable = { event_cluster_order_best_index = event_cluster_order_scan_index }
                    set_temp_variable = { event_cluster_order_best_score = event_cluster_order_candidate_score }
                    set_temp_variable = { event_cluster_order_best_event_id = event_cluster_order_candidate_event_id }
                }
                else_if = {
                    limit = {
                        check_variable = { event_cluster_order_candidate_score = event_cluster_order_best_score }
                        check_variable = {
                            var = event_cluster_order_candidate_event_id
                            value = event_cluster_order_best_event_id
                            compare = less_than
                        }
                    }
                    set_temp_variable = { event_cluster_order_best_index = event_cluster_order_scan_index }
                    set_temp_variable = { event_cluster_order_best_score = event_cluster_order_candidate_score }
                    set_temp_variable = { event_cluster_order_best_event_id = event_cluster_order_candidate_event_id }
                }
            }
            add_to_temp_variable = { event_cluster_order_scan_index = 1 }
        }

        if = {
            limit = { check_variable = { event_cluster_order_best_index > -1 } }
            add_to_array = {
                array = temp_event_cluster_firing_order_event_id_entries
                value = temp_event_cluster_order_candidate_event_id_entries^event_cluster_order_best_index
            }
			add_to_array = {
				array = temp_event_cluster_firing_order_batch_index_entries
				value = temp_event_cluster_order_candidate_batch_index_entries^event_cluster_order_best_index
			}
            set_variable = { temp_event_cluster_order_candidate_event_id_entries^event_cluster_order_best_index = -1 }
            add_to_temp_variable = { event_cluster_order_sorted_count = 1 }
        }
        else = {
            set_temp_variable = { event_cluster_order_sorted_count = event_cluster_order_source_count }
        }
    }
}

build_event_cluster_firing_order = {
	clear_array = temp_event_cluster_firing_order_event_id_entries

	clear_array = temp_event_cluster_firing_order_role_entries
	set_temp_variable = { event_cluster_firing_order_batch_index = 1 }
	# The resolved trigger is dispatched first and synchronously.
	if = {
		limit = {
			check_variable = { event_cluster_trigger_member_row_index > -1 }
			OR = {
				check_variable = { temp_event_cluster_member_status_entries^event_cluster_trigger_member_row_index = constant:event_cluster_member_status.guaranteed }
				check_variable = { temp_event_cluster_member_status_entries^event_cluster_trigger_member_row_index = constant:event_cluster_member_status.manual }
			}
		}
		add_to_array = { array = temp_event_cluster_firing_order_event_id_entries value = temp_event_cluster_member_event_id_entries^event_cluster_trigger_member_row_index }
		add_to_array = { array = temp_event_cluster_firing_order_batch_index_entries value = event_cluster_firing_order_batch_index }
		add_to_array = { array = temp_event_cluster_firing_order_row_id_entries value = temp_event_cluster_member_row_id_entries^event_cluster_trigger_member_row_index }
		add_to_array = { array = temp_event_cluster_firing_order_trigger_entries value = constant:event_cluster_member_primary_trigger.yes }
		add_to_array = { array = temp_event_cluster_firing_order_role_entries value = temp_event_cluster_member_role_entries^event_cluster_trigger_member_row_index }
		add_to_temp_variable = { event_cluster_firing_order_batch_index = 1 }
	}
	# All other required rows follow the trigger in declaration order.
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_member_event_id_entries^num
		value = i
		if = {
			limit = {
				NOT = { check_variable = { i = event_cluster_trigger_member_row_index } }
				check_variable = { temp_event_cluster_member_role_entries^i = constant:event_cluster_member_role.required }
				OR = {
					check_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.guaranteed }
					check_variable = { temp_event_cluster_member_status_entries^i = constant:event_cluster_member_status.manual }
				}
			}
			add_to_array = { array = temp_event_cluster_firing_order_event_id_entries value = temp_event_cluster_member_event_id_entries^i }
			add_to_array = { array = temp_event_cluster_firing_order_batch_index_entries value = event_cluster_firing_order_batch_index }
			add_to_array = { array = temp_event_cluster_firing_order_row_id_entries value = temp_event_cluster_member_row_id_entries^i }
			add_to_array = { array = temp_event_cluster_firing_order_trigger_entries value = constant:event_cluster_member_primary_trigger.no }
			add_to_array = { array = temp_event_cluster_firing_order_role_entries value = temp_event_cluster_member_role_entries^i }
			add_to_temp_variable = { event_cluster_firing_order_batch_index = 1 }
		}
	}
	# Optional rows use the precomputed Low -> Severe randomized order.
	for_loop_effect = {
		start = 0
		end = temp_event_cluster_optional_order_index_entries^num
		value = order_index
		set_temp_variable = { event_cluster_optional_row_index = temp_event_cluster_optional_order_index_entries^order_index }
		if = {
			limit = {
				OR = {
					check_variable = { temp_event_cluster_member_status_entries^event_cluster_optional_row_index = constant:event_cluster_member_status.fired }
					check_variable = { temp_event_cluster_member_status_entries^event_cluster_optional_row_index = constant:event_cluster_member_status.manual }
				}
			}
			add_to_array = { array = temp_event_cluster_firing_order_event_id_entries value = temp_event_cluster_member_event_id_entries^event_cluster_optional_row_index }
			add_to_array = { array = temp_event_cluster_firing_order_batch_index_entries value = event_cluster_firing_order_batch_index }
			add_to_array = { array = temp_event_cluster_firing_order_row_id_entries value = temp_event_cluster_member_row_id_entries^event_cluster_optional_row_index }
			add_to_array = { array = temp_event_cluster_firing_order_trigger_entries value = constant:event_cluster_member_primary_trigger.no }
			add_to_array = { array = temp_event_cluster_firing_order_role_entries value = temp_event_cluster_member_role_entries^event_cluster_optional_row_index }
			add_to_temp_variable = { event_cluster_firing_order_batch_index = 1 }
		}
	}

# ============================================================================
# RANDOM STUFF - DYNAMIC WHOLE-POOL CLUSTER
# ============================================================================
# Random Stuff has no registered member rows. A successfully fired ordinary
# minor event supplies one automatic activation attempt. On success, a fresh
# uniform sample is drawn without replacement from the complete authoritative
# event-system pool. Event type, severity, and current selection weight never
# influence the draw after the positive-weight eligibility gate has passed.

		if = {

			event_cluster_prepare_runtime_context = yes
			if = {
				limit = {
					check_variable = { event_cluster_runtime_ready > 0 }
					NOT = { check_variable = { liberations_joint_dispatch_result = constant:liberations_joint_result.cancelled } }
				}
				# The trigger passed the authoritative evaluator before the activation
				# roll and is dispatched synchronously after runtime preflight. Do not
				# re-evaluate it after preflight: transactional preparations such as the
				# Liberations joint incident intentionally change the state being tested.
				# Delayed members receive their own immediate evaluator recheck.
				set_temp_variable = { event_fire_dispatched = 0 }
				set_temp_variable = { event_id = temp_event_cluster_firing_order_event_id_entries^0 }

					limit = { check_variable = { event_fire_dispatched > 0 } }

					# successful activation. The trigger is always guaranteed, while a
					# manual activation carries the distinct manual status.
					if = {
						limit = { check_variable = { event_cluster_activation_manual > 0 } }
						set_variable = { temp_event_cluster_member_status_entries^event_cluster_trigger_member_row_index = constant:event_cluster_member_status.manual }
					}
					else = {
						set_variable = { temp_event_cluster_member_status_entries^event_cluster_trigger_member_row_index = constant:event_cluster_member_status.guaranteed }
					}
					if = {
						limit = { check_variable = { event_cluster_activation_manual < 1 } }
						event_cluster_commit_successful_automatic_activation = yes
					}
					set_temp_variable = { event_cluster_pending_batch_id = 0 }
					if = {
						limit = { NOT = { has_variable = global.event_cluster_next_batch_id } }
						set_variable = { global.event_cluster_next_batch_id = 0 }
					}
					add_to_variable = { global.event_cluster_next_batch_id = 1 }
					set_temp_variable = { event_cluster_pending_batch_id = global.event_cluster_next_batch_id }
					set_temp_variable = { event_cluster_pending_batch_cluster_id = event_cluster_id }
					set_temp_variable = { event_cluster_pending_batch_manual = event_cluster_activation_manual }
					set_temp_variable = { event_cluster_pending_batch_optional_fired_count = 0 }
					set_temp_variable = { event_cluster_pending_batch_optional_eligible_count = event_cluster_optional_eligible_count }
					set_temp_variable = { event_cluster_pending_batch_fired_count = 1 }
					# Participation skips and base-ineligible rows are already known at
					# activation time. Delayed failures add to this baseline.
					set_temp_variable = { event_cluster_pending_batch_skipped_count = event_cluster_skipped_member_count }
					set_temp_variable = { event_cluster_pending_batch_history_sequence = 0 }
					record_events_log_cluster_entry = yes
					set_temp_variable = { event_cluster_pending_batch_history_sequence = event_cluster_sequence }
					mark_event_cluster_fired_state = yes
					on_event_cluster_global_pacing_update = yes
					set_temp_variable = { event_cluster_queue_start_index = 1 }
					event_cluster_queue_ordered_fired_members = yes

					set_temp_variable = { event_cluster_trigger_result = 1 }

                }
            }
            add_to_temp_variable = { events_log_cluster_detail_scan_index = 1 }
        }

        if = {
            limit = { check_variable = { events_log_cluster_detail_best_index > -1 } }
            events_log_cluster_append_detail_member_from_source = yes
            set_variable = { temp_events_log_cluster_detail_source_event_id_entries^events_log_cluster_detail_best_index = -1 }

        }

        }
    }
