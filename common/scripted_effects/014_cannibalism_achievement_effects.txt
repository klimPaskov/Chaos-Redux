	clr_global_flag = achievement_cannibalism_second_country_infected

	clr_global_flag = achievement_cannibalism_warlord_ever_formed
	clr_global_flag = achievement_cannibalism_silent_larder_ever_reached
	clr_global_flag = achievement_cannibalism_wendigo_merge_occurred
	clr_global_flag = achievement_cannibalism_all_wendigo_anchors_broken_prelock
	clr_global_flag = achievement_cannibalism_exploitation_visibility_open
	clr_global_flag = achievement_cannibalism_island_host_visibility_open
	clr_global_flag = achievement_cannibalism_evolution_ii_visibility_open
	clr_global_flag = achievement_cannibalism_convergence_visibility_open

cannibalism_achievement_mark_initial_host = {
	if = {
		limit = { NOT = { has_global_flag = achievement_cannibalism_first_host_recorded } }
		set_global_flag = achievement_cannibalism_first_host_recorded
		if = {
			limit = { is_ai = no }
			set_country_flag = achievement_cannibalism_first_host
			set_country_flag = achievement_cannibalism_event_participant
		}
	}
}

cannibalism_achievement_record_exploitation = {
	set_global_flag = achievement_cannibalism_exploitation_visibility_open
	set_country_flag = achievement_cannibalism_exploitation_ever
	set_country_flag = cannibalism_clean_route_disqualified
	if = {
		limit = { has_country_flag = achievement_cannibalism_exploitation_renounced }
		set_country_flag = achievement_cannibalism_exploitation_repeated
	}
	if = {
		limit = { has_country_flag = achievement_cannibalism_reinfection_faced }
		set_country_flag = achievement_cannibalism_reinfection_exploitation
	}
}

cannibalism_renounce_terror_exploitation = {
	if = {
		limit = {
			has_country_flag = achievement_cannibalism_exploitation_ever
			has_country_flag = cannibalism_public_court_martial_held
			NOT = { has_country_flag = cannibalism_terror_battalion_in_field }
		}
		cannibalism_clear_current_policy = yes
		clr_country_flag = cannibalism_ritual_response_appropriation
		clr_country_flag = cannibalism_network_response_exploitation
		set_country_flag = cannibalism_policy_open_emergency
		set_country_flag = cannibalism_humane_response_active
		set_country_flag = cannibalism_ritual_response_purge
		set_country_flag = cannibalism_ritual_prosecution_open
		if = {
			limit = { has_global_flag = cannibalism_evolution_ii_active }
			set_country_flag = cannibalism_network_response_disclosure
			set_country_flag = cannibalism_international_tracking_open
			set_country_flag = cannibalism_joint_suppression_open
		}
		set_variable = { cannibalism_policy = constant:cannibalism_policy.open_emergency }
		set_country_flag = achievement_cannibalism_exploitation_renounced
		if = { limit = { has_dynamic_modifier = { modifier = cannibalism_terror_battalion_deployed } } remove_dynamic_modifier = { modifier = cannibalism_terror_battalion_deployed } }
		cannibalism_refresh_country_penalty_variables = yes
	}
}

cannibalism_achievement_record_current_state_as_former_feeding_state = {
	if = {
		limit = {
			OR = {
				check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.feeding_state }
				check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.silent_larder }
			}
			NOT = { has_state_flag = cannibalism_former_feeding_state }
		}
		set_state_flag = cannibalism_former_feeding_state
		add_to_array = { array = global.cannibalism_former_feeding_states value = THIS }
		add_to_variable = { global.achievement_cannibalism_former_feeding_states = 1 }
	}
	if = {
		limit = { check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.silent_larder } }
		set_global_flag = achievement_cannibalism_silent_larder_ever_reached
	}
}

	set_global_flag = achievement_cannibalism_warlord_ever_formed
	add_to_variable = { global.achievement_cannibalism_warlords_formed = 1 }
	if = {
		limit = {
			has_country_flag = cannibalism_origin_island_host
			has_variable = cannibalism_warlord_origin_state
		}
		set_global_flag = achievement_cannibalism_island_host_visibility_open
		var:cannibalism_warlord_origin_state = { set_state_flag = cannibalism_former_island_host_origin }
	}
}

cannibalism_achievement_record_domestic_commune = {
	set_global_flag = achievement_cannibalism_commune_ever_formed

		if = {
			limit = { exists = yes }
			set_country_flag = achievement_cannibalism_commune_formed_domestically
		}
	}

# State scope. Credit is unique by the source country whose cell was removed.
cannibalism_achievement_record_cell_suppression = {
	CONTROLLER = { save_event_target_as = cannibalism_achievement_suppression_actor }
	if = {
		limit = {
			has_event_target = cannibalism_achievement_suppression_actor
			event_target:cannibalism_achievement_suppression_actor = { is_ai = no }
			has_variable = cannibalism_node_source_country
			country_exists = var:cannibalism_node_source_country
		}
		var:cannibalism_node_source_country = {
			if = {
				limit = { NOT = { has_country_flag = achievement_cannibalism_suppression_credit_claimed } }
				set_country_flag = achievement_cannibalism_suppression_credit_claimed
				event_target:cannibalism_achievement_suppression_actor = {
					add_to_variable = { achievement_cannibalism_countries_suppressed = 1 }
					set_country_flag = achievement_cannibalism_event_participant
					set_country_flag = achievement_cannibalism_decisive_aid
				}
			}
		}
	}
}

# State scope. A coastal or island recovery only counts if its controller keeps
# the required convoy/train and support reserves for every recovery stage.

	if = {
		limit = { cannibalism_state_recovery_supply_is_maintained = yes }
		set_state_flag = achievement_cannibalism_supply_recovery_maintained
	}
}

cannibalism_achievement_check_supply_recovery_tracking = {
	if = {
		limit = {
			has_state_flag = achievement_cannibalism_supply_recovery_maintained
			NOT = { cannibalism_state_recovery_supply_is_maintained = yes }
		}
		clr_state_flag = achievement_cannibalism_supply_recovery_maintained
	}
}

cannibalism_achievement_finalize_supply_recovery_tracking = {
	if = {
		limit = {
			has_state_flag = achievement_cannibalism_supply_recovery_maintained
			NOT = { has_state_flag = achievement_cannibalism_supply_recovery_counted }
			OR = { is_island_state = yes is_coastal = yes }
		}
		set_state_flag = achievement_cannibalism_supply_recovery_counted
		CONTROLLER = {
			if = {
				limit = { is_ai = no }
				add_to_variable = { achievement_cannibalism_isolated_nodes_recovered = 1 }
				set_country_flag = achievement_cannibalism_event_participant
			}
		}
	}
}

cannibalism_achievement_refresh_unified_continent_count = {
	set_variable = { achievement_cannibalism_feeding_continent_count = 0 }
	if = { limit = { any_controlled_state = { is_on_continent = europe OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
	if = { limit = { any_controlled_state = { is_on_continent = africa OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
	if = { limit = { any_controlled_state = { is_on_continent = asia OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
	if = { limit = { any_controlled_state = { is_on_continent = middle_east OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
	if = { limit = { any_controlled_state = { is_on_continent = north_america OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
	if = { limit = { any_controlled_state = { is_on_continent = south_america OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
	if = { limit = { any_controlled_state = { is_on_continent = australia OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } } } add_to_variable = { achievement_cannibalism_feeding_continent_count = 1 } }
}

cannibalism_achievement_process_current_actor = {
	if = {
		limit = { is_ai = no }
		set_country_flag = achievement_cannibalism_event_participant
		if = {
			limit = { has_country_flag = cannibalism_external_reinfection_active }
			set_country_flag = achievement_cannibalism_reinfection_faced
			if = {
				limit = {
					OR = {
						has_country_flag = cannibalism_policy_exploitation
						has_country_flag = cannibalism_ritual_response_appropriation
						has_country_flag = cannibalism_network_response_exploitation
					}
				}
				set_country_flag = achievement_cannibalism_reinfection_exploitation
			}
		}
		if = {
			limit = {
				is_cannibalism_warlord_country = yes
				has_country_flag = cannibalism_resistant_warlord
				has_global_flag = cannibalism_reveal_complete
			}
			if = {
				limit = { NOT = { has_variable = achievement_cannibalism_defiant_survival_due_date } }
				set_variable = { achievement_cannibalism_defiant_survival_due_date = global.date }
				add_to_variable = { achievement_cannibalism_defiant_survival_due_date = constant:cannibalism_achievement.defiant_survival_days }
			}
			else_if = {
				limit = {
					check_variable = {
						var = achievement_cannibalism_defiant_survival_due_date
						value = global.date
						compare = less_than_or_equals
					}
					is_subject = no
					has_capitulated = no
				}
				set_country_flag = achievement_cannibalism_defiant_warlord_survived
			}
		}
	}
	if = {
		limit = { is_cannibalism_unified_country = yes }
		cannibalism_achievement_refresh_unified_continent_count = yes
	}
}

cannibalism_achievement_process_pulse = {
	if = {
		limit = {
			check_variable = {
				var = global.cannibalism_infected_country_count
				value = constant:cannibalism_achievement.second_country_count
				compare = greater_than_or_equals
			}
		}
		set_global_flag = achievement_cannibalism_second_country_infected
	}
	if = {
		limit = {
			check_variable = {
				var = global.cannibalism_max_warlords
				value = constant:cannibalism_achievement.minimum_warlord_history
				compare = greater_than_or_equals
			}
		}
		set_global_flag = achievement_cannibalism_warlord_ever_formed
	}
	for_each_scope_loop = {
		array = global.cannibalism_node_states
		if = {
			limit = { has_state_flag = cannibalism_commune_state }
			cannibalism_achievement_record_domestic_commune = yes
		}
		if = {
			limit = { OR = { has_state_flag = cannibalism_feeding_state has_state_flag = cannibalism_silent_larder } }
			set_temp_variable = { cannibalism_new_state_stage = cannibalism_state_stage }
			cannibalism_achievement_record_current_state_as_former_feeding_state = yes
		}
	}
	for_each_scope_loop = {
		array = global.cannibalism_actor_countries
		cannibalism_achievement_process_current_actor = yes
	}
	if = {
		limit = {
			has_global_flag = cannibalism_convergence_broken_once
			NOT = { has_global_flag = cannibalism_reveal_complete }
			check_variable = {
				var = global.cannibalism_network_reach
				value = constant:cannibalism_achievement.pre_reveal_network_reach
				compare = less_than_or_equals
			}
		}
		for_each_scope_loop = {
			array = global.cannibalism_defeat_contributors
			if = {
				limit = {
					is_ai = no
					has_country_flag = achievement_cannibalism_likely_host_destroyed
				}
				set_country_flag = achievement_cannibalism_convergence_mission_won
			}
		}
	}
}

cannibalism_achievement_record_local_victory = {
	if = {
		limit = { is_ai = no }
		set_country_flag = achievement_cannibalism_event_participant
		if = {
			limit = { has_country_flag = cannibalism_external_reinfection_active }
			set_country_flag = achievement_cannibalism_reinfection_faced
			if = {
				limit = {
					NOT = { has_country_flag = achievement_cannibalism_reinfection_exploitation }
					NOT = { has_country_flag = cannibalism_policy_exploitation }
				}
				set_country_flag = achievement_cannibalism_reinfection_defeated
			}
		}
	}
}

cannibalism_achievement_record_global_victory = {

		array = global.cannibalism_defeat_contributors
		if = {
			limit = { is_ai = no }
			set_country_flag = achievement_cannibalism_event_participant
			if = {
				limit = {
					has_global_flag = cannibalism_convergence_broken_once
					NOT = { has_global_flag = cannibalism_reveal_complete }
					has_country_flag = achievement_cannibalism_likely_host_destroyed
				}
				set_country_flag = achievement_cannibalism_convergence_mission_won
			}
		}
	}
}

cannibalism_pay_international_response_cost = {
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_manpower_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_manpower = cannibalism_international_negative_cost
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_command_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_command_power = cannibalism_international_negative_cost
	if = {
		limit = { check_variable = { cannibalism_international_navy_experience_cost > 0 } }
		set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_navy_experience_cost }
		multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
		add_navy_experience = cannibalism_international_negative_cost
	}
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_infantry_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_equipment_to_stockpile = { type = infantry_equipment amount = cannibalism_international_negative_cost }
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_support_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_equipment_to_stockpile = { type = support_equipment_1 amount = cannibalism_international_negative_cost }
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_truck_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_equipment_to_stockpile = { type = motorized_equipment_1 amount = cannibalism_international_negative_cost }
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_train_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_equipment_to_stockpile = { type = train_equipment_1 amount = cannibalism_international_negative_cost }
	set_temp_variable = { cannibalism_international_negative_cost = cannibalism_international_convoy_cost }
	multiply_temp_variable = { cannibalism_international_negative_cost = -1 }
	add_equipment_to_stockpile = { type = convoy_1 amount = cannibalism_international_negative_cost }
}

cannibalism_execute_joint_suppression = {
	if = {
		limit = {
			cannibalism_can_pay_joint_suppression_cost = yes
			has_event_target = cannibalism_joint_suppression_target
			event_target:cannibalism_joint_suppression_target = { cannibalism_is_valid_joint_suppression_target = yes }
		}
		set_temp_variable = { cannibalism_international_manpower_cost = constant:cannibalism_international_response_cost.joint_manpower }
		set_temp_variable = { cannibalism_international_command_cost = constant:cannibalism_international_response_cost.joint_command }
		set_temp_variable = { cannibalism_international_infantry_cost = constant:cannibalism_international_response_cost.joint_infantry }
		set_temp_variable = { cannibalism_international_support_cost = constant:cannibalism_international_response_cost.joint_support }
		set_temp_variable = { cannibalism_international_truck_cost = constant:cannibalism_international_response_cost.joint_trucks }
		set_temp_variable = { cannibalism_international_train_cost = 0 }
		set_temp_variable = { cannibalism_international_convoy_cost = 0 }
		set_temp_variable = { cannibalism_international_navy_experience_cost = 0 }
		cannibalism_pay_international_response_cost = yes
		event_target:cannibalism_joint_suppression_target = {
			cannibalism_expose_detention_exploitation_in_current_country = yes

			add_to_variable = { cannibalism_command_integrity = constant:cannibalism_international_response_cost.joint_target_integrity_gain }
			cannibalism_clamp_current_country_meters = yes
			cannibalism_refresh_country_penalty_variables = yes
			if = {
				limit = { NOT = { has_country_flag = achievement_cannibalism_joint_suppression_counted } }
				set_country_flag = achievement_cannibalism_joint_suppression_counted
				ROOT = { add_to_variable = { achievement_cannibalism_countries_suppressed = 1 } }
			}

cannibalism_refresh_current_country_reconstruction_idea = {
	if = { limit = { has_idea = cannibalism_liberated_feeding_states } remove_ideas = cannibalism_liberated_feeding_states }
	if = { limit = { has_idea = cannibalism_identification_and_burial_emergency } remove_ideas = cannibalism_identification_and_burial_emergency }
	if = { limit = { has_idea = cannibalism_broken_military_trust } remove_ideas = cannibalism_broken_military_trust }
	if = { limit = { has_idea = cannibalism_rebuilt_supply_discipline } remove_ideas = cannibalism_rebuilt_supply_discipline }
	if = {
		limit = {
			has_country_flag = cannibalism_international_compact_member
			NOT = { has_idea = cannibalism_permanent_vigilance }
		}
		add_ideas = cannibalism_permanent_vigilance
	}
	else_if = {
		limit = {
			any_owned_state = {
				cannibalism_state_is_former_feeding_state = yes
				NOT = { has_state_flag = cannibalism_aftermath_identification_complete }
			}
		}
		add_ideas = cannibalism_liberated_feeding_states
	}
	else_if = {
		limit = {
			any_owned_state = {
				cannibalism_state_is_former_feeding_state = yes
				has_state_flag = cannibalism_aftermath_identification_complete
				NOT = { has_state_flag = cannibalism_aftermath_institutions_complete }
			}
		}
		add_ideas = cannibalism_identification_and_burial_emergency
	}
	else_if = {
		limit = {
			any_owned_state = {
				cannibalism_state_is_former_feeding_state = yes
				has_state_flag = cannibalism_aftermath_institutions_complete
				NOT = { has_state_flag = cannibalism_aftermath_reconstruction_complete }
			}
		}
		add_ideas = cannibalism_broken_military_trust
	}
	else_if = {
		limit = {
			has_country_flag = cannibalism_reconstruction_participant
			NOT = { cannibalism_country_has_unfinished_aftermath_state = yes }
		}
		add_ideas = cannibalism_rebuilt_supply_discipline
	}
}

	set_variable = { global.achievement_cannibalism_reconstructed_states = 0 }
	set_variable = { global.achievement_cannibalism_reconstruction_required = global.cannibalism_former_feeding_states^num }
	clear_array = global.cannibalism_reconstruction_participants
	for_each_scope_loop = {
		array = global.cannibalism_defeat_contributors
		if = {
			limit = { exists = yes }
			set_country_flag = cannibalism_reconstruction_participant
			if = {
				limit = { NOT = { is_in_array = { array = global.cannibalism_reconstruction_participants value = THIS } } }
				add_to_array = { array = global.cannibalism_reconstruction_participants value = THIS }
			}
			cannibalism_refresh_current_country_reconstruction_idea = yes
		}
	}
	for_each_scope_loop = {
		array = global.cannibalism_former_feeding_states
		OWNER = {
			if = {
				limit = { exists = yes NOT = { is_cannibalism_country = yes } }
				set_country_flag = cannibalism_reconstruction_participant
				if = {
					limit = { NOT = { is_in_array = { array = global.cannibalism_reconstruction_participants value = THIS } } }
					add_to_array = { array = global.cannibalism_reconstruction_participants value = THIS }
				}
				cannibalism_refresh_current_country_reconstruction_idea = yes
			}
		}
	}
}

			cannibalism_store_captured_warlord_outcome = yes
			if = {
				limit = { NOT = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.escape } } }
				retire_country_leader = yes
			}
		}
		if = {
			limit = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.public_trial } }
			add_stability = constant:cannibalism_character_outcome_result.trial_stability
			add_political_power = constant:cannibalism_character_outcome_result.trial_political_power
		}
		else_if = {
			limit = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.international_tribunal } }
			add_stability = constant:cannibalism_character_outcome_result.tribunal_stability
			add_political_power = constant:cannibalism_character_outcome_result.tribunal_political_power
		}
		else_if = {
			limit = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.local_prosecution } }
			add_stability = constant:cannibalism_character_outcome_result.local_stability
			add_political_power = constant:cannibalism_character_outcome_result.local_political_power
		}
		else_if = {
			limit = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.intelligence_cooperation } }
			add_political_power = constant:cannibalism_character_outcome_result.cooperation_political_power
			add_command_power = constant:cannibalism_character_outcome_result.cooperation_command_power
			set_country_flag = cannibalism_captured_warlord_cooperation
		}
		else_if = {
			limit = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.execution } }
			add_stability = constant:cannibalism_character_outcome_result.execution_stability
			add_war_support = constant:cannibalism_character_outcome_result.execution_war_support
		}
		else_if = {
			limit = { check_variable = { cannibalism_character_outcome_selection = constant:cannibalism_character_outcome.escape } }
			add_stability = constant:cannibalism_character_outcome_result.escape_stability
			set_country_flag = cannibalism_captured_warlord_escaped
		}
	}
}
