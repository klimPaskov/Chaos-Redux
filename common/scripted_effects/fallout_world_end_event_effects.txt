# ============================================================================
# CHAOS REDUX - FALLOUT LIVING-WORLD EVENT EFFECTS
# ============================================================================
# Dormant, idempotent scheduler initialization for the reveal timeline, stable
# successor registry, country runtime rows, orientation receipts, cooldown
# slots, a hard family-repeat veto, and generation-bound ordinary-opening, arc,
# delayed-result, bilateral, cancellation, cleanup, and exact dispatch-issuance
# receipts. No gameplay caller selects a candidate or reserves an event.
# ============================================================================

# Engine contract for public transaction APIs: the outer caller must create
# every documented temporary output before invoking the scripted effect. The
# effect overwrites those existing temporary variables with its final receipt.

# Supported scope: COUNTRY. Input: fallout_event_scheduler_error_to_register.
fallout_event_register_scheduler_error_if_unowned = {
	if = {
		limit = {
			check_variable = { fallout_event_scheduler_error_to_register > constant:fallout_event_scheduler_error.none }
			check_variable = { fallout_event_scheduler_error_to_register < constant:fallout_event_scheduler_error.upper_bound }
			NOT = { has_global_flag = fallout_event_scheduler_error }
		}
		set_global_flag = fallout_event_scheduler_error
		set_variable = { global.fallout_event_scheduler_error_code = fallout_event_scheduler_error_to_register }
		if = {
			limit = { NOT = { has_variable = global.fallout_event_scheduler_error_count } }
			set_variable = { global.fallout_event_scheduler_error_count = constant:fallout_event_runtime.zero }
		}
		add_to_variable = { global.fallout_event_scheduler_error_count = constant:fallout_event_runtime.one }
	}
}

# Supported scope: COUNTRY coordinator. This bounded one-time pass touches only
# the frozen registry and never constructs a game-wide country pool.
fallout_event_promote_registry_empty_ordinary_receipts = {
	if = {
		limit = {
			NOT = { has_global_flag = fallout_event_ordinary_receipt_registry_ready }
			NOT = { has_global_flag = fallout_event_scheduler_activation_approved }
			NOT = { has_global_flag = fallout_event_scheduler_active }
			NOT = { has_global_flag = fallout_event_scheduler_error }
			has_variable = global.fallout_transition_generation
			has_variable = global.fallout_event_schema_version
			has_variable = global.fallout_event_registry_schema_version
			has_variable = global.fallout_event_registry_generation
			has_variable = global.fallout_event_registry_count
			check_variable = { global.fallout_event_schema_version = constant:fallout_event_schema.version }
			OR = {
				check_variable = { global.fallout_event_registry_schema_version = constant:fallout_event_registry_schema.version }
				check_variable = { global.fallout_event_registry_schema_version = constant:fallout_event_registry_schema.previous_without_survival_binding }
			}
			check_variable = { var = global.fallout_event_registry_generation value = global.fallout_transition_generation compare = equals }
			check_variable = { global.fallout_event_registry_count > constant:fallout_event_runtime.zero }
			check_variable = { var = global.fallout_event_registry_count value = global.fallout_event_registry_countries^num compare = equals }
			check_variable = { var = global.fallout_event_registry_count value = global.fallout_event_registry_generation_entries^num compare = equals }
			check_variable = { var = global.fallout_event_registry_count value = global.fallout_event_registry_index_entries^num compare = equals }
		}
		set_temp_variable = { fallout_event_ordinary_promotion_proven_count = constant:fallout_event_runtime.zero }
		for_each_loop = {
			array = global.fallout_event_registry_index_entries
			value = fallout_event_ordinary_promotion_stored_index
			index = fallout_event_ordinary_promotion_index
			set_temp_variable = { fallout_event_ordinary_promotion_context_loaded = constant:fallout_event_runtime.zero }
			if = {
				limit = {
					check_variable = { var = fallout_event_ordinary_promotion_stored_index value = fallout_event_ordinary_promotion_index compare = equals }
					check_variable = { var = global.fallout_event_registry_generation_entries^fallout_event_ordinary_promotion_index value = global.fallout_event_registry_generation compare = equals }
				}
				var:global.fallout_event_registry_countries^fallout_event_ordinary_promotion_index = {
					if = {
						limit = {
							has_country_flag = fallout_event_registry_member
							has_country_flag = fallout_event_runtime_initialized
							check_variable = { var = fallout_event_registry_index value = fallout_event_ordinary_promotion_index compare = equals }
							check_variable = { var = fallout_event_registry_generation value = global.fallout_event_registry_generation compare = equals }
							check_variable = { var = fallout_event_runtime_generation value = global.fallout_event_registry_generation compare = equals }
						}
						fallout_event_promote_empty_ordinary_receipt = yes
						if = {
							limit = { fallout_event_ordinary_receipt_is_consistent = yes }
							set_temp_variable = { fallout_event_ordinary_promotion_context_loaded = constant:fallout_event_runtime.one }
						}
					}
				}
			}
			if = {
				limit = { check_variable = { fallout_event_ordinary_promotion_context_loaded = constant:fallout_event_runtime.one } }
				add_to_temp_variable = { fallout_event_ordinary_promotion_proven_count = constant:fallout_event_runtime.one }
			}
		}
		if = {
			limit = {
				check_variable = { var = fallout_event_ordinary_promotion_proven_count value = global.fallout_event_registry_count compare = equals }
			}
			set_variable = { global.fallout_event_ordinary_receipt_registry_generation = global.fallout_event_registry_generation }
			set_variable = { global.fallout_event_ordinary_receipt_registry_day = global.num_days }
			set_global_flag = fallout_event_ordinary_receipt_registry_ready
		}
		else = {
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.ordinary_receipt_mismatch }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
	}
}

# Supported scope: COUNTRY. Inputs use the fallout_event_requested_dispatch_*
# temporary variables. Output: fallout_event_dispatch_write_accepted. The outer
# caller must pre-seed that output. The ready flag is written last as the commit
# marker. This records routing data only and never fires an event.
fallout_event_write_dispatch_envelope = {
	set_temp_variable = { fallout_event_dispatch_write_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			fallout_event_registry_country_row_is_current = yes
			fallout_event_dispatch_envelope_is_consistent = yes
			exists = yes
			NOT = { has_global_flag = fallout_event_scheduler_error }
			NOT = { has_country_flag = fallout_event_dispatch_ready }
			check_variable = { fallout_event_requested_dispatch_source > constant:fallout_event_dispatch_source.none }
			check_variable = { fallout_event_requested_dispatch_source < constant:fallout_event_dispatch_source.upper_bound }
			check_variable = { fallout_event_requested_dispatch_ticket > constant:fallout_event_runtime.no_ticket }
			check_variable = { fallout_event_requested_dispatch_mode > constant:fallout_event_dispatch_mode.none }
			check_variable = { fallout_event_requested_dispatch_mode < constant:fallout_event_dispatch_mode.upper_bound }
			check_variable = { fallout_event_requested_dispatch_event_token > constant:fallout_event_runtime.zero }
			check_variable = { fallout_event_requested_dispatch_branch > constant:fallout_event_runtime.zero }
		}
		set_temp_variable = { fallout_event_subject_type_to_check = fallout_event_requested_dispatch_target_type }
		set_temp_variable = { fallout_event_subject_value_to_check = fallout_event_requested_dispatch_target }
		if = {
			limit = {
				OR = {
					AND = {
						check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.none }
						check_variable = { fallout_event_subject_value_to_check = constant:fallout_event_runtime.zero }
					}
					fallout_event_subject_shape_is_valid = yes
				}
			}
			set_variable = { fallout_event_dispatch_source = fallout_event_requested_dispatch_source }
			set_variable = { fallout_event_dispatch_ticket = fallout_event_requested_dispatch_ticket }
			set_variable = { fallout_event_dispatch_generation = fallout_event_runtime_generation }
			set_variable = { fallout_event_dispatch_mode = fallout_event_requested_dispatch_mode }
			set_variable = { fallout_event_dispatch_event_token = fallout_event_requested_dispatch_event_token }
			set_variable = { fallout_event_dispatch_branch = fallout_event_requested_dispatch_branch }
			set_variable = { fallout_event_dispatch_target_type = fallout_event_requested_dispatch_target_type }
			set_variable = { fallout_event_dispatch_target = fallout_event_requested_dispatch_target }
			set_country_flag = fallout_event_dispatch_ready
			if = {
				limit = { fallout_event_registry_country_row_is_current = yes }
				set_temp_variable = { fallout_event_dispatch_write_accepted = constant:fallout_event_runtime.one }
			}
			else = {
				fallout_event_clear_dispatch_envelope = yes
			}
		}
	}
}

# Supported scope: COUNTRY. Inputs: requested dispatch source and ticket.
# Output: fallout_event_transaction_accepted. The outer caller must pre-seed
# that output. This helper accepts only a bilateral-response source. Future
# ordinary and delayed consumers must use their atomic issued wrappers.
fallout_event_acknowledge_dispatch_envelope = {
	set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			fallout_event_registry_country_row_is_current = yes
			fallout_event_dispatch_envelope_is_consistent = yes
			has_country_flag = fallout_event_dispatch_ready
			fallout_event_dispatch_issue_receipt_is_current = yes
			check_variable = { fallout_event_requested_dispatch_source = constant:fallout_event_dispatch_source.bilateral_response }
			check_variable = { var = fallout_event_dispatch_source value = fallout_event_requested_dispatch_source compare = equals }
			check_variable = { var = fallout_event_dispatch_ticket value = fallout_event_requested_ticket compare = equals }
			check_variable = { var = fallout_event_dispatch_generation value = fallout_event_runtime_generation compare = equals }
		}
		fallout_event_clear_dispatch_envelope = yes
		set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one }
	}
}

# Supported scope: COUNTRY. Output: fallout_event_dispatch_issue_accepted. The
# outer caller must pre-seed that output. The immutable envelope identity is
# copied first and its issued flag is written last. The dynamic event command
# runs only after that receipt proves current. Future delayed and ordinary
# consumers must use atomic issued wrappers. The bilateral contract retains
# explicit acknowledgement until its reciprocal terminalizers gain that wrapper.
fallout_event_issue_dispatch_envelope = {
	set_temp_variable = { fallout_event_dispatch_issue_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			fallout_event_dispatch_issuance_is_approved = yes
			fallout_event_registry_country_row_is_current = yes
			has_country_flag = fallout_event_dispatch_ready
			fallout_event_dispatch_issue_receipt_is_absent = yes
		}
		set_variable = { fallout_event_dispatch_issued_source = fallout_event_dispatch_source }
		set_variable = { fallout_event_dispatch_issued_ticket = fallout_event_dispatch_ticket }
		set_variable = { fallout_event_dispatch_issued_generation = fallout_event_dispatch_generation }
		set_variable = { fallout_event_dispatch_issued_mode = fallout_event_dispatch_mode }
		set_variable = { fallout_event_dispatch_issued_event_token = fallout_event_dispatch_event_token }
		set_variable = { fallout_event_dispatch_issued_branch = fallout_event_dispatch_branch }
		set_variable = { fallout_event_dispatch_issued_target_type = fallout_event_dispatch_target_type }
		set_variable = { fallout_event_dispatch_issued_target = fallout_event_dispatch_target }
		set_variable = { fallout_event_dispatch_issued_date = global.date }
		set_variable = { fallout_event_dispatch_issued_day = global.num_days }
		set_country_flag = fallout_event_dispatch_issued
		if = {
			limit = {
				fallout_event_dispatch_issue_receipt_is_current = yes
				fallout_event_dispatch_envelope_matches_transaction = yes
			}
			set_temp_variable = { fallout_event_dispatch_issue_accepted = constant:fallout_event_runtime.one }
			meta_effect = {
				text = { country_event = { id = chaosx.fallout.[FALLOUT_EVENT_ID] } }
				FALLOUT_EVENT_ID = "[?fallout_event_dispatch_event_token|.0]"
			}
		}
		else = {
			fallout_event_clear_dispatch_issue_receipt = yes
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.dispatch_issue_receipt_mismatch }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
	}
}

	if = {
		limit = { fallout_event_map_return_receipts_are_current = yes }

		subtract_from_variable = { global.fallout_event_timeline_elapsed_days = global.fallout_event_timeline_start_day }

				limit = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.second_world_end_days } }
				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.open_continuation }

				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.second_world }

				limit = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.new_states_end_days } }
				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.soot_retreat }
			}
			else_if = {
				limit = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.rival_orders_end_days } }
				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.new_states }
			}
			else_if = {
				limit = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.consolidation_end_days } }
				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.rival_orders }

				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.consolidation }

				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.first_winter_year }

				set_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.first_season }

		}
	}

# Supported scope: COUNTRY. One fixed ordinary opening receipt is reconciled
# before compact transaction rows. It writes only a data envelope and leaves
# dynamic event issuance to the single host-owned coordinator.
fallout_event_reconcile_ordinary_receipt = {
	if = {
		limit = { has_country_flag = fallout_event_ordinary_receipt_pending }
		if = {
			limit = { fallout_event_ordinary_receipt_is_consistent = yes }
			set_temp_variable = { fallout_event_subject_type_to_check = fallout_event_ordinary_target_type }
			set_temp_variable = { fallout_event_subject_value_to_check = fallout_event_ordinary_target }
			set_temp_variable = { fallout_event_reconciled_ordinary_target_is_current = constant:fallout_event_runtime.zero }
			if = {
				limit = {
					OR = {
						AND = {
							check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.none }
							check_variable = { fallout_event_subject_value_to_check = constant:fallout_event_runtime.zero }
						}
						fallout_event_subject_is_current = yes
					}
				}
				set_temp_variable = { fallout_event_reconciled_ordinary_target_is_current = constant:fallout_event_runtime.one }
			}
			if = {
				limit = { NOT = { exists = yes } }
				if = {
					limit = { NOT = { fallout_event_ordinary_receipt_has_issued_dispatch = yes } }
					set_temp_variable = { fallout_event_requested_ticket = fallout_event_ordinary_ticket }
					set_temp_variable = { fallout_event_requested_cancellation_reason = constant:fallout_event_cancellation_reason.owner_lost }
					set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
					fallout_event_cancel_ordinary_receipt = yes
				}
			}
			else_if = {
				limit = { check_variable = { fallout_event_reconciled_ordinary_target_is_current = constant:fallout_event_runtime.zero } }
				if = {
					limit = { NOT = { fallout_event_ordinary_receipt_has_issued_dispatch = yes } }
					set_temp_variable = { fallout_event_requested_ticket = fallout_event_ordinary_ticket }
					set_temp_variable = { fallout_event_requested_cancellation_reason = constant:fallout_event_cancellation_reason.target_lost }
					set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
					fallout_event_cancel_ordinary_receipt = yes
				}
			}
			else_if = {
				limit = {
					exists = yes
					NOT = { has_country_flag = fallout_event_dispatch_ready }
					check_variable = { var = fallout_event_ordinary_due_day value = global.num_days compare = less_than_or_equals }
				}
				set_temp_variable = { fallout_event_requested_dispatch_source = constant:fallout_event_dispatch_source.ordinary_event }
				set_temp_variable = { fallout_event_requested_dispatch_ticket = fallout_event_ordinary_ticket }
				if = {
					limit = { is_ai = yes }
					set_temp_variable = { fallout_event_requested_dispatch_mode = constant:fallout_event_dispatch_mode.hidden_ai }
					set_temp_variable = { fallout_event_requested_dispatch_event_token = fallout_event_ordinary_ai_event_token }
				}
				else = {
					set_temp_variable = { fallout_event_requested_dispatch_mode = constant:fallout_event_dispatch_mode.human_visible }
					set_temp_variable = { fallout_event_requested_dispatch_event_token = fallout_event_ordinary_human_event_token }
				}
				set_temp_variable = { fallout_event_requested_dispatch_branch = fallout_event_ordinary_branch }
				set_temp_variable = { fallout_event_requested_dispatch_target_type = fallout_event_ordinary_target_type }
				set_temp_variable = { fallout_event_requested_dispatch_target = fallout_event_ordinary_target }
				set_temp_variable = { fallout_event_dispatch_write_accepted = constant:fallout_event_runtime.zero }
				fallout_event_write_dispatch_envelope = yes
				if = {
					limit = { check_variable = { fallout_event_dispatch_write_accepted = constant:fallout_event_runtime.zero } }
					set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.ordinary_receipt_mismatch }
					fallout_event_register_scheduler_error_if_unowned = yes
				}
			}
		}
		else = {
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.ordinary_receipt_mismatch }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
	}
}

# scans local delayed identities before a due row may write a data-only envelope.

				limit = {
					OR = {

				set_temp_variable = { fallout_event_subject_value_to_check = fallout_event_delayed_target_entries^fallout_event_delayed_reconcile_cursor }
				set_temp_variable = { fallout_event_reconciled_delayed_target_is_current = constant:fallout_event_runtime.zero }
				if = {
					limit = {
						OR = {
							AND = {
								check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.none }
								check_variable = { fallout_event_subject_value_to_check = constant:fallout_event_runtime.zero }
							}
							fallout_event_subject_is_current = yes
						}
					}
					set_temp_variable = { fallout_event_reconciled_delayed_target_is_current = constant:fallout_event_runtime.one }
				}
				if = {
					limit = { NOT = { exists = yes } }
					if = {
						limit = { NOT = { fallout_event_requested_delayed_ticket_has_issued_dispatch = yes } }
						set_temp_variable = { fallout_event_requested_cancellation_reason = constant:fallout_event_cancellation_reason.owner_lost }
						set_temp_variable = { fallout_event_requested_outcome = constant:fallout_event_outcome_type.aborted_after_choice }
						set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
						fallout_event_cancel_delayed_result = yes
					}
				}
				else_if = {
					limit = { check_variable = { fallout_event_reconciled_delayed_target_is_current = constant:fallout_event_runtime.zero } }
					if = {
						limit = { NOT = { fallout_event_requested_delayed_ticket_has_issued_dispatch = yes } }
						set_temp_variable = { fallout_event_requested_cancellation_reason = constant:fallout_event_cancellation_reason.target_lost }
						set_temp_variable = { fallout_event_requested_outcome = constant:fallout_event_outcome_type.aborted_after_choice }
						set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
						fallout_event_cancel_delayed_result = yes
					}
				}
				else_if = {

						check_variable = { var = fallout_event_delayed_due_day_entries^fallout_event_delayed_reconcile_cursor value = global.num_days compare = less_than_or_equals }
					}
					set_temp_variable = { fallout_event_requested_dispatch_source = constant:fallout_event_dispatch_source.delayed_ticket }
					set_temp_variable = { fallout_event_requested_dispatch_ticket = fallout_event_requested_ticket }
					if = {
						limit = { is_ai = yes }

						set_temp_variable = { fallout_event_requested_dispatch_event_token = fallout_event_delayed_ai_event_token_entries^fallout_event_delayed_reconcile_cursor }
					}
					else = {
						set_temp_variable = { fallout_event_requested_dispatch_mode = constant:fallout_event_dispatch_mode.human_visible }
						set_temp_variable = { fallout_event_requested_dispatch_event_token = fallout_event_delayed_human_event_token_entries^fallout_event_delayed_reconcile_cursor }
					}
					set_temp_variable = { fallout_event_requested_dispatch_branch = fallout_event_delayed_branch_entries^fallout_event_delayed_reconcile_cursor }
					set_temp_variable = { fallout_event_requested_dispatch_target_type = fallout_event_delayed_target_type_entries^fallout_event_delayed_reconcile_cursor }
					set_temp_variable = { fallout_event_requested_dispatch_target = fallout_event_delayed_target_entries^fallout_event_delayed_reconcile_cursor }
					set_temp_variable = { fallout_event_dispatch_write_accepted = constant:fallout_event_runtime.zero }
					fallout_event_write_dispatch_envelope = yes
					if = {
						limit = { check_variable = { fallout_event_dispatch_write_accepted = constant:fallout_event_runtime.zero } }
						set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.delayed_receipt_mismatch }
						fallout_event_register_scheduler_error_if_unowned = yes
					}
				}
			}
			else_if = {

						check_variable = { fallout_event_reconciled_delayed_status = constant:fallout_event_queue_status.cancelled }
					}
				}
				set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }

			}
			else_if = {

				set_country_flag = fallout_event_delayed_cleanup_pending
				if = {
					limit = {

					}
					set_temp_variable = { fallout_event_requested_dispatch_source = constant:fallout_event_dispatch_source.delayed_ticket }
					set_temp_variable = { fallout_event_requested_dispatch_ticket = fallout_event_requested_ticket }

	}

# Inputs: current bilateral reconcile index, requested ticket, and cancellation
# reason. This terminalizes a local row whose owner or reciprocal was lost.
fallout_event_cancel_orphaned_bilateral_row = {
	set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			check_variable = { fallout_event_requested_cancellation_reason > constant:fallout_event_cancellation_reason.none }
			check_variable = { fallout_event_requested_cancellation_reason < constant:fallout_event_cancellation_reason.upper_bound }
			OR = {
				check_variable = { fallout_event_bilateral_status_entries^fallout_event_bilateral_reconcile_cursor = constant:fallout_event_bilateral_status.reserved }
				check_variable = { fallout_event_bilateral_status_entries^fallout_event_bilateral_reconcile_cursor = constant:fallout_event_bilateral_status.response_pending }
			}
		}
		set_temp_variable = { fallout_event_bilateral_orphan_previous_status = fallout_event_bilateral_status_entries^fallout_event_bilateral_reconcile_cursor }
		set_temp_variable = { fallout_event_bilateral_orphan_previous_outcome = fallout_event_bilateral_outcome_entries^fallout_event_bilateral_reconcile_cursor }
		set_temp_variable = { fallout_event_bilateral_orphan_previous_cancellation = fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_reconcile_cursor }
		if = {
			limit = { check_variable = { fallout_event_bilateral_status_entries^fallout_event_bilateral_reconcile_cursor = constant:fallout_event_bilateral_status.reserved } }
			set_variable = { fallout_event_bilateral_outcome_entries^fallout_event_bilateral_reconcile_cursor = constant:fallout_event_outcome_type.cancelled_before_choice }
		}
		else = {
			set_variable = { fallout_event_bilateral_outcome_entries^fallout_event_bilateral_reconcile_cursor = constant:fallout_event_outcome_type.aborted_after_choice }
		}
		set_variable = { fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_reconcile_cursor = fallout_event_requested_cancellation_reason }
		set_variable = { fallout_event_bilateral_status_entries^fallout_event_bilateral_reconcile_cursor = constant:fallout_event_bilateral_status.cancelled }
		set_temp_variable = { fallout_event_bilateral_validation_index = fallout_event_bilateral_reconcile_cursor }
		if = {
			limit = { fallout_event_bilateral_row_at_index_is_structurally_current = yes }
			if = {
				limit = {
					has_country_flag = fallout_event_dispatch_ready
					check_variable = { fallout_event_dispatch_source = constant:fallout_event_dispatch_source.bilateral_response }
					check_variable = { var = fallout_event_dispatch_ticket value = fallout_event_requested_ticket compare = equals }
				}
				fallout_event_clear_dispatch_envelope = yes
			}
			set_temp_variable = { fallout_event_cancelled_ticket = fallout_event_requested_ticket }
			set_temp_variable = { fallout_event_cancellation_source_to_record = constant:fallout_event_transaction_source.bilateral }
			set_temp_variable = { fallout_event_cancellation_reason_to_record = fallout_event_requested_cancellation_reason }
			fallout_event_record_cancellation = yes
			set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one }
		}
		else = {
			set_variable = { fallout_event_bilateral_outcome_entries^fallout_event_bilateral_reconcile_cursor = fallout_event_bilateral_orphan_previous_outcome }
			set_variable = { fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_reconcile_cursor = fallout_event_bilateral_orphan_previous_cancellation }
			set_variable = { fallout_event_bilateral_status_entries^fallout_event_bilateral_reconcile_cursor = fallout_event_bilateral_orphan_previous_status }
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.bilateral_reciprocal_mismatch }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
	}
}

# Supported scope: one country selected from the frozen registry. Repeated
# calls on the same date are idempotent even if a caller bypasses the global
# coordinator guard.
fallout_event_reconcile_country_transactions = {
	if = {
		limit = {
			NOT = { has_global_flag = fallout_event_scheduler_error }
			fallout_event_registry_country_row_is_current = yes
			OR = {
				NOT = { has_variable = fallout_event_last_transaction_reconcile_generation }
				NOT = { has_variable = fallout_event_last_transaction_reconcile_day }
				NOT = { check_variable = { var = fallout_event_last_transaction_reconcile_generation value = fallout_event_runtime_generation compare = equals } }
				NOT = { check_variable = { var = fallout_event_last_transaction_reconcile_day value = global.num_days compare = equals } }
			}
		}
		fallout_event_reconcile_ordinary_receipt = yes
		if = {
			limit = { NOT = { has_global_flag = fallout_event_scheduler_error } }
			fallout_event_reconcile_one_arc_row = yes
		}
		if = {
			limit = { NOT = { has_global_flag = fallout_event_scheduler_error } }
			fallout_event_reconcile_one_delayed_row = yes
		}
		if = {
			limit = { NOT = { has_global_flag = fallout_event_scheduler_error } }
			fallout_event_reconcile_one_bilateral_row = yes
		}
		if = {
			limit = {
				NOT = { has_global_flag = fallout_event_scheduler_error }
				fallout_event_registry_country_row_is_current = yes
			}

# Supported scope: COUNTRY. This coordinator hook performs dormant
# initialization, phase-receipt maintenance, schema promotion, bounded
# transaction reconciliation, and exact one-time dispatch issuance.
fallout_event_scheduler_reconcile = {
	if = {
		limit = {
			has_global_flag = fallout_active
			has_global_flag = fallout_transition_complete
			NOT = { has_global_flag = fallout_event_scheduler_error }
		}
		if = {
			limit = {
				has_global_flag = fallout_event_scheduler_registry_ready
				has_variable = global.fallout_event_schema_version
				check_variable = { global.fallout_event_schema_version = constant:fallout_event_schema.version }
				NOT = { has_global_flag = fallout_event_ordinary_receipt_registry_ready }
				NOT = { has_global_flag = fallout_event_scheduler_error }
			}
			fallout_event_promote_registry_empty_ordinary_receipts = yes
		}
		if = {
			limit = {
				has_global_flag = fallout_event_scheduler_registry_ready
				OR = {
					AND = {
						has_variable = global.fallout_event_schema_version
						check_variable = { global.fallout_event_schema_version = constant:fallout_event_schema.previous_initialization_only_version }
					}
					AND = {
						has_variable = global.fallout_event_registry_schema_version
						check_variable = { global.fallout_event_registry_schema_version = constant:fallout_event_registry_schema.previous_without_survival_binding }
					}
				}
				NOT = { fallout_survival_ledger_is_current = yes }
				NOT = { has_global_flag = fallout_event_scheduler_error }
			}
			set_variable = { global.fallout_event_initialization_status = constant:fallout_event_initialization_status.survival_ledger_pending }
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.survival_ledger_not_current }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
		if = {
			limit = {
				has_global_flag = fallout_event_scheduler_registry_ready
				has_variable = global.fallout_event_schema_version
				check_variable = { global.fallout_event_schema_version = constant:fallout_event_schema.previous_initialization_only_version }
				fallout_survival_ledger_is_current = yes
				NOT = { has_global_flag = fallout_event_scheduler_error }
			}
			fallout_event_migrate_dormant_transaction_schema = yes
		}
		if = {
			limit = {
				has_global_flag = fallout_event_scheduler_registry_ready
				has_variable = global.fallout_event_schema_version
				has_variable = global.fallout_event_registry_schema_version
				check_variable = { global.fallout_event_schema_version = constant:fallout_event_schema.version }
				check_variable = { global.fallout_event_registry_schema_version = constant:fallout_event_registry_schema.previous_without_survival_binding }
				fallout_survival_ledger_is_current = yes
				NOT = { has_global_flag = fallout_event_scheduler_error }
			}
			fallout_event_migrate_registry_survival_binding = yes
		}
		if = {
			limit = {
				has_global_flag = fallout_event_scheduler_registry_ready
				NOT = { fallout_survival_ledger_header_is_current = yes }
				NOT = { has_global_flag = fallout_event_scheduler_error }
			}
			set_variable = { global.fallout_event_initialization_status = constant:fallout_event_initialization_status.survival_ledger_pending }
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.survival_ledger_not_current }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
		if = {
			limit = { fallout_event_map_return_receipts_are_current = yes }
			fallout_event_update_timeline_phase = yes
			if = {
				limit = {
					has_global_flag = fallout_event_scheduler_initialization_pending
					NOT = { fallout_survival_ledger_is_current = yes }
					NOT = { has_global_flag = fallout_event_scheduler_error }
				}
				set_variable = { global.fallout_event_initialization_status = constant:fallout_event_initialization_status.survival_ledger_pending }
				set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.survival_ledger_not_current }
				fallout_event_register_scheduler_error_if_unowned = yes
			}
			if = {
				limit = {
					has_global_flag = fallout_event_scheduler_initialization_pending
					NOT = { has_global_flag = fallout_event_scheduler_registry_ready }
					NOT = { has_global_flag = fallout_event_scheduler_error }
					fallout_successor_allocation_is_current = yes
					fallout_survival_ledger_is_current = yes
				}
				fallout_event_build_initial_registry = yes
			}
			else_if = {
				limit = {
					has_global_flag = fallout_event_scheduler_initialization_pending
					has_global_flag = fallout_event_scheduler_registry_ready
					NOT = { fallout_event_registry_is_current = yes }
				}
				set_variable = { global.fallout_event_initialization_status = constant:fallout_event_initialization_status.blocked }
				set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.registry_source_mismatch }
				fallout_event_register_scheduler_error_if_unowned = yes
			}
			else_if = {
				limit = {
					has_global_flag = fallout_event_scheduler_initialization_pending
					NOT = { fallout_successor_allocation_is_current = yes }
					NOT = { has_global_flag = fallout_event_scheduler_registry_ready }
				}
				set_variable = { global.fallout_event_initialization_status = constant:fallout_event_initialization_status.blocked }
				set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.allocation_not_current }
				fallout_event_register_scheduler_error_if_unowned = yes
			}
		}
		else_if = {
			limit = { has_global_flag = fallout_event_scheduler_initialization_pending }
			set_variable = { global.fallout_event_initialization_status = constant:fallout_event_initialization_status.blocked }
			set_temp_variable = { fallout_event_scheduler_error_to_register = constant:fallout_event_scheduler_error.legacy_timeline_unresolved }
			fallout_event_register_scheduler_error_if_unowned = yes
		}
		if = {
			limit = {
				NOT = { has_global_flag = fallout_event_scheduler_error }
				has_global_flag = fallout_event_scheduler_registry_ready
			}
			fallout_event_reconcile_next_registry_country = yes
		}
	}
}

# Supported scope: COUNTRY. Inputs: requested ticket and cancellation reason.
# This never clears an issued engine event. Exact retries are source-bound to
# the generation-bound cancellation receipt.
fallout_event_cancel_ordinary_receipt = {
	set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			fallout_event_registry_country_row_is_current = yes
			fallout_event_ordinary_receipt_is_consistent = yes
			has_country_flag = fallout_event_ordinary_receipt_pending
			check_variable = { var = fallout_event_ordinary_ticket value = fallout_event_requested_ticket compare = equals }
			check_variable = { fallout_event_requested_cancellation_reason > constant:fallout_event_cancellation_reason.none }
			check_variable = { fallout_event_requested_cancellation_reason < constant:fallout_event_cancellation_reason.upper_bound }
			NOT = { fallout_event_ordinary_receipt_has_issued_dispatch = yes }
		}
		if = {
			limit = {
				has_country_flag = fallout_event_dispatch_ready
				check_variable = { fallout_event_dispatch_source = constant:fallout_event_dispatch_source.ordinary_event }
				check_variable = { var = fallout_event_dispatch_ticket value = fallout_event_requested_ticket compare = equals }
			}
			fallout_event_clear_dispatch_envelope = yes
		}
		set_temp_variable = { fallout_event_cancelled_ticket = fallout_event_requested_ticket }
		set_temp_variable = { fallout_event_cancellation_source_to_record = constant:fallout_event_transaction_source.ordinary_event }
		set_temp_variable = { fallout_event_cancellation_reason_to_record = fallout_event_requested_cancellation_reason }
		fallout_event_clear_ordinary_receipt = yes
		fallout_event_record_cancellation = yes
		if = {
			limit = {
				fallout_event_ordinary_receipt_is_consistent = yes
				fallout_event_transaction_history_receipts_are_consistent = yes
			}
			set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one }
		}
	}
	else_if = {
		limit = {
			fallout_event_registry_country_row_is_current = yes
			NOT = { has_country_flag = fallout_event_ordinary_receipt_pending }
			has_variable = fallout_event_last_cancelled_ticket
			has_variable = fallout_event_last_cancellation_source
			has_variable = fallout_event_last_cancellation_reason
			has_variable = fallout_event_last_cancellation_generation
			check_variable = { var = fallout_event_last_cancelled_ticket value = fallout_event_requested_ticket compare = equals }
			check_variable = { fallout_event_last_cancellation_source = constant:fallout_event_transaction_source.ordinary_event }
			check_variable = { var = fallout_event_last_cancellation_reason value = fallout_event_requested_cancellation_reason compare = equals }
			check_variable = { var = fallout_event_last_cancellation_generation value = fallout_event_runtime_generation compare = equals }
			fallout_event_ordinary_receipt_is_consistent = yes
		}
		set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one }
	}
}

# Inputs: requested ticket, bilateral status, outcome, and cancellation reason.
# Both rows are updated inside one effect chain. Status is written after payload.
fallout_event_update_bilateral_pair = {
	set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_bilateral_pair_is_current = constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_bilateral_origin_row_index = constant:fallout_event_runtime.first_array_index }

	set_temp_variable = { fallout_event_transaction_row_found = constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_transaction_row_index = constant:fallout_event_runtime.first_array_index }
	fallout_event_load_current_bilateral_pair = yes
	if = {
		limit = { check_variable = { fallout_event_bilateral_pair_is_current = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_bilateral_current_status = fallout_event_bilateral_status_entries^fallout_event_bilateral_origin_row_index }
		set_temp_variable = { fallout_event_bilateral_origin_previous_outcome = fallout_event_bilateral_outcome_entries^fallout_event_bilateral_origin_row_index }
		set_temp_variable = { fallout_event_bilateral_origin_previous_cancellation = fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_origin_row_index }
		event_target:fallout_event_bilateral_loaded_partner = {
			set_temp_variable = { fallout_event_bilateral_partner_previous_status = fallout_event_bilateral_status_entries^fallout_event_bilateral_partner_row_index }
			set_temp_variable = { fallout_event_bilateral_partner_previous_outcome = fallout_event_bilateral_outcome_entries^fallout_event_bilateral_partner_row_index }
			set_temp_variable = { fallout_event_bilateral_partner_previous_cancellation = fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_partner_row_index }
		}
		set_temp_variable = { fallout_event_bilateral_transition_is_valid = constant:fallout_event_runtime.zero }
		if = {
			limit = {
				check_variable = { var = fallout_event_bilateral_current_status value = fallout_event_requested_bilateral_status compare = equals }
				check_variable = { var = fallout_event_bilateral_outcome_entries^fallout_event_bilateral_origin_row_index value = fallout_event_requested_outcome compare = equals }
				check_variable = { var = fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_origin_row_index value = fallout_event_requested_cancellation_reason compare = equals }
			}
			set_temp_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { fallout_event_bilateral_current_status = constant:fallout_event_bilateral_status.reserved }
						check_variable = { fallout_event_requested_bilateral_status = constant:fallout_event_bilateral_status.response_pending }
						check_variable = { fallout_event_requested_outcome = constant:fallout_event_outcome_type.none }
						check_variable = { fallout_event_requested_cancellation_reason = constant:fallout_event_cancellation_reason.none }
					}
					AND = {
						check_variable = { fallout_event_bilateral_current_status = constant:fallout_event_bilateral_status.response_pending }
						check_variable = { fallout_event_requested_bilateral_status = constant:fallout_event_bilateral_status.resolved }
						check_variable = { fallout_event_requested_outcome > constant:fallout_event_outcome_type.none }
						check_variable = { fallout_event_requested_outcome < constant:fallout_event_outcome_type.aborted_after_choice }
						check_variable = { fallout_event_requested_cancellation_reason = constant:fallout_event_cancellation_reason.none }
					}
					AND = {
						OR = {
							check_variable = { fallout_event_bilateral_current_status = constant:fallout_event_bilateral_status.reserved }
							check_variable = { fallout_event_bilateral_current_status = constant:fallout_event_bilateral_status.response_pending }
						}
						check_variable = { fallout_event_requested_bilateral_status = constant:fallout_event_bilateral_status.cancelled }
						OR = {
							check_variable = { fallout_event_requested_outcome = constant:fallout_event_outcome_type.aborted_after_choice }
							check_variable = { fallout_event_requested_outcome = constant:fallout_event_outcome_type.cancelled_before_choice }
						}
						check_variable = { fallout_event_requested_cancellation_reason > constant:fallout_event_cancellation_reason.none }
						check_variable = { fallout_event_requested_cancellation_reason < constant:fallout_event_cancellation_reason.upper_bound }
					}
					AND = {
						OR = {
							check_variable = { fallout_event_bilateral_current_status = constant:fallout_event_bilateral_status.resolved }
							check_variable = { fallout_event_bilateral_current_status = constant:fallout_event_bilateral_status.cancelled }
						}
						check_variable = { fallout_event_requested_bilateral_status = constant:fallout_event_bilateral_status.cleanup_pending }
						check_variable = { var = fallout_event_bilateral_outcome_entries^fallout_event_bilateral_origin_row_index value = fallout_event_requested_outcome compare = equals }
						check_variable = { var = fallout_event_bilateral_cancellation_reason_entries^fallout_event_bilateral_origin_row_index value = fallout_event_requested_cancellation_reason compare = equals }
					}
				}
			}

			limit = { check_variable = { fallout_event_bilateral_transition_is_valid = constant:fallout_event_runtime.one } }

					}
				}
			}
			if = {

				if = {
					limit = {
						OR = {
							check_variable = { fallout_event_requested_bilateral_status = constant:fallout_event_bilateral_status.resolved }
							check_variable = { fallout_event_requested_bilateral_status = constant:fallout_event_bilateral_status.cancelled }
						}
					}
					if = {
						limit = {
							has_country_flag = fallout_event_dispatch_ready

								check_variable = { fallout_event_dispatch_source = constant:fallout_event_dispatch_source.bilateral_response }
