# ============================================================================
# CHAOS REDUX - FALLOUT SUCCESSOR B7 VERTICAL SLICE EFFECTS
# ============================================================================
# B7 is a dormant, idempotent allocator pilot. It preserves one USA human
# continuation in place, applies a country-specific package receipt, records
# one exact capital, names one deterministic fragmentation candidate, and can
# run the reviewed five-state NZL transfer or one conflict-ledger-gated dynamic
# materialization. It never switches the player or marks the global successor
# matrix complete.
# ============================================================================

# Supported scope: COUNTRY. Clears only the B7 fragmentation probe on this
# source. Durable package and assignment receipts are never erased here.
fallout_successor_b7_clear_source_probe = {
	clr_country_flag = fallout_b7_fragmentation_candidate_named
	clr_country_flag = fallout_b7_fragmentation_candidate_blocked
	clear_variable = fallout_b7_fragmentation_candidate_country
	clear_variable = fallout_b7_fragmentation_candidate_state
	clear_variable = fallout_b7_fragmentation_candidate_generation
	clear_variable = fallout_b7_fragmentation_status
}

# Supported scope: USA. This is the only country package written by B7. It is
# intentionally country-specific and loads an additive Fallout focus layer.

		set_variable = { fallout_regional_package_generation = global.fallout_transition_generation }

		set_variable = { fallout_government_archetype = constant:fallout_government_archetype.continuity_government }
		set_variable = { fallout_government_archetype_generation = global.fallout_transition_generation }

# Supported scope: USA. Records the complete row for the in-place human
# continuation. The capital and source are both the frozen player reservation.
fallout_successor_b7_assign_usa_continuity_row = {
	if = {
		limit = {
			fallout_successor_b7_transaction_is_current = yes
			fallout_successor_b7_usa_continuity_source_is_ready = yes
			fallout_successor_b7_usa_package_receipts_are_current = yes
			NOT = { has_country_flag = fallout_successor_assignment_committed }
		}
		var:fallout_player_primary_capital_state = {
			save_event_target_as = fallout_b7_usa_capital
		}
		set_country_flag = fallout_live_tag_conflict_resolved
		clr_country_flag = fallout_live_tag_conflict_allocation_pending
		set_variable = { fallout_live_tag_conflict_resolution = constant:fallout_tag_conflict_resolution.player_reserved }
		set_variable = { fallout_live_tag_conflict_resolution_generation = global.fallout_transition_generation }
		set_variable = { fallout_live_tag_conflict_cleanup_owner = THIS }
		set_variable = { fallout_live_tag_conflict_cleanup_generation = global.fallout_transition_generation }
		set_variable = { fallout_live_tag_conflict_output_country = THIS }
		set_variable = { fallout_successor_conflict_source_country = THIS }
		set_variable = { fallout_successor_conflict_result = constant:fallout_tag_conflict_resolution.player_reserved }
		set_variable = { fallout_successor_conflict_resolution_generation = global.fallout_transition_generation }
		set_variable = { fallout_successor_cleanup_owner = THIS }
		set_variable = { fallout_successor_cleanup_generation = global.fallout_transition_generation }
		set_variable = { fallout_successor_assignment_generation = global.fallout_transition_generation }
		set_variable = { fallout_successor_assignment_count = constant:fallout_runtime.one }
		set_variable = { fallout_successor_assignment_capital_state = fallout_player_primary_capital_state }
		set_variable = { fallout_successor_assignment_origin_state = fallout_player_primary_capital_state }
		set_country_flag = fallout_successor_assignment_committed
		if = {
			limit = {
				NOT = { has_variable = global.fallout_successor_assignment_count }
			}
			set_variable = { global.fallout_successor_assignment_count = constant:fallout_runtime.zero }
		}
		if = {
			limit = {
				NOT = { is_in_array = { array = global.fallout_successor_assigned_countries value = THIS } }
			}
			add_to_array = { array = global.fallout_successor_assigned_countries value = THIS }
			add_to_variable = { global.fallout_successor_assignment_count = constant:fallout_runtime.one }
		}
		if = {
			limit = {
				NOT = { is_in_array = { array = global.fallout_successor_assigned_capital_states value = event_target:fallout_b7_usa_capital } }
			}
			add_to_array = { array = global.fallout_successor_assigned_capital_states value = event_target:fallout_b7_usa_capital }
		}
		event_target:fallout_b7_usa_capital = {
			set_state_flag = fallout_successor_assignment_capital
			set_variable = { fallout_successor_assignment_country = ROOT }
			set_variable = { fallout_successor_assignment_generation = global.fallout_transition_generation }
			set_variable = { fallout_successor_assignment_count = constant:fallout_runtime.one }
		}
		if = {
			limit = { fallout_successor_b7_usa_assignment_is_current = yes }
			set_country_flag = fallout_b7_usa_continuity_assigned
			set_country_flag = fallout_b7_usa_human_control_receipt
			set_variable = { fallout_b7_human_control_generation = global.fallout_transition_generation }
			set_variable = { fallout_b7_continuity_status = constant:fallout_successor_b7_status.continuity_assigned }
			set_variable = { fallout_b7_continuity_generation = global.fallout_transition_generation }
		}
		else = {
			set_country_flag = fallout_b7_usa_continuity_error
			set_variable = { fallout_b7_continuity_status = constant:fallout_successor_b7_status.error }
			set_variable = { fallout_b7_continuity_generation = global.fallout_transition_generation }
		}
	}
}

# Supported scope: COUNTRY. Names the lowest-id conflict-free country and the
# lowest-id candidate state. No ownership, release, or focus package is
# attempted when the candidate is missing or unproven.
fallout_successor_b7_probe_fragmentation_candidate = {
	if = {
		limit = {
			fallout_successor_b7_transaction_is_current = yes
			fallout_successor_b7_fragmentation_source_is_ready = yes
		}
		fallout_successor_b7_clear_source_probe = yes
		set_temp_variable = { fallout_b7_candidate_found = constant:fallout_runtime.zero }
		set_temp_variable = { fallout_b7_best_candidate_id = constant:fallout_snapshot_limit.maximum_database_id }
		set_temp_variable = { fallout_b7_best_candidate_country = THIS }
		set_temp_variable = { fallout_b7_state_found = constant:fallout_runtime.zero }
		set_temp_variable = { fallout_b7_best_state_id = constant:fallout_snapshot_limit.maximum_database_id }
		set_temp_variable = { fallout_b7_best_state = THIS }
		for_each_scope_loop = {
			array = global.fallout_all_possible_country_scopes
			limit = { fallout_successor_b7_fragmentation_candidate_is_available = yes }
			set_temp_variable = { fallout_b7_candidate_id = THIS.id }
			if = {
				limit = {
					OR = {
						check_variable = { fallout_b7_candidate_found = constant:fallout_runtime.zero }
						check_variable = {
							var = fallout_b7_candidate_id
							value = fallout_b7_best_candidate_id
							compare = less_than
						}
					}
				}
				set_temp_variable = { fallout_b7_candidate_found = constant:fallout_runtime.one }
				set_temp_variable = { fallout_b7_best_candidate_id = fallout_b7_candidate_id }
				set_temp_variable = { fallout_b7_best_candidate_country = THIS }
			}
		}
		for_each_scope_loop = {
			array = global.fallout_successor_candidate_states
			limit = { fallout_successor_state_is_candidate = yes }
			set_temp_variable = { fallout_b7_state_id = THIS.id }
			if = {
				limit = {
					OR = {
						check_variable = { fallout_b7_state_found = constant:fallout_runtime.zero }
						check_variable = {
							var = fallout_b7_state_id
							value = fallout_b7_best_state_id
							compare = less_than
						}
					}
				}
				set_temp_variable = { fallout_b7_state_found = constant:fallout_runtime.one }
				set_temp_variable = { fallout_b7_best_state_id = fallout_b7_state_id }
				set_temp_variable = { fallout_b7_best_state = THIS }
			}
		}
		if = {
			limit = {
				check_variable = { fallout_b7_candidate_found = constant:fallout_runtime.one }
				check_variable = { fallout_b7_state_found = constant:fallout_runtime.one }
			}
			set_country_flag = fallout_b7_fragmentation_candidate_named
			set_variable = { fallout_b7_fragmentation_candidate_country = fallout_b7_best_candidate_country }
			set_variable = { fallout_b7_fragmentation_candidate_state = fallout_b7_best_state }
			set_variable = { fallout_b7_fragmentation_candidate_generation = global.fallout_transition_generation }
			set_variable = { fallout_b7_fragmentation_status = constant:fallout_successor_b7_status.fragmentation_candidate_named }
		}
		else = {
			set_country_flag = fallout_b7_fragmentation_candidate_blocked
			set_variable = { fallout_b7_fragmentation_candidate_generation = global.fallout_transition_generation }
			set_variable = { fallout_b7_fragmentation_status = constant:fallout_successor_b7_status.fragmentation_blocked }
		}
		clear_temp_variable = fallout_b7_candidate_found
		clear_temp_variable = fallout_b7_best_candidate_id
		clear_temp_variable = fallout_b7_best_candidate_country
		clear_temp_variable = fallout_b7_state_found
		clear_temp_variable = fallout_b7_best_state_id
		clear_temp_variable = fallout_b7_best_state
	}
}
