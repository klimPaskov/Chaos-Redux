# ============================================================================
# CHAOS REDUX - FALLOUT SHELTER MARRIAGE LAW EFFECTS
# ============================================================================
# Dormant country-level family-law transaction. The chain freezes the current
# survival ledgers and durable family institutions, then resolves one authored
# policy through a 42-day result, a 300-day callback, Deaths-backed failure,
# Event Log memory, dynamic modifiers, and authenticated cleanup.
# ============================================================================

fallout_event_303_apply_failure_deaths = {
	every_owned_state = {
		if = {
			limit = { check_variable = { state_population_k value = constant:fallout_event_runtime.zero compare = greater_than } }
			set_temp_variable = { fallout_event_303_state_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_303_state_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_303_state_death_request = constant:fallout_event_303_threshold.result_failure_death_percent }
			round_temp_variable = fallout_event_303_state_death_request
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_requested = fallout_event_303_state_death_request }
			set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:fallout_event_303_threshold.minimum_remaining_people }
			set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.fallout_aftermath }
			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }
			apply_exact_state_civilian_population_loss = yes
			clear_temp_variable = fallout_event_303_state_death_request

	}
}
