# ============================================================================
# CHAOS REDUX - EVENT 021 RANDOM CIVIL WAR DYNAMIC TAG CAPACITY
# ============================================================================
# Every ordinary claimant comes from vanilla start_civil_war, which takes a tag
# from the shared dynamic pool declared in
# common/country_tags/zz_dynamic_countries.txt (D01-D75). Other events use the
# same pool, so Event 021 counts the declared tags that currently exist before
# it plans a new ordinary claimant and keeps a centralized reserve free.
# A plan that would need more tags than remain is trimmed, deferred, or
# rejected with the existing capacity skip reason; it is never turned into a
# same-tag contest. Same-tag contests and Event 006 carriers use no dynamic tag.
# ============================================================================

# COUNTRY scope (meta_trigger needs a country or state scope). Counts the
# declared dynamic tags that currently exist. Output:
# `event021_dynamic_tag_live_count`. The probe variable on the calling country
# is cleared again before return.
event021_count_live_dynamic_tags = {
	set_temp_variable = { event021_dynamic_tag_live_count = constant:random_civil_war_value.zero }
	for_loop_effect = {
		start = constant:event021_dynamic_tags.first_index
		end = constant:event021_dynamic_tags.two_digit_first_index
		value = event021_dynamic_tag_index
		set_variable = { event021_dynamic_tag_probe = event021_dynamic_tag_index }
		if = {
			limit = {
				meta_trigger = {
					text = { country_exists = D0[EVENT021_DYNAMIC_TAG] }
					EVENT021_DYNAMIC_TAG = "[?event021_dynamic_tag_probe|.0]"
				}
			}
			add_to_temp_variable = { event021_dynamic_tag_live_count = constant:random_civil_war_value.one }
		}
	}
	for_loop_effect = {
		start = constant:event021_dynamic_tags.two_digit_first_index
		end = constant:event021_dynamic_tags.pool_end_index
		value = event021_dynamic_tag_index
		set_variable = { event021_dynamic_tag_probe = event021_dynamic_tag_index }
		if = {
			limit = {
				meta_trigger = {
					text = { country_exists = D[EVENT021_DYNAMIC_TAG] }
					EVENT021_DYNAMIC_TAG = "[?event021_dynamic_tag_probe|.0]"
				}
			}
			add_to_temp_variable = { event021_dynamic_tag_live_count = constant:random_civil_war_value.one }
		}
	}
	clear_variable = event021_dynamic_tag_probe
}

# COUNTRY scope. Dynamic tags this country may still claim for new ordinary
# claimants: the declared pool, less the tags in use, the centralized reserve,
# and tags held for other countries' frozen Fracture Cascade plans. A country
# with its own frozen plan counts its own reservation as available. Output:
# `event021_dynamic_tag_free`, never below zero.
event021_count_free_dynamic_tags = {
	set_temp_variable = { event021_dynamic_tag_live_count = constant:random_civil_war_value.zero }
	event021_count_live_dynamic_tags = yes
	set_temp_variable = { event021_dynamic_tag_free = constant:event021_dynamic_tags.pool_size }
	subtract_from_temp_variable = { event021_dynamic_tag_free = event021_dynamic_tag_live_count }
	subtract_from_temp_variable = { event021_dynamic_tag_free = constant:event021_dynamic_tags.reserve }
	if = {
		limit = { has_variable = global.random_civil_war_scenario_reserved_dynamic_tags }
		subtract_from_temp_variable = { event021_dynamic_tag_free = global.random_civil_war_scenario_reserved_dynamic_tags }
	}
	if = {
		limit = { has_variable = random_civil_war_scenario_frozen_dynamic_tags }
		add_to_temp_variable = { event021_dynamic_tag_free = random_civil_war_scenario_frozen_dynamic_tags }
	}
	if = {
		limit = { check_variable = { var = event021_dynamic_tag_free value = constant:random_civil_war_value.zero compare = less_than } }
		set_temp_variable = { event021_dynamic_tag_free = constant:random_civil_war_value.zero }
	}
}

# COUNTRY scope on a country whose Fracture Cascade plan just froze. Holds one
# dynamic tag for an ordinary primary claimant plus one for every frozen
# ordinary optional row. Output: `event021_scenario_dynamic_tags_ready`, one
# when the tags were held and zero when the free pool cannot cover the plan.
event021_reserve_scenario_dynamic_tags = {
	set_temp_variable = { event021_scenario_dynamic_tags_ready = constant:random_civil_war_value.one }
	set_temp_variable = { event021_scenario_dynamic_tag_need = constant:random_civil_war_value.zero }
	if = {
		limit = {
			NOT = { check_variable = { var = random_civil_war_archetype value = constant:random_civil_war_archetype.same_tag compare = equals } }
			NOT = { check_variable = { var = random_civil_war_archetype value = constant:random_civil_war_archetype.event006 compare = equals } }
		}
		set_temp_variable = { event021_scenario_dynamic_tag_need = constant:random_civil_war_value.one }
	}
	for_each_loop = {
		array = random_civil_war_scenario_frozen_optional_front_actor_types
		value = event021_scenario_dynamic_tag_actor_type
		if = {
			limit = { check_variable = { var = event021_scenario_dynamic_tag_actor_type value = constant:random_civil_war_planned_actor_type.ordinary_claimant compare = equals } }
			add_to_temp_variable = { event021_scenario_dynamic_tag_need = constant:random_civil_war_value.one }
		}
	}
	event021_release_scenario_dynamic_tags = yes
	set_temp_variable = { event021_dynamic_tag_free = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_dynamic_tag_live_count = constant:random_civil_war_value.zero }
	if = {
		limit = { check_variable = { var = event021_scenario_dynamic_tag_need value = constant:random_civil_war_value.zero compare = greater_than } }
		event021_count_free_dynamic_tags = yes
		if = {
			limit = { check_variable = { var = event021_scenario_dynamic_tag_need value = event021_dynamic_tag_free compare = greater_than } }
			set_temp_variable = { event021_scenario_dynamic_tags_ready = constant:random_civil_war_value.zero }
		}
		else = {
			set_variable = { random_civil_war_scenario_frozen_dynamic_tags = event021_scenario_dynamic_tag_need }
			if = {
				limit = { NOT = { has_variable = global.random_civil_war_scenario_reserved_dynamic_tags } }
				set_variable = { global.random_civil_war_scenario_reserved_dynamic_tags = constant:random_civil_war_value.zero }
			}
			add_to_variable = { global.random_civil_war_scenario_reserved_dynamic_tags = event021_scenario_dynamic_tag_need }
		}
	}
}

# COUNTRY scope. Returns this country's held Fracture Cascade tags to the pool.
event021_release_scenario_dynamic_tags = {
	if = {
		limit = {
			has_variable = random_civil_war_scenario_frozen_dynamic_tags
			has_variable = global.random_civil_war_scenario_reserved_dynamic_tags
		}
		subtract_from_variable = { global.random_civil_war_scenario_reserved_dynamic_tags = random_civil_war_scenario_frozen_dynamic_tags }
		if = {
			limit = { check_variable = { var = global.random_civil_war_scenario_reserved_dynamic_tags value = constant:random_civil_war_value.zero compare = less_than } }
			set_variable = { global.random_civil_war_scenario_reserved_dynamic_tags = constant:random_civil_war_value.zero }
		}
	}
	clear_variable = random_civil_war_scenario_frozen_dynamic_tags
}
