# IW-180 Quebec (QUE)
# ---------------------------------------------------------------------------

independence_wave_que_apply_package_layer = {
	add_ideas = independence_wave_que_power_trusts
	set_country_flag = independence_wave_framework_package_layer_applied
}

	independence_wave_regional_framework_reset_deltas = yes
	add_ideas = independence_wave_que_legislative_assembly
	set_temp_variable = { independence_wave_decision_legitimacy_delta = constant:independence_wave_decision_effect.value_standard }
	set_temp_variable = { independence_wave_decision_instability_delta = constant:independence_wave_decision_effect.value_minor_loss }
	independence_wave_decision_apply_country_deltas = yes
	set_country_flag = independence_wave_que_assembly_seated
}

	add_ideas = independence_wave_que_hydro_commission

independence_wave_que_cleanup_package_layer = {
	if = {
		limit = { original_tag = QUE check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_180 } }
		remove_decision = independence_wave_que_seat_legislative_assembly
		remove_decision = independence_wave_que_nationalise_power_trusts
		remove_decision = independence_wave_que_charter_caisses_populaires
		remove_ideas = { independence_wave_que_power_trusts independence_wave_que_legislative_assembly independence_wave_que_hydro_commission independence_wave_que_caisses_populaires }
		clr_country_flag = independence_wave_que_assembly_seated
		clr_country_flag = independence_wave_que_power_trusts_nationalised
		clr_country_flag = independence_wave_que_caisses_chartered
	}
}
