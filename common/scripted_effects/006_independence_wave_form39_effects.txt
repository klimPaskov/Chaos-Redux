independence_wave_form39_install_autonomous_member = {
	save_event_target_as = independence_wave_form39_autonomous_member_target
	set_country_flag = independence_wave_form39_autonomous_member
	set_variable = { independence_wave_form39_autonomous_family = ROOT.independence_wave_formable_family }
	set_variable = { independence_wave_form39_autonomous_carrier = ROOT }
	set_variable = { independence_wave_form39_autonomous_carrier_generation = ROOT.independence_wave_generation_id }
	set_variable = { independence_wave_form39_autonomy = 55 }
	add_ideas = independence_wave_form39_autonomous_compact
	if = {
		limit = { NOT = { gives_military_access_to = ROOT } }
		give_military_access = ROOT
		set_country_flag = independence_wave_form39_member_granted_access_to_carrier
	}
	if = {
		limit = { NOT = { has_guaranteed = ROOT } }
		give_guarantee = ROOT
		set_country_flag = independence_wave_form39_member_guaranteed_carrier
	}
	ROOT = {
		if = {
			limit = { NOT = { gives_military_access_to = event_target:independence_wave_form39_autonomous_member_target } }
			give_military_access = event_target:independence_wave_form39_autonomous_member_target
			event_target:independence_wave_form39_autonomous_member_target = { set_country_flag = independence_wave_form39_carrier_granted_access_to_member }
		}
		if = {
			limit = { NOT = { has_guaranteed = event_target:independence_wave_form39_autonomous_member_target } }
			give_guarantee = event_target:independence_wave_form39_autonomous_member_target
			event_target:independence_wave_form39_autonomous_member_target = { set_country_flag = independence_wave_form39_carrier_guaranteed_member }
		}
		add_to_variable = { independence_wave_form39_member_consent = constant:independence_wave_form39.member_consent_gain }
	}
}

independence_wave_formable_integration_adapter_39 = {
	if = {
		limit = {
			has_country_flag = independence_wave_formable_mutation_prevalidated
			has_country_flag = independence_wave_formable_identity_committed
			has_cosmetic_tag = MFX
			independence_wave_formable_member_arrays_are_aligned = yes
			has_independence_wave_form39_exact_founding_ledger = yes
		}
		set_temp_variable = { independence_wave_form39_member_index = constant:independence_wave_formable_tuning.first_array_index }
		set_temp_variable = { independence_wave_form39_additional_members_processed = constant:independence_wave_formable_tuning.no_entries }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_form39_member_index value = independence_wave_formable_member_country_entries^num compare = less_than } }
			set_temp_variable = { independence_wave_form39_member_consent = independence_wave_formable_member_consent_entries^independence_wave_form39_member_index }
			var:independence_wave_formable_member_country_entries^independence_wave_form39_member_index = {
				if = {
					limit = {
						NOT = { tag = ROOT }
						is_independence_wave_form39_eligible_member = yes
						check_variable = { independence_wave_form39_member_consent = constant:independence_wave_formable_consent.consenting_member }
					}
					if = {
						limit = { has_country_flag = independence_wave_formable_full_integration_authorized }
						independence_wave_formable_end_consenting_member_origin = yes
						if = { limit = { original_tag = PNG owns_state = @FORM39_PNG_ANCHOR controls_state = @FORM39_PNG_ANCHOR } ROOT = { transfer_state = @FORM39_PNG_ANCHOR add_state_core = @FORM39_PNG_ANCHOR } }
						if = { limit = { original_tag = WPG owns_state = @FORM39_WPG_ANCHOR controls_state = @FORM39_WPG_ANCHOR } ROOT = { transfer_state = @FORM39_WPG_ANCHOR add_state_core = @FORM39_WPG_ANCHOR } }
					}
					else = { independence_wave_form39_install_autonomous_member = yes }
					add_to_temp_variable = { independence_wave_form39_additional_members_processed = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
			add_to_temp_variable = { independence_wave_form39_member_index = constant:independence_wave_formable_tuning.loop_increment }
		}
		if = {
			limit = { check_variable = { var = independence_wave_form39_additional_members_processed value = constant:independence_wave_form39.minimum_non_carrier_members compare = greater_than_or_equals } }
			set_variable = { independence_wave_form39_bound_member_count = independence_wave_form39_additional_members_processed }
			set_country_flag = independence_wave_form39_integration_committed
			set_country_flag = independence_wave_formable_integration_committed
			independence_wave_form39_initialize_ledgers = yes
		}
	}
}

		var:independence_wave_form39_autonomous_carrier = { save_event_target_as = independence_wave_form39_cleanup_carrier }
