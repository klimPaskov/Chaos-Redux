# COUNTRY scope on the proposing actor. FROM is the nominated partner.
africa_world_union_protocol_open = {
	if = {
		limit = {
			africa_world_union_protocol_is_openable = yes
			FROM = { africa_world_union_protocol_target_is_eligible = yes }
		}
		clear_global_event_target = africa_world_union_protocol_host
		clear_global_event_target = africa_world_union_protocol_partner
		save_global_event_target_as = africa_world_union_protocol_host
		FROM = { save_global_event_target_as = africa_world_union_protocol_partner }
		set_global_flag = africa_world_union_protocol_open
		set_country_flag = africa_world_union_protocol_host
		FROM = {
			set_country_flag = africa_world_union_protocol_partner
			set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.proposed }
			set_variable = { africa_world_package_phase = constant:africa_world_package_phase.union_convention }
			set_variable = { africa_world_union_partner_approval = constant:africa_world_union_protocol.approval.pending }
		}
		set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.proposed }
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.union_convention }
		set_variable = { africa_world_union_host_approval = constant:africa_world_union_protocol.approval.pending }
		set_variable = { africa_world_union_host_yes_count = constant:africa_value.zero }
		set_variable = { africa_world_union_host_no_count = constant:africa_value.zero }
		set_variable = { africa_world_union_partner_yes_count = constant:africa_value.zero }
		set_variable = { africa_world_union_partner_no_count = constant:africa_value.zero }
		set_variable = { africa_world_union_host_ballot_total = constant:africa_value.zero }
		set_variable = { africa_world_union_partner_ballot_total = constant:africa_value.zero }
		set_variable = { africa_world_union_strain = constant:africa_value.zero }
		set_country_flag = africa_world_union_protocol_proposal_open
		FROM = { country_event = { id = africa_world_package.700 } }
	}
}

# Dispatches separate voluntary constituent ballots for each actor. Coerced,
# withdrawn, pending, and subject members are excluded from both ballots.
africa_world_union_protocol_dispatch_constituent_ballots = {
	if = {
		limit = {
			has_global_flag = africa_world_union_protocol_open
			has_country_flag = africa_world_union_protocol_host_approved
		}
		set_country_flag = africa_world_union_protocol_ballot_dispatch_pending
		set_variable = { africa_world_union_host_yes_count = constant:africa_value.zero }
		set_variable = { africa_world_union_host_no_count = constant:africa_value.zero }
		set_variable = { africa_world_union_partner_yes_count = constant:africa_value.zero }
		set_variable = { africa_world_union_partner_no_count = constant:africa_value.zero }
		set_variable = { africa_world_union_host_ballot_total = constant:africa_value.zero }
		set_variable = { africa_world_union_partner_ballot_total = constant:africa_value.zero }
		for_each_scope_loop = {
			array = africa_world_constituent_countries
			if = {
				limit = {
					has_country_flag = africa_world_constituent_consented
					NOT = { has_country_flag = africa_world_constituent_coerced }
					NOT = { has_country_flag = africa_world_constituent_withdrawn }
					NOT = { is_subject = yes }
				}
				set_country_flag = africa_world_union_ballot_pending_host
				set_country_flag = africa_world_union_ballot_host_side
				event_target:africa_world_union_protocol_host = { add_to_variable = { africa_world_union_host_ballot_total = constant:africa_value.one } }
				country_event = { id = africa_world_package.702 }
			}
		}
		event_target:africa_world_union_protocol_partner = {
			for_each_scope_loop = {
				array = africa_world_constituent_countries
				if = {
					limit = {
						has_country_flag = africa_world_constituent_consented
						NOT = { has_country_flag = africa_world_constituent_coerced }
						NOT = { has_country_flag = africa_world_constituent_withdrawn }
						NOT = { is_subject = yes }
					}
					set_country_flag = africa_world_union_ballot_pending_partner
					set_country_flag = africa_world_union_ballot_partner_side
					event_target:africa_world_union_protocol_partner = { add_to_variable = { africa_world_union_partner_ballot_total = constant:africa_value.one } }
					country_event = { id = africa_world_package.703 }
				}
			}
		}
		clr_country_flag = africa_world_union_protocol_ballot_dispatch_pending
	}
}

africa_world_union_protocol_try_finalize_ballots = {
	if = {
		limit = {
			has_global_flag = africa_world_union_protocol_open
			event_target:africa_world_union_protocol_host = { africa_world_union_protocol_ballots_are_complete = yes }
			event_target:africa_world_union_protocol_partner = { africa_world_union_protocol_ballots_are_complete = yes }
		}
		event_target:africa_world_union_protocol_host = {
			if = {
				limit = {
					check_variable = {
						var = africa_world_union_host_yes_count
						value = constant:africa_world_union_protocol.tuning.constituent_yes_required
						compare = greater_than_or_equals
					}
				}
				set_country_flag = africa_world_union_protocol_host_ballot_proven
			}
		}
		event_target:africa_world_union_protocol_partner = {
			if = {
				limit = {
					check_variable = {
						var = africa_world_union_partner_yes_count
						value = constant:africa_world_union_protocol.tuning.constituent_yes_required
						compare = greater_than_or_equals
					}
				}
				set_country_flag = africa_world_union_protocol_partner_ballot_proven
			}
		}
		if = {
			limit = {
				event_target:africa_world_union_protocol_host = { has_country_flag = africa_world_union_protocol_host_ballot_proven }
				event_target:africa_world_union_protocol_partner = { has_country_flag = africa_world_union_protocol_partner_ballot_proven }
			}
			event_target:africa_world_union_protocol_host = { set_country_flag = africa_world_union_protocol_ballots_complete }
			event_target:africa_world_union_protocol_partner = { set_country_flag = africa_world_union_protocol_ballots_complete }
			event_target:africa_world_union_protocol_host = { country_event = { id = africa_world_package.704 } }
		}
	}
}

			event_target:africa_world_union_protocol_host = { has_country_flag = africa_world_union_protocol_host_ballot_proven }

		set_country_flag = africa_world_union_protocol_distributed_government
		set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.ratified }

		event_target:africa_world_union_protocol_partner = {
			set_country_flag = africa_world_union_protocol_constitution_ratified
			set_country_flag = africa_world_union_protocol_distributed_government
			set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.ratified }
			set_variable = { africa_world_package_phase = constant:africa_world_package_phase.union_active }
		}
	}
}

africa_world_union_protocol_activate = {
	if = {
		limit = { africa_world_union_protocol_can_activate = yes }
		set_country_flag = africa_world_union_protocol_active
		set_country_flag = africa_world_union_partner
		set_country_flag = africa_dynamic_two_continent_union_active
		set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.active }
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.union_active }
		set_country_flag = africa_world_union_protocol_autonomous_partner_preserved
		event_target:africa_world_union_protocol_partner = {
			set_country_flag = africa_world_union_protocol_active
			set_country_flag = africa_world_union_partner
			set_country_flag = africa_world_union_protocol_autonomous_partner_preserved
			set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.active }
			set_variable = { africa_world_package_phase = constant:africa_world_package_phase.union_active }
		}
		if = {
			limit = { event_target:africa_world_union_protocol_partner = { check_variable = { africa_world_continent_id = constant:africa_world_continent.middle_east } } }
			set_cosmetic_tag = AFRICA_MIDDLE_EASTERN_UNION
		}
		else_if = {
			limit = { event_target:africa_world_union_protocol_partner = { check_variable = { africa_world_continent_id = constant:africa_world_continent.europe } } }
			set_cosmetic_tag = AFRO_EUROPEAN_UNION
		}
		else_if = {
			limit = { event_target:africa_world_union_protocol_partner = { check_variable = { africa_world_continent_id = constant:africa_world_continent.asia } } }
			set_cosmetic_tag = AFRO_ASIAN_UNION
		}
		else_if = {
			limit = { event_target:africa_world_union_protocol_partner = { check_variable = { africa_world_continent_id = constant:africa_world_continent.north_america } } }
			set_cosmetic_tag = AFRO_NORTH_AMERICAN_UNION
		}
		else_if = {
			limit = { event_target:africa_world_union_protocol_partner = { check_variable = { africa_world_continent_id = constant:africa_world_continent.south_america } } }
			set_cosmetic_tag = AFRO_SOUTH_AMERICAN_UNION
		}
		else_if = {
			limit = { event_target:africa_world_union_protocol_partner = { check_variable = { africa_world_continent_id = constant:africa_world_continent.oceania } } }
			set_cosmetic_tag = AFRO_OCEANIAN_UNION
		}
		set_global_flag = africa_dynamic_two_continent_union_active
		set_temp_variable = { africa_event_log_payload = constant:africa_event_log.two_continent_union }
		set_temp_variable = { africa_event_log_actor = THIS.id }
		set_temp_variable = { africa_event_log_has_actor = constant:africa_value.one }
		set_temp_variable = { africa_event_log_secondary_actor = event_target:africa_world_union_protocol_partner.id }
		set_temp_variable = { africa_event_log_has_secondary_actor = constant:africa_value.one }
		africa_record_event_log_world_order_entry = yes
	}
}

africa_world_union_protocol_apply_strain = {
	if = {
		limit = { africa_world_union_protocol_is_active = yes }
		add_to_variable = { africa_world_union_strain = constant:africa_world_union_protocol.tuning.strain_event_gain }
		clamp_variable = { var = africa_world_union_strain min = constant:africa_world_union_protocol.tuning.strain_min max = constant:africa_world_union_protocol.tuning.strain_max }
		set_country_flag = africa_world_union_strain_review_due

		event_target:africa_world_union_protocol_partner = {
			add_to_variable = { africa_world_union_strain = constant:africa_world_union_protocol.tuning.strain_event_gain }
			clamp_variable = { var = africa_world_union_strain min = constant:africa_world_union_protocol.tuning.strain_min max = constant:africa_world_union_protocol.tuning.strain_max }
			set_country_flag = africa_world_union_strain_review_due
			set_variable = { africa_world_union_status = constant:africa_world_union_protocol.status.strained }
		}
		country_event = { id = africa_world_package.709 }
	}
}

# COUNTRY scope on the attacker. FROM is the registered package opponent.
# The caller sets africa_world_continental_war_goal to one of the three W4
# goal constants before invoking this effect.
africa_world_continental_war_protocol_prepare = {
	if = {
		limit = {
			africa_world_continental_war_protocol_can_prepare = yes
			FROM = { africa_world_continental_war_protocol_target_is_registered = yes }
		}
		clear_global_event_target = africa_world_continental_war_protocol_attacker
		clear_global_event_target = africa_world_continental_war_protocol_defender
		save_global_event_target_as = africa_world_continental_war_protocol_attacker
		FROM = { save_global_event_target_as = africa_world_continental_war_protocol_defender }
		set_global_flag = africa_world_continental_war_protocol_open
		set_country_flag = africa_world_continental_war_protocol_attacker
		FROM = {
			set_country_flag = africa_world_continental_war_protocol_defender
			set_country_flag = africa_world_continental_war_protocol_review_pending
			set_variable = { africa_world_continental_war_status = constant:africa_world_war_protocol.status.mobilising }
			set_variable = { africa_world_package_phase = constant:africa_world_package_phase.war_active }
		}
		set_variable = { africa_world_continental_war_status = constant:africa_world_war_protocol.status.mobilising }
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.war_active }
		if = {
			limit = { NOT = { has_variable = africa_world_continental_war_goal } }
			set_variable = { africa_world_continental_war_goal = constant:africa_world_war_protocol.goal.restore_constituent_sovereignty }
		}
		set_country_flag = africa_world_continental_war_protocol_prepared
		FROM = { country_event = { id = africa_world_package.720 } }
	}
}

africa_world_continental_war_protocol_start_war = {
	if = {
		limit = {
			has_global_flag = africa_world_continental_war_protocol_open
			has_country_flag = africa_world_continental_war_protocol_attacker
			event_target:africa_world_continental_war_protocol_defender = { has_country_flag = africa_world_continental_war_protocol_defender }
		}
		set_variable = { africa_world_continental_war_status = constant:africa_world_war_protocol.status.active }
		set_country_flag = africa_world_continental_war_protocol_active
		event_target:africa_world_continental_war_protocol_defender = {
			set_variable = { africa_world_continental_war_status = constant:africa_world_war_protocol.status.active }
			set_country_flag = africa_world_continental_war_protocol_active
		}
		if = {
			limit = { check_variable = { africa_world_continental_war_goal = constant:africa_world_war_protocol.goal.restore_constituent_sovereignty } }
			declare_war_on = { target = event_target:africa_world_continental_war_protocol_defender type = liberate_wargoal }
		}
		else_if = {
			limit = { check_variable = { africa_world_continental_war_goal = constant:africa_world_war_protocol.goal.break_rival_compact } }
			declare_war_on = { target = event_target:africa_world_continental_war_protocol_defender type = topple_government }
		}
		else = {
			declare_war_on = { target = event_target:africa_world_continental_war_protocol_defender type = puppet_wargoal_focus }
		}
		if = {
			limit = { check_variable = { africa_world_continental_war_goal = constant:africa_world_war_protocol.goal.restore_constituent_sovereignty } }
			set_temp_variable = { africa_event_log_payload = constant:africa_event_log.continental_war_restore_constituent_sovereignty }
		}
		else_if = {
			limit = { check_variable = { africa_world_continental_war_goal = constant:africa_world_war_protocol.goal.break_rival_compact } }
			set_temp_variable = { africa_event_log_payload = constant:africa_event_log.continental_war_break_rival_compact }
		}
		else = { set_temp_variable = { africa_event_log_payload = constant:africa_event_log.continental_war_constitutional_settlement } }
		set_temp_variable = { africa_event_log_actor = THIS.id }
		set_temp_variable = { africa_event_log_has_actor = constant:africa_value.one }
		set_temp_variable = { africa_event_log_secondary_actor = event_target:africa_world_continental_war_protocol_defender.id }
		set_temp_variable = { africa_event_log_has_secondary_actor = constant:africa_value.one }
		africa_record_event_log_world_order_entry = yes
		set_country_flag = africa_world_continental_war_protocol_news_sent
		country_event = { id = africa_world_package.723 }
		event_target:africa_world_continental_war_protocol_defender = { country_event = { id = africa_world_package.723 } }
		africa_world_continental_war_protocol_call_registered_partners = yes
	}
}

# No partner is called merely because it has a union or package flag. The
# parent decision must first place the target in the attacker's or defender's
# africa_world_war_registered_partners array.

				array = africa_world_war_registered_partners

					limit = { africa_world_continental_war_protocol_partner_call_is_valid = yes }
					set_country_flag = africa_world_war_call_pending
					set_country_flag = africa_world_war_call_for_attacker
					country_event = { id = africa_world_package.721 }
				}
			}
		}

			for_each_scope_loop = {
				array = africa_world_war_registered_partners
				if = {
					limit = { africa_world_continental_war_protocol_partner_call_is_valid = yes }

					set_country_flag = africa_world_war_call_for_defender
					country_event = { id = africa_world_package.721 }
				}
			}
		}
	}
}
