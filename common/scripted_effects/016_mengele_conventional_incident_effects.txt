# Event 016 private conventional incidents and exact recovery transactions.
# Country scope throughout. Fixed family APIs never write the parent selectors.
# Callers authenticate one real output before invoking the weighted dispatcher.
# Each native decision owns its civilian-factory reservation and timer.
# All history, learned technologies and provider-stage receipts survive cleanup.

# Guidance Phantom: Recalibrate the guidance network and replace unreliable instruments so aircraft and anti-aircraft crews can trust their targets again.
brilliant_scientist_mengele_record_electronics_incident = {
	set_temp_variable = { mengele_event016_conventional_incident_recorded = constant:brilliant_scientist_value.zero }
	if = {
		limit = {
			brilliant_scientist_mengele_conventional_incident_provider_is_valid = yes
			brilliant_scientist_mengele_conventional_incident_stage_is_valid = yes
			NOT = { has_country_flag = mengele_event016_electronics_incident_active }
			NOT = { has_country_flag = mengele_event016_electronics_incident_recovery_active }
		}
		set_country_flag = mengele_event016_electronics_incident_active
		set_country_flag = mengele_event016_electronics_incident_history
		add_to_variable = { mengele_event016_electronics_incident_history_count = constant:brilliant_scientist_value.one }
		add_dynamic_modifier = { modifier = brilliant_scientist_mengele_electronics_incident }
		set_temp_variable = { mengele_event016_conventional_incident_recorded = constant:brilliant_scientist_value.one }
	}
}

brilliant_scientist_mengele_clear_electronics_incident = {
	clr_country_flag = mengele_event016_electronics_incident_active
	if = {
		limit = { has_dynamic_modifier = { modifier = brilliant_scientist_mengele_electronics_incident } }
		remove_dynamic_modifier = { modifier = brilliant_scientist_mengele_electronics_incident }
	}
}

# Called only by the native recovery decision after its affordability gate.
# Native admission owns CIC. Recheck direct costs without double-counting its reservation.
brilliant_scientist_mengele_begin_electronics_incident_recovery = {
	if = {
		limit = {
			brilliant_scientist_mengele_electronics_incident_recovery_requirements_met = yes
			brilliant_scientist_mengele_incident_technical_can_pay_direct = yes
		}
		set_variable = { mengele_event016_electronics_incident_recovery_cost_political_power = constant:brilliant_scientist_project_board.response_technical_political_power }
		set_variable = { mengele_event016_electronics_incident_recovery_cost_support_equipment = constant:brilliant_scientist_project_board.response_technical_support_equipment }
		set_variable = { mengele_event016_electronics_incident_recovery_cost_fuel = constant:brilliant_scientist_project_board.response_technical_fuel }
		set_temp_variable = { mengele_event016_incident_payment_pp = mengele_event016_electronics_incident_recovery_cost_political_power }
		multiply_temp_variable = { mengele_event016_incident_payment_pp = constant:brilliant_scientist_directorate_value.negative_one }
		add_political_power = mengele_event016_incident_payment_pp
		set_temp_variable = { equipment_stockpile_removal_amount = mengele_event016_electronics_incident_recovery_cost_support_equipment }
		remove_support_equipment_from_stockpile = yes
		set_temp_variable = { fuel_stockpile_removal_amount = mengele_event016_electronics_incident_recovery_cost_fuel }
		remove_fuel_from_stockpile = yes
		set_country_flag = mengele_event016_electronics_incident_recovery_active
	}
}

# Refund only the complete private receipt, with no inferred missing payment.
# Snapshot then consume the authority and values before issuing any refund.
brilliant_scientist_mengele_cancel_electronics_incident_recovery = {
	if = {
		limit = { brilliant_scientist_mengele_electronics_incident_recovery_receipt_is_current = yes }
		set_temp_variable = { mengele_event016_incident_refund_pp = mengele_event016_electronics_incident_recovery_cost_political_power }
		set_temp_variable = { mengele_event016_incident_refund_support = mengele_event016_electronics_incident_recovery_cost_support_equipment }
		set_temp_variable = { mengele_event016_incident_refund_fuel = mengele_event016_electronics_incident_recovery_cost_fuel }
		clr_country_flag = mengele_event016_electronics_incident_recovery_active
		clear_variable = mengele_event016_electronics_incident_recovery_cost_political_power
		clear_variable = mengele_event016_electronics_incident_recovery_cost_support_equipment
		clear_variable = mengele_event016_electronics_incident_recovery_cost_fuel
		add_political_power = mengele_event016_incident_refund_pp
		add_equipment_to_stockpile = { type = support_equipment amount = mengele_event016_incident_refund_support }
		add_fuel = mengele_event016_incident_refund_fuel
	}
	# Also retire incomplete receipts without inferring a refund or resolving the incident.
	clr_country_flag = mengele_event016_electronics_incident_recovery_active
	clear_variable = mengele_event016_electronics_incident_recovery_cost_political_power
	clear_variable = mengele_event016_electronics_incident_recovery_cost_support_equipment
	clear_variable = mengele_event016_electronics_incident_recovery_cost_fuel
	if = {
		limit = { NOT = { brilliant_scientist_mengele_conventional_incident_provider_is_valid = yes } }
		brilliant_scientist_mengele_clear_electronics_incident = yes
	}
}

brilliant_scientist_mengele_finish_electronics_incident_recovery = {
	set_temp_variable = { mengele_event016_conventional_recovery_finished = constant:brilliant_scientist_value.zero }
	if = {
		limit = {
			brilliant_scientist_mengele_electronics_incident_recovery_receipt_is_current = yes
			has_country_flag = mengele_event016_electronics_incident_active
			brilliant_scientist_mengele_conventional_incident_provider_is_valid = yes
		}
		clr_country_flag = mengele_event016_electronics_incident_recovery_active
		clear_variable = mengele_event016_electronics_incident_recovery_cost_political_power
		clear_variable = mengele_event016_electronics_incident_recovery_cost_support_equipment
		clear_variable = mengele_event016_electronics_incident_recovery_cost_fuel
		brilliant_scientist_mengele_clear_electronics_incident = yes
		set_country_flag = mengele_event016_electronics_recovery_history
		add_to_variable = { mengele_event016_electronics_recovery_history_count = constant:brilliant_scientist_value.one }
		set_temp_variable = { mengele_event016_conventional_recovery_finished = constant:brilliant_scientist_value.one }
	}
	else = { brilliant_scientist_mengele_cancel_electronics_incident_recovery = yes }
}

# Parent cleanup refunds before removing the native decision and its CIC use.
brilliant_scientist_mengele_cleanup_electronics_incident = {
