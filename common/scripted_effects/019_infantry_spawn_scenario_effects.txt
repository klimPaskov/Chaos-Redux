infantry_spawn_scenario_mark_actor = {
	if = {
		limit = { NOT = { has_country_flag = infantry_spawn_scenario_actor } }
		set_country_flag = infantry_spawn_scenario_actor
		# A queued natural first-family audience is not consumed by SCN-013.
		# Its already scheduled event fails the scenario trigger; clearing only
		# the dispatch latch preserves the pending natural reception for any
		# later return to the ordinary Event 19 lifecycle.
		if = {
			limit = {
				has_country_flag = infantry_spawn_first_family_reception_pending
				NOT = { has_country_flag = infantry_spawn_first_family_reception_resolved }
			}
			clr_country_flag = infantry_spawn_first_family_reception_dispatched
		}
		set_variable = { infantry_spawn_scenario_origin_country_id = global.infantry_spawn_scenario_origin_country_id }
		set_variable = { infantry_spawn_scenario_launch_serial = global.infantry_spawn_scenario_launch_serial }
		set_variable = { infantry_spawn_scenario_actor_type = global.infantry_spawn_scenario_type }
		set_variable = { infantry_spawn_scenario_actor_intensity = global.infantry_spawn_scenario_intensity }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.infantry_spawn_scenario_actor_countries
						value = THIS
					}
				}
			}
			add_to_array = {
				array = global.infantry_spawn_scenario_actor_countries
				value = THIS
			}
		}
		if = {
			limit = {
				check_variable = {
					var = global.infantry_spawn_scenario_origin_country_id
					value = THIS.id
					compare = equals
				}
			}
			set_country_flag = infantry_spawn_scenario_origin_actor
			clr_country_flag = infantry_spawn_scenario_hostile_actor
		}
		else = {
			clr_country_flag = infantry_spawn_scenario_origin_actor
			set_country_flag = infantry_spawn_scenario_hostile_actor
			add_to_variable = {
				global.infantry_spawn_scenario_hostile_actor_total = constant:infantry_spawn_scenario_value.one
			}
		}
	}
	infantry_spawn_reconcile_all_evolution_memberships = yes
}

infantry_spawn_scenario_apply_actor_pressure = {
	set_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.maximum }
	subtract_from_variable = { infantry_spawn_muster_control = global.infantry_spawn_scenario_revolt_pressure }
	set_variable = { infantry_spawn_scenario_pressure_floor = global.infantry_spawn_scenario_revolt_pressure }
	clamp_variable = {
		var = infantry_spawn_muster_control
		min = constant:infantry_spawn_control.minimum
		max = constant:infantry_spawn_control.maximum
	}
	if = {
		limit = {
			check_variable = {
				var = global.infantry_spawn_scenario_type
				value = constant:infantry_spawn_scenario.type_anomalous_rising
				compare = equals
			}
		}
		set_variable = { infantry_spawn_scenario_saturation_floor = global.infantry_spawn_scenario_revolt_pressure }
	}
	else = { clear_variable = infantry_spawn_scenario_saturation_floor }
	infantry_spawn_recalculate_country_pressure = yes
	infantry_spawn_schedule_country_pulse = yes
}

# Country scope. A proved ordinary scenario actor receives the complete
# minimum one-time entry state paired with its generated package before its
# durable scenario identity is exposed. A same-tag participant keeps any fully
# applied higher stage so live claimant, registry, and management transactions
# are not silently dismantled. The temporary profile lock remains set until the
# caller marks the actor, so no already-active global evolution can leak into
# pressure recalculation between this commit and actor registration.
infantry_spawn_scenario_commit_ordinary_actor_evolution_profile = {
	infantry_spawn_scenario_clear_actor_bypass = yes
	set_country_flag = infantry_spawn_scenario_setup_bypass
	set_country_flag = infantry_spawn_scenario_profile_locked
	set_country_flag = infantry_spawn_scenario_force_evolution_i
	set_country_flag = infantry_spawn_scenario_force_evolution_ii
	if = {
		limit = {
			OR = {
				check_variable = {
					var = global.infantry_spawn_scenario_type
					value = constant:infantry_spawn_scenario.type_general_mutiny
					compare = equals
				}
				check_variable = {
					var = global.infantry_spawn_scenario_type
					value = constant:infantry_spawn_scenario.type_anomalous_rising
					compare = equals
				}
			}
		}
		set_country_flag = infantry_spawn_scenario_force_evolution_iii
	}
	if = {
		limit = {
			check_variable = {
				var = global.infantry_spawn_scenario_type
				value = constant:infantry_spawn_scenario.type_anomalous_rising
				compare = equals
			}
		}
		set_country_flag = infantry_spawn_scenario_force_evolution_iv
	}
	set_temp_variable = {
		infantry_spawn_prefire_evolution_entry = constant:infantry_spawn_scenario_value.one
	}
	infantry_spawn_apply_evolution_i_to_country = yes
	infantry_spawn_apply_evolution_ii_to_country = yes
	if = {
		limit = { infantry_spawn_has_evolution_iii = yes }
		infantry_spawn_apply_evolution_iii_to_country = yes
	}
	if = {
		limit = { infantry_spawn_has_evolution_iv = yes }
		infantry_spawn_apply_evolution_iv_to_country = yes
	}
	set_temp_variable = {
		infantry_spawn_prefire_evolution_entry = constant:infantry_spawn_scenario_value.zero
	}
}

infantry_spawn_scenario_build_coherent_revolt_region = {
	clear_temp_array = infantry_spawn_scenario_selected_revolt_states
	set_temp_variable = { infantry_spawn_scenario_region_eligible_states = constant:infantry_spawn_scenario_value.zero }
	every_owned_state = {
		limit = {
			is_controlled_by = PREV
			impassable = no
			NOT = { is_capital = yes }
			NOT = { is_island_state = yes }
		}
		add_to_temp_variable = {
			infantry_spawn_scenario_region_eligible_states = constant:infantry_spawn_scenario_value.one
		}
	}
	if = {
		limit = {
			check_variable = {
				var = infantry_spawn_scenario_region_eligible_states
				value = constant:infantry_spawn_scenario_value.zero
				compare = greater_than
			}
		}
		set_temp_variable = { infantry_spawn_eligible_state_count = infantry_spawn_scenario_region_eligible_states }
		infantry_spawn_calculate_selected_state_target = yes
		set_temp_variable = { infantry_spawn_scenario_region_state_target = infantry_spawn_scenario_region_eligible_states }
		multiply_temp_variable = {
			infantry_spawn_scenario_region_state_target = global.infantry_spawn_scenario_state_coverage
		}
		round_temp_variable = infantry_spawn_scenario_region_state_target
		clamp_temp_variable = {
			var = infantry_spawn_scenario_region_state_target
			min = constant:infantry_spawn_scenario_value.one
			max = infantry_spawn_selected_state_target
		}
		random_owned_state = {
			limit = { infantry_spawn_scenario_revolt_state_is_candidate = yes }
			add_to_temp_array = {
				array = infantry_spawn_scenario_selected_revolt_states
				value = THIS
			}
			save_event_target_as = infantry_spawn_scenario_revolt_anchor_state
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					var = infantry_spawn_scenario_selected_revolt_states^num
					value = infantry_spawn_scenario_region_state_target
					compare = less_than
				}
			}
			set_temp_variable = { infantry_spawn_scenario_region_expansion_found = constant:infantry_spawn_scenario_value.zero }
			random_owned_state = {
				limit = { infantry_spawn_scenario_revolt_state_is_candidate = yes }
				add_to_temp_array = {
					array = infantry_spawn_scenario_selected_revolt_states
					value = THIS
				}
				set_temp_variable = {
					infantry_spawn_scenario_region_expansion_found = constant:infantry_spawn_scenario_value.one
				}
			}
			if = {
				limit = {
					check_variable = {
						var = infantry_spawn_scenario_region_expansion_found
						value = constant:infantry_spawn_scenario_value.zero
						compare = equals
					}
				}
				set_temp_variable = {
					infantry_spawn_scenario_region_state_target = infantry_spawn_scenario_selected_revolt_states^num
				}
			}
		}
	}
}

infantry_spawn_scenario_select_derivative_family = {
	set_temp_variable = { infantry_spawn_scenario_selected_registry_index = constant:infantry_spawn_scenario_value.invalid_index }
	set_temp_variable = { infantry_spawn_scenario_derivative_total_weight = constant:infantry_spawn_scenario_value.zero }
	if = {
		limit = { chaos_unit_family_registry_is_aligned = yes }
		set_temp_variable = { chaos_unit_family_registry_index = constant:infantry_spawn_event.first_index }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = chaos_unit_family_registry_index
					value = global.chaos_unit_family_id_entries^num
					compare = less_than
				}
			}
			if = {
				limit = { infantry_spawn_scenario_current_registry_row_can_form_derivative = yes }
				add_to_temp_variable = {
					infantry_spawn_scenario_derivative_total_weight = global.chaos_unit_family_spawn_weight_entries^chaos_unit_family_registry_index
				}
			}
			add_to_temp_variable = {
				chaos_unit_family_registry_index = constant:infantry_spawn_scenario_value.one
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = infantry_spawn_scenario_derivative_total_weight
				value = constant:infantry_spawn_scenario_value.zero
				compare = greater_than
			}
		}
		set_temp_variable = { infantry_spawn_scenario_derivative_roll_max = infantry_spawn_scenario_derivative_total_weight }
		add_to_temp_variable = {
			infantry_spawn_scenario_derivative_roll_max = constant:infantry_spawn_scenario_value.one
		}
		set_temp_variable_to_random = {
			var = infantry_spawn_scenario_derivative_roll
			min = constant:infantry_spawn_scenario_value.one
			max = infantry_spawn_scenario_derivative_roll_max
			integer = yes
		}
		set_temp_variable = { chaos_unit_family_registry_index = constant:infantry_spawn_event.first_index }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = infantry_spawn_scenario_selected_registry_index
					value = constant:infantry_spawn_event.first_index
					compare = less_than
				}
				check_variable = {
					var = chaos_unit_family_registry_index
					value = global.chaos_unit_family_id_entries^num
					compare = less_than
				}
			}
			if = {
				limit = { infantry_spawn_scenario_current_registry_row_can_form_derivative = yes }
				set_temp_variable = {
					infantry_spawn_scenario_derivative_current_weight = global.chaos_unit_family_spawn_weight_entries^chaos_unit_family_registry_index
				}
				if = {
					limit = {
						check_variable = {
							var = infantry_spawn_scenario_derivative_roll
							value = infantry_spawn_scenario_derivative_current_weight
							compare = greater_than
						}
					}
					subtract_from_temp_variable = {
						infantry_spawn_scenario_derivative_roll = infantry_spawn_scenario_derivative_current_weight
					}
				}
				else = {
					set_temp_variable = {
						infantry_spawn_scenario_selected_registry_index = chaos_unit_family_registry_index
					}
				}
			}
			add_to_temp_variable = {
				chaos_unit_family_registry_index = constant:infantry_spawn_scenario_value.one
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = infantry_spawn_scenario_selected_registry_index
				value = constant:infantry_spawn_scenario_value.invalid_index
				compare = greater_than
			}
		}
		set_temp_variable = {
			infantry_spawn_scenario_selected_family_id = global.chaos_unit_family_id_entries^infantry_spawn_scenario_selected_registry_index
		}
		set_temp_variable = {
			infantry_spawn_scenario_selected_provider_id = global.chaos_unit_family_provider_id_entries^infantry_spawn_scenario_selected_registry_index
		}
	}
}

# deliberately skipping registered-family selection. Conventional Flood and
# Arsenal Lottery can therefore never leak Evolution IV families even when the
# receiving tag is natively compatible with one of those providers.

# Actor-country scope. Package success is proved from the exact rows appended
# during this actor transaction. Every successful branch also proves that the
# scenario/setup identity still excludes the actor from ordinary evolution
# history, so scenario materialization never needs to rewind shared counters.
# General Mutiny's dynamic derivative identity owns a separate private-ledger
# initialization, so its safe proof is the exact lot/claimant materialization
# result plus the completed breakaway identity.
infantry_spawn_scenario_evaluate_actor_package = {
	set_temp_variable = { infantry_spawn_scenario_generation_rows_created = infantry_spawn_generation_uid_entries^num }
	subtract_from_temp_variable = {
		infantry_spawn_scenario_generation_rows_created = infantry_spawn_scenario_generation_rows_before_package
	}
	set_temp_variable = { infantry_spawn_scenario_lot_rows_created = infantry_spawn_lot_uid_entries^num }
	subtract_from_temp_variable = {
		infantry_spawn_scenario_lot_rows_created = infantry_spawn_scenario_lot_rows_before_package
	}
	set_temp_variable = { infantry_spawn_scenario_unit_rows_created = infantry_spawn_unit_uid_entries^num }
	subtract_from_temp_variable = {
		infantry_spawn_scenario_unit_rows_created = infantry_spawn_scenario_unit_rows_before_package
	}
	set_temp_variable = { infantry_spawn_scenario_claimant_rows_created = infantry_spawn_claimant_uid_entries^num }
	subtract_from_temp_variable = {
		infantry_spawn_scenario_claimant_rows_created = infantry_spawn_scenario_claimant_rows_before_package
	}
	set_temp_variable = {
		infantry_spawn_scenario_new_generation_rows_complete = constant:infantry_spawn_scenario_value.one
	}
	set_temp_variable = {
		infantry_spawn_scenario_generation_proof_index = infantry_spawn_scenario_generation_rows_before_package
	}
	while_loop_effect = {
		limit = {
			check_variable = {
				var = infantry_spawn_scenario_generation_proof_index
				value = infantry_spawn_generation_uid_entries^num
				compare = less_than
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = {
						var = infantry_spawn_generation_lot_count_entries^infantry_spawn_scenario_generation_proof_index
						value = constant:infantry_spawn_scenario_value.zero
						compare = less_than_or_equals
					}
					check_variable = {
						var = infantry_spawn_generation_unit_count_entries^infantry_spawn_scenario_generation_proof_index
						value = constant:infantry_spawn_scenario_value.zero
						compare = less_than_or_equals
					}
				}
			}
			set_temp_variable = {
				infantry_spawn_scenario_new_generation_rows_complete = constant:infantry_spawn_scenario_value.zero
			}
		}
		add_to_temp_variable = {
			infantry_spawn_scenario_generation_proof_index = constant:infantry_spawn_scenario_value.one
		}
	}
	set_temp_variable = {
		infantry_spawn_scenario_actor_package_ready = constant:infantry_spawn_scenario_value.zero
	}
	if = {
		limit = {
			NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
			NOT = { infantry_spawn_contributes_to_ordinary_evolution_history = yes }
			infantry_spawn_all_ledgers_are_aligned = yes
			OR = {
				check_variable = {
					var = global.infantry_spawn_scenario_type
					value = constant:infantry_spawn_scenario.type_conventional_flood
					compare = equals
				}
				check_variable = {
					var = global.infantry_spawn_scenario_type
					value = constant:infantry_spawn_scenario.type_arsenal_lottery
					compare = equals
				}
			}
			check_variable = {
				var = infantry_spawn_scenario_generation_rows_created
				value = global.infantry_spawn_scenario_generation_passes
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_lot_rows_created
				value = constant:infantry_spawn_scenario_value.zero
				compare = greater_than
			}
			check_variable = {
				var = infantry_spawn_scenario_unit_rows_created
				value = constant:infantry_spawn_scenario_value.zero
				compare = greater_than
			}
			check_variable = {
				var = infantry_spawn_scenario_new_generation_rows_complete
				value = constant:infantry_spawn_scenario_value.one
				compare = equals
			}
		}
		set_temp_variable = {
			infantry_spawn_scenario_actor_package_ready = constant:infantry_spawn_scenario_value.one
		}
	}
	else_if = {
		limit = {
			NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
			NOT = { infantry_spawn_contributes_to_ordinary_evolution_history = yes }
			infantry_spawn_all_ledgers_are_aligned = yes
			infantry_spawn_claimant_ledgers_are_aligned = yes
			check_variable = {
				var = global.infantry_spawn_scenario_type
				value = constant:infantry_spawn_scenario.type_general_mutiny
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_generation_rows_created
				value = constant:infantry_spawn_scenario_value.one
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_random_lots_materialized
				value = global.infantry_spawn_scenario_random_lot_target
				compare = equals
			}
			check_variable = {

				value = global.infantry_spawn_scenario_random_lot_target
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_lot_rows_created
				value = global.infantry_spawn_scenario_random_lot_target
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_claimant_rows_created
				value = constant:infantry_spawn_scenario_value.one
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_unit_rows_created
				value = global.infantry_spawn_scenario_random_lot_target
				compare = equals
			}
			check_variable = {
				var = infantry_spawn_scenario_new_generation_rows_complete
				value = constant:infantry_spawn_scenario_value.one
				compare = equals
			}
			OR = {
				AND = {
					has_country_flag = infantry_spawn_scenario_takeover_actor
					check_variable = {
						var = infantry_spawn_scenario_general_takeover_prepared
						value = constant:infantry_spawn_scenario_value.one
						compare = equals
					}
				}
				AND = {
					has_country_flag = infantry_spawn_scenario_dynamic_breakaway
					check_variable = {

						value = constant:infantry_spawn_scenario_value.one
						compare = equals
					}
					has_country_flag = infantry_spawn_claimant_breakaway
					has_country_flag = infantry_spawn_derivative_package_initialized
				}
			}
		}
		set_temp_variable = {
			infantry_spawn_scenario_actor_package_ready = constant:infantry_spawn_scenario_value.one
		}

	else_if = {
		limit = {

			NOT = { infantry_spawn_contributes_to_ordinary_evolution_history = yes }

				var = global.infantry_spawn_scenario_type

			}

			}

# Actor-country scope, parent carrier passed as PREV. Vanilla on_annex
# listeners dereference the annexed carrier's capital, and a failed region
# transfer can leave the provisional actor stateless and capitalless. Borrow
# one owned state from the parent and anchor the capital on it; the
# transaction annex returns the borrowed state in the same synchronous step.
# Non-capital states are preferred so the parent's capital never relocates.
infantry_spawn_ensure_dynamic_carrier_annexable_capital = {
	if = {
		limit = { NOT = { check_variable = { num_owned_states > constant:infantry_spawn_scenario_value.zero } } }
		if = {
			limit = { PREV = { any_owned_state = { NOT = { is_capital = yes } } } }
			PREV = {
				random_owned_state = {
					limit = { NOT = { is_capital = yes } }
					ROOT = {
						transfer_state = PREV
						add_state_core = PREV
						set_capital = {
							state = PREV
							remember_old_capital = no
						}
					}
				}
			}
		}
		else_if = {
			limit = { PREV = { check_variable = { num_owned_states > constant:infantry_spawn_scenario_value.zero } } }
			PREV = {
				random_owned_state = {
					ROOT = {
						transfer_state = PREV
						add_state_core = PREV
						set_capital = {
							state = PREV
							remember_old_capital = no
						}
					}
				}
			}
		}
	}
}

# Parent-country scope. This is a transaction rollback, not an alternate
# scenario outcome: a provider contract failure first deletes the fresh actor's
# entire provisional army without refunds and proves that no division remains.
# Only then does it return the exact state base, with troop transfer disabled.
# A failed proof retains the actor, schedules actor-scoped cleanup retries, and
# never leaks units into the parent. Scenario actors are excluded at every
# ordinary-history append site, so retries never write a frozen shared total.
# The host remains processed so setup retries cannot be exploited.

infantry_spawn_scenario_rollback_dynamic_actor_creation = {
	if = {
		limit = {
			has_event_target = infantry_spawn_scenario_dynamic_actor
			has_event_target = infantry_spawn_scenario_parent_country
			event_target:infantry_spawn_scenario_parent_country = { exists = yes }
		}
		set_temp_variable = { infantry_spawn_scenario_rollback_army_cleared = constant:infantry_spawn_scenario_value.zero }
		event_target:infantry_spawn_scenario_dynamic_actor = {
			delete_unit = { disband = no }
			if = {
				limit = { NOT = { num_divisions > constant:infantry_spawn_scenario_value.zero } }
				set_temp_variable = { infantry_spawn_scenario_rollback_army_cleared = constant:infantry_spawn_scenario_value.one }
			}
		}
		if = {
			limit = { check_variable = { infantry_spawn_scenario_rollback_army_cleared = constant:infantry_spawn_scenario_value.one } }
			event_target:infantry_spawn_scenario_dynamic_actor = {
					every_owned_state = {
						if = {
							limit = { is_core_of = event_target:infantry_spawn_scenario_dynamic_actor }
							event_target:infantry_spawn_scenario_dynamic_actor = { remove_state_core = PREV }
					}
				}
			}
			event_target:infantry_spawn_scenario_parent_country = {
				event_target:infantry_spawn_scenario_dynamic_actor = {
					infantry_spawn_ensure_dynamic_carrier_annexable_capital = yes
				}
				annex_country = {
					target = event_target:infantry_spawn_scenario_dynamic_actor
