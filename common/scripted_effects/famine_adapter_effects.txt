# STATE scope after the shared current score has been calculated. This receipt
# records assessment provenance without committing a simulation stage.
famine_record_food_assessment = {
	set_state_flag = famine_food_assessment_current
	if = {
		limit = {
			check_variable = { famine_food_stage = constant:famine_food_stage.stable }
			check_variable = {
				var = famine_food_security_score
				value = constant:famine_food_score.supply_strain_threshold
				compare = greater_than_or_equals
			}
		}
		set_state_flag = famine_food_observation_pending_material
	}
	else = { clr_state_flag = famine_food_observation_pending_material }
}

# STATE scope. Publish only famine-owned current assessment facts. Publication
# never initializes the state or grants simulation, route or service proof.
famine_publish_migration_food_safety = {
	set_temp_variable = { famine_to_migration_food_safety_publish_result = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			has_state_flag = famine_food_assessment_current
			NOT = { has_state_flag = famine_food_observation_pending_material }
			has_variable = famine_food_stage
			has_variable = famine_food_pressure
			has_variable = famine_component_transport
			has_variable = famine_component_governance
			has_variable = famine_component_production
		}
		set_variable = { famine_to_migration_food_safety_schema_version = constant:famine_to_migration_food_safety.schema_version }
		set_variable = { famine_to_migration_food_safety_proven = constant:humanitarian_runtime.one }
		if = { limit = { NOT = { has_variable = famine_to_migration_food_safety_generation } } set_variable = { famine_to_migration_food_safety_generation = constant:humanitarian_runtime.zero } }
		if = { limit = { NOT = { has_variable = famine_to_migration_food_safety_revision } } set_variable = { famine_to_migration_food_safety_revision = constant:humanitarian_runtime.zero } }
		add_to_variable = { famine_to_migration_food_safety_generation = constant:humanitarian_runtime.one }
		add_to_variable = { famine_to_migration_food_safety_revision = constant:humanitarian_runtime.one }
		set_variable = { famine_to_migration_food_stage = famine_food_stage }
		set_variable = { famine_to_migration_food_pressure = famine_food_pressure }
		set_variable = { famine_to_migration_food_exposure_days = constant:humanitarian_runtime.zero }
		set_variable = { famine_to_migration_food_incident_count = constant:humanitarian_runtime.zero }
		if = { limit = { has_variable = famine_food_exposure_days } set_variable = { famine_to_migration_food_exposure_days = famine_food_exposure_days } }
		if = { limit = { has_variable = famine_food_incident_count } set_variable = { famine_to_migration_food_incident_count = famine_food_incident_count } }
		set_variable = { famine_to_migration_transport_component = famine_component_transport }
		set_variable = { famine_to_migration_governance_component = famine_component_governance }
		set_variable = { famine_to_migration_production_component = famine_component_production }
		set_temp_variable = { famine_to_migration_food_safety_publish_result = constant:humanitarian_runtime.one }
	}
	else = { famine_invalidate_migration_food_safety = yes }
}

famine_invalidate_migration_food_safety = {
	clr_state_flag = famine_food_assessment_current
	set_variable = { famine_to_migration_food_safety_proven = constant:humanitarian_runtime.zero }
	clear_variable = famine_to_migration_food_stage
	clear_variable = famine_to_migration_food_pressure
	clear_variable = famine_to_migration_food_exposure_days
	clear_variable = famine_to_migration_food_incident_count
	clear_variable = famine_to_migration_transport_component
	clear_variable = famine_to_migration_governance_component
	clear_variable = famine_to_migration_production_component
	migration_mark_reception_capacity_dirty_if_candidate = yes
}

# STATE scope. Validate one exact state-local food receipt before a source
# adapter is allowed to submit famine pressure. Proof values are caller-owned
# and are never inferred from a death amount, modifier, country flag, or event
# id. The actor target is required so a numeric proof token cannot stand in for
# an unidentified country.
famine_validate_state_local_food_receipt_exact = {
	set_temp_variable = { famine_adapter_contract_valid = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			check_variable = { famine_adapter_state_proven > constant:humanitarian_runtime.zero }
			check_variable = { var = famine_adapter_state_proven value = THIS.id compare = equals }
			check_variable = { famine_adapter_food_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_environment_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_transport_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_policy_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_actor_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_cause_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_source_proven > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_amount > constant:humanitarian_population.zero }
			check_variable = { famine_adapter_generation > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_revision > constant:humanitarian_runtime.zero }
			check_variable = { famine_adapter_request_id > constant:humanitarian_runtime.zero }
			has_event_target = famine_adapter_actor
			event_target:famine_adapter_actor = { exists = yes }
			check_variable = { var = famine_adapter_actor_proven value = event_target:famine_adapter_actor.id compare = equals }
		}
		set_temp_variable = { famine_adapter_contract_valid = constant:humanitarian_runtime.one }
	}

# the receiving country is condemned for obstructing relief. Only a refusal the
# receiver could have avoided counts: its safe front and current reception

			}

			limit = {
				event_target:famine_condemnation_actor = { humanitarian_country_is_valid = yes }
				NOT = { is_owned_by = event_target:famine_condemnation_actor }

			set_temp_variable = { famine_condemnation_actor_proven = humanitarian_corridor_counterpart_country_id }
			set_temp_variable = { famine_condemnation_cause_proven = constant:condemnation_context.relief_obstruction }
			set_temp_variable = { famine_condemnation_people_amount = humanitarian_corridor_transfer_people }
			set_temp_variable = { famine_condemnation_generation = humanitarian_corridor_route_generation }
			set_temp_variable = { famine_condemnation_revision = famine_food_stage }
			set_temp_variable = { famine_condemnation_request_id = humanitarian_corridor_route_generation }
			if = {
				limit = {
					has_variable = humanitarian_corridor_cohort_id
					check_variable = { var = humanitarian_corridor_cohort_id value = constant:humanitarian_runtime.zero compare = greater_than }
				}
				set_temp_variable = { famine_condemnation_request_id = humanitarian_corridor_cohort_id }
			}
			set_temp_variable = { condemnation_civilian_deaths = constant:humanitarian_population.zero }
			set_temp_variable = { condemnation_contamination_value = constant:humanitarian_population.zero }
			event_target:famine_condemnation_actor = { famine_condemn_relief_obstruction = yes }
			if = {
				limit = { check_variable = { var = famine_condemnation_result value = constant:humanitarian_runtime.one compare = equals } }
				set_variable = { famine_corridor_obstruction_condemned_generation = humanitarian_corridor_route_generation }
			}
		}
	}
	famine_clear_condemnation_envelope = yes
}

# STATE scope: a famine state whose foreign relief contract has just failed
# before delivery, while its contract fields still exist. When the donor state
# has been captured by a country at war with the relief actor and this state is
# at the famine or catastrophic stage, that capturing country is condemned for
# obstructing relief. The affected people are those the contract would have
# fed: the live population scaled by the contract amount against the reserve
# import it was sized for, capped at the whole population. One contract
# generation can condemn only once.
# Output: temporary famine_condemnation_result, declared by callers that read it.
famine_condemn_relief_contract_interruption = {
	set_temp_variable = { famine_condemnation_result = constant:humanitarian_runtime.zero }
	set_temp_variable = { famine_relief_interruption_people = constant:humanitarian_population.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			OR = {
				has_state_flag = famine_food_stage_famine
				has_state_flag = famine_food_stage_catastrophic
			}
			has_variable = famine_food_stage
			check_variable = { var = famine_food_stage value = constant:famine_food_stage.famine compare = greater_than_or_equals }
			has_variable = famine_relief_contract_generation
			check_variable = { var = famine_relief_contract_generation value = constant:humanitarian_runtime.zero compare = greater_than }
			has_variable = famine_relief_contract_amount
			check_variable = { var = famine_relief_contract_amount value = constant:famine_food_reserve.zero compare = greater_than }
			has_variable = famine_relief_actor
			has_variable = famine_relief_donor_state
			OR = {
				NOT = { has_variable = famine_relief_obstruction_condemned_generation }
				NOT = { check_variable = { var = famine_relief_obstruction_condemned_generation value = famine_relief_contract_generation compare = equals } }
			}
		}
		var:famine_relief_actor = { save_event_target_as = famine_relief_interruption_actor }
		var:famine_relief_donor_state = {
			save_event_target_as = famine_relief_interruption_donor_state
			CONTROLLER = { save_event_target_as = famine_condemnation_actor }
		}
		if = {
			limit = {
				event_target:famine_relief_interruption_actor = { humanitarian_country_is_valid = yes }
				event_target:famine_condemnation_actor = {
					humanitarian_country_is_valid = yes
					NOT = { tag = event_target:famine_relief_interruption_actor }
					has_war_with = event_target:famine_relief_interruption_actor
					NOT = { is_in_faction_with = event_target:famine_relief_interruption_actor }
				}
				event_target:famine_relief_interruption_donor_state = {
					NOT = { is_owned_by = event_target:famine_condemnation_actor }
				}
			}
			# People the contract would have fed.
			set_temp_variable = { famine_relief_interruption_people = state_population_k }
			multiply_temp_variable = { famine_relief_interruption_people = constant:humanitarian_population.people_per_k }
			if = {
				limit = {
					has_variable = famine_food_reserve_daily_need
					check_variable = { var = famine_food_reserve_daily_need value = constant:famine_food_reserve.zero compare = greater_than }
				}
				set_temp_variable = { famine_relief_interruption_sized_import = famine_food_reserve_daily_need }
				multiply_temp_variable = { famine_relief_interruption_sized_import = constant:famine_mission_contract.reserve_import_days }
				set_temp_variable = { famine_relief_interruption_share = famine_relief_contract_amount }
				divide_temp_variable = { famine_relief_interruption_share = famine_relief_interruption_sized_import }
				clamp_temp_variable = { var = famine_relief_interruption_share min = constant:humanitarian_population.zero max = constant:humanitarian_runtime.one }
				multiply_temp_variable = { famine_relief_interruption_people = famine_relief_interruption_share }
			}
			round_temp_variable = famine_relief_interruption_people
			save_event_target_as = famine_condemnation_state
			set_temp_variable = { famine_condemnation_state_proven = id }
			set_temp_variable = { famine_condemnation_food_proven = famine_food_stage }
			set_temp_variable = { famine_condemnation_environment_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { famine_condemnation_transport_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { famine_condemnation_policy_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { famine_condemnation_actor_proven = event_target:famine_condemnation_actor.id }
			set_temp_variable = { famine_condemnation_cause_proven = constant:condemnation_context.relief_obstruction }
			set_temp_variable = { famine_condemnation_people_amount = famine_relief_interruption_people }
			set_temp_variable = { famine_condemnation_generation = famine_relief_contract_generation }
			set_temp_variable = { famine_condemnation_revision = famine_food_stage }
			set_temp_variable = { famine_condemnation_request_id = event_target:famine_relief_interruption_donor_state.id }
			set_temp_variable = { condemnation_civilian_deaths = constant:humanitarian_population.zero }
			set_temp_variable = { condemnation_contamination_value = constant:humanitarian_population.zero }
			event_target:famine_condemnation_actor = { famine_condemn_relief_obstruction = yes }
			if = {
				limit = { check_variable = { var = famine_condemnation_result value = constant:humanitarian_runtime.one compare = equals } }
				set_variable = { famine_relief_obstruction_condemned_generation = famine_relief_contract_generation }
			}
		}
	}
	set_temp_variable = { famine_relief_interruption_people = constant:humanitarian_population.zero }
	famine_clear_condemnation_envelope = yes
}

# ---------------------------------------------------------------------------
# Hunger Lines scenario (Event 014, SCN-010)
# ---------------------------------------------------------------------------
# Any scope. Called once by the shared scenario launcher after
# trigger_cannibalism_scenario reports a successful setup. It reads only
# Event 014's bounded public node registry, global.cannibalism_node_states,
# and the launch intensity Event 014 records in
# global.cannibalism_scenario_last_intensity; it never scans the map and never
# writes Event 014 state. Each live node state whose previous scenario food
# context has expired:
# - receives a proven famine scenario request of food_share of its population;
# - receives a scenario food context (famine_scenario_food_context_pressure
#   until famine_scenario_food_context_until_date) that the surface collector
#   adds to production, transport, governance and vulnerability while it lasts;
# - loses its local food stores to the seizure (recorded as depleted stock);
# - commits the starting food stage for the intensity at once (acute shortage
#   at Low, famine above), so the famine is visible on the day of the launch.
#   The ordinary stage transition then submits the famine survivor flight to
#   migration, so the scenario adds no separate flight request.
# A state whose context is still running is skipped, so a relaunch cannot
# stack pressure. Output: temporary famine_hunger_lines_state_count, the number
# of node states that received pressure, declared by callers that read it.
famine_apply_hunger_lines_scenario_pressure = {
	set_temp_variable = { famine_hunger_lines_state_count = constant:humanitarian_runtime.zero }
	set_temp_variable = { famine_hunger_lines_food_share = constant:famine_hunger_lines_scenario.low_food_share }
	set_temp_variable = { famine_hunger_lines_context_pressure = constant:famine_hunger_lines_scenario.low_context_pressure }
	set_temp_variable = { famine_hunger_lines_start_stage = constant:famine_hunger_lines_scenario.low_start_stage }
	if = {
		limit = { check_variable = { var = global.cannibalism_scenario_last_intensity value = constant:triggerable_scenario_intensity.medium compare = equals } }
		set_temp_variable = { famine_hunger_lines_food_share = constant:famine_hunger_lines_scenario.medium_food_share }
		set_temp_variable = { famine_hunger_lines_context_pressure = constant:famine_hunger_lines_scenario.medium_context_pressure }
		set_temp_variable = { famine_hunger_lines_start_stage = constant:famine_hunger_lines_scenario.medium_start_stage }
	}
	else_if = {
		limit = { check_variable = { var = global.cannibalism_scenario_last_intensity value = constant:triggerable_scenario_intensity.high compare = equals } }
		set_temp_variable = { famine_hunger_lines_food_share = constant:famine_hunger_lines_scenario.high_food_share }
		set_temp_variable = { famine_hunger_lines_context_pressure = constant:famine_hunger_lines_scenario.high_context_pressure }
		set_temp_variable = { famine_hunger_lines_start_stage = constant:famine_hunger_lines_scenario.high_start_stage }
	}
	else_if = {
		limit = { check_variable = { var = global.cannibalism_scenario_last_intensity value = constant:triggerable_scenario_intensity.maximum compare = equals } }
		set_temp_variable = { famine_hunger_lines_food_share = constant:famine_hunger_lines_scenario.maximum_food_share }
		set_temp_variable = { famine_hunger_lines_context_pressure = constant:famine_hunger_lines_scenario.maximum_context_pressure }
		set_temp_variable = { famine_hunger_lines_start_stage = constant:famine_hunger_lines_scenario.maximum_start_stage }
	}
	set_temp_variable = { chaosx_date_days = constant:famine_hunger_lines_scenario.context_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_temp_variable = { famine_hunger_lines_context_offset = chaosx_date_offset }
	for_each_scope_loop = {
		array = global.cannibalism_node_states
		if = {
			limit = {
				has_state_flag = cannibalism_active_node
				humanitarian_state_is_valid = yes
				OR = {
					NOT = { has_variable = famine_scenario_food_context_until_date }
					check_variable = { var = famine_scenario_food_context_until_date value = global.date compare = less_than_or_equals }
				}
			}
			set_variable = { famine_scenario_food_context_until_date = global.date }
			add_to_variable = { famine_scenario_food_context_until_date = famine_hunger_lines_context_offset }
			set_variable = { famine_scenario_food_context_pressure = famine_hunger_lines_context_pressure }
			set_temp_variable = { famine_hunger_lines_population = state_population_k }
			multiply_temp_variable = { famine_hunger_lines_population = constant:humanitarian_population.people_per_k }
			round_temp_variable = famine_hunger_lines_population
			# Food: the shock enters unmet need through the ordinary request,
			# which also registers the state with the daily food job.
			set_temp_variable = { famine_pressure_request_result = constant:humanitarian_runtime.zero }
			set_temp_variable = { famine_pressure_request_amount = famine_hunger_lines_population }
			multiply_temp_variable = { famine_pressure_request_amount = famine_hunger_lines_food_share }
			round_temp_variable = famine_pressure_request_amount
			set_temp_variable = { famine_pressure_request_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { famine_pressure_request_actor_proven = constant:humanitarian_runtime.one }
			famine_request_scenario_pressure = yes
			if = {
				limit = { check_variable = { var = famine_pressure_request_result value = constant:humanitarian_runtime.one compare = equals } }
				# The seizure: local stores are taken. Initialize the ledger first
				# so the drained amount is recorded as depleted stock.
				famine_refresh_reserve_capacity = yes
				if = {
					limit = { check_variable = { var = famine_food_reserve_amount value = constant:famine_food_reserve.zero compare = greater_than } }
					set_variable = { famine_food_reserve_last_depleted = famine_food_reserve_amount }
					add_to_variable = { famine_food_reserve_total_depleted = famine_food_reserve_amount }
					set_variable = { famine_food_reserve_amount = constant:famine_food_reserve.zero }
				}
				set_state_flag = famine_food_reserve_initialized
				# Commit the starting stage now. The context above keeps the score
				# at that stage for the context window unless relief arrives.
				if = {
					limit = { check_variable = { var = famine_food_stage value = famine_hunger_lines_start_stage compare = less_than } }
					set_temp_variable = { famine_requested_food_stage = famine_hunger_lines_start_stage }
					famine_transition_food_stage = yes
				}
				add_to_temp_variable = { famine_hunger_lines_state_count = constant:humanitarian_runtime.one }
			}
		}
	}
	set_temp_variable = { famine_pressure_request_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { famine_pressure_request_actor_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { famine_pressure_request_amount = constant:humanitarian_population.zero }
}
