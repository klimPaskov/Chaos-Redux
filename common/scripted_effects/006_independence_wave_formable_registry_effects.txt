# Member-country scope. An integration adapter may call this only for a row
# already recorded as consenting. It ends the Event 006 active origin with the
# canonical formable-absorption reason, preserving the historical-origin row
# and avoiding any Event 005 origin mutation.
independence_wave_formable_end_consenting_member_origin = {
	save_event_target_as = independence_wave_formable_member_to_end
	set_temp_variable = { independence_wave_formable_member_authorized = constant:independence_wave_formable_tuning.no_entries }
	event_target:independence_wave_formable_proposer = {
		if = {
			limit = { independence_wave_formable_member_arrays_are_aligned = yes }
			set_temp_variable = { independence_wave_formable_member_index = constant:independence_wave_formable_tuning.first_array_index }
			while_loop_effect = {
				limit = {
					check_variable = { independence_wave_formable_member_authorized = constant:independence_wave_formable_tuning.no_entries }
					check_variable = { var = independence_wave_formable_member_index value = independence_wave_formable_member_country_entries^num compare = less_than }
				}
				set_temp_variable = { independence_wave_formable_member_consent = independence_wave_formable_member_consent_entries^independence_wave_formable_member_index }
				var:independence_wave_formable_member_country_entries^independence_wave_formable_member_index = {
					if = {
						limit = {
							tag = event_target:independence_wave_formable_member_to_end
							check_variable = { independence_wave_formable_member_consent = constant:independence_wave_formable_consent.consenting_member }
						}
						set_temp_variable = { independence_wave_formable_member_authorized = constant:independence_wave_formable_tuning.loop_increment }
					}
				}
				add_to_temp_variable = { independence_wave_formable_member_index = constant:independence_wave_formable_tuning.loop_increment }
			}
		}
	}
	if = {
		limit = {
			check_variable = { independence_wave_formable_member_authorized = constant:independence_wave_formable_tuning.loop_increment }
			has_frozen_accepted_independence_wave_formable_invitation_from_root = yes
			is_independence_wave_active_country = yes
			NOT = { tag = event_target:independence_wave_formable_proposer }
		}
		set_temp_variable = { independence_wave_cleanup_reason = constant:independence_wave_origin_end_reason.formable_absorption }
		independence_wave_end_active_origin = yes
	}
}
