fallout_event_810_pay_branch_cost = {
	if = {
		limit = { NOT = { has_country_flag = fallout_event_810_cost_paid } }
		if = {
			limit = { check_variable = { fallout_event_810_selected_branch value = constant:fallout_event_810_branch.negotiate_safe_roster compare = equals } }
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_810_cost.negotiate_safe_roster_food }
			subtract_from_variable = { fallout_survival_current_entries^3 = constant:fallout_event_810_cost.negotiate_safe_roster_medicine }
			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_810_cost.negotiate_safe_roster_recognition }
		}
		else_if = {
			limit = { check_variable = { fallout_event_810_selected_branch value = constant:fallout_event_810_branch.compel_duty_roster compare = equals } }
			subtract_from_variable = { fallout_survival_current_entries^5 = constant:fallout_event_810_cost.compel_duty_roster_fuel }
			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_810_cost.compel_duty_roster_recognition }
			set_temp_variable = { fallout_survival_cohesion_delta = constant:fallout_event_810_cost.compel_duty_roster_cohesion }
			multiply_temp_variable = { fallout_survival_cohesion_delta = -1 }
			fallout_survival_apply_cohesion_delta = yes
		}
		else_if = {
			limit = { check_variable = { fallout_event_810_selected_branch value = constant:fallout_event_810_branch.cooldown_shutdown compare = equals } }
			subtract_from_variable = { fallout_survival_current_entries^4 = constant:fallout_event_810_cost.cooldown_shutdown_scrap }
			subtract_from_variable = { fallout_survival_current_entries^6 = constant:fallout_event_810_cost.cooldown_shutdown_power }
			subtract_from_variable = { fallout_survival_current_entries^3 = constant:fallout_event_810_cost.cooldown_shutdown_medicine }
		}
		else = {
			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_810_cost.recruit_foreign_team_food }
			subtract_from_variable = { fallout_survival_current_entries^5 = constant:fallout_event_810_cost.recruit_foreign_team_fuel }
			subtract_from_variable = { fallout_survival_current_entries^9 = constant:fallout_event_810_cost.recruit_foreign_team_recognition }
		}
		set_country_flag = fallout_event_810_cost_paid
		fallout_orientation_clamp_resources = yes
		set_temp_variable = { fallout_survival_cohesion_delta = constant:fallout_event_runtime.zero }
	}
}
