# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND CONTROLLER MAINTENANCE
# ============================================================================
# COUNTRY scope, with ROOT equal to the current controller receiving .900.
# Advances nonterminal registered mines even when their ordinary output is zero.
# The existing registry admits at most constant:riches_found_event.registry_cap
# states. This helper never scans unrelated countries or unregistered states.
# Contract, mission, presentation, evolution, and achievement helpers retain
# their own lifecycle rules. Only operating mines receive pulse pressure growth.
# Documentation: docs/plans/029_riches_found_plans/subagent_handoffs/
# maintenance_2026_10_03.md, including inputs, side effects, and integration.
# ============================================================================

# Inputs: the persistent mine registry and existing state-owned records.
# Defaults: an absent registry processes no mines and still rebuilds this
# controller's aggregate. No duration, weight, target, or policy is inferred.
# Output: refreshed mine records and current controller aggregate. Clears the
# delivery guard first and requests the next .900 only for maintenance-active
# controllers. Parent-owned riches_found_run_controller_pulse delegates here.
riches_found_run_maintenance_pulse = {
	clr_country_flag = riches_found_controller_pulse_scheduled
	clear_variable = riches_found_controller_pulse_due_date
	if = {
		limit = { has_variable = global.riches_found_registered_mines^num }
		for_each_scope_loop = {
			array = global.riches_found_registered_mines
			if = {
				limit = {
					riches_found_mine_is_registered = yes
					is_controlled_by = ROOT
					NOT = { riches_found_mine_is_terminal = yes }
				}
				riches_found_refresh_contract_lifecycle = yes
				riches_found_review_foreign_offer = yes
				riches_found_service_account_obligation = yes
				riches_found_review_preentry_exposure = yes
				riches_found_service_closed_mine = yes
				riches_found_apply_revenue_route_drift = yes
				riches_found_review_evolution_containment = yes
				if = {
					limit = { riches_found_mine_is_active = yes }
					riches_found_run_due_incident_review = yes
				}
				set_temp_variable = { riches_found_next_pressure_growth_date = riches_found_last_pressure_growth_date }
				set_temp_variable = { chaosx_date_days = constant:riches_found_duration.controller_pulse_days }
				set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
				chaosx_convert_days_to_date_offset = yes
				add_to_temp_variable = { riches_found_next_pressure_growth_date = chaosx_date_offset }
				if = {
					limit = {
						riches_found_mine_is_operating = yes
						check_variable = { var = global.date value = riches_found_next_pressure_growth_date compare = greater_than_or_equals }
					}
					set_variable = { riches_found_last_pressure_growth_date = global.date }
					set_temp_variable = { riches_found_pulse_pressure_delta = constant:riches_found_scaling.pulse_pressure_add }
					if = { limit = { ROOT = { has_war = yes } } add_to_temp_variable = { riches_found_pulse_pressure_delta = constant:riches_found_scaling.pulse_war_pressure_add } }
					if = { limit = { riches_found_mine_is_severe_pressure = yes } add_to_temp_variable = { riches_found_pulse_pressure_delta = constant:riches_found_scaling.pulse_severe_pressure_add } }
					add_to_variable = { riches_found_extraction_pressure = riches_found_pulse_pressure_delta }
					set_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_collapse_pressure_add }
					if = { limit = { riches_found_mine_is_local_order_crisis = yes } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_low_order_collapse_add } }
					if = { limit = { check_variable = { var = riches_found_sealed_depth_integrity value = constant:riches_found_threshold.sealed_integrity_gate compare = less_than } } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_low_integrity_collapse_add } }
					if = { limit = { check_variable = { var = riches_found_chaos_exposure value = constant:riches_found_value.zero compare = greater_than } } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_chaos_collapse_add } }
					if = { limit = { check_variable = { riches_found_mine_profile = constant:riches_found_profile.deep_vein } NOT = { has_state_flag = riches_found_shafts_reinforced } } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_deep_vein_collapse_add } }
					add_to_variable = { riches_found_collapse_pressure = riches_found_pulse_collapse_delta }
					if = { limit = { riches_found_mine_is_high_pressure = yes } set_state_flag = riches_found_collapse_warning }
				}
				riches_found_clamp_mine_values = yes
				riches_found_review_mine_crisis_flags = yes
				riches_found_refresh_active_mission_progress = yes
				if = {
					limit = {
						riches_found_mine_is_operating = yes
						riches_found_mine_is_collapse_risk = yes
						riches_found_mine_is_severe_pressure = yes
						check_variable = { var = riches_found_sealed_depth_integrity value = constant:riches_found_threshold.sealed_integrity_gate compare = less_than }
					}
					set_temp_variable = { riches_found_collapse_report_text = constant:riches_found_report_text.collapse_neglect }
					riches_found_collapse_mine = yes
				}
				riches_found_refresh_mine_presentation = yes
				riches_found_refresh_mine_contribution = yes
				riches_found_refresh_evolution_gates = yes
				riches_found_refresh_state_achievement_timers = yes
				if = {
					limit = { NOT = { riches_found_evolution_i_gate = yes } }
					clr_state_flag = riches_found_evolution_i_pending
				}
				if = {
					limit = { NOT = { riches_found_evolution_ii_gate = yes } }
					clr_state_flag = riches_found_evolution_ii_pending
				}
				if = {
					limit = { NOT = { riches_found_evolution_iii_gate = yes } }
					clr_state_flag = riches_found_evolution_iii_pending
				}
				if = {
					limit = { NOT = { riches_found_gilded_sovereignty_gate = yes } }
					clr_state_flag = riches_found_gilded_sovereignty_pending
				}
				if = {
					limit = { NOT = { riches_found_bottomless_account_gate = yes } }
					clr_state_flag = riches_found_bottomless_account_pending
				}
				if = {
					limit = { riches_found_evolution_i_gate = yes NOT = { has_state_flag = riches_found_evolution_i_pending } }
					set_state_flag = riches_found_evolution_i_pending
					set_temp_variable = { riches_found_evolution_delay_days = mtth:riches_found_evolution_i_interval }
					clamp_temp_variable = { var = riches_found_evolution_delay_days min = constant:riches_found_duration.evolution_review_floor_days max = constant:riches_found_duration.evolution_review_ceiling_days }
					state_event = { id = chaosx.nr29.901 days = riches_found_evolution_delay_days trigger_for = controller }
				}
				else_if = {
					limit = { riches_found_evolution_ii_gate = yes NOT = { has_state_flag = riches_found_evolution_ii_pending } }
					set_state_flag = riches_found_evolution_ii_pending
					set_temp_variable = { riches_found_evolution_delay_days = mtth:riches_found_evolution_ii_interval }
					clamp_temp_variable = { var = riches_found_evolution_delay_days min = constant:riches_found_duration.evolution_review_floor_days max = constant:riches_found_duration.evolution_review_ceiling_days }
					state_event = { id = chaosx.nr29.902 days = riches_found_evolution_delay_days trigger_for = controller }
				}
				else_if = {
					limit = { riches_found_evolution_iii_gate = yes NOT = { has_state_flag = riches_found_evolution_iii_pending } }
					set_state_flag = riches_found_evolution_iii_pending
					set_temp_variable = { riches_found_evolution_delay_days = mtth:riches_found_evolution_iii_interval }
					clamp_temp_variable = { var = riches_found_evolution_delay_days min = constant:riches_found_duration.evolution_review_floor_days max = constant:riches_found_duration.evolution_review_ceiling_days }
					state_event = { id = chaosx.nr29.903 days = riches_found_evolution_delay_days trigger_for = controller }
				}
				if = {
					limit = { riches_found_gilded_sovereignty_gate = yes NOT = { has_state_flag = riches_found_gilded_sovereignty_pending } }
					set_state_flag = riches_found_gilded_sovereignty_pending
					set_temp_variable = { riches_found_late_outcome_delay_days = constant:riches_found_duration.late_outcome_review_days }
					clamp_temp_variable = { var = riches_found_late_outcome_delay_days min = constant:riches_found_duration.late_outcome_review_floor_days max = constant:riches_found_duration.late_outcome_review_ceiling_days }
					state_event = { id = chaosx.nr29.904 days = riches_found_late_outcome_delay_days trigger_for = controller }
				}
				if = {
					limit = { riches_found_bottomless_account_gate = yes NOT = { has_state_flag = riches_found_bottomless_account_pending } }
					set_state_flag = riches_found_bottomless_account_pending
					set_temp_variable = { riches_found_late_outcome_delay_days = constant:riches_found_duration.late_outcome_review_days }
					clamp_temp_variable = { var = riches_found_late_outcome_delay_days min = constant:riches_found_duration.late_outcome_review_floor_days max = constant:riches_found_duration.late_outcome_review_ceiling_days }
					state_event = { id = chaosx.nr29.905 days = riches_found_late_outcome_delay_days trigger_for = controller }
				}
			}
		}
	}
	riches_found_rebuild_controller_aggregate = yes
	if = { limit = { has_country_flag = riches_found_controller_maintenance_active } riches_found_schedule_controller_pulse = yes }
}

# STATE scope, called once per mine by the controller pulse above. The crisis
# flags follow the mine's current values: each is raised while its value is in
# crisis and lifted on the pulse after the value recovers (Local Order out of
# its crisis band for crime, so a gang outbreak above it lasts one pulse;
# Revenue Legitimacy stable and illicit capture below its gate for a
# compromised administration). A severe mutiny is a wage
# dispute that has broken local order. A raid crisis ends once
# riches_found_duration.raid_memory_days pass without a new raid; Army Cordon,
# Clear the Claimant Barricades and a Mine Truce end it at once.
riches_found_review_mine_crisis_flags = {
	if = {
		limit = {
			OR = {
				riches_found_mine_is_severe_pressure = yes
				riches_found_mine_is_local_order_crisis = yes
				riches_found_mine_is_legitimacy_crisis = yes
			}
		}
		set_state_flag = riches_found_crisis_active
	}
	else = { clr_state_flag = riches_found_crisis_active }
	if = {
		limit = { riches_found_mine_is_local_order_crisis = yes }
		set_state_flag = riches_found_crime_crisis
	}
	else = { clr_state_flag = riches_found_crime_crisis }
	if = {
		limit = { riches_found_mine_is_legitimacy_crisis = yes }
		set_state_flag = riches_found_administration_compromised
	}
	else_if = {
		limit = {
			check_variable = { var = riches_found_revenue_legitimacy value = constant:riches_found_threshold.legitimacy_stable compare = greater_than_or_equals }
			check_variable = { var = riches_found_illicit_capture value = constant:riches_found_threshold.illicit_capture_gate compare = less_than }
		}
		clr_state_flag = riches_found_administration_compromised
	}
	if = {
		limit = {
			has_state_flag = riches_found_wage_dispute
			riches_found_mine_is_local_order_crisis = yes
		}
		set_state_flag = riches_found_severe_mutiny
	}
	else = { clr_state_flag = riches_found_severe_mutiny }
	if = {
		limit = { has_state_flag = riches_found_raid_crisis }
		if = {
			limit = { NOT = { has_variable = riches_found_last_raid_date } }
			set_variable = { riches_found_last_raid_date = global.date }
		}
		set_temp_variable = { riches_found_raid_quiet_date = riches_found_last_raid_date }
		set_temp_variable = { chaosx_date_days = constant:riches_found_duration.raid_memory_days }
		set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
		chaosx_convert_days_to_date_offset = yes
		add_to_temp_variable = { riches_found_raid_quiet_date = chaosx_date_offset }
		if = {
			limit = { check_variable = { var = global.date value = riches_found_raid_quiet_date compare = greater_than_or_equals } }
			clr_state_flag = riches_found_raid_crisis
		}
	}
}

# STATE scope, called once per mine by the controller pulse above. A temporary
# closure cools extraction and collapse pressure each pulse interval, while
# workers left without a standing revenue share or a payment in the current
# revenue cycle lose faith in the mine. When the closure review date passes
# the mine may be reopened through the paid reopening action; a closure left
# in place after that review stops relieving collapse pressure and slowly
# raises it unless the shafts were reinforced. A Demons closure end state is
# kept under watch: it neither strains the shafts nor waits on unpaid crews.
# A collapsed mine left without repair works after its recovery date loses
# development each interval until a Restore the Shaft mission is running.
riches_found_service_closed_mine = {
	if = {
		limit = {
			OR = {
				has_state_flag = riches_found_mine_temporary_closure
				has_state_flag = riches_found_mine_collapsed
			}
		}
		if = {
			limit = { NOT = { has_variable = riches_found_last_closure_service_date } }
			set_variable = { riches_found_last_closure_service_date = global.date }
		}
		set_temp_variable = { riches_found_next_closure_service_date = riches_found_last_closure_service_date }
		set_temp_variable = { chaosx_date_days = constant:riches_found_duration.controller_pulse_days }
		set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
		chaosx_convert_days_to_date_offset = yes
		add_to_temp_variable = { riches_found_next_closure_service_date = chaosx_date_offset }
		if = {
			limit = { check_variable = { var = global.date value = riches_found_next_closure_service_date compare = greater_than_or_equals } }
			set_variable = { riches_found_last_closure_service_date = global.date }
			if = {
				limit = {
					has_state_flag = riches_found_mine_temporary_closure
					NOT = { has_state_flag = riches_found_mine_collapsed }
				}
				add_to_variable = { riches_found_extraction_pressure = constant:riches_found_closure.pressure_decay_per_pulse }
				# Workers keep faith only through a standing revenue share or a
				# payment made within the current revenue cycle; a one-time
				# dividend paid years earlier does not carry a closure. A district
				# contained as a guarded or monitored closure has no waiting crews.
				# A stabilization reserve pays the idle crews through one closure
				# and is then spent.
				if = {
					limit = {
						NOT = { has_state_flag = riches_found_demons_contained_closure }
						NOT = {
							OR = {
								has_state_flag = riches_found_worker_cooperatives
								has_state_flag = riches_found_public_revenue_share_paid
								has_state_flag = riches_found_closure_crews_paid
								AND = {
									has_variable = riches_found_revenue_settlement_expiry_date
									check_variable = { var = riches_found_revenue_settlement_expiry_date value = global.date compare = greater_than }
								}
							}
						}
					}
					if = {
						limit = { riches_found_stabilization_reserve_can_pay = yes }
						riches_found_spend_stabilization_reserve = yes
						set_state_flag = riches_found_closure_crews_paid
					}
					else = {
						add_to_variable = { riches_found_revenue_legitimacy = constant:riches_found_closure.unsupported_legitimacy_per_pulse }
					}
				}
				if = {
					limit = {
						has_variable = riches_found_closure_review_date
						check_variable = { var = global.date value = riches_found_closure_review_date compare = greater_than_or_equals }
					}
					set_state_flag = riches_found_closure_review_complete
				}
				# Before the review the closure relieves the shafts. A contained
				# closure stays under watch and keeps relieving them; any other
				# closure left past its review strains unreinforced shafts.
				if = {
					limit = {
						OR = {
							NOT = { has_state_flag = riches_found_closure_review_complete }
							has_state_flag = riches_found_demons_contained_closure
						}
					}
					add_to_variable = { riches_found_collapse_pressure = constant:riches_found_closure.collapse_decay_per_pulse }
				}
				else_if = {
					limit = { NOT = { has_state_flag = riches_found_shafts_reinforced } }
					add_to_variable = { riches_found_collapse_pressure = constant:riches_found_closure.unmaintained_collapse_per_pulse }
				}
			}
			if = {
				limit = {
					has_state_flag = riches_found_mine_collapsed
					has_variable = riches_found_collapse_recovery_date
					check_variable = { var = global.date value = riches_found_collapse_recovery_date compare = greater_than_or_equals }
					NOT = { has_state_flag = riches_found_mission_restore_shaft_subject }
				}
				add_to_variable = { riches_found_mine_development = constant:riches_found_closure.collapse_neglect_development_per_pulse }
			}
			riches_found_clamp_mine_values = yes
		}
	}
}

# STATE scope, called once per mine by the controller pulse above. The
# standing revenue route keeps working while the mine yields to its owner,
# once each pulse interval: central receipts without an audit or an
# independent revenue authority feed illicit capture, a citizens' dividend and
# a local revenue share steady Local Order and wear down capture that is not
# yet entrenched (below riches_found_threshold.illicit_capture_gate), and
# tolerated patronage keeps feeding capture. Amounts are in
# riches_found_revenue_route; the contribution side of the same route is in
# riches_found_refresh_mine_contribution.
riches_found_apply_revenue_route_drift = {
	if = {
		limit = {
			riches_found_mine_provides_controller_output = yes
			NOT = { riches_found_mine_is_occupied = yes }
		}
		if = {
			limit = { NOT = { has_variable = riches_found_last_revenue_drift_date } }
			set_variable = { riches_found_last_revenue_drift_date = global.date }
		}
		set_temp_variable = { riches_found_next_revenue_drift_date = riches_found_last_revenue_drift_date }
		set_temp_variable = { chaosx_date_days = constant:riches_found_duration.controller_pulse_days }
		set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
		chaosx_convert_days_to_date_offset = yes
		add_to_temp_variable = { riches_found_next_revenue_drift_date = chaosx_date_offset }
		if = {
			limit = { check_variable = { var = global.date value = riches_found_next_revenue_drift_date compare = greater_than_or_equals } }
			set_variable = { riches_found_last_revenue_drift_date = global.date }
			if = {
				limit = {
					has_state_flag = riches_found_central_receipts
					NOT = { riches_found_revenue_has_oversight = yes }
				}
				add_to_variable = { riches_found_illicit_capture = constant:riches_found_revenue_route.central_capture_drift }
			}
			if = {
				limit = { has_state_flag = riches_found_dividend_paid }
				add_to_variable = { riches_found_local_order = constant:riches_found_revenue_route.dividend_order_drift }
				if = {
					limit = { check_variable = { var = riches_found_illicit_capture value = constant:riches_found_threshold.illicit_capture_gate compare = less_than } }
					add_to_variable = { riches_found_illicit_capture = constant:riches_found_revenue_route.dividend_capture_drift }
				}
			}
			if = {
				limit = { has_state_flag = riches_found_local_revenue_settled }
				add_to_variable = { riches_found_local_order = constant:riches_found_revenue_route.local_order_drift }
				if = {
					limit = { check_variable = { var = riches_found_illicit_capture value = constant:riches_found_threshold.illicit_capture_gate compare = less_than } }
					add_to_variable = { riches_found_illicit_capture = constant:riches_found_revenue_route.local_capture_drift }
				}
			}
			if = {
				limit = { has_state_flag = riches_found_patronage_tolerated }
				add_to_variable = { riches_found_illicit_capture = constant:riches_found_revenue_route.patronage_capture_drift }
			}
			riches_found_clamp_mine_values = yes
		}
	}
}

# STATE scope. Spends the mine's stabilization reserve. It is spent once, on
# the first bill it meets (a repair when the crews arrive, or idle crews in a
# closure), and only a new fund builds another.
riches_found_spend_stabilization_reserve = {
	clr_state_flag = riches_found_stabilization_reserve
	set_state_flag = riches_found_stabilization_reserve_spent
}
