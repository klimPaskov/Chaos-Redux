independence_wave_form09_resolve_ai_invitation_from_root = {
	if = {
		limit = {
			is_ai = yes
			has_pending_independence_wave_formable_founding_invitation = yes
			check_variable = { independence_wave_formable_invitation_family = constant:independence_wave_formable_family.balkan_federation }
		}
		if = {
			limit = { should_independence_wave_form09_ai_consent_to_root = yes }
			if = {
				limit = {
					has_independence_wave_form0124_full_compact_control = yes
					OR = {
						has_independence_wave_severe_host_threat = yes
						has_independence_wave_emergency_military_route = yes
					}
				}
				set_country_flag = independence_wave_form0124_full_integration_authorized
				clr_country_flag = independence_wave_form0124_autonomous_membership_authorized
			}
			else = {
				set_country_flag = independence_wave_form0124_autonomous_membership_authorized
				clr_country_flag = independence_wave_form0124_full_integration_authorized
			}
			independence_wave_formable_declare_consent_for_selected_family = yes
		}
		else = { independence_wave_formable_withhold_consent_for_selected_family = yes }
	}
}

independence_wave_formable_integration_adapter_9 = {
	if = {
		limit = {
			has_country_flag = independence_wave_formable_mutation_prevalidated
			has_country_flag = independence_wave_formable_identity_committed
			has_independence_wave_form09_runtime_commit_proof = yes
			independence_wave_formable_member_arrays_are_aligned = yes
		}
		set_temp_variable = { independence_wave_form09_member_index = constant:independence_wave_formable_tuning.first_array_index }
		set_temp_variable = { independence_wave_form09_additional_members_processed = constant:independence_wave_formable_tuning.no_entries }
		set_temp_variable = { independence_wave_form09_required_additional_members = independence_wave_formable_minimum_members }
		subtract_from_temp_variable = { independence_wave_form09_required_additional_members = constant:independence_wave_formable_tuning.loop_increment }
		while_loop_effect = {
			limit = {
				check_variable = { var = independence_wave_form09_member_index value = independence_wave_formable_member_country_entries^num compare = less_than }
			}
			set_temp_variable = { independence_wave_form09_member_consent = independence_wave_formable_member_consent_entries^independence_wave_form09_member_index }
			var:independence_wave_formable_member_country_entries^independence_wave_form09_member_index = {
				if = {
					limit = {
						NOT = { tag = ROOT }
						has_independence_wave_form09_member_candidate = yes
						check_variable = { independence_wave_form09_member_consent = constant:independence_wave_formable_consent.consenting_member }
						has_frozen_accepted_independence_wave_formable_invitation_from_root = yes
					}
					if = {
						limit = { has_country_flag = independence_wave_form0124_full_integration_authorized }
						independence_wave_formable_end_consenting_member_origin = yes
						transfer_units_fraction = { target = ROOT size = 1.0 stockpile_ratio = 1.0 army_ratio = 1.0 navy_ratio = 1.0 air_ratio = 1.0 }
						if = { limit = { original_tag = BBX owns_state = constant:independence_wave_form09_anchor.bbx controls_state = constant:independence_wave_form09_anchor.bbx } ROOT = { transfer_state = constant:independence_wave_form09_anchor.bbx add_state_core = constant:independence_wave_form09_anchor.bbx } }
						if = { limit = { original_tag = BAX owns_state = constant:independence_wave_form09_anchor.bax controls_state = constant:independence_wave_form09_anchor.bax } ROOT = { transfer_state = constant:independence_wave_form09_anchor.bax add_state_core = constant:independence_wave_form09_anchor.bax } }
						if = { limit = { original_tag = BOS owns_state = constant:independence_wave_form09_anchor.bos controls_state = constant:independence_wave_form09_anchor.bos } ROOT = { transfer_state = constant:independence_wave_form09_anchor.bos add_state_core = constant:independence_wave_form09_anchor.bos } }
						if = { limit = { original_tag = MAC owns_state = constant:independence_wave_form09_anchor.mac controls_state = constant:independence_wave_form09_anchor.mac } ROOT = { transfer_state = constant:independence_wave_form09_anchor.mac add_state_core = constant:independence_wave_form09_anchor.mac } }
						if = { limit = { original_tag = MNT owns_state = constant:independence_wave_form09_anchor.mnt controls_state = constant:independence_wave_form09_anchor.mnt } ROOT = { transfer_state = constant:independence_wave_form09_anchor.mnt add_state_core = constant:independence_wave_form09_anchor.mnt } }
					}
					else = {
						independence_wave_form0124_install_autonomous_member = yes
						add_ideas = independence_wave_form09_autonomous_balkan_member
						set_country_flag = independence_wave_form09_autonomous_member
					}
					add_to_temp_variable = { independence_wave_form09_additional_members_processed = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
			add_to_temp_variable = { independence_wave_form09_member_index = constant:independence_wave_formable_tuning.loop_increment }
		}
		if = {
			limit = {
				check_variable = {
					var = independence_wave_form09_additional_members_processed
					value = independence_wave_form09_required_additional_members
					compare = greater_than_or_equals
				}
			}
			if = { limit = { owns_state = constant:independence_wave_form09_anchor.bbx controls_state = constant:independence_wave_form09_anchor.bbx } set_capital = { state = constant:independence_wave_form09_anchor.bbx remember_old_capital = no } }
			else_if = { limit = { owns_state = constant:independence_wave_form09_anchor.bax controls_state = constant:independence_wave_form09_anchor.bax } set_capital = { state = constant:independence_wave_form09_anchor.bax remember_old_capital = no } }
			add_ideas = independence_wave_form09_federal_general_staff
			set_country_flag = independence_wave_form09_border_arbitration_open
			set_country_flag = independence_wave_form09_army_integration_complete
			set_country_flag = independence_wave_formable_integration_committed
		}
	}
}

independence_wave_form09_rollback_identity = {
	if = {
		limit = { NOT = { has_country_flag = independence_wave_formable_integration_committed } }
		clr_country_flag = independence_wave_form09_carrier
		if = { limit = { has_cosmetic_tag = BLX } drop_cosmetic_tag = yes }
		clr_global_flag = independence_wave_form09_blx_identity_in_use
	}
}
