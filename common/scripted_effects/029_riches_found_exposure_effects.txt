# ============================================================================
# CHAOS REDUX - EVENT 029 PRE-ENTRY EXPOSURE
# ============================================================================
# STATE scope. Ordinary extraction can build fictional compulsive pressure and
# disturb the mine's own opened depths before an evolution is recorded. This
# never creates biological exposure, an underground actor or a world threat.
# A state-owned review date prevents controller cycling from accelerating it.
# Managed access can hold Gold pressure down; closed mines do not expose crews.
# ============================================================================

riches_found_review_preentry_exposure = {
	set_temp_variable = { riches_found_next_exposure_review_date = riches_found_last_exposure_review_date }
	set_temp_variable = { chaosx_date_days = constant:riches_found_duration.controller_pulse_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	add_to_temp_variable = { riches_found_next_exposure_review_date = chaosx_date_offset }
	if = {
		limit = {
			riches_found_mine_is_operating = yes
			check_variable = { var = global.date value = riches_found_next_exposure_review_date compare = greater_than_or_equals }
		}
		set_variable = { riches_found_last_exposure_review_date = global.date }
		if = {
			limit = {
				riches_found_evolution_ii_is_enabled = yes
				OR = {
					NOT = { has_state_flag = riches_found_evolution_ii_recorded }
					has_state_flag = riches_found_gold_disease_contained
				}
				check_variable = { var = global.chaos_meter_value value = constant:riches_found_evolution.gold_disease_chaos compare = greater_than_or_equals }
			}
			set_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_exposure.gold_ordinary_extraction }
			# A contained Gold Disease leaves a lasting susceptibility.
			if = { limit = { has_state_flag = riches_found_gold_susceptible } add_to_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_containment.gold_susceptibility_exposure } }
			set_temp_variable = { riches_found_recent_theft_cutoff = global.date }
			set_temp_variable = { chaosx_date_days = constant:riches_found_duration.controller_pulse_days }
			set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
			chaosx_convert_days_to_date_offset = yes
			subtract_from_temp_variable = { riches_found_recent_theft_cutoff = chaosx_date_offset }
			if = { limit = { riches_found_mine_is_high_pressure = yes } add_to_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_exposure.gold_high_pressure } }
			if = { limit = { check_variable = { var = riches_found_mine_profile value = constant:riches_found_profile.gemstone compare = equals } } add_to_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_exposure.gold_gemstone } }
			if = { limit = { has_state_flag = riches_found_private_guards } add_to_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_exposure.gold_private_guards } }
			if = {
				limit = {
					has_variable = riches_found_last_incident_date
					check_variable = { var = riches_found_last_incident_date value = riches_found_recent_theft_cutoff compare = greater_than_or_equals }
					OR = {
						check_variable = { var = riches_found_last_incident_kind value = constant:riches_found_incident_kind.rush_theft compare = equals }
						check_variable = { var = riches_found_last_incident_kind value = constant:riches_found_incident_kind.rush_smuggling compare = equals }
					}
				}
				add_to_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_exposure.gold_theft_or_smuggling }
			}
			if = {
				limit = {
					has_state_flag = riches_found_public_revenue_share_paid
					has_state_flag = riches_found_camp_administration
					check_variable = { var = riches_found_local_order value = constant:riches_found_threshold.order_high compare = greater_than_or_equals }
					check_variable = { var = riches_found_revenue_legitimacy value = constant:riches_found_threshold.legitimacy_high compare = greater_than_or_equals }
					NOT = { riches_found_mine_is_severe_pressure = yes }
				}
				set_temp_variable = { riches_found_gold_exposure_delta = constant:riches_found_exposure.gold_managed_containment }
			}
			add_to_variable = { riches_found_gold_disease_pressure = riches_found_gold_exposure_delta }
		}
		if = {
			limit = {
				riches_found_evolution_iii_is_enabled = yes
				riches_found_no_incompatible_world_end = yes
				NOT = { has_state_flag = riches_found_evolution_iii_recorded }
				check_variable = { var = global.chaos_meter_value value = constant:riches_found_evolution.demons_beneath_mine_chaos compare = greater_than_or_equals }
				OR = {
					check_variable = { var = riches_found_deep_excavation value = constant:riches_found_exposure.deep_disturbance_gate compare = greater_than_or_equals }
					has_state_flag = riches_found_prefire_evolution_iii
				}
			}
			add_to_variable = { riches_found_supernatural_pressure = constant:riches_found_exposure.supernatural_extraction }
			if = {
				limit = { check_variable = { var = riches_found_supernatural_pressure value = constant:riches_found_exposure.supernatural_signs_gate compare = greater_than_or_equals } }
				set_state_flag = riches_found_supernatural_signs
			}
		}
		riches_found_clamp_mine_values = yes
	}
}
