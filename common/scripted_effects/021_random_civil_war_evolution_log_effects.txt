# Event 021 evolution log actor selection.
# ANY scope. The scheduler caller is not implicitly a civil-war participant.
# Caller preinitializes events_log_evolution_has_actor before this helper.

event021_select_verified_evolution_log_actor = {
	set_temp_variable = { events_log_evolution_has_actor = constant:random_civil_war_value.zero }
	if = {
		limit = {
			check_variable = { var = global.random_civil_war_front_ids^num value = global.random_civil_war_front_actors^num compare = equals }
			check_variable = { var = global.random_civil_war_front_ids^num value = global.random_civil_war_front_hosts^num compare = equals }
			check_variable = { var = global.random_civil_war_front_ids^num value = global.random_civil_war_front_crisis_ids^num compare = equals }
			check_variable = { var = global.random_civil_war_front_ids^num value = global.random_civil_war_front_statuses^num compare = equals }
		}
		for_each_loop = {
			array = global.random_civil_war_front_ids
			value = event021_evolution_log_front_id
			index = event021_evolution_log_row
			break = events_log_evolution_has_actor
			if = {
				limit = {
					check_variable = { var = event021_evolution_log_front_id value = constant:random_civil_war_value.zero compare = greater_than }
					check_variable = { var = global.random_civil_war_front_statuses^event021_evolution_log_row value = constant:random_civil_war_front_status.active compare = equals }
				}
				set_temp_variable = { event021_evolution_log_crisis_id = global.random_civil_war_front_crisis_ids^event021_evolution_log_row }
				set_temp_variable = { event021_evolution_log_host_valid = constant:random_civil_war_value.zero }
				if = {
					limit = { check_variable = { var = event021_evolution_log_crisis_id value = constant:random_civil_war_value.zero compare = greater_than } }
					var:global.random_civil_war_front_hosts^event021_evolution_log_row = {
						if = {
							limit = {
								random_civil_war_is_normal_human_country = yes
								has_country_flag = random_civil_war_active
								has_variable = random_civil_war_crisis_id
								check_variable = { var = random_civil_war_crisis_id value = event021_evolution_log_crisis_id compare = equals }
							}
							set_temp_variable = { event021_evolution_log_host_valid = constant:random_civil_war_value.one }
						}
					}
				}
				if = {
					limit = { check_variable = { var = event021_evolution_log_host_valid value = constant:random_civil_war_value.one compare = equals } }
					var:global.random_civil_war_front_actors^event021_evolution_log_row = {
						if = {
							limit = {
								random_civil_war_is_normal_human_country = yes
								has_country_flag = random_civil_war_active
								has_country_flag = random_civil_war_theater_registered
								is_in_array = { array = global.random_civil_war_active_theaters value = THIS }
								has_variable = random_civil_war_front_id
								has_variable = random_civil_war_crisis_id
								check_variable = { var = random_civil_war_front_id value = event021_evolution_log_front_id compare = equals }
								check_variable = { var = random_civil_war_crisis_id value = event021_evolution_log_crisis_id compare = equals }
							}
							# The indicator is caller-owned. A stale regular target alone
							# cannot select the logger's actor branch on a later call.
							save_event_target_as = events_log_evolution_actor
							set_temp_variable = { events_log_evolution_has_actor = constant:random_civil_war_value.one }
						}
					}
				}
			}
		}
	}
}
