# COUNTRY scope: turn exact selected Event 006 state transfers into or out of
# host claims. Event 005 rows, optional states from another host, and unrelated
# historical claims never enter this pass.
independence_wave_host_response_apply_state_claim_policy = {
	if = {
		limit = { has_current_independence_wave_host_response_receipt = yes }
		save_event_target_as = independence_wave_host_response_claim_host
		set_temp_variable = { independence_wave_host_response_claim_state_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_host_response_claim_state_index value = global.liberation_plan_states^num compare = less_than } }
			set_temp_variable = { independence_wave_host_response_claim_state_role = global.liberation_plan_state_roles^independence_wave_host_response_claim_state_index }
			if = {
				limit = {
					check_variable = { var = global.liberation_plan_state_country_owners^independence_wave_host_response_claim_state_index value = constant:liberation_plan_owner.independence_wave compare = equals }
					OR = {
						check_variable = { independence_wave_host_response_claim_state_role = constant:liberation_plan_state_role.anchor }
						check_variable = { independence_wave_host_response_claim_state_role = constant:liberation_plan_state_role.compact }
						check_variable = { independence_wave_host_response_claim_state_role = constant:liberation_plan_state_role.extended }
					}
					var:global.liberation_plan_state_hosts^independence_wave_host_response_claim_state_index = { tag = event_target:independence_wave_host_response_claim_host }
				}
				var:global.liberation_plan_states^independence_wave_host_response_claim_state_index = {
					if = {
						limit = {
							is_claimed_by = event_target:independence_wave_host_response_claim_host
							OR = {
								check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_negotiate compare = equals }
								check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_settlement compare = equals }
								check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_regional_identity compare = equals }
								AND = {
									check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_relief compare = equals }
									OR = {
										check_variable = { var = independence_wave_host_response_claim_state_role value = constant:liberation_plan_state_role.compact compare = equals }
										check_variable = { var = independence_wave_host_response_claim_state_role value = constant:liberation_plan_state_role.extended compare = equals }
									}
								}
								AND = {
									OR = {
										check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_pressure compare = equals }
										check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_command compare = equals }
										check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_continuity compare = equals }
									}
									OR = {
										check_variable = { var = independence_wave_host_response_claim_state_role value = constant:liberation_plan_state_role.compact compare = equals }
										check_variable = { var = independence_wave_host_response_claim_state_role value = constant:liberation_plan_state_role.extended compare = equals }
									}
								}
							}
						}
						remove_claim_by = event_target:independence_wave_host_response_claim_host
					}
					if = {
						limit = {
							NOT = { is_claimed_by = event_target:independence_wave_host_response_claim_host }
							OR = {
								check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_reclamation compare = equals }
								AND = {
									OR = {
										check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_pressure compare = equals }
										check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_command compare = equals }
										check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_continuity compare = equals }
									}
									check_variable = { var = independence_wave_host_response_claim_state_role value = constant:liberation_plan_state_role.anchor compare = equals }
								}
							}
						}
						add_claim_by = event_target:independence_wave_host_response_claim_host
					}
				}
			}
			add_to_temp_variable = { independence_wave_host_response_claim_state_index = constant:independence_wave_plan.loop_increment }
		}
	}
}

# COUNTRY scope: every player option and the hidden AI resolver enter here with
# one action token. The response receipt prevents duplicate application.
independence_wave_host_response_apply_current_action = {
	if = {
		limit = {
			has_current_independence_wave_host_response_receipt = yes
			OR = {
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_negotiate compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_pressure compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_reclamation compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_command compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_relief compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_settlement compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_continuity compare = equals }
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_regional_identity compare = equals }
			}
		}
		set_variable = { independence_wave_host_response_last_action = independence_wave_host_response_action_input }
		set_variable = { independence_wave_host_response_last_tier = independence_wave_host_response_tier }
		if = {
			limit = { NOT = { has_variable = independence_wave_host_response_waves_received } }
			set_variable = { independence_wave_host_response_waves_received = constant:independence_wave_host_response_value.zero }
		}
		if = {
			limit = { NOT = { has_variable = independence_wave_host_response_total_event6_states_lost } }
			set_variable = { independence_wave_host_response_total_event6_states_lost = constant:independence_wave_host_response_value.zero }
		}
		add_to_variable = { independence_wave_host_response_waves_received = constant:independence_wave_host_response_value.one }
		add_to_variable = { independence_wave_host_response_total_event6_states_lost = independence_wave_host_response_event6_loss_count }
		clr_country_flag = independence_wave_host_response_reclamation_prepared
		clr_country_flag = independence_wave_host_response_remnant_legal_continuity
		clr_country_flag = independence_wave_host_response_remnant_regional_identity
		if = {
			limit = {
				OR = {
					check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_command compare = equals }
					check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_relief compare = equals }
					check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.severe_settlement compare = equals }
				}
			}
			set_temp_variable = { independence_wave_host_response_idea_days = constant:independence_wave_host_response_duration.partition_crisis_days }
			add_timed_idea = { idea = independence_wave_host_partition_crisis days = independence_wave_host_response_idea_days }
		}
		if = {
			limit = {
				check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.limited_reclamation compare = equals }
			}
			set_country_flag = independence_wave_host_response_reclamation_prepared
			set_temp_variable = { independence_wave_host_response_idea_days = constant:independence_wave_host_response_duration.reclamation_preparation_days }
			add_timed_idea = { idea = independence_wave_host_reclamation_burden days = independence_wave_host_response_idea_days }
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_continuity compare = equals }
					check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_regional_identity compare = equals }
				}
			}
			set_temp_variable = { independence_wave_host_response_idea_days = constant:independence_wave_host_response_duration.remnant_government_days }
			add_timed_idea = { idea = independence_wave_host_remnant_emergency_government days = independence_wave_host_response_idea_days }
			set_country_flag = independence_wave_host_response_outside_guarantee_opportunity_open
		}
		if = {
			limit = { check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_continuity compare = equals } }
			set_country_flag = independence_wave_host_response_remnant_legal_continuity
		}
		if = {
			limit = { check_variable = { var = independence_wave_host_response_action_input value = constant:independence_wave_host_response_action.remnant_regional_identity compare = equals } }
			set_country_flag = independence_wave_host_response_remnant_regional_identity
		}
		independence_wave_host_response_apply_state_claim_policy = yes
		independence_wave_host_response_update_primary_child_ledgers = yes
		independence_wave_host_response_complete_current_host = yes
	}
	else = { independence_wave_host_response_skip_invalid_current_host = yes }
}
