# ============================================================================
# CHAOS REDUX - CBRN OCCUPATION SCRIPTED EFFECTS
# ============================================================================
# Overview:
# - Owns the exact-state backend for CBRN Coercive Security, Protected
#   Occupation Administration, and targeted Nerve Agent Suppression.
# - Calls the shared CBRN payload, protection, exposure-record, and
#   consequence dispatchers. This file never reimplements deaths,
#   contamination, medical saturation, evidence, attribution, Condemnation,
#   Air Cleanliness, or confirmed-use history.
# - Stores bounded state trauma, short-lived suppression records, responsible
#   country/action snapshots, repeat-use memory, cooldowns, coverup inputs,
#   discovery inputs, and targeted delayed backlash. Callers own event or
#   decision selection and must bind cbrn_occupation_target_state first.
# - No camp, genocide, broad pulse, target-selection fallback, or condition
#   estimator is defined here. Exact state law overrides and output records
#   are deliberate adapters for the owned decision surfaces.
#
# Internal contract:
# - Country-scope operation callers set cbrn_occupation_requested_agent to the
#   cbrn_agent.sarin or cbrn_agent.soman enum. No selected-state condition
#   receipt is synthesized: the route stays closed until a verified
#   current-version hook supplies the complete exact-state receipt.
# - State-scope delayed helpers are called only for the exact state whose
#   stored due-day flag is being processed. The backlash helper is called from
#   the responsible-country scope so ROOT is the recorded occupier.
# - Shared payload debit is the first mutation in the operation chain. The
#   operation cost debit occurs only after a positive shared payload proof.
# - Every public helper leaves a proof or persistent record for its caller;
#   transient input cleanup is explicit and never clears historical records.
# ============================================================================

cbrn_occupation_deliver_external_protective_aid_to_state = {
	set_temp_variable = { cbrn_occupation_protective_aid_proof = constant:cbrn_proof.missing }
	if = {
		limit = {
			has_event_target = cbrn_distribution_external_supplier
			event_target:cbrn_distribution_external_supplier = {
				exists = yes
				cbrn_occupation_country_can_supply_external_protective_aid = yes
			}
			cbrn_occupation_state_can_receive_external_protective_aid = yes
		}
		set_temp_variable = {
			cbrn_distribution_target_fraction = constant:cbrn_protection_program_effect.occupied_distribution_target
		}
		set_temp_variable = {
			cbrn_distribution_base_cost_mult = constant:cbrn_protection_program_effect.occupied_distribution_cost_mult
		}
		set_temp_variable = { cbrn_distribution_effectiveness_mult = constant:cbrn_math.one }
		set_temp_variable = { cbrn_distribution_new_filter_condition = constant:cbrn_mask_filters.condition_full }
		cbrn_distribute_requested_external_masks_to_state = yes
		if = {
			limit = {
				check_variable = {
					var = cbrn_external_distribution_proof
					value = constant:cbrn_proof.supplied
					compare = equals
				}
			}
			set_variable = {
				cbrn_occupation_last_protective_aid_supplier = event_target:cbrn_distribution_external_supplier
			}
			set_variable = { cbrn_occupation_last_protective_aid_date = global.num_days }
			set_variable = { cbrn_occupation_last_protective_aid_consumed = cbrn_distribution_consumed_crates }
			set_variable = { cbrn_occupation_last_protective_aid_delivered = cbrn_distribution_usable_crates }
			if = {
				limit = { NOT = { has_variable = cbrn_occupation_protective_aid_total } }
				set_variable = { cbrn_occupation_protective_aid_total = constant:cbrn_math.zero }
			}
			add_to_variable = { cbrn_occupation_protective_aid_total = cbrn_distribution_usable_crates }
			set_state_flag = cbrn_occupation_external_protective_aid_history
			event_target:cbrn_distribution_external_supplier = {
				cbrn_initialize_country_data = yes
				add_to_variable = { cbrn_protective_aid_exported_crates = cbrn_distribution_consumed_crates }
				cbrn_refresh_country_mask_snapshot = yes
				cbrn_start_protection_maintenance_job = yes
			}
			owner = {
				if = {
					limit = { exists = yes }
					cbrn_initialize_country_data = yes
					add_to_variable = { cbrn_protective_aid_received_crates = cbrn_distribution_usable_crates }
					set_country_flag = cbrn_received_protective_aid
					add_opinion_modifier = {
						target = event_target:cbrn_distribution_external_supplier
						modifier = cbrn_protective_aid_received
					}
				}
			}
			set_temp_variable = { cbrn_occupation_protective_aid_proof = constant:cbrn_proof.supplied }
		}
	}
}

cbrn_occupation_detect_repeat_use = {
	set_temp_variable = { cbrn_occupation_repeat_use = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_occupation_elapsed_days = constant:cbrn_math.zero }
	if = {
		limit = { has_event_target = cbrn_occupation_target_state }
		event_target:cbrn_occupation_target_state = {
			if = {
				limit = { has_variable = cbrn_occupation_last_action_date }
				set_temp_variable = { cbrn_occupation_elapsed_days = global.num_days }
				subtract_from_temp_variable = { cbrn_occupation_elapsed_days = cbrn_occupation_last_action_date }
				if = {
					limit = {
						check_variable = {
							var = cbrn_occupation_elapsed_days
							value = constant:cbrn_math.zero
							compare = greater_than
						}
						check_variable = {
							var = cbrn_occupation_elapsed_days
							value = constant:cbrn_occupation_timing.repeat_window_days
							compare = less_than_or_equals
						}
					}
					set_temp_variable = { cbrn_occupation_repeat_use = constant:cbrn_math.one }
				}
			}
		}
	}
}

						}

							check_variable = {
								var = cbrn_action_dispatch_proof

cbrn_occupation_end_responsible_country_operational_modifiers = {
	if = {
		limit = {
			has_variable = cbrn_occupation_responsible_country
			var:cbrn_occupation_responsible_country = { exists = yes }
		}
		if = {
			limit = {
				has_dynamic_modifier = {
					modifier = cbrn_nerve_suppression_active_state
					scope = var:cbrn_occupation_responsible_country
				}
			}
			remove_dynamic_modifier = {
				modifier = cbrn_nerve_suppression_active_state
				scope = var:cbrn_occupation_responsible_country
			}
		}
		if = {
			limit = {
				has_dynamic_modifier = {
					modifier = cbrn_occupation_backlash_state
					scope = var:cbrn_occupation_responsible_country
				}
			}
			remove_dynamic_modifier = {
				modifier = cbrn_occupation_backlash_state
				scope = var:cbrn_occupation_responsible_country
			}
		}
	}
	clr_state_flag = cbrn_occupation_nerve_suppression_active
	clr_state_flag = cbrn_occupation_backlash_active
	clr_state_flag = cbrn_occupation_state_seal_active
	if = {
		limit = { has_dynamic_modifier = { modifier = cbrn_occupation_state_sealed_modifier } }
		remove_dynamic_modifier = { modifier = cbrn_occupation_state_sealed_modifier }
	}
	force_update_dynamic_modifier = yes
}
