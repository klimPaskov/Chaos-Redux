# Overview:
# - Owns occupation-law state, external protective-aid delivery, exact
#   chemical-record disclosure, and historical nerve-suppression aftermath.
# - Stored action readers preserve deaths, contamination, evidence,
#   attribution, Condemnation, trauma, and discovery without a new attack.
# - State-scope delayed helpers process only the exact recorded state.
# - Aid uses model-specific reserved respirator crates and returns unused
#   stock to the original supplier when delivery cannot finish.
# - No nerve-agent attack preparation, payment, or dispatcher lives here.
#   Historical cover-up decisions pay Political Power through the shared
#   Condemnation-scaled quote and exact-record proof; attacks use native raids.
# ============================================================================

# State-scoped UI lifetime count for independently targeted aid projects.
# The receipt flag prevents a second close from decrementing the donor twice.
cbrn_occupation_register_external_aid_reservation = {
	if = {
		limit = {
			has_state_flag = cbrn_external_mask_escrow_active
			NOT = { has_state_flag = cbrn_external_aid_reservation_counted }
		}
		set_state_flag = cbrn_external_aid_reservation_counted
		ROOT = {
			if = { limit = { NOT = { has_variable = cbrn_external_aid_active_reservations_count } } set_variable = { cbrn_external_aid_active_reservations_count = constant:cbrn_math.zero } }
			add_to_variable = { cbrn_external_aid_active_reservations_count = constant:cbrn_math.one }
		}
	}
}

cbrn_occupation_close_external_aid_reservation = {
	clr_state_flag = cbrn_external_aid_sale_factory_commitment
	clear_variable = cbrn_external_aid_committed_pp_fee
	if = {
		limit = { has_state_flag = cbrn_external_aid_reservation_counted }
		if = {
			limit = { has_variable = cbrn_external_mask_escrow_donor var:cbrn_external_mask_escrow_donor = { exists = yes } }
			var:cbrn_external_mask_escrow_donor = {
				subtract_from_variable = { cbrn_external_aid_active_reservations_count = constant:cbrn_math.one }
				clamp_variable = { var = cbrn_external_aid_active_reservations_count min = constant:cbrn_math.zero }
			}
		}
		clr_state_flag = cbrn_external_aid_reservation_counted
	}
}

cbrn_occupation_deliver_external_protective_aid_to_state = {
	set_temp_variable = { cbrn_occupation_protective_aid_proof = constant:cbrn_proof.missing }
	if = {
		limit = {
			has_event_target = cbrn_distribution_external_supplier
			event_target:cbrn_distribution_external_supplier = {
				exists = yes
				cbrn_occupation_country_can_continue_external_protective_aid = yes
			}
			cbrn_occupation_state_has_donor_external_aid_reservation = yes
			cbrn_occupation_state_remains_external_protective_aid_target = yes
		}
		set_temp_variable = {
			cbrn_distribution_target_fraction = constant:cbrn_protection_program_effect.occupied_distribution_target
		}
		set_temp_variable = {
			cbrn_distribution_base_cost_mult = constant:cbrn_protection_program_effect.occupied_distribution_cost_mult
		}
		set_temp_variable = { cbrn_distribution_effectiveness_mult = constant:cbrn_math.one }
		set_temp_variable = { cbrn_distribution_new_filter_condition = constant:cbrn_mask_filters.condition_full }
		set_temp_variable = { cbrn_external_distribution_proof = constant:cbrn_proof.missing }
		set_temp_variable = { cbrn_distribution_consumed_crates = constant:cbrn_math.zero }
		set_temp_variable = { cbrn_distribution_usable_crates = constant:cbrn_math.zero }
		cbrn_distribute_reserved_external_masks_to_state = yes
		if = {
			limit = {
				check_variable = {
					var = cbrn_external_distribution_proof
					value = constant:cbrn_proof.supplied
					compare = equals
				}
			}
			set_variable = {
				cbrn_occupation_last_protective_aid_supplier = event_target:cbrn_distribution_external_supplier
			}
			set_variable = { cbrn_occupation_last_protective_aid_date = global.num_days }
			set_variable = { cbrn_occupation_last_protective_aid_consumed = cbrn_distribution_consumed_crates }
			set_variable = { cbrn_occupation_last_protective_aid_delivered = cbrn_distribution_usable_crates }
			if = {
				limit = { NOT = { has_variable = cbrn_occupation_protective_aid_total } }
				set_variable = { cbrn_occupation_protective_aid_total = constant:cbrn_math.zero }
			}
			add_to_variable = { cbrn_occupation_protective_aid_total = cbrn_distribution_usable_crates }
			set_state_flag = cbrn_occupation_external_protective_aid_history
			event_target:cbrn_distribution_external_supplier = {
				cbrn_initialize_country_data = yes
				add_to_variable = { cbrn_protective_aid_exported_crates = cbrn_distribution_consumed_crates }
				cbrn_refresh_country_mask_snapshot = yes
				cbrn_start_protection_maintenance_job = yes
			}
			owner = {
				if = {
					limit = { exists = yes }
					cbrn_initialize_country_data = yes
					add_to_variable = { cbrn_protective_aid_received_crates = cbrn_distribution_usable_crates }
					set_country_flag = cbrn_received_protective_aid
					add_opinion_modifier = {
						target = event_target:cbrn_distribution_external_supplier
						modifier = cbrn_protective_aid_received
					}
				}
			}
			set_temp_variable = { cbrn_occupation_protective_aid_proof = constant:cbrn_proof.supplied }
		}
	}
}

cbrn_occupation_execute_external_protective_aid = {
	set_temp_variable = { cbrn_occupation_protective_aid_proof = constant:cbrn_proof.missing }
	if = {
		limit = {
			has_event_target = cbrn_occupation_target_state
			cbrn_occupation_country_can_continue_external_protective_aid = yes
			event_target:cbrn_occupation_target_state = {
				cbrn_occupation_state_has_donor_external_aid_reservation = yes
				cbrn_occupation_state_remains_external_protective_aid_target = yes
			}
		}
		save_event_target_as = cbrn_distribution_external_supplier
		event_target:cbrn_occupation_target_state = {
			cbrn_occupation_deliver_external_protective_aid_to_state = yes
		}
		if = {
			limit = {
				check_variable = {
					var = cbrn_occupation_protective_aid_proof
					value = constant:cbrn_proof.supplied
					compare = equals
				}
			}
			if = {
				limit = { NOT = { has_variable = cbrn_occupation_external_aid_action_count } }
				set_variable = { cbrn_occupation_external_aid_action_count = constant:cbrn_math.zero }
			}
			add_to_variable = { cbrn_occupation_external_aid_action_count = constant:cbrn_math.one }
			set_variable = { cbrn_occupation_last_external_aid_date = global.num_days }
		}
	}
	else_if = {
		limit = { has_event_target = cbrn_occupation_target_state }
		event_target:cbrn_occupation_target_state = { cbrn_refund_external_state_mask_reservation = yes }
	}
}

cbrn_occupation_refresh_trauma_modifier = {
	set_variable = { cbrn_occupation_trauma_resistance_target = constant:cbrn_math.zero }
	set_variable = { cbrn_occupation_trauma_resistance_growth = constant:cbrn_math.zero }
	set_variable = { cbrn_occupation_trauma_compliance_gain = constant:cbrn_math.zero }
	if = {
		limit = {
			has_variable = cbrn_occupation_nerve_suppression_trauma
			check_variable = { cbrn_occupation_nerve_suppression_trauma > constant:cbrn_math.zero }
		}
		set_variable = { cbrn_occupation_trauma_resistance_target = cbrn_occupation_nerve_suppression_trauma }
		multiply_variable = {
			cbrn_occupation_trauma_resistance_target = constant:cbrn_occupation_state_modifier.trauma_resistance_target_per_point
		}
		set_variable = { cbrn_occupation_trauma_resistance_growth = cbrn_occupation_nerve_suppression_trauma }
		multiply_variable = {
			cbrn_occupation_trauma_resistance_growth = constant:cbrn_occupation_state_modifier.trauma_resistance_growth_per_point
		}
		set_variable = { cbrn_occupation_trauma_compliance_gain = cbrn_occupation_nerve_suppression_trauma }
		multiply_variable = {
			cbrn_occupation_trauma_compliance_gain = constant:cbrn_occupation_state_modifier.trauma_compliance_gain_per_point
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = cbrn_nerve_suppression_trauma_state } } }
			add_dynamic_modifier = { modifier = cbrn_nerve_suppression_trauma_state }
		}
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = cbrn_nerve_suppression_trauma_state } }
			remove_dynamic_modifier = { modifier = cbrn_nerve_suppression_trauma_state }
		}
	}
	force_update_dynamic_modifier = yes
}

cbrn_occupation_end_responsible_country_operational_modifiers = {
	if = {
		limit = {
			has_variable = cbrn_occupation_responsible_country
			var:cbrn_occupation_responsible_country = { exists = yes }
		}
		if = {
			limit = {
				has_dynamic_modifier = {
					modifier = cbrn_nerve_suppression_active_state
					scope = var:cbrn_occupation_responsible_country
				}
			}
			remove_dynamic_modifier = {
				modifier = cbrn_nerve_suppression_active_state
				scope = var:cbrn_occupation_responsible_country
			}
		}
		if = {
			limit = {
				has_dynamic_modifier = {
					modifier = cbrn_occupation_backlash_state
					scope = var:cbrn_occupation_responsible_country
				}
			}
			remove_dynamic_modifier = {
				modifier = cbrn_occupation_backlash_state
				scope = var:cbrn_occupation_responsible_country
			}
		}
	}
	clr_state_flag = cbrn_occupation_nerve_suppression_active
	clr_state_flag = cbrn_occupation_backlash_active
	clr_state_flag = cbrn_occupation_state_seal_active
	if = {
		limit = { has_variable = migration_persecution_cbrn_generation }
		set_variable = {
			migration_persecution_pending_owner = constant:migration_persecution_owner_class.cbrn_coercive_operation
		}
		set_variable = {
			migration_persecution_pending_clear_generation = migration_persecution_cbrn_generation
		}
	}
	migration_persecution_clear_owner_state = yes
	if = {
		limit = { has_dynamic_modifier = { modifier = cbrn_occupation_state_sealed_modifier } }
		remove_dynamic_modifier = { modifier = cbrn_occupation_state_sealed_modifier }
	}
	force_update_dynamic_modifier = yes
}

		clamp_temp_variable = {
			var = cbrn_occupation_backlash_resistance
			min = constant:cbrn_math.zero
			max = constant:cbrn_occupation_suppression.backlash_resistance_maximum
		}
		add_resistance = cbrn_occupation_backlash_resistance
		set_variable = { cbrn_occupation_last_backlash_resistance = cbrn_occupation_backlash_resistance }
		set_temp_variable = { cbrn_occupation_backlash_target_days = constant:cbrn_occupation_timing.backlash_target_days }

			flag = cbrn_occupation_backlash_active

		add_dynamic_modifier = {
			modifier = cbrn_occupation_backlash_state
			days = cbrn_occupation_backlash_target_days
			scope = var:cbrn_occupation_responsible_country
		}
		force_update_dynamic_modifier = yes
		set_temp_variable = { cbrn_occupation_compliance_debit = constant:cbrn_occupation_suppression.backlash_compliance_loss }
		multiply_temp_variable = { cbrn_occupation_compliance_debit = constant:cbrn_payload_math.negation_multiplier }
		add_compliance = cbrn_occupation_compliance_debit
		if = {
			limit = {
				OR = {
					check_variable = {

# Country scope. Retire inherited human CBRN authority without altering other laws.
cbrn_occupation_clear_nonhuman_country_policy = {
	if = {
		limit = { uses_normal_civilian_systems = no }
		THIS = {
			if = {
				limit = { OR = { occupation_law = cbrn_coercive_security_occupation occupation_law = cbrn_protected_occupation_administration } }
				set_occupation_law = default_law
			}
		}
		every_occupied_country = {
			if = {
				limit = { OR = { occupation_law = cbrn_coercive_security_occupation occupation_law = cbrn_protected_occupation_administration } }
				set_occupation_law = default_law
			}
		}
		clr_country_flag = cbrn_occupation_coercive_security_authorized
		clr_country_flag = cbrn_occupation_protected_administration_authorized
		clr_country_flag = cbrn_nerve_suppression_policy_authorized
		every_owned_state = { cbrn_occupation_clear_nonhuman_state_policy = yes }
		every_controlled_state = { cbrn_occupation_clear_nonhuman_state_policy = yes }
	}
}
