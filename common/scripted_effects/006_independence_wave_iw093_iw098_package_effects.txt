# Each package conference locks the matching shared government route and then
# installs the exact package politics. The result temporary lets a timed
# decision fail cleanly if another route locked while the conference ran.
independence_wave_lock_iw093_royal_confederacy_route = {
	set_temp_variable = { independence_wave_iw093_route_transaction_result = constant:independence_wave_iw093.zero }
	if = {
		limit = { can_lock_independence_wave_iw093_royal_confederacy_route = yes }
		if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.undecided } }
			set_temp_variable = { independence_wave_government_route_input = constant:independence_wave_government_route.traditional_restoration }
			independence_wave_select_government_route = yes
		}
		if = {
			limit = { check_variable = { independence_wave_government_route = constant:independence_wave_government_route.traditional_restoration } }
			set_country_flag = independence_wave_iw093_route_royal_confederacy
			set_country_flag = independence_wave_iw093_route_politics_receipt
			set_politics = { ruling_party = neutrality elections_allowed = no }
			set_popularities = {
				democratic = constant:independence_wave_iw093_route_politics.royal_democratic
				communism = constant:independence_wave_iw093_route_politics.royal_communism
				neutrality = constant:independence_wave_iw093_route_politics.royal_neutrality
				fascism = constant:independence_wave_iw093_route_politics.royal_fascism
			}
			subtract_from_variable = { iw093_court_cabinet_balance = constant:independence_wave_iw093.route_balance_shift }
			add_to_variable = { iw093_confederated_authority = constant:independence_wave_iw093.focus_standard_gain }
			set_temp_variable = { independence_wave_decision_legitimacy_delta = constant:independence_wave_decision_effect.value_standard }
			set_temp_variable = { independence_wave_decision_recognition_delta = constant:independence_wave_decision_effect.value_minor }
			set_temp_variable = { independence_wave_decision_capacity_delta = constant:independence_wave_value.minimum }
			set_temp_variable = { independence_wave_decision_security_delta = constant:independence_wave_decision_effect.value_minor }
			set_temp_variable = { independence_wave_decision_instability_delta = constant:independence_wave_decision_effect.value_minor_loss }
			independence_wave_decision_apply_country_deltas = yes
			independence_wave_clamp_iw093_values = yes
			independence_wave_refresh_country_state = yes
			set_temp_variable = { independence_wave_iw093_route_transaction_result = constant:independence_wave_iw093.one }
		}
	}
}
