independence_wave_iw098_select_sultan_by_date = {
	set_temp_variable = { independence_wave_iw098_succession_result = constant:independence_wave_iw098.zero }
	clr_country_flag = independence_wave_iw098_sultan_selection_complete
	clr_country_flag = independence_wave_iw098_pre_cutover_sultan_selected
	clr_country_flag = independence_wave_iw098_post_cutover_sultan_selected
	if = {
		limit = {
			OR = {
				is_independence_wave_iw098_prepared_scope = yes
				is_independence_wave_iw098_active_scope = yes
			}
			has_independence_wave_iw098_date_appropriate_role_attestation = yes
		}
		if = {
			limit = { is_independence_wave_iw098_pre_cutover = yes }
			set_country_flag = independence_wave_iw098_pre_cutover_sultan_selected
		}
		else_if = {
			limit = { is_independence_wave_iw098_post_cutover = yes }
			set_country_flag = independence_wave_iw098_post_cutover_sultan_selected
		}
		set_country_flag = independence_wave_iw098_sultan_selection_complete
		set_temp_variable = { independence_wave_iw098_succession_result = constant:independence_wave_iw098.one }
	}
}

		set_variable = { independence_wave_iw093_kumasi_victory_point = constant:independence_wave_iw093.kumasi_victory_point }
