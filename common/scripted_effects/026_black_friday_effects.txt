# ============================================================================
# CHAOS REDUX - EVENT 026 BLACK FRIDAY EFFECTS
# ============================================================================
# Event-owned lifecycle for the actorless global sale. Selection reserves the
# event, while the existing bounded global pulse activates it on the first
# eligible Friday. Activation snapshots one payment ratio, records one
# fire-once history row, refreshes human status markers, and expires on the
# next daily tick. No recurring country-wide scan is introduced here.
# ============================================================================

black_friday_update_calendar_flag = {
	set_temp_variable = { black_friday_calendar_remainder = global.num_days }
	modulo_temp_variable = {
		black_friday_calendar_remainder = constant:black_friday_event.weekday_cycle
	}
	if = {
		limit = {
			check_variable = {
				black_friday_calendar_remainder = constant:black_friday_event.friday_remainder
			}
		}
		set_global_flag = black_friday_friday
	}
	else = {
		clr_global_flag = black_friday_friday
	}
}

black_friday_initialize_runtime = {
	if = {
		limit = { NOT = { has_global_flag = black_friday_runtime_initialized } }
		set_variable = { global.black_friday_state = constant:black_friday_event.state_unavailable }
		set_variable = { global.black_friday_discount_ratio = constant:black_friday_event.baseline_payment_ratio }
		set_variable = { global.black_friday_discount_percent = constant:black_friday_event.baseline_discount_percent }
		set_variable = { global.black_friday_reservation_date = constant:black_friday_event.zero_date }
		set_variable = { global.black_friday_activation_date = constant:black_friday_event.zero_date }
		set_variable = { global.black_friday_expiration_date = constant:black_friday_event.zero_date }
		set_variable = { global.black_friday_sale_sequence = constant:black_friday_event.zero_date }
		set_variable = { global.black_friday_launch_result = constant:black_friday_event.launch_no_op }
		set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.launch_no_op }
		set_variable = { global.black_friday_achievement_family_count = constant:black_friday_event.zero_date }
		clr_global_flag = black_friday_reserved
		clr_global_flag = black_friday_active
		clr_global_flag = black_friday_natural_reservation
		clr_global_flag = black_friday_natural_activation
		clr_global_flag = black_friday_force_disqualified
		clr_global_flag = black_friday_manual_disqualified
		clr_global_flag = black_friday_evolution_i_recorded
		set_global_flag = black_friday_runtime_initialized
	}
	black_friday_update_calendar_flag = yes
}

# Clear only the current sale's achievement progress. The receipt ledger is
# deliberately retained so a refundable payment can still be settled or
# refunded after the one-day sale has expired.
black_friday_clear_current_sale_transaction_state = {
	every_country = {
		clear_array = black_friday_achievement_family_ids
		set_variable = { black_friday_achievement_family_count = constant:black_friday_event.zero_date }
		clr_country_flag = black_friday_achievement_has_institutional_family
		clr_country_flag = black_friday_achievement_has_material_family
		clr_country_flag = black_friday_achievement_has_non_political_family
	}
}

black_friday_prepare_achievement_progress = {
	clear_array = black_friday_achievement_family_ids
	set_variable = { black_friday_achievement_family_count = constant:black_friday_event.zero_date }
	set_variable = { black_friday_achievement_active_sale_sequence = global.black_friday_sale_sequence }
	clr_country_flag = black_friday_achievement_has_institutional_family
	clr_country_flag = black_friday_achievement_has_material_family
	clr_country_flag = black_friday_achievement_has_non_political_family
}

black_friday_add_achievement_family = {
	if = {
		limit = {
			check_variable = {
				var = black_friday_achievement_active_sale_sequence
				value = global.black_friday_sale_sequence
				compare = not_equals
			}
		}
		black_friday_prepare_achievement_progress = yes
	}
	set_temp_variable = { black_friday_purchase_has_non_political_paid_component = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_purchase_component_cursor = constant:universal_cost_framework.zero }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = black_friday_purchase_component_cursor
				value = universal_cost_component_transaction_ids^num
				compare = less_than
			}
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_component_transaction_ids^black_friday_purchase_component_cursor
					value = black_friday_purchase_transaction_id
					compare = equals
				}
				check_variable = {
					var = universal_cost_component_actual_paid_costs^black_friday_purchase_component_cursor
					value = constant:universal_cost_framework.zero
					compare = greater_than
				}
				NOT = {
					check_variable = {
						var = universal_cost_component_resource_kinds^black_friday_purchase_component_cursor
						value = constant:universal_cost_resource_kind.political_power
						compare = equals
					}
				}
			}
			set_temp_variable = { black_friday_purchase_has_non_political_paid_component = constant:universal_cost_framework.one }
		}
		add_to_temp_variable = { black_friday_purchase_component_cursor = constant:universal_cost_framework.one }
	}
	if = {
		limit = {
			NOT = {
				is_in_array = {
					array = black_friday_achievement_family_ids
					value = black_friday_purchase_family_id
				}
			}
		}
		add_to_array = {
			array = black_friday_achievement_family_ids
			value = black_friday_purchase_family_id
		}
		add_to_variable = { black_friday_achievement_family_count = constant:black_friday_event.one }
		if = {
			limit = {
				OR = {
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.laws
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.personnel
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.officer_corps
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.intelligence
						compare = equals
					}
				}
			}
			set_country_flag = black_friday_achievement_has_institutional_family
		}
		if = {
			limit = {
				OR = {
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.equipment
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.convoys
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.trains
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.fuel
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.factory_commitment
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.consumer_goods
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.special_project
						compare = equals
					}
					check_variable = {
						var = black_friday_purchase_family_id
						value = constant:universal_cost_family.state_project
						compare = equals
					}
				}
			}
			set_country_flag = black_friday_achievement_has_material_family
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_purchase_has_non_political_paid_component
					value = constant:universal_cost_framework.one
					compare = equals
				}
			}
			set_country_flag = black_friday_achievement_has_non_political_family
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_achievement_family_count
					value = constant:black_friday_event.achievement_target_family_count
					compare = greater_than_or_equals
				}
				has_country_flag = black_friday_achievement_has_institutional_family
				has_country_flag = black_friday_achievement_has_material_family
				has_country_flag = black_friday_achievement_has_non_political_family
			}
			set_country_flag = black_friday_achievement_awarded
		}
	}
}

# Inputs: black_friday_purchase_transaction_id, black_friday_purchase_family_id,
# black_friday_purchase_ordinary_cost, black_friday_purchase_actual_paid_cost,
# and optional positive black_friday_purchase_is_refundable. Owners call this
# after the shared receipt was recorded and after the real debit succeeded.
black_friday_record_achievement_transaction = {
	if = {
		limit = {
			black_friday_achievement_is_valid_sale = yes
			check_variable = { black_friday_purchase_transaction_id > 0 }
			check_variable = { black_friday_purchase_family_id > 0 }
			check_variable = { black_friday_purchase_ordinary_cost > 0 }
			check_variable = { black_friday_purchase_actual_paid_cost > 0 }
			NOT = {
				is_in_array = {
					array = black_friday_achievement_transaction_ids
					value = black_friday_purchase_transaction_id
				}
			}
		}
		add_to_array = {
			array = black_friday_achievement_transaction_ids
			value = black_friday_purchase_transaction_id
		}
		add_to_array = {
			array = black_friday_achievement_transaction_family_ids
			value = black_friday_purchase_family_id
		}
		add_to_array = {
			array = black_friday_achievement_transaction_sale_sequences
			value = global.black_friday_sale_sequence
		}
		if = {
			limit = {
				has_variable = black_friday_purchase_is_refundable
				check_variable = { black_friday_purchase_is_refundable > 0 }
			}
			add_to_array = {
				array = black_friday_achievement_transaction_states
				value = constant:black_friday_event.achievement_transaction_pending
			}
		}
		else = {
			add_to_array = {
				array = black_friday_achievement_transaction_states
				value = constant:black_friday_event.achievement_transaction_committed
			}
			black_friday_add_achievement_family = yes
		}
	}
}

black_friday_find_achievement_transaction = {
	set_temp_variable = { black_friday_achievement_transaction_index = constant:universal_cost_framework.invalid_index }
	set_temp_variable = { black_friday_achievement_transaction_cursor = constant:black_friday_event.zero_date }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = black_friday_achievement_transaction_cursor
				value = black_friday_achievement_transaction_ids^num
				compare = less_than
			}
			check_variable = {
				var = black_friday_achievement_transaction_index
				value = constant:universal_cost_framework.invalid_index
				compare = equals
			}
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_achievement_transaction_ids^black_friday_achievement_transaction_cursor
					value = black_friday_purchase_transaction_id
					compare = equals
				}
			}
			set_temp_variable = {
				black_friday_achievement_transaction_index = black_friday_achievement_transaction_cursor
			}
		}
		add_to_temp_variable = { black_friday_achievement_transaction_cursor = constant:black_friday_event.one }
	}
}

black_friday_confirm_achievement_transaction = {
	if = {
		limit = {
			black_friday_is_active = yes
			black_friday_achievement_is_valid_sale = yes
			check_variable = { black_friday_purchase_transaction_id > 0 }
		}
		black_friday_find_achievement_transaction = yes
		if = {
			limit = {
				check_variable = {
					var = black_friday_achievement_transaction_index
					value = constant:universal_cost_framework.invalid_index
					compare = greater_than
				}
				check_variable = {
					var = black_friday_achievement_transaction_states^black_friday_achievement_transaction_index
					value = constant:black_friday_event.achievement_transaction_pending
					compare = equals
				}
			}
			set_variable = {
				black_friday_achievement_transaction_states^black_friday_achievement_transaction_index = constant:black_friday_event.achievement_transaction_committed
			}
			set_temp_variable = {
				black_friday_purchase_family_id = black_friday_achievement_transaction_family_ids^black_friday_achievement_transaction_index
			}
			black_friday_add_achievement_family = yes
		}
	}
}

black_friday_retract_achievement_transaction = {
	if = {
		limit = { check_variable = { black_friday_purchase_transaction_id > 0 } }
		black_friday_find_achievement_transaction = yes
		if = {
			limit = {
				check_variable = {
					var = black_friday_achievement_transaction_index
					value = constant:universal_cost_framework.invalid_index
					compare = greater_than
				}
				NOT = {
					check_variable = {
						var = black_friday_achievement_transaction_states^black_friday_achievement_transaction_index
						value = constant:black_friday_event.achievement_transaction_refunded
						compare = equals
					}
				}
			}
			set_variable = {
				black_friday_achievement_transaction_states^black_friday_achievement_transaction_index = constant:black_friday_event.achievement_transaction_refunded
			}
			if = {
				limit = {
					check_variable = {
						var = black_friday_achievement_active_sale_sequence
						value = global.black_friday_sale_sequence
						compare = equals
					}
					check_variable = {
						var = black_friday_achievement_transaction_sale_sequences^black_friday_achievement_transaction_index
						value = global.black_friday_sale_sequence
						compare = equals
					}
					NOT = { has_country_flag = black_friday_achievement_awarded }
				}
				set_temp_variable = {
					black_friday_purchase_family_id = black_friday_achievement_transaction_family_ids^black_friday_achievement_transaction_index
				}
				set_temp_variable = { black_friday_achievement_family_still_used = constant:black_friday_event.zero_date }
				set_temp_variable = { black_friday_achievement_transaction_cursor = constant:black_friday_event.zero_date }
				while_loop_effect = {
					limit = {
						check_variable = {
							var = black_friday_achievement_transaction_cursor
							value = black_friday_achievement_transaction_ids^num
							compare = less_than
						}
					}
					if = {
						limit = {
							check_variable = {
								var = black_friday_achievement_transaction_family_ids^black_friday_achievement_transaction_cursor
								value = black_friday_purchase_family_id
								compare = equals
							}
							check_variable = {
								var = black_friday_achievement_transaction_sale_sequences^black_friday_achievement_transaction_cursor
								value = global.black_friday_sale_sequence
								compare = equals
							}
							check_variable = {
								var = black_friday_achievement_transaction_states^black_friday_achievement_transaction_cursor
								value = constant:black_friday_event.achievement_transaction_committed
								compare = equals
							}
						}
						set_temp_variable = { black_friday_achievement_family_still_used = constant:black_friday_event.one }
					}
					add_to_temp_variable = { black_friday_achievement_transaction_cursor = constant:black_friday_event.one }
				}
				if = {
					limit = {
						check_variable = {
							var = black_friday_achievement_family_still_used
							value = constant:black_friday_event.zero_date
							compare = equals
						}
					}
					remove_from_array = {
						array = black_friday_achievement_family_ids
						value = black_friday_purchase_family_id
					}
					subtract_from_variable = { black_friday_achievement_family_count = constant:black_friday_event.one }
				}
			}
		}
	}
}

# Apply the fire-once pacing transaction without recalculating the event timer.
# The selection transaction already resolved that timer when the reservation
# was created. A delayed Friday activation must therefore update the pacing
# counters and major-event gain in place while allowing that timer to continue.
black_friday_apply_activation_pacing = {
	add_to_variable = { global.total_events_fired = constant:black_friday_event.one }
	add_to_variable = { global.minor_events_since_major = constant:black_friday_event.one }
	add_to_variable = { timer_day_decrement = constant:event_system_timer_defaults.minor_event_decrement_step }
	clamp_variable = {
		var = timer_day_decrement
		max = timer_min_decrement_cap
	}
	set_temp_variable = { black_friday_pacing_timer_check = global.minor_events_since_major }
	modulo_temp_variable = {
		black_friday_pacing_timer_check = constant:event_system_timer_defaults.minor_event_timer_increment
	}
	if = {
		limit = {
			check_variable = {
				var = black_friday_pacing_timer_check
				value = constant:black_friday_event.zero
				compare = equals
			}
			check_variable = {
				var = timer_max_cap_reduction
				value = timer_max_decrement_cap
				compare = less_than
			}
		}
		add_to_variable = { timer_max_cap_reduction = constant:event_system_timer_defaults.minor_event_compression_step }
	}
	else_if = {
		limit = {
			check_variable = {
				var = timer_max_cap_reduction
				value = timer_max_decrement_cap
				compare = greater_than
			}
		}
		set_variable = { timer_max_cap_reduction = timer_max_decrement_cap }
	}
	update_repeatable_event_weights = yes
	apply_dynamic_major_weight_gain_after_minor = yes
	update_unfired_events_counts = yes
}

# Allocate one payer-scoped transaction id for an owner adapter.
black_friday_allocate_owner_transaction = {
	if = {
		limit = { NOT = { has_variable = black_friday_transaction_sequence } }
		set_variable = { black_friday_transaction_sequence = constant:black_friday_event.zero_date }
	}
	add_to_variable = { black_friday_transaction_sequence = constant:black_friday_event.one }
	set_temp_variable = { black_friday_owner_transaction_id = black_friday_transaction_sequence }
}

# The owner-family bridge below accepts one logical Fury purchase as four
# aligned temporary arrays. Each component is quoted, preflighted, paid, and
# receipted before the action's normal effect is allowed to continue. The
# primary component supplies the single achievement family for the logical
# action; secondary components never create a second achievement family.
black_friday_fury_begin_component_list = {
	clear_temp_array = black_friday_fury_component_ids
	clear_temp_array = black_friday_fury_component_resource_kinds
	clear_temp_array = black_friday_fury_component_family_masks
	clear_temp_array = black_friday_fury_component_ordinary_costs
}

# Inputs: black_friday_fury_next_component_id,
# black_friday_fury_next_resource_kind, black_friday_fury_next_family_mask,
# and black_friday_fury_next_ordinary_cost.
black_friday_fury_append_component = {
	add_to_temp_array = {
		array = black_friday_fury_component_ids
		value = black_friday_fury_next_component_id
	}
	add_to_temp_array = {
		array = black_friday_fury_component_resource_kinds
		value = black_friday_fury_next_resource_kind
	}
	add_to_temp_array = {
		array = black_friday_fury_component_family_masks
		value = black_friday_fury_next_family_mask
	}
	add_to_temp_array = {
		array = black_friday_fury_component_ordinary_costs
		value = black_friday_fury_next_ordinary_cost
	}
}

# Inputs are the same as black_friday_fury_append_component, except that
# black_friday_fury_next_spend is the owner's negative payment constant. The
# shared ledger stores positive ordinary costs, so the sign is inverted before
# the component is appended.
black_friday_fury_append_spend_component = {
	set_temp_variable = { black_friday_fury_next_ordinary_cost = black_friday_fury_next_spend }
	multiply_temp_variable = {
		black_friday_fury_next_ordinary_cost = constant:universal_cost_framework.negative_one
	}
	black_friday_fury_append_component = yes
}

# Inputs are the aligned arrays prepared by black_friday_fury_append_component,
# black_friday_fury_primary_component_id, and
# black_friday_fury_primary_family_id. Output:
# black_friday_fury_payment_success. All resources are native framework kinds,
# so the generic refund path can return the actual paid amounts if a later
# component or receipt operation fails.
black_friday_pay_fury_bundle = {
	set_temp_variable = { black_friday_fury_payment_success = constant:universal_cost_framework.zero }
	black_friday_allocate_owner_transaction = yes
	set_temp_variable = { black_friday_fury_all_affordable = constant:universal_cost_framework.one }
	set_temp_variable = {
		black_friday_fury_component_count = black_friday_fury_component_ordinary_costs^num
	}
	set_temp_variable = { black_friday_fury_primary_paid = constant:universal_cost_framework.zero }
	clear_temp_array = black_friday_fury_component_final_costs
	set_temp_variable = { universal_cost_source_query_id = constant:universal_cost_framework.black_friday_source_id }
	if = {
		limit = { NOT = { universal_cost_source_is_active = yes } }
		set_temp_variable = { black_friday_fury_all_affordable = constant:universal_cost_framework.zero }
	}
	for_each_loop = {
		array = black_friday_fury_component_ordinary_costs
		value = black_friday_fury_component_ordinary_cost
		index = black_friday_fury_component_index
		set_temp_variable = { universal_cost_quote_ordinary_cost = black_friday_fury_component_ordinary_cost }
		set_temp_variable = {
			universal_cost_quote_family_mask = black_friday_fury_component_family_masks^black_friday_fury_component_index
		}
		set_temp_variable = { universal_cost_quote_rounding_quantum = constant:universal_cost_framework.default_rounding_quantum }
		if = {
			limit = {
				check_variable = {
					var = black_friday_fury_component_resource_kinds^black_friday_fury_component_index
					value = constant:universal_cost_resource_kind.stability
					compare = equals
				}
			}
			set_temp_variable = { universal_cost_quote_rounding_quantum = constant:universal_cost_fixed_point_quantum.stability }
		}
		else_if = {
			limit = {
				check_variable = {
					var = black_friday_fury_component_resource_kinds^black_friday_fury_component_index
					value = constant:universal_cost_resource_kind.war_support
					compare = equals
				}
			}
			set_temp_variable = { universal_cost_quote_rounding_quantum = constant:universal_cost_fixed_point_quantum.war_support }
		}
		universal_cost_quote_integer = yes
		add_to_temp_array = {
			array = black_friday_fury_component_final_costs
			value = universal_cost_quote_final_cost
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_fury_component_ids^black_friday_fury_component_index
					value = black_friday_fury_primary_component_id
					compare = equals
				}
			}
			set_temp_variable = { black_friday_fury_primary_paid = universal_cost_quote_final_cost }
			set_temp_variable = { black_friday_fury_primary_ordinary = black_friday_fury_component_ordinary_cost }
		}
		set_temp_variable = {
			universal_cost_quote_resource_kind = black_friday_fury_component_resource_kinds^black_friday_fury_component_index
		}
		set_temp_variable = { universal_cost_quote_final_cost = universal_cost_quote_final_cost }
		universal_cost_check_quote_affordable = yes
		if = {
			limit = {
				check_variable = {
					var = universal_cost_quote_affordable
					value = constant:universal_cost_framework.result_success
					compare = not_equals
				}
			}
			set_temp_variable = { black_friday_fury_all_affordable = constant:universal_cost_framework.zero }
		}
	}
	if = {
		limit = {
			check_variable = {
				var = black_friday_fury_all_affordable
				value = constant:universal_cost_framework.one
				compare = equals
			}
		}
		set_temp_variable = { black_friday_fury_payment_cursor = constant:universal_cost_framework.zero }
		set_temp_variable = { black_friday_fury_payment_break = constant:universal_cost_framework.zero }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = black_friday_fury_payment_cursor
					value = black_friday_fury_component_count
					compare = less_than
				}
				check_variable = {
					var = black_friday_fury_payment_break
					value = constant:universal_cost_framework.zero
					compare = equals
				}
			}
			set_temp_variable = {
				universal_cost_payment_resource_kind = black_friday_fury_component_resource_kinds^black_friday_fury_payment_cursor
			}
			set_temp_variable = {
				universal_cost_payment_amount = black_friday_fury_component_final_costs^black_friday_fury_payment_cursor
			}
			universal_cost_pay_component = yes
			if = {
				limit = {
					check_variable = {
						var = universal_cost_payment_result
						value = constant:universal_cost_framework.result_success
						compare = equals
					}
				}
				set_temp_variable = {
					universal_cost_transaction_id = black_friday_owner_transaction_id
				}
				set_temp_variable = {
					universal_cost_transaction_component_id = black_friday_fury_component_ids^black_friday_fury_payment_cursor
				}
				set_temp_variable = {
					universal_cost_transaction_resource_kind = black_friday_fury_component_resource_kinds^black_friday_fury_payment_cursor
				}
				set_temp_variable = { universal_cost_transaction_resource_id = constant:universal_cost_framework.zero }
				set_temp_variable = {
					universal_cost_transaction_ordinary_cost = black_friday_fury_component_ordinary_costs^black_friday_fury_payment_cursor
				}
				set_temp_variable = {
					universal_cost_transaction_quoted_cost = black_friday_fury_component_final_costs^black_friday_fury_payment_cursor
				}
				set_temp_variable = { universal_cost_transaction_actual_paid_cost = universal_cost_transaction_quoted_cost }
				set_temp_variable = { universal_cost_transaction_source_id = constant:universal_cost_framework.black_friday_source_id }
			set_temp_variable = { universal_cost_transaction_primary_component = constant:universal_cost_framework.zero }
			if = {
				limit = {
					check_variable = {
						var = universal_cost_transaction_component_id
						value = black_friday_fury_primary_component_id
						compare = equals
					}
				}
				set_temp_variable = { universal_cost_transaction_primary_component = black_friday_fury_primary_component_id }
			}
			universal_cost_record_transaction = yes
			if = {
				limit = {
					check_variable = {
						var = universal_cost_record_transaction_result
						value = constant:universal_cost_framework.result_success
						compare = not_equals
					}
				}
				set_temp_variable = { black_friday_fury_all_affordable = constant:universal_cost_framework.zero }
				set_temp_variable = { black_friday_fury_payment_break = constant:universal_cost_framework.one }
			}
		}
		else = {
			set_temp_variable = { black_friday_fury_all_affordable = constant:universal_cost_framework.zero }
			set_temp_variable = { black_friday_fury_payment_break = constant:universal_cost_framework.one }
		}
		add_to_temp_variable = {
			black_friday_fury_payment_cursor = constant:universal_cost_framework.one
		}
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_fury_all_affordable
					value = constant:universal_cost_framework.one
					compare = equals
				}
			}
			set_temp_variable = { universal_cost_transaction_id = black_friday_owner_transaction_id }
			universal_cost_settle_transaction = yes
			set_temp_variable = { black_friday_purchase_transaction_id = black_friday_owner_transaction_id }
			set_temp_variable = { black_friday_purchase_family_id = black_friday_fury_primary_family_id }
			set_temp_variable = { black_friday_purchase_ordinary_cost = black_friday_fury_primary_ordinary }
			set_temp_variable = { black_friday_purchase_actual_paid_cost = black_friday_fury_primary_paid }
			black_friday_record_achievement_transaction = yes
			set_temp_variable = { black_friday_fury_payment_success = constant:universal_cost_framework.one }
		}
		else = {
			set_temp_variable = { universal_cost_transaction_id = black_friday_owner_transaction_id }
			universal_cost_refund_transaction = yes
		}
	}
}

black_friday_activate_sale = {
	set_variable = { global.black_friday_launch_result = constant:black_friday_event.launch_no_op }
	if = {
		limit = { NOT = { black_friday_is_active = yes } }
		add_to_variable = { global.black_friday_sale_sequence = constant:black_friday_event.one }
		set_variable = { global.black_friday_activation_date = global.num_days }
		set_variable = { global.black_friday_expiration_date = global.num_days }
		add_to_variable = { global.black_friday_expiration_date = constant:black_friday_event.active_duration_days }
		set_variable = { global.black_friday_discount_ratio = constant:black_friday_event.baseline_payment_ratio }
		set_variable = { global.black_friday_discount_percent = constant:black_friday_event.baseline_discount_percent }
		if = {
			limit = {
				check_variable = {
					var = global.chaos_meter_value
					value = constant:black_friday_event.evolution_chaos
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { events_log_evolution_event_id = constant:black_friday_event.id }
			set_temp_variable = { events_log_evolution_type = constant:black_friday_event.evolution_type }
			set_temp_variable = { events_log_evolution_stage = constant:black_friday_event.evolution_stage_i }
			set_temp_variable = { events_log_evolution_tier = constant:black_friday_event.evolution_tier_i }
			if = {
				limit = { is_current_evolution_enabled = yes }
				set_variable = { global.black_friday_discount_ratio = constant:black_friday_event.evolution_payment_ratio }
				set_variable = { global.black_friday_discount_percent = constant:black_friday_event.evolution_discount_percent }
			}
		}
		if = {
			limit = { check_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_force } }
			set_global_flag = black_friday_force_disqualified
		}
		else_if = {
			limit = { check_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_manual } }
			set_global_flag = black_friday_manual_disqualified
		}
		else_if = {
			limit = { check_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_natural } }
			set_global_flag = black_friday_natural_activation
		}
		else_if = {
			limit = { has_global_flag = black_friday_natural_reservation }
			set_global_flag = black_friday_natural_activation
		}
		else = {
			set_global_flag = black_friday_manual_disqualified
		}
		clr_global_flag = black_friday_natural_reservation
		clr_global_flag = black_friday_reserved
		set_variable = { global.black_friday_reservation_date = constant:black_friday_event.zero_date }
		set_global_flag = black_friday_active
		set_global_flag = black_friday_fired
		set_variable = { global.black_friday_state = constant:black_friday_event.state_active }
		set_temp_variable = { universal_cost_source_ratio = global.black_friday_discount_ratio }
		set_temp_variable = { universal_cost_source_eligibility_mask = constant:universal_cost_framework.source_all_mask }
		universal_cost_source_register_black_friday = yes
		black_friday_record_activation_history = yes
		if = {
			limit = { check_variable = { global.black_friday_discount_ratio = constant:black_friday_event.evolution_payment_ratio } }
			black_friday_record_evolution_i = yes
		}
		every_country = {
			black_friday_refresh_active_cost_modifier = yes
			if = {
				limit = { is_ai = no }
				black_friday_prepare_achievement_progress = yes
				country_event = { id = chaosx.nr26.2 }
			}
		}
		set_variable = { global.black_friday_launch_result = constant:black_friday_event.launch_success }
		log = "CHAOS REDUX BLACK FRIDAY: Sale active at [?global.black_friday_discount_percent]% for the current date"
	}
}

black_friday_reserve_event = {
	if = {
		limit = { black_friday_can_reserve = yes }
		set_global_flag = black_friday_reserved
		set_variable = { global.black_friday_state = constant:black_friday_event.state_reserved }
		set_variable = { global.black_friday_reservation_date = global.num_days }
		set_variable = { global.black_friday_launch_result = constant:black_friday_event.launch_success }
		# Natural provenance must survive after the transient dispatch mode resets.
		if = {
			limit = { check_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_natural } }
			set_global_flag = black_friday_natural_reservation
		}
		else = {
			clr_global_flag = black_friday_natural_reservation
		}
		log = "CHAOS REDUX BLACK FRIDAY: Event 26 reserved while waiting for Friday"
		if = {
			limit = { check_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_natural } }
			# The selection transaction resolves its timer before same-day activation
			# can clear the pending reservation.
			calculate_next_timer_value = yes
		}
		if = {
			limit = { black_friday_is_friday = yes }
			black_friday_activate_sale = yes
		}
	}
}

black_friday_select_or_activate = {
	set_variable = { global.black_friday_launch_result = constant:black_friday_event.launch_no_op }
	black_friday_initialize_runtime = yes
	black_friday_update_calendar_flag = yes
	if = {
		limit = {
			black_friday_event_system_is_running = yes
			NOT = { black_friday_is_disabled = yes }
			check_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_force }
			NOT = { black_friday_is_active = yes }
			NOT = { black_friday_is_fired = yes }
		}
		black_friday_activate_sale = yes
	}
	else_if = {
		limit = { black_friday_is_active = yes }
		log = "CHAOS REDUX BLACK FRIDAY: Duplicate launch ignored while sale is active"
	}
	else_if = {
		limit = { black_friday_can_activate_naturally = yes }
		black_friday_activate_sale = yes
	}
	else_if = {
		limit = { black_friday_is_reserved = yes }
		if = {
			limit = { black_friday_can_activate_naturally = yes }
			black_friday_activate_sale = yes
		}
	}
	else = {
		black_friday_reserve_event = yes
	}
}

black_friday_dispatch_selected_event = {
	set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_manual }
	if = {
		limit = {
			has_variable = temp_bypass_checks
			check_variable = { temp_bypass_checks > 0 }
		}
		set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_force }
	}
	else_if = {
		limit = { has_country_flag = force_trigger_mode_enabled }
		set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_force }
	}
	else_if = {
		limit = {
			has_variable = automatic_event_firing_context
			check_variable = { automatic_event_firing_context > 0 }
		}
		set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_natural }
	}
	set_variable = { global.black_friday_launch_result = constant:black_friday_event.launch_no_op }
	country_event = { id = chaosx.nr26.1 }
	if = {
		limit = {
			check_variable = { global.black_friday_launch_result > 0 }
			black_friday_is_active = yes
		}
		set_temp_variable = { event_fire_dispatched = 1 }
	}
	else = {
		set_temp_variable = { event_fire_dispatched = 0 }
	}
	set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.launch_no_op }
}

black_friday_daily_pulse = {
	black_friday_initialize_runtime = yes
	black_friday_update_calendar_flag = yes
	if = {
		limit = { has_global_flag = world_end }
		black_friday_terminal_cleanup = yes
	}
	else = {
		black_friday_on_event_disabled = yes
		if = {
			limit = {
				black_friday_is_active = yes
				check_variable = {
					var = global.num_days
					value = global.black_friday_expiration_date
					compare = greater_than_or_equals
				}
			}
			black_friday_expire_sale = yes
		}
		else_if = {
			limit = { black_friday_can_activate_naturally = yes }
			black_friday_activate_sale = yes
		}
	}
}

black_friday_on_chaos_changed = {
	black_friday_initialize_runtime = yes
	black_friday_update_calendar_flag = yes
	if = {
		limit = { black_friday_can_activate_naturally = yes }
		black_friday_activate_sale = yes
	}
}

# Resume hook for the edge case where a reservation survives the daily pulse
# while the event system is paused and the same eligible Friday is resumed
# later that day. The normal daily pulse remains the primary path.
black_friday_on_event_system_resumed = {
	black_friday_initialize_runtime = yes
	black_friday_update_calendar_flag = yes
	if = {
		limit = { black_friday_can_activate_naturally = yes }
		black_friday_activate_sale = yes
	}
}

		set_variable = { global.black_friday_dispatch_mode = constant:black_friday_event.dispatch_manual }
	}
	black_friday_select_or_activate = yes
}
