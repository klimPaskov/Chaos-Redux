# Shared building blocks for the rewards of the actor focus tree
# (common/national_focus/031_random_terror_focus.txt) and of the actor
# decisions that raise formations.  COUNTRY scope throughout: the actor
# completing the focus or taking the decision (ROOT).  Amounts live in the
# random_terror_focus_reward, random_terror_unit and random_terror_mobilization
# script constants.
#
# Focus context.  A focus reward that reuses an actor decision helper sets the
# temp variable random_terror_focus_context first, so
# random_terror_actor_action_common skips the decision cooldown.
# ============================================================================

# Raises one formation through random_terror_actor_raise_formation (same temp
# inputs: random_terror_unit_experience, _equipment, _manpower and _id).  At
# the actor division limit the command takes the formation's rifles into its
# stockpile instead, so no reward can raise forces past the cap.
random_terror_focus_raise_formation = {
	if = {
		limit = { num_divisions < constant:random_terror_tuning.actor_division_cap }
		random_terror_actor_raise_formation = yes
	}
	else = {
		add_equipment_to_stockpile = { type = infantry_equipment amount = constant:random_terror_cost.actor_defector_equipment }
	}
}

# command, economy and legitimacy (Part 6 idea lifecycle table).  A route
# focus moves a slot with random_terror_set_<slot>_form, which removes every
# other stable form of that slot, so the tree never stacks two forms of one
# institution and an actor holds at most three tree spirits.  Input: temp
# random_terror_idea_form, a random_terror_idea_form constant.
#
# The failure overlay owned by random_terror_refresh_actor_lifecycles stays
# outside the slots.  Looting Spiral and Sponsor Dependency are both economy
# forms and failure overlays: a slot change removes them only when the overlay
# does not own them, and a focus that chooses one takes it over from the
# overlay, so recovery cannot strip a chosen form.
# ----------------------------------------------------------------------------

# One military factory in an owned state the command holds with a free slot.
# When no such slot exists, the seat gains one shared building slot for it.
random_terror_focus_add_arms_factory = {
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				free_building_slots = { building = arms_factory size > 0 }
			}
		}
		random_owned_controlled_state = {
			limit = { free_building_slots = { building = arms_factory size > 0 } }
			add_building_construction = { type = arms_factory level = 1 instant_build = yes }
		}
	}
	else = {
		capital_scope = {
			add_extra_state_shared_building_slots = 1
			add_building_construction = { type = arms_factory level = 1 instant_build = yes }
		}
	}
}

# Saves random_terror_focus_state: an owned state the command holds that faces
# an enemy, else one that borders a foreign state, else the seat.
random_terror_focus_select_border_state = {
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
			}
		}
		random_owned_controlled_state = {
			limit = { any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } } }
			save_event_target_as = random_terror_focus_state
		}
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				any_neighbor_state = { NOT = { is_owned_by = ROOT } }
			}
		}
		random_owned_controlled_state = {
			limit = { any_neighbor_state = { NOT = { is_owned_by = ROOT } } }
			save_event_target_as = random_terror_focus_state
		}
	}
	else = {
		capital_scope = { save_event_target_as = random_terror_focus_state }
	}
}

# Hides up to protected_cell_count threatened cells abroad
# (random_terror_state_is_threatened_cell): the next government raid in each
# finds only empty safe houses, as after Relocate the Network.
random_terror_focus_protect_threatened_cells = {
	set_temp_variable = { random_terror_focus_cells_done = constant:random_terror_value.zero }
	while_loop_effect = {
		limit = {
			check_variable = { random_terror_focus_cells_done < constant:random_terror_focus_reward.protected_cell_count }
			any_state = { random_terror_state_is_threatened_cell = yes }
		}
		random_state = {
			limit = { random_terror_state_is_threatened_cell = yes }
			set_state_flag = random_terror_cell_relocated
		}
		add_to_temp_variable = { random_terror_focus_cells_done = constant:random_terror_value.one }
	}
}

# Seal the Inner Circle: exposed members are cut off, ending Compromised
# Network, and threatened cells abroad go to ground before the next raid.

	if = {
		limit = { has_idea = random_terror_compromised_network }
		remove_ideas = random_terror_compromised_network

}

# Layered Safehouses: bunkers in the key towns of every held state outside
# the seat, or in the seat when the command holds nothing else.
random_terror_focus_reward_layered_safehouses = {
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				NOT = { is_capital = yes }
			}
		}
		every_owned_state = {
			limit = {
				is_controlled_by = ROOT
				NOT = { is_capital = yes }
			}
			add_building_construction = {
				type = bunker
				level = 1
				province = { all_provinces = yes limit_to_victory_point = yes }
				instant_build = yes
			}
		}
	}
	else = {
		capital_scope = {
			add_building_construction = {
				type = bunker
				level = 1
				province = { all_provinces = yes limit_to_victory_point = yes }
				instant_build = yes
			}
		}
	}
	add_to_variable = { global.random_terror_network_reach = constant:random_terror_tuning.network_reach_gain }
	clamp_variable = { var = global.random_terror_network_reach min = constant:random_terror_value.minimum max = constant:random_terror_value.maximum }
	set_country_flag = random_terror_focus_safehouses_layered
}

# Links up to cell_select_count cells abroad
# (random_terror_state_is_focus_cell_candidate) to the command's network and,
# when temp random_terror_focus_activity_push is above zero, raises their
# activity by it.  Cells bordering the command come first.  With temp
# random_terror_focus_require_link = 1 only cells that already carry a
# transnational link qualify; the input is reset afterwards.
random_terror_focus_reach_foreign_cells = {
	set_temp_variable = { random_terror_focus_cells_done = constant:random_terror_value.zero }
	while_loop_effect = {
		limit = {
			check_variable = { random_terror_focus_cells_done < constant:random_terror_focus_reward.cell_select_count }
			any_state = { random_terror_state_is_focus_cell_candidate = yes }
		}
		if = {
			limit = {
				any_state = {
					random_terror_state_is_focus_cell_candidate = yes
					any_neighbor_state = { is_controlled_by = ROOT }
				}
			}
			random_state = {
				limit = {
					random_terror_state_is_focus_cell_candidate = yes
					any_neighbor_state = { is_controlled_by = ROOT }
				}
				random_terror_focus_mark_foreign_cell = yes
			}
		}
		else = {
			random_state = {
				limit = { random_terror_state_is_focus_cell_candidate = yes }
				random_terror_focus_mark_foreign_cell = yes
			}
		}
		add_to_temp_variable = { random_terror_focus_cells_done = constant:random_terror_value.one }
	}
	every_state = {
		limit = { has_state_flag = random_terror_focus_cell_selected }
		clr_state_flag = random_terror_focus_cell_selected
	}
	set_temp_variable = { random_terror_focus_require_link = constant:random_terror_value.zero }
}

# Lays one railway level from the seat to random_terror_focus_state when that
# state is not the seat.
random_terror_focus_build_rail_to_focus_state = {
	if = {
		limit = { event_target:random_terror_focus_state = { NOT = { is_capital = yes } } }
		capital_scope = { set_temp_variable = { random_terror_rail_start_state = THIS } }
		event_target:random_terror_focus_state = { set_temp_variable = { random_terror_rail_target_state = THIS } }
		meta_effect = {
			text = {
				build_railway = {
					level = 1
					start_state = [RT_START]
					target_state = [RT_TARGET]
				}
			}
			RT_START = "[?random_terror_rail_start_state|.0]"
			RT_TARGET = "[?random_terror_rail_target_state|.0]"
		}
	}
}

# Secure Forward Depots: roads and a fuel silo in a state facing the enemy
# (else a border state, else the seat), and fuel for the stores.
random_terror_focus_reward_forward_depots = {
	random_terror_focus_select_border_state = yes
	event_target:random_terror_focus_state = {
		add_building_construction = { type = infrastructure level = 1 instant_build = yes }
		if = {
			limit = { free_building_slots = { building = fuel_silo size > 0 } }
			add_building_construction = { type = fuel_silo level = 1 instant_build = yes }
		}
	}
	add_fuel = constant:random_terror_focus_reward.forward_depot_fuel
	add_to_variable = { random_terror_external_supply = constant:random_terror_tuning.actor_supply_depot_gain }
	add_to_variable = { random_terror_territorial_control = constant:random_terror_tuning.actor_control_minor_gain }
	set_country_flag = random_terror_focus_forward_depots_secured
}

# Relocate the Command: a fallback seat with bunkers and anti-air in another
# held state, or anti-air over the seat when the command holds no other.
random_terror_focus_reward_fallback_seat = {
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				NOT = { is_capital = yes }
			}
		}
		random_owned_controlled_state = {
			limit = { NOT = { is_capital = yes } }
			add_building_construction = {
				type = bunker
				level = 1
				province = { all_provinces = yes limit_to_victory_point = yes }
				instant_build = yes
			}
			add_building_construction = { type = anti_air_building level = 1 instant_build = yes }
		}
	}
	else = {
		capital_scope = {
			add_building_construction = { type = anti_air_building level = 1 instant_build = yes }
		}
	}
	add_to_variable = { random_terror_network_authority = constant:random_terror_tuning.network_reach_gain }
	add_to_variable = { random_terror_external_supply = constant:random_terror_tuning.actor_supply_depot_gain }
	set_country_flag = random_terror_focus_command_relocated
}

# Identify the Sponsor: a sponsor request (chaosx.nr31.42) goes to a country
# that could back the command, a neighbour first.  The country decides.
random_terror_focus_request_sponsor = {
	if = {
		limit = { any_country = { random_terror_focus_sponsor_candidate = yes } }
		if = {
			limit = {
				any_country = {
					random_terror_focus_sponsor_candidate = yes
					is_neighbor_of = ROOT
				}
			}
			random_country = {
				limit = {
					random_terror_focus_sponsor_candidate = yes
					is_neighbor_of = ROOT
				}
				save_event_target_as = random_terror_sponsor_target
			}
		}
		else = {
			random_country = {
				limit = { random_terror_focus_sponsor_candidate = yes }
				save_event_target_as = random_terror_sponsor_target
			}
		}
		set_country_flag = random_terror_sponsor_contact_recorded
		random_terror_commit_sponsor_request = yes
	}
}

# Prepare the Parent Offensive: every populated state of the parent or of
# another enemy that borders the command is claimed, linked and pressed by
# its cells.
random_terror_focus_reward_parent_offensive = {
	set_temp_variable = { random_terror_focus_activity_push = constant:random_terror_focus_reward.parent_offensive_activity }
	every_controlled_state = {
		every_neighbor_state = {
			limit = {
				NOT = { is_controlled_by = ROOT }
				NOT = { has_state_flag = random_terror_focus_offensive_marked }
				state_population_k > constant:random_terror_value.zero
				CONTROLLER = {
					OR = {
						has_country_flag = random_terror_parent_of_@ROOT
						has_war_with = ROOT
					}
				}
			}
			set_state_flag = random_terror_focus_offensive_marked
			if = {
				limit = {
					NOT = { is_owned_by = ROOT }
					NOT = { is_claimed_by = ROOT }
				}
				add_claim_by = ROOT
			}
			set_state_flag = random_terror_transnational_link_active
			random_terror_focus_push_state_activity = yes
		}
	}
	every_controlled_state = {
		every_neighbor_state = {
			limit = { has_state_flag = random_terror_focus_offensive_marked }
			clr_state_flag = random_terror_focus_offensive_marked
		}
	}
	add_to_variable = { random_terror_territorial_control = constant:random_terror_tuning.actor_control_convert_gain }
	set_country_flag = random_terror_focus_parent_offensive_prepared
}
