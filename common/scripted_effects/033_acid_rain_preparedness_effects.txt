# COUNTRY SCOPE. Reconstructs the selected state from the frozen registry.
# The saved state id is the source of truth; the resolved scope variable is
# rebuilt before each transaction and cleared when commitments are released.
# Using a regular scope variable permits explicit invalidation on failed lookup.
acid_rain_resolve_action_target_state = {
	clear_variable = acid_rain_selected_action_state
	if = {
		limit = { has_variable = acid_rain_action_target_state }
		for_each_scope_loop = {
			array = global.acid_rain_frozen_states
			if = {
				limit = {
					check_variable = { var = THIS.id value = ROOT.acid_rain_action_target_state compare = equals }
					acid_rain_state_is_frozen_valid = yes
				}
				set_variable = { ROOT.acid_rain_selected_action_state = THIS }
			}
		}
	}
}
