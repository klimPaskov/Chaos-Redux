# ---------------------------------------------------------------------------

# Shelter Protocols, Water Distribution, Medical Surge, Reroute Transport and
# demobilization run as national operations, one per family at a time, in the
# acid_rain_op_*_entries rows indexed by acid_rain_action.family_*.
# Evacuations and recovery work run as state operations, one per state, held
# on the state itself and listed in the owner's acid_rain_op_states array.
# acid_rain_evacuations_active and acid_rain_recovery_ops_active count the
# operations in progress against the band caps.
# Every operation keeps its own paid reservation, schedules its own completion
# callback and is released exactly once by completion, interruption or cleanup.
# Completion, release and binding effects never read ROOT, so they run from
# callbacks and participant loops alike; only acid_rain_start_action expects
# the deciding country as ROOT.

# COUNTRY SCOPE. Applies a finished action. National families act on the
# country; evacuations and recovery act on var:acid_rain_selected_action_state.
acid_rain_apply_action_effect = {
	set_temp_variable = { acid_rain_completed_action_type = acid_rain_selected_action }
	if = { limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_shelter compare = equals } } set_country_flag = { flag = acid_rain_action_shelter_active days = acid_rain_quote_effect_days } }
	if = { limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_water_land compare = equals } } set_country_flag = { flag = acid_rain_action_water_active days = acid_rain_quote_effect_days } }
	if = { limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_water_island compare = equals } } set_country_flag = { flag = acid_rain_action_water_active days = acid_rain_quote_effect_days } }
	if = { limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_medical compare = equals } } set_country_flag = { flag = acid_rain_action_medical_active days = acid_rain_quote_effect_days } set_temp_variable = { acid_rain_medical_fatigue_days = constant:acid_rain_preparedness.urgent_medical_effect_days } set_country_flag = { flag = acid_rain_medical_fatigue days = acid_rain_medical_fatigue_days } }
	if = { limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_transport compare = equals } } set_country_flag = { flag = acid_rain_action_transport_active days = acid_rain_quote_effect_days } }
	if = {
		limit = {
			OR = {
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
			}
		}
		var:acid_rain_selected_action_state = {
			if = {
				limit = { NOT = { has_state_flag = acid_rain_evacuated } }
				set_state_flag = acid_rain_evacuated
				set_state_flag = acid_rain_evacuation_receipt
				set_variable = { acid_rain_evacuation_generation = global.acid_rain_generation_id }
				set_variable = { acid_rain_evacuation_country_id = CONTROLLER.id }
				set_variable = { acid_rain_evacuation_completion_day = global.date }
				acid_rain_033_achievement_record_runtime_evacuation_completion = yes
			}
		}
	}
	if = {
		limit = {
			check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_transport compare = greater_than_or_equals }
			check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_industry compare = less_than_or_equals }
		}
		var:acid_rain_selected_action_state = { acid_rain_apply_recovery_result = yes }
	}
	if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_demobilize compare = equals } }
		acid_rain_demobilize_emergency_apparatus = yes
	}
}

		var:acid_rain_action_owner = {
			set_variable = { acid_rain_selected_action_state = PREV }

							check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_island compare = equals }

				}
				else = { event_target:acid_rain_completed_action_state = { acid_rain_record_paid_recovery_consequence = yes } }
				set_country_flag = acid_rain_action_receipt
			}
			else = { set_country_flag = acid_rain_action_interrupted_stale_target }

			acid_rain_project_country_obligations = yes
			acid_rain_gui_refresh_read_only = yes

	}

# COUNTRY SCOPE. Returns the reservation of the state operation in
# var:acid_rain_selected_action_state once, unlocks the state and lowers the
# matching in-progress counter. Only the paying country may release it.
acid_rain_release_state_operation = {
	if = {
		limit = {
			has_variable = acid_rain_selected_action_state
			var:acid_rain_selected_action_state = {
				has_state_flag = acid_rain_action_lock
				check_variable = { var = acid_rain_action_owner_country_id value = PREV.id compare = equals }
			}
		}
		set_temp_variable = { acid_rain_reserved_manpower = constant:acid_rain_runtime.zero }
		set_temp_variable = { acid_rain_reserved_civ = constant:acid_rain_runtime.zero }
		set_temp_variable = { acid_rain_released_type = constant:acid_rain_runtime.zero }
		var:acid_rain_selected_action_state = {
			set_temp_variable = { acid_rain_reserved_manpower = acid_rain_action_reserved_manpower }
			set_temp_variable = { acid_rain_reserved_civ = acid_rain_action_reserved_civ }
			set_temp_variable = { acid_rain_released_type = acid_rain_action_type }
			# A failed evacuation ends its disruption now; a finished one keeps
			# it until the severe episode closes.
			if = {
				limit = { NOT = { has_state_flag = acid_rain_evacuated } }
				remove_dynamic_modifier = { modifier = acid_rain_evacuation_disruption }
			}
			clear_variable = acid_rain_evacuation_partial_day
			clr_state_flag = acid_rain_action_lock
			clear_variable = acid_rain_action_owner
			clear_variable = acid_rain_action_owner_generation
			clear_variable = acid_rain_action_owner_country_id
			clear_variable = acid_rain_action_type
			clear_variable = acid_rain_action_start_day
			clear_variable = acid_rain_action_due_day
			clear_variable = acid_rain_action_reserved_manpower
			clear_variable = acid_rain_action_reserved_civ
			clear_variable = acid_rain_action_key
			clear_variable = acid_rain_action_visit
			clear_variable = acid_rain_action_region
			clear_variable = acid_rain_action_front
			clear_variable = acid_rain_action_warning_day
			PREV = { remove_from_array = { array = acid_rain_op_states value = PREV } }
		}
		add_manpower = acid_rain_reserved_manpower
		subtract_from_variable = { acid_rain_committed_manpower = acid_rain_reserved_manpower }
		subtract_from_variable = { acid_rain_committed_civilian_factories = acid_rain_reserved_civ }
		subtract_from_variable = { acid_rain_action_civ_committed = acid_rain_reserved_civ }
		if = {
			limit = {
				OR = {
					check_variable = { var = acid_rain_released_type value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
					check_variable = { var = acid_rain_released_type value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
				}
			}
			subtract_from_variable = { acid_rain_evacuations_active = constant:acid_rain_runtime.loop_increment }
			clamp_variable = { var = acid_rain_evacuations_active min = constant:acid_rain_runtime.zero }
		}
		else = {
			subtract_from_variable = { acid_rain_recovery_ops_active = constant:acid_rain_runtime.loop_increment }
			clamp_variable = { var = acid_rain_recovery_ops_active min = constant:acid_rain_runtime.zero }
		}
		clear_variable = acid_rain_selected_action_state
		acid_rain_refresh_operation_state = yes
		acid_rain_refresh_country_project_quotes = yes
		acid_rain_refresh_country_action_quotes = yes
	}
}

# COUNTRY SCOPE. Interrupts running operations and returns their
# reservations; spent equipment, fuel, trains, convoys, trucks and political
# power stay spent. Input: temporary acid_rain_keep_recovery_operations, one
# to keep recovery work and demobilization running through acute dissipation.
acid_rain_release_running_operations = {
	for_loop_effect = {
		start = constant:acid_rain_runtime.one
		end = constant:acid_rain_action.family_rows
		value = acid_rain_op_release_family
		set_temp_variable = { acid_rain_op_family = acid_rain_op_release_family }
		if = {
			limit = {
				check_variable = { var = acid_rain_op_action_entries^acid_rain_op_family value = constant:acid_rain_runtime.zero compare = greater_than }
				OR = {
					check_variable = { var = acid_rain_keep_recovery_operations value = constant:acid_rain_runtime.zero compare = equals }
					check_variable = { var = acid_rain_op_family value = constant:acid_rain_action.family_demobilize compare = less_than }
				}
			}
			set_country_flag = acid_rain_action_interrupted_cleanup
			acid_rain_release_national_operation = yes
		}
	}
	clear_temp_array = acid_rain_op_release_states
	for_each_scope_loop = {
		array = acid_rain_op_states
		add_to_temp_array = { array = acid_rain_op_release_states value = THIS }
	}
	for_each_scope_loop = {
		array = acid_rain_op_release_states
		if = {
			limit = {
				OR = {
					check_variable = { var = acid_rain_keep_recovery_operations value = constant:acid_rain_runtime.zero compare = equals }
					check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.recovery_transport compare = less_than }
				}
			}
			PREV = {
				set_variable = { acid_rain_selected_action_state = PREV }
				set_country_flag = acid_rain_action_interrupted_cleanup
				acid_rain_release_state_operation = yes
			}
		}
	}
}
