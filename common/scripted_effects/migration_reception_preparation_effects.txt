# Paid exact-state service preparation. The caller saves FROM as the regular
# migration_reception_preparation_state target and predeclares the public result.
# This helper owns one debit and a complete failure refund, never population/load.

migration_prepare_reception_services = {
	set_temp_variable = { migration_reception_preparation_result = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_target_ok = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_pending = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_shelter_changed = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_medical_changed = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_policy_changed = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_policy_attempted = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_shelter_attempted = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_preparation_medical_attempted = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_reception_shelter_service_result = constant:migration_capacity.zero }
	set_temp_variable = { migration_reception_medical_service_result = constant:migration_capacity.zero }
	set_temp_variable = { migration_reception_candidate_result = constant:migration_capacity.zero }
	set_temp_variable = { migration_decision_policy_result = constant:civilian_transfer_route_result.invalid }
	save_event_target_as = migration_reception_preparation_receiver
	if = {
		limit = {
			has_event_target = migration_reception_preparation_state
			migration_reception_preparation_is_visible = yes
			migration_reception_preparation_is_affordable = yes
			num_of_civilian_factories > constant:humanitarian_population.zero
		}
		event_target:migration_reception_preparation_state = {
			set_temp_variable = { migration_famine_food_safety_result = constant:humanitarian_runtime.zero }
			migration_consume_famine_food_safety = yes
			if = {
				limit = {
					is_owned_and_controlled_by = event_target:migration_reception_preparation_receiver
					migration_reception_preparation_state_is_safe = yes
					migration_reception_preparation_can_change_state = yes
					event_target:migration_reception_preparation_receiver = {
						OR = {
							NOT = { has_country_flag = humanitarian_corridor_offer_pending }
							AND = {
								migration_pending_reception_issue_is_valid = yes
								check_variable = { humanitarian_corridor_offer_front_state_id = PREV.id }
							}
						}
					}
				}
				set_temp_variable = { migration_reception_preparation_target_ok = constant:humanitarian_runtime.one }
			}
		}
		if = {
			limit = { check_variable = { migration_reception_preparation_target_ok = constant:humanitarian_runtime.one } }
			if = {
				limit = { has_country_flag = humanitarian_corridor_offer_pending }
				set_temp_variable = { migration_reception_preparation_pending = constant:humanitarian_runtime.one }
				set_temp_variable = { migration_reception_preparation_generation = humanitarian_corridor_offer_route_generation }
			}
			set_temp_variable = { migration_reception_policy_prep_had = constant:humanitarian_runtime.zero }
			if = { limit = { has_variable = migration_reception_policy } set_temp_variable = { migration_reception_policy_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_reception_policy_prep_before = migration_reception_policy } }
			set_temp_variable = { migration_departure_policy_prep_had = constant:humanitarian_runtime.zero }
			if = { limit = { has_variable = migration_departure_policy } set_temp_variable = { migration_departure_policy_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_departure_policy_prep_before = migration_departure_policy } }
			set_temp_variable = { migration_border_policy_prep_had = constant:humanitarian_runtime.zero }
			if = { limit = { has_variable = migration_border_policy } set_temp_variable = { migration_border_policy_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_border_policy_prep_before = migration_border_policy } }
			set_temp_variable = { migration_reception_preparation_border_flag_before = constant:humanitarian_runtime.zero }
			if = { limit = { has_country_flag = migration_border_policy_active } set_temp_variable = { migration_reception_preparation_border_flag_before = constant:humanitarian_runtime.one } }
			event_target:migration_reception_preparation_state = {
				set_temp_variable = { migration_reception_shelter_capacity_prep_had = constant:humanitarian_runtime.zero }
				if = { limit = { has_variable = migration_reception_shelter_capacity } set_temp_variable = { migration_reception_shelter_capacity_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_reception_shelter_capacity_prep_before = migration_reception_shelter_capacity } }
				set_temp_variable = { migration_reception_shelter_service_proven_prep_had = constant:humanitarian_runtime.zero }
				if = { limit = { has_variable = migration_reception_shelter_service_proven } set_temp_variable = { migration_reception_shelter_service_proven_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_reception_shelter_service_proven_prep_before = migration_reception_shelter_service_proven } }
				set_temp_variable = { migration_reception_medical_capacity_prep_had = constant:humanitarian_runtime.zero }
				if = { limit = { has_variable = migration_reception_medical_capacity } set_temp_variable = { migration_reception_medical_capacity_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_reception_medical_capacity_prep_before = migration_reception_medical_capacity } }
				set_temp_variable = { migration_reception_medical_service_proven_prep_had = constant:humanitarian_runtime.zero }
				if = { limit = { has_variable = migration_reception_medical_service_proven } set_temp_variable = { migration_reception_medical_service_proven_prep_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_reception_medical_service_proven_prep_before = migration_reception_medical_service_proven } }
				set_temp_variable = { migration_reception_preparation_candidate_before = constant:humanitarian_runtime.zero }
				set_temp_variable = { migration_reception_preparation_candidate_revision_had = constant:humanitarian_runtime.zero }
				if = { limit = { has_variable = migration_reception_candidate_revision } set_temp_variable = { migration_reception_preparation_candidate_revision_had = constant:humanitarian_runtime.one } set_temp_variable = { migration_reception_preparation_candidate_revision_before = migration_reception_candidate_revision } }
				if = { limit = { has_state_flag = migration_reception_candidate_active } set_temp_variable = { migration_reception_preparation_candidate_before = constant:humanitarian_runtime.one } }
				set_temp_variable = { migration_reception_preparation_array_before = constant:humanitarian_runtime.zero }
				if = { limit = { is_in_array = { array = global.migration_reception_candidate_states value = THIS } } set_temp_variable = { migration_reception_preparation_array_before = constant:humanitarian_runtime.one } }
				set_temp_variable = { migration_reception_preparation_population = state_population_k }
				multiply_temp_variable = { migration_reception_preparation_population = constant:migration_capacity.people_per_k }
				set_temp_variable = { migration_reception_preparation_shelter = migration_reception_preparation_population }
				multiply_temp_variable = { migration_reception_preparation_shelter = constant:migration_reception_bootstrap.shelter_population_share }
				round_temp_variable = migration_reception_preparation_shelter
				set_temp_variable = { migration_reception_preparation_medical = migration_reception_preparation_population }
				multiply_temp_variable = { migration_reception_preparation_medical = constant:migration_reception_bootstrap.medical_population_share }
				round_temp_variable = migration_reception_preparation_medical
			}
			set_temp_variable = { migration_reception_preparation_pp_payment = constant:migration_reception_preparation_cost.political_power }
			multiply_temp_variable = { migration_reception_preparation_pp_payment = constant:migration_reception_preparation_cost.debit_sign }
			set_temp_variable = { migration_reception_preparation_support_payment = constant:migration_reception_preparation_cost.support_equipment }
			multiply_temp_variable = { migration_reception_preparation_support_payment = constant:migration_reception_preparation_cost.debit_sign }
			set_temp_variable = { migration_reception_preparation_fuel_payment = constant:migration_reception_preparation_cost.fuel }
			multiply_temp_variable = { migration_reception_preparation_fuel_payment = constant:migration_reception_preparation_cost.debit_sign }
			add_political_power = migration_reception_preparation_pp_payment
			add_equipment_to_stockpile = { type = support_equipment amount = migration_reception_preparation_support_payment }
			add_fuel = migration_reception_preparation_fuel_payment
			event_target:migration_reception_preparation_state = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable = migration_reception_shelter_capacity }
							NOT = { check_variable = { migration_reception_shelter_service_proven = constant:migration_capacity.one } }
							check_variable = { migration_reception_shelter_capacity < migration_reception_preparation_shelter }
						}
					}
					set_temp_variable = { migration_reception_shelter_service_request = migration_reception_preparation_shelter }
					set_temp_variable = { migration_reception_preparation_shelter_attempted = constant:humanitarian_runtime.one }
					set_temp_variable = { migration_reception_shelter_service_request_proven = constant:migration_capacity.one }
					migration_set_reception_shelter_service = yes
					if = { limit = { check_variable = { migration_reception_shelter_service_result = constant:migration_capacity.one } } set_temp_variable = { migration_reception_preparation_shelter_changed = constant:humanitarian_runtime.one } }
				}
				if = {
					limit = {
						OR = {
							NOT = { has_variable = migration_reception_medical_capacity }
							NOT = { check_variable = { migration_reception_medical_service_proven = constant:migration_capacity.one } }
							check_variable = { migration_reception_medical_capacity < migration_reception_preparation_medical }
						}
					}
					set_temp_variable = { migration_reception_medical_service_request = migration_reception_preparation_medical }
					set_temp_variable = { migration_reception_preparation_medical_attempted = constant:humanitarian_runtime.one }
					set_temp_variable = { migration_reception_medical_service_request_proven = constant:migration_capacity.one }
					migration_set_reception_medical_service = yes
					if = { limit = { check_variable = { migration_reception_medical_service_result = constant:migration_capacity.one } } set_temp_variable = { migration_reception_preparation_medical_changed = constant:humanitarian_runtime.one } }
				}
			}
			if = {
				limit = { NOT = { check_variable = { migration_reception_policy = constant:migration_border_policy.humanitarian_open } } }
				set_temp_variable = { migration_reception_policy_request = constant:migration_border_policy.humanitarian_open }
				set_temp_variable = { migration_departure_policy_request = constant:humanitarian_runtime.zero }
				set_temp_variable = { migration_reception_preparation_policy_attempted = constant:humanitarian_runtime.one }
				set_temp_variable = { migration_border_policy_result = constant:humanitarian_runtime.zero }
				migration_decision_set_policy = yes
				if = { limit = { check_variable = { migration_decision_policy_result = constant:civilian_transfer_route_result.valid } } set_temp_variable = { migration_reception_preparation_policy_changed = constant:humanitarian_runtime.one } }
			}
			migration_mark_reception_capacity_dirty = yes
			migration_recalculate_reception_capacity = yes
			if = {
				limit = {
					OR = {
						check_variable = { migration_reception_preparation_shelter_changed = constant:humanitarian_runtime.one }
						check_variable = { migration_reception_preparation_medical_changed = constant:humanitarian_runtime.one }
						check_variable = { migration_reception_preparation_policy_changed = constant:humanitarian_runtime.one }
					}
					migration_reception_capacity_is_valid = yes
					OR = { check_variable = { migration_reception_preparation_policy_attempted = constant:humanitarian_runtime.zero } check_variable = { migration_decision_policy_result = constant:civilian_transfer_route_result.valid } }
					OR = { check_variable = { migration_reception_preparation_shelter_attempted = constant:humanitarian_runtime.zero } check_variable = { migration_reception_shelter_service_result = constant:migration_capacity.one } }
					OR = { check_variable = { migration_reception_preparation_medical_attempted = constant:humanitarian_runtime.zero } check_variable = { migration_reception_medical_service_result = constant:migration_capacity.one } }
					event_target:migration_reception_preparation_state = {
						is_owned_and_controlled_by = event_target:migration_reception_preparation_receiver
						migration_reception_preparation_state_is_safe = yes
						migration_reception_candidate_is_valid = yes
					}
					OR = {
						check_variable = { migration_reception_preparation_pending = constant:humanitarian_runtime.zero }
						AND = {
							humanitarian_corridor_offer_can_be_accepted = yes
							check_variable = { humanitarian_corridor_offer_route_generation = migration_reception_preparation_generation }
						}
					}
				}
				# Pending success also re-resolves the entire live row before publication.
				set_temp_variable = { migration_reception_preparation_result = constant:humanitarian_runtime.one }
				if = {
					limit = { check_variable = { migration_reception_preparation_pending = constant:humanitarian_runtime.one } }
					var:humanitarian_corridor_offer_origin_state_id = {
						set_temp_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.zero }
						humanitarian_corridor_resolve_live_cohort = yes
						if = {
							limit = { NOT = { check_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.one } } }
							set_temp_variable = { migration_reception_preparation_result = constant:humanitarian_runtime.zero }
						}
					}
				}
			}
			# Qualitative intent advances only after genuine, fully revalidated
			# paid success. Projection failure cannot charge/refund or reject setup.
			if = {
				limit = {
					check_variable = { migration_reception_preparation_result = constant:humanitarian_runtime.one }
					check_variable = { migration_reception_preparation_pending = constant:humanitarian_runtime.one }
					migration_pending_reception_issue_is_valid = yes
				}
				set_temp_variable = { migration_reception_front_intent_record_result = constant:migration_reception_front_intent.zero }
				set_temp_variable = { migration_reception_front_intent_reveal_proven = constant:migration_reception_front_intent.one }
				set_temp_variable = { migration_reception_front_intent_setup_proven = constant:migration_reception_front_intent.one }
				set_temp_variable = { migration_reception_front_intent_phase_request = constant:migration_reception_front_intent.prepared }
				var:humanitarian_corridor_offer_origin_state_id = { migration_publish_reception_front_intent_from_contract = yes }
			}
			if = {
				limit = { check_variable = { migration_reception_preparation_result = constant:humanitarian_runtime.zero } }
				event_target:migration_reception_preparation_state = {
					if = {
						limit = { check_variable = { migration_reception_preparation_shelter_changed = constant:humanitarian_runtime.one } check_variable = { migration_reception_shelter_capacity = migration_reception_preparation_shelter } check_variable = { migration_reception_shelter_service_proven = constant:migration_capacity.one } }
						if = { limit = { check_variable = { migration_reception_shelter_capacity_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_reception_shelter_capacity = migration_reception_shelter_capacity_prep_before } }
						else = { clear_variable = migration_reception_shelter_capacity }
						if = { limit = { check_variable = { migration_reception_shelter_service_proven_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_reception_shelter_service_proven = migration_reception_shelter_service_proven_prep_before } }
						else = { clear_variable = migration_reception_shelter_service_proven }
					}
					if = {
						limit = { check_variable = { migration_reception_preparation_medical_changed = constant:humanitarian_runtime.one } check_variable = { migration_reception_medical_capacity = migration_reception_preparation_medical } check_variable = { migration_reception_medical_service_proven = constant:migration_capacity.one } }
						if = { limit = { check_variable = { migration_reception_medical_capacity_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_reception_medical_capacity = migration_reception_medical_capacity_prep_before } }
						else = { clear_variable = migration_reception_medical_capacity }
						if = { limit = { check_variable = { migration_reception_medical_service_proven_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_reception_medical_service_proven = migration_reception_medical_service_proven_prep_before } }
						else = { clear_variable = migration_reception_medical_service_proven }
					}
					if = { limit = { check_variable = { migration_reception_preparation_candidate_before = constant:humanitarian_runtime.zero } } clr_state_flag = migration_reception_candidate_active }
					else = { set_state_flag = migration_reception_candidate_active }
					if = { limit = { check_variable = { migration_reception_preparation_array_before = constant:humanitarian_runtime.zero } } remove_from_array = { array = global.migration_reception_candidate_states value = THIS } }
					else_if = { limit = { NOT = { is_in_array = { array = global.migration_reception_candidate_states value = THIS } } } add_to_array = { array = global.migration_reception_candidate_states value = THIS } }
					set_variable = { global.migration_reception_candidate_states_count = global.migration_reception_candidate_states^num }
					if = { limit = { check_variable = { migration_reception_preparation_candidate_revision_had = constant:humanitarian_runtime.one } } set_variable = { migration_reception_candidate_revision = migration_reception_preparation_candidate_revision_before } }
					else = { clear_variable = migration_reception_candidate_revision }
				}
				if = {
					limit = { check_variable = { migration_reception_preparation_policy_attempted = constant:humanitarian_runtime.one } check_variable = { migration_reception_policy = constant:migration_border_policy.humanitarian_open } }
					if = { limit = { check_variable = { migration_reception_policy_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_reception_policy = migration_reception_policy_prep_before } }
					else = { clear_variable = migration_reception_policy }
					if = { limit = { check_variable = { migration_departure_policy_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_departure_policy = migration_departure_policy_prep_before } }
					else = { clear_variable = migration_departure_policy }
					if = { limit = { check_variable = { migration_border_policy_prep_had = constant:humanitarian_runtime.one } } set_variable = { migration_border_policy = migration_border_policy_prep_before } }
					else = { clear_variable = migration_border_policy }
					if = { limit = { check_variable = { migration_reception_preparation_border_flag_before = constant:humanitarian_runtime.zero } } clr_country_flag = migration_border_policy_active }
				}
				migration_mark_reception_capacity_dirty = yes
				migration_recalculate_reception_capacity = yes
				set_temp_variable = { migration_reception_preparation_pp_refund = constant:migration_reception_preparation_cost.political_power }
				set_temp_variable = { migration_reception_preparation_support_refund = constant:migration_reception_preparation_cost.support_equipment }
				set_temp_variable = { migration_reception_preparation_fuel_refund = constant:migration_reception_preparation_cost.fuel }
				add_political_power = migration_reception_preparation_pp_refund
				add_equipment_to_stockpile = { type = support_equipment amount = migration_reception_preparation_support_refund }
				add_fuel = migration_reception_preparation_fuel_refund
			}
		}
	}
}
