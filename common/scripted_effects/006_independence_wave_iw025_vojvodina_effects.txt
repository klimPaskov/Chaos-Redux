independence_wave_iw025_vojvodina_remove_overlay_ideas = {
	remove_ideas = {
		independence_wave_iw025_vojvodina_contested_danube_administration
		independence_wave_iw025_vojvodina_coordinated_grain_authority
		independence_wave_iw025_vojvodina_municipal_minority_charter
		independence_wave_iw025_vojvodina_federal_frontier_compact
	}
}

independence_wave_iw025_vojvodina_refresh_idea_lifecycle = {
	if = {
		limit = { is_independence_wave_iw025_vojvodina_overlay_active = yes }
		independence_wave_iw025_vojvodina_remove_overlay_ideas = yes
		if = {
			limit = { has_country_flag = independence_wave_iw025_vojvodina_municipal_charter }
			add_ideas = independence_wave_iw025_vojvodina_municipal_minority_charter
		}
		else_if = {
			limit = { has_country_flag = independence_wave_iw025_vojvodina_federal_agrarian_compact }
			add_ideas = independence_wave_iw025_vojvodina_federal_frontier_compact
		}
		else_if = {
			limit = {
				check_variable = {
					var = independence_wave_iw025_vojvodina_river_logistics
					value = constant:independence_wave_iw025_vojvodina_value.settlement_threshold
					compare = greater_than_or_equals
				}
				check_variable = {
					var = independence_wave_iw025_vojvodina_border_mobility
					value = constant:independence_wave_iw025_vojvodina_value.settlement_threshold
					compare = greater_than_or_equals
				}
			}
			add_ideas = independence_wave_iw025_vojvodina_coordinated_grain_authority
		}
		else = {
			add_ideas = independence_wave_iw025_vojvodina_contested_danube_administration
		}
	}
}

independence_wave_iw025_vojvodina_update_watch_hold = {
	if = {
		limit = {
			is_independence_wave_iw025_vojvodina_overlay_active = yes
			has_country_flag = independence_wave_iw025_vojvodina_watch_running
			has_independence_wave_iw025_vojvodina_watch_objective = yes
		}
		add_to_variable = { independence_wave_iw025_vojvodina_watch_hold_days = constant:independence_wave_iw025_vojvodina_value.watch_daily_gain }
		clamp_variable = {
			var = independence_wave_iw025_vojvodina_watch_hold_days
			min = constant:independence_wave_iw025_vojvodina_value.minimum
			max = constant:independence_wave_iw025_vojvodina_value.watch_required
		}
	}
	else_if = {
		limit = {
			is_independence_wave_iw025_vojvodina_overlay_active = yes
			has_country_flag = independence_wave_iw025_vojvodina_watch_running
		}
		set_variable = { independence_wave_iw025_vojvodina_watch_hold_days = constant:independence_wave_iw025_vojvodina_value.minimum }
	}
}

independence_wave_iw025_vojvodina_initialize_overlay = {
	if = {
		limit = {
			is_independence_wave_iw025_vojvodina_route_active = yes
			NOT = { has_country_flag = independence_wave_iw025_vojvodina_overlay_ever_activated }
		}
		set_country_flag = independence_wave_iw025_vojvodina_overlay_ever_activated
		set_country_flag = independence_wave_iw025_vojvodina_package_initialized
		set_country_flag = independence_wave_iw025_vojvodina_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw025_vojvodina_overlay_suspended
		set_variable = { independence_wave_iw025_vojvodina_river_logistics = constant:independence_wave_iw025_vojvodina_value.river_logistics_start }
		set_variable = { independence_wave_iw025_vojvodina_provincial_legitimacy = constant:independence_wave_iw025_vojvodina_value.provincial_legitimacy_start }
		set_variable = { independence_wave_iw025_vojvodina_border_mobility = constant:independence_wave_iw025_vojvodina_value.border_mobility_start }
		set_variable = { independence_wave_iw025_vojvodina_watch_hold_days = constant:independence_wave_iw025_vojvodina_value.minimum }
		set_variable = { independence_wave_anchor_state = 45 }
		set_variable = { independence_wave_package_archetype = constant:independence_wave_package_archetype.river_or_corridor }
		set_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.mounted_mobile }
		independence_wave_iw025_vojvodina_refresh_idea_lifecycle = yes
	}
}

independence_wave_iw025_vojvodina_suspend_overlay = {
	if = {
		limit = { has_country_flag = independence_wave_iw025_vojvodina_overlay_active }
		if = {
			limit = {
				has_country_flag = independence_wave_iw025_vojvodina_watch_running
				NOT = { has_country_flag = independence_wave_iw025_vojvodina_watch_completed }
			}
			set_country_flag = independence_wave_iw025_vojvodina_watch_interrupted
		}
		independence_wave_iw025_vojvodina_remove_overlay_ideas = yes
		clr_country_flag = independence_wave_iw025_vojvodina_overlay_active
		clr_country_flag = independence_wave_route_overlay_active
		set_country_flag = independence_wave_iw025_vojvodina_overlay_suspended
	}
}

independence_wave_iw025_vojvodina_pause_watch_mission = {
	if = {
		limit = {
			has_country_flag = independence_wave_iw025_vojvodina_overlay_suspended
			has_country_flag = independence_wave_iw025_vojvodina_watch_interrupted
			has_country_flag = independence_wave_iw025_vojvodina_watch_running
			has_active_mission = independence_wave_iw025_hold_vojvodina_border_watch
		}
		set_temp_variable = { independence_wave_iw025_vojvodina_pause_days = constant:independence_wave_iw025_vojvodina_duration.watch_pause_daily_extension }
		add_days_mission_timeout = {
			mission = independence_wave_iw025_hold_vojvodina_border_watch
			days = independence_wave_iw025_vojvodina_pause_days
		}
	}
}

independence_wave_iw025_vojvodina_resume_overlay = {
	if = {
		limit = {
			is_independence_wave_iw025_vojvodina_route_active = yes
			has_country_flag = independence_wave_iw025_vojvodina_overlay_ever_activated
			NOT = { has_country_flag = independence_wave_iw025_vojvodina_overlay_active }
		}
		set_country_flag = independence_wave_iw025_vojvodina_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw025_vojvodina_overlay_suspended
		independence_wave_iw025_vojvodina_refresh_idea_lifecycle = yes
	}
}

# Called only from the dynamic-country on_daily_Dxx hooks below. There is no
# global country iteration.
independence_wave_iw025_vojvodina_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw025_vojvodina_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw025_vojvodina_overlay_ever_activated } }
			independence_wave_iw025_vojvodina_initialize_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw025_vojvodina_overlay_active } }
			independence_wave_iw025_vojvodina_resume_overlay = yes
		}
		else = {
			clr_country_flag = independence_wave_iw025_vojvodina_overlay_suspended
			if = {
				limit = { has_country_flag = independence_wave_iw025_vojvodina_watch_interrupted }
				if = {
					limit = { has_country_flag = independence_wave_iw025_vojvodina_watch_running has_active_mission = independence_wave_iw025_hold_vojvodina_border_watch }
					clr_country_flag = independence_wave_iw025_vojvodina_watch_interrupted
				}
				else = { independence_wave_iw025_vojvodina_fail_watch_mission = yes }
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw025_vojvodina_watch_running
					NOT = { has_active_mission = independence_wave_iw025_hold_vojvodina_border_watch }
				}
				independence_wave_iw025_vojvodina_fail_watch_mission = yes
			}
			independence_wave_iw025_vojvodina_update_watch_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw025_vojvodina_contested_danube_administration
							has_idea = independence_wave_iw025_vojvodina_coordinated_grain_authority
							has_idea = independence_wave_iw025_vojvodina_municipal_minority_charter
							has_idea = independence_wave_iw025_vojvodina_federal_frontier_compact
						}
					}
				}
				independence_wave_iw025_vojvodina_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw025_vojvodina_overlay_active }
			independence_wave_iw025_vojvodina_suspend_overlay = yes
		}
		independence_wave_iw025_vojvodina_pause_watch_mission = yes
	}
}
