# ============================================================================
# CHAOS REDUX - UNIVERSAL DISCOUNTED-COST EFFECTS
# ============================================================================
# This file is the neutral source/transaction adapter layer used by events,
# decisions, missions, focuses, and scripted GUI consumers. It never decides
# whether a purchase is eligible and never replaces an owner's ordinary cost
# provider. Owners pass the ordinary current payable cost, call the quote at
# display and confirmation time, and record the amount actually paid only after
# their payment provider succeeds.
#
# Registry storage is global and uses four aligned arrays:
#   global.universal_cost_source_ids
#   global.universal_cost_source_ratios
#   global.universal_cost_source_priorities
#   global.universal_cost_source_eligibility_masks
#
# Transaction storage is payer-country scoped and uses regular aligned arrays.
# All public inputs and outputs are temporary variables unless explicitly
# written to one of those arrays by the transaction effects.
# ============================================================================

# ---------------------------------------------------------------------------
# Source registry
# ---------------------------------------------------------------------------

# Inputs: universal_cost_source_id, universal_cost_source_ratio,
# universal_cost_source_priority, and universal_cost_source_eligibility_mask.
# Ratios use the universal basis and are composed in ascending priority order.
# A source id is active while its row exists; clearing its row is its expiry.
universal_cost_source_register = {
	set_temp_variable = { universal_cost_source_register_result = constant:universal_cost_framework.result_no_op }
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_id
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_source_ratio
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_source_ratio
				value = constant:universal_cost_framework.basis
				compare = less_than_or_equals
			}
			check_variable = {
				var = universal_cost_source_priority
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_source_priority
				value = constant:universal_cost_framework.max_source_priority
				compare = less_than
			}
			check_variable = {
				var = universal_cost_source_eligibility_mask
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_source_eligibility_mask
				value = constant:universal_cost_framework.max_eligibility_mask_exclusive
				compare = less_than
			}
		}
		set_temp_variable = { universal_cost_source_found_index = constant:universal_cost_framework.invalid_index }
		set_temp_variable = { universal_cost_source_cursor = constant:universal_cost_framework.zero }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = universal_cost_source_cursor
					value = global.universal_cost_source_ids^num
					compare = less_than
				}
				check_variable = {
					var = universal_cost_source_found_index
					value = constant:universal_cost_framework.invalid_index
					compare = equals
				}
			}
			if = {
				limit = {
					check_variable = {
						var = global.universal_cost_source_ids^universal_cost_source_cursor
						value = universal_cost_source_id
						compare = equals
					}
				}
				set_temp_variable = { universal_cost_source_found_index = universal_cost_source_cursor }
			}
			add_to_temp_variable = {
				universal_cost_source_cursor = constant:universal_cost_framework.one
			}
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_source_found_index
					value = constant:universal_cost_framework.invalid_index
					compare = greater_than
				}
			}
			set_variable = {
				global.universal_cost_source_ratios^universal_cost_source_found_index = universal_cost_source_ratio
			}
			set_variable = {
				global.universal_cost_source_priorities^universal_cost_source_found_index = universal_cost_source_priority
			}
			set_variable = {
				global.universal_cost_source_eligibility_masks^universal_cost_source_found_index = universal_cost_source_eligibility_mask
			}
		}
		else = {
			add_to_array = {
				array = global.universal_cost_source_ids
				value = universal_cost_source_id
			}
			add_to_array = {
				array = global.universal_cost_source_ratios
				value = universal_cost_source_ratio
			}
			add_to_array = {
				array = global.universal_cost_source_priorities
				value = universal_cost_source_priority
			}
			add_to_array = {
				array = global.universal_cost_source_eligibility_masks
				value = universal_cost_source_eligibility_mask
			}
		}
		set_temp_variable = { universal_cost_source_register_result = constant:universal_cost_framework.result_success }
	}
}

# Input: universal_cost_source_id. Removes only the matching source row and
# leaves all payer transaction receipts intact.
universal_cost_source_clear = {
	set_temp_variable = { universal_cost_source_clear_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_source_found_index = constant:universal_cost_framework.invalid_index }
	set_temp_variable = { universal_cost_source_cursor = constant:universal_cost_framework.zero }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = universal_cost_source_cursor
				value = global.universal_cost_source_ids^num
				compare = less_than
			}
			check_variable = {
				var = universal_cost_source_found_index
				value = constant:universal_cost_framework.invalid_index
				compare = equals
			}
		}
		if = {
			limit = {
				check_variable = {
					var = global.universal_cost_source_ids^universal_cost_source_cursor
					value = universal_cost_source_id
					compare = equals
				}
			}
			set_temp_variable = { universal_cost_source_found_index = universal_cost_source_cursor }
		}
		add_to_temp_variable = {
			universal_cost_source_cursor = constant:universal_cost_framework.one
		}
	}
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_found_index
				value = constant:universal_cost_framework.invalid_index
				compare = greater_than
			}
		}
		remove_from_array = {
			array = global.universal_cost_source_ids
			index = universal_cost_source_found_index
		}
		remove_from_array = {
			array = global.universal_cost_source_ratios
			index = universal_cost_source_found_index
		}
		remove_from_array = {
			array = global.universal_cost_source_priorities
			index = universal_cost_source_found_index
		}
		remove_from_array = {
			array = global.universal_cost_source_eligibility_masks
			index = universal_cost_source_found_index
		}
		set_temp_variable = { universal_cost_source_clear_result = constant:universal_cost_framework.result_success }
	}
}

# Inputs: universal_cost_source_mask and universal_cost_quote_family_mask.
# Output: universal_cost_source_matches_quote. A zero source mask matches all
# families. Non-zero masks use the low thirty bits and are matched with
# division/modulo because no generic Clausewitz bitwise effect is assumed.
universal_cost_source_matches_quote_family = {
	set_temp_variable = { universal_cost_source_matches_quote = constant:universal_cost_framework.zero }
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_mask
				value = constant:universal_cost_framework.source_all_mask
				compare = equals
			}
			check_variable = {
				var = universal_cost_quote_family_mask
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
		}
		set_temp_variable = { universal_cost_source_matches_quote = constant:universal_cost_framework.one }
	}
	else = {
		set_temp_variable = { universal_cost_source_mask_bit = constant:universal_cost_framework.one }
		set_temp_variable = { universal_cost_source_mask_break = constant:universal_cost_framework.zero }
		for_loop_effect = {
			start = constant:universal_cost_framework.zero
			end = constant:universal_cost_framework.eligibility_bit_count
			value = universal_cost_source_mask_bit_index
			break = universal_cost_source_mask_break
			set_temp_variable = {
				universal_cost_source_mask_bit_presence = {
					value = universal_cost_source_mask
					divide = universal_cost_source_mask_bit
					mod = constant:universal_cost_framework.two
				}
			}
			set_temp_variable = {
				universal_cost_quote_mask_bit_presence = {
					value = universal_cost_quote_family_mask
					divide = universal_cost_source_mask_bit
					mod = constant:universal_cost_framework.two
				}
			}
			if = {
				limit = {
					check_variable = {
						var = universal_cost_source_mask_bit_presence
						value = constant:universal_cost_framework.zero
						compare = greater_than
					}
					check_variable = {
						var = universal_cost_quote_mask_bit_presence
						value = constant:universal_cost_framework.zero
						compare = greater_than
					}
				}
				set_temp_variable = { universal_cost_source_matches_quote = constant:universal_cost_framework.one }
				set_temp_variable = { universal_cost_source_mask_break = constant:universal_cost_framework.one }
			}
			multiply_temp_variable = {
				universal_cost_source_mask_bit = constant:universal_cost_framework.two
			}
		}
	}
}

# Input: universal_cost_quantize_amount and optional
# universal_cost_quantize_quantum. Output: universal_cost_quantize_result and
# universal_cost_quantize_units. Positive values are rounded upward to the
# next quantum; zero and negative values are preserved exactly.
universal_cost_round_up_to_quantum = {
	set_temp_variable = { universal_cost_quantize_result = universal_cost_quantize_amount }
	set_temp_variable = { universal_cost_quantize_units = constant:universal_cost_framework.zero }
	if = {
		limit = {
			check_variable = {
				var = universal_cost_quantize_amount
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_quantize_quantum
					value = constant:universal_cost_framework.zero
					compare = less_than_or_equals
				}
			}
			set_temp_variable = {
				universal_cost_quantize_quantum = constant:universal_cost_framework.default_rounding_quantum
			}
		}
		set_temp_variable = { universal_cost_quantize_raw_units = universal_cost_quantize_amount }
		divide_temp_variable = { universal_cost_quantize_raw_units = universal_cost_quantize_quantum }
		set_temp_variable = { universal_cost_quantize_units = universal_cost_quantize_raw_units }
		round_temp_variable = universal_cost_quantize_units
		if = {
			limit = {
				check_variable = {
					var = universal_cost_quantize_units
					value = universal_cost_quantize_raw_units
					compare = less_than
				}
			}
			add_to_temp_variable = {
				universal_cost_quantize_units = constant:universal_cost_framework.one
			}
		}
		set_temp_variable = { universal_cost_quantize_result = universal_cost_quantize_units }
		multiply_temp_variable = {
			universal_cost_quantize_result = universal_cost_quantize_quantum
		}
	}
}

# Inputs: universal_cost_quote_ordinary_cost, universal_cost_quote_family_mask,
# and universal_cost_quote_rounding_quantum. Output: final/displayed/saved
# quote fields, composed ratio, primary source id, and source count.
# Ordinary cost must already include the consumer's ordinary current modifiers.
universal_cost_quote_integer = {
	set_temp_variable = { universal_cost_quote_final_cost = universal_cost_quote_ordinary_cost }
	set_temp_variable = { universal_cost_quote_displayed_cost = universal_cost_quote_ordinary_cost }
	set_temp_variable = { universal_cost_quote_saved_cost = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quote_composed_ratio = constant:universal_cost_framework.basis }
	set_temp_variable = { universal_cost_quote_source_count = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quote_primary_source_id = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quote_valid = constant:universal_cost_framework.one }

	if = {
		limit = {
			check_variable = {
				var = universal_cost_quote_ordinary_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_quote_family_mask
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
		}
		clear_temp_array = universal_cost_quote_priority_work
		clear_temp_array = universal_cost_quote_index_work
		for_each_loop = {
			array = global.universal_cost_source_priorities
			value = universal_cost_quote_priority_entry
			index = universal_cost_quote_registry_index_entry
			add_to_temp_array = {
				array = universal_cost_quote_priority_work
				value = universal_cost_quote_priority_entry
			}
			add_to_temp_array = {
				array = universal_cost_quote_index_work
				value = universal_cost_quote_registry_index_entry
			}
		}
		set_temp_variable = {
			universal_cost_quote_sources_remaining = global.universal_cost_source_priorities^num
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					var = universal_cost_quote_sources_remaining
					value = constant:universal_cost_framework.zero
					compare = greater_than
				}
			}
			find_lowest_in_array = {
				array = universal_cost_quote_priority_work
				value = universal_cost_quote_selected_priority
				index = universal_cost_quote_selected_slot
			}
			set_temp_variable = {
				universal_cost_quote_registry_index = universal_cost_quote_index_work^universal_cost_quote_selected_slot
			}
			set_temp_variable = {
				universal_cost_source_id = global.universal_cost_source_ids^universal_cost_quote_registry_index
			}
			set_temp_variable = {
				universal_cost_source_ratio = global.universal_cost_source_ratios^universal_cost_quote_registry_index
			}
			set_temp_variable = {
				universal_cost_source_mask = global.universal_cost_source_eligibility_masks^universal_cost_quote_registry_index
			}
			remove_from_temp_array = {
				array = universal_cost_quote_priority_work
				index = universal_cost_quote_selected_slot
			}
			remove_from_temp_array = {
				array = universal_cost_quote_index_work
				index = universal_cost_quote_selected_slot
			}
			if = {
				limit = {
					check_variable = {
						var = universal_cost_source_id
						value = constant:universal_cost_framework.zero
						compare = greater_than
					}
					check_variable = {
						var = universal_cost_source_ratio
						value = constant:universal_cost_framework.zero
						compare = greater_than
					}
					check_variable = {
						var = universal_cost_source_ratio
						value = constant:universal_cost_framework.basis
						compare = less_than_or_equals
					}
				}
				universal_cost_source_matches_quote_family = yes
				if = {
					limit = {
						check_variable = {
							var = universal_cost_source_matches_quote
							value = constant:universal_cost_framework.one
							compare = equals
						}
					}
					multiply_temp_variable = {
						universal_cost_quote_composed_ratio = universal_cost_source_ratio
					}
					divide_temp_variable = {
						universal_cost_quote_composed_ratio = constant:universal_cost_framework.basis
					}
					if = {
						limit = {
							check_variable = {
								var = universal_cost_quote_source_count
								value = constant:universal_cost_framework.zero
								compare = equals
							}
						}
						set_temp_variable = {
							universal_cost_quote_primary_source_id = universal_cost_source_id
						}
					}
					add_to_temp_variable = {
						universal_cost_quote_source_count = constant:universal_cost_framework.one
					}
				}
			}
			subtract_from_temp_variable = {
				universal_cost_quote_sources_remaining = constant:universal_cost_framework.one
			}
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_quote_source_count
					value = constant:universal_cost_framework.zero
					compare = greater_than
				}
			}
			set_temp_variable = { universal_cost_quantize_amount = universal_cost_quote_ordinary_cost }
			multiply_temp_variable = {
				universal_cost_quantize_amount = universal_cost_quote_composed_ratio
			}
			divide_temp_variable = {
				universal_cost_quantize_amount = constant:universal_cost_framework.basis
			}
			set_temp_variable = { universal_cost_quantize_quantum = universal_cost_quote_rounding_quantum }
			universal_cost_round_up_to_quantum = yes
			set_temp_variable = { universal_cost_quote_final_cost = universal_cost_quantize_result }
		}
	}

	set_temp_variable = { universal_cost_quote_displayed_cost = universal_cost_quote_final_cost }
	if = {
		limit = {
			check_variable = {
				var = universal_cost_quote_ordinary_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
		}
		set_temp_variable = { universal_cost_quote_saved_cost = universal_cost_quote_ordinary_cost }
		subtract_from_temp_variable = {
			universal_cost_quote_saved_cost = universal_cost_quote_final_cost
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_quote_saved_cost
					value = constant:universal_cost_framework.zero
					compare = less_than
				}
			}
			set_temp_variable = { universal_cost_quote_saved_cost = constant:universal_cost_framework.zero }
		}
	}
}

# Convert a quoted component into an affordability result without changing
# resources. Inputs: universal_cost_quote_resource_kind and
# universal_cost_quote_final_cost. Output: universal_cost_quote_affordable.
universal_cost_check_quote_affordable = {
	set_temp_variable = { universal_cost_quote_affordable = constant:universal_cost_framework.result_no_op }
	set_temp_variable = {
		universal_cost_affordability_resource_kind = universal_cost_quote_resource_kind
	}
	set_temp_variable = {
		universal_cost_affordability_amount = universal_cost_quote_final_cost
	}
	if = {
		limit = { universal_cost_component_is_affordable = yes }
		set_temp_variable = { universal_cost_quote_affordable = constant:universal_cost_framework.result_success }
	}
}

# Inputs: universal_cost_payment_resource_kind and positive
# universal_cost_payment_amount. Output: universal_cost_payment_result.
# Unsupported kinds return no_op and do not mutate the payer.
universal_cost_pay_component = {
	set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = {
		universal_cost_affordability_resource_kind = universal_cost_payment_resource_kind
	}
	set_temp_variable = {
		universal_cost_affordability_amount = universal_cost_payment_amount
	}
	if = {
		limit = {
			check_variable = {
				var = universal_cost_payment_amount
				value = constant:universal_cost_framework.zero
				compare = less_than_or_equals
			}
		}
		set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_success }
	}
	else = {
		if = {
			limit = {
				check_variable = {
					var = universal_cost_payment_resource_kind
					value = constant:universal_cost_resource_kind.political_power
					compare = equals
				}
				universal_cost_component_is_affordable = yes
			}
			set_temp_variable = { universal_cost_payment_delta = universal_cost_payment_amount }
			multiply_temp_variable = {
				universal_cost_payment_delta = constant:universal_cost_framework.negative_one
			}
			add_political_power = universal_cost_payment_delta
			set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_success }
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_payment_resource_kind
					value = constant:universal_cost_resource_kind.command_power
					compare = equals
				}
				universal_cost_component_is_affordable = yes
			}
			set_temp_variable = { universal_cost_payment_delta = universal_cost_payment_amount }
			multiply_temp_variable = {
				universal_cost_payment_delta = constant:universal_cost_framework.negative_one
			}
			add_command_power = universal_cost_payment_delta
			set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_success }
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_payment_resource_kind
					value = constant:universal_cost_resource_kind.manpower
					compare = equals
				}
				universal_cost_component_is_affordable = yes
			}
			set_temp_variable = { universal_cost_payment_delta = universal_cost_payment_amount }
			multiply_temp_variable = {
				universal_cost_payment_delta = constant:universal_cost_framework.negative_one
			}
			add_manpower = universal_cost_payment_delta
			set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_success }
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_payment_resource_kind
					value = constant:universal_cost_resource_kind.fuel
					compare = equals
				}
				universal_cost_component_is_affordable = yes
			}
			set_temp_variable = { universal_cost_payment_delta = universal_cost_payment_amount }
			multiply_temp_variable = {
				universal_cost_payment_delta = constant:universal_cost_framework.negative_one
			}
			add_fuel = universal_cost_payment_delta
			set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_success }
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_payment_resource_kind
					value = constant:universal_cost_resource_kind.infantry_equipment
					compare = equals
				}
				universal_cost_component_is_affordable = yes
			}
			set_temp_variable = { universal_cost_payment_delta = universal_cost_payment_amount }
			multiply_temp_variable = {
				universal_cost_payment_delta = constant:universal_cost_framework.negative_one
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = universal_cost_payment_delta
			}
			set_temp_variable = { universal_cost_payment_result = constant:universal_cost_framework.result_success }
		}
		if = {

# Inputs: universal_cost_transaction_lookup_id. Output:
# universal_cost_transaction_found_index. The lookup is payer-country scoped.
universal_cost_find_transaction_row = {
	set_temp_variable = { universal_cost_transaction_found_index = constant:universal_cost_framework.invalid_index }
	set_temp_variable = { universal_cost_transaction_cursor = constant:universal_cost_framework.zero }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = universal_cost_transaction_cursor
				value = universal_cost_transaction_ids^num
				compare = less_than
			}
			check_variable = {
				var = universal_cost_transaction_found_index
				value = constant:universal_cost_framework.invalid_index
				compare = equals
			}
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_transaction_ids^universal_cost_transaction_cursor
					value = universal_cost_transaction_lookup_id
					compare = equals
				}
			}
			set_temp_variable = {
				universal_cost_transaction_found_index = universal_cost_transaction_cursor
			}
		}
		add_to_temp_variable = {
			universal_cost_transaction_cursor = constant:universal_cost_framework.one
		}
	}
}

		limit = {

				var = universal_cost_transaction_id

			}
			check_variable = {
				var = universal_cost_transaction_component_id
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}

			}
		}
		set_temp_variable = {
			universal_cost_transaction_lookup_id = universal_cost_transaction_id
		}
		universal_cost_find_transaction_row = yes
		set_temp_variable = { universal_cost_record_allowed = constant:universal_cost_framework.zero }
		if = {
			limit = {
				check_variable = {
					var = universal_cost_transaction_found_index
					value = constant:universal_cost_framework.invalid_index
					compare = equals
				}
			}
			add_to_array = {
				array = universal_cost_transaction_ids

			}

			}

				universal_cost_transaction_found_index = constant:universal_cost_framework.one
			}
			set_temp_variable = { universal_cost_record_allowed = constant:universal_cost_framework.one }
		}
		else = {
			if = {
				limit = {
					check_variable = {
						var = universal_cost_transaction_states^universal_cost_transaction_found_index
						value = constant:universal_cost_framework.transaction_state_recorded
						compare = equals
					}

					universal_cost_component_lookup_id = universal_cost_transaction_component_id

				check_variable = {
					var = universal_cost_record_allowed
					value = constant:universal_cost_framework.one
					compare = equals
				}
			}
			add_to_array = {
				array = universal_cost_component_transaction_ids
				value = universal_cost_transaction_id
			}
			add_to_array = {

			}

			}
			add_to_array = {
				array = universal_cost_component_resource_ids
				value = universal_cost_transaction_resource_id
			}
			add_to_array = {
				array = universal_cost_component_ordinary_costs

				array = universal_cost_component_quoted_costs

			}
			add_to_array = {
				array = universal_cost_component_actual_paid_costs

			}
			add_to_array = {
				array = universal_cost_component_source_ids
				value = universal_cost_transaction_source_id
			}

				array = universal_cost_component_refunded_states

			}
			set_temp_variable = {
				universal_cost_recorded_component_index = universal_cost_component_transaction_ids^num

# Input: universal_cost_transaction_id. Refunds every still-unrefunded
# component from the stored actual-paid ledger amount. Native resource kinds
# are credited here; unsupported owner-adapter kinds remain pending and return
# result_partial. A successful refund changes the receipt state so a second
# call is a no-op.
universal_cost_refund_transaction = {
	set_temp_variable = { universal_cost_refund_transaction_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_refund_total = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_refund_unhandled_total = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_lookup_id = universal_cost_transaction_id }
	universal_cost_find_transaction_row = yes
	if = {
		limit = {
			check_variable = {
				var = universal_cost_transaction_found_index
				value = constant:universal_cost_framework.invalid_index
				compare = greater_than
			}
			OR = {
				check_variable = {
					var = universal_cost_transaction_states^universal_cost_transaction_found_index
					value = constant:universal_cost_framework.transaction_state_recorded
					compare = equals
				}
				check_variable = {
					var = universal_cost_transaction_states^universal_cost_transaction_found_index
					value = constant:universal_cost_framework.transaction_state_refund_partial
					compare = equals
				}
			}
		}
		set_temp_variable = { universal_cost_refund_cursor = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_refund_all_complete = constant:universal_cost_framework.one }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = universal_cost_refund_cursor
					value = universal_cost_component_transaction_ids^num
					compare = less_than
				}
			}
			if = {
				limit = {
					check_variable = {
						var = universal_cost_component_transaction_ids^universal_cost_refund_cursor
						value = universal_cost_transaction_id
						compare = equals
					}
					check_variable = {
						var = universal_cost_component_refunded_states^universal_cost_refund_cursor
						value = constant:universal_cost_component_state.not_refunded
						compare = equals
					}
				}
				set_temp_variable = {
					universal_cost_payment_resource_kind = universal_cost_component_resource_kinds^universal_cost_refund_cursor
				}
				set_temp_variable = {
					universal_cost_payment_amount = universal_cost_component_actual_paid_costs^universal_cost_refund_cursor
				}
				if = {
					limit = {

			}
		}
