# ============================================================================
# CHAOS REDUX - UNIVERSAL DISCOUNTED-COST EFFECTS
# ============================================================================
# This file is the neutral source/transaction adapter layer used by events,
# decisions, missions, focuses, and scripted GUI consumers. It never decides
# whether a purchase is eligible and never replaces an owner's ordinary cost
# provider. Owners pass the ordinary current payable cost, call the quote at
# display and confirmation time, and record the amount actually paid only after
# their payment provider succeeds.
#
# Registry storage is global and uses four aligned arrays:
#   global.universal_cost_source_ids
#   global.universal_cost_source_ratios
#   global.universal_cost_source_priorities
#   global.universal_cost_source_eligibility_masks
#
# Transaction storage is payer-country scoped and uses regular aligned arrays.
# All public inputs and outputs are temporary variables unless explicitly
# written to one of those arrays by the transaction effects.
# ============================================================================

# ---------------------------------------------------------------------------
# Source registry
# ---------------------------------------------------------------------------

# Inputs: universal_cost_source_id, universal_cost_source_ratio,
# universal_cost_source_priority, and universal_cost_source_eligibility_mask.
# Ratios use the universal basis. Priority selects the primary source; exact
# multiplication is commutative, with equal-priority identity in registry order.
# A source id is active while its row exists; clearing its row is its expiry.
universal_cost_source_register = {
	set_temp_variable = { universal_cost_source_register_result = constant:universal_cost_framework.result_no_op }
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_id
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_source_ratio
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_source_ratio
				value = constant:universal_cost_framework.basis
				compare = less_than_or_equals
			}
			check_variable = {
				var = universal_cost_source_priority
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_source_priority
				value = constant:universal_cost_framework.max_source_priority
				compare = less_than
			}
			check_variable = {
				var = universal_cost_source_eligibility_mask
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_source_eligibility_mask
				value = constant:universal_cost_framework.max_eligibility_mask_exclusive
				compare = less_than
			}
		}
		set_temp_variable = { universal_cost_source_found_index = constant:universal_cost_framework.invalid_index }
		if = {
			limit = { universal_cost_exact_find_source = yes }
			set_temp_variable = { universal_cost_source_found_index = universal_cost_source_found_index }
		}
		if = {
			limit = {
				check_variable = {
					var = universal_cost_source_found_index
					value = constant:universal_cost_framework.invalid_index
					compare = greater_than
				}
			}
			set_variable = {
				global.universal_cost_source_ratios^universal_cost_source_found_index = universal_cost_source_ratio
			}
			set_variable = {
				global.universal_cost_source_priorities^universal_cost_source_found_index = universal_cost_source_priority
			}
			set_variable = {
				global.universal_cost_source_eligibility_masks^universal_cost_source_found_index = universal_cost_source_eligibility_mask
			}
		}
		else = {
			add_to_array = {
				array = global.universal_cost_source_ids
				value = universal_cost_source_id
			}
			add_to_array = {
				array = global.universal_cost_source_ratios
				value = universal_cost_source_ratio
			}
			add_to_array = {
				array = global.universal_cost_source_priorities
				value = universal_cost_source_priority
			}
			add_to_array = {
				array = global.universal_cost_source_eligibility_masks
				value = universal_cost_source_eligibility_mask
			}
		}
		set_temp_variable = { universal_cost_source_register_result = constant:universal_cost_framework.result_success }
		if = {
			limit = {
				is_in_array = {
					array = global.universal_cost_source_ids
					value = constant:universal_cost_framework.black_friday_source_id
				}
			}
			universal_cost_rebuild_custom_quote_ratio_cache = yes
		}
	}
}

# Input: universal_cost_source_id. Removes only the matching source row and
# leaves all payer transaction receipts intact.
universal_cost_source_clear = {
	set_temp_variable = { universal_cost_source_clear_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_source_found_index = constant:universal_cost_framework.invalid_index }
	if = {
		limit = { universal_cost_exact_find_source = yes }
		set_temp_variable = { universal_cost_source_found_index = universal_cost_source_found_index }
	}
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_found_index
				value = constant:universal_cost_framework.invalid_index
				compare = greater_than
			}
		}
		remove_from_array = {
			array = global.universal_cost_source_ids
			index = universal_cost_source_found_index
		}
		remove_from_array = {
			array = global.universal_cost_source_ratios
			index = universal_cost_source_found_index
		}
		remove_from_array = {
			array = global.universal_cost_source_priorities
			index = universal_cost_source_found_index
		}
		remove_from_array = {
			array = global.universal_cost_source_eligibility_masks
			index = universal_cost_source_found_index
		}
		set_temp_variable = { universal_cost_source_clear_result = constant:universal_cost_framework.result_success }
		if = {
			limit = {
				is_in_array = {
					array = global.universal_cost_source_ids
					value = constant:universal_cost_framework.black_friday_source_id
				}
			}
			universal_cost_rebuild_custom_quote_ratio_cache = yes
		}
		else = {
			universal_cost_clear_custom_quote_ratio_cache = yes
		}
	}
}

# Stable owner-facing source API for Event 026. A caller may supply
# universal_cost_source_ratio and universal_cost_source_eligibility_mask; an
# omitted/zero ratio uses the baseline sale ratio and a zero mask means all
# families. The event owner still controls when this effect is called.
universal_cost_source_register_black_friday = {
	set_temp_variable = {
		universal_cost_source_id = constant:universal_cost_framework.black_friday_source_id
	}
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_ratio
				value = constant:universal_cost_framework.zero
				compare = less_than_or_equals
			}
		}
		set_temp_variable = {
			universal_cost_source_ratio = constant:universal_cost_framework.black_friday_default_ratio
		}
	}
	if = {
		limit = {
			check_variable = {
				var = universal_cost_source_priority
				value = constant:universal_cost_framework.zero
				compare = less_than_or_equals
			}
		}
		set_temp_variable = {
			universal_cost_source_priority = constant:universal_cost_framework.black_friday_source_priority
		}
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = universal_cost_source_eligibility_mask }
				check_variable = {
					var = universal_cost_source_eligibility_mask
					value = constant:universal_cost_framework.zero
					compare = less_than_or_equals
				}
			}
		}
	set_temp_variable = {
			universal_cost_source_eligibility_mask = constant:universal_cost_framework.source_all_mask
	}
	}
	set_temp_variable = { universal_cost_source_register_result = constant:universal_cost_framework.result_no_op }
	universal_cost_source_register = yes
}

# Stable owner-facing expiry API for Event 026.
universal_cost_source_clear_black_friday = {
	set_temp_variable = {
		universal_cost_source_id = constant:universal_cost_framework.black_friday_source_id
	}
	set_temp_variable = { universal_cost_source_clear_result = constant:universal_cost_framework.result_no_op }
	universal_cost_source_clear = yes
}

# Inputs: universal_cost_source_mask and universal_cost_quote_family_mask.
# Output: universal_cost_source_matches_quote. A zero source mask matches all
# families. Non-zero masks use the low thirty bits and are matched with
# division/modulo because no generic Clausewitz bitwise effect is assumed.
universal_cost_source_matches_quote_family = {
	set_temp_variable = { universal_cost_source_matches_quote = constant:universal_cost_framework.zero }
	if = {
		limit = {
			check_variable = { universal_cost_quote_family_mask > constant:universal_cost_framework.zero }
			universal_cost_exact_match_family = yes
		}
		set_temp_variable = { universal_cost_source_matches_quote = universal_cost_source_matches_quote }
	}
}

# Input: universal_cost_quantize_amount and optional
# universal_cost_quantize_quantum. Output: universal_cost_quantize_result and
# universal_cost_quantize_units. Positive values are rounded upward to the
# next quantum; zero and negative values are preserved exactly.
universal_cost_round_up_to_quantum = {
	set_temp_variable = { universal_cost_quantize_result = universal_cost_quantize_amount }
	set_temp_variable = { universal_cost_quantize_valid = constant:universal_cost_framework.one }
	set_temp_variable = { universal_cost_quantize_units = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quantize_units_valid = constant:universal_cost_framework.one }
	clear_temp_array = uc_exact_units
	resize_temp_array = { array = uc_exact_units size = constant:universal_cost_framework.one }
	if = {
		limit = { NOT = { has_variable = universal_cost_quantize_quantum } }
		set_temp_variable = { universal_cost_quantize_quantum = constant:universal_cost_framework.zero }
	}
	if = {
		limit = { universal_cost_exact_quantize_scalar = yes }
		set_temp_variable = { universal_cost_quantize_result = universal_cost_quantize_result }
	}
}

# Inputs: universal_cost_quote_ordinary_cost, universal_cost_quote_family_mask,
# and universal_cost_quote_rounding_quantum. Output: final/displayed/saved
# quote fields, composed ratio, primary source id, and source count.
# Ordinary cost must already include the consumer's ordinary current modifiers.
universal_cost_quote_integer = {
	# Initialize outputs in the effect chain before the nested producer predicate.
	set_temp_variable = { universal_cost_quote_final_cost = universal_cost_quote_ordinary_cost }
	set_temp_variable = { universal_cost_quote_displayed_cost = universal_cost_quote_ordinary_cost }
	set_temp_variable = { universal_cost_quote_saved_cost = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quote_composed_ratio = constant:universal_cost_framework.basis }
	set_temp_variable = { universal_cost_quote_source_count = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quote_primary_source_id = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_quote_valid = constant:universal_cost_framework.one }
	if = {
		limit = { NOT = { has_variable = universal_cost_quote_rounding_quantum } }
		set_temp_variable = { universal_cost_quote_rounding_quantum = constant:universal_cost_framework.zero }
	}
	if = {
		limit = { universal_cost_quote_exact = yes }
		set_temp_variable = { universal_cost_quote_displayed_cost = universal_cost_quote_final_cost }
	}
}

# Publish the existing Event 026 readiness marker and retire scalar cache rows.
# Trigger/effect frontends read the same exact aligned source registry directly.
# Clearing readiness preserves other source rows and every payer receipt.
universal_cost_rebuild_custom_quote_ratio_cache = {
	# The aligned source registry retains exact ratios for EVERY family mask.
	# Retire scalar cache rows so no stale scalar can govern a quote.
	universal_cost_clear_custom_quote_ratio_cache = yes
	set_variable = { global.universal_cost_custom_quote_cache_valid = constant:universal_cost_framework.one }
}

universal_cost_clear_custom_quote_ratio_cache = {
	clear_variable = global.universal_cost_custom_quote_ratio_political_power
	clear_variable = global.universal_cost_custom_quote_ratio_command_power
	clear_variable = global.universal_cost_custom_quote_ratio_manpower
	clear_variable = global.universal_cost_custom_quote_ratio_fuel
	clear_variable = global.universal_cost_custom_quote_ratio_equipment
	clear_variable = global.universal_cost_custom_quote_ratio_trains
	clear_variable = global.universal_cost_custom_quote_ratio_convoy
	clear_variable = global.universal_cost_custom_quote_ratio_army_experience
	clear_variable = global.universal_cost_custom_quote_ratio_navy_experience
	clear_variable = global.universal_cost_custom_quote_ratio_air_experience
	clear_variable = global.universal_cost_custom_quote_ratio_stability
	clear_variable = global.universal_cost_custom_quote_ratio_war_support
	set_variable = { global.universal_cost_custom_quote_cache_valid = constant:universal_cost_framework.zero }
}

# Convert a quoted component into an affordability result without changing
# resources. Inputs: universal_cost_quote_resource_kind and
# universal_cost_quote_final_cost. Output: universal_cost_quote_affordable.
universal_cost_check_quote_affordable = {
	set_temp_variable = { universal_cost_quote_affordable = constant:universal_cost_framework.result_no_op }
	set_temp_variable = {
		universal_cost_affordability_resource_kind = universal_cost_quote_resource_kind
	}
	set_temp_variable = {
		universal_cost_affordability_amount = universal_cost_quote_final_cost
	}
	if = {
		limit = {
			check_variable = { universal_cost_quote_valid = constant:universal_cost_framework.one }
			universal_cost_component_is_affordable = yes
		}
		set_temp_variable = { universal_cost_quote_affordable = constant:universal_cost_framework.result_success }
	}
}

# Inputs: universal_cost_transaction_lookup_id. Output:
# universal_cost_transaction_found_index. The lookup is payer-country scoped.
universal_cost_find_transaction_row = {
	set_temp_variable = { universal_cost_transaction_found_index = constant:universal_cost_framework.invalid_index }
	set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_transaction_is_indexed = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_pointer = universal_cost_transaction_row_plus_one_@var:universal_cost_transaction_lookup_id }
	if = {
		limit = { check_variable = { var = universal_cost_transaction_pointer value = constant:universal_cost_framework.zero compare = greater_than } }
		set_temp_variable = { universal_cost_transaction_candidate_index = universal_cost_transaction_pointer }
		subtract_from_temp_variable = { universal_cost_transaction_candidate_index = constant:universal_cost_framework.one }
		if = {
			limit = {
				check_variable = { var = universal_cost_transaction_candidate_index value = universal_cost_transaction_ids^num compare = less_than }
				check_variable = { var = universal_cost_transaction_ids^universal_cost_transaction_candidate_index value = universal_cost_transaction_lookup_id compare = equals }
			}
			set_temp_variable = { universal_cost_transaction_found_index = universal_cost_transaction_candidate_index }
			set_temp_variable = { universal_cost_transaction_is_indexed = constant:universal_cost_framework.one }
			set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_success }
		}
		else = { set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_rejected } }
	}
	else = {
		if = {
			limit = { check_variable = { var = universal_cost_transaction_ids^num value = constant:universal_cost_framework.legacy_scan_max_rows compare = less_than_or_equals } }
			set_temp_variable = { universal_cost_transaction_cursor = constant:universal_cost_framework.zero }
			while_loop_effect = {
				limit = {
					check_variable = { var = universal_cost_transaction_cursor value = universal_cost_transaction_ids^num compare = less_than }
					check_variable = { var = universal_cost_transaction_found_index value = constant:universal_cost_framework.invalid_index compare = equals }
				}
				if = {
					limit = { check_variable = { var = universal_cost_transaction_ids^universal_cost_transaction_cursor value = universal_cost_transaction_lookup_id compare = equals } }
					set_temp_variable = { universal_cost_transaction_found_index = universal_cost_transaction_cursor }
					set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_success }
				}
				add_to_temp_variable = { universal_cost_transaction_cursor = constant:universal_cost_framework.one }
			}
		}
		else = { set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_rejected } }
	}
}

		set_temp_variable = { universal_cost_component_walk_count = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_component_last_pointer = constant:universal_cost_framework.zero }
		while_loop_effect = {
			limit = {
				check_variable = { var = universal_cost_component_pointer value = constant:universal_cost_framework.zero compare = greater_than }
				check_variable = { var = universal_cost_component_walk_count value = constant:universal_cost_framework.max_components_per_transaction compare = less_than }
			}
			set_temp_variable = { universal_cost_component_cursor = universal_cost_component_pointer }
			subtract_from_temp_variable = { universal_cost_component_cursor = constant:universal_cost_framework.one }
			if = {
				limit = {
					check_variable = { var = universal_cost_component_cursor value = universal_cost_component_transaction_ids^num compare = less_than }
					check_variable = { var = universal_cost_component_cursor value = universal_cost_component_next_plus_one^num compare = less_than }
					check_variable = { var = universal_cost_component_transaction_ids^universal_cost_component_cursor value = universal_cost_component_lookup_transaction_id compare = equals }
				}
				if = {
					limit = {
						check_variable = { var = universal_cost_component_ids^universal_cost_component_cursor value = universal_cost_component_lookup_id compare = equals }
					}
					if = {
						limit = { check_variable = { var = universal_cost_component_found_index value = constant:universal_cost_framework.invalid_index compare = equals } }
						set_temp_variable = { universal_cost_component_found_index = universal_cost_component_cursor }
						set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_success }
					}
					else = { set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_rejected } }
				}
				set_temp_variable = { universal_cost_component_last_pointer = universal_cost_component_pointer }
				set_temp_variable = { universal_cost_component_pointer = universal_cost_component_next_plus_one^universal_cost_component_cursor }
			}
			else = {
				set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_rejected }
				set_temp_variable = { universal_cost_component_found_index = constant:universal_cost_framework.invalid_index }
				set_temp_variable = { universal_cost_component_pointer = constant:universal_cost_framework.zero }
			}
			add_to_temp_variable = { universal_cost_component_walk_count = constant:universal_cost_framework.one }
		}
		if = {
			limit = { check_variable = { var = universal_cost_component_pointer value = constant:universal_cost_framework.zero compare = greater_than } }
			set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_rejected }
			set_temp_variable = { universal_cost_component_found_index = constant:universal_cost_framework.invalid_index }
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = universal_cost_component_walk_count value = universal_cost_transaction_component_counts^universal_cost_transaction_found_index compare = not_equals }
					check_variable = { var = universal_cost_component_last_pointer value = universal_cost_transaction_component_tail_plus_one^universal_cost_transaction_found_index compare = not_equals }
				}
			}
			set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_rejected }
			set_temp_variable = { universal_cost_component_found_index = constant:universal_cost_framework.invalid_index }
		}
		if = {
			limit = { check_variable = { var = universal_cost_component_lookup_result value = constant:universal_cost_framework.result_rejected compare = equals } }
			set_temp_variable = { universal_cost_component_found_index = constant:universal_cost_framework.invalid_index }
		}
	}
	else_if = {
		limit = {
			check_variable = { var = universal_cost_transaction_lookup_result value = constant:universal_cost_framework.result_success compare = equals }
			check_variable = { var = universal_cost_component_transaction_ids^num value = constant:universal_cost_framework.legacy_scan_max_rows compare = less_than_or_equals }
		}
		set_temp_variable = { universal_cost_component_cursor = constant:universal_cost_framework.zero }
		while_loop_effect = {
			limit = {
				check_variable = { var = universal_cost_component_cursor value = universal_cost_component_transaction_ids^num compare = less_than }
				check_variable = { var = universal_cost_component_found_index value = constant:universal_cost_framework.invalid_index compare = equals }
			}
			if = {
				limit = {
					check_variable = { var = universal_cost_component_transaction_ids^universal_cost_component_cursor value = universal_cost_component_lookup_transaction_id compare = equals }
					check_variable = { var = universal_cost_component_ids^universal_cost_component_cursor value = universal_cost_component_lookup_id compare = equals }
				}
				set_temp_variable = { universal_cost_component_found_index = universal_cost_component_cursor }
				set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_success }
			}
			add_to_temp_variable = { universal_cost_component_cursor = constant:universal_cost_framework.one }
		}
	}
	else = { set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_rejected } }
}

# Lazy payer-local schema extension. Existing values retain their indices;
# legacy rows receive UNKNOWN proof metadata, never an inferred logical family.
# Scope: payer. Inputs: existing receipt row counts. Outputs: aligned normal
# metadata arrays. Defaults: zero/unknown for missing suffix entries. Persistent
# side effects: preserves prefixes and resizes metadata only; no amount/state edit.
universal_cost_prepare_achievement_receipt_metadata = {
	resize_array = { array = universal_cost_transaction_achievement_family_ids size = universal_cost_transaction_ids^num }
	resize_array = { array = universal_cost_transaction_achievement_primary_component_ids size = universal_cost_transaction_ids^num }
	resize_array = { array = universal_cost_transaction_achievement_cost_bases size = universal_cost_transaction_ids^num }

	resize_array = { array = universal_cost_component_achievement_primary_members size = universal_cost_component_ids^num }
}

# Called only after a successful component record. Optional owner metadata is
# independent of resource kind and never changes payment/refund amounts.
# All snapshot inputs and row indices belong to the calling record frame.
# Scope: payer. Inputs: record_metadata_* snapshots and recorded row indices.
# Outputs: normal role metadata only. Defaults: unknown for absent/invalid data.
# Side effects: conflicting continuation invalidates family proof, not the receipt.
universal_cost_bind_achievement_receipt_metadata = {
	universal_cost_prepare_achievement_receipt_metadata = yes
	set_temp_variable = { universal_cost_record_metadata_valid = constant:universal_cost_framework.zero }
	if = {
		limit = {
			check_variable = { universal_cost_record_metadata_family > constant:universal_cost_framework.zero }
			OR = {
				AND = {
					check_variable = { universal_cost_record_metadata_basis = constant:universal_cost_achievement_basis.primary_component }
					check_variable = { universal_cost_record_metadata_primary > constant:universal_cost_framework.zero }
					OR = {
						AND = {
							check_variable = { universal_cost_transaction_component_id = universal_cost_record_metadata_primary }
							check_variable = { universal_cost_record_metadata_member = constant:universal_cost_framework.one }
						}
						AND = {
							check_variable = { var = universal_cost_transaction_component_id value = universal_cost_record_metadata_primary compare = not_equals }
							check_variable = { universal_cost_record_metadata_member = constant:universal_cost_framework.zero }
						}
					}
				}
				AND = {
					check_variable = { universal_cost_record_metadata_basis = constant:universal_cost_achievement_basis.primary_group }
					check_variable = { universal_cost_record_metadata_primary = constant:universal_cost_framework.zero }
					OR = {
						check_variable = { universal_cost_record_metadata_member = constant:universal_cost_framework.zero }
						check_variable = { universal_cost_record_metadata_member = constant:universal_cost_framework.one }
					}
				}
			}
		}
		set_temp_variable = { universal_cost_record_metadata_valid = constant:universal_cost_framework.one }
	}
	if = {
		limit = { check_variable = { universal_cost_transaction_component_counts^universal_cost_transaction_found_index = constant:universal_cost_framework.one } }
		if = {
			limit = { check_variable = { universal_cost_record_metadata_valid = constant:universal_cost_framework.one } }
			set_variable = { universal_cost_transaction_achievement_family_ids^universal_cost_transaction_found_index = universal_cost_record_metadata_family }
			set_variable = { universal_cost_transaction_achievement_primary_component_ids^universal_cost_transaction_found_index = universal_cost_record_metadata_primary }
			set_variable = { universal_cost_transaction_achievement_cost_bases^universal_cost_transaction_found_index = universal_cost_record_metadata_basis }
		}
	}
	if = {
		limit = {
			check_variable = { universal_cost_record_metadata_valid = constant:universal_cost_framework.one }
			check_variable = { universal_cost_transaction_achievement_family_ids^universal_cost_transaction_found_index = universal_cost_record_metadata_family }
			check_variable = { universal_cost_transaction_achievement_primary_component_ids^universal_cost_transaction_found_index = universal_cost_record_metadata_primary }
			check_variable = { universal_cost_transaction_achievement_cost_bases^universal_cost_transaction_found_index = universal_cost_record_metadata_basis }
		}
		set_variable = { universal_cost_component_achievement_family_ids^universal_cost_recorded_component_index = universal_cost_record_metadata_family }
		set_variable = { universal_cost_component_achievement_primary_members^universal_cost_recorded_component_index = universal_cost_record_metadata_member }
	}
	else = {
		# A missing/conflicting continuation can never retroactively type a row.
		set_variable = { universal_cost_transaction_achievement_family_ids^universal_cost_transaction_found_index = constant:universal_cost_framework.zero }
	}
}

# Record one paid component in payer scope after payment succeeds.
# Inputs: transaction/component IDs, resource kind/id, ordinary/quoted/actual-paid
# amounts, source ID, optional legacy primary flag, and four optional explicit
# achievement-role inputs. Actual-paid must equal quoted. Defaults: unknown role
# metadata for omitted tuples; generic/source-zero receipt semantics unchanged.
# Outputs (caller-owned): record_transaction_result, recorded_component_index.
# Persistent side effects: receipt append/totals/links and proof metadata binding.
universal_cost_record_transaction = {
	set_temp_variable = { universal_cost_record_transaction_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_recorded_component_index = constant:universal_cost_framework.invalid_index }
	universal_cost_prepare_achievement_receipt_metadata = yes
	if = { limit = { NOT = { has_variable = universal_cost_transaction_achievement_family_id } } set_temp_variable = { universal_cost_transaction_achievement_family_id = constant:universal_cost_framework.zero } }
	if = { limit = { NOT = { has_variable = universal_cost_transaction_achievement_primary_component_id } } set_temp_variable = { universal_cost_transaction_achievement_primary_component_id = constant:universal_cost_framework.zero } }
	if = { limit = { NOT = { has_variable = universal_cost_transaction_achievement_cost_basis } } set_temp_variable = { universal_cost_transaction_achievement_cost_basis = constant:universal_cost_framework.zero } }
	if = { limit = { NOT = { has_variable = universal_cost_transaction_achievement_primary_member } } set_temp_variable = { universal_cost_transaction_achievement_primary_member = constant:universal_cost_framework.zero } }
	set_temp_variable = { universal_cost_record_metadata_family = universal_cost_transaction_achievement_family_id }
	set_temp_variable = { universal_cost_record_metadata_primary = universal_cost_transaction_achievement_primary_component_id }
	set_temp_variable = { universal_cost_record_metadata_basis = universal_cost_transaction_achievement_cost_basis }
	set_temp_variable = { universal_cost_record_metadata_member = universal_cost_transaction_achievement_primary_member }
	# Consume each optional tuple, requiring explicit forwarding on each record.
	set_temp_variable = { universal_cost_transaction_achievement_family_id = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_achievement_primary_component_id = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_achievement_cost_basis = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_achievement_primary_member = constant:universal_cost_framework.zero }
	if = {
		limit = {
			check_variable = {
				var = universal_cost_transaction_id
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_transaction_component_id
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_transaction_resource_kind
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_transaction_ordinary_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_transaction_quoted_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_transaction_actual_paid_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = universal_cost_transaction_actual_paid_cost
				value = universal_cost_transaction_quoted_cost
				compare = equals
			}
		}
		set_temp_variable = {
			universal_cost_transaction_lookup_id = universal_cost_transaction_id
		}
		# Caller-owned lookup outputs; do not reset unrelated outer cursors.
		set_temp_variable = { universal_cost_transaction_found_index = constant:universal_cost_framework.invalid_index }
		set_temp_variable = { universal_cost_transaction_is_indexed = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_no_op }
		set_temp_variable = { universal_cost_transaction_pointer = constant:universal_cost_framework.zero }
		universal_cost_find_transaction_row = yes
		set_temp_variable = { universal_cost_allocation_proof = universal_cost_allocated_transaction_id_@var:universal_cost_transaction_id }
		set_temp_variable = { universal_cost_record_allowed = constant:universal_cost_framework.zero }
		if = {
			limit = {
				check_variable = {
					var = universal_cost_transaction_found_index
					value = constant:universal_cost_framework.invalid_index
					compare = equals
				}
				check_variable = { var = universal_cost_transaction_pointer value = constant:universal_cost_framework.zero compare = equals }
				check_variable = { var = universal_cost_allocation_proof value = universal_cost_transaction_id compare = equals }
			}
			add_to_array = {
				array = universal_cost_transaction_ids
				value = universal_cost_transaction_id
			}
			add_to_array = {
				array = universal_cost_transaction_states
				value = constant:universal_cost_framework.transaction_state_recorded
			}
			add_to_array = {
				array = universal_cost_transaction_component_counts
				value = constant:universal_cost_framework.zero
			}
			add_to_array = { array = universal_cost_transaction_component_head_plus_one value = constant:universal_cost_framework.zero }
			add_to_array = { array = universal_cost_transaction_component_tail_plus_one value = constant:universal_cost_framework.zero }
			add_to_array = { array = universal_cost_transaction_non_political_paid_component_count value = constant:universal_cost_framework.zero }
			add_to_array = {
				array = universal_cost_transaction_source_ids
				value = universal_cost_transaction_source_id
			}
			add_to_array = {
				array = universal_cost_transaction_primary_component_ids
				value = constant:universal_cost_framework.zero

			add_to_array = {
				array = universal_cost_transaction_total_ordinary_costs

				array = universal_cost_transaction_total_refunded_costs
				value = constant:universal_cost_framework.zero

				universal_cost_transaction_found_index = constant:universal_cost_framework.one
			}
			set_temp_variable = { universal_cost_transaction_pointer = universal_cost_transaction_ids^num }
			set_variable = { universal_cost_transaction_row_plus_one_@var:universal_cost_transaction_id = universal_cost_transaction_pointer }
			clear_variable = universal_cost_allocated_transaction_id_@var:universal_cost_transaction_id
			set_temp_variable = { universal_cost_transaction_is_indexed = constant:universal_cost_framework.one }

		else_if = {
			limit = { check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals } }
			if = {
				limit = {
					check_variable = { var = universal_cost_transaction_component_counts^universal_cost_transaction_found_index value = constant:universal_cost_framework.max_components_per_transaction compare = less_than }
					check_variable = {
						var = universal_cost_transaction_states^universal_cost_transaction_found_index

					}

					universal_cost_component_lookup_id = universal_cost_transaction_component_id

			}
		}
		if = {

					value = constant:universal_cost_framework.one
					compare = equals
				}
			}
			add_to_array = {
				array = universal_cost_component_transaction_ids
				value = universal_cost_transaction_id
			}
			add_to_array = {
				array = universal_cost_component_ids
				value = universal_cost_transaction_component_id
			}
			add_to_array = {
				array = universal_cost_component_resource_kinds
				value = universal_cost_transaction_resource_kind
			}
			add_to_array = {
				array = universal_cost_component_resource_ids

			}

				array = universal_cost_component_ordinary_costs
				value = universal_cost_transaction_ordinary_cost
			}
			add_to_array = {
				array = universal_cost_component_quoted_costs
				value = universal_cost_transaction_quoted_cost
			}
			add_to_array = {
				array = universal_cost_component_actual_paid_costs
				value = universal_cost_transaction_actual_paid_cost
			}
			add_to_array = {
				array = universal_cost_component_source_ids
				value = universal_cost_transaction_source_id
			}

				universal_cost_recorded_component_index = universal_cost_component_transaction_ids^num
			}

			if = {
				limit = { check_variable = { var = universal_cost_old_component_pointer value = constant:universal_cost_framework.zero compare = greater_than } }
				set_temp_variable = { universal_cost_old_component_index = universal_cost_old_component_pointer }
				subtract_from_temp_variable = { universal_cost_old_component_index = constant:universal_cost_framework.one }
				set_variable = { universal_cost_component_next_plus_one^universal_cost_old_component_index = universal_cost_new_component_pointer }
			}
			else = { set_variable = { universal_cost_transaction_component_head_plus_one^universal_cost_transaction_found_index = universal_cost_new_component_pointer } }
			set_variable = { universal_cost_transaction_component_tail_plus_one^universal_cost_transaction_found_index = universal_cost_new_component_pointer }

			set_temp_variable = {
				universal_cost_transaction_current_count = universal_cost_transaction_component_counts^universal_cost_transaction_found_index
			}
			add_to_temp_variable = {
				universal_cost_transaction_current_count = constant:universal_cost_framework.one
			}
			set_variable = {
				universal_cost_transaction_component_counts^universal_cost_transaction_found_index = universal_cost_transaction_current_count
			}
			set_temp_variable = {
				universal_cost_transaction_current_total = universal_cost_transaction_total_ordinary_costs^universal_cost_transaction_found_index
			}
			add_to_temp_variable = {

			}
			set_variable = {
				universal_cost_transaction_total_ordinary_costs^universal_cost_transaction_found_index = universal_cost_transaction_current_total
			}
			set_temp_variable = {

			}
			add_to_temp_variable = {
				universal_cost_transaction_current_total = universal_cost_transaction_quoted_cost
			}

			}
			set_temp_variable = {
				universal_cost_transaction_current_total = universal_cost_transaction_total_paid_costs^universal_cost_transaction_found_index
			}
			add_to_temp_variable = {
				universal_cost_transaction_current_total = universal_cost_transaction_actual_paid_cost
			}
			set_variable = {
				universal_cost_transaction_total_paid_costs^universal_cost_transaction_found_index = universal_cost_transaction_current_total
			}
			if = {
				limit = {
					check_variable = { var = universal_cost_transaction_actual_paid_cost value = constant:universal_cost_framework.zero compare = greater_than }
					NOT = { check_variable = { var = universal_cost_transaction_resource_kind value = constant:universal_cost_resource_kind.political_power compare = equals } }
				}

				add_to_temp_variable = { universal_cost_transaction_current_count = constant:universal_cost_framework.one }
				set_variable = { universal_cost_transaction_non_political_paid_component_count^universal_cost_transaction_found_index = universal_cost_transaction_current_count }
			}
			if = {
				limit = {
					check_variable = {
						var = universal_cost_transaction_primary_component
						value = constant:universal_cost_framework.zero
						compare = greater_than
					}
					check_variable = {
						var = universal_cost_transaction_primary_component_ids^universal_cost_transaction_found_index
						value = constant:universal_cost_framework.zero
						compare = equals
					}
				}
				set_variable = {
					universal_cost_transaction_primary_component_ids^universal_cost_transaction_found_index = universal_cost_transaction_component_id
				}
			}
			universal_cost_bind_achievement_receipt_metadata = yes
			set_temp_variable = { universal_cost_record_transaction_result = constant:universal_cost_framework.result_success }
		}
	}

# Input: universal_cost_transaction_id. Marks an otherwise refundable receipt
# settled after the owning action has succeeded. A settled receipt is retained
# to prevent a later duplicate refund and cannot be appended to.
universal_cost_settle_transaction = {

	if = {

		limit = {
			check_variable = {

			}

}

# Input: universal_cost_component_found_index and universal_cost_transaction_found_index.
# Called only when a positive non-political component first becomes refunded.
universal_cost_decrement_non_political_paid_count = {
	if = {
		limit = {
			check_variable = { var = universal_cost_component_actual_paid_costs^universal_cost_component_found_index value = constant:universal_cost_framework.zero compare = greater_than }
			NOT = { check_variable = { var = universal_cost_component_resource_kinds^universal_cost_component_found_index value = constant:universal_cost_resource_kind.political_power compare = equals } }
			check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals }
			check_variable = { var = universal_cost_transaction_non_political_paid_component_count^universal_cost_transaction_found_index value = constant:universal_cost_framework.zero compare = greater_than }
		}
		set_temp_variable = { universal_cost_transaction_non_political_count = universal_cost_transaction_non_political_paid_component_count^universal_cost_transaction_found_index }
		subtract_from_temp_variable = { universal_cost_transaction_non_political_count = constant:universal_cost_framework.one }
		set_variable = { universal_cost_transaction_non_political_paid_component_count^universal_cost_transaction_found_index = universal_cost_transaction_non_political_count }
	}
}

# Inputs: universal_cost_transaction_id and universal_cost_transaction_component_id. This
# acknowledgement is for an owner adapter that already credited an unsupported
# commitment/component. It uses the stored actual-paid value for the receipt;
# the external adapter must credit exactly that amount before calling it.
universal_cost_mark_component_refunded = {
	set_temp_variable = { universal_cost_mark_component_refunded_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_transaction_lookup_id = universal_cost_transaction_id }
	# Caller-owned lookup outputs; do not reset unrelated outer cursors.
	set_temp_variable = { universal_cost_transaction_found_index = constant:universal_cost_framework.invalid_index }
	set_temp_variable = { universal_cost_transaction_is_indexed = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_transaction_pointer = constant:universal_cost_framework.zero }
	universal_cost_find_transaction_row = yes
	if = {
		limit = { check_variable = { var = universal_cost_transaction_lookup_result value = constant:universal_cost_framework.result_rejected compare = equals } }
		set_temp_variable = { universal_cost_mark_component_refunded_result = constant:universal_cost_framework.result_rejected }
	}
	if = {
		limit = {
			check_variable = {
				var = universal_cost_transaction_found_index
				value = constant:universal_cost_framework.invalid_index
				compare = greater_than
			}
			check_variable = {
				var = universal_cost_transaction_states^universal_cost_transaction_found_index
				value = constant:universal_cost_framework.transaction_state_settled
				compare = not_equals
			}
		}
		set_temp_variable = { universal_cost_component_lookup_transaction_id = universal_cost_transaction_id }
		set_temp_variable = { universal_cost_component_lookup_id = universal_cost_transaction_component_id }
		set_temp_variable = { universal_cost_component_found_index = constant:universal_cost_framework.invalid_index }
		set_temp_variable = { universal_cost_component_lookup_result = constant:universal_cost_framework.result_no_op }
		universal_cost_find_component_row = yes
		if = {
			limit = { check_variable = { var = universal_cost_component_lookup_result value = constant:universal_cost_framework.result_rejected compare = equals } }
			set_temp_variable = { universal_cost_mark_component_refunded_result = constant:universal_cost_framework.result_rejected }
		}
		if = {
			limit = {
				check_variable = { var = universal_cost_component_lookup_result value = constant:universal_cost_framework.result_success compare = equals }
				check_variable = {
					var = universal_cost_component_found_index
					value = constant:universal_cost_framework.invalid_index
					compare = greater_than
				}
				check_variable = {
					var = universal_cost_component_refunded_states^universal_cost_component_found_index
					value = constant:universal_cost_component_state.not_refunded
					compare = equals
				}
			}
			set_variable = {
				universal_cost_component_refunded_states^universal_cost_component_found_index = constant:universal_cost_component_state.refunded
			}
			universal_cost_decrement_non_political_paid_count = yes
			set_temp_variable = {
				universal_cost_transaction_current_total = universal_cost_transaction_total_refunded_costs^universal_cost_transaction_found_index
			}
			add_to_temp_variable = {
				universal_cost_transaction_current_total = universal_cost_component_actual_paid_costs^universal_cost_component_found_index
			}
			set_variable = {
				universal_cost_transaction_total_refunded_costs^universal_cost_transaction_found_index = universal_cost_transaction_current_total
			}
			set_temp_variable = { universal_cost_mark_component_refunded_result = constant:universal_cost_framework.result_success }
		}
	}
}

# Validate every link before the first credit. A damaged or cyclic chain never
# becomes a successful refund, even if its first component is readable.
universal_cost_validate_component_chain = {
	set_temp_variable = { universal_cost_chain_valid = constant:universal_cost_framework.zero }
	if = {
		limit = {
			check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals }
			check_variable = { var = universal_cost_transaction_component_counts^universal_cost_transaction_found_index value = constant:universal_cost_framework.zero compare = greater_than }
			check_variable = { var = universal_cost_transaction_component_counts^universal_cost_transaction_found_index value = constant:universal_cost_framework.max_components_per_transaction compare = less_than_or_equals }
		}
		set_temp_variable = { universal_cost_chain_valid = constant:universal_cost_framework.one }
		set_temp_variable = { universal_cost_chain_pointer = universal_cost_transaction_component_head_plus_one^universal_cost_transaction_found_index }
		set_temp_variable = { universal_cost_chain_walked = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_chain_last_pointer = constant:universal_cost_framework.zero }
		while_loop_effect = {
			limit = {
				check_variable = { var = universal_cost_chain_pointer value = constant:universal_cost_framework.zero compare = greater_than }
				check_variable = { var = universal_cost_chain_walked value = universal_cost_transaction_component_counts^universal_cost_transaction_found_index compare = less_than }
				check_variable = { var = universal_cost_chain_valid value = constant:universal_cost_framework.one compare = equals }
			}
			set_temp_variable = { universal_cost_chain_index = universal_cost_chain_pointer }
			subtract_from_temp_variable = { universal_cost_chain_index = constant:universal_cost_framework.one }
			if = {
				limit = {
					check_variable = { var = universal_cost_chain_index value = universal_cost_component_transaction_ids^num compare = less_than }
					check_variable = { var = universal_cost_chain_index value = universal_cost_component_next_plus_one^num compare = less_than }
					check_variable = { var = universal_cost_component_transaction_ids^universal_cost_chain_index value = universal_cost_transaction_id compare = equals }
				}
				set_temp_variable = { universal_cost_chain_last_pointer = universal_cost_chain_pointer }
				set_temp_variable = { universal_cost_chain_pointer = universal_cost_component_next_plus_one^universal_cost_chain_index }
			}
			else = {
				set_temp_variable = { universal_cost_chain_valid = constant:universal_cost_framework.zero }
			}
			add_to_temp_variable = { universal_cost_chain_walked = constant:universal_cost_framework.one }
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = universal_cost_chain_walked value = universal_cost_transaction_component_counts^universal_cost_transaction_found_index compare = not_equals }
					check_variable = { var = universal_cost_chain_pointer value = constant:universal_cost_framework.zero compare = not_equals }
					check_variable = { var = universal_cost_chain_last_pointer value = universal_cost_transaction_component_tail_plus_one^universal_cost_transaction_found_index compare = not_equals }
				}
			}
			set_temp_variable = { universal_cost_chain_valid = constant:universal_cost_framework.zero }
		}
	}
}

# Input: universal_cost_refund_cursor. The transaction row is already known and
# a linked receipt has passed the full-chain check.
universal_cost_refund_component_at_cursor = {
	if = {
		limit = {
			check_variable = { var = universal_cost_component_transaction_ids^universal_cost_refund_cursor value = universal_cost_transaction_id compare = equals }
			check_variable = { var = universal_cost_component_refunded_states^universal_cost_refund_cursor value = constant:universal_cost_component_state.not_refunded compare = equals }
		}
		set_temp_variable = { universal_cost_payment_resource_kind = universal_cost_component_resource_kinds^universal_cost_refund_cursor }
		set_temp_variable = { universal_cost_payment_amount = universal_cost_component_actual_paid_costs^universal_cost_refund_cursor }
		if = {
			limit = { check_variable = { var = universal_cost_payment_amount value = constant:universal_cost_framework.zero compare = less_than_or_equals } }
			set_variable = { universal_cost_component_refunded_states^universal_cost_refund_cursor = constant:universal_cost_component_state.refunded }
		}
		else = {
			set_temp_variable = { universal_cost_credit_result = constant:universal_cost_framework.result_no_op }
			universal_cost_credit_component = yes
			if = {
				limit = { check_variable = { var = universal_cost_credit_result value = constant:universal_cost_framework.result_success compare = equals } }
				set_variable = { universal_cost_component_refunded_states^universal_cost_refund_cursor = constant:universal_cost_component_state.refunded }
				add_to_temp_variable = { universal_cost_refund_total = universal_cost_payment_amount }
				set_temp_variable = { universal_cost_component_found_index = universal_cost_refund_cursor }
				universal_cost_decrement_non_political_paid_count = yes
			}
			else = {
				set_temp_variable = { universal_cost_refund_all_complete = constant:universal_cost_framework.zero }
				add_to_temp_variable = { universal_cost_refund_unhandled_total = universal_cost_payment_amount }
			}
		}
	}
	if = {
		limit = {
			check_variable = { var = universal_cost_component_transaction_ids^universal_cost_refund_cursor value = universal_cost_transaction_id compare = equals }
			check_variable = { var = universal_cost_component_refunded_states^universal_cost_refund_cursor value = constant:universal_cost_component_state.not_refunded compare = equals }
		}
		set_temp_variable = { universal_cost_refund_all_complete = constant:universal_cost_framework.zero }
	}
}

# Refund only stored actual-paid amounts. The indexed path visits this
# transaction's linked components, regardless of historical array length.
universal_cost_refund_transaction = {
	set_temp_variable = { universal_cost_refund_transaction_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_refund_total = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_refund_unhandled_total = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_lookup_id = universal_cost_transaction_id }
	# Caller-owned lookup outputs; do not reset unrelated outer cursors.
	set_temp_variable = { universal_cost_transaction_found_index = constant:universal_cost_framework.invalid_index }
	set_temp_variable = { universal_cost_transaction_is_indexed = constant:universal_cost_framework.zero }
	set_temp_variable = { universal_cost_transaction_lookup_result = constant:universal_cost_framework.result_no_op }
	set_temp_variable = { universal_cost_transaction_pointer = constant:universal_cost_framework.zero }
	universal_cost_find_transaction_row = yes
	if = {
		limit = { check_variable = { var = universal_cost_transaction_lookup_result value = constant:universal_cost_framework.result_rejected compare = equals } }
		set_temp_variable = { universal_cost_refund_transaction_result = constant:universal_cost_framework.result_rejected }
	}
	if = {
		limit = {
			check_variable = { var = universal_cost_transaction_found_index value = constant:universal_cost_framework.invalid_index compare = greater_than }
			OR = {
				check_variable = { var = universal_cost_transaction_states^universal_cost_transaction_found_index value = constant:universal_cost_framework.transaction_state_recorded compare = equals }
				check_variable = { var = universal_cost_transaction_states^universal_cost_transaction_found_index value = constant:universal_cost_framework.transaction_state_refund_partial compare = equals }
			}
		}
		set_temp_variable = { universal_cost_refund_ready = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_refund_all_complete = constant:universal_cost_framework.one }
		if = {
			limit = { check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals } }
			set_temp_variable = { universal_cost_chain_valid = constant:universal_cost_framework.zero }
			universal_cost_validate_component_chain = yes
			if = {
				limit = { check_variable = { var = universal_cost_chain_valid value = constant:universal_cost_framework.one compare = equals } }
				set_temp_variable = { universal_cost_refund_pointer = universal_cost_transaction_component_head_plus_one^universal_cost_transaction_found_index }
				set_temp_variable = { universal_cost_refund_ready = constant:universal_cost_framework.one }
			}
		}
		else_if = {
			limit = { check_variable = { var = universal_cost_component_transaction_ids^num value = constant:universal_cost_framework.legacy_scan_max_rows compare = less_than_or_equals } }
			set_temp_variable = { universal_cost_refund_cursor = constant:universal_cost_framework.zero }
			set_temp_variable = { universal_cost_refund_ready = constant:universal_cost_framework.one }
		}
		if = {
			limit = { check_variable = { var = universal_cost_refund_ready value = constant:universal_cost_framework.one compare = equals } }
			while_loop_effect = {
				limit = {
					OR = {
						AND = {
							check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals }
							check_variable = { var = universal_cost_refund_pointer value = constant:universal_cost_framework.zero compare = greater_than }
						}
						AND = {
							check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.zero compare = equals }
							check_variable = { var = universal_cost_refund_cursor value = universal_cost_component_transaction_ids^num compare = less_than }
						}
					}
				}
				if = {
					limit = { check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals } }
					set_temp_variable = { universal_cost_refund_cursor = universal_cost_refund_pointer }
					subtract_from_temp_variable = { universal_cost_refund_cursor = constant:universal_cost_framework.one }
				}
				universal_cost_refund_component_at_cursor = yes
				if = {
					limit = { check_variable = { var = universal_cost_transaction_is_indexed value = constant:universal_cost_framework.one compare = equals } }
					set_temp_variable = { universal_cost_refund_pointer = universal_cost_component_next_plus_one^universal_cost_refund_cursor }
				}
				else = { add_to_temp_variable = { universal_cost_refund_cursor = constant:universal_cost_framework.one } }
			}
			set_temp_variable = { universal_cost_transaction_current_total = universal_cost_transaction_total_refunded_costs^universal_cost_transaction_found_index }
			add_to_temp_variable = { universal_cost_transaction_current_total = universal_cost_refund_total }
			set_variable = { universal_cost_transaction_total_refunded_costs^universal_cost_transaction_found_index = universal_cost_transaction_current_total }
			if = {
				limit = { check_variable = { var = universal_cost_refund_all_complete value = constant:universal_cost_framework.one compare = equals } }
				set_variable = { universal_cost_transaction_states^universal_cost_transaction_found_index = constant:universal_cost_framework.transaction_state_refunded }
				set_temp_variable = { universal_cost_refund_transaction_result = constant:universal_cost_framework.result_success }
			}
			else = {
				set_variable = { universal_cost_transaction_states^universal_cost_transaction_found_index = constant:universal_cost_framework.transaction_state_refund_partial }
				set_temp_variable = { universal_cost_refund_transaction_result = constant:universal_cost_framework.result_partial }
			}
		}
		else = { set_temp_variable = { universal_cost_refund_transaction_result = constant:universal_cost_framework.result_rejected } }
	}
}
