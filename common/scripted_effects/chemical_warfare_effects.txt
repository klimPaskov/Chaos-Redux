chem_sync_chemical_tactic_preferred_weight_ideas = {
	if = {
		limit = {
			has_tech = basic_gas_masks
			NOT = { has_country_flag = chem_gas_mask_preferred_tactic_suppression_synced }
		}
		add_ideas = chemical_gas_mask_preferred_tactic_suppression
		set_country_flag = chem_gas_mask_preferred_tactic_suppression_synced
	}

	if = {
		limit = {
			OR = {
				has_tech = chlorine_gas
				has_tech = phosgene
				has_tech = mustard_gas
				has_tech = lewisite
				has_tech = sarin
				has_tech = soman
			}
			NOT = { has_country_flag = chem_shelling_preferred_tactic_suppression_synced }
		}
		add_ideas = chemical_shelling_preferred_tactic_suppression
		set_country_flag = chem_shelling_preferred_tactic_suppression_synced
	}
}

chem_warfare_register_attack_use_no_livens = {
    if = {
        limit = { NOT = { has_variable = chem_condemnation_base } }
        set_temp_variable = { var = chem_condemnation_base value = 0 }
    }
    if = {
        limit = { NOT = { has_variable = chem_condemnation_per_unit } }
        set_temp_variable = { var = chem_condemnation_per_unit value = 0 }
    }
    set_temp_variable = { var = chem_condemnation_add value = chem_condemnation_base }

    if = {
        limit = { has_variable = chem_consumed_amount }
        set_temp_variable = { var = chem_condemnation_scale value = chem_consumed_amount }
    }
    else_if = {
        limit = { has_variable = chem_required_amount }
        set_temp_variable = { var = chem_condemnation_scale value = chem_required_amount }
        if = {
            limit = { has_variable = chem_attack_scale }
            multiply_temp_variable = { var = chem_condemnation_scale value = chem_attack_scale }
        }
        else_if = {
            limit = { has_variable = chem_ratio_loc }
            multiply_temp_variable = { var = chem_condemnation_scale value = chem_ratio_loc }
        }
    }
    else = {
        set_temp_variable = { var = chem_condemnation_scale value = 0 }
    }

    multiply_temp_variable = { var = chem_condemnation_scale value = chem_condemnation_per_unit }
    add_to_temp_variable = { var = chem_condemnation_add value = chem_condemnation_scale }
    set_temp_variable = { chem_operations_condemnation_mult = constant:cbrn_math.one }
    chem_set_integrated_operations_condemnation_mult_from_country = yes
    multiply_temp_variable = { var = chem_condemnation_add value = chem_operations_condemnation_mult }
    if = {
        limit = { has_variable = chem_condemnation_mult }
        multiply_temp_variable = { var = chem_condemnation_add value = chem_condemnation_mult }
    }

    if = {
        limit = { check_variable = { var = chem_condemnation_add value = 0 compare = greater_than } }
        set_temp_variable = { condemnation_source_type = constant:condemnation_source.chemical }
        set_temp_variable = { condemnation_base_gain = chem_condemnation_add }
        set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.public }
        set_temp_variable = { condemnation_severity_type = constant:condemnation_severity.combat }
        set_temp_variable = { condemnation_civilian_deaths = 0 }
        if = {
            limit = { has_variable = chem_condemnation_civilian_deaths }
            set_temp_variable = { condemnation_civilian_deaths = chem_condemnation_civilian_deaths }
        }
        set_temp_variable = { condemnation_contamination_value = 0 }
        if = {
            limit = { has_variable = chem_condemnation_contamination_value }
            set_temp_variable = { condemnation_contamination_value = chem_condemnation_contamination_value }
        }
		if = {
			limit = { var:chem_condemnation_victim_country = { exists = yes } }
			set_temp_variable = { condemnation_victim_country = chem_condemnation_victim_country }
		}
        set_temp_variable = { condemnation_context_type = constant:condemnation_context.chemical_combat }
        if = {
            limit = { check_variable = { chem_air_bomb_active_regions > 0 } }
            set_temp_variable = { condemnation_context_type = constant:condemnation_context.chemical_air_strike }
        }
        else_if = {
            limit = { has_variable = chem_condemnation_context_type }
            set_temp_variable = { condemnation_context_type = chem_condemnation_context_type }
        }
        condemnation_add_source = yes
        air_cleanliness_treaty_on_unconventional_use = yes
    }
}

chem_try_grant_chemical_operations_trait_from_academy_spirit = {
	if = {
		limit = {
			FROM = { has_idea = chemical_operations_academy_spirit }
			is_army_leader = yes
			NOT = { has_trait = chemical_operations_commander }
		}
		randomize_temp_variable = {
			var = chem_operations_academy_trait_roll
			min = 0
			max = @CHEM_OPERATIONS_ACADEMY_TRAIT_ROLL_MAX
		}
		if = {
			limit = {
				check_variable = {
					var = chem_operations_academy_trait_roll
					value = constant:chem_chaos_warfare_spirit.academy.chemical_operations_trait_gain_chance
					compare = less_than
				}
			}
			add_unit_leader_trait = chemical_operations_commander
		}
	}
}
