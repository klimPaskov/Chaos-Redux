# Immediate owner publication for Migration, Famine and Camps.
# No clock, world scan, simulation registration, payment or gameplay assessment.

# COUNTRY terminal/rebirth cleanup, including obsolete observer fields.
national_policy_panel_retire_country = {
	clr_country_flag = national_policy_panel_observer_pending
	clear_variable = national_policy_panel_observer_owner
	clear_variable = national_policy_panel_observer_earliest
	clear_variable = national_policy_panel_owner
	clear_variable = national_policy_panel_update_depth
	clr_country_flag = national_policy_panel_publication_pending
	clr_country_flag = national_policy_panel_publishing
	humanitarian_panel_invalidate_actor = yes
	clr_country_flag = camp_repression_ledger_open
	camp_admin_close_pickers = yes
	clear_variable = camp_selected_state_id
	clear_array = camp_admin_gui_states
}

# COUNTRY initialization. Copied presentation belongs to its original actor.
national_policy_panel_reconcile_identity = {
	if = {
		limit = {
			OR = {
				AND = { has_variable = national_policy_panel_owner NOT = { check_variable = { national_policy_panel_owner = THIS } } }
				AND = { has_variable = national_policy_panel_observer_owner NOT = { check_variable = { national_policy_panel_observer_owner = THIS } } }
			}
		}
		national_policy_panel_retire_country = yes
	}
	clr_country_flag = national_policy_panel_observer_pending
	clear_variable = national_policy_panel_observer_owner
	clear_variable = national_policy_panel_observer_earliest
}

# COUNTRY. Refresh only open windows. Existing owner callbacks may call this paused.
national_policy_panel_project_open = {
	if = {
		limit = { OR = { has_country_flag = famine_panel_open has_country_flag = migration_panel_open } }
		if = {
			limit = { humanitarian_panel_is_eligible = yes }
			if = { limit = { has_country_flag = famine_panel_open } famine_panel_refresh = yes }
			if = { limit = { has_country_flag = migration_panel_open } migration_panel_refresh = yes }
		}
		else = { humanitarian_panel_invalidate_actor = yes }
	}
	if = {
		limit = { has_country_flag = camp_repression_ledger_open }
		if = {
			limit = { exists = yes is_ai = no camp_rework_country_is_eligible = yes has_camp_category_visible_action = yes }
			# Presentation only. The ledger opener/rebuild can register and reconcile gameplay.
			camp_admin_refresh_interface = yes
		}
		else = {
			clr_country_flag = camp_repression_ledger_open
			camp_admin_close_pickers = yes
			clear_variable = camp_selected_state_id
			clear_array = camp_admin_gui_states
		}
	}
}

# COUNTRY nested transaction boundaries. Only an actual open human is tracked.
national_policy_panel_begin_update = {
	if = {
		limit = { exists = yes is_ai = no national_policy_panel_has_open_window = yes }
		add_to_variable = { national_policy_panel_update_depth = constant:humanitarian_panel.one }
	}
}
# COUNTRY matching end; the final flush occurs before this effect returns.
national_policy_panel_end_update = {
	if = {
		limit = { check_variable = { national_policy_panel_update_depth > constant:humanitarian_panel.zero } }
		subtract_from_variable = { national_policy_panel_update_depth = constant:humanitarian_panel.one }
		if = {
			limit = { NOT = { check_variable = { national_policy_panel_update_depth > constant:humanitarian_panel.zero } } }
			clear_variable = national_policy_panel_update_depth
			if = { limit = { has_country_flag = national_policy_panel_publication_pending } national_policy_panel_refresh_open = yes }
		}
	}
}

# COUNTRY immediate publication. Closed panels are idle; nested requests coalesce.
# Projection callees are read-only gameplay consumers and cannot publish recursively.
national_policy_panel_refresh_open = {
	if = {
		limit = { exists = yes is_ai = no national_policy_panel_has_open_window = yes }
		set_country_flag = national_policy_panel_publication_pending
		if = {
			limit = {
				NOT = { check_variable = { national_policy_panel_update_depth > constant:humanitarian_panel.zero } }
				NOT = { has_country_flag = national_policy_panel_publishing }
			}
			set_country_flag = national_policy_panel_publishing
			clr_country_flag = national_policy_panel_publication_pending
			# A no-delay hidden event establishes actor ROOT and isolated temporary storage.
			country_event = { id = national_policy_panel.2 }
			clr_country_flag = national_policy_panel_publishing
		}
	}
}

# STATE, after committed publication. Notify owner and distinct controller.
# Former controllers and independent receivers require an explicit owner callback.
national_policy_panel_publish_state = {
	owner = { save_event_target_as = national_policy_panel_state_owner national_policy_panel_refresh_open = yes }
	controller = {
		if = {
			limit = { NOT = { tag = event_target:national_policy_panel_state_owner } }
			national_policy_panel_refresh_open = yes
		}
	}
}
