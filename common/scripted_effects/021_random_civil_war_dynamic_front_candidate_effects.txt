# Event 021 dynamic front candidate effects
# Builds variable-length optional-front candidates from viable state anchors and
# complete dormant Event 006 packages before ownership changes. The candidate
# count is derived from the live state/package ledgers and remaining shared
# front capacity; this file contains no named secondary slots.

# COUNTRY scope on the opening host. Appends one ordinary claimant candidate
# from the current viable anchor and its complete connected state region.
event021_parent_append_dynamic_ordinary_front_candidate = {
	set_temp_variable = { event021_planned_route_archetype = event021_dynamic_candidate_archetype }
	event021_parent_set_planned_route_source = yes
	event021_parent_set_planned_front_goal = yes
	set_temp_variable = { event021_dynamic_candidate_state_offset = event021_parent_dynamic_optional_front_selected_states^num }
	set_temp_variable = { event021_dynamic_candidate_state_count = constant:random_civil_war_value.zero }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_archetypes value = event021_dynamic_candidate_archetype }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_routes value = event021_planned_front_route }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_anchors value = THIS }
	if = {
		limit = { has_variable = global.random_civil_war_pending_opposition_front_id }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_sources value = global.random_civil_war_pending_opposition_front_id }
	}
	else = { add_to_temp_array = { array = event021_parent_dynamic_optional_front_sources value = constant:random_civil_war_value.zero } }
	event_target:random_civil_war_dynamic_candidate_host = { add_to_temp_array = { array = event021_parent_dynamic_optional_front_actor_scopes value = THIS } }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_actor_types value = constant:random_civil_war_planned_actor_type.ordinary_claimant }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_relationships value = constant:random_civil_war_planned_relationship.nested_war }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_goals value = event021_planned_front_goal }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_packages value = constant:random_civil_war_value.zero }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_package_regions value = constant:random_civil_war_value.zero }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_package_depths value = constant:random_civil_war_value.zero }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_package_archetypes value = event021_dynamic_candidate_archetype }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_state_weights value = event021_dynamic_candidate_region_weight }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_selected_routes value = event021_dynamic_candidate_archetype }
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_state_offsets value = event021_dynamic_candidate_state_offset }
	for_each_scope_loop = {
		array = event021_parent_dynamic_candidate_region_states
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_selected_states value = THIS }
		add_to_temp_variable = { event021_dynamic_candidate_state_count = constant:random_civil_war_value.one }
	}
	add_to_temp_array = { array = event021_parent_dynamic_optional_front_state_counts value = event021_dynamic_candidate_state_count }
}

# COUNTRY scope on the opening host. Captures one already-admitted complete
# Event 006 package before its adapter scratch is cleared.
event021_parent_append_dynamic_event6_front_candidate = {
	if = {
		limit = {
			has_variable = random_civil_war_event6_package_id
			has_event_target = random_civil_war_event6_selected_anchor
			NOT = { is_in_array = { array = event021_parent_dynamic_optional_front_packages value = random_civil_war_event6_package_id } }
			NOT = { is_in_array = { array = event021_parent_dynamic_optional_front_selected_states value = event_target:random_civil_war_event6_selected_anchor } }
			OR = {
				NOT = { is_in_array = { array = global.random_civil_war_scenario_reserved_states value = event_target:random_civil_war_event6_selected_anchor } }
				is_in_array = { array = random_civil_war_scenario_frozen_states value = event_target:random_civil_war_event6_selected_anchor }
			}
		}
		set_temp_variable = { event021_dynamic_candidate_anchor_weight = constant:random_civil_war_value.zero }
		event_target:random_civil_war_event6_selected_anchor = {
			event021_parent_score_anchor_state = yes
			set_temp_variable = { event021_dynamic_candidate_anchor_weight = event021_anchor_state_score }
		}
		set_temp_variable = { event021_dynamic_candidate_state_offset = event021_parent_dynamic_optional_front_selected_states^num }
		set_temp_variable = { event021_dynamic_candidate_state_count = constant:random_civil_war_value.zero }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_archetypes value = constant:random_civil_war_archetype.event006 }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_routes value = constant:random_civil_war_evidence_source.event006 }
		event_target:random_civil_war_event6_selected_anchor = {
			add_to_temp_array = { array = event021_parent_dynamic_optional_front_anchors value = THIS }
			add_to_temp_array = { array = event021_parent_dynamic_optional_front_selected_states value = THIS }
			add_to_temp_variable = { event021_dynamic_candidate_state_count = constant:random_civil_war_value.one }
		}
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_state_offsets value = event021_dynamic_candidate_state_offset }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_state_counts value = event021_dynamic_candidate_state_count }
		if = {
			limit = { has_variable = global.random_civil_war_pending_host_front_id }
			add_to_temp_array = { array = event021_parent_dynamic_optional_front_sources value = global.random_civil_war_pending_host_front_id }
		}
		else = { add_to_temp_array = { array = event021_parent_dynamic_optional_front_sources value = constant:random_civil_war_value.zero } }
		if = {
			limit = { has_event_target = random_civil_war_event6_selected_actor }
			event_target:random_civil_war_event6_selected_actor = { add_to_temp_array = { array = event021_parent_dynamic_optional_front_actor_scopes value = THIS } }
		}
		else = { event_target:random_civil_war_dynamic_candidate_host = { add_to_temp_array = { array = event021_parent_dynamic_optional_front_actor_scopes value = THIS } } }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_actor_types value = constant:random_civil_war_planned_actor_type.event006_carrier }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_relationships value = constant:random_civil_war_planned_relationship.independence_war }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_goals value = constant:random_civil_war_front_goal.independence }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_packages value = random_civil_war_event6_package_id }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_package_regions value = random_civil_war_event6_candidate_region }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_package_depths value = random_civil_war_event6_candidate_depth }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_package_archetypes value = random_civil_war_event6_candidate_archetype }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_state_weights value = event021_dynamic_candidate_anchor_weight }
		add_to_temp_array = { array = event021_parent_dynamic_optional_front_selected_routes value = constant:random_civil_war_archetype.event006 }
		if = {
			limit = { has_global_flag = random_civil_war_manual_scenario_launch_in_progress }
			event_target:random_civil_war_event6_selected_anchor = {
				if = {
					limit = { NOT = { is_in_array = { array = global.random_civil_war_scenario_reserved_states value = THIS } } }
					add_to_array = { array = global.random_civil_war_scenario_reserved_states value = THIS }
				}
			}
			add_to_array = { array = global.random_civil_war_scenario_reserved_event6_packages value = random_civil_war_event6_package_id }
		}
	}
}

# COUNTRY scope. Releases only this host's frozen optional-front receipt and
# the shared state, package, and front-capacity reservations it owns.
event021_parent_clear_scenario_optional_front_receipt = {
	for_each_scope_loop = {
		array = random_civil_war_scenario_frozen_optional_front_state_entries
		clr_state_flag = random_civil_war_secondary_transfer_state
		remove_from_array = { array = global.random_civil_war_scenario_reserved_states value = THIS }
	}
	set_temp_variable = { event021_scenario_optional_receipt_index = constant:random_civil_war_value.zero }
	set_temp_variable = { event021_scenario_optional_release_break = constant:random_civil_war_value.zero }
	for_each_loop = {
		array = random_civil_war_scenario_frozen_optional_front_packages
		value = event021_scenario_optional_package_to_release
		index = event021_scenario_optional_receipt_index
		break = event021_scenario_optional_release_break
		remove_from_array = {
			array = global.random_civil_war_scenario_reserved_event6_packages
			value = random_civil_war_scenario_frozen_optional_front_packages^event021_scenario_optional_receipt_index
		}
	}
	if = {
		limit = {
			has_country_flag = random_civil_war_scenario_optional_front_plan_frozen
			has_variable = global.random_civil_war_scenario_reserved_front_count
			has_variable = random_civil_war_scenario_frozen_front_reservation
		}
		subtract_from_variable = { global.random_civil_war_scenario_reserved_front_count = random_civil_war_scenario_frozen_front_reservation }
		if = {
			limit = { check_variable = { var = global.random_civil_war_scenario_reserved_front_count value = constant:random_civil_war_value.zero compare = less_than } }
			set_variable = { global.random_civil_war_scenario_reserved_front_count = constant:random_civil_war_value.zero }
		}
	}
	clear_array = random_civil_war_scenario_frozen_optional_front_archetypes
	clear_array = random_civil_war_scenario_frozen_optional_front_routes
	clear_array = random_civil_war_scenario_frozen_optional_front_anchors
	clear_array = random_civil_war_scenario_frozen_optional_front_actor_scopes
	clear_array = random_civil_war_scenario_frozen_optional_front_actor_types
	clear_array = random_civil_war_scenario_frozen_optional_front_relationships
	clear_array = random_civil_war_scenario_frozen_optional_front_goals
	clear_array = random_civil_war_scenario_frozen_optional_front_packages
	clear_array = random_civil_war_scenario_frozen_optional_front_package_regions
	clear_array = random_civil_war_scenario_frozen_optional_front_package_depths
	clear_array = random_civil_war_scenario_frozen_optional_front_package_archetypes
	clear_array = random_civil_war_scenario_frozen_optional_front_state_weights
	clear_array = random_civil_war_scenario_frozen_optional_front_state_counts
	clear_array = random_civil_war_scenario_frozen_optional_front_state_offsets
	clear_array = random_civil_war_scenario_frozen_optional_front_state_entries
	clear_variable = random_civil_war_scenario_frozen_optional_front_count
	clear_variable = random_civil_war_scenario_frozen_front_reservation
	clear_variable = random_civil_war_scenario_frozen_belligerent_cap
	clr_country_flag = random_civil_war_scenario_optional_front_plan_frozen
}

			clr_state_flag = random_civil_war_secondary_transfer_state

	set_temp_variable = { event021_scenario_optional_input_valid = event021_parent_dynamic_optional_front_valid }
	clear_array = random_civil_war_scenario_frozen_optional_front_archetypes
	clear_array = random_civil_war_scenario_frozen_optional_front_routes
	clear_array = random_civil_war_scenario_frozen_optional_front_anchors
	clear_array = random_civil_war_scenario_frozen_optional_front_actor_scopes

	clear_array = random_civil_war_scenario_frozen_optional_front_packages

	if = {
		limit = {
			check_variable = { var = event021_scenario_optional_front_count value = constant:random_civil_war_value.zero compare = greater_than }

		}
		set_temp_variable = { event021_scenario_optional_freeze_break = constant:random_civil_war_value.zero }

			array = event021_parent_dynamic_optional_front_archetypes
			value = event021_scenario_optional_freeze_archetype
			index = event021_scenario_optional_front_index
			break = event021_scenario_optional_freeze_break
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_archetypes value = event021_parent_dynamic_optional_front_archetypes^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_routes value = event021_parent_dynamic_optional_front_routes^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_actor_types value = event021_parent_dynamic_optional_front_actor_types^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_relationships value = event021_parent_dynamic_optional_front_relationships^event021_scenario_optional_front_index }

			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_packages value = event021_parent_dynamic_optional_front_packages^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_package_regions value = event021_parent_dynamic_optional_front_package_regions^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_package_depths value = event021_parent_dynamic_optional_front_package_depths^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_package_archetypes value = event021_parent_dynamic_optional_front_package_archetypes^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_state_weights value = event021_parent_dynamic_optional_front_state_weights^event021_scenario_optional_front_index }
			set_temp_variable = { event021_scenario_optional_state_offset = random_civil_war_scenario_frozen_optional_front_state_entries^num }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_state_offsets value = event021_scenario_optional_state_offset }
			set_temp_variable = { event021_scenario_optional_state_count = event021_parent_dynamic_optional_front_state_counts^event021_scenario_optional_front_index }
			add_to_array = { array = random_civil_war_scenario_frozen_optional_front_state_counts value = event021_scenario_optional_state_count }
			var:event021_parent_dynamic_optional_front_anchors^event021_scenario_optional_front_index = {
				event_target:random_civil_war_scenario_receipt_owner = {
					add_to_array = { array = random_civil_war_scenario_frozen_optional_front_anchors value = PREV }
				}
			}
			var:event021_parent_dynamic_optional_front_actor_scopes^event021_scenario_optional_front_index = {
				event_target:random_civil_war_scenario_receipt_owner = {
					add_to_array = { array = random_civil_war_scenario_frozen_optional_front_actor_scopes value = PREV }
				}
			}
			set_temp_variable = { event021_scenario_optional_source_offset = event021_parent_dynamic_optional_front_state_offsets^event021_scenario_optional_front_index }
			set_temp_variable = { event021_scenario_optional_source_end = event021_scenario_optional_source_offset }
			add_to_temp_variable = { event021_scenario_optional_source_end = event021_scenario_optional_state_count }
			set_temp_variable = { event021_scenario_optional_source_index = constant:random_civil_war_value.zero }
			for_each_scope_loop = {
				array = event021_parent_dynamic_optional_front_selected_states
				if = {
					limit = {
						check_variable = { var = event021_scenario_optional_source_index value = event021_scenario_optional_source_offset compare = greater_than_or_equals }
						check_variable = { var = event021_scenario_optional_source_index value = event021_scenario_optional_source_end compare = less_than }
					}
					event_target:random_civil_war_scenario_receipt_owner = {
						add_to_array = { array = random_civil_war_scenario_frozen_optional_front_state_entries value = PREV }

	set_variable = { random_civil_war_scenario_frozen_optional_front_count = random_civil_war_scenario_frozen_optional_front_archetypes^num }

		limit = {
			OR = {
				NOT = { check_variable = { var = event021_scenario_optional_input_valid value = constant:random_civil_war_value.one compare = equals } }
				NOT = { check_variable = { var = random_civil_war_scenario_frozen_optional_front_count value = event021_scenario_optional_front_count compare = equals } }

		limit = {
			check_variable = { var = event021_parent_dynamic_optional_front_valid value = constant:random_civil_war_value.one compare = equals }
			NOT = { has_variable = random_civil_war_scenario_base_front_reservation }

	}
	if = {
		limit = {
			check_variable = { var = event021_parent_dynamic_optional_front_valid value = constant:random_civil_war_value.one compare = equals }
			has_country_flag = random_civil_war_scenario_base_front_reserved
			has_variable = random_civil_war_scenario_base_front_reservation
		}
		set_variable = { random_civil_war_scenario_frozen_front_reservation = random_civil_war_scenario_frozen_optional_front_count }
		if = {
			limit = { NOT = { has_variable = global.random_civil_war_scenario_reserved_front_count } }
			set_variable = { global.random_civil_war_scenario_reserved_front_count = constant:random_civil_war_value.zero }
		}
		add_to_variable = { global.random_civil_war_scenario_reserved_front_count = random_civil_war_scenario_frozen_optional_front_count }
		set_country_flag = random_civil_war_scenario_optional_front_plan_frozen
	}
	else = {
		set_temp_variable = { event021_parent_scenario_plan_candidate = constant:random_civil_war_value.zero }
		event021_parent_clear_scenario_optional_front_receipt = yes
	}
}
