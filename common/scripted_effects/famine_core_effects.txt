# Historical profile candidates are registered only after the profile resolver
# proves the audited map/date/control/causal contract. The scheduler below
# iterates this sparse state array; it never discovers profiles by scanning map
# states.
famine_register_historical_profile_candidate_state = {
	humanitarian_initialize_runtime = yes
	set_temp_variable = { famine_profile_candidate_registry_result = constant:humanitarian_runtime.zero }
	if = {
		limit = { famine_historical_profile_context_is_resolved = yes }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.famine_historical_profile_candidate_states
						value = THIS
					}
				}
			}
			add_to_array = { array = global.famine_historical_profile_candidate_states value = THIS }
			set_variable = { global.famine_historical_profile_candidate_states_count = global.famine_historical_profile_candidate_states^num }
		}
		set_state_flag = famine_historical_profile_candidate_active
		set_temp_variable = { famine_profile_candidate_registry_result = constant:humanitarian_runtime.one }
	}
}


famine_unregister_historical_profile_candidate_state = {
	humanitarian_initialize_runtime = yes
	if = {
		limit = {
			is_in_array = {
				array = global.famine_historical_profile_candidate_states
				value = THIS
			}
		}
		remove_from_array = { array = global.famine_historical_profile_candidate_states value = THIS }
		set_variable = { global.famine_historical_profile_candidate_states_count = global.famine_historical_profile_candidate_states^num }
	}
	clr_state_flag = famine_historical_profile_candidate_active
}

# State registry entry points are called only by explicit pressure or trap
# requests. They therefore remain bounded by the active request set.
famine_register_active_state = {
	humanitarian_initialize_runtime = yes
	set_temp_variable = { humanitarian_registry_result = constant:humanitarian_runtime.zero }
	if = {
		limit = { humanitarian_state_is_valid = yes }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.famine_active_states
						value = THIS
					}
				}
			}
			add_to_array = { array = global.famine_active_states value = THIS }
			set_variable = { global.famine_active_states_count = global.famine_active_states^num }
		}
		set_state_flag = famine_food_security_active
		migration_mark_reception_capacity_dirty_if_candidate = yes
		set_temp_variable = { humanitarian_registry_result = constant:humanitarian_runtime.one }
	}
}


famine_unregister_active_state = {
	humanitarian_initialize_runtime = yes
	if = {
		limit = {
			is_in_array = {
				array = global.famine_active_states
				value = THIS
			}
		}
		remove_from_array = { array = global.famine_active_states value = THIS }
		set_variable = { global.famine_active_states_count = global.famine_active_states^num }
	}
	clr_state_flag = famine_food_security_active
	migration_mark_reception_capacity_dirty_if_candidate = yes
	clear_variable = famine_food_pressure
	clear_variable = famine_food_stage
}


# Famine and migration accept separate request bundles. A caller that proves
# both consequences invokes both public endpoints explicitly; neither endpoint
# can initialize, reveal, mutate, or retire the other mechanic.
famine_apply_pressure_request = {
	set_temp_variable = { famine_pressure_request_result = constant:humanitarian_runtime.zero }
	if = {
		limit = { famine_pressure_request_is_valid = yes }
		famine_initialize_state = yes
		set_temp_variable = { famine_food_pressure_population = state_population_k }
		multiply_temp_variable = { famine_food_pressure_population = constant:humanitarian_population.people_per_k }
		set_temp_variable = { famine_food_pressure_delta = famine_pressure_request_amount }
		if = {
			limit = { check_variable = { var = famine_food_pressure_population value = constant:humanitarian_population.zero compare = greater_than } }
			divide_temp_variable = { famine_food_pressure_delta = famine_food_pressure_population }
			divide_temp_variable = { famine_food_pressure_delta = constant:famine_food_shock.population_share_for_maximum }
			multiply_temp_variable = { famine_food_pressure_delta = constant:famine_food_component.maximum }
			clamp_temp_variable = { var = famine_food_pressure_delta min = constant:famine_food_component.zero max = constant:famine_food_component.maximum }
			add_to_variable = { famine_food_pressure = famine_food_pressure_delta }
			clamp_variable = { var = famine_food_pressure min = constant:famine_food_component.zero max = constant:famine_food_component.maximum }
		}
		set_variable = { famine_last_pressure_source = famine_pressure_request_source }
		famine_register_active_state = yes
		set_temp_variable = { famine_pressure_request_result = constant:humanitarian_runtime.one }
	}
	set_temp_variable = { famine_pressure_request_proven = constant:humanitarian_runtime.zero }
	set_temp_variable = { famine_pressure_request_amount = constant:humanitarian_population.zero }
	set_temp_variable = { famine_pressure_request_source = constant:humanitarian_pressure_source.unknown }
	set_temp_variable = { famine_pressure_request_actor_proven = constant:humanitarian_runtime.zero }
}


# STATE scope. Decays the normalized external shock at most once per game date.
# Relief access accelerates recovery, while stable states clear residual memory
# more quickly. This never erases a live component owned by another system.
famine_decay_pressure = {
	if = {
		limit = {
			check_variable = { var = famine_food_pressure value = constant:famine_food_component.zero compare = greater_than }
			OR = {
				NOT = { has_variable = famine_food_pressure_last_decay_date }
				check_variable = { var = famine_food_pressure_last_decay_date value = global.date compare = less_than }
			}
		}
		set_variable = { famine_food_pressure_last_decay_date = global.date }
		set_temp_variable = { famine_food_pressure_decay = constant:famine_food_shock.base_daily_decay }
		if = {
			limit = { has_state_flag = famine_relief_access_active }
			add_to_temp_variable = { famine_food_pressure_decay = constant:famine_food_shock.relief_access_daily_decay }
		}
		if = {
			limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.stable compare = equals } }
			add_to_temp_variable = { famine_food_pressure_decay = constant:famine_food_shock.stable_daily_decay }
		}
		subtract_from_variable = { famine_food_pressure = famine_food_pressure_decay }
		clamp_variable = { var = famine_food_pressure min = constant:famine_food_component.zero max = constant:famine_food_component.maximum }
	}
}

# State scope. Refresh every clause of the island-blockade contract from live
# engine facts. The caller is the bounded active-food-state registry job.
famine_refresh_blockade_proof = {
	clear_variable = famine_blockade_island_proven
	clear_variable = famine_blockade_isolation_proven
	clear_variable = famine_blockade_maritime_dependence_proven
	clear_variable = famine_blockade_route_disrupted_proven
	clear_variable = famine_blockade_port_disrupted_proven
	clear_variable = famine_blockade_convoy_shortage_proven
	clear_variable = famine_blockade_escort_shortage_proven
	clear_variable = famine_blockade_no_humanitarian_corridor_proven
	clear_variable = famine_blockade_local_food_insufficient_proven

	if = {
		limit = {
			OR = {
				is_island_state = yes
				is_one_state_island = yes
			}
		}
		set_variable = { famine_blockade_island_proven = constant:humanitarian_runtime.one }
		set_variable = { famine_blockade_isolation_proven = constant:humanitarian_runtime.one }
	}
	if = {
		limit = {
			OR = {
				is_island_state = yes
				is_one_state_island = yes
			}
			is_coastal = yes
		}
		set_variable = { famine_blockade_maritime_dependence_proven = constant:humanitarian_runtime.one }
	}
	if = {
		limit = {
			OR = {
				has_state_flag = civilian_transfer_route_damaged
				AND = {
					naval_base > 0
					OWNER = { convoy_threat > constant:famine_blockade_runtime.convoy_route_threat_threshold }
				}
			}
		}
		set_variable = { famine_blockade_route_disrupted_proven = constant:humanitarian_runtime.one }
	}
	if = {
		limit = {
			naval_base > 0
			NOT = {
				any_province_building_level = { province = { all_provinces = yes limit_to_naval_base = yes } building = naval_base level > @FAMINE_NON_DAMAGED_MINIMUM }
			}
		}
		set_variable = { famine_blockade_port_disrupted_proven = constant:humanitarian_runtime.one }
	}
	if = {
		limit = { OWNER = { NOT = { has_equipment = { convoy > constant:famine_blockade_runtime.minimum_available_convoys } } } }
		set_variable = { famine_blockade_convoy_shortage_proven = constant:humanitarian_runtime.one }
	}
	if = {
		limit = {
			OWNER = {
				has_navy_size = {
					archetype = ship_hull_light
					size < constant:famine_blockade_runtime.minimum_light_escorts
				}
			}
		}
		set_variable = { famine_blockade_escort_shortage_proven = constant:humanitarian_runtime.one }
	}
	if = {
		limit = { NOT = { has_state_flag = famine_relief_access_active } }
		set_variable = { famine_blockade_no_humanitarian_corridor_proven = constant:humanitarian_runtime.one }
	}

	set_temp_variable = { famine_blockade_reserve_days = constant:famine_food_reserve.zero }
	if = {
		limit = { check_variable = { var = famine_food_reserve_daily_need value = constant:famine_food_reserve.zero compare = greater_than } }
		set_temp_variable = { famine_blockade_reserve_days = famine_food_reserve_amount }
		divide_temp_variable = { famine_blockade_reserve_days = famine_food_reserve_daily_need }
	}
	if = {
		limit = {
			check_variable = {
				var = famine_component_production
				value = constant:famine_blockade_runtime.local_production_pressure_threshold
				compare = greater_than_or_equals
			}
			check_variable = {
				var = famine_blockade_reserve_days
				value = constant:famine_blockade_runtime.maximum_local_reserve_days
				compare = less_than
			}
		}
		set_variable = { famine_blockade_local_food_insufficient_proven = constant:humanitarian_runtime.one }
	}
}


# State scope after famine_refresh_blockade_proof. A full proof adds
# pressure through named transport and need components and accumulates one
# bounded day of isolation per registered daily evaluation. Any broken clause
# resets the consecutive-duration term instead of leaving a stale blockade.
famine_apply_blockade_components = {
	if = {
		limit = { famine_blockade_proof = yes }
		add_to_variable = { famine_blockade_exposure_days = constant:humanitarian_runtime.one }
		set_temp_variable = { famine_blockade_duration_pressure = famine_blockade_exposure_days }
		multiply_temp_variable = { famine_blockade_duration_pressure = constant:famine_blockade_runtime.pressure_per_isolation_day }
		clamp_temp_variable = {
			var = famine_blockade_duration_pressure
			min = constant:famine_food_component.zero
			max = constant:famine_blockade_runtime.maximum_duration_pressure
		}
		add_to_variable = { famine_component_transport = constant:famine_blockade_runtime.transport_pressure_base }
		add_to_variable = { famine_component_transport = famine_blockade_duration_pressure }
		clamp_variable = {
			var = famine_component_transport
			min = constant:famine_food_component.zero
			max = constant:famine_food_component.maximum
		}
		add_to_variable = { famine_component_need = constant:famine_blockade_runtime.need_pressure_base }
		add_to_variable = { famine_component_need = famine_blockade_duration_pressure }
		clamp_variable = {
			var = famine_component_need
			min = constant:famine_food_component.zero
			max = constant:famine_food_component.maximum
		}
		set_variable = { famine_last_pressure_source = constant:humanitarian_pressure_source.blockade }
	}
	else = {
		set_variable = { famine_blockade_exposure_days = constant:humanitarian_runtime.zero }
	}
}

# A food-stage commit may submit one separate survivor-flight request. The
# request is people-denominated, stage-generation-bound, and never mutates
# state population, deaths, or the aligned cohort ledger. Existing flight,
# trapped, and live-cohort reservations use the conservative maximum because
# those ledgers can describe the same threatened people. Migration owns the
# pressure, incident, registration, and accepted-request ledgers through its
# explicit request contract below.
famine_submit_survivor_flight_request = {
	set_temp_variable = { famine_flight_request_amount = constant:humanitarian_population.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			check_variable = { var = famine_food_stage value = constant:famine_food_stage.supply_strain compare = greater_than_or_equals }
		}
		if = { limit = { NOT = { has_variable = famine_flight_stage_generation } } set_variable = { famine_flight_stage_generation = constant:humanitarian_runtime.zero } }
		add_to_variable = { famine_flight_stage_generation = constant:migration_core_reconciliation.projection_generation_start }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = famine_flight_request_generation }
					NOT = { check_variable = { var = famine_flight_request_generation value = famine_flight_stage_generation compare = equals } }
				}
			}
			set_temp_variable = { famine_flight_stage_share = constant:migration_core_reconciliation.flight_share_supply_strain }
			if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.acute_shortage compare = greater_than_or_equals } } set_temp_variable = { famine_flight_stage_share = constant:migration_core_reconciliation.flight_share_acute_shortage } }
			if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.famine compare = greater_than_or_equals } } set_temp_variable = { famine_flight_stage_share = constant:migration_core_reconciliation.flight_share_famine } }
			if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.catastrophic_famine compare = greater_than_or_equals } } set_temp_variable = { famine_flight_stage_share = constant:migration_core_reconciliation.flight_share_catastrophic } }
		}
		set_temp_variable = { famine_flight_live_population = state_population_k }
		multiply_temp_variable = { famine_flight_live_population = constant:humanitarian_population.people_per_k }
		round_temp_variable = famine_flight_live_population
		set_temp_variable = { famine_flight_desired_total = famine_flight_live_population }
		multiply_temp_variable = { famine_flight_desired_total = famine_flight_stage_share }
		if = { limit = { has_variable = famine_flight_request_accepted_total } subtract_from_temp_variable = { famine_flight_desired_total = famine_flight_request_accepted_total } }
		clamp_temp_variable = { var = famine_flight_desired_total min = constant:humanitarian_population.zero }
		set_temp_variable = { famine_flight_request_amount = famine_flight_desired_total }
		clamp_temp_variable = { var = famine_flight_request_amount min = constant:humanitarian_population.zero max = famine_flight_live_population }
		round_temp_variable = famine_flight_request_amount
		if = {
			limit = { check_variable = { var = famine_flight_request_amount value = constant:migration_core_reconciliation.flight_request_minimum compare = greater_than_or_equals } }
			# The migration owner applies the exact pressure and incident receipt.
			# Famine supplies only the people amount, source, actor proof, and its
			# own stage generation; it does not write migration ledgers directly.
			set_temp_variable = { migration_pressure_request_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { migration_pressure_request_amount = famine_flight_request_amount }
			set_temp_variable = { migration_pressure_request_source = constant:humanitarian_pressure_source.resource_shortage }
			set_temp_variable = { migration_pressure_request_actor_proven = OWNER.id }
			set_temp_variable = { migration_pressure_request_generation = famine_flight_stage_generation }
			migration_accept_famine_survivor_request = yes
			if = {
				limit = { check_variable = { var = migration_famine_survivor_request_result value = constant:humanitarian_runtime.one compare = equals } }
				if = { limit = { NOT = { has_variable = famine_flight_request_accepted_total } } set_variable = { famine_flight_request_accepted_total = constant:humanitarian_population.zero } }
				add_to_variable = { famine_flight_request_accepted_total = migration_famine_survivor_accepted_amount }
				# Stage pressure has not selected a lane or destination yet. Keep the
				# famine source at the food/resource shortage channel; movement owners
				# assign internal or cross-border outcome roles after exact transfer.
				set_variable = { famine_last_pressure_source = constant:humanitarian_pressure_source.resource_shortage }
			}
		}
		set_variable = { famine_flight_request_generation = famine_flight_stage_generation }
	}
}


famine_transition_food_stage = {
	set_temp_variable = { famine_previous_food_stage = famine_food_stage }
	if = {
		limit = {
			check_variable = { var = famine_requested_food_stage value = constant:famine_food_stage.stable compare = greater_than }
			check_variable = { var = famine_requested_food_stage value = constant:famine_food_stage.catastrophic_famine compare = less_than_or_equals }
		}
		set_variable = { famine_food_stage = famine_requested_food_stage }
		if = {
			limit = { check_variable = { var = famine_requested_food_stage value = famine_previous_food_stage compare = greater_than } }
			famine_submit_survivor_flight_request = yes
		}
		set_variable = { famine_food_stage_started_date = global.date }
		set_variable = { famine_food_stage_candidate = famine_requested_food_stage }
		set_variable = { famine_food_exposure_days = constant:humanitarian_runtime.zero }
		set_variable = { famine_food_next_mortality_date = global.date }
		clr_state_flag = famine_food_stage_stable
		clr_state_flag = famine_food_stage_supply_strain
		clr_state_flag = famine_food_stage_acute_shortage
		clr_state_flag = famine_food_stage_famine
		clr_state_flag = famine_food_stage_catastrophic
		if = { limit = { check_variable = { var = famine_requested_food_stage value = constant:famine_food_stage.supply_strain   compare = equals } } set_state_flag = famine_food_stage_supply_strain }
		if = { limit = { check_variable = { var = famine_requested_food_stage value = constant:famine_food_stage.acute_shortage   compare = equals } } set_state_flag = famine_food_stage_acute_shortage }
		if = { limit = { check_variable = { var = famine_requested_food_stage value = constant:famine_food_stage.famine   compare = equals } } set_state_flag = famine_food_stage_famine }
		if = { limit = { check_variable = { var = famine_requested_food_stage value = constant:famine_food_stage.catastrophic_famine   compare = equals } } set_state_flag = famine_food_stage_catastrophic }

		set_variable = { famine_food_reserve_target = famine_food_reserve_target_calculated }
		clamp_variable = { var = famine_food_reserve_amount min = constant:famine_food_reserve.zero }
		if = {
			limit = { NOT = { has_state_flag = famine_food_reserve_initialized } }
			if = {
				limit = {
					check_variable = { var = famine_food_reserve_initialization_proven value = constant:humanitarian_runtime.zero compare = greater_than }
					check_variable = { var = famine_food_reserve_initial_amount value = constant:famine_food_reserve.zero compare = greater_than }
				}
				set_variable = { famine_food_reserve_amount = famine_food_reserve_initial_amount }
				round_variable = famine_food_reserve_amount
				clamp_variable = {
					var = famine_food_reserve_amount
					min = constant:famine_food_reserve.zero
					max = famine_food_reserve_capacity
				}
				set_state_flag = famine_food_reserve_initialized
			}
			if = {
				limit = {
					NOT = { has_state_flag = famine_food_reserve_initialized }
					check_variable = { var = famine_food_reserve_amount value = constant:famine_food_reserve.zero compare = greater_than }
				}
				# A positive amount already present in a save is durable evidence;
				# zero is not promoted to an invented starting stockpile.
				set_state_flag = famine_food_reserve_initialized
			}
		}
		set_temp_variable = { famine_food_reserve_capacity_output = famine_food_reserve_capacity }
		set_temp_variable = { famine_food_reserve_target_output = famine_food_reserve_target }
		set_temp_variable = { famine_food_reserve_amount_output = famine_food_reserve_amount }
		set_temp_variable = { famine_food_reserve_daily_need_output = famine_food_reserve_daily_need }
		set_temp_variable = { famine_food_reserve_logistics_factor_output = famine_food_reserve_logistics_factor }
		set_temp_variable = { famine_food_reserve_refresh_result = constant:humanitarian_runtime.one }
	}
}


# Applies at most one reserve update per game date. A stable state replenishes
# toward target, while an active shortage depletes stock by stage and need.
famine_update_reserve = {
	set_temp_variable = { famine_food_reserve_update_result = constant:humanitarian_runtime.zero }
	set_temp_variable = { famine_food_reserve_replenished_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_depleted_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_amount_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_daily_need_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_relief_output = constant:famine_food_reserve.zero }
	if = {
		limit = { humanitarian_state_is_valid = yes }
		famine_refresh_reserve_capacity = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = famine_food_reserve_last_update_date }
					check_variable = { var = famine_food_reserve_last_update_date value = global.date compare = less_than }
				}
			}
			set_variable = { famine_food_reserve_last_update_date = global.date }
			set_variable = { famine_food_reserve_last_replenished = constant:famine_food_reserve.zero }
			set_variable = { famine_food_reserve_last_depleted = constant:famine_food_reserve.zero }
			if = {
				limit = { check_variable = { var = famine_food_reserve_relief value = constant:famine_food_reserve.zero compare = greater_than } }
				subtract_from_variable = { famine_food_reserve_relief = constant:famine_food_reserve.relief_decay_per_day }
				clamp_variable = { var = famine_food_reserve_relief min = constant:famine_food_reserve.zero max = constant:famine_food_reserve.relief_maximum }
			}
			if = {
				limit = { famine_food_stage_is_active = yes }
				set_temp_variable = { famine_food_reserve_depletion_share = constant:famine_food_reserve.zero }
				if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.supply_strain compare = equals } } set_temp_variable = { famine_food_reserve_depletion_share = constant:famine_food_reserve.supply_strain_depletion_share } }
				if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.acute_shortage compare = equals } } set_temp_variable = { famine_food_reserve_depletion_share = constant:famine_food_reserve.acute_shortage_depletion_share } }
				if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.famine compare = equals } } set_temp_variable = { famine_food_reserve_depletion_share = constant:famine_food_reserve.famine_depletion_share } }
				if = { limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.catastrophic_famine compare = equals } } set_temp_variable = { famine_food_reserve_depletion_share = constant:famine_food_reserve.catastrophic_depletion_share } }
				set_temp_variable = { famine_food_reserve_depletion_factor = famine_component_need }
				divide_temp_variable = { famine_food_reserve_depletion_factor = constant:famine_food_component.component_scale }
				clamp_temp_variable = {
					var = famine_food_reserve_depletion_factor
					min = constant:famine_food_reserve.depletion_factor_minimum
					max = constant:famine_food_weight.component_factor_base
				}
				set_temp_variable = { famine_food_reserve_depletion_amount = famine_food_reserve_daily_need }
				multiply_temp_variable = { famine_food_reserve_depletion_amount = famine_food_reserve_depletion_share }
				multiply_temp_variable = { famine_food_reserve_depletion_amount = famine_food_reserve_depletion_factor }
				round_temp_variable = famine_food_reserve_depletion_amount
				clamp_temp_variable = {
					var = famine_food_reserve_depletion_amount
					min = constant:famine_food_reserve.zero
					max = famine_food_reserve_amount
				}
				if = {
					limit = { check_variable = { var = famine_food_reserve_depletion_amount value = constant:famine_food_reserve.zero compare = greater_than } }
					subtract_from_variable = { famine_food_reserve_amount = famine_food_reserve_depletion_amount }
					set_variable = { famine_food_reserve_last_depleted = famine_food_reserve_depletion_amount }
					add_to_variable = { famine_food_reserve_total_depleted = famine_food_reserve_depletion_amount }
				}
			}
			else = {
				if = {
					limit = {
						check_variable = { var = famine_food_reserve_amount value = famine_food_reserve_target compare = less_than }
					}
					set_temp_variable = { famine_food_reserve_production_factor = constant:famine_food_weight.component_factor_base }
					set_temp_variable = { famine_food_reserve_production_pressure = famine_component_production }
					divide_temp_variable = { famine_food_reserve_production_pressure = constant:famine_food_component.component_scale }
					subtract_from_temp_variable = { famine_food_reserve_production_factor = famine_food_reserve_production_pressure }
					clamp_temp_variable = {
						var = famine_food_reserve_production_factor
						min = constant:famine_food_reserve.replenishment_factor_minimum
						max = constant:famine_food_weight.component_factor_base
					}
					set_temp_variable = { famine_food_reserve_transport_factor = constant:famine_food_weight.component_factor_base }
					set_temp_variable = { famine_food_reserve_transport_pressure = famine_component_transport }
					divide_temp_variable = { famine_food_reserve_transport_pressure = constant:famine_food_component.component_scale }
					subtract_from_temp_variable = { famine_food_reserve_transport_factor = famine_food_reserve_transport_pressure }
					clamp_temp_variable = {
						var = famine_food_reserve_transport_factor
						min = constant:famine_food_reserve.replenishment_factor_minimum
						max = constant:famine_food_weight.component_factor_base
					}
					set_temp_variable = { famine_food_reserve_replenishment_amount = famine_food_reserve_daily_need }
					multiply_temp_variable = { famine_food_reserve_replenishment_amount = constant:famine_food_reserve.replenishment_per_k_per_day }
					multiply_temp_variable = { famine_food_reserve_replenishment_amount = famine_food_reserve_production_factor }
					multiply_temp_variable = { famine_food_reserve_replenishment_amount = famine_food_reserve_transport_factor }
					multiply_temp_variable = { famine_food_reserve_replenishment_amount = famine_food_reserve_logistics_factor }
					round_temp_variable = famine_food_reserve_replenishment_amount
					set_temp_variable = { famine_food_reserve_available_capacity = famine_food_reserve_target }
					subtract_from_temp_variable = { famine_food_reserve_available_capacity = famine_food_reserve_amount }
					clamp_temp_variable = { var = famine_food_reserve_available_capacity min = constant:famine_food_reserve.zero }
					clamp_temp_variable = {
						var = famine_food_reserve_replenishment_amount
						min = constant:famine_food_reserve.zero
						max = famine_food_reserve_available_capacity
					}
					if = {
						limit = { check_variable = { var = famine_food_reserve_replenishment_amount value = constant:famine_food_reserve.zero compare = greater_than } }
						add_to_variable = { famine_food_reserve_amount = famine_food_reserve_replenishment_amount }
						set_state_flag = famine_food_reserve_initialized
						set_variable = { famine_food_reserve_last_replenished = famine_food_reserve_replenishment_amount }
						add_to_variable = { famine_food_reserve_total_replenished = famine_food_reserve_replenishment_amount }
					}
				}
			}
		}
		set_temp_variable = { famine_food_reserve_replenished_output = famine_food_reserve_last_replenished }
		set_temp_variable = { famine_food_reserve_depleted_output = famine_food_reserve_last_depleted }
		set_temp_variable = { famine_food_reserve_amount_output = famine_food_reserve_amount }
		set_temp_variable = { famine_food_reserve_daily_need_output = famine_food_reserve_daily_need }
		set_temp_variable = { famine_food_reserve_relief_output = famine_food_reserve_relief }
		set_temp_variable = { famine_food_reserve_update_result = constant:humanitarian_runtime.one }
	}
}

# Exact state-to-state reserve transfer. The accepted amount is debited once
# from the source and credited once to the destination; a non-zero residual
# triggers rollback before either cumulative transfer ledger is advanced.
famine_transfer_reserves = {
	set_temp_variable = { famine_food_reserve_transfer_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { famine_food_reserve_transfer_source_debit_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_transfer_destination_credit_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_transfer_remaining_output = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_destination_available = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_destination_credit = constant:famine_food_reserve.zero }
	set_temp_variable = { famine_food_reserve_transfer_same_state = constant:humanitarian_runtime.zero }

		limit = { famine_food_reserve_transfer_request_is_valid = yes }

			famine_refresh_reserve_capacity = yes
			set_temp_variable = { famine_food_reserve_destination_available = famine_food_reserve_capacity }
			subtract_from_temp_variable = { famine_food_reserve_destination_available = famine_food_reserve_amount }
			clamp_temp_variable = { var = famine_food_reserve_destination_available min = constant:famine_food_reserve.zero }
			if = {
				limit = { state = event_target:famine_food_reserve_source }
				set_temp_variable = { famine_food_reserve_transfer_same_state = constant:humanitarian_runtime.one }
			}
		}
		if = {
			limit = { check_variable = { var = famine_food_reserve_transfer_same_state value = constant:humanitarian_runtime.zero compare = equals } }
			set_temp_variable = { famine_food_reserve_transfer_applied = famine_food_reserve_transfer_amount }

				var = famine_food_reserve_transfer_applied
				min = constant:famine_food_reserve.zero
				max = famine_food_reserve_amount
			}
			clamp_temp_variable = {
				var = famine_food_reserve_transfer_applied
				min = constant:famine_food_reserve.zero
				max = famine_food_reserve_destination_available
			}
			round_temp_variable = famine_food_reserve_transfer_applied
			if = {
				limit = { check_variable = { var = famine_food_reserve_transfer_applied value = constant:famine_food_reserve.zero compare = greater_than } }
				set_temp_variable = { famine_food_reserve_transfer_source_before = famine_food_reserve_amount }

				subtract_from_temp_variable = { famine_food_reserve_transfer_source_debit = famine_food_reserve_amount }
				event_target:famine_food_reserve_destination = {
					set_temp_variable = { famine_food_reserve_destination_before = famine_food_reserve_amount }
					add_to_variable = { famine_food_reserve_amount = famine_food_reserve_transfer_applied }
					set_temp_variable = { famine_food_reserve_destination_credit = famine_food_reserve_amount }
					subtract_from_temp_variable = { famine_food_reserve_destination_credit = famine_food_reserve_destination_before }
				}
				set_temp_variable = { famine_food_reserve_transfer_residual = famine_food_reserve_transfer_source_debit }

				}

					limit = { check_variable = { var = famine_food_reserve_transfer_residual value = constant:famine_food_reserve.zero compare = less_than } }
					set_temp_variable = { famine_food_reserve_transfer_excess = famine_food_reserve_transfer_residual }
					multiply_temp_variable = { famine_food_reserve_transfer_excess = constant:chaos_meter_deaths.negative_one }
					event_target:famine_food_reserve_destination = {
						subtract_from_variable = { famine_food_reserve_amount = famine_food_reserve_transfer_excess }
					}
					subtract_from_temp_variable = { famine_food_reserve_destination_credit = famine_food_reserve_transfer_excess }
				}
				set_temp_variable = { famine_food_reserve_transfer_source_debit = famine_food_reserve_transfer_source_before }
				subtract_from_temp_variable = { famine_food_reserve_transfer_source_debit = famine_food_reserve_amount }
				set_temp_variable = { famine_food_reserve_transfer_residual = famine_food_reserve_transfer_source_debit }
				subtract_from_temp_variable = { famine_food_reserve_transfer_residual = famine_food_reserve_destination_credit }
				if = {
					limit = { check_variable = { var = famine_food_reserve_transfer_residual value = constant:famine_food_reserve.zero compare = equals } }
					add_to_variable = { famine_food_reserve_total_transfer_out = famine_food_reserve_transfer_source_debit }
					set_variable = { famine_food_reserve_last_transfer_out = famine_food_reserve_transfer_source_debit }
					event_target:famine_food_reserve_destination = {
						add_to_variable = { famine_food_reserve_total_transfer_in = famine_food_reserve_destination_credit }
						set_variable = { famine_food_reserve_last_transfer_in = famine_food_reserve_destination_credit }
					}
					set_temp_variable = { famine_food_reserve_transfer_source_debit_output = famine_food_reserve_transfer_source_debit }
					set_temp_variable = { famine_food_reserve_transfer_destination_credit_output = famine_food_reserve_destination_credit }

	}
	clear_variable = famine_food_reserve_transfer_amount
	clear_variable = famine_food_reserve_transfer_request_proven
	clear_variable = famine_food_reserve_transfer_route_proven
	clear_variable = famine_food_reserve_transfer_actor_proven
}

		}
		if = {
			limit = { check_variable = { var = famine_food_stage value = constant:famine_food_stage.catastrophic_famine   compare = equals } }
			set_temp_variable = { famine_mortality_minimum_exposure = constant:famine_mortality.catastrophic_minimum_exposure_days }
			set_temp_variable = { famine_mortality_interval = constant:famine_mortality.catastrophic_interval_days }
			set_temp_variable = { famine_mortality_rate = constant:famine_mortality.catastrophic_rate }
		}
		if = {
			limit = { check_variable = { var = famine_food_exposure_days value = famine_mortality_minimum_exposure compare = greater_than_or_equals } }
			set_temp_variable = { famine_mortality_population = state_population_k }
			multiply_temp_variable = { famine_mortality_population = constant:chaos_meter_deaths.people_per_k }
			round_temp_variable = famine_mortality_population
			set_temp_variable = { famine_mortality_floor = famine_mortality_population }
			multiply_temp_variable = { famine_mortality_floor = constant:famine_mortality.protected_floor_share }
			round_temp_variable = famine_mortality_floor
			if = {
				limit = { check_variable = { var = famine_mortality_floor value = constant:famine_mortality.protected_floor_people compare = less_than } }
				set_temp_variable = { famine_mortality_floor = constant:famine_mortality.protected_floor_people }
			}
			clamp_temp_variable = {
				var = famine_mortality_floor
				min = constant:humanitarian_population.zero
				max = famine_mortality_population
			}
			set_temp_variable = { famine_mortality_available = famine_mortality_population }
			subtract_from_temp_variable = { famine_mortality_available = famine_mortality_floor }
			clamp_temp_variable = { var = famine_mortality_available min = constant:humanitarian_population.zero }

			set_temp_variable = { famine_mortality_exposure_factor = famine_food_exposure_days }
			multiply_temp_variable = { famine_mortality_exposure_factor = constant:famine_mortality.exposure_multiplier_per_day }
			add_to_temp_variable = { famine_mortality_exposure_factor = constant:famine_mortality.minimum_exposure_multiplier }
			clamp_temp_variable = {
				var = famine_mortality_exposure_factor
				min = constant:famine_mortality.minimum_exposure_multiplier
				max = constant:famine_mortality.maximum_exposure_multiplier
			}

			set_temp_variable = { famine_mortality_vulnerability_factor = constant:famine_food_weight.component_factor_base }
		set_temp_variable = { famine_mortality_component_factor = famine_component_vulnerability }
		divide_temp_variable = { famine_mortality_component_factor = constant:famine_food_component.component_scale }
		add_to_temp_variable = { famine_mortality_vulnerability_factor = famine_mortality_component_factor }
			set_temp_variable = { famine_mortality_access_factor = constant:famine_food_weight.component_factor_base }
			set_temp_variable = { famine_mortality_component_factor = famine_component_transport }
			multiply_temp_variable = { famine_mortality_component_factor = constant:famine_mortality.access_transport_share }
			set_temp_variable = { famine_mortality_need_factor = famine_component_need }
			multiply_temp_variable = { famine_mortality_need_factor = constant:famine_mortality.access_need_share }
			add_to_temp_variable = { famine_mortality_component_factor = famine_mortality_need_factor }
			divide_temp_variable = { famine_mortality_component_factor = constant:famine_food_component.component_scale }
			add_to_temp_variable = { famine_mortality_access_factor = famine_mortality_component_factor }
			set_temp_variable = { famine_mortality_policy_factor = constant:famine_food_weight.component_factor_base }
			set_temp_variable = { famine_mortality_component_factor = famine_component_extraction }
			divide_temp_variable = { famine_mortality_component_factor = constant:famine_food_component.component_scale }
			add_to_temp_variable = { famine_mortality_policy_factor = famine_mortality_component_factor }
			set_temp_variable = { famine_mortality_environment_factor = constant:famine_food_weight.component_factor_base }
			set_temp_variable = { famine_mortality_component_factor = famine_component_environment }
			divide_temp_variable = { famine_mortality_component_factor = constant:famine_food_component.component_scale }
			add_to_temp_variable = { famine_mortality_environment_factor = famine_mortality_component_factor }
			set_temp_variable = { famine_mortality_governance_factor = constant:famine_food_weight.component_factor_base }
			set_temp_variable = { famine_mortality_component_factor = famine_component_governance }
			divide_temp_variable = { famine_mortality_component_factor = constant:famine_food_component.component_scale }
			add_to_temp_variable = { famine_mortality_governance_factor = famine_mortality_component_factor }
			set_temp_variable = { famine_mortality_relief_factor = constant:famine_food_weight.component_factor_base }
			set_temp_variable = { famine_mortality_component_factor = famine_component_relief }
			divide_temp_variable = { famine_mortality_component_factor = constant:famine_food_component.component_scale }
			subtract_from_temp_variable = { famine_mortality_relief_factor = famine_mortality_component_factor }
			clamp_temp_variable = {
				var = famine_mortality_relief_factor
				min = constant:famine_mortality.minimum_relief_factor
				max = constant:famine_mortality.maximum_relief_factor
			}

			set_temp_variable = { famine_mortality_request = famine_mortality_population }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_rate }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_exposure_factor }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_vulnerability_factor }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_access_factor }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_policy_factor }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_environment_factor }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_governance_factor }
			multiply_temp_variable = { famine_mortality_request = famine_mortality_relief_factor }
			clamp_temp_variable = {
				var = famine_mortality_request
				min = constant:humanitarian_population.zero
				max = famine_mortality_available
			}
			round_temp_variable = famine_mortality_request
			if = {
				limit = { check_variable = { var = famine_mortality_request value = constant:famine_mortality.minimum_pulse_people compare = greater_than_or_equals } }
				set_temp_variable = { state_civilian_population_loss_requested = famine_mortality_request }
				set_temp_variable = { state_civilian_population_loss_minimum_remaining = famine_mortality_floor }
				set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.famine }
				set_temp_variable = { state_civilian_population_loss_log_deaths = constant:chaos_meter_deaths.one }
				set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
				set_temp_variable = { state_civilian_population_loss_has_target_country = constant:chaos_meter_deaths.one }
				set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.one }
				apply_exact_state_civilian_population_loss = yes
				set_variable = { famine_last_deaths_applied = state_civilian_population_loss_applied }
				add_to_variable = { famine_total_deaths = state_civilian_population_loss_applied }
				set_temp_variable = { famine_mortality_result = state_civilian_population_loss_applied }
				famine_record_exact_mortality_condemnation = yes
			}
			set_variable = { famine_food_next_mortality_date = global.date }
			add_to_variable = { famine_food_next_mortality_date = famine_mortality_interval }
		}
	}
}


# Called only from the host country's registry loop in the dedicated on_action
# file. The loop scope is the registered state only.
famine_process_registered_state = {
	famine_consume_registered_state_reassessment = yes
	civilian_transfer_refresh_route_damage_projection = yes
	if = {
		limit = {
			has_state_flag = famine_blockade_candidate_active
			NOT = {
				OR = {
					OWNER = { has_war = yes }
					CONTROLLER = { has_war = yes }
				}
			}
		}
		clr_state_flag = famine_blockade_candidate_active
	}
	if = {
		limit = {
			has_state_flag = famine_food_security_active
			humanitarian_state_is_valid = yes
		}
		famine_evaluate_food_security = yes
	}
	else = {
		famine_relief_unregister_donor_state = yes
		famine_cleanup_state_registration = yes
	}
}

# COUNTRY scope. Called only for the two countries in the exact war-relation
# callback. This one-time owned-state pass seeds true islands and archipelagos
# into the sparse food registry; it is not a recurring world scan.
famine_register_wartime_island_candidates = {
	if = {
		limit = {
			humanitarian_country_is_valid = yes
			has_war = yes
		}
		every_owned_state = {
			limit = {
				humanitarian_state_is_valid = yes
				is_coastal = yes
				OR = {
					is_island_state = yes
					is_one_state_island = yes
				}
			}
			famine_initialize_state = yes
			set_state_flag = famine_blockade_candidate_active
			famine_register_active_state = yes
		}
	}
}

# STATE scope. Consume only the versioned migration-owned reception-demand
# projection. Famine stores a famine-owned input and never reads the migration
# trapped ledger directly.
famine_consume_migration_reception_demand = {
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			check_variable = { var = migration_to_famine_reception_demand_proven value = constant:humanitarian_runtime.one compare = equals }
			check_variable = { var = migration_to_famine_reception_demand_schema_version value = constant:migration_to_famine_reception_demand.schema_version compare = equals }
			check_variable = { var = migration_to_famine_reception_demand_generation value = constant:migration_to_famine_reception_demand.minimum_revision compare = greater_than_or_equals }
			check_variable = { var = migration_to_famine_reception_demand_revision value = constant:migration_to_famine_reception_demand.minimum_revision compare = greater_than_or_equals }
			check_variable = { var = migration_to_famine_reception_demand_cause value = constant:migration_to_famine_reception_demand.cause_trapped_population compare = equals }
			check_variable = { var = migration_to_famine_reception_demand_amount value = constant:humanitarian_population.zero compare = greater_than }
		}
		set_variable = { famine_reception_demand_population = migration_to_famine_reception_demand_amount }
		set_variable = { famine_reception_demand_generation = migration_to_famine_reception_demand_generation }
		set_variable = { famine_reception_demand_revision = migration_to_famine_reception_demand_revision }
		set_state_flag = famine_food_security_active
		famine_register_active_state = yes
	}
	else = {
		clear_variable = famine_reception_demand_population
		clear_variable = famine_reception_demand_generation
		clear_variable = famine_reception_demand_revision
	}
}


# STATE scope. Emits bounded Condemnation sources only after the exact famine
# mortality owner has applied a positive population loss. The state, actor,
# causal components, exact people amount, generation, revision, and request
# identity are all taken from this completed pulse; no decision click is used
# as a substitute for civilian harm.
famine_record_exact_mortality_condemnation = {
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			check_variable = { var = famine_mortality_result value = constant:humanitarian_population.zero compare = greater_than }
			has_variable = famine_food_stage
			has_variable = famine_component_environment
			has_variable = famine_component_transport
			OWNER = { humanitarian_country_is_valid = yes }
		}
		if = { limit = { NOT = { has_variable = famine_mortality_condemnation_generation } } set_variable = { famine_mortality_condemnation_generation = constant:humanitarian_runtime.zero } }
		add_to_variable = { famine_mortality_condemnation_generation = constant:humanitarian_runtime.one }
		save_event_target_as = famine_condemnation_state
		OWNER = { save_event_target_as = famine_condemnation_actor }
		set_temp_variable = { famine_condemnation_state_proven = id }
		set_temp_variable = { famine_condemnation_food_proven = famine_food_stage }
		set_temp_variable = { famine_condemnation_environment_proven = constant:humanitarian_runtime.one }
		set_temp_variable = { famine_condemnation_transport_proven = constant:humanitarian_runtime.one }
		set_temp_variable = { famine_condemnation_actor_proven = OWNER.id }
		set_temp_variable = { famine_condemnation_people_amount = famine_mortality_result }
		set_temp_variable = { famine_condemnation_generation = famine_mortality_condemnation_generation }
		set_temp_variable = { famine_condemnation_revision = famine_food_stage }
		set_temp_variable = { famine_condemnation_request_id = global.date }
		set_temp_variable = { condemnation_civilian_deaths = famine_mortality_result }
		set_temp_variable = { condemnation_contamination_value = constant:humanitarian_population.zero }
		if = {
			limit = {
				has_state_flag = famine_extraction_active
				NOT = { has_state_flag = famine_condemnation_deliberate_starvation_recorded }
			}
			set_temp_variable = { famine_condemnation_policy_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { famine_condemnation_cause_proven = constant:condemnation_context.deliberate_starvation }
			OWNER = { famine_condemn_deliberate_starvation = yes }
			if = {
				limit = { check_variable = { var = famine_condemnation_result value = constant:humanitarian_runtime.one compare = equals } }
				set_state_flag = famine_condemnation_deliberate_starvation_recorded
			}
		}
		if = {
			limit = {
				has_state_flag = famine_crisis_concealed
				NOT = { has_state_flag = famine_condemnation_concealment_recorded }
			}
			set_temp_variable = { famine_condemnation_policy_proven = constant:humanitarian_runtime.one }
			set_temp_variable = { famine_condemnation_cause_proven = constant:condemnation_context.concealment_operation }
			OWNER = { famine_condemn_concealment = yes }
			if = {
				limit = { check_variable = { var = famine_condemnation_result value = constant:humanitarian_runtime.one compare = equals } }
				set_state_flag = famine_condemnation_concealment_recorded
			}
		}
	}
}

famine_initialize_runtime = {
	if = {
		limit = { NOT = { has_global_flag = famine_runtime_initialized } }
		if = { limit = { NOT = { has_variable = global.famine_active_states_count } } set_variable = { global.famine_active_states_count = global.famine_active_states^num } }
		if = { limit = { NOT = { has_variable = global.famine_historical_profile_candidate_states_count } } set_variable = { global.famine_historical_profile_candidate_states_count = global.famine_historical_profile_candidate_states^num } }
		if = { limit = { NOT = { has_variable = global.famine_historical_profile_anchor_states_count } } set_variable = { global.famine_historical_profile_anchor_states_count = global.famine_historical_profile_anchor_states^num } }
		set_global_flag = famine_runtime_initialized
	}
}

famine_claim_mission_slot = {
	if = {
		limit = { famine_country_has_free_mission_slot = yes }
		if = {
			limit = { NOT = { has_variable = famine_active_mission_count } }
			set_variable = { famine_active_mission_count = constant:humanitarian_runtime.zero }
		}
		add_to_variable = { famine_active_mission_count = constant:humanitarian_runtime.one }
	}
}

famine_release_mission_slot = {
	if = {
		limit = { check_variable = { var = famine_active_mission_count value = constant:humanitarian_runtime.zero compare = greater_than } }
		subtract_from_variable = { famine_active_mission_count = constant:humanitarian_runtime.one }
	}
	else = { set_variable = { famine_active_mission_count = constant:humanitarian_runtime.zero } }
}

famine_recount_active_mission_slots = {

}
