@FORM07_CAT_ANCHOR = 165
# The installed-map compact Basque anchor is País Vasco (792). Navarra (172)
# and French Basque (806) remain optional extension objectives.
@FORM07_NAV_ANCHOR = 792
@FORM07_GLC_ANCHOR = 171

# Country scope. The generic family-registration effect calls this immediately
# after loading FORM-07. The identity contract is deliberately absent in the
# current source tree, so the generic readiness flags remain cleared.
independence_wave_form07_register_readiness = {
	independence_wave_formable_clear_selected_family_readiness = yes
	if = {
		limit = { can_independence_wave_form07_register_readiness = yes }
		set_variable = { independence_wave_formable_readiness_family = constant:independence_wave_formable_family.iberian_federation }
		set_country_flag = independence_wave_formable_territory_adapter_ready
		set_country_flag = independence_wave_formable_x_tag_reserved
		set_country_flag = independence_wave_formable_flag_package_ready
		set_country_flag = independence_wave_formable_identity_adapter_ready
		set_country_flag = independence_wave_formable_integration_adapter_ready
		set_country_flag = independence_wave_formable_member_policy_audited
		set_country_flag = independence_wave_form07_readiness_attested
	}
}

# Identity dispatch target. FORM-07 has no source-approved X-ending tag or
# complete identity flag package, so this adapter never writes a cosmetic tag,
# global identity lock, or generic identity-committed flag. The blocked marker
# is transaction-local and is removed by rollback/cleanup.
independence_wave_formable_identity_adapter_7 = {
	if = {
		limit = { has_country_flag = independence_wave_formable_mutation_prevalidated }
		set_country_flag = independence_wave_form07_identity_adapter_blocked
	}
}

# Member-country scope with ROOT as the future reviewed carrier. Only the
# bilateral access and guarantee relations created by this adapter are marked
# for later removal; no faction, subject, idea, or tree replacement is implied.
independence_wave_form07_install_autonomous_member = {
	save_event_target_as = independence_wave_form07_autonomous_member_target
	set_country_flag = independence_wave_form07_autonomous_member
	set_variable = { independence_wave_form07_autonomous_family = ROOT.independence_wave_formable_family }
	set_variable = { independence_wave_form07_autonomous_carrier = ROOT }
	set_variable = { independence_wave_form07_autonomous_carrier_generation = ROOT.independence_wave_generation_id }
	if = {
		limit = { NOT = { has_military_access_to = ROOT } }
		give_military_access = ROOT
		set_country_flag = independence_wave_form07_member_granted_access_to_carrier
	}
	if = {
		limit = { NOT = { has_guaranteed = ROOT } }
		give_guarantee = ROOT
		set_country_flag = independence_wave_form07_member_guaranteed_carrier
	}
	ROOT = {
		if = {
			limit = { NOT = { has_military_access_to = event_target:independence_wave_form07_autonomous_member_target } }
			give_military_access = event_target:independence_wave_form07_autonomous_member_target
			event_target:independence_wave_form07_autonomous_member_target = { set_country_flag = independence_wave_form07_carrier_granted_access_to_member }
		}
		if = {
			limit = { NOT = { has_guaranteed = event_target:independence_wave_form07_autonomous_member_target } }
			give_guarantee = event_target:independence_wave_form07_autonomous_member_target
			event_target:independence_wave_form07_autonomous_member_target = { set_country_flag = independence_wave_form07_carrier_guaranteed_member }
		}
		set_country_flag = independence_wave_form07_has_autonomous_member
	}
}

# Carrier-country scope. The frozen member ledger remains the only iteration
# source. Full integration is opt-in through the existing Event 006 decision
# authorization; every other consenting CAT/NAV/GLC row stays autonomous.
independence_wave_formable_integration_adapter_7 = {
	if = {
		limit = {
			has_country_flag = independence_wave_formable_mutation_prevalidated
			has_country_flag = independence_wave_formable_identity_committed
			independence_wave_formable_member_arrays_are_aligned = yes
			has_independence_wave_form07_corridor_proof = yes
		}
		set_temp_variable = { independence_wave_form07_member_index = constant:independence_wave_formable_tuning.first_array_index }
		set_temp_variable = { independence_wave_form07_additional_members_processed = constant:independence_wave_formable_tuning.no_entries }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_form07_member_index value = independence_wave_formable_member_country_entries^num compare = less_than } }
			set_temp_variable = { independence_wave_form07_member_consent = independence_wave_formable_member_consent_entries^independence_wave_form07_member_index }
			var:independence_wave_formable_member_country_entries^independence_wave_form07_member_index = {
				if = {
					limit = {
						NOT = { tag = ROOT }
						is_independence_wave_form07_member_candidate = yes
						has_independence_wave_form07_exact_member_anchor = yes
						has_independence_wave_form07_connection_to_root = yes
						has_independence_wave_form07_declared_consent = yes
						check_variable = { independence_wave_form07_member_consent = constant:independence_wave_formable_consent.consenting_member }
					}
					if = {
						limit = { has_country_flag = independence_wave_form0124_full_integration_authorized }
						independence_wave_formable_end_consenting_member_origin = yes
						if = { limit = { original_tag = CAT owns_state = @FORM07_CAT_ANCHOR controls_state = @FORM07_CAT_ANCHOR } ROOT = { transfer_state = @FORM07_CAT_ANCHOR add_state_core = @FORM07_CAT_ANCHOR } }
						if = { limit = { original_tag = NAV owns_state = @FORM07_NAV_ANCHOR controls_state = @FORM07_NAV_ANCHOR } ROOT = { transfer_state = @FORM07_NAV_ANCHOR add_state_core = @FORM07_NAV_ANCHOR } }
						if = { limit = { original_tag = GLC owns_state = @FORM07_GLC_ANCHOR controls_state = @FORM07_GLC_ANCHOR } ROOT = { transfer_state = @FORM07_GLC_ANCHOR add_state_core = @FORM07_GLC_ANCHOR } }
					}
					else = { independence_wave_form07_install_autonomous_member = yes }
					ROOT = { add_to_temp_variable = { independence_wave_form07_additional_members_processed = constant:independence_wave_formable_tuning.loop_increment } }
				}
			}
			add_to_temp_variable = { independence_wave_form07_member_index = constant:independence_wave_formable_tuning.loop_increment }
		}
		if = {
			limit = { check_variable = { var = independence_wave_form07_additional_members_processed value = constant:independence_wave_form07.minimum_additional_members compare = greater_than_or_equals } }
			set_variable = { independence_wave_form07_bound_member_count = independence_wave_form07_additional_members_processed }
			set_country_flag = independence_wave_form07_integration_committed
			set_country_flag = independence_wave_formable_integration_committed
		}
	}
}

# Country scope. There is no cosmetic identity to drop yet; this rollback only
# clears the transaction-local blocker and any future global identity lock.
independence_wave_form07_rollback_identity = {
	if = {
		limit = { NOT = { has_country_flag = independence_wave_formable_integration_committed } }
		clr_global_flag = independence_wave_form07_identity_in_use
		clr_country_flag = independence_wave_form07_identity_adapter_blocked
		clr_country_flag = independence_wave_form07_federal_carrier
	}
}

# Autonomous-member country scope. Relations are removed only when their
# directional ownership flags were written by this adapter.
independence_wave_form07_remove_event6_autonomous_relations = {
	save_event_target_as = independence_wave_form07_cleanup_member
	if = {
		limit = {
			has_variable = independence_wave_form07_autonomous_carrier
			var:independence_wave_form07_autonomous_carrier = { exists = yes }
		}
		var:independence_wave_form07_autonomous_carrier = { save_event_target_as = independence_wave_form07_cleanup_carrier }
		if = {
			limit = { has_country_flag = independence_wave_form07_member_granted_access_to_carrier }
			diplomatic_relation = { country = event_target:independence_wave_form07_cleanup_carrier relation = military_access active = no }
		}
		if = {
			limit = { has_country_flag = independence_wave_form07_member_guaranteed_carrier }
			diplomatic_relation = { country = event_target:independence_wave_form07_cleanup_carrier relation = guarantee active = no }
		}
		if = {
			limit = { has_country_flag = independence_wave_form07_carrier_granted_access_to_member }
			event_target:independence_wave_form07_cleanup_carrier = {
				diplomatic_relation = { country = event_target:independence_wave_form07_cleanup_member relation = military_access active = no }
			}
		}
		if = {
			limit = { has_country_flag = independence_wave_form07_carrier_guaranteed_member }
			event_target:independence_wave_form07_cleanup_carrier = {
				diplomatic_relation = { country = event_target:independence_wave_form07_cleanup_member relation = guarantee active = no }
			}
		}
	}
	clr_country_flag = independence_wave_form07_member_granted_access_to_carrier
	clr_country_flag = independence_wave_form07_member_guaranteed_carrier
	clr_country_flag = independence_wave_form07_carrier_granted_access_to_member
	clr_country_flag = independence_wave_form07_carrier_guaranteed_member
	clear_variable = independence_wave_form07_autonomous_carrier
	clear_variable = independence_wave_form07_autonomous_carrier_generation
}

independence_wave_form07_clear_member_state = {
	independence_wave_form07_remove_event6_autonomous_relations = yes
	clr_country_flag = independence_wave_form07_autonomous_member
	clr_country_flag = independence_wave_form0124_full_integration_authorized
	clear_variable = independence_wave_form07_autonomous_family
}

# Carrier-country scope. Cleanup traverses only the frozen FORM-07 ledger and
# validates generation/family pointers before touching an autonomous member.
independence_wave_form07_cleanup_frozen_autonomous_members = {
	if = {
		limit = { independence_wave_formable_member_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_form07_cleanup_member_index = constant:independence_wave_formable_tuning.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_form07_cleanup_member_index value = independence_wave_formable_member_country_entries^num compare = less_than } }
			var:independence_wave_formable_member_country_entries^independence_wave_form07_cleanup_member_index = {
				if = {
					limit = {
						has_country_flag = independence_wave_form07_autonomous_member
						has_variable = independence_wave_form07_autonomous_carrier
						has_variable = independence_wave_form07_autonomous_carrier_generation
						var:independence_wave_form07_autonomous_carrier = { tag = ROOT }
						check_variable = { var = independence_wave_form07_autonomous_carrier_generation value = ROOT.independence_wave_generation_id compare = equals }
						check_variable = { var = independence_wave_form07_autonomous_family value = ROOT.independence_wave_formable_family compare = equals }
					}
					independence_wave_form07_clear_member_state = yes
				}
			}
			add_to_temp_variable = { independence_wave_form07_cleanup_member_index = constant:independence_wave_formable_tuning.loop_increment }
		}
	}
}

independence_wave_form07_cleanup_runtime = {
	if = {
		limit = { has_country_flag = independence_wave_form07_autonomous_member }
		independence_wave_form07_clear_member_state = yes
	}
	independence_wave_form07_cleanup_frozen_autonomous_members = yes
	independence_wave_form07_rollback_identity = yes
	clr_country_flag = independence_wave_form07_has_autonomous_member
	clr_country_flag = independence_wave_form07_identity_adapter_blocked
	clr_country_flag = independence_wave_form07_federal_carrier
	clr_country_flag = independence_wave_form07_integration_committed
	clr_country_flag = independence_wave_form07_readiness_attested
	clr_global_flag = independence_wave_form07_identity_in_use
	clear_variable = independence_wave_form07_bound_member_count
}
