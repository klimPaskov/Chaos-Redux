# Wider persecution and occupation consequences: finite noncustody cohorts, actual receipts,
# immutable authority records, conservative outside-loss reconciliation and bounded processing.
# Public contract, historical sources and limitations: camp_administration_occupation_effects.md.

camp_occ_initialize_country = {
	if = {
		limit = { NOT = { has_country_flag = camp_occ_initialized } }
		set_country_flag = camp_occ_initialized
		clear_array = camp_occ_states
		clear_array = camp_occ_eligible_states
		camp_occ_initialize_records = yes
	}
}
camp_occ_initialize_records = {
	set_variable = { camp_occ_total_deaths_k = 0 }
	set_variable = { camp_occ_month_deaths_k = 0 }
	set_variable = { camp_occ_total_campaign_violence_k = 0 }
	set_variable = { camp_occ_month_campaign_violence_k = 0 }
	set_variable = { camp_occ_total_deprivation_k = 0 }
	set_variable = { camp_occ_month_deprivation_k = 0 }
	set_variable = { camp_occ_total_exposure_removed_k = 0 }
	set_variable = { camp_occ_month_exposure_removed_k = 0 }
	set_variable = { camp_occ_total_released_k = 0 }
}
camp_occ_reset_month_records = {
	set_variable = { camp_occ_month_deaths_k = 0 }
	set_variable = { camp_occ_month_campaign_violence_k = 0 }
	set_variable = { camp_occ_month_deprivation_k = 0 }
	set_variable = { camp_occ_month_exposure_removed_k = 0 }
}

# State; one-shot explicit source contract, never repopulates an exhausted or closed cohort.
camp_occ_register_cohort = {
	set_temp_variable = { camp_occ_admitted_k = 0 }
	if = {
		limit = {
			check_variable = { camp_occ_source_proven = 1 }
			check_variable = { camp_occ_source_requested_k > 0 }
			check_variable = { camp_occ_source_membership_k > 0 }
			has_variable = camp_occ_source_authority
			NOT = { has_state_flag = camp_occ_registered }
			is_controlled_by = var:camp_occ_source_authority
			CONTROLLER = { camp_occ_authority_has_campaign = yes }
			OR = {
				AND = { check_variable = { camp_occ_source_profile = constant:camp_occ_profile.japan } camp_occ_state_is_china_region = yes CONTROLLER = { original_tag = JAP } }
				AND = { check_variable = { camp_occ_source_profile = constant:camp_occ_profile.germany } has_state_flag = camp_occ_origin_assigned CONTROLLER = { original_tag = GER } }
			}
		}
		set_temp_variable = { camp_occ_admitted_k = state_population_k }
		subtract_from_temp_variable = { camp_occ_admitted_k = constant:camp_occ_value.minimum_population_k }
		if = { limit = { has_variable = camp_admin_detainees_k } subtract_from_temp_variable = { camp_occ_admitted_k = camp_admin_detainees_k } }
		clamp_temp_variable = { var = camp_occ_admitted_k min = 0 max = camp_occ_source_membership_k }
		clamp_temp_variable = { var = camp_occ_admitted_k min = 0 max = camp_occ_source_requested_k }
		if = {
			limit = { check_variable = { camp_occ_admitted_k > 0 } }
			set_state_flag = camp_occ_registered
			set_variable = { camp_occ_responsible_country = camp_occ_source_authority }
			set_variable = { camp_occ_profile_id = camp_occ_source_profile }
			set_variable = { camp_occ_initial_k = camp_occ_admitted_k }
			set_variable = { camp_occ_remaining_k = camp_occ_admitted_k }
			set_variable = { camp_occ_observed_population_k = state_population_k }
			set_variable = { camp_occ_observed_state_deaths = chaos_state_civilian_deaths_total }
			camp_occ_initialize_records = yes
			var:camp_occ_responsible_country = {
				camp_occ_initialize_country = yes
				add_to_array = { array = camp_occ_states value = PREV }
				camp_rework_register_country_for_monthly_pulse = yes
			}
		}
	}
	set_temp_variable = { camp_occ_source_proven = 0 }
	set_temp_variable = { camp_occ_source_requested_k = 0 }
	set_temp_variable = { camp_occ_source_membership_k = 0 }
	set_temp_variable = { camp_occ_source_authority = 0 }
	set_temp_variable = { camp_occ_source_profile = 0 }
}

# State: outside declines remove exposure conservatively; never assign deaths or blame from them.
camp_occ_reconcile_external_population = {
	set_temp_variable = { camp_occ_external_removed_k = camp_occ_observed_population_k }
		subtract_from_temp_variable = { camp_occ_external_removed_k = state_population_k }
		# Net population can hide recorded deaths behind growth or inward transfers.
		# This all-cause delta removes exposure only; it never establishes a perpetrator.
		set_temp_variable = { camp_occ_external_recorded_k = chaos_state_civilian_deaths_total }
		subtract_from_temp_variable = { camp_occ_external_recorded_k = camp_occ_observed_state_deaths }
		divide_temp_variable = { camp_occ_external_recorded_k = constant:camp_occ_value.people_per_k }
		clamp_temp_variable = { var = camp_occ_external_recorded_k min = 0 }
		clamp_temp_variable = { var = camp_occ_external_removed_k min = camp_occ_external_recorded_k }
		# Revalidate current noncustody membership even if another owner changed reservations.
		set_temp_variable = { camp_occ_current_headroom_k = state_population_k }
		subtract_from_temp_variable = { camp_occ_current_headroom_k = constant:camp_occ_value.minimum_population_k }
		if = { limit = { has_variable = camp_admin_detainees_k } subtract_from_temp_variable = { camp_occ_current_headroom_k = camp_admin_detainees_k } }
		clamp_temp_variable = { var = camp_occ_current_headroom_k min = 0 }
		set_temp_variable = { camp_occ_membership_shortfall_k = camp_occ_remaining_k }
		subtract_from_temp_variable = { camp_occ_membership_shortfall_k = camp_occ_current_headroom_k }
		clamp_temp_variable = { var = camp_occ_membership_shortfall_k min = 0 }
		clamp_temp_variable = { var = camp_occ_external_removed_k min = camp_occ_membership_shortfall_k }
		clamp_temp_variable = { var = camp_occ_external_removed_k min = 0 max = camp_occ_remaining_k }
		subtract_from_variable = { camp_occ_remaining_k = camp_occ_external_removed_k }
		add_to_variable = { camp_occ_total_exposure_removed_k = camp_occ_external_removed_k }
		add_to_variable = { camp_occ_month_exposure_removed_k = camp_occ_external_removed_k }
		var:camp_occ_responsible_country = {
			add_to_variable = { camp_occ_total_exposure_removed_k = camp_occ_external_removed_k }
			add_to_variable = { camp_occ_month_exposure_removed_k = camp_occ_external_removed_k }
		}
		set_variable = { camp_occ_observed_population_k = state_population_k }
		set_variable = { camp_occ_observed_state_deaths = chaos_state_civilian_deaths_total }
}

# State: requested amount is never the receipt. Shared helper debits once with logging disabled.
camp_occ_record_deaths = {
	set_temp_variable = { camp_occ_actual_k = 0 }
	if = {
		limit = {
			camp_occ_state_can_pulse = yes
			check_variable = { camp_occ_loss_contract = 1 }
			check_variable = { camp_occ_requested_k > 0 }
			check_variable = { camp_occ_loss_cause = constant:camp_occ_cause.campaign_violence }
		}
		camp_occ_reconcile_external_population = yes
		set_temp_variable = { camp_occ_before_k = state_population_k }
		set_temp_variable = { camp_occ_request_k = camp_occ_requested_k }
		clamp_temp_variable = { var = camp_occ_request_k min = 0 max = camp_occ_remaining_k }
		clamp_temp_variable = { var = camp_occ_request_k min = 0 max = constant:camp_occ_value.maximum_receipt_k }
		set_temp_variable = { state_civilian_population_loss_requested = camp_occ_request_k }
		multiply_temp_variable = { state_civilian_population_loss_requested = constant:camp_occ_value.people_per_k }
		# The shared API rounds to people; never round above the finite cohort request.
		set_temp_variable = { camp_occ_request_people_ceiling = state_civilian_population_loss_requested }
		round_temp_variable = state_civilian_population_loss_requested
		if = {
			limit = { check_variable = { state_civilian_population_loss_requested > camp_occ_request_people_ceiling } }
			subtract_from_temp_variable = { state_civilian_population_loss_requested = constant:camp_occ_value.one_person }
		}
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:camp_occ_value.minimum_population_k }
		if = { limit = { has_variable = camp_admin_detainees_k } add_to_temp_variable = { state_civilian_population_loss_minimum_remaining = camp_admin_detainees_k } }
		multiply_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:camp_occ_value.people_per_k }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.occupation_repression }
		set_temp_variable = { state_civilian_population_loss_log_deaths = 0 }
		set_temp_variable = { state_civilian_population_loss_target_country = camp_occ_responsible_country }
		set_temp_variable = { state_civilian_population_loss_has_target_country = 1 }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = 1 }
		apply_exact_state_civilian_population_loss = yes
		set_temp_variable = { camp_occ_actual_k = camp_occ_before_k }
		subtract_from_temp_variable = { camp_occ_actual_k = state_population_k }
		clamp_temp_variable = { var = camp_occ_actual_k min = 0 }
		subtract_from_variable = { camp_occ_remaining_k = camp_occ_actual_k }
		set_variable = { camp_occ_observed_population_k = state_population_k }
		if = {
			limit = { check_variable = { camp_occ_actual_k > 0 } }
			set_temp_variable = { chaos_deaths_change = camp_occ_actual_k }
			multiply_temp_variable = { chaos_deaths_change = constant:camp_occ_value.people_per_k }
			set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.occupation_repression }
			set_temp_variable = { chaos_deaths_is_civilian = 1 }
			set_temp_variable = { chaos_deaths_apply_state_pop = 0 }
			set_temp_variable = { chaos_deaths_record_state_ledger = 1 }
			set_temp_variable = { chaos_deaths_target_country = camp_occ_responsible_country }
			set_temp_variable = { chaos_deaths_has_target_country = 1 }
			chaos_meter_register_deaths = yes
			# Advance after projection so the next reconciliation excludes this exact own receipt.
			set_variable = { camp_occ_observed_state_deaths = chaos_state_civilian_deaths_total }
			camp_occ_project_receipt = yes
			var:camp_occ_responsible_country = { camp_occ_project_receipt = yes }
		}
	}
	set_temp_variable = { camp_occ_loss_contract = 0 }
	set_temp_variable = { camp_occ_requested_k = 0 }
	set_temp_variable = { camp_occ_loss_cause = 0 }
}
camp_occ_project_receipt = {
	add_to_variable = { camp_occ_month_deaths_k = camp_occ_actual_k }
	add_to_variable = { camp_occ_total_deaths_k = camp_occ_actual_k }
	add_to_variable = { camp_occ_month_campaign_violence_k = camp_occ_actual_k }
	add_to_variable = { camp_occ_total_campaign_violence_k = camp_occ_actual_k }
}

camp_occ_monthly_country = {
	camp_occ_initialize_country = yes
	camp_admin_prepare_month_key = yes
	if = {
		limit = { NOT = { check_variable = { camp_occ_last_month = camp_admin_month_key } } }
		set_variable = { camp_occ_last_month = camp_admin_month_key }
		camp_occ_reset_month_records = yes
		for_each_scope_loop = { array = camp_occ_eligible_states camp_occ_try_register_current_state = yes }
		for_each_scope_loop = {
			array = camp_occ_states
			camp_occ_reset_month_records = yes
			camp_occ_reconcile_external_population = yes
			if = { limit = { NOT = { camp_occ_state_has_current_authority = yes } } camp_occ_close_state = yes }
			if = {
				limit = { camp_occ_state_can_pulse = yes }
				set_temp_variable = { camp_occ_requested_k = camp_occ_remaining_k }
				if = {
					limit = { check_variable = { camp_occ_profile_id = constant:camp_occ_profile.japan } }
					multiply_temp_variable = { camp_occ_requested_k = constant:camp_occ_value.japan_campaign_monthly_fraction }
				}
				else = { multiply_temp_variable = { camp_occ_requested_k = constant:camp_occ_value.german_campaign_monthly_fraction } }
				set_temp_variable = { camp_occ_loss_cause = constant:camp_occ_cause.campaign_violence }
				set_temp_variable = { camp_occ_loss_contract = 1 }
				camp_occ_record_deaths = yes
			}
		}
	}
}

# Closure releases surviving exposure in place and permanently prevents reseeding this state row.
camp_occ_close_state = {
	if = {
		limit = { has_state_flag = camp_occ_registered NOT = { has_state_flag = camp_occ_closed } }
		camp_occ_reconcile_external_population = yes
		set_temp_variable = { camp_occ_release_k = camp_occ_remaining_k }
		add_to_variable = { camp_occ_total_released_k = camp_occ_release_k }
		var:camp_occ_responsible_country = { add_to_variable = { camp_occ_total_released_k = camp_occ_release_k } }
		set_variable = { camp_occ_remaining_k = 0 }
		set_state_flag = camp_occ_closed
	}
}
camp_occ_close_country = {
	set_country_flag = camp_occ_country_closed
	for_each_scope_loop = { array = camp_occ_states camp_occ_close_state = yes }
}

# Country of origin, one-time source snapshot. The denominator includes ALL its European owned
# states, so partial occupation cannot concentrate a whole national cohort into the occupied share.
camp_occ_seed_origin = {
	if = {
		limit = { NOT = { has_country_flag = camp_occ_origin_seeded } }
		set_country_flag = camp_occ_origin_seeded
		set_variable = { camp_occ_origin_initial_ceiling_k = camp_occ_origin_ceiling_input_k }
		set_variable = { camp_occ_origin_unassigned_k = camp_occ_origin_initial_ceiling_k }
		set_temp_variable = { camp_occ_origin_population_k = 0 }
		every_owned_state = {
			limit = { is_on_continent = europe }
			add_to_temp_variable = { camp_occ_origin_population_k = state_population_k }
		}
		if = {
			limit = { check_variable = { camp_occ_origin_population_k > 0 } }
			divide_temp_variable = { camp_occ_origin_population_k = constant:camp_occ_value.people_per_k }
			every_owned_state = {
				limit = { is_on_continent = europe NOT = { has_state_flag = camp_occ_origin_assigned } }
				set_state_flag = camp_occ_origin_assigned
				set_variable = { camp_occ_origin_country = PREV }
				set_variable = { camp_occ_origin_observed_population_k = state_population_k }
				set_variable = { camp_occ_origin_observed_state_deaths = chaos_state_civilian_deaths_total }
				set_temp_variable = { camp_occ_origin_share = state_population_k }
				divide_temp_variable = { camp_occ_origin_share = camp_occ_origin_population_k }
				set_variable = { camp_occ_origin_ceiling_k = PREV.camp_occ_origin_initial_ceiling_k }
				divide_variable = { camp_occ_origin_ceiling_k = constant:camp_occ_value.people_per_k }
				multiply_variable = { camp_occ_origin_ceiling_k = camp_occ_origin_share }
				clamp_variable = { var = camp_occ_origin_ceiling_k min = 0 max = PREV.camp_occ_origin_unassigned_k }
				set_temp_variable = { camp_occ_origin_assigned_k = camp_occ_origin_ceiling_k }
				PREV = { subtract_from_variable = { camp_occ_origin_unassigned_k = camp_occ_origin_assigned_k } }
			}
		}
	}
}

# State; event-driven registry insertion, distinct from population admission.
camp_occ_track_controlled_state = {
	if = {
		limit = {
			OR = {
				AND = { has_state_flag = camp_occ_origin_assigned CONTROLLER = { original_tag = GER } }
				AND = { camp_occ_state_is_china_region = yes CONTROLLER = { original_tag = JAP } }
			}
		}
		CONTROLLER = {
			camp_occ_initialize_country = yes
			if = {
				limit = { NOT = { is_in_array = { array = camp_occ_eligible_states value = PREV } } }
				add_to_array = { array = camp_occ_eligible_states value = PREV }
			}
			camp_rework_register_country_for_monthly_pulse = yes
		}
		camp_occ_try_register_current_state = yes
	}
}
camp_occ_try_register_current_state = {
	if = {
		limit = {
			NOT = { has_state_flag = camp_occ_registered }
			CONTROLLER = { camp_occ_authority_has_campaign = yes }
		}
		set_temp_variable = { camp_occ_source_authority = CONTROLLER }
		set_temp_variable = { camp_occ_source_proven = 0 }
		set_temp_variable = { camp_occ_source_requested_k = 0 }
		if = {
			limit = { has_state_flag = camp_occ_origin_assigned CONTROLLER = { original_tag = GER } }
			set_temp_variable = { camp_occ_source_requested_k = camp_occ_origin_ceiling_k }
			set_temp_variable = { camp_occ_pre_enrollment_decline_k = camp_occ_origin_observed_population_k }
			subtract_from_temp_variable = { camp_occ_pre_enrollment_decline_k = state_population_k }
			set_temp_variable = { camp_occ_pre_enrollment_recorded_k = chaos_state_civilian_deaths_total }
			subtract_from_temp_variable = { camp_occ_pre_enrollment_recorded_k = camp_occ_origin_observed_state_deaths }
			divide_temp_variable = { camp_occ_pre_enrollment_recorded_k = constant:camp_occ_value.people_per_k }
			clamp_temp_variable = { var = camp_occ_pre_enrollment_recorded_k min = 0 }
			clamp_temp_variable = { var = camp_occ_pre_enrollment_decline_k min = camp_occ_pre_enrollment_recorded_k }
			clamp_temp_variable = { var = camp_occ_pre_enrollment_decline_k min = 0 max = camp_occ_source_requested_k }
			subtract_from_temp_variable = { camp_occ_source_requested_k = camp_occ_pre_enrollment_decline_k }
			set_temp_variable = { camp_occ_source_profile = constant:camp_occ_profile.germany }
			set_temp_variable = { camp_occ_source_proven = 1 }
		}
		else_if = {
			limit = { camp_occ_state_is_china_region = yes CONTROLLER = { original_tag = JAP } }
			# Snapshot actual present civilians; no claim that this fraction is a historical census.
			set_temp_variable = { camp_occ_source_requested_k = state_population_k }
			multiply_temp_variable = { camp_occ_source_requested_k = constant:camp_occ_value.japan_exposure_share }
			set_temp_variable = { camp_occ_source_profile = constant:camp_occ_profile.japan }
			set_temp_variable = { camp_occ_source_proven = 1 }
		}
		set_temp_variable = { camp_occ_source_membership_k = camp_occ_source_requested_k }
		camp_occ_register_cohort = yes
	}
}
camp_occ_on_state_control_changed = {
	if = {
		limit = { has_state_flag = camp_occ_registered NOT = { camp_occ_state_has_current_authority = yes } }
		camp_occ_close_state = yes
	}
	camp_occ_track_controlled_state = yes
}
