# Country-owned sparse rows. All arrays are append-only and aligned; IDs, not indexes, identify cohorts.
# State camp_occ_remaining_k remains the aggregate live reservation read by custody admission.
camp_occ_initialize_rows = {
	if = {
		limit = { NOT = { has_country_flag = camp_occ_rows_initialized } }
		set_country_flag = camp_occ_rows_initialized
		clear_array = camp_occ_row_ids
		clear_array = camp_occ_row_root_ids
		clear_array = camp_occ_row_origins
		clear_array = camp_occ_row_hosts
		clear_array = camp_occ_row_authorities
		clear_array = camp_occ_row_profiles
		clear_array = camp_occ_row_amounts_k
		clear_array = camp_occ_row_deaths_k
		clear_array = camp_occ_row_removed_k
		clear_array = camp_occ_row_released_k
		clear_array = camp_occ_row_moved_out_k
		clear_array = camp_occ_row_moved_in_k
		clear_array = camp_occ_row_statuses
		set_variable = { camp_occ_next_row_id = constant:camp_occ_value.row_first_id }
		for_each_scope_loop = { array = camp_occ_states camp_occ_adopt_legacy_state_row = yes }
	}
}

# Country. Inputs are one-shot temporary scope/value tokens staged by a state caller.
camp_occ_append_row = {
	set_temp_variable = { camp_occ_appended_row_id = camp_occ_next_row_id }
	if = { limit = { check_variable = { camp_occ_append_root_id = 0 } } set_temp_variable = { camp_occ_append_root_id = camp_occ_appended_row_id } }
	add_to_array = { array = camp_occ_row_ids value = camp_occ_appended_row_id }
	add_to_array = { array = camp_occ_row_root_ids value = camp_occ_append_root_id }
	add_to_array = { array = camp_occ_row_origins value = camp_occ_append_origin }
	add_to_array = { array = camp_occ_row_hosts value = camp_occ_append_host }
	add_to_array = { array = camp_occ_row_authorities value = THIS }
	add_to_array = { array = camp_occ_row_profiles value = camp_occ_append_profile }
	add_to_array = { array = camp_occ_row_amounts_k value = camp_occ_append_amount_k }
	add_to_array = { array = camp_occ_row_deaths_k value = 0 }
	add_to_array = { array = camp_occ_row_removed_k value = 0 }
	add_to_array = { array = camp_occ_row_released_k value = 0 }
	add_to_array = { array = camp_occ_row_moved_out_k value = 0 }
	add_to_array = { array = camp_occ_row_moved_in_k value = 0 }

	add_to_variable = { camp_occ_next_row_id = constant:camp_occ_value.row_index_step }
	set_variable = { camp_occ_row_count = camp_occ_row_ids^num }
}

# State. The old state-keyed row is the sole cohort in a pre-row country.
camp_occ_adopt_legacy_state_row = {
	camp_occ_backfill_cohort_origin = yes
	if = {
		limit = { has_state_flag = camp_occ_registered NOT = { has_variable = camp_occ_native_row_id } }
		set_temp_variable = { camp_occ_append_origin = camp_occ_cohort_origin_state }
		set_temp_variable = { camp_occ_append_host = THIS }
		set_temp_variable = { camp_occ_append_profile = camp_occ_profile_id }
		set_temp_variable = { camp_occ_append_amount_k = camp_occ_remaining_k }
		set_temp_variable = { camp_occ_append_root_id = 0 }
		set_temp_variable = { camp_occ_legacy_deaths_k = camp_occ_total_deaths_k }
		set_temp_variable = { camp_occ_legacy_removed_k = camp_occ_total_exposure_removed_k }
		set_temp_variable = { camp_occ_legacy_released_k = camp_occ_total_released_k }
		set_temp_variable = { camp_occ_legacy_closed = 0 }
		if = { limit = { has_state_flag = camp_occ_closed } set_temp_variable = { camp_occ_legacy_closed = 1 } }
		var:camp_occ_responsible_country = {
			camp_occ_append_row = yes
			set_temp_variable = { camp_occ_legacy_row_index = camp_occ_row_count }
			subtract_from_temp_variable = { camp_occ_legacy_row_index = constant:camp_occ_value.row_index_step }
			set_variable = { camp_occ_row_deaths_k^camp_occ_legacy_row_index = camp_occ_legacy_deaths_k }
			set_variable = { camp_occ_row_removed_k^camp_occ_legacy_row_index = camp_occ_legacy_removed_k }
			set_variable = { camp_occ_row_released_k^camp_occ_legacy_row_index = camp_occ_legacy_released_k }
			if = { limit = { check_variable = { camp_occ_legacy_closed = 1 } } set_variable = { camp_occ_row_statuses^camp_occ_legacy_row_index = constant:camp_occ_row_status.released } }
		}
		set_variable = { camp_occ_native_row_id = camp_occ_appended_row_id }
	}
}

# State; one-shot explicit source contract, never repopulates an exhausted or closed cohort.
camp_occ_register_cohort = {
	set_temp_variable = { camp_occ_admitted_k = 0 }
	camp_origin_reconcile_host_membership = yes
	if = {
		limit = {
			camp_origin_host_provenance_ready = yes
			check_variable = { camp_occ_source_proven = 1 }
			check_variable = { camp_occ_source_requested_k > 0 }
			check_variable = { camp_occ_source_membership_k > 0 }
			has_variable = camp_occ_source_authority
			NOT = { has_state_flag = camp_occ_registered }
			OR = {
				AND = {
					OR = {
						AND = {
							check_variable = { camp_occ_source_profile = constant:camp_occ_profile.japan }
							camp_occ_state_is_china_region = yes
							camp_occ_state_has_japanese_client_authority = yes
							var:camp_occ_source_authority = { original_tag = JAP camp_occ_authority_has_campaign = yes }
						}
						AND = {
							check_variable = { camp_occ_source_profile = constant:camp_occ_profile.germany }
							camp_occ_has_foreign_german_campaign_origin = yes
							camp_occ_state_has_german_authority = yes
							var:camp_occ_source_authority = { original_tag = GER camp_occ_authority_has_campaign = yes }
						}
					}
				}
				# Dated partner deportation: the German authority directs it while
				# the dominated partner keeps control of the state.
				AND = {
					check_variable = { camp_occ_source_partner = constant:camp_occ_value.transfer_valid }
					check_variable = { camp_occ_source_profile = constant:camp_occ_profile.germany }
					camp_occ_partner_deportation_candidate = yes
					var:camp_occ_source_authority = { original_tag = GER camp_occ_authority_has_campaign = yes }
				}
			}
		}
		set_temp_variable = { camp_occ_admitted_k = state_population_k }
		subtract_from_temp_variable = { camp_occ_admitted_k = constant:camp_occ_value.minimum_population_k }
		if = { limit = { has_variable = camp_admin_detainees_k } subtract_from_temp_variable = { camp_occ_admitted_k = camp_admin_detainees_k } }
		clamp_temp_variable = { var = camp_occ_admitted_k min = 0 max = camp_occ_source_membership_k }
		clamp_temp_variable = { var = camp_occ_admitted_k min = 0 max = camp_occ_source_requested_k }
		if = {
			limit = { check_variable = { camp_occ_admitted_k > 0 } }
			set_state_flag = camp_occ_registered
			if = { limit = { check_variable = { camp_occ_source_partner = constant:camp_occ_value.transfer_valid } } set_state_flag = camp_occ_partner_host }
			set_variable = { camp_occ_responsible_country = camp_occ_source_authority }
			# Stable provenance for the cohort even if a later approved route moves survivors.
			set_variable = { camp_occ_cohort_origin_state = THIS }
			set_variable = { camp_occ_profile_id = camp_occ_source_profile }
			set_variable = { camp_occ_initial_k = camp_occ_admitted_k }
			set_variable = { camp_occ_remaining_k = camp_occ_admitted_k }
			set_variable = { camp_occ_observed_population_k = state_population_k }
			set_variable = { camp_occ_observed_state_deaths = chaos_state_civilian_deaths_total }
			camp_occ_initialize_records = yes
			set_temp_variable = { camp_occ_append_origin = THIS }
			set_temp_variable = { camp_occ_append_host = THIS }
			set_temp_variable = { camp_occ_append_profile = camp_occ_profile_id }
			set_temp_variable = { camp_occ_append_amount_k = camp_occ_admitted_k }
			set_temp_variable = { camp_occ_append_root_id = 0 }
			var:camp_occ_responsible_country = {
				camp_occ_initialize_country = yes
				camp_occ_initialize_rows = yes
				camp_occ_append_row = yes
				add_to_array = { array = camp_occ_states value = PREV }
				camp_rework_register_country_for_monthly_pulse = yes
			}
			set_variable = { camp_occ_native_row_id = camp_occ_appended_row_id }
			camp_origin_observe_host = yes
			var:camp_occ_responsible_country = {
				if = { limit = { original_tag = GER } camp_occ_refresh_manual_transfer_source = yes }
			}
		}
	}
	set_temp_variable = { camp_occ_source_proven = 0 }
	set_temp_variable = { camp_occ_source_requested_k = 0 }
	set_temp_variable = { camp_occ_source_membership_k = 0 }
	set_temp_variable = { camp_occ_source_authority = 0 }
	set_temp_variable = { camp_occ_source_profile = 0 }
	set_temp_variable = { camp_occ_source_partner = 0 }
}

# State. Positive physical credit with owner/controller recruitable-manpower reconciliation.
# Input camp_occ_credit_request_people; output camp_occ_credit_actual_people.
camp_occ_apply_population_credit = {
	set_temp_variable = { camp_occ_credit_actual_people = 0 }
	if = {
		limit = { check_variable = { camp_occ_credit_request_people > 0 } }
		set_temp_variable = { camp_occ_credit_before_people = state_population_k }
		multiply_temp_variable = { camp_occ_credit_before_people = constant:camp_occ_value.people_per_k }
		round_temp_variable = camp_occ_credit_before_people
		OWNER = { set_temp_variable = { camp_occ_credit_owner_before_k = manpower_k } }
		if = { limit = { NOT = { is_owned_and_controlled_by = OWNER } } CONTROLLER = { set_temp_variable = { camp_occ_credit_controller_before_k = manpower_k } } }
		add_manpower = var:camp_occ_credit_request_people
		set_temp_variable = { camp_occ_credit_actual_people = state_population_k }
		multiply_temp_variable = { camp_occ_credit_actual_people = constant:camp_occ_value.people_per_k }
		round_temp_variable = camp_occ_credit_actual_people
		subtract_from_temp_variable = { camp_occ_credit_actual_people = camp_occ_credit_before_people }
		clamp_temp_variable = { var = camp_occ_credit_actual_people min = 0 }
		OWNER = {
			set_temp_variable = { camp_occ_credit_owner_gain_people = manpower_k }
			subtract_from_temp_variable = { camp_occ_credit_owner_gain_people = camp_occ_credit_owner_before_k }
			multiply_temp_variable = { camp_occ_credit_owner_gain_people = constant:camp_occ_value.people_per_k }
			round_temp_variable = camp_occ_credit_owner_gain_people
			clamp_temp_variable = { var = camp_occ_credit_owner_gain_people min = 0 max = camp_occ_credit_actual_people }
			if = {
				limit = { check_variable = { camp_occ_credit_owner_gain_people > 0 } }
				set_temp_variable = { camp_occ_credit_owner_correction = camp_occ_credit_owner_gain_people }
				multiply_temp_variable = { camp_occ_credit_owner_correction = constant:camp_occ_value.negative_one }
				add_manpower = var:camp_occ_credit_owner_correction
			}
		}
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = {
				set_temp_variable = { camp_occ_credit_controller_gain_people = manpower_k }
				subtract_from_temp_variable = { camp_occ_credit_controller_gain_people = camp_occ_credit_controller_before_k }
				multiply_temp_variable = { camp_occ_credit_controller_gain_people = constant:camp_occ_value.people_per_k }
				round_temp_variable = camp_occ_credit_controller_gain_people
				clamp_temp_variable = { var = camp_occ_credit_controller_gain_people min = 0 max = camp_occ_credit_actual_people }
				if = {
					limit = { check_variable = { camp_occ_credit_controller_gain_people > 0 } }
					set_temp_variable = { camp_occ_credit_controller_correction = camp_occ_credit_controller_gain_people }
					multiply_temp_variable = { camp_occ_credit_controller_correction = constant:camp_occ_value.negative_one }
					add_manpower = var:camp_occ_credit_controller_correction
				}
			}
		}
	}
}

				camp_origin_quarantine_transaction = yes
			}
		}
		else = {

		}
	}

	set_temp_variable = { camp_occ_transfer_escape = 0 }

# Source state. Inverse the exact custody bind before undoing either physical population effect.
# A failed inverse is quarantined rather than deleting population underneath a live custody row.
camp_occ_rollback_custody_transaction = {
	set_temp_variable = { camp_occ_custody_can_undo_physical = constant:camp_occ_value.transfer_valid }
	if = {
		limit = { check_variable = { camp_admin_import_bind_pending_inverse = constant:camp_occ_value.transfer_valid } }
		event_target:camp_occ_custody_destination = { camp_admin_inverse_import_bind = yes }
		set_temp_variable = { camp_occ_custody_inverse_people = camp_admin_import_inverse_accepted_people }
		if = {
			limit = { NOT = { check_variable = { camp_occ_custody_inverse_people = camp_occ_custody_requested_people } } }
			set_temp_variable = { camp_occ_custody_can_undo_physical = constant:camp_occ_value.transfer_invalid }
		}
	}
	else_if = {
		limit = { OR = { check_variable = { camp_occ_custody_bind_people > 0 } check_variable = { camp_occ_custody_bound_row_id > 0 } } }
		set_temp_variable = { camp_occ_custody_can_undo_physical = constant:camp_occ_value.transfer_invalid }
	}
	if = {
		limit = { check_variable = { camp_occ_custody_can_undo_physical = constant:camp_occ_value.transfer_valid } }
		if = {
			limit = { check_variable = { camp_occ_custody_destination_credit_people > 0 } }
			set_temp_variable = { camp_occ_debit_request_people = camp_occ_custody_destination_credit_people }
			event_target:camp_occ_custody_destination = { camp_occ_apply_population_debit = yes }
			set_temp_variable = { camp_occ_custody_destination_undone_people = camp_occ_debit_actual_people }
		}
		if = {
			limit = { check_variable = { camp_occ_custody_source_debit_people > 0 } }
			set_temp_variable = { camp_occ_credit_request_people = camp_occ_custody_source_debit_people }
			camp_occ_apply_population_credit = yes
			set_temp_variable = { camp_occ_custody_source_restored_people = camp_occ_credit_actual_people }
		}
		if = {
			limit = {
				check_variable = { camp_occ_custody_destination_undone_people = camp_occ_custody_destination_credit_people }
				check_variable = { camp_occ_custody_source_restored_people = camp_occ_custody_source_debit_people }
			}
			set_temp_variable = { camp_occ_custody_rollback_complete = constant:camp_occ_value.transfer_valid }
		}
	}
	if = { limit = { check_variable = { camp_occ_custody_rollback_complete = constant:camp_occ_value.transfer_invalid } } set_state_flag = camp_occ_custody_rollback_failed }
}

# subset of the existing killing-center cause and never a transfer death.
camp_occ_apply_killing_arrival_loss = {
	set_temp_variable = { camp_occ_arrival_actual_k = 0 }
	if = {
		limit = {
			check_variable = { camp_occ_custody_result = constant:camp_occ_value.transfer_valid }
			check_variable = { camp_occ_custody_committed_people > 0 }
			check_variable = { camp_occ_custody_bound_row_id > 0 }
			camp_occ_authority_has_campaign = yes
			has_event_target = camp_occ_custody_destination
			NOT = { has_global_flag = settings_chaos_deaths_disabled }
			event_target:camp_occ_custody_destination = { camp_occ_killing_destination_active = yes }
		}
		set_temp_variable = { camp_occ_arrival_death_request_people = camp_occ_custody_committed_people }
		multiply_temp_variable = { camp_occ_arrival_death_request_people = constant:camp_occ_arrival.violence_request_fraction }
		set_temp_variable = { camp_occ_arrival_death_request_ceiling_people = camp_occ_arrival_death_request_people }
		round_temp_variable = camp_occ_arrival_death_request_people
		if = {
			limit = { check_variable = { camp_occ_arrival_death_request_people > camp_occ_arrival_death_request_ceiling_people } }
			subtract_from_temp_variable = { camp_occ_arrival_death_request_people = constant:camp_occ_value.one_person }
		}
		if = {
			limit = { check_variable = { camp_occ_arrival_death_request_people > 0 } }
			set_temp_variable = { camp_admin_death_requested_k = camp_occ_arrival_death_request_people }
			divide_temp_variable = { camp_admin_death_requested_k = constant:camp_occ_value.people_per_k }
			set_temp_variable = { camp_admin_death_row_id = camp_occ_custody_bound_row_id }
			set_temp_variable = { camp_admin_death_cause = constant:camp_admin_cause.violence }
			set_temp_variable = { camp_admin_death_contract = constant:camp_admin_value.one }
			set_temp_variable = { camp_admin_arrival_phase_contract = constant:camp_admin_value.one }
			event_target:camp_occ_custody_destination = { camp_admin_record_deaths = yes }
			set_temp_variable = { camp_occ_arrival_actual_k = camp_admin_death_actual_k }
		}
	}
}

# Country scope. Reserve one share of a proven living Lodz row during the documented
# August-September 1944 window. The pending share is a movement ceiling, not a death quota.
# Supported monthly throughput and resident capacity are recalculated for every wave.
camp_occ_try_lodz_auschwitz_transfer = {
	set_temp_variable = { camp_occ_custody_result = constant:camp_occ_value.transfer_invalid }
	if = {
		limit = {
			date > 1944.9.30
			has_variable = camp_occ_lodz_pending_people
		}
		clear_variable = camp_occ_lodz_pending_people
		clear_variable = camp_occ_lodz_source_row_id
	}
	if = {
		limit = {
			original_tag = GER
			date > 1944.8.8
			date < 1944.10.1
			camp_occ_authority_has_campaign = yes
			87 = {
				camp_occ_state_has_current_authority = yes
				has_variable = camp_occ_native_row_id
				check_variable = { camp_occ_remaining_k > 0 }
			}
			88 = { camp_occ_killing_destination_active = yes }
		}
		if = {
			limit = {
				NOT = { has_country_flag = camp_occ_lodz_auschwitz_transfer_recorded }
				NOT = { has_variable = camp_occ_lodz_pending_people }
			}
			87 = {
				set_temp_variable = { camp_occ_custody_row_id = camp_occ_native_row_id }
				set_temp_variable = { camp_occ_custody_requested_people = constant:camp_occ_value.one_person }

						subtract_from_temp_variable = { camp_occ_lodz_pending_request_people = constant:camp_occ_value.one_person }
					}
					if = {
						limit = { check_variable = { camp_occ_lodz_pending_request_people > 0 } }

							set_variable = { camp_occ_lodz_pending_people = camp_occ_lodz_pending_request_people }
							set_variable = { camp_occ_lodz_source_row_id = PREV.camp_occ_native_row_id }
							set_country_flag = camp_occ_lodz_auschwitz_transfer_recorded
						}
					}
				}
			}
		}
		if = {
			limit = { check_variable = { camp_occ_lodz_pending_people > 0 } }
			88 = {
				set_temp_variable = { camp_occ_lodz_monthly_throughput_people = non_damaged_building_level@extermination_camp }
				multiply_temp_variable = { camp_occ_lodz_monthly_throughput_people = constant:camp_occ_arrival.throughput_per_supported_level_k }
				multiply_temp_variable = { camp_occ_lodz_monthly_throughput_people = camp_admin_support_ratio }
				multiply_temp_variable = { camp_occ_lodz_monthly_throughput_people = constant:camp_occ_value.people_per_k }
			}
			set_temp_variable = { camp_occ_lodz_batch_count = 0 }
			set_temp_variable = { camp_occ_lodz_monthly_moved_people = 0 }
			set_temp_variable = { camp_occ_lodz_continue = constant:camp_occ_value.transfer_valid }
			while_loop_effect = {
				limit = {
					check_variable = { camp_occ_lodz_continue = constant:camp_occ_value.transfer_valid }
					check_variable = { camp_occ_lodz_pending_people > 0 }
					check_variable = { camp_occ_lodz_monthly_throughput_people > 0 }
					check_variable = { camp_occ_lodz_batch_count < constant:camp_occ_arrival.max_batches_per_month }
					87 = { camp_occ_state_has_current_authority = yes }
					88 = { camp_occ_killing_destination_active = yes }
				}
				add_to_temp_variable = { camp_occ_lodz_batch_count = constant:camp_occ_value.row_index_step }
				88 = {
					camp_admin_refresh_capacity = yes
					set_temp_variable = { camp_occ_lodz_free_capacity_k = camp_admin_capacity_k }
					subtract_from_temp_variable = { camp_occ_lodz_free_capacity_k = camp_admin_detainees_k }
					set_temp_variable = { camp_occ_lodz_share_headroom_k = state_population_k }
					multiply_temp_variable = { camp_occ_lodz_share_headroom_k = constant:camp_admin_value.maximum_population_share }
					subtract_from_temp_variable = { camp_occ_lodz_share_headroom_k = camp_admin_detainees_k }
					clamp_temp_variable = { var = camp_occ_lodz_share_headroom_k min = 0 }
					clamp_temp_variable = { var = camp_occ_lodz_free_capacity_k min = 0 max = camp_occ_lodz_share_headroom_k }
					camp_occ_stage_custody_destination = yes
				}
				87 = {
					set_temp_variable = { camp_occ_custody_row_id = PREV.camp_occ_lodz_source_row_id }
					set_temp_variable = { camp_occ_custody_requested_people = constant:camp_occ_value.one_person }
					camp_occ_preflight_custody_source = yes
				}
				if = {
					limit = { check_variable = { camp_occ_custody_source_valid = constant:camp_occ_value.transfer_invalid } }
					set_temp_variable = { camp_occ_lodz_continue = constant:camp_occ_value.transfer_invalid }
				}
				set_temp_variable = { camp_occ_lodz_wave_ceiling_people = camp_occ_lodz_free_capacity_k }
				multiply_temp_variable = { camp_occ_lodz_wave_ceiling_people = constant:camp_occ_value.people_per_k }
				clamp_temp_variable = { var = camp_occ_lodz_wave_ceiling_people min = 0 max = camp_occ_lodz_pending_people }
				clamp_temp_variable = { var = camp_occ_lodz_wave_ceiling_people min = 0 max = camp_occ_lodz_monthly_throughput_people }
				set_temp_variable = { camp_occ_lodz_live_row_people = camp_occ_custody_row_before_k }
				multiply_temp_variable = { camp_occ_lodz_live_row_people = constant:camp_occ_value.people_per_k }
				clamp_temp_variable = { var = camp_occ_lodz_wave_ceiling_people min = 0 max = camp_occ_lodz_live_row_people }
				set_temp_variable = { camp_occ_lodz_source_headroom_people = camp_occ_custody_source_available_people }
				subtract_from_temp_variable = { camp_occ_lodz_source_headroom_people = constant:camp_occ_value.one_person }
				clamp_temp_variable = { var = camp_occ_lodz_source_headroom_people min = 0 }
				clamp_temp_variable = { var = camp_occ_lodz_wave_ceiling_people min = 0 max = camp_occ_lodz_source_headroom_people }
				set_temp_variable = { camp_occ_lodz_wave_before_round_people = camp_occ_lodz_wave_ceiling_people }
				round_temp_variable = camp_occ_lodz_wave_ceiling_people
				if = {
					limit = { check_variable = { camp_occ_lodz_wave_ceiling_people > camp_occ_lodz_wave_before_round_people } }
					subtract_from_temp_variable = { camp_occ_lodz_wave_ceiling_people = constant:camp_occ_value.one_person }
				}
				if = {
					limit = {
						check_variable = { camp_occ_lodz_continue = constant:camp_occ_value.transfer_valid }
						check_variable = { camp_occ_lodz_wave_ceiling_people > 0 }
					}
					87 = {
						set_temp_variable = { camp_occ_custody_row_id = PREV.camp_occ_lodz_source_row_id }
						set_temp_variable = { camp_occ_custody_requested_people = camp_occ_lodz_wave_ceiling_people }
						set_temp_variable = { camp_occ_custody_route_proven = constant:camp_occ_value.transfer_valid }
						camp_occ_transfer_to_custody_exact = yes
					}
					if = {
						limit = { check_variable = { camp_occ_custody_result = constant:camp_occ_value.transfer_valid } }
						subtract_from_variable = { camp_occ_lodz_pending_people = camp_occ_custody_committed_people }
						subtract_from_temp_variable = { camp_occ_lodz_monthly_throughput_people = camp_occ_custody_committed_people }
						add_to_temp_variable = { camp_occ_lodz_monthly_moved_people = camp_occ_custody_committed_people }
						camp_occ_apply_killing_arrival_loss = yes
					}
					else = { set_temp_variable = { camp_occ_lodz_continue = constant:camp_occ_value.transfer_invalid } }
				}
				else = { set_temp_variable = { camp_occ_lodz_continue = constant:camp_occ_value.transfer_invalid } }
			}
			if = {
				limit = { check_variable = { camp_occ_lodz_monthly_moved_people > 0 } }
				set_temp_variable = { camp_admin_record_kind = constant:camp_admin_history_kind.killing_center_transfer }
				set_temp_variable = { camp_admin_record_state_id = 87 }
				camp_admin_record_history = yes
			}
		}
	}
}

# Country, responsible German authority, from the October 1943 order. Each
# living Danish-origin row still at its home state sends its escaping share
# (camp_occ_rescue.danish_escape_share) to Sweden once, through the exact
# movement receipt in escape mode: one measured debit and one measured credit,
# origin and responsibility kept, no death recorded. Without a valid refuge
# nothing moves and nothing is recorded.
camp_occ_try_danish_rescue = {
	if = {
		limit = { camp_occ_danish_refuge_available = yes }
		set_temp_variable = { camp_occ_rescue_refuge_state = constant:camp_occ_rescue.refuge_anchor_state }
		if = {
			limit = { var:camp_occ_rescue_refuge_state = { camp_occ_rescue_refuge_state_valid = yes } }
			var:camp_occ_rescue_refuge_state = { save_event_target_as = camp_occ_rescue_refuge }
		}
		else = {
			SWE = { capital_scope = { if = { limit = { camp_occ_rescue_refuge_state_valid = yes } save_event_target_as = camp_occ_rescue_refuge } } }
		}
		if = {
			limit = { has_event_target = camp_occ_rescue_refuge }
			set_temp_variable = { camp_occ_rescue_row_index = camp_occ_row_ids^num }
			while_loop_effect = {
				limit = { check_variable = { camp_occ_rescue_row_index > 0 } }
				subtract_from_temp_variable = { camp_occ_rescue_row_index = constant:camp_occ_value.row_index_step }
				if = {
					limit = {
						check_variable = { camp_occ_row_statuses^camp_occ_rescue_row_index = constant:camp_occ_row_status.active }
						check_variable = { camp_occ_row_amounts_k^camp_occ_rescue_row_index > 0 }
						var:camp_occ_row_origins^camp_occ_rescue_row_index = { is_core_of = DEN }
						var:camp_occ_row_hosts^camp_occ_rescue_row_index = { camp_occ_danish_rescue_source_ready = yes }
					}
					set_temp_variable = { camp_occ_rescue_requested_people = camp_occ_row_amounts_k^camp_occ_rescue_row_index }
					multiply_temp_variable = { camp_occ_rescue_requested_people = constant:camp_occ_rescue.danish_escape_share }
					multiply_temp_variable = { camp_occ_rescue_requested_people = constant:camp_occ_value.people_per_k }
					# Whole people, never above the escaping share of the finite row.
					set_temp_variable = { camp_occ_rescue_ceiling_people = camp_occ_rescue_requested_people }
					round_temp_variable = camp_occ_rescue_requested_people
					if = {
						limit = { check_variable = { camp_occ_rescue_requested_people > camp_occ_rescue_ceiling_people } }
						subtract_from_temp_variable = { camp_occ_rescue_requested_people = constant:camp_occ_value.one_person }
					}
					if = {
						limit = { check_variable = { camp_occ_rescue_requested_people > 0 } }
						set_temp_variable = { camp_occ_rescue_row_id = camp_occ_row_ids^camp_occ_rescue_row_index }
						set_temp_variable = { camp_occ_transfer_result = constant:camp_occ_value.transfer_invalid }
						var:camp_occ_row_hosts^camp_occ_rescue_row_index = {
							event_target:camp_occ_rescue_refuge = { camp_occ_stage_transfer_destination = yes }
							set_temp_variable = { camp_occ_transfer_route_proven = constant:camp_occ_value.transfer_valid }
							set_temp_variable = { camp_occ_transfer_escape = constant:camp_occ_value.transfer_valid }
							set_temp_variable = { camp_occ_transfer_row_id = camp_occ_rescue_row_id }
							set_temp_variable = { camp_occ_transfer_requested_people = camp_occ_rescue_requested_people }
							camp_occ_transfer_survivors_exact = yes
							if = {
								limit = { check_variable = { camp_occ_transfer_result = constant:camp_occ_value.transfer_valid } }
								set_state_flag = camp_occ_danish_rescue_done
								set_temp_variable = { camp_occ_rescue_record_state = THIS.id }
							}
						}
						if = {
							limit = { check_variable = { camp_occ_transfer_result = constant:camp_occ_value.transfer_valid } }
							set_temp_variable = { camp_admin_record_kind = constant:camp_admin_history_kind.escape }
							set_temp_variable = { camp_admin_record_state_id = camp_occ_rescue_record_state }
							camp_admin_record_history = yes
						}
					}
				}
			}
		}
	}
}

# Country, the German authority, while the dated partner order stands (Hungary,
# 15 May to 9 July 1944). Registers each partner-held foreign-origin cohort placed
# under deportation orders, then moves living people from those cohorts to the
# state-88 killing center in exact waves bounded by supported throughput, resident
# capacity and source headroom, as the Lodz route does. Each committed wave keeps
# origin, root and German responsibility, and arrival-phase violence applies only
# to its bound row. Without a live killing destination nothing is registered,
# moved or recorded; cohorts still at home when the order lapses are released in
# place by the monthly pass.
camp_occ_try_partner_deportation = {
	if = {
		limit = {
			original_tag = GER
			camp_occ_authority_has_campaign = yes
			camp_occ_partner_deportation_window_open = yes
			country_exists = HUN
			HUN = { camp_occ_partner_order_stands = yes }
			88 = { camp_occ_killing_destination_active = yes }
		}
		set_temp_variable = { camp_occ_partner_authority = THIS }
		HUN = {
			every_controlled_state = {
				limit = { camp_occ_partner_deportation_candidate = yes }
				camp_occ_register_partner_state = yes
			}
		}
		# The Hungarian government learns of the demand once, when its first
		# cohorts are placed under German orders, and answers in camp_occ_partner.1.
		if = {
			limit = {
				HUN = { NOT = { has_country_flag = camp_occ_partner_order_notified } }
				any_of_scopes = { array = camp_occ_states camp_occ_state_has_partner_authority = yes }
			}
			HUN = {
				set_country_flag = camp_occ_partner_order_notified
				country_event = { id = camp_occ_partner.1 }
			}
		}
		88 = {
			set_temp_variable = { camp_occ_partner_throughput_people = non_damaged_building_level@extermination_camp }
			multiply_temp_variable = { camp_occ_partner_throughput_people = constant:camp_occ_arrival.throughput_per_supported_level_k }
			multiply_temp_variable = { camp_occ_partner_throughput_people = camp_admin_support_ratio }
			multiply_temp_variable = { camp_occ_partner_throughput_people = constant:camp_occ_value.people_per_k }
		}
		# Delaying Hungarian officials hold back part of every month's trains.
		if = {
			limit = { HUN = { has_country_flag = camp_occ_partner_deportations_obstructed } }
			multiply_temp_variable = { camp_occ_partner_throughput_people = constant:camp_occ_partner.obstructed_wave_share }
		}
		set_temp_variable = { camp_occ_partner_batch_count = 0 }
		for_each_scope_loop = {
			array = camp_occ_states
			if = {
				limit = {
					camp_occ_state_has_partner_authority = yes
					has_variable = camp_occ_native_row_id
					check_variable = { camp_occ_remaining_k > 0 }
				}
				set_temp_variable = { camp_occ_partner_source_moved_people = 0 }
				set_temp_variable = { camp_occ_partner_continue = constant:camp_occ_value.transfer_valid }
				while_loop_effect = {
					limit = {
						check_variable = { camp_occ_partner_continue = constant:camp_occ_value.transfer_valid }
						check_variable = { camp_occ_partner_throughput_people > 0 }
						check_variable = { camp_occ_partner_batch_count < constant:camp_occ_arrival.max_batches_per_month }
						camp_occ_state_has_partner_authority = yes
						88 = { camp_occ_killing_destination_active = yes }
					}
					add_to_temp_variable = { camp_occ_partner_batch_count = constant:camp_occ_value.row_index_step }
					88 = {
						camp_admin_refresh_capacity = yes

# Country of origin, one-time source snapshot. The denominator includes ALL its European owned
# states, so partial occupation cannot concentrate a whole national cohort into the occupied share.
# Optional temp inputs camp_occ_origin_anchor_state_input and camp_occ_origin_anchor_input_k
# reserve a sourced city concentration for one owned European state inside the same ceiling;
# an unowned anchor reserves nothing. Both inputs are consumed.
camp_occ_seed_origin = {
	if = {
		limit = { NOT = { has_country_flag = camp_occ_origin_seeded } }
		set_country_flag = camp_occ_origin_seeded
		set_variable = { camp_occ_origin_initial_ceiling_k = camp_occ_origin_ceiling_input_k }
		set_variable = { camp_occ_origin_unassigned_k = camp_occ_origin_initial_ceiling_k }
		set_variable = { camp_occ_origin_proportional_k = camp_occ_origin_initial_ceiling_k }
		if = {
			limit = { check_variable = { camp_occ_origin_anchor_state_input > 0 } }
			set_variable = { camp_occ_origin_anchor_state = camp_occ_origin_anchor_state_input }
			if = {
				limit = {
					var:camp_occ_origin_anchor_state = {
						is_owned_by = PREV
						is_on_continent = europe
						NOT = { has_state_flag = camp_occ_origin_assigned }
					}
				}
				set_temp_variable = { camp_occ_origin_assigned_input_k = camp_occ_origin_anchor_input_k }
				clamp_temp_variable = { var = camp_occ_origin_assigned_input_k min = 0 max = camp_occ_origin_initial_ceiling_k }
				var:camp_occ_origin_anchor_state = {
					set_state_flag = camp_occ_origin_sourced_anchor
					camp_occ_seed_assign_state = yes
				}
				subtract_from_variable = { camp_occ_origin_proportional_k = camp_occ_origin_assigned_k }
			}
			else = { clear_variable = camp_occ_origin_anchor_state }
		}
		set_temp_variable = { camp_occ_origin_population_k = 0 }
		every_owned_state = {
			limit = { is_on_continent = europe NOT = { has_state_flag = camp_occ_origin_sourced_anchor } }
			add_to_temp_variable = { camp_occ_origin_population_k = state_population_k }
		}
		if = {
			limit = { check_variable = { camp_occ_origin_population_k > 0 } }
			divide_temp_variable = { camp_occ_origin_population_k = constant:camp_occ_value.people_per_k }
			every_owned_state = {
				limit = { is_on_continent = europe NOT = { has_state_flag = camp_occ_origin_assigned } }
				set_temp_variable = { camp_occ_origin_share = state_population_k }
				divide_temp_variable = { camp_occ_origin_share = camp_occ_origin_population_k }
				set_temp_variable = { camp_occ_origin_assigned_input_k = PREV.camp_occ_origin_proportional_k }
				divide_temp_variable = { camp_occ_origin_assigned_input_k = constant:camp_occ_value.people_per_k }
				multiply_temp_variable = { camp_occ_origin_assigned_input_k = camp_occ_origin_share }
				camp_occ_seed_assign_state = yes
			}
		}
	}
	set_temp_variable = { camp_occ_origin_anchor_state_input = 0 }
	set_temp_variable = { camp_occ_origin_anchor_input_k = 0 }
}
