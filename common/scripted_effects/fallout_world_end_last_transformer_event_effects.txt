fallout_event_243_select_partner = {
	set_variable = { fallout_event_243_partner_id = constant:fallout_event_runtime.zero }
	set_variable = { fallout_event_243_partner_score = constant:fallout_event_runtime.zero }
	set_variable = { fallout_event_243_partner_power = constant:fallout_event_runtime.zero }
	set_variable = { fallout_event_243_partner_reclamation = constant:fallout_event_runtime.zero }
	every_neighbor_country = {
		limit = {
			exists = yes
			is_ai = yes
			NOT = { has_war_with = ROOT }
			fallout_survival_country_identity_row_is_current = yes
			fallout_survival_country_resource_row_is_durable = yes
			check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_243_partner.minimum_power compare = greater_than_or_equals }
			any_owned_state = {
				check_variable = { air_winter_reclamation value = constant:fallout_event_243_partner.minimum_reclamation compare = greater_than_or_equals }
			}
		}
		set_variable = { ROOT.fallout_event_243_neighbor_reclamation = constant:fallout_event_runtime.zero }
		every_owned_state = {
			if = {
				limit = { check_variable = { air_winter_reclamation value = ROOT.fallout_event_243_neighbor_reclamation compare = greater_than } }
				set_variable = { ROOT.fallout_event_243_neighbor_reclamation = air_winter_reclamation }
			}
		}
		set_variable = { ROOT.fallout_event_243_neighbor_score = fallout_survival_current_entries^6 }
		add_to_variable = { ROOT.fallout_event_243_neighbor_score = ROOT.fallout_event_243_neighbor_reclamation }
		if = {
			limit = {
				OR = {
					check_variable = { ROOT.fallout_event_243_partner_id = constant:fallout_event_runtime.zero }
					check_variable = { ROOT.fallout_event_243_neighbor_score value = ROOT.fallout_event_243_partner_score compare = greater_than }
					AND = {
						check_variable = { ROOT.fallout_event_243_neighbor_score value = ROOT.fallout_event_243_partner_score compare = equals }
						check_variable = { var = THIS.id value = ROOT.fallout_event_243_partner_id compare = less_than }
					}
				}
			}
			set_variable = { ROOT.fallout_event_243_partner_id = THIS.id }
			set_variable = { ROOT.fallout_event_243_partner_score = ROOT.fallout_event_243_neighbor_score }
			set_variable = { ROOT.fallout_event_243_partner_power = fallout_survival_current_entries^6 }
			set_variable = { ROOT.fallout_event_243_partner_reclamation = ROOT.fallout_event_243_neighbor_reclamation }
		}
		clear_variable = fallout_event_243_neighbor_reclamation
		clear_variable = fallout_event_243_neighbor_score
	}
	clear_variable = fallout_event_243_neighbor_reclamation
	clear_variable = fallout_event_243_neighbor_score
}

	if = {

			NOT = { has_country_flag = fallout_event_243_factory_removal_applied }

			if = {
				limit = {
					check_variable = { ROOT.fallout_event_243_selected_building_type = constant:fallout_event_243_building_type.civilian }
					check_variable = { var = non_damaged_building_level@industrial_complex value = constant:fallout_event_runtime.zero compare = greater_than }

		}
	}
}
