				}
			}
		}
		if = {
			limit = {
				has_country_flag = africa_first_proof_outward_partner_eligible
				africa_resolved_first_proof_action_supplies_outward_partner = yes
			}
			set_country_flag = africa_first_proof_actor_semantic_valid
			event_target:africa_host = {

					limit = { NOT = { is_in_array = { array = africa_first_proof_actor_credits value = PREV } } }
					add_to_array = { array = africa_first_proof_actor_credits value = PREV }

				has_country_flag = africa_first_proof_corridor_actor_eligible
				africa_resolved_first_proof_action_supplies_corridor_partner = yes
			}
			set_country_flag = africa_first_proof_corridor_actor_semantic_valid
			if = {
				limit = { check_variable = { africa_resolved_first_proof_action_id = constant:africa_action.connect_member_capitals } }
				set_country_flag = africa_first_proof_capital_rail_route
			}

					add_to_array = { array = africa_first_proof_corridor_actor_credits value = PREV }

			}

			limit = {
				has_country_flag = africa_first_proof_domestic_settlement_eligible
				africa_resolved_first_proof_action_supplies_domestic_settlement = yes
			}
			event_target:africa_host = { set_country_flag = africa_first_proof_domestic_settlement_durable }
		}
		if = {
			limit = {
				has_country_flag = africa_first_proof_reform_eligible
				africa_resolved_first_proof_action_supplies_reform = yes
			}
			event_target:africa_host = { set_country_flag = africa_first_proof_reform_durable }

				has_country_flag = africa_first_proof_sovereignty_eligible
				is_subject = no

		}
		if = {

			event_target:africa_host = { set_country_flag = africa_first_proof_no_coercion_durable }

	set_country_flag = africa_first_proof_live_primary_target_valid

			limit = {
				OR = {
					NOT = { exists = yes }
					has_capitulated = yes
					NOT = { has_country_flag = africa_first_proof_primary_semantic_valid }
					NOT = {
						OR = {
							africa_is_current_host = yes
							africa_first_proof_country_is_current_playbook_geographic_partner = yes
						}
					}
				}
			}
			event_target:africa_host = { clr_country_flag = africa_first_proof_live_primary_target_valid }
		}
	}
	for_each_scope_loop = {
		array = africa_first_proof_actor_credits
		if = {
			limit = {
				OR = {
					NOT = { exists = yes }
					has_capitulated = yes
					NOT = { has_country_flag = africa_first_proof_actor_semantic_valid }
					NOT = { africa_first_proof_country_is_current_playbook_geographic_partner = yes }
				}
			}

		}
	}
	for_each_scope_loop = {
		array = africa_first_proof_corridor_actor_credits

			limit = {
				OR = {
					NOT = { exists = yes }
					has_capitulated = yes

					NOT = { africa_first_proof_country_is_current_playbook_geographic_partner = yes }
					NOT = { africa_first_proof_corridor_actor_route_is_live = yes }
				}
			}
			event_target:africa_host = { clr_country_flag = africa_first_proof_live_corridors_valid }
		}
	}
	for_each_scope_loop = {
		array = africa_first_proof_sovereignty_actor_credits
		if = {
			limit = {
				OR = {
					NOT = { exists = yes }

					NOT = { has_state_flag = africa_first_proof_region_semantic_valid }
					NOT = { africa_first_proof_state_project_evidence_is_live = yes }

					NOT = { africa_first_proof_corridor_state_evidence_is_live_and_exact = yes }

			}

		limit = {
			has_country_flag = africa_first_proof_requires_regional_consent

			limit = { has_country_flag = africa_first_proof_recovery_partial_ready }
			set_country_flag = africa_first_proof_recovered_under_oversight
			set_country_flag = africa_mandatory_autonomy_clause
			add_to_variable = { africa_charter_authority = constant:africa_action_result.failure_authority }
		}
		africa_cleanup_compound_first_proof_ledger = yes
		africa_clamp_host_values = yes
		africa_refresh_continental_focus_tree_layout = yes
	}
}

# COUNTRY scope on the resolved target before action cleanup. Every full proof
# action credits the exact immutable target/state snapshot; a partial or
# failed primary action closes the first mandate and opens the one-shot
# recovery contract. Action 101 has its own full-only recovery result.
africa_handle_first_proof_action_resolution = {
	set_temp_variable = { africa_resolved_first_proof_action_id = africa_active_action_id }
	set_temp_variable = { africa_resolved_first_proof_full_context = constant:africa_value.zero }
	if = {
		limit = { check_variable = { africa_active_action_outcome = constant:africa_action_outcome.full } }
		set_temp_variable = { africa_resolved_first_proof_full_context = constant:africa_value.one }
	}
	if = {
		limit = {
			check_variable = { africa_active_action_outcome = constant:africa_action_outcome.full }
			event_target:africa_host = { africa_resolved_first_proof_action_is_allowed = yes }
		}
		africa_credit_first_proof_from_full_action = yes
		africa_confirm_active_first_proof_corridor_states = yes
		africa_confirm_active_first_proof_region_states = yes
		event_target:africa_host = {
			africa_credit_first_proof_compressed_institutions_from_full_action = yes
			if = {
				limit = { has_country_flag = africa_first_proof_requires_sovereignty }
				africa_confirm_first_proof_sovereignty = yes
			}
			if = {
				limit = { has_country_flag = africa_first_proof_requires_no_coercion }
				africa_confirm_first_proof_no_coercion = yes
			}
			africa_finalize_compound_first_proof_ledger = yes
			africa_consume_first_proof_ledger_ready = yes
		}
		if = {
			limit = { event_target:africa_host = { has_country_flag = africa_first_proof_succeeded } }
			set_country_flag = africa_first_proof_success_target
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { africa_active_action_outcome = constant:africa_action_outcome.partial }
				check_variable = { africa_active_action_outcome = constant:africa_action_outcome.failure }
				check_variable = { africa_active_action_outcome = constant:africa_action_outcome.cancelled }
			}
			event_target:africa_host = {
				africa_resolved_first_proof_action_is_allowed = yes
				africa_resolved_first_proof_action_is_primary = yes
			}
		}
		set_country_flag = africa_original_failed_proof_target
		event_target:africa_host = {
			clr_country_flag = africa_first_proof_active
			set_country_flag = africa_first_proof_failed
			set_country_flag = africa_first_proof_failure_remembered
			add_to_variable = { africa_first_proof_failures = constant:africa_value.one }
			if = {
				limit = { PREV = { check_variable = { africa_active_action_outcome = constant:africa_action_outcome.partial } } }
				set_country_flag = africa_first_proof_partial_result
			}
			else = { set_country_flag = africa_first_proof_failed_result }
			africa_cleanup_compound_first_proof_ledger = yes
			africa_refresh_continental_focus_tree_layout = yes
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { africa_active_action_outcome = constant:africa_action_outcome.partial }
				check_variable = { africa_active_action_outcome = constant:africa_action_outcome.failure }
				check_variable = { africa_active_action_outcome = constant:africa_action_outcome.cancelled }
			}
			event_target:africa_host = {
				africa_resolved_first_proof_action_is_allowed = yes
				NOT = { africa_resolved_first_proof_action_is_primary = yes }
			}
		}
		set_country_flag = africa_first_proof_auxiliary_unresolved_target
		event_target:africa_host = {
			set_country_flag = africa_first_proof_auxiliary_evidence_unresolved
			clr_country_flag = africa_first_proof_ledger_ready
			if = {
				limit = { PREV = { check_variable = { africa_active_action_outcome = constant:africa_action_outcome.partial } } }
				set_country_flag = africa_first_proof_auxiliary_partial_result
			}
			else_if = {
				limit = { PREV = { check_variable = { africa_active_action_outcome = constant:africa_action_outcome.cancelled } } }
				set_country_flag = africa_first_proof_auxiliary_cancelled_result
			}
			else = { set_country_flag = africa_first_proof_auxiliary_failed_result }
		}
	}
	if = {
		limit = { check_variable = { africa_active_action_id = constant:africa_action.recover_failed_host_proof } }
		africa_credit_first_proof_recovery_from_full_action = yes
		if = {
			limit = { check_variable = { africa_active_action_outcome = constant:africa_action_outcome.partial } }
			event_target:africa_host = { set_country_flag = africa_first_proof_recovery_partial_ready }
		}
		africa_close_first_proof_recovery_contract = yes
		event_target:africa_host = { africa_consume_first_proof_recovery_ready = yes }
		if = {
			limit = { check_variable = { africa_active_action_outcome = constant:africa_action_outcome.failure } }
			event_target:africa_host = {
				add_to_variable = { africa_first_proof_failures = constant:africa_value.one }
				africa_cleanup_compound_first_proof_ledger = yes
				africa_refresh_continental_focus_tree_layout = yes
			}
		}
	}
}
