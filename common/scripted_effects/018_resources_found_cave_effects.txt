# strength, continuous-control resource anchors, automatic brood spawning,
# neighbor wars, focus rewards, exact continent verification, resource-weighted

# STATE scope. Doctrine and player preference change the automatically created
# template. Locked templates remain unavailable to the ordinary recruit queue.
resources_found_cave_spawn_selected_brood_here = {
	if = {
		limit = { DHO = { has_country_flag = resources_found_spawn_preference_stone } }
		resources_found_cave_spawn_stone_brood_here = yes
	}
	else_if = {
		limit = { DHO = { has_country_flag = resources_found_spawn_preference_burrow } }
		resources_found_cave_spawn_burrow_brood_here = yes
	}
	else_if = {
		limit = { DHO = { has_country_flag = resources_found_spawn_preference_scree } }
		resources_found_cave_spawn_scree_brood_here = yes
	}
	else = {
		set_temp_variable = { resources_found_cave_default_brood_weight = constant:resources_found_cave_random_weight.default_brood }
		set_temp_variable = { resources_found_cave_stone_brood_weight = constant:resources_found_value.zero }
		set_temp_variable = { resources_found_cave_burrow_brood_weight = constant:resources_found_value.zero }
		set_temp_variable = { resources_found_cave_scree_brood_weight = constant:resources_found_value.zero }
		set_temp_variable = { resources_found_cave_guard_brood_weight = constant:resources_found_value.zero }
		if = {
			limit = { DHO = { has_country_flag = cave_stone_phalanx_doctrine } }
			set_temp_variable = { resources_found_cave_stone_brood_weight = constant:resources_found_cave_random_weight.doctrine_brood }
		}
		if = {
			limit = { DHO = { has_country_flag = cave_burrow_war_doctrine } }
			set_temp_variable = { resources_found_cave_burrow_brood_weight = constant:resources_found_cave_random_weight.doctrine_brood }
		}
		if = {
			limit = { DHO = { has_country_flag = cave_scree_tide_doctrine } }
			set_temp_variable = { resources_found_cave_scree_brood_weight = constant:resources_found_cave_random_weight.doctrine_brood }
		}
		if = {
			limit = { DHO = { has_country_flag = cave_feeding_chamber_guards_unlocked } }
			set_temp_variable = { resources_found_cave_guard_brood_weight = constant:resources_found_cave_random_weight.guard_brood }
		}
		random_list = {
			resources_found_cave_default_brood_weight = { resources_found_cave_spawn_war_brood_here = yes }
			resources_found_cave_stone_brood_weight = { resources_found_cave_spawn_stone_brood_here = yes }
			resources_found_cave_burrow_brood_weight = { resources_found_cave_spawn_burrow_brood_here = yes }
			resources_found_cave_scree_brood_weight = { resources_found_cave_spawn_scree_brood_here = yes }
			resources_found_cave_guard_brood_weight = { resources_found_cave_spawn_guard_brood_here = yes }
		}
	}
}

# STATE origin scope.
resources_found_cave_spawn_opening_allocation = {
	set_temp_variable = { resources_found_opening_broods_remaining = DHO.resources_found_cave_starting_divisions }
	while_loop_effect = {
		limit = { check_variable = { resources_found_opening_broods_remaining > constant:resources_found_value.zero } }
		resources_found_cave_spawn_war_brood_here = yes
		subtract_from_temp_variable = { resources_found_opening_broods_remaining = constant:resources_found_value.one }
	}
	DHO = { set_variable = { resources_found_active_brood_divisions = num_divisions } }
}

# DHO country scope.
resources_found_cave_refresh_division_and_capacity_totals = {
	set_variable = { resources_found_active_brood_divisions = num_divisions }
	set_variable = { resources_found_total_brood_capacity = resources_found_protected_origin_allocation }
	add_to_variable = { resources_found_total_brood_capacity = resources_found_total_anchor_capacity }
	if = {
		limit = { check_variable = { resources_found_total_brood_capacity < resources_found_protected_origin_allocation } }
		set_variable = { resources_found_total_brood_capacity = resources_found_protected_origin_allocation }
	}
}

}

# STATE field scope. Prepares the first persistent DHO incarnation, then asks
# the former owner how play should continue before the origin state transfers.
# This ordering preserves the choice even when the field is that country's
# final owned state.
resources_found_begin_first_cave_emergence = {
	if = {
		limit = { resources_found_state_can_form_first_cave_origin = yes }
		save_event_target_as = resources_found_emergence_state
		OWNER = {
			save_global_event_target_as = resources_found_cave_former_owner
			set_country_flag = resources_found_cave_origin_lost
			set_country_flag = resources_found_cave_aid_eligible
		}
		DHO = { resources_found_cave_setup_country = yes }
		resources_found_calculate_cave_starting_strength = yes
		resources_found_record_cave_emergence_achievement_state = yes
		resources_found_record_cave_origin = yes
		event_target:resources_found_cave_former_owner = {
			country_event = { id = chaosx.nr18.80 }
		}
	}
}

# STATE origin scope. Completes the first emergence only after Event .80 has
# resolved. All three options call this exact helper, so AI countries, players
# retaining their country, and players continuing as DHO share one lifecycle.
resources_found_complete_first_cave_emergence = {
	save_event_target_as = resources_found_emergence_state
	if = {
		limit = {
			has_state_flag = resources_found_cave_origin
			NOT = { has_state_flag = resources_found_first_cave_emergence_complete }
			DHO = {
				has_country_flag = resources_found_cave_country_created
				has_variable = resources_found_cave_origin_state
			}
		}
		resources_found_cave_declare_war_on_current_owner_and_controller = yes
		transfer_state_to = DHO
		add_core_of = DHO
		set_state_controller_to = DHO
		set_state_flag = resources_found_first_cave_emergence_complete
		set_state_flag = resources_found_cave_cleanup_required
		add_dynamic_modifier = { modifier = resources_found_cave_origin_chamber }
		force_update_dynamic_modifier = yes
		if = {
			limit = { NOT = { supply_node > constant:resources_found_value.zero } }
			add_building_construction = { type = supply_node level = @CR_SC_RESOURCES_FOUND_VALUE_ONE instant_build = yes }
		}
		DHO = {
			set_capital = {
				state = event_target:resources_found_emergence_state
				remember_old_capital = no
			}
			set_variable = { resources_found_protected_origin_allocation = resources_found_cave_starting_divisions }
			set_variable = { resources_found_total_brood_capacity = resources_found_cave_starting_divisions }
			set_country_flag = resources_found_continent_registry_dirty
		}
		resources_found_cave_spawn_opening_allocation = yes
		set_global_flag = world_threat_source_resources_found_caves
		refresh_world_threat_state = yes
		DHO = {
			resources_found_cave_declare_neighbor_wars = yes
			resources_found_cave_refresh_continent_progress = yes
			resources_found_cave_prepare_foothold_candidates = yes
		}
		DHO = {
			resources_found_emit_cave_emergence_news = yes
			resources_found_emit_emergence_super_event = yes
			country_event = { id = chaosx.nr18.82 hours = @CAVE_FOLLOWUP_HOURS }
		}
	}
}

# STATE field scope. A later breach strengthens the one persistent Host.
resources_found_reinforce_existing_cave_host_from_field = {
	if = {
		limit = { resources_found_state_can_reinforce_existing_cave_host = yes }
		save_event_target_as = resources_found_emergence_state
		save_event_target_as = resources_found_unbinding_state
		OWNER = {
			save_event_target_as = resources_found_reinforcement_former_owner
			set_country_flag = resources_found_cave_aid_eligible
		}
		resources_found_cave_declare_war_on_current_owner_and_controller = yes
		set_state_flag = resources_found_field_converted_to_cave_breach
		if = {
			limit = { has_variable = resources_found_recorded_owner }
			var:resources_found_recorded_owner = {
				save_event_target_as = resources_found_reinforcement_registry_owner
				resources_found_remove_field_from_country_registry = yes
			}
		}
		# The registry helper normally selects another active field immediately.
		# This state is still physically active until the registry mutation is
		# complete, so prevent it from surviving as the former owner's selection.
		if = {
			limit = { has_event_target = resources_found_reinforcement_registry_owner }
			event_target:resources_found_reinforcement_registry_owner = {
				if = {
					limit = {
						has_variable = resources_found_selected_field
						check_variable = {
							var = resources_found_selected_field
							value = event_target:resources_found_unbinding_state
							compare = equals
						}
					}
					clear_variable = resources_found_selected_field
				}
			}
		}
		clr_state_flag = resources_found_field_active
		clr_state_flag = resources_found_field_suspended
		if = {
			limit = { has_event_target = resources_found_reinforcement_registry_owner }
			event_target:resources_found_reinforcement_registry_owner = { resources_found_validate_country_field_selection = yes }
		}
		resources_found_clear_field_diplomacy_and_incidents = yes
		transfer_state_to = DHO
		add_core_of = DHO
		set_state_controller_to = DHO
		set_variable = { resources_found_recorded_owner = DHO }
		set_variable = { resources_found_recorded_controller = DHO }
		set_state_flag = resources_found_cave_cleanup_required
		resources_found_cave_begin_anchor_activation = yes
		DHO = {
			resources_found_cave_refresh_division_and_capacity_totals = yes
			resources_found_cave_declare_neighbor_wars = yes
			resources_found_cave_refresh_continent_progress = yes
		}
		event_target:resources_found_reinforcement_former_owner = { country_event = { id = chaosx.nr18.81 hours = @CAVE_NOTICE_HOURS } }
	}
}

# STATE scope. Selects first or later Evolution IV behavior without a fallback.

	resources_found_record_evolution_iv_from_field = yes
	if = {
		limit = { resources_found_state_can_form_first_cave_origin = yes }
		resources_found_begin_first_cave_emergence = yes
	}
	else_if = {
		limit = { resources_found_state_can_reinforce_existing_cave_host = yes }
		resources_found_reinforce_existing_cave_host_from_field = yes
	}
}

# STATE scope.
resources_found_cave_declare_war_on_current_owner_and_controller = {
	OWNER = { save_event_target_as = resources_found_breach_owner_target }
	CONTROLLER = { save_event_target_as = resources_found_breach_controller_target }
	if = {
		limit = {
			has_event_target = resources_found_breach_owner_target
			event_target:resources_found_breach_owner_target = {
				exists = yes
				NOT = { is_resources_found_cave_country = yes }
				NOT = { has_war_with = DHO }
			}
			DHO = { can_declare_war_on = event_target:resources_found_breach_owner_target }
		}
		DHO = { declare_war_on = { target = event_target:resources_found_breach_owner_target type = annex_everything } }
	}
	if = {
		limit = {
			has_event_target = resources_found_breach_controller_target
			event_target:resources_found_breach_controller_target = {
				exists = yes
				NOT = { is_resources_found_cave_country = yes }
				NOT = { has_war_with = DHO }
			}
			DHO = { can_declare_war_on = event_target:resources_found_breach_controller_target }
		}
		DHO = { declare_war_on = { target = event_target:resources_found_breach_controller_target type = annex_everything } }
	}
}

# STATE scope for a land state adjacent to DHO control. The war helper covers
# distinct owner/controller actors, while the flag prevents repeated notices.
resources_found_cave_attack_adjacent_state_actors = {
	resources_found_cave_declare_war_on_current_owner_and_controller = yes
	OWNER = {
		if = {
			limit = {
				exists = yes
				NOT = { is_resources_found_cave_country = yes }
				has_war_with = DHO
				NOT = { has_country_flag = resources_found_cave_frontier_notice_received }
			}
			set_country_flag = resources_found_cave_frontier_notice_received
			country_event = { id = chaosx.nr18.82 hours = @CAVE_NOTICE_HOURS }
		}
	}
	CONTROLLER = {
		if = {
			limit = {
				exists = yes
				NOT = { is_resources_found_cave_country = yes }
				has_war_with = DHO
				NOT = { has_country_flag = resources_found_cave_frontier_notice_received }
			}
			set_country_flag = resources_found_cave_frontier_notice_received
			country_event = { id = chaosx.nr18.82 hours = @CAVE_NOTICE_HOURS }
		}
	}
}

# DHO country scope. Every present and newly adjacent land neighbor is attacked.
# Controlled frontier states are used so conquests extend the war frontier even
# before a peace conference changes state ownership.
resources_found_cave_declare_neighbor_wars = {
	every_controlled_state = {
		every_neighbor_state = {
			limit = {
				impassable = no
				NOT = { is_controlled_by = DHO }
			}
			resources_found_cave_attack_adjacent_state_actors = yes
		}
	}
}

			}
			subtract_from_variable = { resources_found_cave_anchor_activation_date = constant:resources_found_cave_runtime.chamber_autonomy_activation_reduction_days }

				has_variable = resources_found_cave_denial_capacity_penalty
			}

			# and capacity penalty remain prepared until an activation succeeds.

		}

		force_update_dynamic_modifier = yes
		save_event_target_as = resources_found_anchor_registration_state

							array = resources_found_cave_anchor_states

					}

					array = resources_found_cave_anchor_states
					value = event_target:resources_found_anchor_registration_state
				}
			}
		}
	}

# STATE origin scope. Losing the first chamber removes its protected opening
# allocation from live capacity, so surviving broods become unfed until the
# origin is recovered or the distributed hierarchy has reduced the command
# shock. Recapture restores the exact protected allocation recorded at loss.
resources_found_cave_handle_origin_control_change = {
	if = {
		limit = {
			resources_found_is_cave_origin_state = yes
			has_event_target = resources_found_old_controller
			event_target:resources_found_old_controller = { is_resources_found_cave_country = yes }
			has_event_target = resources_found_new_controller
			event_target:resources_found_new_controller = { NOT = { is_resources_found_cave_country = yes } }
		}
		DHO = {
			if = {
				limit = { NOT = { has_variable = resources_found_origin_allocation_before_loss } }
				set_variable = { resources_found_origin_allocation_before_loss = resources_found_protected_origin_allocation }
			}
			set_country_flag = cave_origin_lost
			set_country_flag = cave_origin_threatened
			if = {
				limit = {
					has_country_flag = cave_host_without_a_head
					has_variable = resources_found_cave_secondary_capital_state
					var:resources_found_cave_secondary_capital_state = { is_owned_and_controlled_by = PREV }
				}
				set_variable = { resources_found_protected_origin_allocation = resources_found_origin_allocation_before_loss }
				divide_variable = { resources_found_protected_origin_allocation = constant:resources_found_value.two }
				var:resources_found_cave_secondary_capital_state = {
					save_event_target_as = resources_found_emergency_cave_capital
				}
				set_capital = {
					state = event_target:resources_found_emergency_cave_capital
					remember_old_capital = no
				}
				if = { limit = { has_idea = cave_origin_command_crisis } remove_ideas = cave_origin_command_crisis }
			}
			else = {
				set_variable = { resources_found_protected_origin_allocation = constant:resources_found_value.zero }
				if = { limit = { NOT = { has_idea = cave_origin_command_crisis } } add_ideas = cave_origin_command_crisis }
			}
			resources_found_cave_refresh_division_and_capacity_totals = yes
			resources_found_cave_refresh_unfed_status = yes
			country_event = { id = chaosx.nr18.87 hours = @CAVE_NOTICE_HOURS }
		}
	}
	else_if = {
		limit = {
			resources_found_is_cave_origin_state = yes
			has_event_target = resources_found_new_controller
			event_target:resources_found_new_controller = { is_resources_found_cave_country = yes }
			DHO = { has_country_flag = cave_origin_lost }
		}
		DHO = {
			if = {
				limit = { has_variable = resources_found_origin_allocation_before_loss }
				set_variable = { resources_found_protected_origin_allocation = resources_found_origin_allocation_before_loss }
				clear_variable = resources_found_origin_allocation_before_loss
			}
			clr_country_flag = cave_origin_lost
			clr_country_flag = cave_origin_threatened
			if = { limit = { has_idea = cave_origin_command_crisis } remove_ideas = cave_origin_command_crisis }
			resources_found_cave_refresh_division_and_capacity_totals = yes
			resources_found_cave_refresh_unfed_status = yes
		}
	}
}

					NOT = { is_controlled_by = DHO }

				NOT = { has_state_flag = resources_found_cave_anchor_disrupted }

					impassable = no
					NOT = { is_controlled_by = DHO }
				}
			}
			save_global_event_target_as = resources_found_cave_spawn_state
		}
	}
	else_if = {
		limit = {
			has_country_flag = cave_continental_network
			any_controlled_state = {
				has_state_flag = resources_found_cave_anchor_active
				NOT = { has_state_flag = resources_found_cave_anchor_disrupted }
				has_state_flag = resources_found_cave_tunnel_link_endpoint
			}
		}
		random_controlled_state = {
			limit = {
				has_state_flag = resources_found_cave_anchor_active
				NOT = { has_state_flag = resources_found_cave_anchor_disrupted }
				has_state_flag = resources_found_cave_tunnel_link_endpoint
			}
			save_global_event_target_as = resources_found_cave_spawn_state
		}
	}
	else_if = {
		limit = {
			any_controlled_state = {
				has_state_flag = resources_found_cave_anchor_active
				NOT = { has_state_flag = resources_found_cave_anchor_disrupted }
			}
		}
		random_controlled_state = {
			limit = {
				has_state_flag = resources_found_cave_anchor_active
				NOT = { has_state_flag = resources_found_cave_anchor_disrupted }
			}
			save_global_event_target_as = resources_found_cave_spawn_state
		}
	}
	else_if = {
		limit = { resources_found_world_end_origin_inputs_valid = yes }
		var:resources_found_cave_origin_state = { save_global_event_target_as = resources_found_cave_spawn_state }
	}

# DHO country scope. One-shot exact scan, called only from state changes and
# terminal milestones. Eligible states must be legitimate land states on the
# stored origin continent. Ownership and control are counted independently.
resources_found_cave_refresh_continent_progress = {
	set_variable = { resources_found_continent_eligible_state_count = constant:resources_found_value.zero }
	set_variable = { resources_found_continent_scanned_state_count = constant:resources_found_value.zero }
	set_variable = { resources_found_continent_owned_state_count = constant:resources_found_value.zero }
	set_variable = { resources_found_continent_controlled_state_count = constant:resources_found_value.zero }
	set_variable = { resources_found_continent_owned_controlled_state_count = constant:resources_found_value.zero }
	set_variable = { resources_found_continent_remaining_state_count = constant:resources_found_value.zero }
	clr_country_flag = resources_found_continent_registry_ready
	clr_country_flag = resources_found_continent_scan_mismatch
	every_state = {
		limit = { has_state_flag = resources_found_cave_continental_capital_objective }
		clr_state_flag = resources_found_cave_continental_capital_objective
	}

	every_state = {
		limit = {
			resources_found_state_is_valid_continent_member = yes
			resources_found_state_on_cave_origin_continent = yes
		}
		DHO = {
			add_to_variable = { resources_found_continent_eligible_state_count = constant:resources_found_value.one }
			add_to_variable = { resources_found_continent_scanned_state_count = constant:resources_found_value.one }
		}
		if = {
			limit = { is_owned_by = DHO }
			DHO = { add_to_variable = { resources_found_continent_owned_state_count = constant:resources_found_value.one } }
		}
		if = {
			limit = { is_controlled_by = DHO }
			DHO = { add_to_variable = { resources_found_continent_controlled_state_count = constant:resources_found_value.one } }
		}
		else_if = {
			limit = { is_capital = yes }
			set_state_flag = resources_found_cave_continental_capital_objective
		}
		if = {
			limit = {
				is_owned_by = DHO
				is_controlled_by = DHO
			}
			DHO = { add_to_variable = { resources_found_continent_owned_controlled_state_count = constant:resources_found_value.one } }
		}
		else = { DHO = { add_to_variable = { resources_found_continent_remaining_state_count = constant:resources_found_value.one } } }
	}

	if = {
		limit = {
			check_variable = { resources_found_continent_eligible_state_count > constant:resources_found_value.zero }
			check_variable = {
				var = resources_found_continent_scanned_state_count
				value = resources_found_continent_eligible_state_count
				compare = equals
			}
		}
		set_country_flag = resources_found_continent_registry_ready
	}
	else = { set_country_flag = resources_found_continent_scan_mismatch }
	resources_found_refresh_near_control_status = yes

	set_temp_variable = { resources_found_continent_quarter_test = resources_found_continent_owned_controlled_state_count }
	multiply_temp_variable = { resources_found_continent_quarter_test = constant:resources_found_cave_runtime.quarter_multiplier }
	if = {
		limit = {
			NOT = { has_country_flag = resources_found_continent_quarter_reached }
			check_variable = {
				var = resources_found_continent_quarter_test
				value = resources_found_continent_eligible_state_count
				compare = greater_than_or_equals
			}
		}
		set_country_flag = resources_found_continent_quarter_reached
		country_event = { id = chaosx.nr18.90 hours = @CAVE_NOTICE_HOURS }
	}

	set_temp_variable = { resources_found_continent_half_test = resources_found_continent_owned_controlled_state_count }
	multiply_temp_variable = { resources_found_continent_half_test = constant:resources_found_cave_runtime.half_multiplier }
	if = {
		limit = {
			NOT = { has_country_flag = resources_found_continent_half_reached }
			check_variable = {
				var = resources_found_continent_half_test
				value = resources_found_continent_eligible_state_count
				compare = greater_than_or_equals
			}
		}
		set_country_flag = resources_found_continent_half_reached
		country_event = { id = chaosx.nr18.91 hours = @CAVE_NOTICE_HOURS }
	}

	set_temp_variable = { resources_found_continent_three_quarter_left = resources_found_continent_owned_controlled_state_count }
	multiply_temp_variable = { resources_found_continent_three_quarter_left = constant:resources_found_cave_runtime.three_quarter_multiplier }
	set_temp_variable = { resources_found_continent_three_quarter_right = resources_found_continent_eligible_state_count }
	multiply_temp_variable = { resources_found_continent_three_quarter_right = constant:resources_found_cave_runtime.three_quarter_required }
	if = {
		limit = {
			NOT = { has_country_flag = resources_found_continent_three_quarter_reached }
			check_variable = {
				var = resources_found_continent_three_quarter_left
				value = resources_found_continent_three_quarter_right
				compare = greater_than_or_equals
			}
		}
		set_country_flag = resources_found_continent_three_quarter_reached
		country_event = { id = chaosx.nr18.92 hours = @CAVE_NOTICE_HOURS }
	}

	if = {
		limit = {
			check_variable = {
				var = resources_found_continent_remaining_state_count
				value = constant:resources_found_world_end.near_control_remaining_states
				compare = less_than_or_equals
			}
			check_variable = { resources_found_continent_remaining_state_count > constant:resources_found_value.zero }
			NOT = { has_country_flag = resources_found_last_continent_states_notified }
		}
		set_country_flag = resources_found_last_continent_states_notified
		country_event = { id = chaosx.nr18.93 hours = @CAVE_NOTICE_HOURS }
	}
}

# STATE scope. Repeated entries make selection proportional to resource bands.
resources_found_cave_append_weighted_foothold_candidate = {
	resources_found_refresh_captured_state_capacity = yes
	if = {
		limit = {
			check_variable = {
				var = resources_found_capacity_source_total
				value = constant:resources_found_cave_runtime.foothold_minimum_resource_total
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.zero }
		if = { limit = { check_variable = { resource@oil > constant:resources_found_value.zero } } add_to_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.one } }
		if = { limit = { check_variable = { resource@aluminium > constant:resources_found_value.zero } } add_to_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.one } }
		if = { limit = { check_variable = { resource@rubber > constant:resources_found_value.zero } } add_to_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.one } }
		if = { limit = { check_variable = { resource@tungsten > constant:resources_found_value.zero } } add_to_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.one } }
		if = { limit = { check_variable = { resource@steel > constant:resources_found_value.zero } } add_to_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.one } }
		if = { limit = { check_variable = { resource@chromium > constant:resources_found_value.zero } } add_to_temp_variable = { resources_found_foothold_distinct_resources = constant:resources_found_value.one } }

		resources_found_cave_append_foothold_candidate_once = yes
		if = {
			limit = { check_variable = { resources_found_capacity_source_total > constant:resources_found_cave_runtime.resource_weight_second_entry } }
			resources_found_cave_append_foothold_candidate_once = yes
		}
		if = {
			limit = { check_variable = { resources_found_capacity_source_total > constant:resources_found_cave_runtime.resource_weight_third_entry } }
			resources_found_cave_append_foothold_candidate_once = yes
		}
		if = {
			limit = {
				check_variable = {
					var = resources_found_foothold_distinct_resources
					value = constant:resources_found_cave_runtime.foothold_diversity_weight_threshold
					compare = greater_than_or_equals
				}
			}
			resources_found_cave_append_foothold_candidate_once = yes
		}
		if = {
			limit = {
				OR = {
					industrial_complex > constant:resources_found_cave_runtime.foothold_industry_weight_threshold
					arms_factory > constant:resources_found_cave_runtime.foothold_industry_weight_threshold
					dockyard > constant:resources_found_cave_runtime.foothold_industry_weight_threshold
				}
			}
			resources_found_cave_append_foothold_candidate_once = yes
		}
		if = {
			limit = {
				OR = {
					infrastructure > constant:resources_found_cave_runtime.foothold_transport_weight_threshold
					supply_node > constant:resources_found_value.zero
					rail_way > constant:resources_found_value.zero
				}
			}
			resources_found_cave_append_foothold_candidate_once = yes
		}
		if = { limit = { is_coastal = no } resources_found_cave_append_foothold_candidate_once = yes }
		if = { limit = { OWNER = { is_major = yes } } resources_found_cave_append_foothold_candidate_once = yes }
	}
}
