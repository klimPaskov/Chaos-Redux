# ============================================================================
# CHAOS REDUX - BIOLOGICAL COUNTERMEASURE SCRIPTED EFFECTS
# ============================================================================
# Overview:
# - Builds national Medical Capacity and Biological Security from real stores.
# - Commits and restores exact medical capacity for selected-state responses.
# - Applies surveillance, quarantine, hospitals, agent-specific antibiotics,
#   vaccination, and international-transparency consequences.
# - Releases exact-state medical reservations when a treated agent recovers or
#   a paid deployment reaches a state that is no longer eligible.
# - Resolves exact-state containment missions from recorded intensity and
#   event-driven outbound-spread history.
# - No helper scans the world or infers a target state.
# ============================================================================

@BIO_ANTIBIOTIC_COURSE_COOLDOWN_DAYS = 45

bio_countermeasure_debit_instrument_equipment = {
	set_temp_variable = { bio_countermeasure_instrument_remaining = bio_countermeasure_instrument_cost }
	clamp_temp_variable = { var = bio_countermeasure_instrument_remaining min = constant:cbrn_math.zero }

	set_temp_variable = { bio_countermeasure_instrument_available = num_equipment@cbrn_instrument_equipment_1 }
	set_temp_variable = { bio_countermeasure_instrument_take = bio_countermeasure_instrument_remaining }
	clamp_temp_variable = { var = bio_countermeasure_instrument_take min = constant:cbrn_math.zero max = bio_countermeasure_instrument_available }
	if = {
		limit = { check_variable = { bio_countermeasure_instrument_take > constant:cbrn_math.zero } }
		set_temp_variable = { bio_countermeasure_equipment_debit = bio_countermeasure_instrument_take }
		multiply_temp_variable = { bio_countermeasure_equipment_debit = constant:bio_lifecycle_math.negation }
		add_equipment_to_stockpile = {
			type = cbrn_instrument_equipment_1
			amount = bio_countermeasure_equipment_debit
		}
		subtract_from_temp_variable = { bio_countermeasure_instrument_remaining = bio_countermeasure_instrument_take }
	}

	set_temp_variable = { bio_countermeasure_instrument_available = num_equipment@cbrn_instrument_equipment_2 }
	set_temp_variable = { bio_countermeasure_instrument_take = bio_countermeasure_instrument_remaining }
	clamp_temp_variable = { var = bio_countermeasure_instrument_take min = constant:cbrn_math.zero max = bio_countermeasure_instrument_available }
	if = {
		limit = { check_variable = { bio_countermeasure_instrument_take > constant:cbrn_math.zero } }
		set_temp_variable = { bio_countermeasure_equipment_debit = bio_countermeasure_instrument_take }
		multiply_temp_variable = { bio_countermeasure_equipment_debit = constant:bio_lifecycle_math.negation }
		add_equipment_to_stockpile = {
			type = cbrn_instrument_equipment_2
			amount = bio_countermeasure_equipment_debit
		}
		subtract_from_temp_variable = { bio_countermeasure_instrument_remaining = bio_countermeasure_instrument_take }
	}

	set_temp_variable = { bio_countermeasure_instrument_available = num_equipment@cbrn_instrument_equipment_3 }
	set_temp_variable = { bio_countermeasure_instrument_take = bio_countermeasure_instrument_remaining }
	clamp_temp_variable = { var = bio_countermeasure_instrument_take min = constant:cbrn_math.zero max = bio_countermeasure_instrument_available }
	if = {
		limit = { check_variable = { bio_countermeasure_instrument_take > constant:cbrn_math.zero } }
		set_temp_variable = { bio_countermeasure_equipment_debit = bio_countermeasure_instrument_take }
		multiply_temp_variable = { bio_countermeasure_equipment_debit = constant:bio_lifecycle_math.negation }
		add_equipment_to_stockpile = {
			type = cbrn_instrument_equipment_3
			amount = bio_countermeasure_equipment_debit
		}
		subtract_from_temp_variable = { bio_countermeasure_instrument_remaining = bio_countermeasure_instrument_take }
	}
}

bio_countermeasure_reserve_field_hospital_capacity = {
	if = {
		limit = {
			NOT = { has_state_flag = bio_field_hospital_capacity_committed }
			NOT = { has_state_flag = bio_field_hospital_deployment_pending }
			NOT = { has_state_flag = field_hospitals_deployed }
			has_variable = bio_ordinary_outbreak_cycle
		}
		set_state_flag = bio_field_hospital_capacity_committed
		set_state_flag = bio_field_hospital_deployment_pending
		set_variable = { bio_field_hospital_medical_provider = ROOT }
		set_variable = { bio_field_hospital_medical_commitment = constant:bio_countermeasure_capacity.field_hospital_medical_commitment }
		set_variable = { bio_field_hospital_order_cycle = bio_ordinary_outbreak_cycle }
		ROOT = {
			cbrn_initialize_country_data = yes
			subtract_from_variable = { cbrn_medical_capacity = constant:bio_countermeasure_capacity.field_hospital_medical_commitment }
			clamp_variable = { var = cbrn_medical_capacity min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
		}
	}
}

bio_countermeasure_release_field_hospital_reservation = {

		limit = {
			has_state_flag = bio_field_hospital_capacity_committed
			has_variable = bio_field_hospital_medical_provider
			has_variable = bio_field_hospital_medical_commitment
			var:bio_field_hospital_medical_provider = { exists = yes }

			add_to_variable = { cbrn_medical_capacity = bio_countermeasure_refund_value }

	clr_state_flag = bio_field_hospital_deployment_pending
	clear_variable = bio_field_hospital_medical_provider
	clear_variable = bio_field_hospital_medical_commitment
	clear_variable = bio_field_hospital_order_cycle
	clear_temp_variable = bio_countermeasure_refund_value
}

bio_countermeasure_resolve_containment_mission = {
	if = {
		limit = { bio_countermeasure_containment_mission_record_is_current = yes }
		if = {
			limit = { NOT = { bio_lifecycle_state_has_any_ordinary_episode = yes } }
			set_state_flag = bio_containment_mission_success_history
			if = {
				limit = {
					has_variable = bio_containment_mission_provider
					var:bio_containment_mission_provider = { exists = yes }
				}
				var:bio_containment_mission_provider = {
					add_stability = constant:bio_countermeasure_response.containment_success_stability
				}
			}
		}
		else_if = {
			limit = { bio_countermeasure_containment_mission_is_success = yes }
			var:bio_containment_mission_provider = {
				bio_countermeasure_pay_containment_sustainment = yes
				add_stability = constant:bio_countermeasure_response.containment_success_stability
			}
			bio_countermeasure_apply_containment_success = yes
		}
		else_if = {
			limit = { bio_countermeasure_containment_mission_is_partial_success = yes }
			var:bio_containment_mission_provider = {
				bio_countermeasure_pay_containment_sustainment = yes
			}
			bio_countermeasure_apply_containment_partial_success = yes
		}
		else = {
			bio_countermeasure_apply_containment_failure = yes
		}
		bio_countermeasure_clear_containment_mission = yes
	}
	else_if = {
		limit = {
			OR = {
				has_state_flag = bio_containment_mission_active
				has_state_flag = bio_containment_mission_capacity_committed
			}
		}
		bio_countermeasure_clear_containment_mission = yes
	}
}
