# Chaos Redux humanitarian panel presentation.
# Read-only gameplay inputs. Writes are country UI state and temporary chain-local projections only.

humanitarian_panel_invalidate_actor = {
	clr_country_flag = famine_panel_open
	clr_country_flag = migration_panel_open
	clear_variable = famine_panel_selected_state
	clear_variable = migration_panel_selected_state
	clear_array = famine_panel_states
	clear_array = migration_panel_states
	clear_array = famine_panel_candidates
	clear_array = migration_panel_candidates
	set_variable = { famine_panel_state_count = constant:humanitarian_panel.zero }
	set_variable = { migration_panel_state_count = constant:humanitarian_panel.zero }
}

# STATE output temp, no state writes. Missing assessment stays a separate UI flag.
humanitarian_panel_read_food_stage = {
	set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.stable }
	if = { limit = { has_variable = famine_food_stage } set_temp_variable = { humanitarian_panel_food_stage = famine_food_stage } }
	else_if = { limit = { has_state_flag = famine_food_stage_catastrophic } set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.catastrophic_famine } }
	else_if = { limit = { has_state_flag = famine_food_stage_famine } set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.famine } }
	else_if = { limit = { has_state_flag = famine_food_stage_acute_shortage } set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.acute_shortage } }
	else_if = { limit = { has_state_flag = famine_food_stage_supply_strain } set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.supply_strain } }
}
humanitarian_panel_read_food_cause = {
	set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_none }
	set_temp_variable = { humanitarian_panel_food_cause_strength = constant:humanitarian_panel.zero }
	if = {
		limit = { has_variable = famine_component_production check_variable = { famine_component_production > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_production }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_production }
	}
	if = {
		limit = { has_variable = famine_component_transport check_variable = { famine_component_transport > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_transport }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_transport }
	}
	if = {
		limit = { has_variable = famine_component_extraction check_variable = { famine_component_extraction > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_extraction }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_extraction }
	}
	if = {
		limit = { has_variable = famine_component_environment check_variable = { famine_component_environment > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_environment }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_environment }
	}
	if = {
		limit = { has_variable = famine_component_governance check_variable = { famine_component_governance > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_governance }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_governance }
	}
	if = {
		limit = { has_variable = famine_component_need check_variable = { famine_component_need > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_need }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_need }
	}
	if = {
		limit = { has_variable = famine_component_vulnerability check_variable = { famine_component_vulnerability > humanitarian_panel_food_cause_strength } }
		set_temp_variable = { humanitarian_panel_food_cause_strength = famine_component_vulnerability }
		set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_vulnerability }
	}
}

famine_panel_open_window = {
	if = {
		limit = { famine_panel_is_eligible = yes }
		clr_country_flag = migration_panel_open
		set_country_flag = famine_panel_open
		if = { limit = { NOT = { has_variable = famine_panel_view } } set_variable = { famine_panel_view = constant:humanitarian_panel.overview } }
		if = { limit = { NOT = { has_variable = famine_panel_group } } set_variable = { famine_panel_group = constant:humanitarian_panel.group_first } }
		famine_panel_refresh = yes
		national_policy_panel_request_observer = yes
	}
	else = { humanitarian_panel_invalidate_actor = yes }
}
famine_panel_close_window = {
	if = { limit = { famine_panel_is_eligible = yes } clr_country_flag = famine_panel_open }
	else = { humanitarian_panel_invalidate_actor = yes }
}
# STATE, called only from the actor-bounded native iteration after relevance qualification.
famine_panel_collect_candidate = {
	if = {
		limit = { humanitarian_panel_state_belongs_to_actor = yes }
		event_target:humanitarian_panel_actor = {
			if = {
				limit = { NOT = { is_in_array = { array = famine_panel_candidates value = PREV } } }
				add_to_array = { array = famine_panel_candidates value = PREV }
			}
		}
	}
}
famine_panel_refresh = {
	set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.stable }
	set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_none }
	if = {
		limit = { famine_panel_is_eligible = yes }
		save_event_target_as = humanitarian_panel_actor
		clear_array = famine_panel_candidates
		clear_array = famine_panel_states
		# Native actor-bounded iterators cannot lose a late state in another country's registry prefix.
		# Qualification reads published subsystem facts/work only and never registers or assesses a state.
		every_owned_state = { limit = { famine_panel_state_is_relevant = yes } famine_panel_collect_candidate = yes }
		every_controlled_state = { limit = { famine_panel_state_is_relevant = yes } famine_panel_collect_candidate = yes }
		set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
		for_each_scope_loop = {
			array = famine_panel_candidates
			break = humanitarian_panel_list_break
			humanitarian_panel_read_food_stage = yes
			if = { limit = { check_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.catastrophic_famine } } event_target:humanitarian_panel_actor = { add_to_array = { array = famine_panel_states value = PREV } } }
		}
		set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
		for_each_scope_loop = {
			array = famine_panel_candidates
			break = humanitarian_panel_list_break
			humanitarian_panel_read_food_stage = yes
			if = { limit = { check_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.famine } } event_target:humanitarian_panel_actor = { add_to_array = { array = famine_panel_states value = PREV } } }
		}
		set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
		for_each_scope_loop = {
			array = famine_panel_candidates
			break = humanitarian_panel_list_break
			humanitarian_panel_read_food_stage = yes
			if = { limit = { check_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.acute_shortage } } event_target:humanitarian_panel_actor = { add_to_array = { array = famine_panel_states value = PREV } } }
		}
		set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
		for_each_scope_loop = {
			array = famine_panel_candidates
			break = humanitarian_panel_list_break
			humanitarian_panel_read_food_stage = yes
			if = { limit = { check_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.supply_strain } } event_target:humanitarian_panel_actor = { add_to_array = { array = famine_panel_states value = PREV } } }
		}
		set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
		for_each_scope_loop = {
			array = famine_panel_candidates
			break = humanitarian_panel_list_break
			humanitarian_panel_read_food_stage = yes
			if = { limit = { check_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.stable } } event_target:humanitarian_panel_actor = { add_to_array = { array = famine_panel_states value = PREV } } }
		}
		set_variable = { famine_panel_state_count = famine_panel_states^num }
		if = { limit = { NOT = { famine_panel_has_selected_state = yes } } clear_variable = famine_panel_selected_state }
		if = {
			limit = { NOT = { has_variable = famine_panel_selected_state } check_variable = { famine_panel_states^num > constant:humanitarian_panel.zero } }
			set_variable = { famine_panel_selected_state = famine_panel_states^0 }
		}
		famine_panel_project_overview = yes
		famine_panel_project_selection = yes
		famine_panel_project_frames = yes
		famine_panel_project_work = yes
		famine_gui_refresh_selected_action_status = yes
	}
	else = { humanitarian_panel_invalidate_actor = yes }
}

famine_panel_project_overview = {
	set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.stable }
	set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_none }
	clr_country_flag = famine_panel_food_assessed
	clr_country_flag = famine_panel_reserves_assessed
	set_variable = { famine_panel_food_stage = constant:famine_food_stage.stable }
	set_variable = { famine_panel_reserve_amount = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_reserve_need = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_reserve_days = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_relief = constant:humanitarian_panel.relief_not_required }
	set_variable = { famine_panel_priority_cause = constant:humanitarian_panel.cause_none }
	set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
	for_each_scope_loop = {
		array = famine_panel_states
		break = humanitarian_panel_list_break
		if = {
			limit = { famine_panel_state_has_food_assessment = yes }
			humanitarian_panel_read_food_stage = yes
			humanitarian_panel_read_food_cause = yes
			event_target:humanitarian_panel_actor = {
				set_country_flag = famine_panel_food_assessed
				if = {
					limit = { check_variable = { humanitarian_panel_food_stage > famine_panel_food_stage } }
					set_variable = { famine_panel_food_stage = humanitarian_panel_food_stage }
					set_variable = { famine_panel_priority_cause = humanitarian_panel_food_cause }
				}
			}
		}
		if = {
			limit = { has_state_flag = famine_food_reserve_initialized has_variable = famine_food_reserve_amount has_variable = famine_food_reserve_daily_need check_variable = { famine_food_reserve_daily_need > constant:humanitarian_panel.zero } }
			event_target:humanitarian_panel_actor = {
				set_country_flag = famine_panel_reserves_assessed
				add_to_variable = { famine_panel_reserve_amount = PREV.famine_food_reserve_amount }
				add_to_variable = { famine_panel_reserve_need = PREV.famine_food_reserve_daily_need }
			}
		}
		if = {
			limit = { has_state_flag = famine_food_security_active NOT = { has_state_flag = famine_relief_access_active } }
			event_target:humanitarian_panel_actor = { set_variable = { famine_panel_relief = constant:humanitarian_panel.relief_blocked } }
		}
		else_if = {
			limit = { has_state_flag = famine_food_security_active has_state_flag = famine_relief_access_active }
			event_target:humanitarian_panel_actor = { if = { limit = { NOT = { check_variable = { famine_panel_relief = constant:humanitarian_panel.relief_blocked } } } set_variable = { famine_panel_relief = constant:humanitarian_panel.relief_open } } }
		}
	}
	if = {
		limit = { has_country_flag = famine_panel_reserves_assessed check_variable = { famine_panel_reserve_need > constant:humanitarian_panel.zero } }
		set_variable = { famine_panel_reserve_days = famine_panel_reserve_amount }
		divide_variable = { famine_panel_reserve_days = famine_panel_reserve_need }
	}
}
migration_panel_project_overview = {
	clr_country_flag = migration_panel_displacement_assessed
	clr_country_flag = migration_panel_capacity_assessed
	set_variable = { migration_panel_fleeing = constant:humanitarian_panel.zero }
	set_variable = { migration_panel_trapped = constant:humanitarian_panel.zero }
	set_variable = { migration_panel_sheltered = constant:humanitarian_panel.zero }
	set_variable = { migration_panel_capacity = constant:humanitarian_panel.zero }
	set_variable = { migration_panel_capacity_used = constant:humanitarian_panel.zero }
	set_temp_variable = { humanitarian_panel_list_break = constant:humanitarian_panel.zero }
	for_each_scope_loop = {
		array = migration_panel_states
		break = humanitarian_panel_list_break
		if = { limit = { has_variable = migration_state_flight_population } event_target:humanitarian_panel_actor = { set_country_flag = migration_panel_displacement_assessed add_to_variable = { migration_panel_fleeing = PREV.migration_state_flight_population } } }
		if = { limit = { has_variable = migration_state_trapped_population } event_target:humanitarian_panel_actor = { set_country_flag = migration_panel_displacement_assessed add_to_variable = { migration_panel_trapped = PREV.migration_state_trapped_population } } }
		if = { limit = { has_variable = migration_state_reception_load } event_target:humanitarian_panel_actor = { add_to_variable = { migration_panel_sheltered = PREV.migration_state_reception_load } } }
	}
	if = {
		limit = { migration_reception_capacity_is_valid = yes has_variable = migration_reception_load }
		set_country_flag = migration_panel_capacity_assessed
		set_variable = { migration_panel_capacity = migration_reception_capacity }
		set_variable = { migration_panel_capacity_used = migration_reception_load }
	}
}

famine_panel_project_selection = {
	set_temp_variable = { humanitarian_panel_food_stage = constant:famine_food_stage.stable }
	set_temp_variable = { humanitarian_panel_food_cause = constant:humanitarian_panel.cause_none }
	clr_country_flag = famine_panel_selected_food_assessed
	clr_country_flag = famine_panel_selected_reserves_assessed
	set_variable = { famine_panel_selected_food_stage = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_selected_reserve_amount = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_selected_reserve_need = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_selected_reserve_days = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_selected_relief = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_selected_priority_cause = constant:humanitarian_panel.zero }
	if = {
		limit = { famine_panel_has_selected_state = yes }
		var:famine_panel_selected_state = {
			if = {
				limit = { famine_panel_state_has_food_assessment = yes }
				humanitarian_panel_read_food_stage = yes
				humanitarian_panel_read_food_cause = yes
				event_target:humanitarian_panel_actor = { set_country_flag = famine_panel_selected_food_assessed set_variable = { famine_panel_selected_food_stage = humanitarian_panel_food_stage } set_variable = { famine_panel_selected_priority_cause = humanitarian_panel_food_cause } }
			}
			if = {
				limit = { has_state_flag = famine_food_reserve_initialized has_variable = famine_food_reserve_amount has_variable = famine_food_reserve_daily_need check_variable = { famine_food_reserve_daily_need > constant:humanitarian_panel.zero } }
				event_target:humanitarian_panel_actor = { set_country_flag = famine_panel_selected_reserves_assessed set_variable = { famine_panel_selected_reserve_amount = PREV.famine_food_reserve_amount } set_variable = { famine_panel_selected_reserve_need = PREV.famine_food_reserve_daily_need } }
			}
			if = { limit = { has_state_flag = famine_food_security_active } event_target:humanitarian_panel_actor = { set_variable = { famine_panel_selected_relief = constant:humanitarian_panel.relief_blocked } } }
			if = { limit = { has_state_flag = famine_food_security_active has_state_flag = famine_relief_access_active } event_target:humanitarian_panel_actor = { set_variable = { famine_panel_selected_relief = constant:humanitarian_panel.relief_open } } }
		}
	}
	if = { limit = { has_country_flag = famine_panel_selected_reserves_assessed check_variable = { famine_panel_selected_reserve_need > constant:humanitarian_panel.zero } } set_variable = { famine_panel_selected_reserve_days = famine_panel_selected_reserve_amount } divide_variable = { famine_panel_selected_reserve_days = famine_panel_selected_reserve_need } }
}

			if = { limit = { has_variable = migration_state_trapped_population } event_target:humanitarian_panel_actor = { set_country_flag = migration_panel_selected_displacement_assessed set_variable = { migration_panel_selected_trapped = PREV.migration_state_trapped_population } } }
			if = { limit = { has_variable = migration_state_reception_load } event_target:humanitarian_panel_actor = { set_country_flag = migration_panel_selected_displacement_assessed set_variable = { migration_panel_selected_sheltered = PREV.migration_state_reception_load } } }
			if = { limit = { has_state_flag = migration_evacuation_prepared } event_target:humanitarian_panel_actor = { set_country_flag = migration_panel_selected_prepared } }
		}
	}
}

famine_panel_project_frames = {
	set_variable = { famine_panel_food_stage_frame = constant:humanitarian_panel.frame_unknown }
	if = { limit = { has_country_flag = famine_panel_food_assessed } set_variable = { famine_panel_food_stage_frame = famine_panel_food_stage } add_to_variable = { famine_panel_food_stage_frame = constant:humanitarian_panel.food_stage_frame_offset } }
	set_variable = { famine_panel_relief_frame = constant:humanitarian_panel.frame_unknown }
	if = { limit = { has_country_flag = famine_panel_food_assessed check_variable = { famine_panel_relief = constant:humanitarian_panel.relief_open } } set_variable = { famine_panel_relief_frame = constant:humanitarian_panel.frame_ready } }
	else_if = { limit = { has_country_flag = famine_panel_food_assessed check_variable = { famine_panel_relief = constant:humanitarian_panel.relief_blocked } } set_variable = { famine_panel_relief_frame = constant:humanitarian_panel.frame_urgent } }
}
migration_panel_project_frames = {
	set_variable = { migration_panel_displacement_frame = constant:humanitarian_panel.frame_unknown }
	if = { limit = { check_variable = { migration_panel_fleeing > constant:humanitarian_panel.zero } } set_variable = { migration_panel_displacement_frame = constant:humanitarian_panel.frame_ready } }
	if = { limit = { check_variable = { migration_panel_trapped > constant:humanitarian_panel.zero } } set_variable = { migration_panel_displacement_frame = constant:humanitarian_panel.frame_urgent } }
	set_variable = { migration_panel_capacity_frame = constant:humanitarian_panel.frame_unknown }
	if = {
		limit = { has_country_flag = migration_panel_capacity_assessed }
		set_variable = { migration_panel_capacity_frame = constant:humanitarian_panel.frame_ready }
		if = { limit = { NOT = { check_variable = { migration_panel_capacity_used < migration_panel_capacity } } } set_variable = { migration_panel_capacity_frame = constant:humanitarian_panel.frame_urgent } }
	}
}

# COUNTRY, exact pending UID rows and saved start/due dates. No timing estimate or work dispatch.

	clr_country_flag = famine_panel_has_active_work
	set_variable = { famine_panel_work_action = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_work_uid = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_work_start = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_work_due = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_work_remaining_days = constant:humanitarian_panel.zero }
	set_variable = { famine_panel_work_frame = constant:humanitarian_panel.frame_unknown }
	if = {
		limit = {
			famine_panel_has_selected_state = yes
			check_variable = { famine_action_ids^num = famine_action_types^num }
			check_variable = { famine_action_ids^num = famine_action_states^num }
			check_variable = { famine_action_ids^num = famine_action_dates^num }
			check_variable = { famine_action_ids^num = famine_action_starts^num }
		}
