	if = {
		limit = { check_variable = { fallout_event_114_selected_branch = constant:fallout_event_114_branch.use_forced_labor } }

			limit = { check_variable = { fallout_event_114_result_outcome = constant:fallout_event_outcome_type.partial } }

		}
		else = { set_temp_variable = { fallout_event_114_death_percent = constant:fallout_event_114_threshold.forced_failure_death_percent } }
	}
	else_if = {
		limit = { check_variable = { fallout_event_114_selected_branch = constant:fallout_event_114_branch.abandon_line } }
		set_temp_variable = { fallout_event_114_death_percent = constant:fallout_event_114_threshold.abandon_failure_death_percent }

		set_temp_variable = { fallout_event_114_death_percent = constant:fallout_event_114_threshold.neighbor_failure_death_percent }
	}
	if = {
		limit = {
			has_variable = fallout_event_114_result_state_id
			fallout_event_114_rail_registry_is_current = yes
		}
		var:fallout_event_114_result_state_id = {
			set_temp_variable = { fallout_event_114_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_114_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_114_death_request = fallout_event_114_death_percent }
			round_temp_variable = fallout_event_114_death_request
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:fallout_event_runtime.one }

			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }

		}
	}
	clear_temp_variable = fallout_event_114_death_percent
	clear_temp_variable = fallout_event_114_death_request
}
