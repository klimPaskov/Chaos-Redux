# ============================================================================
# CHAOS REDUX - EVENT 012 PRIORITY-MEMBER SOVEREIGN EFFECTS
# ============================================================================
# Event 012 registers one package-specific sovereign after full Action 102
# validation. The seven niche Event 006 shells own their dormant sovereign
# in country history. The nine normal vanilla carriers queue a hidden,
# event-owned recruitment transaction from the package registration effect.
# A settlement adds and promotes exactly one leader role, changes the ruling
# party only by explicit country choice, and names that party after the
# selected institution. Prior leaders are retained rather than retired. A
# historical sovereign role is fail-closed until its source-locked portrait
# has passed the runtime source gate; held rows keep their package politics
# but do not expose an unreviewed generated ruler.
# ============================================================================

africa_priority_member_queue_vanilla_sovereign_recruitment = {
	if = {
		limit = {
			africa_priority_member_is_vanilla_sovereign_carrier = yes
			NOT = { has_country_flag = africa_priority_member_vanilla_sovereign_recruitment_complete }
		}
		country_event = { id = africa_priority_member.1240 }
	}
}

africa_priority_member_install_council_government = {
