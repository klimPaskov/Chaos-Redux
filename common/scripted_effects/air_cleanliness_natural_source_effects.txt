# ============================================================================
# CHAOS REDUX - AIR CLEANLINESS NATURAL SOURCE EFFECTS
# ============================================================================
# Registers small wildfire-smoke and volcanic-ash contributions without a new
# periodic iterator. Event 013 calls the impact helper from STATE scope after
# family and severity are resolved. The host monthly Air Cleanliness pulse
# consumes and decays the capped global reservoir once per date.
#
# Public effects:
# - air_contamination_register_natural_disaster_source
# - air_contamination_prepare_natural_source_monthly
# ============================================================================

# Supported scope: STATE
# Inputs: natural_disaster_current_family and natural_disaster_current_severity
# temporary variables, plus persisted natural_disaster_sequence_id and
# natural_disaster_impact_index on the state.
air_contamination_register_natural_disaster_source = {
	set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.zero }

	if = {
		limit = {
			check_variable = { natural_disaster_current_family = constant:natural_disaster_family.wildfire }
			check_variable = {
				var = natural_disaster_current_severity
				value = constant:natural_disaster_severity.regional
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.wildfire_regional_bp }
		if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.wildfire_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.wildfire_abnormal_bp }
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.volcanic_eruption } }
		set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.volcanic_local_bp }
		if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.severe } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.volcanic_severe_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.regional } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.volcanic_regional_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.volcanic_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.volcanic_abnormal_bp }
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.ashfall } }
		set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.ashfall_local_bp }
		if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.severe } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.ashfall_severe_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.regional } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.ashfall_regional_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.ashfall_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.ashfall_abnormal_bp }
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.massive_eruption } }
		set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.massive_eruption_local_bp }
		if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.severe } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.massive_eruption_severe_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.regional } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.massive_eruption_regional_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.massive_eruption_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_current_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.massive_eruption_abnormal_bp }
		}
	}
	if = {
		limit = {
			OR = {
				has_global_flag = settings_air_cleanliness_disabled
				has_global_flag = air_contamination_final_silence_locked
			}
		}
		set_temp_variable = { air_natural_source_add_bp = constant:air_contamination_natural_source.zero }
	}

	if = {
		limit = { check_variable = { air_natural_source_add_bp > constant:air_contamination_natural_source.zero } }
		set_temp_variable = { air_natural_source_impact_is_duplicate = constant:air_contamination_natural_source.zero }
		if = {
			limit = {
				has_variable = air_contamination_natural_source_sequence_id
				has_variable = air_contamination_natural_source_impact_index
				has_variable = air_contamination_natural_source_family
				check_variable = { air_contamination_natural_source_sequence_id = natural_disaster_sequence_id }
				check_variable = { air_contamination_natural_source_impact_index = natural_disaster_impact_index }
				check_variable = { air_contamination_natural_source_family = natural_disaster_current_family }
			}
			set_temp_variable = { air_natural_source_impact_is_duplicate = constant:air_contamination_natural_source.one }
		}
		if = {
			limit = { check_variable = { air_natural_source_impact_is_duplicate < constant:air_contamination_natural_source.one } }
			set_variable = { air_contamination_natural_source_sequence_id = natural_disaster_sequence_id }
			set_variable = { air_contamination_natural_source_impact_index = natural_disaster_impact_index }
			set_variable = { air_contamination_natural_source_family = natural_disaster_current_family }
			if = {
				limit = { NOT = { has_variable = global.air_contamination_natural_source_reservoir_bp } }
				set_variable = { global.air_contamination_natural_source_reservoir_bp = constant:air_contamination_natural_source.zero }
			}
			add_to_variable = { global.air_contamination_natural_source_reservoir_bp = air_natural_source_add_bp }
			clamp_variable = {
				var = global.air_contamination_natural_source_reservoir_bp
				min = constant:air_contamination_natural_source.zero
				max = constant:air_contamination_natural_source.monthly_cap_bp
			}
		}
	}
}

# Supported scope: STATE, called when a qualifying ash aftermath card opens.
# The impact writer already covers the physical event. This second receipt
# represents settled ash in exposed streets and roof systems. It is limited to
# ash-heavy families and can write only once for a stored disaster sequence.
air_contamination_register_natural_disaster_aftermath_source = {
	set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.zero }
	if = {
		limit = {
			check_variable = { natural_disaster_family = constant:natural_disaster_family.ashfall }
			check_variable = {
				var = natural_disaster_severity
				value = constant:natural_disaster_severity.regional
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.ashfall_aftermath_regional_bp }
		if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.ashfall_aftermath_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.ashfall_aftermath_abnormal_bp }
		}
	}
	else_if = {
		limit = {
			check_variable = { natural_disaster_family = constant:natural_disaster_family.volcanic_eruption }
			check_variable = {
				var = natural_disaster_severity
				value = constant:natural_disaster_severity.severe
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.volcanic_aftermath_severe_bp }
		if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.regional } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.volcanic_aftermath_regional_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.volcanic_aftermath_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.volcanic_aftermath_abnormal_bp }
		}
	}
	else_if = {
		limit = {
			check_variable = { natural_disaster_family = constant:natural_disaster_family.massive_eruption }
			check_variable = {
				var = natural_disaster_severity
				value = constant:natural_disaster_severity.severe
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.massive_eruption_aftermath_severe_bp }
		if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.regional } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.massive_eruption_aftermath_regional_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.catastrophic } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.massive_eruption_aftermath_catastrophic_bp }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.abnormal } }
			set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.massive_eruption_aftermath_abnormal_bp }
		}
	}
	if = {
		limit = {
			OR = {
				has_global_flag = settings_air_cleanliness_disabled
				has_global_flag = air_contamination_final_silence_locked
			}
		}
		set_temp_variable = { air_natural_aftermath_add_bp = constant:air_contamination_natural_source.zero }
	}
	if = {
		limit = {
			check_variable = { air_natural_aftermath_add_bp > constant:air_contamination_natural_source.zero }
			has_variable = natural_disaster_sequence_id
			OR = {
				NOT = { has_variable = air_contamination_natural_aftermath_sequence_id }
				NOT = {
					check_variable = {
						var = air_contamination_natural_aftermath_sequence_id
						value = natural_disaster_sequence_id
						compare = equals
					}
				}
			}
		}
		set_variable = { air_contamination_natural_aftermath_sequence_id = natural_disaster_sequence_id }
		if = {
			limit = { NOT = { has_variable = global.air_contamination_natural_source_reservoir_bp } }
			set_variable = { global.air_contamination_natural_source_reservoir_bp = constant:air_contamination_natural_source.zero }
		}
		add_to_variable = { global.air_contamination_natural_source_reservoir_bp = air_natural_aftermath_add_bp }
		clamp_variable = {
			var = global.air_contamination_natural_source_reservoir_bp
			min = constant:air_contamination_natural_source.zero
			max = constant:air_contamination_natural_source.monthly_cap_bp
		}
	}
	clear_temp_variable = air_natural_aftermath_add_bp
}

# Supported scope: COUNTRY, called by the existing host monthly pulse.
# Output: air_monthly_natural_bp temporary variable.
air_contamination_prepare_natural_source_monthly = {
	if = {
		limit = { NOT = { has_variable = global.air_contamination_natural_source_reservoir_bp } }
		set_variable = { global.air_contamination_natural_source_reservoir_bp = constant:air_contamination_natural_source.zero }
	}
	clamp_variable = {
		var = global.air_contamination_natural_source_reservoir_bp
		min = constant:air_contamination_natural_source.zero
		max = constant:air_contamination_natural_source.monthly_cap_bp
	}

	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.air_contamination_natural_source_last_tick_date }
				NOT = {
					check_variable = {
						var = global.air_contamination_natural_source_last_tick_date
						value = global.date
						compare = equals
					}
				}
			}
		}
		set_variable = { global.air_contamination_natural_source_last_tick_date = global.date }
		set_variable = { global.air_contamination_natural_source_bp = global.air_contamination_natural_source_reservoir_bp }
		subtract_from_variable = { global.air_contamination_natural_source_reservoir_bp = constant:air_contamination_natural_source.monthly_decay_bp }
		clamp_variable = {
			var = global.air_contamination_natural_source_reservoir_bp
			min = constant:air_contamination_natural_source.zero
			max = constant:air_contamination_natural_source.monthly_cap_bp
		}
	}
	if = {
		limit = { NOT = { has_variable = global.air_contamination_natural_source_bp } }
		set_variable = { global.air_contamination_natural_source_bp = constant:air_contamination_natural_source.zero }
	}
	if = {
		limit = { has_global_flag = air_contamination_final_silence_locked }
		set_variable = { global.air_contamination_natural_source_bp = constant:air_contamination_natural_source.zero }
	}
	set_temp_variable = { air_monthly_natural_bp = global.air_contamination_natural_source_bp }
}
