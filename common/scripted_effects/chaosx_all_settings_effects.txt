# ============================================================================
# CHAOS REDUX - ALL SETTINGS WINDOW EFFECTS
# ============================================================================
# Open and close chaosx_all_settings_window (interface/chaosx_all_settings.gui).
# The window's controls call the existing settings effects directly; these helpers
# only manage its visibility flag. Supported scope: COUNTRY (the player).
# ============================================================================

toggle_chaosx_all_settings_window = {
	if = {
		limit = { has_country_flag = chaosx_all_settings_open }
		clr_country_flag = chaosx_all_settings_open
	}
	else = {
		set_country_flag = chaosx_all_settings_open
	}
}

close_chaosx_all_settings_window = {
	clr_country_flag = chaosx_all_settings_open
}
