africa_clear_prefire_context = {
	set_temp_variable = { africa_prefire_ready = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_selection_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_selection_weight = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_pool_weight_remaining = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_prefire_contact_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_prefire_contact_target_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_prefire_contact_slot_cursor = @CR_SC_AFRICA_VALUE_ZERO }
	clear_temp_array = africa_host_selection_pool
	clear_temp_array = africa_prefire_contact_pool
}

# COUNTRY scope. Produces an integer host weight from every accepted design
# factor. There is deliberately no human-player bonus.
africa_prepare_host_selection_weight = {
	set_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.base_weight }
	if = {
		limit = { is_subject = no }
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.independent_bonus }
	}
	else_if = {
		limit = {
			is_subject = yes
			compare_autonomy_state > autonomy_colony
			capital_scope = { is_controlled_by = PREV }
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.high_autonomy_subject_bonus }
	}
	if = {
		limit = {
			has_war = yes
			any_enemy_country = {
				capital_scope = { NOT = { is_on_continent = africa } }
				any_owned_state = { is_on_continent = africa }
			}
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.colonial_war_bonus }
	}
	if = {
		limit = {
			OR = {
				any_guaranteed_country = {
					capital_scope = { is_on_continent = africa }
				}
				any_country = {
					capital_scope = { is_on_continent = africa }
					has_volunteers_amount_from = {
						tag = PREV
						count > @CR_SC_AFRICA_VALUE_ZERO
					}
				}
			}
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.african_support_bonus }
	}
	if = {
		limit = {
			is_major = no
			num_of_factories > constant:africa_host_selection.medium_factory_minimum
			num_of_factories < constant:africa_host_selection.medium_factory_maximum
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.medium_power_bonus }
	}
	if = {
		limit = { is_major = yes }
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.major_penalty }
	}
	if = {
		limit = {
			capital_scope = { is_island_state = yes }
			num_of_factories < constant:africa_host_selection.tiny_factory_maximum
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.tiny_island_penalty }
	}
	if = {
		limit = { surrender_progress > constant:africa_host_selection.near_capitulation_threshold }
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.near_capitulation_penalty }
	}
	set_temp_variable = { africa_host_african_neighbor_count = @CR_SC_AFRICA_VALUE_ZERO }
	every_neighbor_country = {
		limit = {
			capital_scope = { is_on_continent = africa }
		}
		add_to_temp_variable = { africa_host_african_neighbor_count = constant:africa_value.one }
	}
	if = {
		limit = {
			check_variable = {
				var = africa_host_african_neighbor_count
				value = constant:africa_host_selection.several_neighbors_minimum
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.several_neighbors_bonus }
	}
	if = {
		limit = {
			capital_scope = { is_island_state = yes }
			check_variable = { africa_host_african_neighbor_count = @CR_SC_AFRICA_VALUE_ZERO }
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.overseas_contact_penalty }
	}
	round_temp_variable = africa_host_selection_weight
	clamp_temp_variable = {
		var = africa_host_selection_weight
		min = constant:africa_host_selection.minimum_weight
		max = constant:africa_host_selection.maximum_weight
	}
	set_temp_variable = { africa_host_african_neighbor_count = @CR_SC_AFRICA_VALUE_ZERO }
}

# COUNTRY scope on a prospective host. This is called only from the explicit
# fire-once selector. It creates a temporary, distinct roster using the
# playbook-mapped regional predicate and never persists a world scan.
africa_build_prefire_contact_pool = {
	clear_temp_array = africa_prefire_contact_pool
	set_temp_variable = { africa_prefire_contact_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	save_event_target_as = africa_prefire_host_candidate
	every_country = {
		limit = {
			africa_is_valid_relationship_target = yes
			africa_matches_prefire_host_contact_pool = yes
			NOT = { tag = event_target:africa_prefire_host_candidate }
		}
		add_to_temp_array = { africa_prefire_contact_pool = THIS }
		add_to_temp_variable = { africa_prefire_contact_pool_count = constant:africa_value.one }
	}
}

# COUNTRY scope on the selected prospective host. Three to five distinct real
# governments are frozen before Event 12 commits. The delayed contact event
# consumes these regular targets and never reopens a country scan.
africa_freeze_prefire_contact_roster = {
	set_temp_variable_to_random = {
		var = africa_prefire_contact_target_count
		min = constant:africa_opening_contact.minimum_frozen_contacts
		max = constant:africa_opening_contact.maximum_frozen_contacts
		integer = yes
	}
	if = {
		limit = { check_variable = { africa_prefire_contact_target_count > africa_prefire_contact_pool_count } }
		set_temp_variable = { africa_prefire_contact_target_count = africa_prefire_contact_pool_count }
	}
	set_temp_variable = { africa_prefire_contact_slot_cursor = @CR_SC_AFRICA_VALUE_ZERO }
	clear_array = africa_frozen_first_contact_targets
	while_loop_effect = {
		limit = {
			check_variable = { africa_prefire_contact_slot_cursor < africa_prefire_contact_target_count }
			check_variable = { africa_prefire_contact_pool_count > @CR_SC_AFRICA_VALUE_ZERO }
		}
		random_scope_in_array = {
			array = africa_prefire_contact_pool
			add_to_temp_variable = { africa_prefire_contact_slot_cursor = constant:africa_value.one }
			if = { limit = { check_variable = { africa_prefire_contact_slot_cursor = constant:africa_value.one } } save_event_target_as = africa_first_contact_slot_1 }
			else_if = { limit = { check_variable = { africa_prefire_contact_slot_cursor = constant:africa_value.two } } save_event_target_as = africa_first_contact_slot_2 }
			else_if = { limit = { check_variable = { africa_prefire_contact_slot_cursor = constant:africa_value.three } } save_event_target_as = africa_first_contact_slot_3 }
			else_if = { limit = { check_variable = { africa_prefire_contact_slot_cursor = constant:africa_value.four } } save_event_target_as = africa_first_contact_slot_4 }
			else_if = { limit = { check_variable = { africa_prefire_contact_slot_cursor = constant:africa_value.five } } save_event_target_as = africa_first_contact_slot_5 }
			set_country_flag = africa_prefire_contact_slot_member
			if = {
				limit = { has_war = yes }
				set_country_flag = africa_prefire_contact_threatened
			}
			else_if = {
				limit = { is_subject = yes }
				set_country_flag = africa_prefire_contact_liberation
			}
			else = { set_country_flag = africa_prefire_contact_proof_demand }
			event_target:africa_prefire_host_candidate = {
				add_to_array = { array = africa_frozen_first_contact_targets value = PREV }
			}
			remove_from_temp_array = { array = africa_prefire_contact_pool value = THIS }
		}
		subtract_from_temp_variable = { africa_prefire_contact_pool_count = constant:africa_value.one }
	}
	set_variable = { africa_frozen_first_contact_count = africa_prefire_contact_slot_cursor }
	if = {
		limit = {
			check_variable = {
				var = africa_frozen_first_contact_count
				value = constant:africa_opening_contact.minimum_frozen_contacts
				compare = greater_than_or_equals
			}
		}
		set_country_flag = africa_prefire_contacts_frozen
	}
	clear_temp_array = africa_prefire_contact_pool
	set_temp_variable = { africa_prefire_contact_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_prefire_contact_target_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_prefire_contact_slot_cursor = @CR_SC_AFRICA_VALUE_ZERO }
}

# COUNTRY scope. Duplicated scope entries are the engine-supported weighted
# array pattern already used by Fury and Event 16.
africa_add_current_country_to_host_selection_pool = {
	africa_build_prefire_contact_pool = yes
	if = {
		limit = {
			OR = {
				NOT = { tag = SAF }
				africa_rsa_prefire_contact_pool_has_patron = yes
			}
			check_variable = {
				var = africa_prefire_contact_pool_count
				value = constant:africa_opening_contact.minimum_frozen_contacts
				compare = greater_than_or_equals
			}
		}
		africa_prepare_host_selection_weight = yes
		set_temp_variable = { africa_host_pool_weight_remaining = africa_host_selection_weight }
		while_loop_effect = {
			limit = { check_variable = { africa_host_pool_weight_remaining > @CR_SC_AFRICA_VALUE_ZERO } }
			add_to_temp_array = { africa_host_selection_pool = THIS }
			add_to_temp_variable = { africa_host_selection_pool_count = constant:africa_value.one }
			subtract_from_temp_variable = { africa_host_pool_weight_remaining = constant:africa_value.one }
		}
	}
	clear_temp_array = africa_prefire_contact_pool
	set_temp_variable = { africa_prefire_contact_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_pool_weight_remaining = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_selection_weight = @CR_SC_AFRICA_VALUE_ZERO }
}

# Any scope. This explicit fire-once scan is never called from a periodic hook.
africa_select_weighted_prefire_host = {
	clear_temp_array = africa_host_selection_pool
	set_temp_variable = { africa_host_selection_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	every_country = {
		limit = {
			OR = {
				africa_is_eligible_host = yes
				africa_rsa_shared_entry_is_eligible = yes
			}
		}
		africa_add_current_country_to_host_selection_pool = yes
	}
	if = {
		limit = { check_variable = { africa_host_selection_pool_count > @CR_SC_AFRICA_VALUE_ZERO } }
			random_scope_in_array = {
				array = africa_host_selection_pool
				save_event_target_as = africa_prefire_host
			}
		event_target:africa_prefire_host = {
			africa_build_prefire_contact_pool = yes
			africa_freeze_prefire_contact_roster = yes
		}
	}
	clear_temp_array = africa_host_selection_pool
	set_temp_variable = { africa_host_selection_pool_count = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_pool_weight_remaining = @CR_SC_AFRICA_VALUE_ZERO }
	set_temp_variable = { africa_host_selection_weight = @CR_SC_AFRICA_VALUE_ZERO }
}

# Any scope. Output is the regular africa_prefire_host target that survives into
# the entry event fired from the same effect chain.
africa_prepare_random_event_fire = {
	africa_clear_prefire_context = yes
	if = {
		limit = { africa_automatic_event_is_available = yes }
		africa_select_weighted_prefire_host = yes
		if = {
			limit = {
				has_event_target = africa_prefire_host
				event_target:africa_prefire_host = {
					OR = {
						africa_is_eligible_host = yes
						africa_rsa_shared_entry_is_eligible = yes
					}
					has_country_flag = africa_prefire_contacts_frozen
					check_variable = {
						var = africa_frozen_first_contact_count
						value = constant:africa_opening_contact.minimum_frozen_contacts
						compare = greater_than_or_equals
					}
				}
			}
			set_temp_variable = { africa_prefire_ready = constant:africa_value.one }
		}
		else_if = {
			limit = { has_event_target = africa_prefire_host }
			# A rejected frozen roster must not poison a later explicit fire
			# attempt with stale contact flags or an old patron count.
			event_target:africa_prefire_host = {
				africa_clear_frozen_first_contact_context = yes
			}
		}
	}
}

# COUNTRY scope. Weak hosts receive emergency capacity and narrower early work;
# strong hosts receive less free help and begin under greater suspicion.
africa_classify_host_support = {
	set_variable = { africa_host_support_class = constant:africa_host_support.regional }
	if = {
		limit = {
			OR = {
				num_of_factories < constant:africa_host_selection.weak_factory_maximum
				num_of_controlled_states < constant:africa_host_selection.weak_state_maximum
			}
		}
		set_variable = { africa_host_support_class = constant:africa_host_support.weak }
	}
	else_if = {
		limit = {
			OR = {
				num_of_factories > constant:africa_host_selection.strong_factory_minimum
				num_of_controlled_states > constant:africa_host_selection.strong_state_minimum
			}
		}
		set_variable = { africa_host_support_class = constant:africa_host_support.strong }
	}
}

# COUNTRY scope on the initialized host. The three distinct contact roles are
# optional independently. The fire-once gate guarantees at least one mapped
# contact; losing every delegation during the short delay records a real failure.
africa_prepare_first_contact_targets = {
	random_country = {
		limit = {
			africa_is_valid_relationship_target = yes
			africa_matches_current_host_overlay = yes
			NOT = { tag = ROOT }
			has_war = yes
		}
		save_event_target_as = africa_first_contact_threatened
	}
	random_country = {
		limit = {
			africa_is_valid_relationship_target = yes
			africa_matches_current_host_overlay = yes
			NOT = { tag = ROOT }
			is_subject = yes
			NOT = { tag = event_target:africa_first_contact_threatened }
		}
		save_event_target_as = africa_first_contact_subject
	}
	random_country = {
		limit = {
			africa_is_valid_relationship_target = yes
			africa_matches_current_host_overlay = yes
			NOT = { tag = ROOT }
			is_subject = no
			NOT = { tag = event_target:africa_first_contact_threatened }
			NOT = { tag = event_target:africa_first_contact_subject }
		}
		save_event_target_as = africa_first_contact_regional_power
	}
}

# COUNTRY scope on the host. Input is africa_contact_to_register, copied from
# one of the five frozen slots. The selected government becomes a real bounded
# relationship/action target; no option grants membership or integration.
africa_register_frozen_first_contact = {
	if = {
		limit = {
			africa_is_current_host = yes
			has_event_target = africa_contact_to_register
			event_target:africa_contact_to_register = {
				africa_is_valid_relationship_target = yes
				has_country_flag = africa_prefire_contact_slot_member
			}
		}
		africa_register_contact_target = yes
		if = {
			limit = { check_variable = { africa_contact_registration_result > @CR_SC_AFRICA_VALUE_ZERO } }
			event_target:africa_contact_to_register = {
				if = {
					limit = { has_country_flag = africa_prefire_contact_threatened }
					set_country_flag = africa_contact_protection_priority
					set_country_flag = africa_member_requested_protection
					if = { limit = { NOT = { has_variable = africa_member_confidence } } set_variable = { africa_member_confidence = constant:africa_measure.medium } }
					set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.protected }
					africa_apply_relationship_transition = yes
				}
				else_if = {
					limit = { has_country_flag = africa_prefire_contact_liberation }
					set_country_flag = africa_contact_liberation_priority
					set_country_flag = africa_liberation_movement_recognised
					if = { limit = { NOT = { has_variable = africa_member_confidence } } set_variable = { africa_member_confidence = constant:africa_measure.medium } }
				}
				else = {
					set_country_flag = africa_contact_proof_demand
					if = { limit = { NOT = { has_variable = africa_member_confidence } } set_variable = { africa_member_confidence = constant:africa_measure.medium } }
				}
				set_country_flag = africa_first_contact_priority
			}
			set_country_flag = africa_first_contact_selected
			set_country_flag = africa_first_contact_roster_consumed
		}
	}
	if = {
		limit = { has_country_flag = africa_first_contact_selected }
		africa_clear_frozen_first_contact_context = yes
	}
	else = { africa_record_lost_first_delegation = yes }
}

# COUNTRY scope on a registered Event 012 member from relationship registration,
# on_join_faction, or on_leave_faction. Foreign patronage is a derived receipt:
# it exists only while a current-generation member is in a faction whose leader
# is neither the Event 012 host, another current-generation member, nor a leader
# sharing the host's faction. The helper is deliberately bounded to the member
# country; it does not scan the world or infer patronage from opinion.
africa_reconcile_foreign_patronage_state = {
	if = {
		limit = {
			has_event_target = africa_host
			africa_member_host_generation_is_current = yes
			NOT = { africa_is_current_host = yes }
			is_in_faction = yes
			OR = {
				africa_relationship_is_protected = yes
				africa_relationship_is_associate = yes
				africa_relationship_is_chartered = yes
				africa_relationship_is_autonomous_federal = yes
				africa_relationship_is_integrated_region = yes
				africa_relationship_is_resistant = yes
				africa_relationship_is_leaving = yes
				africa_relationship_is_rival_bloc = yes
				africa_relationship_is_occupied_settlement = yes
			}
			faction_leader = {
				exists = yes
				is_faction_leader = yes
				NOT = { africa_is_current_host = yes }
				NOT = { africa_member_host_generation_is_current = yes }
				NOT = { is_in_faction_with = event_target:africa_host }
			}
		}
		set_country_flag = africa_foreign_patronage_active
	}
	else = {
		clr_country_flag = africa_foreign_patronage_active
	}
}

				compare = greater_than_or_equals

# controlled replacement tree for generic hosts; original SAF keeps its
# meaningful vanilla tree while the RSA rupture runs alongside it.
africa_load_continental_focus_tree = {
	if = {
		limit = {
			africa_is_current_host = yes
			NOT = { original_tag = SAF }
			NOT = { has_focus_tree = africa_continental_focus_tree }
			has_event_target = africa_focus_completion_source
		}
		load_focus_tree = {
			tree = africa_continental_focus_tree
			keep_completed = yes
			copy_completed_from = event_target:africa_focus_completion_source
		}
	}
	else_if = {
		limit = {
			africa_is_current_host = yes
			NOT = { original_tag = SAF }
			NOT = { has_focus_tree = africa_continental_focus_tree }
		}
		load_focus_tree = {
			tree = africa_continental_focus_tree
			keep_completed = yes
		}
	}
	if = {
		limit = {
			africa_is_current_host = yes
			has_focus_tree = africa_continental_focus_tree
			NOT = { has_country_flag = africa_continental_focus_tree_loaded }
		}
		set_country_flag = africa_continental_focus_tree_loaded
		mark_focus_tree_layout_dirty = yes
	}
}

africa_refresh_continental_focus_tree_layout = {
	if = {
		limit = {
			africa_is_current_host = yes
			has_focus_tree = africa_continental_focus_tree
		}
		mark_focus_tree_layout_dirty = yes
	}
}
