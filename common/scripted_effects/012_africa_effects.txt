africa_score_unifier_candidate_for_selection = {
	set_temp_variable = { africa_unifier_selection_weight = 0 }
	if = {
		limit = { is_africa_valid_unifier_candidate = yes }
		set_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.base }
		if = {
			limit = { is_africa_independent_unifier_candidate = yes }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.independent_african_capital }
		}
		if = {
			limit = { is_africa_subject_liberation_unifier_candidate = yes }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.subject_liberation_story }
		}
		if = {
			limit = {
				has_war = yes
				any_enemy_country = {
					NOT = { capital_scope = { is_on_continent = africa } }
				}
			}
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.at_war_with_foreign_holder }
		}
		if = {
			limit = { has_country_flag = independence_wave_breakaway }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.recent_chaos_liberation_state }
		}
		if = {
			limit = { is_africa_rsa_allies_unifier_candidate = yes }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.rsa_allies_branch }
		}
		if = {
			limit = { num_of_controlled_states < 3 }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.small_country_bonus }
		}
	}
}

africa_add_current_unifier_candidate_to_selection_pool = {
	africa_score_unifier_candidate_for_selection = yes
	round_temp_variable = africa_unifier_selection_weight
	clamp_temp_variable = {
		var = africa_unifier_selection_weight
		min = constant:africa_selection_weight.minimum
		max = constant:africa_selection_weight.maximum
	}
	set_temp_variable = { africa_unifier_pool_weight_remaining = africa_unifier_selection_weight }
	while_loop_effect = {
		limit = { check_variable = { africa_unifier_pool_weight_remaining > 0 } }
		add_to_temp_array = { africa_unifier_selection_pool = THIS }
		add_to_temp_variable = { africa_unifier_selection_pool_count = 1 }
		subtract_from_temp_variable = { africa_unifier_pool_weight_remaining = 1 }
	}
	set_temp_variable = { africa_unifier_pool_weight_remaining = 0 }
	set_temp_variable = { africa_unifier_selection_weight = 0 }
}

africa_select_weighted_unifier_candidate = {
	clear_temp_array = africa_unifier_selection_pool
	set_temp_variable = { africa_unifier_selection_pool_count = 0 }
	set_temp_variable = { africa_weighted_unifier_selected = 0 }
	every_country = {
		limit = { is_africa_valid_unifier_candidate = yes }
		africa_add_current_unifier_candidate_to_selection_pool = yes
	}
	if = {
		limit = { check_variable = { africa_unifier_selection_pool_count > 0 } }
		random_scope_in_array = {
			array = africa_unifier_selection_pool
			save_event_target_as = africa_weighted_unifier_candidate
			set_temp_variable = { africa_weighted_unifier_selected = 1 }
		}
	}
	if = {
		limit = { check_variable = { africa_weighted_unifier_selected > 0 } }
		event_target:africa_weighted_unifier_candidate = {
			africa_prepare_runtime_context_from_this = yes
		}
	}
	clear_temp_array = africa_unifier_selection_pool
	set_temp_variable = { africa_unifier_selection_pool_count = 0 }
	set_temp_variable = { africa_unifier_pool_weight_remaining = 0 }
	set_temp_variable = { africa_unifier_selection_weight = 0 }
}

africa_triggerable_scenario_launch_selected = {
	set_global_flag = africa_triggerable_scenario_started
	set_variable = { global.africa_triggerable_scenario_intensity = triggerable_scenarios_intensity }
	set_variable = { global.africa_triggerable_scenario_type = triggerable_scenarios_africa_type }
	if = {
		limit = {
			check_variable = { triggerable_scenarios_africa_type = constant:triggerable_scenario_africa_type.rsa_civil_war }
			SAF = { is_africa_rsa_allies_unifier_candidate = yes }
		}
		set_global_flag = africa_triggerable_scenario_rsa_civil_war
		SAF = {
			africa_prepare_runtime_context_from_this = yes
			africa_start_rsa_allies_civil_war = yes
		}
		if = {
			limit = { has_africa_runtime_context = yes }
			event_target:africa_unifier_country = {
				set_country_flag = africa_triggerable_scenario_actor
				set_country_flag = africa_triggerable_scenario_rsa_civil_war
				africa_apply_triggerable_scenario_starting_forces = yes
			}
		}
	}
	else = {
		africa_select_random_unifier_candidate = yes
		if = {
			limit = { NOT = { has_africa_runtime_context = yes } }
			africa_triggerable_scenario_seed_wac_host = yes
		}
		if = {
			limit = { has_africa_runtime_context = yes }
			event_target:africa_unifier_country = {
				africa_establish_union_start = yes
				africa_apply_triggerable_scenario_setup = yes
			}
		}
	}
}

africa_select_next_unopened_dossier = {
	clear_variable = africa_selected_dossier_id
	clear_variable = africa_selected_dossier_seat_state
	set_temp_variable = { africa_dossier_scan_index = 0 }
	set_temp_variable = { africa_dossier_found = 0 }
	while = {
		limit = {
			check_variable = { africa_dossier_scan_index < global.africa_authority_atlas_dossier_ids^num }
			check_variable = { africa_dossier_found < 1 }
		}
		set_temp_variable = { africa_candidate_dossier_id = global.africa_authority_atlas_dossier_ids^africa_dossier_scan_index }
		if = {
			limit = {
				NOT = { is_in_array = { array = africa_opened_dossier_ids value = africa_candidate_dossier_id } }
			}
			set_variable = { africa_selected_dossier_id = africa_candidate_dossier_id }
			set_temp_variable = { africa_dossier_found = 1 }
		}
		add_to_temp_variable = { africa_dossier_scan_index = 1 }
	}
	if = {
		limit = { check_variable = { africa_dossier_found < 1 } }
		set_country_flag = africa_all_historical_dossiers_opened
	}
	else = {
		africa_set_selected_dossier_seat_state = yes
	}
}

	if = {
		limit = {
			has_country_flag = africa_sponsored_europe_charter
			OR = {
				has_country_flag = africa_sponsored_asia_charter
				has_country_flag = africa_sponsored_middle_east_charter
			}
		}
		africa_apply_dynamic_union_afro_eurasian_identity = yes
	}
	else_if = {
		limit = { has_country_flag = africa_sponsored_asia_charter }
		africa_apply_dynamic_union_afro_asian_identity = yes
	}
	else_if = {
		limit = {
			has_country_flag = africa_sponsored_middle_east_charter
			has_country_flag = africa_sponsored_south_atlantic_charter
		}
		africa_apply_dynamic_union_congress_of_continents_identity = yes
	}
	else_if = {
		limit = { has_country_flag = africa_sponsored_middle_east_charter }
		africa_apply_dynamic_union_african_middle_eastern_identity = yes
	}
	else_if = {
		limit = { has_country_flag = africa_sponsored_south_atlantic_charter }
		africa_apply_dynamic_union_afro_atlantic_identity = yes
	}
	else = {
		africa_apply_dynamic_union_congress_of_continents_identity = yes
	}
}

africa_refresh_external_continent_unifier_proofs = {
	set_variable = { africa_external_continent_unifier_proof_count = 0 }
	if = {
		limit = { has_country_flag = africa_middle_east_unifier_proof_verified }
		add_to_variable = { africa_external_continent_unifier_proof_count = 1 }
	}
	if = {
		limit = { has_country_flag = africa_asia_unifier_proof_verified }
		add_to_variable = { africa_external_continent_unifier_proof_count = 1 }
	}
	if = {
		limit = { has_country_flag = africa_europe_unifier_proof_verified }
		add_to_variable = { africa_external_continent_unifier_proof_count = 1 }
	}
	if = {
		limit = { has_country_flag = africa_south_atlantic_unifier_proof_verified }
		add_to_variable = { africa_external_continent_unifier_proof_count = 1 }
	}
	if = {
		limit = {
			check_variable = {
				var = africa_external_continent_unifier_proof_count
				value = constant:africa_world_is_one_requirement.external_continent_unifier_proofs
				compare = greater_than_or_equals
			}
		}
		if = {
			limit = { NOT = { has_country_flag = africa_external_continent_unifier_proofs_ready } }
			africa_clear_dynamic_union_identity_flags = yes
			set_country_flag = africa_dynamic_union_congress_of_continents
			set_cosmetic_tag = AFR_CONGRESS_OF_CONTINENTS
			add_to_variable = { africa_authority = constant:africa_value_delta.external_unifier_proof_ready_authority }
			add_to_variable = { africa_league_cohesion = constant:africa_value_delta.external_unifier_proof_ready_cohesion }
			add_to_variable = { africa_colonial_alarm = constant:africa_value_delta.external_unifier_proof_alarm }
			africa_clamp_core_values = yes
		}
		set_country_flag = africa_external_continent_unifier_proofs_ready
		set_global_flag = africa_external_continent_unifier_proofs_ready
	}
	else = {
		clr_country_flag = africa_external_continent_unifier_proofs_ready
		clr_global_flag = africa_external_continent_unifier_proofs_ready
	}
}
