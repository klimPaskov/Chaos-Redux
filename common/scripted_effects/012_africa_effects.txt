# Overview:
# - Resolves one bounded weighted host only when Event 12 is explicitly fired.

#   government, leaders, territory, armies, and public country identity.

# - Adds no daily, weekly, monthly, or other recurring world scan.

# ---------------------------------------------------------------------------
# Fire-once host selection
# ---------------------------------------------------------------------------

africa_clear_prefire_context = {
	set_temp_variable = { africa_prefire_ready = constant:africa_value.zero }
	set_temp_variable = { africa_host_selection_pool_count = constant:africa_value.zero }
	set_temp_variable = { africa_host_selection_weight = constant:africa_value.zero }
	set_temp_variable = { africa_host_pool_weight_remaining = constant:africa_value.zero }
	clear_temp_array = africa_host_selection_pool
}

# COUNTRY scope. Produces an integer host weight from every accepted design
# factor. There is deliberately no human-player bonus.
africa_prepare_host_selection_weight = {
	set_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.base_weight }
	if = {
		limit = { is_subject = no }
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.independent_bonus }
	}
	else_if = {
		limit = {
			is_subject = yes
			compare_autonomy_state > autonomy_colony
			capital_scope = { is_controlled_by = ROOT }
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.high_autonomy_subject_bonus }
	}
	if = {
		limit = {
			has_war = yes
			any_enemy_country = {
				capital_scope = { NOT = { is_on_continent = africa } }
				any_owned_state = { is_on_continent = africa }
			}
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.colonial_war_bonus }
	}
	if = {
		limit = {
			OR = {
				any_guaranteed_country = {
					capital_scope = { is_on_continent = africa }
				}
				any_country = {
					capital_scope = { is_on_continent = africa }
					has_volunteers_amount_from = {
						tag = ROOT
						count > constant:africa_value.zero
					}
				}
			}
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.african_support_bonus }
	}
	if = {
		limit = {
			is_major = no
			num_of_factories > constant:africa_host_selection.medium_factory_minimum
			num_of_factories < constant:africa_host_selection.medium_factory_maximum
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.medium_power_bonus }
	}
	if = {
		limit = { is_major = yes }
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.major_penalty }
	}
	if = {
		limit = {
			capital_scope = { is_island_state = yes }
			num_of_factories < constant:africa_host_selection.tiny_factory_maximum
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.tiny_island_penalty }
	}
	if = {
		limit = { surrender_progress > constant:africa_host_selection.near_capitulation_threshold }
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.near_capitulation_penalty }
	}
	set_temp_variable = { africa_host_african_neighbor_count = constant:africa_value.zero }
	every_neighbor_country = {
		limit = {
			capital_scope = { is_on_continent = africa }
		}
		add_to_temp_variable = { africa_host_african_neighbor_count = constant:africa_value.one }
	}
	if = {
		limit = {
			check_variable = {
				var = africa_host_african_neighbor_count
				value = constant:africa_host_selection.several_neighbors_minimum
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.several_neighbors_bonus }
	}
	if = {
		limit = {
			capital_scope = { is_island_state = yes }
			check_variable = { africa_host_african_neighbor_count = constant:africa_value.zero }
		}
		add_to_temp_variable = { africa_host_selection_weight = constant:africa_host_selection.overseas_contact_penalty }
	}
	round_temp_variable = africa_host_selection_weight
	clamp_temp_variable = {
		var = africa_host_selection_weight
		min = constant:africa_host_selection.minimum_weight
		max = constant:africa_host_selection.maximum_weight
	}
	set_temp_variable = { africa_host_african_neighbor_count = constant:africa_value.zero }
}

# COUNTRY scope. Duplicated scope entries are the engine-supported weighted
# array pattern already used by Fury and Event 16.
africa_add_current_country_to_host_selection_pool = {
	africa_prepare_host_selection_weight = yes
	set_temp_variable = { africa_host_pool_weight_remaining = africa_host_selection_weight }
	while_loop_effect = {
		limit = { check_variable = { africa_host_pool_weight_remaining > constant:africa_value.zero } }
		add_to_temp_array = { africa_host_selection_pool = THIS }
		add_to_temp_variable = { africa_host_selection_pool_count = constant:africa_value.one }
		subtract_from_temp_variable = { africa_host_pool_weight_remaining = constant:africa_value.one }
	}
	set_temp_variable = { africa_host_pool_weight_remaining = constant:africa_value.zero }
	set_temp_variable = { africa_host_selection_weight = constant:africa_value.zero }
}

# Any scope. This explicit fire-once scan is never called from a periodic hook.
africa_select_weighted_prefire_host = {
	clear_temp_array = africa_host_selection_pool
	set_temp_variable = { africa_host_selection_pool_count = constant:africa_value.zero }
	every_country = {
		limit = { africa_is_eligible_host = yes }
		africa_add_current_country_to_host_selection_pool = yes
	}
	if = {
		limit = { check_variable = { africa_host_selection_pool_count > constant:africa_value.zero } }
		random_scope_in_array = {
			array = africa_host_selection_pool
			save_event_target_as = africa_prefire_host
		}
	}
	clear_temp_array = africa_host_selection_pool
	set_temp_variable = { africa_host_selection_pool_count = constant:africa_value.zero }
	set_temp_variable = { africa_host_pool_weight_remaining = constant:africa_value.zero }
	set_temp_variable = { africa_host_selection_weight = constant:africa_value.zero }
}

# Any scope. Output is the regular africa_prefire_host target that survives into
# the entry event fired from the same effect chain.
africa_prepare_random_event_fire = {
	africa_clear_prefire_context = yes
	if = {
		limit = { africa_automatic_event_is_available = yes }
		africa_select_weighted_prefire_host = yes
		if = {
			limit = {
				has_event_target = africa_prefire_host
				event_target:africa_prefire_host = { africa_is_eligible_host = yes }
			}
			set_temp_variable = { africa_prefire_ready = constant:africa_value.one }
		}
	}
}

# COUNTRY scope. Weak hosts receive emergency capacity and narrower early work;
# strong hosts receive less free help and begin under greater suspicion.
africa_classify_host_support = {
	set_variable = { africa_host_support_class = constant:africa_host_support.regional }
	if = {
		limit = {
			OR = {
				num_of_factories < constant:africa_host_selection.weak_factory_maximum
				num_of_controlled_states < constant:africa_host_selection.weak_state_maximum
			}
		}
		set_variable = { africa_host_support_class = constant:africa_host_support.weak }

	}

# COUNTRY scope on the host. Input is the regular africa_contact_to_register
# event target. The helper is idempotent and never exceeds the roster cap.
africa_register_contact_target = {
	set_temp_variable = { africa_contact_registration_result = constant:africa_value.zero }
	if = {
		limit = {
			has_event_target = africa_contact_to_register
			event_target:africa_contact_to_register = { africa_is_valid_relationship_target = yes }
			OR = {
				is_in_array = { array = africa_selected_targets value = event_target:africa_contact_to_register }
				africa_has_selected_target_capacity = yes
			}
		}
		if = {
			limit = { NOT = { is_in_array = { array = africa_selected_targets value = event_target:africa_contact_to_register } } }
			add_to_array = { array = africa_selected_targets value = event_target:africa_contact_to_register }
		}
		event_target:africa_contact_to_register = { set_country_flag = africa_selected_roster_target }
		africa_clear_selected_country_target = yes
		set_variable = { africa_selected_country_id = event_target:africa_contact_to_register.id }
		add_to_array = { array = africa_selected_country_targets value = event_target:africa_contact_to_register }
		event_target:africa_contact_to_register = { set_country_flag = africa_selected_country_target }
		set_temp_variable = { africa_contact_registration_result = constant:africa_value.one }
	}
}

	}

# COUNTRY scope on a member. Applies the accepted confidence consequence and
# records lifetime evidence after a successful relationship transition.
africa_apply_relationship_transition_consequence = {
	if = { limit = { africa_relationship_is_outside = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.outside_delta } }
	else_if = { limit = { africa_relationship_is_protected = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.protected_delta } set_country_flag = africa_lifetime_was_protected }
	else_if = { limit = { africa_relationship_is_associate = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.associate_delta } set_country_flag = africa_lifetime_was_associate }
	else_if = { limit = { africa_relationship_is_chartered = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.chartered_delta } set_country_flag = africa_lifetime_was_chartered }
	else_if = { limit = { africa_relationship_is_autonomous_federal = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.autonomous_federal_delta } set_country_flag = africa_lifetime_was_autonomous_federal }
	else_if = { limit = { africa_relationship_is_integrated_region = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.integrated_region_delta } set_country_flag = africa_lifetime_was_integrated_region }
	else_if = { limit = { africa_relationship_is_resistant = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.resistant_delta } set_country_flag = africa_lifetime_was_resistant }
	else_if = { limit = { africa_relationship_is_leaving = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.leaving_delta } set_country_flag = africa_lifetime_began_withdrawal }
	else_if = { limit = { africa_relationship_is_rival_bloc = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.rival_bloc_delta } set_country_flag = africa_lifetime_joined_rival_bloc }
	else_if = { limit = { africa_relationship_is_occupied_settlement = yes } add_to_variable = { africa_member_confidence = constant:africa_relationship_confidence.occupied_settlement_delta } set_country_flag = africa_lifetime_received_occupied_settlement }
	clamp_variable = { var = africa_member_confidence min = constant:africa_measure.minimum max = constant:africa_measure.maximum }
}

# COUNTRY scope on the proposed member. Input is the temporary
# africa_requested_relationship_state value set by the calling action/event.
# Output africa_relationship_transition_result is one only after the full
# unregister/mutate/register/reconcile transaction succeeds.
africa_cancel_incompatible_action_before_relationship_transition = {
	if = {
		limit = {
			has_country_flag = africa_action_record_active
			check_variable = { africa_active_action_status = constant:africa_action_status.active }
			OR = {
				check_variable = { africa_requested_relationship_state = constant:africa_relationship.outside }
				check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant }
				check_variable = { africa_requested_relationship_state = constant:africa_relationship.leaving }
				check_variable = { africa_requested_relationship_state = constant:africa_relationship.rival_bloc }
				check_variable = { africa_requested_relationship_state = constant:africa_relationship.occupied_settlement }
				check_variable = { africa_active_action_family = constant:africa_action_family.accession }
				check_variable = { africa_active_action_family = constant:africa_action_family.integration }
			}
		}
		africa_cancel_action = yes
	}
}

africa_apply_relationship_transition = {
	set_temp_variable = { africa_relationship_transition_result = constant:africa_value.zero }
	set_temp_variable = { africa_relationship_transition_allowed = constant:africa_value.zero }
	set_temp_variable = { africa_transition_host_generation = constant:africa_value.invalid }
	if = {
		limit = {
			has_event_target = africa_host
			event_target:africa_host = {
				exists = yes
				has_country_flag = africa_host_commit_completed
			}
		}
		event_target:africa_host = {
			set_temp_variable = { africa_transition_host_generation = africa_host_generation }
		}
	}
	if = {
		limit = {
			has_event_target = africa_host
			event_target:africa_host = {
				exists = yes
				has_country_flag = africa_host_commit_completed
			}
			africa_is_valid_relationship_target = yes
			OR = {
				NOT = { has_variable = africa_member_host_generation }
				check_variable = { var = africa_member_host_generation value = africa_transition_host_generation compare = equals }
			}
			OR = {
				NOT = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.integrated_region } }
				africa_member_can_begin_integration = yes
			}
			OR = {
				NOT = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal } }
				africa_member_can_enter_autonomous_federal = yes
			}
			OR = {
				AND = { africa_relationship_is_outside = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.protected } check_variable = { africa_requested_relationship_state = constant:africa_relationship.associate } check_variable = { africa_requested_relationship_state = constant:africa_relationship.rival_bloc } check_variable = { africa_requested_relationship_state = constant:africa_relationship.occupied_settlement } } }
				AND = { africa_relationship_is_protected = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.associate } check_variable = { africa_requested_relationship_state = constant:africa_relationship.outside } check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant } } }
				AND = { africa_relationship_is_associate = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered } check_variable = { africa_requested_relationship_state = constant:africa_relationship.outside } check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant } } }
				AND = { africa_relationship_is_chartered = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal } check_variable = { africa_requested_relationship_state = constant:africa_relationship.integrated_region } check_variable = { africa_requested_relationship_state = constant:africa_relationship.leaving } check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant } } }
				AND = { africa_relationship_is_autonomous_federal = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.integrated_region } check_variable = { africa_requested_relationship_state = constant:africa_relationship.leaving } check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant } } }
				AND = { africa_relationship_is_integrated_region = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal } check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant } } }
				AND = { africa_relationship_is_resistant = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered } check_variable = { africa_requested_relationship_state = constant:africa_relationship.leaving } check_variable = { africa_requested_relationship_state = constant:africa_relationship.rival_bloc } check_variable = { africa_requested_relationship_state = constant:africa_relationship.occupied_settlement } } }
				AND = { africa_relationship_is_leaving = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.outside } check_variable = { africa_requested_relationship_state = constant:africa_relationship.associate } check_variable = { africa_requested_relationship_state = constant:africa_relationship.rival_bloc } check_variable = { africa_requested_relationship_state = constant:africa_relationship.occupied_settlement } } }
				AND = { africa_relationship_is_rival_bloc = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.associate } check_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal } check_variable = { africa_requested_relationship_state = constant:africa_relationship.outside } check_variable = { africa_requested_relationship_state = constant:africa_relationship.occupied_settlement } } }
				AND = { africa_relationship_is_occupied_settlement = yes OR = { check_variable = { africa_requested_relationship_state = constant:africa_relationship.integrated_region } check_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal } check_variable = { africa_requested_relationship_state = constant:africa_relationship.resistant } check_variable = { africa_requested_relationship_state = constant:africa_relationship.outside } } }
			}
		}
		set_temp_variable = { africa_relationship_transition_allowed = constant:africa_value.one }
	}
	if = {
		limit = { check_variable = { africa_relationship_transition_allowed > constant:africa_value.zero } }
		set_temp_variable = { africa_relationship_previous_state = constant:africa_relationship.outside }
		if = { limit = { has_variable = africa_relationship_state } set_temp_variable = { africa_relationship_previous_state = africa_relationship_state } }
		africa_cancel_incompatible_action_before_relationship_transition = yes
		africa_unregister_current_relationship_state = yes
		set_variable = { africa_relationship_state = africa_requested_relationship_state }
		set_variable = { africa_member_host_generation = africa_transition_host_generation }
		if = { limit = { NOT = { has_variable = africa_member_confidence } } set_variable = { africa_member_confidence = constant:africa_measure.medium } }
		africa_apply_relationship_transition_consequence = yes
		africa_register_current_relationship_state = yes
		event_target:africa_host = { africa_reconcile_relationship_counts_and_caps = yes }
		set_country_flag = africa_relationship_transition_pending_notice
		set_temp_variable = { africa_relationship_transition_result = constant:africa_value.one }
	}
	set_temp_variable = { africa_relationship_transition_allowed = constant:africa_value.zero }
}
