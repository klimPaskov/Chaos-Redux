africa_score_unifier_candidate_for_selection = {
	set_temp_variable = { africa_unifier_selection_weight = 0 }
	if = {
		limit = { is_africa_valid_unifier_candidate = yes }
		set_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.base }
		if = {
			limit = { is_africa_independent_unifier_candidate = yes }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.independent_african_capital }
		}
		if = {
			limit = { is_africa_subject_liberation_unifier_candidate = yes }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.subject_liberation_story }
		}
		if = {
			limit = {
				has_war = yes
				any_enemy_country = {
					NOT = { capital_scope = { is_on_continent = africa } }
				}
			}
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.at_war_with_foreign_holder }
		}
		if = {
			limit = { has_country_flag = independence_wave_breakaway }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.recent_chaos_liberation_state }
		}
		if = {
			limit = { is_africa_rsa_allies_unifier_candidate = yes }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.rsa_allies_branch }
		}
		if = {
			limit = { num_of_controlled_states < 3 }
			add_to_temp_variable = { africa_unifier_selection_weight = constant:africa_selection_weight.small_country_bonus }
		}
	}
}

africa_select_random_unifier_candidate = {
	set_variable = { global.africa_runtime_ready = constant:africa_runtime_state.not_ready }
	if = {
		limit = { has_africa_valid_unifier_candidate = yes }
		random_country = {
			limit = { is_africa_valid_unifier_candidate = yes }
			africa_prepare_runtime_context_from_this = yes
		}
	}
}

africa_prepare_runtime_context_from_this = {
	save_global_event_target_as = africa_unifier_country
	set_variable = { global.africa_unifier_country = ROOT }
	set_variable = { global.africa_unifier_country_id = ROOT.id }
	set_variable = { global.africa_runtime_ready = constant:africa_runtime_state.ready }
	set_global_flag = africa_runtime_context_ready
	set_global_flag = event_012_africa_fired
	set_country_flag = africa_unifier_active
	africa_initialize_core_values = yes
}

africa_clear_runtime_context = {
	if = {
		limit = { has_event_target = africa_unifier_country }
		event_target:africa_unifier_country = {
			clr_country_flag = africa_unifier_active
			clr_country_flag = africa_charter_league_leader
			clr_country_flag = africa_rsa_continental_side
			clr_country_flag = africa_rsa_loyalist_side
		}
		clear_global_event_target = africa_unifier_country
	}
	if = {
		limit = { has_event_target = africa_rsa_continental_side }
		clear_global_event_target = africa_rsa_continental_side
	}
	if = {
		limit = { has_event_target = africa_rsa_loyalist_side }
		clear_global_event_target = africa_rsa_loyalist_side
	}
	if = {
		limit = { has_event_target = africa_rsa_union_origin }
		event_target:africa_rsa_union_origin = {
			clr_country_flag = africa_rsa_union_origin
		}
		clear_global_event_target = africa_rsa_union_origin
	}
	if = {
		limit = { has_event_target = africa_charter_league_leader }
		event_target:africa_charter_league_leader = {
			clr_country_flag = africa_charter_league_leader
		}
		clear_global_event_target = africa_charter_league_leader
	}
	clear_variable = global.africa_unifier_country
	clear_variable = global.africa_unifier_country_id
	clear_variable = global.africa_rsa_branch_state
	clear_variable = global.africa_world_is_one_gate_state
	set_variable = { global.africa_runtime_ready = constant:africa_runtime_state.not_ready }
	clr_global_flag = africa_runtime_context_ready
	clr_global_flag = africa_rsa_civil_war_active
}

africa_grant_continental_paper_claims = {
	every_state = {
		limit = { is_on_continent = africa }
		if = {
			limit = { NOT = { is_claimed_by = ROOT } }
			add_claim_by = ROOT
		}
		set_state_flag = africa_paper_claim
		set_variable = { africa_paper_core_stage = constant:africa_paper_core_stage.claim }
	}
	set_country_flag = africa_continental_paper_claims_seeded
}

africa_unlock_state_integration_from_from = {
	FROM = {
		if = {
			limit = {
				is_on_continent = africa
				has_state_flag = africa_paper_claim
			}
			set_state_flag = africa_integration_unlocked
			set_variable = { africa_paper_core_stage = constant:africa_paper_core_stage.integration_unlocked }
		}
	}
}

	if = {
		limit = {
			has_event_target = africa_rsa_continental_side
			ENG = { exists = yes }
		}
		ENG = {
			every_country = {
				limit = {
					exists = yes
					is_in_faction_with = ENG
					has_war_with = event_target:africa_rsa_continental_side
				}
				event_target:africa_rsa_continental_side = {
					white_peace = PREV
				}
			}
		}
		set_global_flag = africa_rsa_allied_peace_done
		set_variable = { global.africa_rsa_branch_state = constant:africa_rsa_branch.allied_peace_done }
	}
}
