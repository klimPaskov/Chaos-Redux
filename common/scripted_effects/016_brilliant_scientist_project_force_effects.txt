# Country-scope reconstruction and one-time formation dispatch for the military
# outputs carried by Warren Kruger's exact project history. The runtime rebuild

# COUNTRY scope. Existing formations survive a project shutdown, but every
# matching template immediately loses both queue access and its division cap.
# The runtime rebuild selectively restores only operational project families.
brilliant_scientist_revoke_project_force_template_authorizations = {
	set_temp_variable = { brilliant_scientist_project_force_zero_division_cap = constant:brilliant_scientist_value.zero }

# COUNTRY scope. The agent flags are the exact payload ledger. No biological
# project, agent, or stockpile is fabricated by this reconstruction.
brilliant_scientist_apply_biological_force_history = {
	if = {
		limit = { brilliant_scientist_has_biological_force_history_prototype = yes }
		set_country_flag = brilliant_scientist_project_force_biological_weapons_active
	}

	if = {
		limit = { brilliant_scientist_has_biological_force_history_deployment = yes }
		if = { limit = { NOT = { has_country_flag = bioweapon_available } } set_country_flag = bioweapon_available }
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_anthrax } }
			set_technology = { anthrax_bomb_delivery_systems = 1 popup = no }
		}
		if = {
			limit = {
				KRG_warren_kruger = {
					OR = {
						has_character_flag = brilliant_scientist_personal_biological_agent_plague
						has_character_flag = brilliant_scientist_personal_biological_agent_black_plague
					}
				}
			}
			set_technology = { plague_bomb_delivery_systems = 1 popup = no }
		}
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_tularemia } }
			set_technology = { tularemia_bomb_delivery_systems = 1 popup = no }
		}
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_smallpox } }
			set_technology = { smallpox_bomb_delivery_systems = 1 popup = no }
		}
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_weaponized_zombies } }
			set_technology = { zombie_disease_bomb_delivery_systems = 1 popup = no }
		}
	}

	if = {
		limit = { brilliant_scientist_has_biological_force_history_weaponization = yes }
		set_country_flag = brilliant_scientist_project_force_biological_weapons_weaponized
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_anthrax } }
			add_ideas = brilliant_scientist_biological_delivery_anthrax
		}
		if = {
			limit = {
				KRG_warren_kruger = {
					OR = {
						has_character_flag = brilliant_scientist_personal_biological_agent_plague
						has_character_flag = brilliant_scientist_personal_biological_agent_black_plague
					}
				}
			}
			add_ideas = brilliant_scientist_biological_delivery_plague
		}
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_tularemia } }
			add_ideas = brilliant_scientist_biological_delivery_tularemia
		}
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_smallpox } }
			add_ideas = brilliant_scientist_biological_delivery_smallpox
		}
		if = {
			limit = { KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_weaponized_zombies } }
			add_ideas = brilliant_scientist_biological_delivery_weaponized_zombies
		}
	}
