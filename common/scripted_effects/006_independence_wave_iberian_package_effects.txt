independence_wave_refresh_nav_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_nav_package = yes }
		remove_ideas = { nav_contested_fueros nav_mountain_industrial_compact }
		if = {
			limit = { has_stable_independence_wave_nav_compact = yes }
			add_ideas = nav_mountain_industrial_compact
			set_country_flag = independence_wave_nav_compact_stabilized
		}
		else = {
			add_ideas = nav_contested_fueros
			clr_country_flag = independence_wave_nav_compact_stabilized
		}
	}
}
