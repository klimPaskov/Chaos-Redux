# ============================================================================
# CHAOS REDUX - EVENT 006 FORCE PACKAGE RUNTIME DISPATCHER
# ============================================================================
# Overview:
# - Probes p1 through p206 through one dynamic meta-effect lookup.
# - Decodes exact reinforcement and inheritance masks with structural constants.
# - Exposes a temporary, non-mutating readiness contract for pre-transfer checks.
# - Loads profile, military tradition, accepted path flags, inheritance flags,
#   and research sensitivity into a country only after the probe validates.
# - Never marks a command roster ready and never materializes starting forces.
# ============================================================================

# Clears only the force mapping state owned by this dispatcher.
independence_wave_clear_force_package_mapping_runtime = {
	clr_country_flag = independence_wave_force_mapping_loaded

	independence_wave_clear_reinforcement_pathways = yes
}

# Non-mutating pre-transfer probe.
# Input: temporary independence_wave_setup_package_id.
# Outputs are temporary independence_wave_force_package_probe_* values.
independence_wave_probe_force_package_mapping = {
	set_temp_variable = { independence_wave_force_package_probe_ready = constant:independence_wave_force_package_runtime.false_value }
	set_temp_variable = { independence_wave_force_package_probe_package_id = independence_wave_setup_package_id }
	round_temp_variable = independence_wave_force_package_probe_package_id
	set_temp_variable = { independence_wave_force_package_probe_profile = constant:independence_wave_force_package_runtime.false_value }
	set_temp_variable = { independence_wave_force_package_probe_military_tradition = constant:independence_wave_force_package_runtime.false_value }
	set_temp_variable = { independence_wave_force_package_probe_reinforcement_mask = constant:independence_wave_force_package_runtime.false_value }
	set_temp_variable = { independence_wave_force_package_probe_inheritance_mask = constant:independence_wave_force_package_runtime.false_value }
	set_temp_variable = { independence_wave_force_package_probe_research_sensitive = constant:independence_wave_force_package_runtime.false_value }

	if = {
		limit = { has_valid_independence_wave_force_package_probe_id = yes }
		meta_effect = {
			text = {
				set_temp_variable = {
					independence_wave_force_package_probe_profile = constant:independence_wave_force_package_profile.p[PACKAGE_ID]
				}
				set_temp_variable = {
					independence_wave_force_package_probe_military_tradition = constant:independence_wave_force_package_military_tradition.p[PACKAGE_ID]
				}
				set_temp_variable = {
					independence_wave_force_package_probe_reinforcement_mask = constant:independence_wave_force_package_reinforcement_mask.p[PACKAGE_ID]
				}
				set_temp_variable = {
					independence_wave_force_package_probe_inheritance_mask = constant:independence_wave_force_package_inheritance_mask.p[PACKAGE_ID]
				}
				set_temp_variable = {
					independence_wave_force_package_probe_research_sensitive = constant:independence_wave_force_package_research_sensitive.p[PACKAGE_ID]
				}
			}
			PACKAGE_ID = "[?independence_wave_force_package_probe_package_id|.0]"
		}
	}

	independence_wave_decode_force_package_probe_masks = yes
	if = {
		limit = { has_valid_independence_wave_force_package_mapping_probe = yes }
		set_temp_variable = { independence_wave_force_package_probe_ready = constant:independence_wave_force_package_runtime.true_value }
	}
}

# Applies only the five path bits validated by the probe. Each call uses the
# existing idempotent pathway helper, which owns its flag and count update.
independence_wave_apply_force_package_probe_paths = {
	if = {
		limit = { check_variable = { independence_wave_force_package_probe_path_integrate_militias = constant:independence_wave_force_package_runtime.true_value } }
		independence_wave_add_reinforce_integrate_militias = yes
	}

		independence_wave_add_reinforce_regional_guards = yes
	}
	if = {
		limit = { check_variable = { independence_wave_force_package_probe_path_secure_depots = constant:independence_wave_force_package_runtime.true_value } }
		independence_wave_add_reinforce_secure_depots = yes

# Public mutating loader.
# Call on an initialized active Event 006 country whose persistent package ID
# already matches temporary independence_wave_setup_package_id in the 1..206 set.
independence_wave_load_force_package_mapping = {
	independence_wave_clear_force_package_mapping_runtime = yes
	independence_wave_probe_force_package_mapping = yes

	if = {
		limit = { has_ready_independence_wave_force_package_mapping_probe = yes }
		set_variable = {
			independence_wave_force_mapping_revision = constant:independence_wave_force_package_runtime.mapping_revision
		}
		set_variable = { independence_wave_force_mapping_package_id = independence_wave_force_package_probe_package_id }
		set_variable = { independence_wave_force_profile = independence_wave_force_package_probe_profile }
		set_variable = { independence_wave_military_tradition = independence_wave_force_package_probe_military_tradition }
		independence_wave_apply_force_package_probe_paths = yes

		if = {
			limit = { check_variable = { independence_wave_force_package_probe_inherit_navy = constant:independence_wave_force_package_runtime.true_value } }
			set_country_flag = independence_wave_force_navy_inheritance_enabled
		}
		if = {
			limit = { check_variable = { independence_wave_force_package_probe_inherit_air = constant:independence_wave_force_package_runtime.true_value } }
			set_country_flag = independence_wave_force_air_inheritance_enabled
		}
		if = {
			limit = { check_variable = { independence_wave_force_package_probe_research_sensitive = constant:independence_wave_force_package_runtime.true_value } }
			set_country_flag = independence_wave_force_mapping_research_sensitive
		}

		if = {
			limit = { has_valid_independence_wave_loaded_force_package_mapping = yes }
			set_country_flag = independence_wave_force_mapping_loaded
		}
		else = {
			independence_wave_clear_force_package_mapping_runtime = yes
		}
	}
}
