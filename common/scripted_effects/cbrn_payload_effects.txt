# ============================================================================
# CHAOS REDUX - CBRN PAYLOAD EFFECTS
# ============================================================================
# Overview:
# - Owns selected shell and air profiles, strategic-payload conversion,
#   route-specific payload requirements, exact stock debits, and one-time
#   legacy stock migration.
# - Country scope is required. Action helpers consume the temporary action
#   record created by cbrn_reset_action_context.
# - cbrn_try_debit_action_payload supplies the payload proof only after the
#   exact equipment removal succeeds. Callers must treat a missing proof as a
#   rejected operation and must not dispatch exposure.
# - Profile changes impose a real delay and discard a bounded part of prepared
#   stock. They never alter evidence, attribution, casualties, contamination,
#   medical load, Condemnation history, or confirmed-use records.
# ============================================================================

cbrn_initialize_payload_logistics = {
	if = {
		limit = { NOT = { has_variable = cbrn_shell_filling_agent } }
		set_variable = { cbrn_shell_filling_agent = constant:cbrn_agent.none }
	}
	if = {
		limit = { NOT = { has_variable = cbrn_air_payload_agent } }
		set_variable = { cbrn_air_payload_agent = constant:cbrn_agent.none }
	}
}

cbrn_try_debit_action_payload = {
	set_temp_variable = { cbrn_action_payload_consumed = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_action_payload_consumed_proof = constant:cbrn_proof.missing }
	set_temp_variable = { cbrn_payload_debit_model_matched = constant:cbrn_proof.missing }

	if = {
		limit = {
			cbrn_chemical_action_static_metadata_is_valid = yes
			cbrn_action_agent_is_unlocked = yes
			cbrn_action_payload_profile_is_ready = yes
			cbrn_action_payload_stock_is_sufficient = yes
		}
		if = {
			limit = { cbrn_action_uses_strategic_agent_lots = yes }
			cbrn_debit_strategic_agent_lots_internal = yes
		}
		else_if = {
			limit = { cbrn_action_uses_shell_lots = yes }
			cbrn_debit_shell_lots_internal = yes
		}
		else_if = {
			limit = { cbrn_action_uses_air_payload_lots = yes }
			cbrn_debit_air_payload_lots_internal = yes
		}

		if = {
			limit = { check_variable = { var = cbrn_payload_debit_model_matched value = constant:cbrn_proof.supplied compare = equals } }
			set_temp_variable = { cbrn_action_payload_consumed = cbrn_action_payload_required }
			set_temp_variable = { cbrn_action_payload_consumed_proof = constant:cbrn_proof.supplied }
		}
	}
}

cbrn_remove_shell_profile_wastage_internal = {
	set_temp_variable = { cbrn_payload_profile_available = num_equipment@chemical_shell_lot_1 }
	set_temp_variable = { cbrn_payload_profile_retained = cbrn_payload_profile_available }
	multiply_temp_variable = { cbrn_payload_profile_retained = constant:cbrn_payload_conversion.shell_profile_switch_recovery }
	set_temp_variable = { cbrn_payload_profile_loss = cbrn_payload_profile_available }
	subtract_from_temp_variable = { cbrn_payload_profile_loss = cbrn_payload_profile_retained }
	round_temp_variable = cbrn_payload_profile_loss
	if = {
		limit = { check_variable = { var = cbrn_payload_profile_loss value = constant:cbrn_math.zero compare = greater_than } }
		multiply_temp_variable = { cbrn_payload_profile_loss = constant:cbrn_payload_math.negation_multiplier }
		add_equipment_to_stockpile = { type = chemical_shell_lot_1 amount = cbrn_payload_profile_loss }
	}
}

cbrn_remove_air_profile_wastage_internal = {
	set_temp_variable = { cbrn_payload_profile_available = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_payload_profile_old_class = constant:cbrn_agent_class.none }

	if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_air_payload_agent value = constant:cbrn_agent.chlorine compare = equals }
				check_variable = { var = cbrn_air_payload_agent value = constant:cbrn_agent.phosgene compare = equals }
			}
		}
		set_temp_variable = { cbrn_payload_profile_old_class = constant:cbrn_agent_class.choking }
		set_temp_variable = { cbrn_payload_profile_available = num_equipment@choking_chemical_air_payload_lot_1 }

				check_variable = { var = cbrn_air_payload_agent value = constant:cbrn_agent.lewisite compare = equals }

	round_temp_variable = cbrn_payload_profile_loss

	if = {
		limit = { check_variable = { var = cbrn_payload_profile_loss value = constant:cbrn_math.zero compare = greater_than } }
		multiply_temp_variable = { cbrn_payload_profile_loss = constant:cbrn_payload_math.negation_multiplier }
		if = {
			limit = { check_variable = { var = cbrn_payload_profile_old_class value = constant:cbrn_agent_class.choking compare = equals } }
			add_equipment_to_stockpile = { type = choking_chemical_air_payload_lot_1 amount = cbrn_payload_profile_loss }

			limit = { check_variable = { var = cbrn_payload_profile_old_class value = constant:cbrn_agent_class.blister compare = equals } }
			add_equipment_to_stockpile = { type = blister_chemical_air_payload_lot_1 amount = cbrn_payload_profile_loss }

cbrn_change_shell_filling_profile = {
	cbrn_initialize_payload_logistics = yes
	set_temp_variable = { cbrn_payload_profile_change_accepted = constant:cbrn_proof.missing }
	if = {
		limit = {
			NOT = { has_country_flag = cbrn_shell_profile_change_in_progress }
			cbrn_requested_payload_agent_is_unlocked = yes
			check_variable = { var = cbrn_shell_filling_agent value = cbrn_requested_payload_agent compare = not_equals }
		}
		if = {
			limit = { check_variable = { var = cbrn_shell_filling_agent value = constant:cbrn_agent.none compare = not_equals } }
			cbrn_remove_shell_profile_wastage_internal = yes
		}
		set_variable = { cbrn_shell_filling_agent = cbrn_requested_payload_agent }
		set_temp_variable = { cbrn_payload_profile_change_days = constant:cbrn_payload_profile_timing.shell_change_days }
		set_country_flag = { flag = cbrn_shell_profile_change_in_progress days = cbrn_payload_profile_change_days }
		set_temp_variable = { cbrn_payload_profile_change_accepted = constant:cbrn_proof.supplied }
	}
}
