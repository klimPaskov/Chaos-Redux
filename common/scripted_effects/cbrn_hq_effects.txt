# - Calculates readiness-adjusted preparation and exact affected-force bands.

}

cbrn_hq_calculate_preparation_days = {
	set_temp_variable = { cbrn_hq_preparation_days = cbrn_hq_preparation_base }
	if = {
		limit = {
			OWNER = {
				check_variable = {
					var = chemical_readiness
					value = constant:cbrn_hq_readiness.limited_threshold
					compare = less_than
				}
			}
		}
		multiply_temp_variable = { cbrn_hq_preparation_days = constant:cbrn_hq_readiness.limited_preparation_mult }
	}
	else_if = {
		limit = {
			OWNER = {
				check_variable = {
					var = chemical_readiness
					value = constant:cbrn_hq_readiness.integrated_threshold
					compare = less_than
				}
			}
		}
		multiply_temp_variable = { cbrn_hq_preparation_days = constant:cbrn_hq_readiness.standard_preparation_mult }
	}
	else_if = {
		limit = {
			OWNER = {
				check_variable = {
					var = chemical_readiness
					value = constant:cbrn_hq_readiness.full_threshold
					compare = less_than
				}
			}
		}
		multiply_temp_variable = { cbrn_hq_preparation_days = constant:cbrn_hq_readiness.integrated_preparation_mult }
	}
	else = {
		multiply_temp_variable = { cbrn_hq_preparation_days = constant:cbrn_hq_readiness.full_preparation_mult }
	}
	clamp_temp_variable = {
		var = cbrn_hq_preparation_days
		min = cbrn_hq_preparation_min
		max = cbrn_hq_preparation_max
	}
	round_temp_variable = cbrn_hq_preparation_days
}

			add_to_variable = { cbrn_hq_mask_consumption_total = cbrn_mask_stock_debit_completed }
			add_to_variable = { cbrn_mask_replacement_demand = cbrn_mask_stock_debit_completed }
		}
		if = {
			limit = { check_variable = { var = cbrn_hq_filter_condition_debit value = constant:cbrn_math.zero compare = greater_than } }
			set_temp_variable = { cbrn_hq_filter_condition_before_debit = cbrn_military_mask_filter_condition }

			set_temp_variable = { cbrn_hq_filter_condition_debit_completed = cbrn_hq_filter_condition_before_debit }
			subtract_from_temp_variable = { cbrn_hq_filter_condition_debit_completed = cbrn_military_mask_filter_condition }

cbrn_hq_schedule_next_upkeep_tick = {
	if = {
		limit = {
			check_variable = {
				var = cbrn_hq_upkeep_ticks_remaining
				value = constant:cbrn_math.zero
				compare = greater_than
			}
		}
		set_temp_variable = { cbrn_hq_next_upkeep_days = constant:cbrn_hq_force.weekly_tick_days }
		unit_leader_event = { id = cbrn_hq.2 days = cbrn_hq_next_upkeep_days }
	}
}

cbrn_hq_complete_upkeep_tick = {
	subtract_from_variable = { cbrn_hq_upkeep_ticks_remaining = constant:cbrn_math.one }
	clamp_variable = { var = cbrn_hq_upkeep_ticks_remaining min = constant:cbrn_math.zero }
	cbrn_hq_schedule_next_upkeep_tick = yes
}

cbrn_hq_fail_upkeep = {
	cbrn_hq_stop_operation_benefits = yes
	clear_variable = cbrn_hq_upkeep_ticks_remaining
}
