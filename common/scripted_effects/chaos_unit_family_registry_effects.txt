# ============================================================================
# CHAOS REDUX - OPT-IN CHAOS UNIT FAMILY REGISTRY EFFECTS
# ============================================================================
# Provider registration inputs are temporary variables named
# chaos_unit_family_register_*. Registration is idempotent by family id.
# Consumers resolve a row, then use the recorded provider id in meta effects.
# ============================================================================

chaos_unit_family_initialize_registry = {
	if = {
		limit = { NOT = { hidden_trigger = { has_global_flag = chaos_unit_family_registry_initialized } } }
		set_global_flag = chaos_unit_family_registry_initialized
		set_variable = { global.chaos_unit_family_registry_revision = constant:chaos_unit_family_registry.contract_version }
	}
}

chaos_unit_family_mark_registry_failure = {
	set_global_flag = chaos_unit_family_registry_invariant_failure
}

chaos_unit_family_find_registration_row = {
	set_temp_variable = { chaos_unit_family_registry_index = constant:chaos_unit_family_registry.not_found }
	set_temp_variable = { chaos_unit_family_registry_scan_index = constant:chaos_unit_family_registry.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { chaos_unit_family_registry_index < constant:chaos_unit_family_registry.first_index }
			check_variable = {
				var = chaos_unit_family_registry_scan_index
				value = global.chaos_unit_family_id_entries^num
				compare = less_than
			}
		}
		if = {
			limit = {
				check_variable = {
					var = global.chaos_unit_family_id_entries^chaos_unit_family_registry_scan_index
					value = chaos_unit_family_register_id
					compare = equals
				}
			}
			set_temp_variable = { chaos_unit_family_registry_index = chaos_unit_family_registry_scan_index }
		}
		add_to_temp_variable = { chaos_unit_family_registry_scan_index = 1 }
	}
}

chaos_unit_family_register_current_provider = {
	chaos_unit_family_initialize_registry = yes
	if = {
		limit = { chaos_unit_family_registry_is_aligned = no }
		chaos_unit_family_mark_registry_failure = yes
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { chaos_unit_family_register_id < constant:chaos_unit_family_registry.minimum_family_id }
				check_variable = { chaos_unit_family_register_provider_id < constant:chaos_unit_family_registry.minimum_provider_id }
				NOT = { check_variable = { chaos_unit_family_register_contract_version = constant:chaos_unit_family_registry.contract_version } }
				NOT = { chaos_unit_family_registration_has_supported_event19_visual_contract = yes }
			}
		}
		chaos_unit_family_mark_registry_failure = yes
	}
	else = {
		chaos_unit_family_find_registration_row = yes
		if = {
			limit = { check_variable = { chaos_unit_family_registry_index < constant:chaos_unit_family_registry.first_index } }
			add_to_array = { array = global.chaos_unit_family_id_entries value = chaos_unit_family_register_id }
			add_to_array = { array = global.chaos_unit_family_provider_id_entries value = chaos_unit_family_register_provider_id }
			add_to_array = { array = global.chaos_unit_family_source_event_id_entries value = chaos_unit_family_register_source_event_id }
			add_to_array = { array = global.chaos_unit_family_availability_mode_entries value = chaos_unit_family_register_availability_mode }
			add_to_array = { array = global.chaos_unit_family_event19_lot_policy_entries value = chaos_unit_family_register_event19_lot_policy }
			add_to_array = { array = global.chaos_unit_family_derivative_profile_entries value = chaos_unit_family_register_derivative_profile }
			add_to_array = { array = global.chaos_unit_family_sustainment_profile_entries value = chaos_unit_family_register_sustainment_profile }
			add_to_array = { array = global.chaos_unit_family_containment_profile_entries value = chaos_unit_family_register_containment_profile }
			add_to_array = { array = global.chaos_unit_family_ai_class_entries value = chaos_unit_family_register_ai_class }
			add_to_array = { array = global.chaos_unit_family_visual_profile_entries value = chaos_unit_family_register_visual_profile }
			add_to_array = { array = global.chaos_unit_family_cleanup_profile_entries value = chaos_unit_family_register_cleanup_profile }
			add_to_array = { array = global.chaos_unit_family_parent_isolation_profile_entries value = chaos_unit_family_register_parent_isolation_profile }
			add_to_array = { array = global.chaos_unit_family_spawn_weight_entries value = chaos_unit_family_register_spawn_weight }
			add_to_array = { array = global.chaos_unit_family_contract_version_entries value = chaos_unit_family_register_contract_version }
		}

			limit = {

					NOT = {
						check_variable = {
							var = global.chaos_unit_family_provider_id_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_provider_id
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_source_event_id_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_source_event_id
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_availability_mode_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_availability_mode
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_event19_lot_policy_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_event19_lot_policy
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_derivative_profile_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_derivative_profile
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_sustainment_profile_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_sustainment_profile
							compare = equals

							value = chaos_unit_family_register_containment_profile
							compare = equals

					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_ai_class_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_ai_class
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_visual_profile_entries^chaos_unit_family_registry_index
							value = chaos_unit_family_register_visual_profile
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.chaos_unit_family_cleanup_profile_entries^chaos_unit_family_registry_index

chaos_unit_family_resolve_requested_id = {
	set_temp_variable = { chaos_unit_family_registry_index = constant:chaos_unit_family_registry.not_found }
	set_temp_variable = { chaos_unit_family_registry_scan_index = constant:chaos_unit_family_registry.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { chaos_unit_family_registry_index < constant:chaos_unit_family_registry.first_index }
			check_variable = {
				var = chaos_unit_family_registry_scan_index
				value = global.chaos_unit_family_id_entries^num
				compare = less_than
			}
		}
		if = {
			limit = {
				check_variable = {
					var = global.chaos_unit_family_id_entries^chaos_unit_family_registry_scan_index
					value = chaos_unit_family_requested_id
					compare = equals
				}
			}
			set_temp_variable = { chaos_unit_family_registry_index = chaos_unit_family_registry_scan_index }
		}
		add_to_temp_variable = { chaos_unit_family_registry_scan_index = 1 }
	}
	if = {
		limit = { check_variable = { chaos_unit_family_registry_index < constant:chaos_unit_family_registry.first_index } }
		chaos_unit_family_mark_registry_failure = yes
	}
}

# ----------------------------------------------------------------------------
# Shared default Event 19 provider hooks
# ----------------------------------------------------------------------------
# Event 19 dispatches each provider hook by name with
# chaos_unit_family_provider_[PROVIDER]_event19_<hook>. Providers whose hook
# behaviour is the ordinary one keep that dispatch name as a one-line wrapper
# around the matching default below, so the dispatch contract is unchanged.
# Providers with their own cost or token logic implement the hook directly.
# Scope and variables are the caller's: the defaults read and write only the
# infantry_spawn_family_* temporary variables of the Event 19 management contract.
# ----------------------------------------------------------------------------

# Default pay: payment succeeds when the requested action (train_or_spawn or
# sustainment) is available for the family. No provider-owned cost is paid.
chaos_unit_family_provider_default_event19_pay_management_action = {
	set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
			check_variable = { infantry_spawn_family_provider_can_spawn > 0 }
		}
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
	else_if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.sustainment }
			check_variable = { infantry_spawn_family_provider_can_sustain > 0 }
		}
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
}

# Default refund: clears the payment receipt. Nothing was debited by the default pay.
chaos_unit_family_provider_default_event19_refund_management_action = {
	if = {
		limit = { check_variable = { infantry_spawn_family_provider_payment_succeeded > 0 } }
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	}
}

# Default equipment token: the standard (non-specialist) equipment token for the current obligation profile.
chaos_unit_family_provider_default_event19_get_equipment_token = {
	set_temp_variable = { infantry_spawn_family_provider_equipment_is_specialist = 0 }
	infantry_spawn_provider_get_standard_equipment_token = yes
}
