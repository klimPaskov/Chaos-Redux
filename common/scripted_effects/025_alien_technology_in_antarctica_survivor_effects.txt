# ============================================================================
# CHAOS REDUX - EVENT 025 SOMETHING SURVIVED (EVOLUTION II)
# ============================================================================
# Overview:
# - chaosx_nr25_pick_survivor_profile chooses one hidden profile for the whole
#   race when Evolution II activates (global.chaosx_nr25_survivor_profile).
#   The profile only weights which evidence appears; no text names it.
# - chaosx_nr25_survivor_pulse is called by each expedition's own pulse while
#   the evolution is in force. An expedition with a camp on the ice rolls a
#   survivor incident through the MTTH entry
#   chaosx_nr25_survivor_incident_chance; the incident kind is drawn from the
#   profile's weights. Incidents stop for everyone once the survivor is
#   contained.
# - Stances are exclusive: avoiding contact makes incidents rarer and keeps
#   people out of reach, guarding every movement halves the damage and keeps
#   everyone alive, attempting capture draws the survivor closer, and offering
#   a controlled path draws it away but can lead it back to the camp.
# - Capture is a single attempt per order with a real chance of failure; a
#   success contains the survivor for the whole race, records the Event 036
#   reaction marker and gives the captor a specimen that speeds containment
#   in the Evolution V aftermath.
# - Turning the trace toward a rival strikes the rival selected on the board
#   (or the leading rival): its outpost or its readiness. The rival may trace
#   it back to the actor.
# - Only the expedition's own country, the global race values and the stored
#   participant registry are read. Nothing scans the world, and nothing runs
#   while the evolution is disabled. The survivor never blocks recovery.
# ============================================================================

# GLOBAL value set from any scope. Chooses the hidden survivor profile once.
chaosx_nr25_pick_survivor_profile = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.chaosx_nr25_survivor_profile }
				check_variable = { var = global.chaosx_nr25_survivor_profile value = constant:chaosx_nr25_survivor_profile.none compare = equals }
			}
		}
		set_temp_variable_to_random = chaosx_nr25_profile_roll
		multiply_temp_variable = { chaosx_nr25_profile_roll = constant:chaosx_nr25_survivor_profile.count }
		set_variable = { global.chaosx_nr25_survivor_profile = constant:chaosx_nr25_survivor_profile.swarm }
		if = {
			limit = { check_variable = { var = chaosx_nr25_profile_roll value = constant:chaosx_nr25_value.one compare = less_than } }
			set_variable = { global.chaosx_nr25_survivor_profile = constant:chaosx_nr25_survivor_profile.biological }
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_profile_roll value = constant:chaosx_nr25_value.two compare = less_than } }
			set_variable = { global.chaosx_nr25_survivor_profile = constant:chaosx_nr25_survivor_profile.repair_machine }
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_profile_roll value = constant:chaosx_nr25_value.three compare = less_than } }
			set_variable = { global.chaosx_nr25_survivor_profile = constant:chaosx_nr25_survivor_profile.crew_presence }
		}
	}
}

# COUNTRY scope. Sets one survivor stance and clears the others. The caller
# sets the stance flag itself after this call.
chaosx_nr25_clear_survivor_stance = {
	clr_country_flag = chaosx_nr25_survivor_avoid_route
	clr_country_flag = chaosx_nr25_survivor_guard_route
	clr_country_flag = chaosx_nr25_survivor_capture_route
	clr_country_flag = chaosx_nr25_survivor_path_route
}

# COUNTRY scope (the expedition, from its pulse while Evolution II is in force).
chaosx_nr25_survivor_pulse = {
	if = {
		limit = { NOT = { has_variable = global.chaosx_nr25_survivor_profile } }
		chaosx_nr25_pick_survivor_profile = yes
	}
	# Stance upkeep for an expedition on the ice.
	if = {
		limit = { chaosx_nr25_survivor_incident_possible = yes }
		if = {
			limit = { has_country_flag = chaosx_nr25_survivor_guard_route }
			subtract_from_variable = { chaosx_nr25_logistics_readiness = constant:chaosx_nr25_survivor.guard_readiness_upkeep }
		}
		if = {
			limit = {
				has_country_flag = chaosx_nr25_survivor_avoid_route
				OR = {
					check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.survey compare = equals }
					check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.final_recovery compare = equals }
				}
			}
			subtract_from_variable = { chaosx_nr25_progress = constant:chaosx_nr25_survivor.avoid_progress_drag }
		}
	}
	if = {
		limit = { chaosx_nr25_survivor_incident_possible = yes }
		set_temp_variable = { chaosx_nr25_survivor_chance = mtth:chaosx_nr25_survivor_incident_chance }
		clamp_temp_variable = {
			var = chaosx_nr25_survivor_chance
			min = constant:chaosx_nr25_value.zero
			max = constant:chaosx_nr25_survivor.max_chance
		}
		set_temp_variable_to_random = chaosx_nr25_survivor_roll
		multiply_temp_variable = { chaosx_nr25_survivor_roll = constant:chaosx_nr25_survivor.percent }
		if = {
			limit = { check_variable = { var = chaosx_nr25_survivor_roll value = chaosx_nr25_survivor_chance compare = less_than } }
			chaosx_nr25_apply_survivor_incident = yes
		}
	}
}

# COUNTRY scope. Avoid Contact: route around the trace. Lower exposure;
# incidents come less often and never take people, but surveys go slower.
chaosx_nr25_survivor_response_avoid = {
	chaosx_nr25_clear_survivor_stance = yes
	subtract_from_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_cost.exposure_survivor_avoid_reduction }
	set_country_flag = chaosx_nr25_survivor_avoid_route
}

# COUNTRY scope. Guard Every Movement: escorts for every field team. Readiness
# now and a small upkeep each pulse; incidents do half the damage and kill no
# one; a later capture attempt is likelier to succeed.
chaosx_nr25_survivor_response_guard = {
	chaosx_nr25_clear_survivor_stance = yes
	add_to_variable = { chaosx_nr25_logistics_readiness = constant:chaosx_nr25_survivor.guard_readiness }
	add_to_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_cost.exposure_survivor_guard }
	set_country_flag = chaosx_nr25_survivor_guard_route
}

# COUNTRY scope. Attempt Capture: one attempt per order. The chance rises with
# Intelligence and with escorts already guarding the teams. A success contains
# the survivor for the whole race; a failure costs a crew and readiness.
chaosx_nr25_survivor_response_capture = {
	set_temp_variable = { chaosx_nr25_capture_chance = chaosx_nr25_intelligence }
	multiply_temp_variable = { chaosx_nr25_capture_chance = constant:chaosx_nr25_survivor.capture_intelligence_factor }
	add_to_temp_variable = { chaosx_nr25_capture_chance = constant:chaosx_nr25_survivor.capture_base_chance }
	if = {
		limit = { has_country_flag = chaosx_nr25_survivor_guard_route }
		add_to_temp_variable = { chaosx_nr25_capture_chance = constant:chaosx_nr25_survivor.capture_guard_bonus }
	}
	clamp_temp_variable = {
		var = chaosx_nr25_capture_chance
		min = constant:chaosx_nr25_value.zero
		max = constant:chaosx_nr25_survivor.capture_max_chance
	}
	chaosx_nr25_clear_survivor_stance = yes
	set_country_flag = chaosx_nr25_survivor_capture_route
	add_to_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_cost.exposure_survivor_capture }
	set_temp_variable_to_random = chaosx_nr25_capture_roll

# COUNTRY scope. Turn It Toward a Rival: drive the trace at the rival selected
# on the board, or otherwise the rival with the most progress. Its outpost
# loses security, or its readiness falls if it has none. The rival traces it
# back to us when its intelligence is good enough, and otherwise on a fixed
# chance; a traced diversion costs us opinion and exposure.
chaosx_nr25_survivor_response_turn = {
	add_to_variable = { chaosx_nr25_intelligence = constant:chaosx_nr25_survivor.turn_intelligence }
	set_temp_variable = { chaosx_nr25_rival_picked = constant:chaosx_nr25_value.zero }
	chaosx_nr25_pick_rival_target = yes
	clr_country_flag = chaosx_nr25_survivor_turn_traced
	if = {
		limit = { check_variable = { var = chaosx_nr25_rival_picked value = constant:chaosx_nr25_value.one compare = equals } }
		set_variable = { chaosx_nr25_last_turn_target = event_target:chaosx_nr25_picked_rival }
		set_temp_variable_to_random = chaosx_nr25_turn_roll
		multiply_temp_variable = { chaosx_nr25_turn_roll = constant:chaosx_nr25_survivor.percent }
		event_target:chaosx_nr25_picked_rival = {
			set_variable = { chaosx_nr25_survivor_readiness_loss = constant:chaosx_nr25_value.zero }
			set_variable = { chaosx_nr25_survivor_security_loss = constant:chaosx_nr25_value.zero }
			if = {
				limit = { has_country_flag = chaosx_nr25_outpost_established }
				set_variable = { chaosx_nr25_survivor_security_loss = constant:chaosx_nr25_survivor.turn_outpost_security }
			}
			else = { set_variable = { chaosx_nr25_survivor_readiness_loss = constant:chaosx_nr25_survivor.turn_readiness } }
			if = {
				limit = { has_country_flag = chaosx_nr25_survivor_guard_route }
				multiply_variable = { chaosx_nr25_survivor_readiness_loss = constant:chaosx_nr25_survivor.guard_severity }
				multiply_variable = { chaosx_nr25_survivor_security_loss = constant:chaosx_nr25_survivor.guard_severity }
			}
			subtract_from_variable = { chaosx_nr25_logistics_readiness = chaosx_nr25_survivor_readiness_loss }
			subtract_from_variable = { chaosx_nr25_outpost_security = chaosx_nr25_survivor_security_loss }
			chaosx_nr25_clamp_values = yes
			clr_country_flag = chaosx_nr25_survivor_turner_identified
			if = {
				limit = {
					OR = {
						check_variable = { var = chaosx_nr25_intelligence value = constant:chaosx_nr25_report.sabotage_attribution_intelligence compare = greater_than_or_equals }
						check_variable = { var = chaosx_nr25_turn_roll value = constant:chaosx_nr25_survivor.turn_attribution_chance compare = less_than }
					}
				}
				set_country_flag = chaosx_nr25_survivor_turner_identified
				set_variable = { chaosx_nr25_survivor_turner_id = ROOT.id }
				add_opinion_modifier = { target = ROOT modifier = chaosx_nr25_sabotage_suspicion }
				ROOT = { set_country_flag = chaosx_nr25_survivor_turn_traced }
			}
			country_event = { id = chaosx.nr25.340 days = 1 }
		}
		if = {
			limit = { has_country_flag = chaosx_nr25_survivor_turn_traced }
			add_to_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_survivor.turn_attribution_exposure }
		}
		else = {
			subtract_from_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_cost.exposure_survivor_turn_reduction }
		}
		country_event = { id = chaosx.nr25.341 days = 1 }
	}
	set_country_flag = chaosx_nr25_survivor_turn_route
}
