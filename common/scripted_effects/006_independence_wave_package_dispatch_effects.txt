	independence_wave_dispatch_wallonia_frisia_package_final_validation = yes

	independence_wave_dispatch_iw093_iw098_package_final_validation = yes
	independence_wave_dispatch_montenegro_package_final_validation = yes
	independence_wave_dispatch_iberian_package_final_validation = yes
	independence_wave_dispatch_catalonia_package_final_validation = yes
	independence_wave_dispatch_transylvania_package_final_validation = yes
	independence_wave_dispatch_transcaucasus_package_final_validation = yes
	independence_wave_dispatch_karelia_crimea_package_final_validation = yes
	independence_wave_dispatch_macedonia_package_final_validation = yes
	independence_wave_dispatch_banat_package_final_validation = yes
	independence_wave_dispatch_thrace_package_final_validation = yes
	independence_wave_dispatch_epirus_package_final_validation = yes
	independence_wave_dispatch_bosnia_package_final_validation = yes
	# Every released Event 006 country must expose the one shared generic tree
	# or a reviewed additive carrier overlay. Package adapters own their
	# identity-specific checks, but this common barrier prevents a future
	# adapter from admitting a country with no usable Event 006 focus surface.
	if = {
		limit = { check_variable = { independence_wave_package_final_validation_success = constant:independence_wave_planner.selected } }
		if = {
			limit = {
				OR = {
					NOT = { has_independence_wave_generic_focus_contract = yes }
					NOT = { has_country_flag = independence_wave_generic_ai_profile }
				}
			}
			set_temp_variable = { independence_wave_package_final_validation_success = constant:independence_wave_planner.no_candidates }
		}
	}
}
