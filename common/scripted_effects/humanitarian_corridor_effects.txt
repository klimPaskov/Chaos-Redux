# ============================================================================
# CHAOS REDUX - HUMANITARIAN CORRIDOR CONTRACT EFFECTS
# ============================================================================
# Package-owned lifecycle and transaction adapters for an exact humanitarian
# corridor. The effects persist origin/front/counterpart database IDs, bind a
# single adjacent route, and reuse the existing exact civilian and reserve
# transfer helpers. They do not create events, register an event pool, add a
# GUI, add a mapmode, or scan the world. Attack receipts and owner call sites
# remain explicit inputs.
# ============================================================================

# COUNTRY scope. Schedule this country's next sparse corridor pulse
# pulse_cooldown_days after today. Date variables count one game hour as
# 0.00001, so every corridor day count passes through the shared
# chaosx_convert_days_to_date_offset helper before it is added to a date.
humanitarian_corridor_schedule_country_pulse = {
	set_temp_variable = { chaosx_date_days = constant:humanitarian_corridor_timing.pulse_cooldown_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { humanitarian_corridor_country_pulse_due = global.date }
	add_to_variable = { humanitarian_corridor_country_pulse_due = chaosx_date_offset }
}

# STATE scope. Set the corridor mission deadline mission_hold_days after today.
humanitarian_corridor_set_mission_deadline = {
	set_temp_variable = { chaosx_date_days = constant:humanitarian_corridor_timing.mission_hold_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { humanitarian_corridor_mission_deadline = global.date }
	add_to_variable = { humanitarian_corridor_mission_deadline = chaosx_date_offset }
}

# STATE scope. Copy caller-owned route evidence into durable contract fields.
# A missing request is represented as zero and is intentionally rejected by
# humanitarian_corridor_route_geometry_is_valid.
humanitarian_corridor_copy_request_proofs = {
	set_variable = { humanitarian_corridor_route_geometry_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_border_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_transport_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_safety_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_actor_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_ceasefire_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_food_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_route_reception_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_food_proof = constant:humanitarian_corridor_runtime.zero }
	set_variable = { humanitarian_corridor_reception_proof = constant:humanitarian_corridor_runtime.zero }
	if = { limit = { has_variable = humanitarian_corridor_route_geometry_proof_request } set_variable = { humanitarian_corridor_route_geometry_proof = humanitarian_corridor_route_geometry_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_border_proof_request } set_variable = { humanitarian_corridor_route_border_proof = humanitarian_corridor_route_border_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_transport_proof_request } set_variable = { humanitarian_corridor_route_transport_proof = humanitarian_corridor_route_transport_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_safety_proof_request } set_variable = { humanitarian_corridor_route_safety_proof = humanitarian_corridor_route_safety_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_actor_proof_request } set_variable = { humanitarian_corridor_route_actor_proof = humanitarian_corridor_route_actor_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_ceasefire_proof_request } set_variable = { humanitarian_corridor_route_ceasefire_proof = humanitarian_corridor_route_ceasefire_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_food_proof_request } set_variable = { humanitarian_corridor_route_food_proof = humanitarian_corridor_route_food_proof_request } }
	if = { limit = { has_variable = humanitarian_corridor_route_reception_proof_request } set_variable = { humanitarian_corridor_route_reception_proof = humanitarian_corridor_route_reception_proof_request } }
	set_variable = { humanitarian_corridor_food_proof = humanitarian_corridor_route_food_proof }
	set_variable = { humanitarian_corridor_reception_proof = humanitarian_corridor_route_reception_proof }
	if = {
		limit = { has_variable = humanitarian_corridor_route_generation_request }
		set_variable = { humanitarian_corridor_route_generation = humanitarian_corridor_route_generation_request }
	}
	else = { set_variable = { humanitarian_corridor_route_generation = global.date } }
	set_variable = { humanitarian_corridor_route_proof_generation = humanitarian_corridor_route_generation }
}

# STATE scope. Prepare one exact contract from the state-target decision. A
# caller may submit a front-state ID, but the ID still passes the same unique
# adjacency/controller predicate. Without an explicit ID the bounded neighbour
# loop must find exactly one candidate; zero or multiple candidates fail closed.
# STATE scope. Resolve the current whole origin row without movement/binding.
# All child outputs used here are declared before the resolver call.
humanitarian_corridor_resolve_live_cohort = {
	set_temp_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.zero }
	set_temp_variable = { humanitarian_corridor_live_requester_bound = constant:humanitarian_runtime.zero }
	save_event_target_as = humanitarian_corridor_live_origin
	if = {
		limit = { has_variable = humanitarian_corridor_requester_country_id var:humanitarian_corridor_requester_country_id = { humanitarian_country_is_valid = yes } }
		var:humanitarian_corridor_requester_country_id = { save_event_target_as = humanitarian_corridor_live_requester }
		set_temp_variable = { humanitarian_corridor_live_requester_bound = constant:humanitarian_runtime.one }
	}
	set_temp_variable = { migration_cohort_origin_resolution_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_destination_resolution_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { migration_cohort_resolved_amount = constant:humanitarian_population.zero }
	set_temp_variable = { migration_cohort_resolved_status = constant:migration_cohort_status.none }
	set_temp_variable = { migration_cohort_resolved_source = constant:humanitarian_pressure_source.unknown }
	set_temp_variable = { migration_cohort_resolution_count = constant:humanitarian_runtime.zero }
	set_temp_variable = { migration_cohort_id_request = humanitarian_corridor_cohort_id }
	migration_resolve_cohort_origin = yes
	if = {
		limit = {
			humanitarian_corridor_origin_cohort_is_current = yes
			check_variable = { humanitarian_corridor_live_requester_bound = constant:humanitarian_runtime.one }
			migration_cohort_origin_resolution_is_valid = yes
			check_variable = { migration_cohort_resolved_status = constant:migration_cohort_status.active }
			check_variable = { migration_cohort_resolved_amount = humanitarian_corridor_transfer_people }
			event_target:migration_cohort_origin = { state = event_target:humanitarian_corridor_live_origin }
			event_target:migration_cohort_host = { state = event_target:humanitarian_corridor_live_origin }
			event_target:migration_cohort_owner = { tag = event_target:humanitarian_corridor_live_requester }
		}
		set_temp_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.one }
	}
}

humanitarian_corridor_prepare_contract = {
	set_temp_variable = { humanitarian_corridor_request_result = constant:humanitarian_corridor_result.invalid }
	set_temp_variable = { humanitarian_corridor_request_reason = constant:humanitarian_corridor_reason.invalid_origin }
	if = {
		limit = { NOT = { check_variable = { humanitarian_corridor_requester_override_proven = constant:humanitarian_runtime.one } } }
		ROOT = { save_event_target_as = humanitarian_corridor_requester }
	}
	if = {
		limit = { humanitarian_corridor_origin_is_valid = yes }
		save_event_target_as = humanitarian_corridor_origin
		set_state_flag = humanitarian_corridor_contract_active
		set_state_flag = humanitarian_corridor_preparation_pending
		set_variable = { humanitarian_corridor_origin_state_id = THIS.id }
		set_variable = { humanitarian_corridor_requester_country_id = event_target:humanitarian_corridor_requester.id }
		set_variable = { humanitarian_corridor_origin_controller_country_id = CONTROLLER.id }
		set_variable = { humanitarian_corridor_origin_owner_country_id = OWNER.id }
		set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.prepared }
		# A new contract cannot inherit another operation's terminal display.
		clear_variable = humanitarian_corridor_last_terminal_status
		clear_variable = humanitarian_corridor_last_terminal_operation
		if = {
			limit = { has_variable = humanitarian_corridor_operation_request }
			set_variable = { humanitarian_corridor_operation = humanitarian_corridor_operation_request }
		}
		else = { set_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation } }
		if = {
			limit = { has_variable = humanitarian_corridor_relation_request }
			set_variable = { humanitarian_corridor_relation = humanitarian_corridor_relation_request }
		}
		else = { set_variable = { humanitarian_corridor_relation = constant:humanitarian_corridor_relation.enemy_front } }
		if = {
			limit = { has_variable = humanitarian_corridor_cohort_id_request }
			set_variable = { humanitarian_corridor_cohort_id = humanitarian_corridor_cohort_id_request }
		}
		else = { set_variable = { humanitarian_corridor_cohort_id = migration_current_cohort_id } }
		if = {
			limit = { has_variable = humanitarian_corridor_transfer_people_request }
			set_variable = { humanitarian_corridor_transfer_people = humanitarian_corridor_transfer_people_request }
		}
		else = {
			if = {
				limit = { has_variable = migration_trapped_population }
				set_variable = { humanitarian_corridor_transfer_people = migration_trapped_population }
			}
		}
		if = {
			limit = { has_variable = humanitarian_corridor_relief_amount_request }
			set_variable = { humanitarian_corridor_relief_amount = humanitarian_corridor_relief_amount_request }
		}
		else = { set_variable = { humanitarian_corridor_relief_amount = constant:famine_food_reserve.zero } }
		clear_variable = humanitarian_corridor_counterpart_country_id
		clear_variable = humanitarian_corridor_front_state_id
		clear_variable = humanitarian_corridor_offer_deadline
		clear_variable = humanitarian_corridor_operation_deadline
		clear_variable = humanitarian_corridor_mission_deadline
		clear_variable = humanitarian_corridor_evacuation_proven
		clear_variable = humanitarian_corridor_relief_proven
		clear_variable = humanitarian_corridor_attack_receipt_proven
		clear_variable = humanitarian_corridor_attack_attacker_country_id
		clear_variable = humanitarian_corridor_route_binding_active
		clear_variable = humanitarian_corridor_mission_armed
		clr_state_flag = humanitarian_corridor_offer_pending
		clr_state_flag = humanitarian_corridor_response_accepted
		clr_state_flag = humanitarian_corridor_operation_pending
		clr_state_flag = humanitarian_corridor_mission_armed
		clr_state_flag = humanitarian_corridor_attack_received
		humanitarian_corridor_copy_request_proofs = yes
		# Every prepared corridor receives a fresh persistent identity nonce. The
		# caller-supplied route generation is not allowed to replay an older crisis.
		humanitarian_achievement_advance_generation = yes
		set_variable = { humanitarian_corridor_route_generation = global.humanitarian_achievement_generation_counter }
		set_variable = { humanitarian_corridor_route_proof_generation = humanitarian_corridor_route_generation }
		set_variable = { humanitarian_corridor_candidate_count = constant:humanitarian_corridor_runtime.zero }
		clear_variable = humanitarian_corridor_candidate_front_state_id
		clear_variable = humanitarian_corridor_candidate_country_id
		if = {
			limit = {
				has_variable = humanitarian_corridor_front_state_request
				var:humanitarian_corridor_front_state_request = { humanitarian_corridor_front_candidate_is_valid = yes }
			}
			var:humanitarian_corridor_front_state_request = {
				save_event_target_as = humanitarian_corridor_candidate_front
				CONTROLLER = { save_event_target_as = humanitarian_corridor_candidate_controller }
			}
			set_variable = { humanitarian_corridor_candidate_count = constant:humanitarian_corridor_runtime.one }
		set_variable = { humanitarian_corridor_candidate_front_state_id = event_target:humanitarian_corridor_candidate_front.id }
		set_variable = { humanitarian_corridor_candidate_country_id = event_target:humanitarian_corridor_candidate_controller.id }
		}
		else = {
			every_neighbor_state = {
				limit = { humanitarian_corridor_front_candidate_is_valid = yes }
				save_event_target_as = humanitarian_corridor_candidate_front
				CONTROLLER = { save_event_target_as = humanitarian_corridor_candidate_controller }
				event_target:humanitarian_corridor_origin = {
					if = {
						limit = { check_variable = { var = humanitarian_corridor_candidate_count value = constant:humanitarian_corridor_runtime.zero compare = equals } }
						set_variable = { humanitarian_corridor_candidate_front_state_id = event_target:humanitarian_corridor_candidate_front.id }
						set_variable = { humanitarian_corridor_candidate_country_id = event_target:humanitarian_corridor_candidate_controller.id }
					}
					add_to_variable = { humanitarian_corridor_candidate_count = constant:humanitarian_corridor_runtime.candidate_increment }
				}
			}
		}
		if = {
			limit = { check_variable = { var = humanitarian_corridor_candidate_count value = constant:humanitarian_corridor_runtime.one compare = equals } }
			set_variable = { humanitarian_corridor_front_state_id = humanitarian_corridor_candidate_front_state_id }
			set_variable = { humanitarian_corridor_counterpart_country_id = humanitarian_corridor_candidate_country_id }
			set_variable = { humanitarian_corridor_origin_controller_id = humanitarian_corridor_origin_controller_country_id }
			set_variable = { humanitarian_corridor_front_controller_id = event_target:humanitarian_corridor_candidate_controller.id }
			set_variable = { humanitarian_corridor_route_geometry_proof = constant:humanitarian_corridor_runtime.one }
			set_variable = { humanitarian_corridor_route_border_proof = constant:humanitarian_corridor_runtime.one }
			set_variable = { humanitarian_corridor_route_actor_proof = constant:humanitarian_corridor_runtime.one }
			set_variable = { humanitarian_corridor_route_transport_proof = constant:humanitarian_corridor_runtime.zero }
			set_variable = { humanitarian_corridor_route_food_proof = constant:humanitarian_corridor_runtime.zero }
			var:humanitarian_corridor_front_state_id = {
				set_temp_variable = { migration_famine_food_safety_result = constant:humanitarian_runtime.zero }
				migration_consume_famine_food_safety = yes
			}
			if = {
				limit = {
					infrastructure > constant:migration_destination_selection_threshold.minimum_route_infrastructure
					has_variable = famine_component_transport
					var:humanitarian_corridor_front_state_id = {
						infrastructure > constant:migration_destination_selection_threshold.minimum_route_infrastructure
						has_variable = famine_component_transport
					}
				}
				set_variable = { humanitarian_corridor_route_transport_proof = constant:humanitarian_corridor_runtime.one }
			}
			if = {
				limit = {
					var:humanitarian_corridor_front_state_id = {
						OR = {
							famine_to_migration_food_safety_is_valid = yes
						}
						humanitarian_corridor_front_is_preparable = yes
					}
				}
				set_variable = { humanitarian_corridor_route_food_proof = constant:humanitarian_corridor_runtime.one }
			}
			set_variable = { humanitarian_corridor_food_proof = humanitarian_corridor_route_food_proof }
			set_temp_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.zero }
			if = { limit = { check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation } } humanitarian_corridor_resolve_live_cohort = yes }
			if = {
				limit = {
					humanitarian_corridor_offer_route_is_valid = yes
					OR = { check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.relief } check_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.one } }
				}
				if = {
					limit = {
						check_variable = { var = humanitarian_corridor_transfer_people value = constant:humanitarian_population.zero compare = greater_than }
						check_variable = { var = humanitarian_corridor_route_generation value = constant:humanitarian_corridor_runtime.zero compare = greater_than }
					}
					set_temp_variable = { migration_presentation_corridor_denominator = state_population_k }
					multiply_temp_variable = { migration_presentation_corridor_denominator = constant:chaos_meter_deaths.people_per_k }
					round_temp_variable = migration_presentation_corridor_denominator
					set_temp_variable = { migration_presentation_input_preparing_proven = constant:migration_presentation.one }
					set_temp_variable = { migration_presentation_input_preparing_people = humanitarian_corridor_transfer_people }
					set_temp_variable = { migration_presentation_input_preparing_denominator = migration_presentation_corridor_denominator }
					set_temp_variable = { migration_presentation_input_cohort_id = constant:migration_presentation.zero }
					if = {
						limit = { has_variable = humanitarian_corridor_cohort_id }
						set_temp_variable = { migration_presentation_input_cohort_id = humanitarian_corridor_cohort_id }
					}
					set_temp_variable = { migration_presentation_input_generation = humanitarian_corridor_route_generation }
					set_temp_variable = { migration_presentation_input_role = constant:migration_core_reconciliation.transfer_projection_role_organized_evacuation }
					migration_presentation_record_preparing_projection = yes
					humanitarian_refresh_state_modifiers = yes
				}
				clr_state_flag = humanitarian_corridor_preparation_pending
				set_temp_variable = { humanitarian_corridor_request_result = constant:humanitarian_corridor_result.valid }
				set_temp_variable = { humanitarian_corridor_request_reason = constant:humanitarian_corridor_reason.unknown }
				migration_register_active_state = yes
			}
			else = {
				set_state_flag = humanitarian_corridor_route_invalid
				set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.missing_route_proof }
				set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.missing_route_proof }
				set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.invalidated }
				set_temp_variable = { humanitarian_corridor_request_reason = constant:humanitarian_corridor_reason.missing_route_proof }
				humanitarian_corridor_cleanup = yes
			}
		}
		else_if = {
			limit = { check_variable = { var = humanitarian_corridor_candidate_count value = constant:humanitarian_corridor_runtime.one compare = greater_than } }
			set_state_flag = humanitarian_corridor_selection_ambiguous
			set_state_flag = humanitarian_corridor_route_invalid
			set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.ambiguous_front }
			set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.ambiguous_front }
			set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.invalidated }
			set_temp_variable = { humanitarian_corridor_request_result = constant:humanitarian_corridor_result.ambiguous }
			set_temp_variable = { humanitarian_corridor_request_reason = constant:humanitarian_corridor_reason.ambiguous_front }
			humanitarian_corridor_cleanup = yes
		}
		else = {
			set_state_flag = humanitarian_corridor_route_invalid
			set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.no_front }
			set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.no_front }
			set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.invalidated }
			set_temp_variable = { humanitarian_corridor_request_result = constant:humanitarian_corridor_result.no_front }
			set_temp_variable = { humanitarian_corridor_request_reason = constant:humanitarian_corridor_reason.no_front }
			humanitarian_corridor_cleanup = yes
		}
	}
	else = {
		set_temp_variable = { humanitarian_corridor_request_result = constant:humanitarian_corridor_result.invalid }
		set_temp_variable = { humanitarian_corridor_request_reason = constant:humanitarian_corridor_reason.invalid_origin }
	}
	clear_variable = humanitarian_corridor_candidate_count
	clear_variable = humanitarian_corridor_candidate_front_state_id
	clear_variable = humanitarian_corridor_candidate_country_id
	clear_variable = humanitarian_corridor_front_state_request
	clear_variable = humanitarian_corridor_relation_request
	clear_variable = humanitarian_corridor_operation_request
	clear_variable = humanitarian_corridor_cohort_id_request
	clear_variable = humanitarian_corridor_transfer_people_request
	clear_variable = humanitarian_corridor_relief_amount_request
	clear_variable = humanitarian_corridor_route_geometry_proof_request
	clear_variable = humanitarian_corridor_route_border_proof_request
	clear_variable = humanitarian_corridor_route_transport_proof_request
	clear_variable = humanitarian_corridor_route_safety_proof_request
	clear_variable = humanitarian_corridor_route_actor_proof_request
	clear_variable = humanitarian_corridor_route_ceasefire_proof_request
	clear_variable = humanitarian_corridor_route_food_proof_request
		clear_variable = humanitarian_corridor_route_reception_proof_request
		clear_variable = humanitarian_corridor_route_generation_request
	set_temp_variable = { humanitarian_corridor_requester_override_proven = constant:humanitarian_runtime.zero }
}

# STATE scope. Submit a pending offer to the exact stored counterpart. The
# counterpart's country mirror is the only response surface; no event is fired.
humanitarian_corridor_submit_offer = {
	set_temp_variable = { humanitarian_corridor_offer_result = constant:humanitarian_corridor_result.invalid }
	if = {
		limit = { humanitarian_corridor_offer_request_is_valid = yes }
		if = {
			limit = { var:humanitarian_corridor_counterpart_country_id = { has_country_flag = humanitarian_corridor_offer_pending } }
			set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.duplicate_offer }
			set_temp_variable = { humanitarian_corridor_offer_result = constant:humanitarian_corridor_result.duplicate }
		}
		else = {
			save_event_target_as = humanitarian_corridor_origin
			set_temp_variable = { chaosx_date_days = constant:humanitarian_corridor_timing.offer_response_days }
			set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
			chaosx_convert_days_to_date_offset = yes
			set_variable = { humanitarian_corridor_offer_deadline = global.date }
			add_to_variable = { humanitarian_corridor_offer_deadline = chaosx_date_offset }
			set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.offer_pending }
			set_state_flag = humanitarian_corridor_offer_pending
			var:humanitarian_corridor_counterpart_country_id = {
				set_country_flag = humanitarian_corridor_offer_pending
				set_variable = { humanitarian_corridor_offer_origin_state_id = event_target:humanitarian_corridor_origin.id }
				set_variable = { humanitarian_corridor_offer_front_state_id = event_target:humanitarian_corridor_origin.humanitarian_corridor_front_state_id }
				set_variable = { humanitarian_corridor_offer_requester_country_id = event_target:humanitarian_corridor_origin.humanitarian_corridor_requester_country_id }
				set_variable = { humanitarian_corridor_offer_deadline = event_target:humanitarian_corridor_origin.humanitarian_corridor_offer_deadline }
				set_variable = { humanitarian_corridor_offer_operation = event_target:humanitarian_corridor_origin.humanitarian_corridor_operation }
				set_variable = { humanitarian_corridor_offer_cohort_id = event_target:humanitarian_corridor_origin.humanitarian_corridor_cohort_id }
				set_variable = { humanitarian_corridor_offer_route_generation = event_target:humanitarian_corridor_origin.humanitarian_corridor_route_generation }
				set_country_flag = humanitarian_corridor_country_pulse_pending
				humanitarian_corridor_schedule_country_pulse = yes
				if = {
					limit = {
						check_variable = { humanitarian_corridor_offer_operation = constant:humanitarian_corridor_operation.evacuation }

# COUNTRY scope. The counterpart's acceptance is a real state transition only;
# population, reserve, and cohort transactions remain separate exact calls.
humanitarian_corridor_accept_offer = {
	set_temp_variable = { humanitarian_corridor_response_result = constant:humanitarian_corridor_result.invalid }
	set_temp_variable = { migration_reception_front_intent_record_result = constant:migration_reception_front_intent.zero }
	set_temp_variable = { migration_reception_front_intent_reveal_proven = constant:migration_reception_front_intent.zero }
	save_event_target_as = humanitarian_corridor_response_country
	if = {
		limit = { has_country_flag = humanitarian_corridor_offer_pending has_variable = humanitarian_corridor_offer_front_state_id }
		var:humanitarian_corridor_offer_front_state_id = { migration_consume_famine_food_safety = yes OWNER = { if = { limit = { check_variable = { migration_reception_capacity_dirty = constant:migration_capacity.one } } migration_recalculate_reception_capacity = yes } } }
	}
	# Snapshot while the offer is still pending. After status changes, the
	# pending-issue predicate intentionally ceases to be true.
	if = { limit = { migration_pending_reception_issue_is_valid = yes } set_temp_variable = { migration_reception_front_intent_reveal_proven = constant:migration_reception_front_intent.one } }
	if = {
		limit = { humanitarian_corridor_offer_can_be_accepted = yes }
		var:humanitarian_corridor_offer_origin_state_id = {
			set_temp_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.zero }
			if = { limit = { check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation } } humanitarian_corridor_resolve_live_cohort = yes }
			if = {
				limit = {
					humanitarian_corridor_acceptance_is_valid = yes
					OR = {
						AND = {
							check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation }
							humanitarian_corridor_full_request_headroom_is_valid = yes
							check_variable = { humanitarian_corridor_live_cohort_result = constant:humanitarian_runtime.one }
						}
						AND = {
							check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.relief }
							var:humanitarian_corridor_front_state_id = { humanitarian_corridor_front_destination_is_safe = yes }
						}
					}
				}
				set_variable = { humanitarian_corridor_route_safety_proof = constant:humanitarian_corridor_runtime.one }
				set_variable = { humanitarian_corridor_route_ceasefire_proof = constant:humanitarian_corridor_runtime.one }
				set_variable = { humanitarian_corridor_route_reception_proof = constant:humanitarian_corridor_runtime.one }
				set_variable = { humanitarian_corridor_reception_proof = constant:humanitarian_corridor_runtime.one }
				set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.accepted }
				clr_state_flag = humanitarian_corridor_offer_pending
				set_state_flag = humanitarian_corridor_response_accepted
				set_state_flag = humanitarian_corridor_accepted
				set_state_flag = humanitarian_corridor_operation_pending
				set_temp_variable = { chaosx_date_days = constant:humanitarian_corridor_timing.mission_hold_days }
				set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
				chaosx_convert_days_to_date_offset = yes
				set_variable = { humanitarian_corridor_operation_deadline = global.date }
				add_to_variable = { humanitarian_corridor_operation_deadline = chaosx_date_offset }
				clear_variable = humanitarian_corridor_offer_deadline
				# Bind the receiving country's achievement ledger to this exact
				# accepted origin/front/counterpart/cohort contract. The old aggregate
				# trapped-population proxy is deliberately not used.
				save_event_target_as = humanitarian_corridor_origin
				event_target:humanitarian_corridor_response_country = {
					set_variable = { humanitarian_achievement_gate_generation_request = PREV.humanitarian_corridor_route_generation }
					set_variable = { humanitarian_achievement_gate_origin_state_id_request = PREV.id }
					set_variable = { humanitarian_achievement_gate_front_state_id_request = PREV.humanitarian_corridor_front_state_id }
					set_variable = { humanitarian_achievement_gate_counterpart_country_id_request = THIS.id }
					set_variable = { humanitarian_achievement_gate_offered_cohort_id_request = PREV.humanitarian_corridor_cohort_id }
					humanitarian_achievement_record_gate_crisis = yes
				}
				var:humanitarian_corridor_requester_country_id = {
					set_country_flag = humanitarian_corridor_response_accepted
					clr_country_flag = humanitarian_corridor_request_pending
					if = {
						limit = { PREV = { NOT = { check_variable = { migration_spontaneous_reception_cohort_id = humanitarian_corridor_cohort_id } } } }
						set_temp_variable = { famine_achievement_corridor_generation_request = PREV.humanitarian_corridor_route_generation }
						set_temp_variable = { famine_achievement_corridor_origin_state_id_request = PREV.id }
						set_temp_variable = { famine_achievement_corridor_front_state_id_request = PREV.humanitarian_corridor_front_state_id }
						set_temp_variable = { famine_achievement_corridor_counterpart_country_id_request = PREV.humanitarian_corridor_counterpart_country_id }
						famine_achievement_start_corridor_attempt = yes
					}
					add_stability = constant:migration_decision_outcome.decision_standard_stability
					humanitarian_corridor_schedule_country_pulse = yes
					set_country_flag = humanitarian_corridor_country_pulse_pending
				}
				event_target:humanitarian_corridor_response_country = { set_temp_variable = { humanitarian_corridor_response_result = constant:humanitarian_corridor_result.valid } }
				set_temp_variable = { migration_reception_front_intent_setup_proven = constant:migration_reception_front_intent.zero }
				set_temp_variable = { migration_reception_front_intent_phase_request = constant:migration_reception_front_intent.agreed }
				migration_publish_reception_front_intent_from_contract = yes
			}
		}
		if = {
			limit = { check_variable = { humanitarian_corridor_response_result = constant:humanitarian_corridor_result.valid } }
			clr_country_flag = humanitarian_corridor_offer_pending
			set_country_flag = humanitarian_corridor_response_accepted
			clear_variable = humanitarian_corridor_offer_origin_state_id
			clear_variable = humanitarian_corridor_offer_front_state_id
			clear_variable = humanitarian_corridor_offer_requester_country_id
			clear_variable = humanitarian_corridor_offer_deadline
			clear_variable = humanitarian_corridor_offer_operation
			clear_variable = humanitarian_corridor_offer_cohort_id
			clear_variable = humanitarian_corridor_offer_route_generation
			if = {
				limit = { NOT = { has_variable = humanitarian_corridor_origin_state_id } }
				clr_country_flag = humanitarian_corridor_country_pulse_pending
				clear_variable = humanitarian_corridor_country_pulse_due
			}
		}

	}
	set_temp_variable = { migration_reception_front_intent_reveal_proven = constant:migration_reception_front_intent.zero }
}

# STATE scope. Expiry is terminal and never rolls back an already committed
# civilian/reserve transaction. The response deadline and mission deadline are
# distinguished so audit tools can identify where the contract ended.
humanitarian_corridor_expire = {
	set_temp_variable = { humanitarian_corridor_expire_result = constant:humanitarian_corridor_result.invalid }
	if = {
		limit = { has_state_flag = humanitarian_corridor_contract_active }
		set_variable = { humanitarian_corridor_last_terminal_status = constant:humanitarian_corridor_status.expired }
		set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.mission_deadline }
		set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.mission_deadline }
		if = {
			limit = { check_variable = { var = humanitarian_corridor_status value = constant:humanitarian_corridor_status.offer_pending compare = equals } }
			set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.response_deadline }
			set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.response_deadline }
			# Letting a still-acceptable request lapse unanswered obstructs relief
			# like a refusal; an offer that became invalid is nobody's refusal.
			if = {
				limit = {
					has_variable = humanitarian_corridor_offer_deadline
					check_variable = { var = global.date value = humanitarian_corridor_offer_deadline compare = greater_than_or_equals }
					# Everything but the deadline still allowed acceptance.
					humanitarian_state_is_valid = yes
					humanitarian_corridor_counterpart_is_valid = yes
					humanitarian_corridor_offer_route_is_valid = yes
				}
				set_temp_variable = { famine_condemnation_result = constant:humanitarian_runtime.zero }
				famine_condemn_corridor_obstruction = yes
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = humanitarian_corridor_route_safety_proof value = constant:humanitarian_corridor_runtime.zero compare = equals }
					check_variable = { var = humanitarian_corridor_route_ceasefire_proof value = constant:humanitarian_corridor_runtime.zero compare = equals }
				}
			}
			set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.route_unsafe }
			set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.route_unsafe }
		}
		set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.expired }
		humanitarian_corridor_cleanup = yes
		set_temp_variable = { humanitarian_corridor_expire_result = constant:humanitarian_corridor_result.expired }
	}
}

# STATE scope. Revalidates the persisted corridor relation after an exact war
# or peace callback. The route is invalidated only when its stored relation no
# longer matches the current controller relationship; no war graph is scanned.
humanitarian_corridor_revalidate_route_relation = {
	if = {
		limit = { humanitarian_state_is_valid = yes }
		if = {
			limit = { CONTROLLER = { humanitarian_country_is_valid = yes } }
			CONTROLLER = { save_event_target_as = humanitarian_corridor_relation_origin_controller }
		}
		if = {
			limit = {
				has_event_target = humanitarian_corridor_relation_origin_controller
				CONTROLLER = { humanitarian_country_is_valid = yes }
				humanitarian_corridor_front_state_is_valid = yes
				has_variable = humanitarian_corridor_relation
				OR = {
					AND = {
						check_variable = { var = humanitarian_corridor_relation value = constant:humanitarian_corridor_relation.enemy_front compare = equals }
						var:humanitarian_corridor_counterpart_country_id = { has_war_with = event_target:humanitarian_corridor_relation_origin_controller }
					}
					AND = {
						check_variable = { var = humanitarian_corridor_relation value = constant:humanitarian_corridor_relation.neutral_border compare = equals }
						var:humanitarian_corridor_counterpart_country_id = {
							NOT = { has_war_with = event_target:humanitarian_corridor_relation_origin_controller }
							NOT = { is_in_faction_with = event_target:humanitarian_corridor_relation_origin_controller }
						}
					}
					AND = {
						check_variable = { var = humanitarian_corridor_relation value = constant:humanitarian_corridor_relation.faction_partner compare = equals }
						var:humanitarian_corridor_counterpart_country_id = {
							NOT = { has_war_with = event_target:humanitarian_corridor_relation_origin_controller }
							is_in_faction_with = event_target:humanitarian_corridor_relation_origin_controller
						}
					}
					AND = {
						check_variable = { var = humanitarian_corridor_relation value = constant:humanitarian_corridor_relation.neighbor compare = equals }
						var:humanitarian_corridor_counterpart_country_id = { NOT = { has_war_with = event_target:humanitarian_corridor_relation_origin_controller } }
					}
				}
			}
		}
		else = {
			set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.route_closed }
			set_variable = { humanitarian_corridor_cleanup_reason = constant:humanitarian_corridor_reason.route_closed }
			set_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.invalidated }
			humanitarian_corridor_cleanup = yes
		}
	}
}

# STATE scope. Idempotent terminal cleanup. Last status/reason, transaction
# amounts, and attacker evidence remain for audits; active IDs, response
# mirrors, mission slots, and generic route proof do not.
humanitarian_corridor_cleanup = {
	# Capture retirement identity before any active contract field is cleared.
	# An explicitly supplied old tuple is preserved by the bridge.
	set_temp_variable = { migration_reception_front_intent_clear_result = constant:migration_reception_front_intent.zero }
	if = {
		limit = {
			OR = {
				check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation }
				check_variable = { migration_reception_front_intent_clear_proven = constant:migration_reception_front_intent.one }
			}
		}
		migration_retire_reception_front_intent_from_contract = yes
	}
	# Corridor cleanup is an exact terminal for its preparation/organized
	# presentation receipts. Durable depopulation and resettlement/return
	# history is intentionally untouched here.
	set_temp_variable = { humanitarian_corridor_presentation_clear_refresh = constant:humanitarian_corridor_runtime.zero }
	if = {
		limit = { has_state_flag = migration_presentation_preparing_projection_active }
		set_temp_variable = { humanitarian_corridor_presentation_clear_refresh = constant:humanitarian_corridor_runtime.one }
		migration_presentation_clear_preparing_projection = yes
	}
	if = {
		limit = { has_state_flag = migration_presentation_organized_projection_active }
		set_temp_variable = { humanitarian_corridor_presentation_clear_refresh = constant:humanitarian_corridor_runtime.one }
		migration_presentation_clear_organized_evacuation_projection = yes
	}
	if = {
		limit = {
			check_variable = { var = humanitarian_corridor_presentation_clear_refresh value = constant:humanitarian_corridor_runtime.one compare = equals }
		}
		humanitarian_refresh_state_modifiers = yes
	}
	set_temp_variable = { humanitarian_corridor_presentation_clear_refresh = constant:humanitarian_corridor_runtime.zero }
	if = {
		limit = { has_variable = humanitarian_corridor_status }
		set_variable = { humanitarian_corridor_last_terminal_status = humanitarian_corridor_status }
		# Preserve only explicitly known operation provenance before live cleanup.
		# Repeated cleanup without a new status leaves the previous pair intact.
		clear_variable = humanitarian_corridor_last_terminal_operation
		if = {
			limit = {
				has_variable = humanitarian_corridor_operation
				OR = {
					check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation }
					check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.relief }
				}
			}
			set_variable = { humanitarian_corridor_last_terminal_operation = humanitarian_corridor_operation }
		}
	}
	set_state_flag = humanitarian_corridor_cleanup_pending
	if = {
		limit = { has_variable = humanitarian_corridor_counterpart_country_id }
		# Snapshot the receiving country's terminal cohort identity before this
		# cleanup clears the route mirror and counterpart IDs.
		var:humanitarian_corridor_counterpart_country_id = {
			humanitarian_achievement_record_gate_terminal_cleanup = yes
		}
	}
	if = {
		limit = { has_variable = humanitarian_corridor_front_state_id }
		humanitarian_corridor_clear_route_binding = yes
	}
	if = {
		limit = { has_variable = humanitarian_corridor_counterpart_country_id }
		var:humanitarian_corridor_counterpart_country_id = {
			if = {
				limit = {
					check_variable = { humanitarian_corridor_offer_route_generation = PREV.humanitarian_corridor_route_generation }
					check_variable = { humanitarian_corridor_offer_origin_state_id = PREV.id }
				}
			clr_country_flag = humanitarian_corridor_offer_pending

# COUNTRY scope. Sparse country pulse checks one mirrored origin only. A stale
# offer expires through the normal lifecycle; a changed front/controller also
# expires through the same path. No any_owned_state/any_country scan occurs.
humanitarian_corridor_clear_stale_incoming_mirror = {
	if = {
		limit = { humanitarian_country_is_valid = yes NOT = { humanitarian_corridor_incoming_mirror_matches_live_contract = yes } }
	clr_country_flag = humanitarian_corridor_offer_pending
	clr_country_flag = humanitarian_corridor_offer_accepted
	clear_variable = humanitarian_corridor_offer_origin_state_id
	clear_variable = humanitarian_corridor_offer_front_state_id
	clear_variable = humanitarian_corridor_offer_requester_country_id
	clear_variable = humanitarian_corridor_offer_deadline
	clear_variable = humanitarian_corridor_offer_operation
	clear_variable = humanitarian_corridor_offer_cohort_id
	clear_variable = humanitarian_corridor_offer_route_generation
	humanitarian_corridor_rearm_surviving_country_identity = yes
	}
}

humanitarian_corridor_sparse_country_pulse = {
	set_temp_variable = { humanitarian_corridor_pulse_result = constant:humanitarian_corridor_result.invalid }
	save_event_target_as = humanitarian_corridor_pulse_country
	if = {
		limit = { humanitarian_corridor_country_pulse_is_valid = yes }
		clear_variable = humanitarian_corridor_country_pulse_due
		clr_country_flag = humanitarian_corridor_country_pulse_pending
	set_variable = { humanitarian_corridor_country_pulse_last_date = global.date }
		if = {
			limit = { has_variable = humanitarian_corridor_offer_origin_state_id }
			if = {
				limit = { humanitarian_corridor_incoming_mirror_matches_live_contract = yes }
				var:humanitarian_corridor_offer_origin_state_id = {
					if = {
						limit = { event_target:humanitarian_corridor_pulse_country = { NOT = { humanitarian_corridor_offer_is_valid = yes } } }
					set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.response_deadline }
					humanitarian_corridor_expire = yes
				}
				else = {
					event_target:humanitarian_corridor_pulse_country = {
						set_temp_variable = { humanitarian_corridor_pulse_result = constant:humanitarian_corridor_result.valid }
						humanitarian_corridor_schedule_country_pulse = yes
						set_country_flag = humanitarian_corridor_country_pulse_pending
					}
				}
			}
			}
			else = { humanitarian_corridor_clear_stale_incoming_mirror = yes }
		}
		else_if = {
			limit = { has_variable = humanitarian_corridor_origin_state_id }
			if = {
				limit = {
					var:humanitarian_corridor_origin_state_id = {
						humanitarian_state_is_valid = yes
						has_state_flag = humanitarian_corridor_contract_active
						check_variable = { humanitarian_corridor_requester_country_id = PREV.id }
						check_variable = { humanitarian_corridor_route_generation = PREV.humanitarian_corridor_request_route_generation }
					}
				}
			var:humanitarian_corridor_origin_state_id = {
				if = {
					limit = {
						OR = {
							NOT = { humanitarian_corridor_front_state_is_valid = yes }
							AND = { has_state_flag = humanitarian_corridor_offer_pending NOT = { humanitarian_corridor_acceptance_is_valid = yes } }
							AND = { has_state_flag = humanitarian_corridor_operation_pending NOT = { has_variable = humanitarian_corridor_evacuation_proven } has_variable = humanitarian_corridor_operation_deadline check_variable = { var = global.date value = humanitarian_corridor_operation_deadline compare = greater_than_or_equals } }
						}
					}
					set_variable = { humanitarian_corridor_last_cleanup_reason = constant:humanitarian_corridor_reason.control_change }
					humanitarian_corridor_expire = yes
				}
				else = {
					event_target:humanitarian_corridor_pulse_country = {
						set_temp_variable = { humanitarian_corridor_pulse_result = constant:humanitarian_corridor_result.valid }
						humanitarian_corridor_schedule_country_pulse = yes
						set_country_flag = humanitarian_corridor_country_pulse_pending
					}
				}
			}
			}
			else = {
				clr_country_flag = humanitarian_corridor_request_pending
				clr_country_flag = humanitarian_corridor_response_accepted
				clear_variable = humanitarian_corridor_origin_state_id
				clear_variable = humanitarian_corridor_request_route_generation
				clear_variable = humanitarian_corridor_counterpart_country_id
				clear_variable = humanitarian_corridor_front_state_id
				humanitarian_corridor_rearm_surviving_country_identity = yes
			}
		}
	}
}
