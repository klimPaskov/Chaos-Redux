# Shared native/panel action implementations and dispatch.
# Called in an immediate actor-country event so legacy deep ROOT consumers keep their actor.

# COUNTRY scope after the complete action tail. The transfer owner has already
# published its atomic receipt; this final pass republishes endpoints after
# action-owned preparation, trapped, mission, return, or reception flags settle.
migration_gui_publish_transfer_participants = {
	if = {
		limit = {
			check_variable = { var = civilian_transfer_finalize_result value = constant:migration_core_reconciliation.transfer_finalize_valid compare = equals }
			check_variable = { var = civilian_transfer_actual_origin_debit value = constant:humanitarian_population.zero compare = greater_than }
		}
		if = {
			limit = { has_event_target = civilian_transfer_route_origin }
			event_target:civilian_transfer_route_origin = { national_policy_panel_publish_state = yes }
		}
		if = {
			limit = { has_event_target = civilian_transfer_route_destination }
			event_target:civilian_transfer_route_destination = { national_policy_panel_publish_state = yes }
		}
	}
}

migration_prepare_evacuation_action_begin_work = {
	if = {
			limit = { migration_prepare_evacuation_can_start_in_context = yes }
			set_temp_variable = { migration_transport_payment_result = constant:migration_transport_cost.zero }
			set_temp_variable = { migration_transport_fuel_land = constant:migration_transport_cost.light_fuel_land } set_temp_variable = { migration_transport_fuel_sea = constant:migration_transport_cost.light_fuel_sea } set_temp_variable = { migration_transport_convoys_sea = constant:migration_transport_cost.light_convoys_sea } set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:migration_prepare_evacuation } migration_transport_pay_state_cost = yes
			if = {
				limit = { check_variable = { migration_transport_payment_result = constant:migration_transport_cost.one } }
				add_equipment_to_stockpile = { type = train_equipment amount = constant:humanitarian_legacy_cost.train_light_debit }
				var:humanitarian_action_target = {
					set_state_flag = migration_prepare_evacuation_cost_paid
					set_variable = { migration_prepare_evacuation_cost_payer = ROOT }
				}
				set_country_flag = migration_decision_active
				clr_country_flag = migration_decision_emerging
				clr_country_flag = migration_decision_dormant
				clr_country_flag = migration_decision_resolution
			}
		}
}

migration_prepare_evacuation_action_resolve_work = {
	if = {
			limit = { migration_prepare_evacuation_cost_receipt_is_valid_in_context = yes }
			var:humanitarian_action_target = {
				clr_state_flag = migration_prepare_evacuation_cost_paid
				clear_variable = migration_prepare_evacuation_cost_payer
			}
			custom_effect_tooltip = migration_decision_effect_tt
			var:humanitarian_action_target = {
				set_state_flag = migration_evacuation_prepared
				national_policy_panel_publish_state = yes
			}
			add_stability = constant:migration_decision_outcome.decision_minor_stability_negative
		}
		else_if = {
			limit = {
				var:humanitarian_action_target = {
					has_state_flag = migration_prepare_evacuation_cost_paid
					has_variable = migration_prepare_evacuation_cost_payer
					var:migration_prepare_evacuation_cost_payer = { tag = ROOT }
				}
			}
			var:humanitarian_action_target = {
				clr_state_flag = migration_prepare_evacuation_cost_paid
				clear_variable = migration_prepare_evacuation_cost_payer
			}
		}
		set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:migration_prepare_evacuation } migration_transport_forget_state_cost = yes
}
