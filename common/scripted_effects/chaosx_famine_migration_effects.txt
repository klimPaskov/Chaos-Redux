# ============================================================================
# CHAOS REDUX - FAMINE AND MIGRATION SCRIPTED EFFECTS
# ============================================================================
# Reusable, fail-closed contracts for food-security pressure, displacement
# registries, route validation, exact civilian movement, border policy,
# reception, return, integration, resettlement, and invalidation cleanup.
# The file never performs a whole-world scan. Owner systems submit explicit
# proof and amounts, while the exact transfer keeps one population ledger.
# ============================================================================

famine_migration_initialize_runtime = {
	if = {
		limit = { NOT = { has_global_flag = famine_migration_runtime_initialized } }
		clear_array = global.famine_migration_active_food_states
		clear_array = global.famine_migration_active_displacement_states
		clear_array = global.famine_migration_active_displacement_countries
		clear_array = global.famine_migration_historical_profile_candidate_states
		clear_array = global.famine_migration_historical_profile_anchor_states
		clear_array = global.famine_migration_cohort_ids
		clear_array = global.famine_migration_cohort_origins
		clear_array = global.famine_migration_cohort_hosts
		clear_array = global.famine_migration_cohort_destinations
		clear_array = global.famine_migration_cohort_owners
		clear_array = global.famine_migration_cohort_amounts
		clear_array = global.famine_migration_cohort_sources
		clear_array = global.famine_migration_cohort_statuses
		set_variable = { global.famine_migration_active_food_states_count = constant:famine_migration_runtime.zero }
		set_variable = { global.famine_migration_active_displacement_states_count = constant:famine_migration_runtime.zero }
		set_variable = { global.famine_migration_active_displacement_countries_count = constant:famine_migration_runtime.zero }
		set_variable = { global.famine_migration_historical_profile_candidate_states_count = constant:famine_migration_runtime.zero }
		set_variable = { global.famine_migration_historical_profile_anchor_states_count = constant:famine_migration_runtime.zero }
		set_variable = { global.famine_migration_cohort_count = constant:famine_migration_runtime.zero }
		set_variable = { global.famine_migration_next_cohort_id = constant:famine_migration_runtime.next_cohort_id_start }
		set_variable = { global.famine_migration_conservation_debit = constant:famine_migration_population.zero }
		set_variable = { global.famine_migration_conservation_credit = constant:famine_migration_population.zero }
		set_variable = { global.famine_migration_conservation_route_deaths = constant:famine_migration_population.zero }
		set_variable = { global.famine_migration_conservation_residual = constant:famine_migration_population.zero }
		set_global_flag = famine_migration_runtime_initialized
	}
}

# ---------------------------------------------------------------------------
# Persisted cohort ledger
# ---------------------------------------------------------------------------
# Every row keeps aligned identity, ownership, amount, and lifecycle arrays:
# original state, current host, and (only after a real destination scope is
# entered) destination. A missing destination remains unbound and is never
# replaced with a random neighbour.

famine_migration_record_displaced_cohort = {
	famine_migration_initialize_runtime = yes
	set_temp_variable = { famine_migration_cohort_record_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_cohort_id_result = constant:famine_migration_runtime.zero }
	if = {
		limit = { famine_migration_cohort_record_request_is_valid = yes }
		set_temp_variable = { famine_migration_cohort_new_id = global.famine_migration_next_cohort_id }
		add_to_array = { array = global.famine_migration_cohort_ids value = famine_migration_cohort_new_id }
		add_to_array = { array = global.famine_migration_cohort_origins value = THIS }
		add_to_array = { array = global.famine_migration_cohort_hosts value = THIS }
		# The origin placeholder is not a destination identity. Status controls
		# whether this slot may ever be used as a route endpoint.
		add_to_array = { array = global.famine_migration_cohort_destinations value = THIS }
		add_to_array = { array = global.famine_migration_cohort_owners value = OWNER }
		add_to_array = { array = global.famine_migration_cohort_amounts value = famine_migration_cohort_record_amount }
		add_to_array = { array = global.famine_migration_cohort_sources value = famine_migration_cohort_record_source }
		add_to_array = { array = global.famine_migration_cohort_statuses value = constant:famine_migration_cohort_status.active }
		add_to_variable = { global.famine_migration_next_cohort_id = constant:famine_migration_runtime.array_index_increment }
		set_variable = { global.famine_migration_cohort_count = global.famine_migration_cohort_ids^num }
		set_temp_variable = { famine_migration_cohort_id_result = famine_migration_cohort_new_id }
		set_temp_variable = { famine_migration_cohort_record_result = constant:famine_migration_route_result.valid }
		if = {
			limit = {
				has_variable = famine_migration_current_cohort_id
				NOT = { check_variable = { var = famine_migration_current_cohort_id value = famine_migration_cohort_new_id compare = equals } }
			}
			clear_variable = famine_migration_current_cohort_id
			set_state_flag = famine_migration_cohort_selection_ambiguous
		}
		else_if = {
			limit = { NOT = { has_state_flag = famine_migration_cohort_selection_ambiguous } }
			set_variable = { famine_migration_current_cohort_id = famine_migration_cohort_new_id }
		}
		OWNER = {
			if = {
				limit = {
					has_variable = famine_migration_current_country_cohort_id
					NOT = { check_variable = { var = famine_migration_current_country_cohort_id value = famine_migration_cohort_new_id compare = equals } }
				}
				clear_variable = famine_migration_current_country_cohort_id
				set_country_flag = famine_migration_country_cohort_selection_ambiguous
			}
			else_if = {
				limit = { NOT = { has_country_flag = famine_migration_country_cohort_selection_ambiguous } }
				set_variable = { famine_migration_current_country_cohort_id = famine_migration_cohort_new_id }
			}
		}
	}
	set_temp_variable = { famine_migration_cohort_record_request_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_record_amount = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_cohort_record_owner_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_record_source = constant:famine_migration_pressure_source.unknown }
}

# Bind a destination only from inside that actual destination state scope.
# The regular route target is saved for the current effect chain so callers
# may return to origin safely; the global row remains the durable identity.
famine_migration_bind_cohort_destination = {
	famine_migration_initialize_runtime = yes
	set_temp_variable = { famine_migration_cohort_destination_bind_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_cohort_destination_bind_break = constant:famine_migration_runtime.zero }
	if = {
		limit = { famine_migration_cohort_destination_bind_request_is_valid = yes }
		set_temp_variable = { famine_migration_cohort_destination_bind_index = global.famine_migration_cohort_ids^num }
		while_loop_effect = {
			break = famine_migration_cohort_destination_bind_break
			limit = {
				check_variable = { var = famine_migration_cohort_destination_bind_index value = constant:famine_migration_runtime.zero compare = greater_than }
				check_variable = { var = famine_migration_cohort_destination_bind_break value = constant:famine_migration_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { famine_migration_cohort_destination_bind_index = constant:famine_migration_runtime.array_index_increment }
			set_temp_variable = { famine_migration_cohort_row_id = global.famine_migration_cohort_ids^famine_migration_cohort_destination_bind_index }
			set_temp_variable = { famine_migration_cohort_row_status = global.famine_migration_cohort_statuses^famine_migration_cohort_destination_bind_index }
			if = {
				limit = {
					check_variable = { var = famine_migration_cohort_row_id value = famine_migration_cohort_id_request compare = equals }
					check_variable = { var = famine_migration_cohort_row_status value = constant:famine_migration_cohort_status.active compare = equals }
				}
				remove_from_array = { array = global.famine_migration_cohort_destinations index = famine_migration_cohort_destination_bind_index }
				add_to_array = { array = global.famine_migration_cohort_destinations value = THIS index = famine_migration_cohort_destination_bind_index }
				remove_from_array = { array = global.famine_migration_cohort_hosts index = famine_migration_cohort_destination_bind_index }
				add_to_array = { array = global.famine_migration_cohort_hosts value = THIS index = famine_migration_cohort_destination_bind_index }
				remove_from_array = { array = global.famine_migration_cohort_statuses index = famine_migration_cohort_destination_bind_index }
				add_to_array = { array = global.famine_migration_cohort_statuses value = constant:famine_migration_cohort_status.destination_bound index = famine_migration_cohort_destination_bind_index }
				save_event_target_as = famine_migration_route_destination
				if = {
					limit = {
						has_variable = famine_migration_current_cohort_id
						NOT = { check_variable = { var = famine_migration_current_cohort_id value = famine_migration_cohort_row_id compare = equals } }
					}
					clear_variable = famine_migration_current_cohort_id
					set_state_flag = famine_migration_cohort_selection_ambiguous
				}
				else_if = {
					limit = { NOT = { has_state_flag = famine_migration_cohort_selection_ambiguous } }
					set_variable = { famine_migration_current_cohort_id = famine_migration_cohort_row_id }
				}
				# Country selection remains tied to the persisted owner row. The
				# host-state selection above is authoritative after this destination
				# bind, so a different destination owner is never assigned the row.
				set_temp_variable = { famine_migration_cohort_destination_bind_result = constant:famine_migration_route_result.valid }
				set_temp_variable = { famine_migration_cohort_destination_bind_break = constant:famine_migration_runtime.one }
			}
		}
	}
	set_temp_variable = { famine_migration_cohort_destination_food_safe_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_destination_reception_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_id_request = constant:famine_migration_runtime.zero }
}

# Called inside the actual new destination state after a safe voluntary
# resettlement route has been proven. This accepts only a live destination_bound
# row, keeps its survivor amount and original owner/origin unchanged, and moves
# the aligned host/destination identity once. No population transaction occurs.
famine_migration_rebind_cohort_destination_safe = {
	famine_migration_initialize_runtime = yes
	set_temp_variable = { famine_migration_cohort_resettlement_rebind_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_cohort_resettlement_rebind_break = constant:famine_migration_runtime.zero }
	if = {
		limit = { famine_migration_cohort_resettlement_rebind_request_is_valid = yes }
		set_temp_variable = { famine_migration_cohort_resettlement_rebind_index = global.famine_migration_cohort_ids^num }
		while_loop_effect = {
			break = famine_migration_cohort_resettlement_rebind_break
			limit = {
				check_variable = { var = famine_migration_cohort_resettlement_rebind_index value = constant:famine_migration_runtime.zero compare = greater_than }
				check_variable = { var = famine_migration_cohort_resettlement_rebind_break value = constant:famine_migration_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { famine_migration_cohort_resettlement_rebind_index = constant:famine_migration_runtime.array_index_increment }
			set_temp_variable = { famine_migration_cohort_row_id = global.famine_migration_cohort_ids^famine_migration_cohort_resettlement_rebind_index }
			set_temp_variable = { famine_migration_cohort_row_status = global.famine_migration_cohort_statuses^famine_migration_cohort_resettlement_rebind_index }
			if = {
				limit = {
					check_variable = { var = famine_migration_cohort_row_id value = famine_migration_cohort_id_request compare = equals }
					check_variable = { var = famine_migration_cohort_row_status value = constant:famine_migration_cohort_status.destination_bound compare = equals }
				}
				remove_from_array = { array = global.famine_migration_cohort_destinations index = famine_migration_cohort_resettlement_rebind_index }
				add_to_array = { array = global.famine_migration_cohort_destinations value = THIS index = famine_migration_cohort_resettlement_rebind_index }
				remove_from_array = { array = global.famine_migration_cohort_hosts index = famine_migration_cohort_resettlement_rebind_index }
				add_to_array = { array = global.famine_migration_cohort_hosts value = THIS index = famine_migration_cohort_resettlement_rebind_index }
				remove_from_array = { array = global.famine_migration_cohort_statuses index = famine_migration_cohort_resettlement_rebind_index }
				add_to_array = { array = global.famine_migration_cohort_statuses value = constant:famine_migration_cohort_status.destination_bound index = famine_migration_cohort_resettlement_rebind_index }
				event_target:famine_migration_cohort_host = {
					if = {
						limit = {
							has_variable = famine_migration_current_cohort_id
							check_variable = { var = famine_migration_current_cohort_id value = famine_migration_cohort_row_id compare = equals }
						}
						clear_variable = famine_migration_current_cohort_id
						clr_state_flag = famine_migration_cohort_selection_ambiguous
					}
				}
				save_event_target_as = famine_migration_route_destination
				if = {
					limit = {
						has_variable = famine_migration_current_cohort_id
						NOT = { check_variable = { var = famine_migration_current_cohort_id value = famine_migration_cohort_row_id compare = equals } }
					}
					clear_variable = famine_migration_current_cohort_id
					set_state_flag = famine_migration_cohort_selection_ambiguous
				}
				else_if = {
					limit = { NOT = { has_state_flag = famine_migration_cohort_selection_ambiguous } }
					set_variable = { famine_migration_current_cohort_id = famine_migration_cohort_row_id }
				}
				set_temp_variable = { famine_migration_cohort_resettlement_rebind_result = constant:famine_migration_route_result.valid }
				set_temp_variable = { famine_migration_cohort_resettlement_rebind_break = constant:famine_migration_runtime.one }
			}
		}
	}
	set_temp_variable = { famine_migration_cohort_resettlement_rebind_request_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_resettlement_food_safe_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_resettlement_reception_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_resettlement_route_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_resettlement_actor_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_id_request = constant:famine_migration_runtime.zero }
}

# Resolve a row's original state and current host. A destination target is
# returned only for a bound row; unsafe binding remains visible to the forced
# return contract but is rejected by the normal destination safety trigger.
famine_migration_resolve_cohort_origin = {
	famine_migration_initialize_runtime = yes
	set_temp_variable = { famine_migration_cohort_origin_resolution_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_cohort_destination_resolution_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_cohort_resolve_break = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_cohort_resolved_amount = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_cohort_resolved_source = constant:famine_migration_pressure_source.unknown }
	set_temp_variable = { famine_migration_cohort_resolved_status = constant:famine_migration_cohort_status.none }
	if = {
		limit = { NOT = { has_variable = famine_migration_cohort_id_request } }
		if = {
			limit = {
				famine_migration_state_is_valid = yes
				NOT = { has_state_flag = famine_migration_cohort_selection_ambiguous }
				has_variable = famine_migration_current_cohort_id
			}
			set_temp_variable = { famine_migration_cohort_id_request = famine_migration_current_cohort_id }
		}
		else_if = {
			limit = {
				famine_migration_country_is_valid = yes
				NOT = { has_country_flag = famine_migration_country_cohort_selection_ambiguous }
				has_variable = famine_migration_current_country_cohort_id
			}
			set_temp_variable = { famine_migration_cohort_id_request = famine_migration_current_country_cohort_id }
		}
	}
	if = {
		limit = { check_variable = { var = famine_migration_cohort_id_request value = constant:famine_migration_runtime.zero compare = greater_than } }
		set_temp_variable = { famine_migration_cohort_resolve_index = global.famine_migration_cohort_ids^num }
		while_loop_effect = {
			break = famine_migration_cohort_resolve_break
			limit = {
				check_variable = { var = famine_migration_cohort_resolve_index value = constant:famine_migration_runtime.zero compare = greater_than }
				check_variable = { var = famine_migration_cohort_resolve_break value = constant:famine_migration_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { famine_migration_cohort_resolve_index = constant:famine_migration_runtime.array_index_increment }
			set_temp_variable = { famine_migration_cohort_row_id = global.famine_migration_cohort_ids^famine_migration_cohort_resolve_index }
			if = {
				limit = { check_variable = { var = famine_migration_cohort_row_id value = famine_migration_cohort_id_request compare = equals } }
				set_temp_variable = { famine_migration_cohort_resolved_amount = global.famine_migration_cohort_amounts^famine_migration_cohort_resolve_index }
				set_temp_variable = { famine_migration_cohort_resolved_source = global.famine_migration_cohort_sources^famine_migration_cohort_resolve_index }
				set_temp_variable = { famine_migration_cohort_resolved_status = global.famine_migration_cohort_statuses^famine_migration_cohort_resolve_index }
				var:global.famine_migration_cohort_origins^famine_migration_cohort_resolve_index = {
					save_event_target_as = famine_migration_cohort_origin
				}
				var:global.famine_migration_cohort_hosts^famine_migration_cohort_resolve_index = {
					save_event_target_as = famine_migration_cohort_host
				}
				var:global.famine_migration_cohort_owners^famine_migration_cohort_resolve_index = {
					save_event_target_as = famine_migration_cohort_owner
				}
				set_temp_variable = { famine_migration_cohort_origin_resolution_result = constant:famine_migration_route_result.valid }
				if = {
					limit = {
						OR = {
							check_variable = { var = famine_migration_cohort_resolved_status value = constant:famine_migration_cohort_status.destination_bound compare = equals }
							check_variable = { var = famine_migration_cohort_resolved_status value = constant:famine_migration_cohort_status.destination_bound_unsafe compare = equals }
						}
					}
					var:global.famine_migration_cohort_destinations^famine_migration_cohort_resolve_index = {
						save_event_target_as = famine_migration_cohort_destination
					}
					set_temp_variable = { famine_migration_cohort_destination_resolution_result = constant:famine_migration_route_result.valid }
				}
				set_temp_variable = { famine_migration_cohort_resolve_break = constant:famine_migration_runtime.one }
			}
		}
	}
}

# Remove a completed or invalid row by id. All aligned arrays are removed at
# the same index, keeping identity/amount/owner conservation intact.
famine_migration_cleanup_cohort_record = {
	famine_migration_initialize_runtime = yes

# COUNTRY scope. Runtime retirement removes only live scheduler and selection
# state. Historical integration, resettlement, and return ledgers remain
# available to achievements, map modes, and documentation consumers.
famine_migration_retire_inactive_displacement_country = {
	if = {
		limit = {
			famine_migration_country_is_valid = yes
			has_country_flag = famine_migration_decision_dormant
			NOT = { has_variable = famine_migration_current_country_cohort_id }
			OR = {
				NOT = { has_variable = famine_migration_reception_load }
				NOT = { check_variable = { var = famine_migration_reception_load value = constant:famine_migration_population.zero compare = greater_than } }
			}
			NOT = {
				any_owned_state = {
					OR = {
						has_state_flag = famine_migration_food_security_active
						has_state_flag = famine_migration_displacement_active
					}
				}
			}
		}
		if = {
			limit = {
				is_in_array = {
					array = global.famine_migration_active_displacement_countries
					value = THIS
				}
			}
			remove_from_array = { array = global.famine_migration_active_displacement_countries value = THIS }
			set_variable = { global.famine_migration_active_displacement_countries_count = global.famine_migration_active_displacement_countries^num }
		}
		clr_country_flag = famine_migration_displacement_country_active
		clr_country_flag = famine_migration_decision_emerging
		clr_country_flag = famine_migration_decision_active
		clr_country_flag = famine_migration_decision_resolution
		clear_variable = famine_migration_reception_capacity
		clear_variable = famine_migration_reception_load
		clear_variable = famine_migration_current_country_cohort_id
		clear_variable = famine_migration_cohort_resolution_transaction_proven
		clr_country_flag = famine_migration_country_cohort_selection_ambiguous
	}
}

# STATE scope. Reveal the shared migration category only after the accepted
# repeated/large thresholds are met. This projection is invoked only for an
# already registered state; it never scans the world.
famine_migration_refresh_decision_phase_from_state = {
	if = {
		limit = {
			famine_migration_state_is_valid = yes
			OR = {
				check_variable = { var = famine_migration_food_exposure_days value = constant:famine_migration_decision_threshold.category_food_exposure_days compare = greater_than_or_equals }
				check_variable = { var = famine_migration_food_incident_count value = constant:famine_migration_decision_threshold.category_food_incidents compare = greater_than_or_equals }
				check_variable = { var = famine_migration_flight_pressure value = constant:famine_migration_decision_threshold.category_flight_people compare = greater_than_or_equals }
				check_variable = { var = famine_migration_trapped_population value = constant:famine_migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }
				check_variable = { var = famine_migration_state_reception_load value = constant:famine_migration_decision_threshold.category_reception_load compare = greater_than_or_equals }
			}
		}
		OWNER = {
			famine_migration_register_active_displacement_country = yes
			if = {
				limit = {
					NOT = { has_country_flag = famine_migration_decision_active }
					NOT = { has_country_flag = famine_migration_decision_resolution }
				}
				set_country_flag = famine_migration_decision_emerging
				clr_country_flag = famine_migration_decision_dormant
			}
		}
	}
}

# Generic pressure adapter. The caller supplies one positive amount and may
# route it to food, flight, or both with the two apply flags. Missing apply
# flags default to food only; missing requests fail closed.
famine_migration_apply_pressure_request = {
	famine_migration_initialize_food_state = yes
	set_temp_variable = { famine_migration_pressure_request_result = constant:famine_migration_runtime.zero }
	if = {
		limit = {
			NOT = { has_variable = famine_migration_pressure_request_apply_food }
		}
		set_temp_variable = { famine_migration_pressure_request_apply_food = constant:famine_migration_runtime.one }
	}
	if = {
		limit = { NOT = { has_variable = famine_migration_pressure_request_apply_flight } }
		set_temp_variable = { famine_migration_pressure_request_apply_flight = constant:famine_migration_runtime.zero }
	}
	if = {
		limit = { famine_migration_pressure_request_is_valid = yes }
		if = {
			limit = { check_variable = { var = famine_migration_pressure_request_apply_food > constant:famine_migration_runtime.zero  } }
			add_to_variable = { famine_migration_food_pressure = famine_migration_pressure_request_amount }
			famine_migration_register_active_food_state = yes
		}
		if = {
			limit = { check_variable = { var = famine_migration_pressure_request_apply_flight > constant:famine_migration_runtime.zero  } }
			add_to_variable = { famine_migration_flight_pressure = famine_migration_pressure_request_amount }
			add_to_variable = { famine_migration_state_flight_population = famine_migration_pressure_request_amount }
			famine_migration_register_active_displacement_state = yes
			OWNER = { famine_migration_register_active_displacement_country = yes }
			famine_migration_refresh_decision_phase_from_state = yes
		}
		set_variable = { famine_migration_last_pressure_source = famine_migration_pressure_request_source }
		set_temp_variable = { famine_migration_pressure_request_result = constant:famine_migration_runtime.one }
	}
	set_temp_variable = { famine_migration_pressure_request_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_pressure_request_amount = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_pressure_request_source = constant:famine_migration_pressure_source.unknown }
	set_temp_variable = { famine_migration_pressure_request_actor_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_pressure_request_apply_food = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_pressure_request_apply_flight = constant:famine_migration_runtime.zero }
}

# Exact state-to-state civilian transfer. Origin debit is performed exactly
# once by the existing no-recruitable-manpower helper. Route deaths are a
# ledger slice of that debit; only survivors are credited at the destination.
famine_migration_transfer_civilians_exact = {
	set_temp_variable = { famine_migration_transfer_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_transfer_origin_country_id = OWNER.id }
	set_temp_variable = { famine_migration_transfer_actual_origin_debit = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_route_deaths = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_survivor_credit = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_conservation_ledger = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_destination_actual_credit = constant:famine_migration_population.zero }
	if = {
		limit = { NOT = { has_variable = famine_migration_transfer_minimum_origin_remaining } }
		set_temp_variable = { famine_migration_transfer_minimum_origin_remaining = constant:famine_migration_population.zero }
	}
	if = {
		limit = { NOT = { has_variable = famine_migration_transfer_route_deaths_requested } }
		set_temp_variable = { famine_migration_transfer_route_deaths_requested = constant:famine_migration_population.zero }
	}
	if = {
		limit = { NOT = { has_variable = famine_migration_transfer_log_deaths } }
		set_temp_variable = { famine_migration_transfer_log_deaths = constant:famine_migration_runtime.zero }
	}
	if = {
		limit = { NOT = { has_variable = famine_migration_transfer_route_death_reason } }
		set_temp_variable = { famine_migration_transfer_route_death_reason = constant:chaos_meter_deaths_reason.unknown }
	}
	if = {
		limit = { NOT = { has_variable = famine_migration_transfer_target_country } }
		set_temp_variable = { famine_migration_transfer_target_country = OWNER }
	}
	if = {
		limit = { NOT = { has_variable = famine_migration_transfer_has_target_country } }
		set_temp_variable = { famine_migration_transfer_has_target_country = constant:famine_migration_runtime.zero }
	}
	famine_migration_resolve_route = yes
	if = {
		limit = { check_variable = { var = famine_migration_route_resolution_result value = constant:famine_migration_route_result.valid   compare = equals } }
		set_temp_variable = { famine_migration_transfer_origin_before = state_population_k }
		multiply_temp_variable = { famine_migration_transfer_origin_before = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = famine_migration_transfer_origin_before
		set_temp_variable = { famine_migration_transfer_available = famine_migration_transfer_origin_before }
		subtract_from_temp_variable = {
			famine_migration_transfer_available = famine_migration_transfer_minimum_origin_remaining
		}
		clamp_temp_variable = {
			var = famine_migration_transfer_available
			min = constant:famine_migration_population.zero
		}
		set_temp_variable = { famine_migration_transfer_request = famine_migration_transfer_people }
		clamp_temp_variable = {
			var = famine_migration_transfer_request
			min = constant:famine_migration_population.zero
			max = famine_migration_transfer_available
		}
		round_temp_variable = famine_migration_transfer_request
		if = {
			limit = { check_variable = { var = famine_migration_transfer_request > constant:famine_migration_population.zero  } }
			set_temp_variable = { state_population_transaction_loss = famine_migration_transfer_request }
			set_temp_variable = { state_population_transaction_contract_supplied = constant:famine_migration_runtime.one }
			apply_state_population_loss_without_recruitable_manpower_gain = yes
			set_temp_variable = { famine_migration_transfer_actual_origin_after = state_population_k }
			multiply_temp_variable = { famine_migration_transfer_actual_origin_after = constant:chaos_meter_deaths.people_per_k }
			round_temp_variable = famine_migration_transfer_actual_origin_after
			set_temp_variable = { famine_migration_transfer_actual_origin_debit = famine_migration_transfer_origin_before }
			subtract_from_temp_variable = { famine_migration_transfer_actual_origin_debit = famine_migration_transfer_actual_origin_after }
			clamp_temp_variable = {
				var = famine_migration_transfer_actual_origin_debit
				min = constant:famine_migration_population.zero
				max = famine_migration_transfer_request
			}
			round_temp_variable = famine_migration_transfer_actual_origin_debit
			set_temp_variable = { famine_migration_transfer_route_deaths = famine_migration_transfer_route_deaths_requested }
			clamp_temp_variable = {
				var = famine_migration_transfer_route_deaths
				min = constant:famine_migration_population.zero
				max = famine_migration_transfer_actual_origin_debit
			}
			round_temp_variable = famine_migration_transfer_route_deaths
			set_temp_variable = { famine_migration_transfer_survivor_request = famine_migration_transfer_actual_origin_debit }
			subtract_from_temp_variable = { famine_migration_transfer_survivor_request = famine_migration_transfer_route_deaths }
			if = {
				limit = { check_variable = { var = famine_migration_transfer_survivor_request > constant:famine_migration_population.zero  } }
				set_temp_variable = { famine_migration_transfer_destination_request = famine_migration_transfer_survivor_request }
				event_target:famine_migration_route_destination = {
					famine_migration_apply_destination_credit = yes
				}
			}
			if = {
				limit = {
					check_variable = { var = famine_migration_transfer_route_deaths > constant:famine_migration_population.zero  }
					check_variable = { var = famine_migration_transfer_log_deaths > constant:famine_migration_runtime.zero  }
				}
				set_temp_variable = { chaos_deaths_change = famine_migration_transfer_route_deaths }
				set_temp_variable = { chaos_deaths_reason = famine_migration_transfer_route_death_reason }
				set_temp_variable = { chaos_deaths_is_civilian = constant:famine_migration_runtime.one }
				set_temp_variable = { chaos_deaths_apply_state_pop = constant:famine_migration_runtime.zero }
				set_temp_variable = { chaos_deaths_record_state_ledger = constant:famine_migration_runtime.one }
				set_temp_variable = { chaos_deaths_target_country = famine_migration_transfer_target_country }
				set_temp_variable = { chaos_deaths_has_target_country = famine_migration_transfer_has_target_country }
				chaos_meter_register_deaths = yes
			}
			set_temp_variable = { famine_migration_transfer_survivor_credit = famine_migration_transfer_destination_actual_credit }
			set_temp_variable = { famine_migration_transfer_conservation_ledger = famine_migration_transfer_actual_origin_debit }
			subtract_from_temp_variable = { famine_migration_transfer_conservation_ledger = famine_migration_transfer_route_deaths }
			subtract_from_temp_variable = { famine_migration_transfer_conservation_ledger = famine_migration_transfer_survivor_credit }
			if = {
				limit = { check_variable = { var = famine_migration_transfer_conservation_ledger value = constant:famine_migration_population.zero compare = greater_than } }
				set_temp_variable = { famine_migration_origin_restore_request = famine_migration_transfer_conservation_ledger }
				famine_migration_restore_origin_population_residual = yes
				subtract_from_temp_variable = { famine_migration_transfer_actual_origin_debit = famine_migration_origin_restore_actual }
				set_temp_variable = { famine_migration_transfer_conservation_ledger = famine_migration_transfer_actual_origin_debit }
				subtract_from_temp_variable = { famine_migration_transfer_conservation_ledger = famine_migration_transfer_route_deaths }
				subtract_from_temp_variable = { famine_migration_transfer_conservation_ledger = famine_migration_transfer_survivor_credit }
			}
			add_to_variable = { global.famine_migration_conservation_debit = famine_migration_transfer_actual_origin_debit }
			add_to_variable = { global.famine_migration_conservation_credit = famine_migration_transfer_survivor_credit }
			add_to_variable = { global.famine_migration_conservation_route_deaths = famine_migration_transfer_route_deaths }
			add_to_variable = { global.famine_migration_conservation_residual = famine_migration_transfer_conservation_ledger }
			if = {
				limit = {
					check_variable = { var = famine_migration_transfer_conservation_ledger value = constant:famine_migration_population.zero compare = equals }
					check_variable = { var = famine_migration_transfer_actual_origin_debit value = constant:famine_migration_population.zero compare = greater_than }
				}
				set_temp_variable = { famine_migration_transfer_result = constant:famine_migration_route_result.valid }
				if = {
					limit = {
						has_variable = famine_migration_cohort_id_request
						has_event_target = famine_migration_route_destination
					}
					event_target:famine_migration_route_destination = {
						famine_migration_update_cohort_host_after_transfer = yes
					}
				}
			}
			else = {
				if = {
					limit = {
						check_variable = { var = famine_migration_transfer_actual_origin_debit value = constant:famine_migration_population.zero compare = greater_than }
						OR = {
							check_variable = { var = famine_migration_transfer_conservation_ledger value = constant:famine_migration_population.zero compare = less_than }
							check_variable = { var = famine_migration_transfer_conservation_ledger value = constant:famine_migration_population.zero compare = greater_than }
						}
					}
					OWNER = { famine_migration_achievement_record_transfer_cycle_failure = yes }
				}
			}
		}
	}
	set_temp_variable = { famine_migration_transfer_request_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_transfer_people = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_route_deaths_requested = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_minimum_origin_remaining = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_transfer_route_death_reason = constant:chaos_meter_deaths_reason.unknown }
	set_temp_variable = { famine_migration_transfer_log_deaths = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_transfer_target_country = OWNER }
	set_temp_variable = { famine_migration_transfer_has_target_country = constant:famine_migration_runtime.zero }
	if = {
		limit = { has_event_target = famine_migration_route_destination }
		event_target:famine_migration_route_destination = {
			clear_variable = famine_migration_forced_return_destination_proven
		}
	}
	clear_variable = famine_migration_forced_return_destination_proven
}

# Destination-side positive state add with the same manpower reconciliation
# used by the existing exact debit helper. This prevents an arrival from
# creating a second recruitable-manpower credit in owner or controller pools.
famine_migration_apply_destination_credit = {
	set_temp_variable = { famine_migration_transfer_destination_actual_credit = constant:famine_migration_population.zero }
	if = {
		limit = {
			famine_migration_destination_is_valid = yes
			check_variable = { var = famine_migration_transfer_destination_request > constant:famine_migration_population.zero  }
		}
		set_temp_variable = { famine_migration_destination_before = state_population_k }
		multiply_temp_variable = { famine_migration_destination_before = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = famine_migration_destination_before
		set_temp_variable = { famine_migration_destination_owner_before_k = constant:famine_migration_population.zero }
		set_temp_variable = { famine_migration_destination_controller_before_k = constant:famine_migration_population.zero }
		OWNER = { set_temp_variable = { famine_migration_destination_owner_before_k = manpower_k } }
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = { set_temp_variable = { famine_migration_destination_controller_before_k = manpower_k } }
		}
		add_manpower = var:famine_migration_transfer_destination_request
		set_temp_variable = { famine_migration_transfer_destination_actual_credit = state_population_k }
		multiply_temp_variable = { famine_migration_transfer_destination_actual_credit = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = famine_migration_transfer_destination_actual_credit
		subtract_from_temp_variable = { famine_migration_transfer_destination_actual_credit = famine_migration_destination_before }
		clamp_temp_variable = {
			var = famine_migration_transfer_destination_actual_credit
			min = constant:famine_migration_population.zero
			max = famine_migration_transfer_destination_request
		}
		round_temp_variable = famine_migration_transfer_destination_actual_credit
		OWNER = {
			set_temp_variable = { famine_migration_destination_owner_gain = manpower_k }
			subtract_from_temp_variable = { famine_migration_destination_owner_gain = famine_migration_destination_owner_before_k }
			multiply_temp_variable = { famine_migration_destination_owner_gain = constant:chaos_meter_deaths.people_per_k }
			round_temp_variable = famine_migration_destination_owner_gain
			clamp_temp_variable = {
				var = famine_migration_destination_owner_gain
				min = constant:famine_migration_population.zero
				max = famine_migration_transfer_destination_actual_credit
			}
			if = {
				limit = { check_variable = { var = famine_migration_destination_owner_gain > constant:famine_migration_population.zero  } }
				set_temp_variable = { famine_migration_destination_owner_correction = famine_migration_destination_owner_gain }
				multiply_temp_variable = { famine_migration_destination_owner_correction = constant:chaos_meter_deaths.negative_one }
				add_manpower = var:famine_migration_destination_owner_correction
			}
		}
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = {
				set_temp_variable = { famine_migration_destination_controller_gain = manpower_k }
				subtract_from_temp_variable = { famine_migration_destination_controller_gain = famine_migration_destination_controller_before_k }
				multiply_temp_variable = { famine_migration_destination_controller_gain = constant:chaos_meter_deaths.people_per_k }
				round_temp_variable = famine_migration_destination_controller_gain
				clamp_temp_variable = {
					var = famine_migration_destination_controller_gain
					min = constant:famine_migration_population.zero
					max = famine_migration_transfer_destination_actual_credit
				}
				if = {
					limit = { check_variable = { var = famine_migration_destination_controller_gain > constant:famine_migration_population.zero  } }
					set_temp_variable = { famine_migration_destination_controller_correction = famine_migration_destination_controller_gain }
					multiply_temp_variable = { famine_migration_destination_controller_correction = constant:chaos_meter_deaths.negative_one }
					add_manpower = var:famine_migration_destination_controller_correction
				}
			}
		}
	}
}

# Origin-state rollback used only when the destination API credited fewer
# survivors than the actual origin debit supplied. It restores the uncredited
# residual and removes any incidental owner/controller recruitable-manpower
# gain, leaving the transaction equation debit = deaths + survivor credit.
famine_migration_restore_origin_population_residual = {
	set_temp_variable = { famine_migration_origin_restore_actual = constant:famine_migration_population.zero }
	if = {
		limit = {
			famine_migration_state_is_valid = yes
			check_variable = { var = famine_migration_origin_restore_request value = constant:famine_migration_population.zero compare = greater_than }
		}
		set_temp_variable = { famine_migration_origin_restore_before = state_population_k }
		multiply_temp_variable = { famine_migration_origin_restore_before = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = famine_migration_origin_restore_before
		set_temp_variable = { famine_migration_origin_restore_owner_before_k = constant:famine_migration_population.zero }
		set_temp_variable = { famine_migration_origin_restore_controller_before_k = constant:famine_migration_population.zero }
		OWNER = { set_temp_variable = { famine_migration_origin_restore_owner_before_k = manpower_k } }
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = { set_temp_variable = { famine_migration_origin_restore_controller_before_k = manpower_k } }
		}
		add_manpower = var:famine_migration_origin_restore_request
		set_temp_variable = { famine_migration_origin_restore_actual = state_population_k }
		multiply_temp_variable = { famine_migration_origin_restore_actual = constant:chaos_meter_deaths.people_per_k }
		round_temp_variable = famine_migration_origin_restore_actual
		subtract_from_temp_variable = { famine_migration_origin_restore_actual = famine_migration_origin_restore_before }
		clamp_temp_variable = {
			var = famine_migration_origin_restore_actual
			min = constant:famine_migration_population.zero
			max = famine_migration_origin_restore_request
		}
		round_temp_variable = famine_migration_origin_restore_actual
		OWNER = {
			set_temp_variable = { famine_migration_origin_restore_owner_gain = manpower_k }
			subtract_from_temp_variable = { famine_migration_origin_restore_owner_gain = famine_migration_origin_restore_owner_before_k }
			multiply_temp_variable = { famine_migration_origin_restore_owner_gain = constant:chaos_meter_deaths.people_per_k }
			round_temp_variable = famine_migration_origin_restore_owner_gain
			clamp_temp_variable = { var = famine_migration_origin_restore_owner_gain min = constant:famine_migration_population.zero max = famine_migration_origin_restore_actual }
			if = {
				limit = { check_variable = { var = famine_migration_origin_restore_owner_gain value = constant:famine_migration_population.zero compare = greater_than } }
				set_temp_variable = { famine_migration_origin_restore_owner_correction = famine_migration_origin_restore_owner_gain }
				multiply_temp_variable = { famine_migration_origin_restore_owner_correction = constant:chaos_meter_deaths.negative_one }
				add_manpower = var:famine_migration_origin_restore_owner_correction
			}
		}
		if = {
			limit = { NOT = { is_owned_and_controlled_by = OWNER } }
			CONTROLLER = {
				set_temp_variable = { famine_migration_origin_restore_controller_gain = manpower_k }
				subtract_from_temp_variable = { famine_migration_origin_restore_controller_gain = famine_migration_origin_restore_controller_before_k }
				multiply_temp_variable = { famine_migration_origin_restore_controller_gain = constant:chaos_meter_deaths.people_per_k }
				round_temp_variable = famine_migration_origin_restore_controller_gain
				clamp_temp_variable = { var = famine_migration_origin_restore_controller_gain min = constant:famine_migration_population.zero max = famine_migration_origin_restore_actual }

# Called in the actual destination state scope after an exact transfer. The
# persisted row follows the surviving cohort; original state and ownership
# never change. No row or target means no update.
famine_migration_update_cohort_host_after_transfer = {
	set_temp_variable = { famine_migration_cohort_host_update_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_cohort_host_update_break = constant:famine_migration_runtime.zero }
	if = {
		limit = {
			famine_migration_state_is_valid = yes
			check_variable = { var = famine_migration_cohort_id_request value = constant:famine_migration_runtime.zero compare = greater_than }
		}
		set_temp_variable = { famine_migration_cohort_host_update_index = global.famine_migration_cohort_ids^num }
		while_loop_effect = {
			break = famine_migration_cohort_host_update_break
			limit = {
				check_variable = { var = famine_migration_cohort_host_update_index value = constant:famine_migration_runtime.zero compare = greater_than }
				check_variable = { var = famine_migration_cohort_host_update_break value = constant:famine_migration_runtime.zero compare = equals }
			}
			subtract_from_temp_variable = { famine_migration_cohort_host_update_index = constant:famine_migration_runtime.array_index_increment }
			set_temp_variable = { famine_migration_cohort_row_id = global.famine_migration_cohort_ids^famine_migration_cohort_host_update_index }
			if = {
				limit = { check_variable = { var = famine_migration_cohort_row_id value = famine_migration_cohort_id_request compare = equals } }
				if = {
					limit = { check_variable = { var = famine_migration_transfer_survivor_credit value = constant:famine_migration_population.zero compare = greater_than } }
					remove_from_array = { array = global.famine_migration_cohort_hosts index = famine_migration_cohort_host_update_index }
					add_to_array = { array = global.famine_migration_cohort_hosts value = THIS index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_amounts index = famine_migration_cohort_host_update_index }
					add_to_array = { array = global.famine_migration_cohort_amounts value = famine_migration_transfer_survivor_credit index = famine_migration_cohort_host_update_index }
					if = {
						limit = {
							has_variable = famine_migration_current_cohort_id
							NOT = { check_variable = { var = famine_migration_current_cohort_id value = famine_migration_cohort_row_id compare = equals } }
						}
						clear_variable = famine_migration_current_cohort_id
						set_state_flag = famine_migration_cohort_selection_ambiguous
					}
					else_if = {
						limit = { NOT = { has_state_flag = famine_migration_cohort_selection_ambiguous } }
						set_variable = { famine_migration_current_cohort_id = famine_migration_cohort_row_id }
					}
				}
				else = {
					remove_from_array = { array = global.famine_migration_cohort_ids index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_origins index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_hosts index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_destinations index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_owners index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_amounts index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_sources index = famine_migration_cohort_host_update_index }
					remove_from_array = { array = global.famine_migration_cohort_statuses index = famine_migration_cohort_host_update_index }
					set_variable = { global.famine_migration_cohort_count = global.famine_migration_cohort_ids^num }
					clear_variable = famine_migration_current_cohort_id
					clr_state_flag = famine_migration_cohort_selection_ambiguous
				}
				set_temp_variable = { famine_migration_cohort_host_update_result = constant:famine_migration_route_result.valid }
				set_temp_variable = { famine_migration_cohort_host_update_break = constant:famine_migration_runtime.one }
			}
		}
	}
}

# State scope. Rebuilds reception/overcrowding projection flags from the
# authoritative state load and the owner-country capacity/load contract.

		limit = { famine_migration_state_is_valid = yes }

			limit = { check_variable = { var = famine_migration_state_reception_load value = constant:famine_migration_population.zero compare = greater_than } }

			if = {
				limit = {
					has_variable = famine_migration_reception_capacity
					has_variable = famine_migration_reception_load

			}
			else = { event_target:famine_migration_reception_context_state = { clr_state_flag = famine_migration_overcrowded_context_active } }
		}
		famine_migration_refresh_dynamic_modifiers = yes
	}
}

# State scope. Applies one positive, already-proven actual survivor credit or
# debit to both the state-local load and its owner-country aggregate. Debit
# requests fail closed unless both ledgers contain the full requested amount;
# this makes the helper conservation-safe and prevents double counting. The
# caller must invoke it once per exact transfer transaction.
famine_migration_apply_reception_delta = {
	set_temp_variable = { famine_migration_reception_delta_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_reception_delta_applied = constant:famine_migration_population.zero }
	if = {
		limit = { famine_migration_reception_delta_request_is_valid = yes }
		if = {
			limit = { check_variable = { var = famine_migration_reception_delta_mode value = constant:famine_migration_reception_delta.credit compare = equals } }
			add_to_variable = { famine_migration_state_reception_load = famine_migration_reception_delta_amount }
			OWNER = {
				add_to_variable = { famine_migration_reception_load = famine_migration_reception_delta_amount }
				famine_migration_register_active_displacement_country = yes
			}
		}
		else_if = {
			limit = { check_variable = { var = famine_migration_reception_delta_mode value = constant:famine_migration_reception_delta.debit compare = equals } }
			subtract_from_variable = { famine_migration_state_reception_load = famine_migration_reception_delta_amount }
			OWNER = { subtract_from_variable = { famine_migration_reception_load = famine_migration_reception_delta_amount } }
		}
		set_temp_variable = { famine_migration_reception_delta_applied = famine_migration_reception_delta_amount }
		famine_migration_refresh_reception_context = yes
		famine_migration_refresh_decision_phase_from_state = yes
		set_temp_variable = { famine_migration_reception_delta_result = constant:famine_migration_route_result.valid }
	}
	set_temp_variable = { famine_migration_reception_delta_request_proven = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_reception_delta_amount = constant:famine_migration_population.zero }
	set_temp_variable = { famine_migration_reception_delta_mode = constant:famine_migration_runtime.zero }
}

# Bounded country-side selector maintenance. A country may resolve a cohort
# without passing a numeric id only when exactly one row is owned by it; more
# than one row sets an explicit ambiguity flag and fails closed.
famine_migration_reconcile_country_cohort_selection = {
	famine_migration_initialize_runtime = yes
	save_event_target_as = famine_migration_reconcile_country
	set_temp_variable = { famine_migration_reconcile_count = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_reconcile_candidate = constant:famine_migration_runtime.zero }
	if = {
		limit = {
			famine_migration_country_is_valid = yes
			NOT = { has_country_flag = famine_migration_country_cohort_selection_ambiguous }
		}
		set_temp_variable = { famine_migration_reconcile_index = global.famine_migration_cohort_ids^num }
		while_loop_effect = {
			limit = { check_variable = { var = famine_migration_reconcile_index value = constant:famine_migration_runtime.zero compare = greater_than } }
			subtract_from_temp_variable = { famine_migration_reconcile_index = constant:famine_migration_runtime.array_index_increment }
			set_temp_variable = { famine_migration_reconcile_owner_match = constant:famine_migration_runtime.zero }
			var:global.famine_migration_cohort_owners^famine_migration_reconcile_index = {
				if = { limit = { tag = event_target:famine_migration_reconcile_country } set_temp_variable = { famine_migration_reconcile_owner_match = constant:famine_migration_runtime.one } }
			}
			if = {
				limit = { check_variable = { var = famine_migration_reconcile_owner_match value = constant:famine_migration_runtime.one compare = equals } }
				add_to_temp_variable = { famine_migration_reconcile_count = constant:famine_migration_runtime.one }
				set_temp_variable = { famine_migration_reconcile_candidate = global.famine_migration_cohort_ids^famine_migration_reconcile_index }
			}
		}
		if = {
			limit = { check_variable = { var = famine_migration_reconcile_count value = constant:famine_migration_runtime.one compare = equals } }
			set_variable = { famine_migration_current_country_cohort_id = famine_migration_reconcile_candidate }
		}
		if = {
			limit = { check_variable = { var = famine_migration_reconcile_count value = constant:famine_migration_runtime.one compare = greater_than } }
			clear_variable = famine_migration_current_country_cohort_id
			set_country_flag = famine_migration_country_cohort_selection_ambiguous
		}
		if = {
			limit = { check_variable = { var = famine_migration_reconcile_count value = constant:famine_migration_runtime.zero compare = equals } }
			clear_variable = famine_migration_current_country_cohort_id
			clr_country_flag = famine_migration_country_cohort_selection_ambiguous
		}
	}
}

# COUNTRY scope. The registered-country scheduler owns this bounded
# achievement projection. It sums only aligned, nonterminal cohort rows whose
# persisted owner tag is THIS country; it never discovers states or countries.
# The achievement tracker consumes the sum to start/continue the major-cohort
# duration window. A zero after an already-started major window is explicit
# resolution evidence only when a valid transaction/cleanup marker was supplied
# by the movement consumer.
famine_migration_reconcile_country_cohort_achievement = {
	if = {
		limit = { famine_migration_country_is_valid = yes }
		save_event_target_as = famine_migration_achievement_owner_country
		set_temp_variable = { famine_migration_achievement_cohort_total = constant:famine_migration_population.zero }
		set_temp_variable = { famine_migration_achievement_cohort_index = global.famine_migration_cohort_ids^num }
		while_loop_effect = {
			limit = { check_variable = { var = famine_migration_achievement_cohort_index value = constant:famine_migration_runtime.zero compare = greater_than } }
			subtract_from_temp_variable = { famine_migration_achievement_cohort_index = constant:famine_migration_runtime.array_index_increment }
			set_temp_variable = { famine_migration_achievement_cohort_row_status = global.famine_migration_cohort_statuses^famine_migration_achievement_cohort_index }
			set_temp_variable = { famine_migration_achievement_cohort_owner_match = constant:famine_migration_runtime.zero }
			var:global.famine_migration_cohort_owners^famine_migration_achievement_cohort_index = {
				if = {
					limit = { tag = event_target:famine_migration_achievement_owner_country }
					set_temp_variable = { famine_migration_achievement_cohort_owner_match = constant:famine_migration_runtime.one }
				}
			}
			if = {
				limit = {
					check_variable = { var = famine_migration_achievement_cohort_owner_match value = constant:famine_migration_runtime.one compare = equals }
					OR = {
						check_variable = { var = famine_migration_achievement_cohort_row_status value = constant:famine_migration_cohort_status.active compare = equals }
						check_variable = { var = famine_migration_achievement_cohort_row_status value = constant:famine_migration_cohort_status.destination_bound compare = equals }
						check_variable = { var = famine_migration_achievement_cohort_row_status value = constant:famine_migration_cohort_status.destination_bound_unsafe compare = equals }
					}
				}
				add_to_temp_variable = { famine_migration_achievement_cohort_total = global.famine_migration_cohort_amounts^famine_migration_achievement_cohort_index }
			}
		}
		set_variable = { famine_migration_achievement_cohort_people = famine_migration_achievement_cohort_total }
		famine_migration_achievement_record_major_cohort_duration = yes
		if = {
			limit = {
				has_variable = famine_migration_achievement_major_cohort_start_date
				check_variable = { var = famine_migration_achievement_cohort_total value = constant:famine_migration_population.zero compare = equals }
				check_variable = { var = famine_migration_cohort_resolution_transaction_proven value = constant:famine_migration_runtime.one compare = equals }
			}
			set_country_flag = famine_migration_achievement_all_major_cohorts_resolved
			clear_variable = famine_migration_cohort_resolution_transaction_proven
		}
	}
}

			set_variable = { famine_migration_food_stage_candidate = famine_migration_requested_food_stage }
			set_variable = { famine_migration_food_stage_candidate_since_date = global.date }
		}
		set_temp_variable = { famine_migration_food_stage_candidate_elapsed = global.date }
		subtract_from_temp_variable = { famine_migration_food_stage_candidate_elapsed = famine_migration_food_stage_candidate_since_date }
		set_temp_variable = { famine_migration_food_candidate_days = constant:famine_migration_food_timing.stable_candidate_days }
		if = { limit = { check_variable = { var = famine_migration_requested_food_stage value = constant:famine_migration_food_stage.supply_strain   compare = equals } } set_temp_variable = { famine_migration_food_candidate_days = constant:famine_migration_food_timing.supply_strain_candidate_days } }
		if = { limit = { check_variable = { var = famine_migration_requested_food_stage value = constant:famine_migration_food_stage.acute_shortage   compare = equals } } set_temp_variable = { famine_migration_food_candidate_days = constant:famine_migration_food_timing.acute_shortage_candidate_days } }
		if = { limit = { check_variable = { var = famine_migration_requested_food_stage value = constant:famine_migration_food_stage.famine   compare = equals } } set_temp_variable = { famine_migration_food_candidate_days = constant:famine_migration_food_timing.famine_candidate_days } }
		if = { limit = { check_variable = { var = famine_migration_requested_food_stage value = constant:famine_migration_food_stage.catastrophic_famine   compare = equals } } set_temp_variable = { famine_migration_food_candidate_days = constant:famine_migration_food_timing.catastrophic_candidate_days } }
		if = {
			limit = { check_variable = { var = famine_migration_food_stage_candidate_elapsed value = famine_migration_food_candidate_days compare = greater_than_or_equals } }
			famine_migration_transition_food_stage = yes
		}
	}
	else_if = {
		limit = { check_variable = { var = famine_migration_requested_food_stage value = famine_migration_food_stage compare = less_than } }
		set_temp_variable = { famine_migration_food_recovery_threshold = constant:famine_migration_food_score.zero }
		set_temp_variable = { famine_migration_food_recovery_days = constant:famine_migration_food_timing.stable_recovery_days }
		if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.supply_strain   compare = equals } } set_temp_variable = { famine_migration_food_recovery_threshold = constant:famine_migration_food_score.supply_strain_recovery_threshold } set_temp_variable = { famine_migration_food_recovery_days = constant:famine_migration_food_timing.supply_strain_recovery_days } }
		if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.acute_shortage   compare = equals } } set_temp_variable = { famine_migration_food_recovery_threshold = constant:famine_migration_food_score.acute_shortage_recovery_threshold } set_temp_variable = { famine_migration_food_recovery_days = constant:famine_migration_food_timing.acute_shortage_recovery_days } }
		if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.famine   compare = equals } } set_temp_variable = { famine_migration_food_recovery_threshold = constant:famine_migration_food_score.famine_recovery_threshold } set_temp_variable = { famine_migration_food_recovery_days = constant:famine_migration_food_timing.famine_recovery_days } }
		if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.catastrophic_famine   compare = equals } } set_temp_variable = { famine_migration_food_recovery_threshold = constant:famine_migration_food_score.catastrophic_recovery_threshold } set_temp_variable = { famine_migration_food_recovery_days = constant:famine_migration_food_timing.catastrophic_recovery_days } }
		if = {
			limit = { check_variable = { var = famine_migration_food_score_raw < famine_migration_food_recovery_threshold  } }
			if = { limit = { NOT = { has_variable = famine_migration_food_recovery_since_date } } set_variable = { famine_migration_food_recovery_since_date = global.date } }
			set_temp_variable = { famine_migration_food_recovery_elapsed = global.date }
			subtract_from_temp_variable = { famine_migration_food_recovery_elapsed = famine_migration_food_recovery_since_date }
			if = { limit = { check_variable = { var = famine_migration_food_recovery_elapsed value = famine_migration_food_recovery_days compare = greater_than_or_equals } } famine_migration_transition_food_stage = yes }
		}
		else = { clear_variable = famine_migration_food_recovery_since_date }
	}
	else = {
		clear_variable = famine_migration_food_stage_candidate_since_date
		clear_variable = famine_migration_food_recovery_since_date
	}

	if = {
		limit = { famine_migration_food_stage_requires_mortality = yes }
		set_temp_variable = { famine_migration_food_exposure_elapsed = global.date }
		subtract_from_temp_variable = { famine_migration_food_exposure_elapsed = famine_migration_food_stage_started_date }
		clamp_temp_variable = { var = famine_migration_food_exposure_elapsed min = constant:famine_migration_runtime.zero max = constant:famine_migration_food_timing.maximum_exposure_days }
		set_variable = { famine_migration_food_exposure_days = famine_migration_food_exposure_elapsed }
		famine_migration_refresh_decision_phase_from_state = yes
		famine_migration_apply_famine_mortality = yes
	}
	else = { set_variable = { famine_migration_food_exposure_days = constant:famine_migration_runtime.zero } }
	famine_migration_refresh_dynamic_modifiers = yes
	set_temp_variable = { famine_migration_food_evaluation_result = constant:famine_migration_runtime.one }
	famine_migration_achievement_process_state = yes
	famine_migration_retire_recovered_state = yes
	}
}

# Refreshes the state-owned reserve ledger from live population and logistics.
# Reserve units are thousand-person-days; this effect never changes population
# and preserves the amount and cumulative totals across state retirement.
famine_migration_refresh_food_reserve_capacity = {
	set_temp_variable = { famine_migration_food_reserve_refresh_result = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_food_reserve_capacity_output = constant:famine_migration_food_reserve.zero }

	set_temp_variable = { famine_migration_food_reserve_daily_need_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_logistics_factor_output = constant:famine_migration_food_reserve.zero }

		limit = { famine_migration_state_is_valid = yes }
		famine_migration_initialize_food_state = yes

		add_to_temp_variable = { famine_migration_food_reserve_logistics_factor_calculated = famine_migration_food_reserve_infrastructure }
		set_temp_variable = { famine_migration_food_reserve_transport_penalty = famine_migration_component_transport }
		multiply_temp_variable = { famine_migration_food_reserve_transport_penalty = constant:famine_migration_food_reserve.transport_capacity_penalty }
		subtract_from_temp_variable = { famine_migration_food_reserve_logistics_factor_calculated = famine_migration_food_reserve_transport_penalty }
		set_temp_variable = { famine_migration_food_reserve_production_penalty = famine_migration_component_production }
		multiply_temp_variable = { famine_migration_food_reserve_production_penalty = constant:famine_migration_food_reserve.production_capacity_penalty }
		subtract_from_temp_variable = { famine_migration_food_reserve_logistics_factor_calculated = famine_migration_food_reserve_production_penalty }
		clamp_temp_variable = {
			var = famine_migration_food_reserve_logistics_factor_calculated
			min = constant:famine_migration_food_reserve.logistics_factor_minimum
			max = constant:famine_migration_food_reserve.logistics_factor_maximum
		}
		set_variable = { famine_migration_food_reserve_logistics_factor = famine_migration_food_reserve_logistics_factor_calculated }
		set_variable = { famine_migration_food_reserve_daily_need = state_population_k }
		multiply_variable = { famine_migration_food_reserve_daily_need = constant:famine_migration_food_reserve.daily_need_per_k }
		round_variable = famine_migration_food_reserve_daily_need
		clamp_variable = { var = famine_migration_food_reserve_daily_need min = constant:famine_migration_food_reserve.minimum_units }
		set_temp_variable = { famine_migration_food_reserve_capacity_calculated = famine_migration_food_reserve_daily_need }
		multiply_temp_variable = { famine_migration_food_reserve_capacity_calculated = constant:famine_migration_food_reserve.capacity_days }
		multiply_temp_variable = { famine_migration_food_reserve_capacity_calculated = famine_migration_food_reserve_logistics_factor }
		round_temp_variable = famine_migration_food_reserve_capacity_calculated
		clamp_temp_variable = { var = famine_migration_food_reserve_capacity_calculated min = constant:famine_migration_food_reserve.minimum_units }
		set_variable = { famine_migration_food_reserve_capacity = famine_migration_food_reserve_capacity_calculated }
		set_temp_variable = { famine_migration_food_reserve_target_calculated = famine_migration_food_reserve_daily_need }
		multiply_temp_variable = { famine_migration_food_reserve_target_calculated = constant:famine_migration_food_reserve.target_days }
		multiply_temp_variable = { famine_migration_food_reserve_target_calculated = famine_migration_food_reserve_logistics_factor }
		round_temp_variable = famine_migration_food_reserve_target_calculated
		clamp_temp_variable = {
			var = famine_migration_food_reserve_target_calculated
			min = constant:famine_migration_food_reserve.minimum_units
			max = famine_migration_food_reserve_capacity
		}
		set_variable = { famine_migration_food_reserve_target = famine_migration_food_reserve_target_calculated }
		clamp_variable = { var = famine_migration_food_reserve_amount min = constant:famine_migration_food_reserve.zero }
		if = {
			limit = { NOT = { has_state_flag = famine_migration_food_reserve_initialized } }

				limit = {
					check_variable = { var = famine_migration_food_reserve_initialization_proven value = constant:famine_migration_runtime.zero compare = greater_than }

				clamp_variable = {
					var = famine_migration_food_reserve_amount
					min = constant:famine_migration_food_reserve.zero
					max = famine_migration_food_reserve_capacity
				}
				set_state_flag = famine_migration_food_reserve_initialized
			}
			if = {
				limit = {
					NOT = { has_state_flag = famine_migration_food_reserve_initialized }
					check_variable = { var = famine_migration_food_reserve_amount value = constant:famine_migration_food_reserve.zero compare = greater_than }
				}
				# A positive amount already present in a save is durable evidence;
				# zero is not promoted to an invented starting stockpile.
				set_state_flag = famine_migration_food_reserve_initialized
			}
		}
		set_temp_variable = { famine_migration_food_reserve_capacity_output = famine_migration_food_reserve_capacity }
		set_temp_variable = { famine_migration_food_reserve_target_output = famine_migration_food_reserve_target }

		set_temp_variable = { famine_migration_food_reserve_daily_need_output = famine_migration_food_reserve_daily_need }
		set_temp_variable = { famine_migration_food_reserve_logistics_factor_output = famine_migration_food_reserve_logistics_factor }
		set_temp_variable = { famine_migration_food_reserve_refresh_result = constant:famine_migration_runtime.one }
	}
}

# Applies at most one reserve update per game date. A stable state replenishes
# toward target, while an active shortage depletes stock by stage and need.
famine_migration_update_food_reserve = {
	set_temp_variable = { famine_migration_food_reserve_update_result = constant:famine_migration_runtime.zero }
	set_temp_variable = { famine_migration_food_reserve_replenished_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_depleted_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_amount_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_daily_need_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_relief_output = constant:famine_migration_food_reserve.zero }
	if = {
		limit = { famine_migration_state_is_valid = yes }
		famine_migration_refresh_food_reserve_capacity = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = famine_migration_food_reserve_last_update_date }
					check_variable = { var = famine_migration_food_reserve_last_update_date value = global.date compare = less_than }
				}
			}
			set_variable = { famine_migration_food_reserve_last_update_date = global.date }
			set_variable = { famine_migration_food_reserve_last_replenished = constant:famine_migration_food_reserve.zero }
			set_variable = { famine_migration_food_reserve_last_depleted = constant:famine_migration_food_reserve.zero }
			if = {
				limit = { check_variable = { var = famine_migration_food_reserve_relief value = constant:famine_migration_food_reserve.zero compare = greater_than } }
				subtract_from_variable = { famine_migration_food_reserve_relief = constant:famine_migration_food_reserve.relief_decay_per_day }
				clamp_variable = { var = famine_migration_food_reserve_relief min = constant:famine_migration_food_reserve.zero max = constant:famine_migration_food_reserve.relief_maximum }
			}
			if = {
				limit = { famine_migration_food_stage_is_active = yes }
				set_temp_variable = { famine_migration_food_reserve_depletion_share = constant:famine_migration_food_reserve.zero }
				if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.supply_strain compare = equals } } set_temp_variable = { famine_migration_food_reserve_depletion_share = constant:famine_migration_food_reserve.supply_strain_depletion_share } }
				if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.acute_shortage compare = equals } } set_temp_variable = { famine_migration_food_reserve_depletion_share = constant:famine_migration_food_reserve.acute_shortage_depletion_share } }
				if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.famine compare = equals } } set_temp_variable = { famine_migration_food_reserve_depletion_share = constant:famine_migration_food_reserve.famine_depletion_share } }
				if = { limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.catastrophic_famine compare = equals } } set_temp_variable = { famine_migration_food_reserve_depletion_share = constant:famine_migration_food_reserve.catastrophic_depletion_share } }
				set_temp_variable = { famine_migration_food_reserve_depletion_factor = famine_migration_component_need }
				divide_temp_variable = { famine_migration_food_reserve_depletion_factor = constant:famine_migration_food_component.component_scale }
				clamp_temp_variable = {
					var = famine_migration_food_reserve_depletion_factor
					min = constant:famine_migration_food_reserve.depletion_factor_minimum
					max = constant:famine_migration_food_weight.component_factor_base
				}
				set_temp_variable = { famine_migration_food_reserve_depletion_amount = famine_migration_food_reserve_daily_need }
				multiply_temp_variable = { famine_migration_food_reserve_depletion_amount = famine_migration_food_reserve_depletion_share }
				multiply_temp_variable = { famine_migration_food_reserve_depletion_amount = famine_migration_food_reserve_depletion_factor }
				round_temp_variable = famine_migration_food_reserve_depletion_amount
				clamp_temp_variable = {
					var = famine_migration_food_reserve_depletion_amount
					min = constant:famine_migration_food_reserve.zero
					max = famine_migration_food_reserve_amount
				}
				if = {
					limit = { check_variable = { var = famine_migration_food_reserve_depletion_amount value = constant:famine_migration_food_reserve.zero compare = greater_than } }
					subtract_from_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_depletion_amount }
					set_variable = { famine_migration_food_reserve_last_depleted = famine_migration_food_reserve_depletion_amount }
					add_to_variable = { famine_migration_food_reserve_total_depleted = famine_migration_food_reserve_depletion_amount }
				}
			}
			else = {
				if = {
					limit = {
						has_state_flag = famine_migration_food_reserve_initialized
						check_variable = { var = famine_migration_food_reserve_amount value = famine_migration_food_reserve_target compare = less_than }
					}
					set_temp_variable = { famine_migration_food_reserve_production_factor = constant:famine_migration_food_weight.component_factor_base }
					set_temp_variable = { famine_migration_food_reserve_production_pressure = famine_migration_component_production }
					divide_temp_variable = { famine_migration_food_reserve_production_pressure = constant:famine_migration_food_component.component_scale }
					subtract_from_temp_variable = { famine_migration_food_reserve_production_factor = famine_migration_food_reserve_production_pressure }
					clamp_temp_variable = {
						var = famine_migration_food_reserve_production_factor
						min = constant:famine_migration_food_reserve.replenishment_factor_minimum
						max = constant:famine_migration_food_weight.component_factor_base
					}
					set_temp_variable = { famine_migration_food_reserve_transport_factor = constant:famine_migration_food_weight.component_factor_base }
					set_temp_variable = { famine_migration_food_reserve_transport_pressure = famine_migration_component_transport }
					divide_temp_variable = { famine_migration_food_reserve_transport_pressure = constant:famine_migration_food_component.component_scale }
					subtract_from_temp_variable = { famine_migration_food_reserve_transport_factor = famine_migration_food_reserve_transport_pressure }
					clamp_temp_variable = {
						var = famine_migration_food_reserve_transport_factor
						min = constant:famine_migration_food_reserve.replenishment_factor_minimum
						max = constant:famine_migration_food_weight.component_factor_base
					}
					set_temp_variable = { famine_migration_food_reserve_replenishment_amount = famine_migration_food_reserve_daily_need }
					multiply_temp_variable = { famine_migration_food_reserve_replenishment_amount = constant:famine_migration_food_reserve.replenishment_per_k_per_day }
					multiply_temp_variable = { famine_migration_food_reserve_replenishment_amount = famine_migration_food_reserve_production_factor }
					multiply_temp_variable = { famine_migration_food_reserve_replenishment_amount = famine_migration_food_reserve_transport_factor }
					multiply_temp_variable = { famine_migration_food_reserve_replenishment_amount = famine_migration_food_reserve_logistics_factor }
					round_temp_variable = famine_migration_food_reserve_replenishment_amount
					set_temp_variable = { famine_migration_food_reserve_available_capacity = famine_migration_food_reserve_target }
					subtract_from_temp_variable = { famine_migration_food_reserve_available_capacity = famine_migration_food_reserve_amount }
					clamp_temp_variable = { var = famine_migration_food_reserve_available_capacity min = constant:famine_migration_food_reserve.zero }
					clamp_temp_variable = {
						var = famine_migration_food_reserve_replenishment_amount
						min = constant:famine_migration_food_reserve.zero
						max = famine_migration_food_reserve_available_capacity
					}
					if = {
						limit = { check_variable = { var = famine_migration_food_reserve_replenishment_amount value = constant:famine_migration_food_reserve.zero compare = greater_than } }
						add_to_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_replenishment_amount }
						set_variable = { famine_migration_food_reserve_last_replenished = famine_migration_food_reserve_replenishment_amount }
						add_to_variable = { famine_migration_food_reserve_total_replenished = famine_migration_food_reserve_replenishment_amount }
					}
				}
			}
		}
		set_temp_variable = { famine_migration_food_reserve_replenished_output = famine_migration_food_reserve_last_replenished }
		set_temp_variable = { famine_migration_food_reserve_depleted_output = famine_migration_food_reserve_last_depleted }
		set_temp_variable = { famine_migration_food_reserve_amount_output = famine_migration_food_reserve_amount }
		set_temp_variable = { famine_migration_food_reserve_daily_need_output = famine_migration_food_reserve_daily_need }
		set_temp_variable = { famine_migration_food_reserve_relief_output = famine_migration_food_reserve_relief }
		set_temp_variable = { famine_migration_food_reserve_update_result = constant:famine_migration_runtime.one }
	}
}

# Consumes actual stock and grants a bounded transient relief projection. The
# amount output is the exact debit; an over-sized request is safely partial.
famine_migration_consume_food_reserve_for_relief = {
	set_temp_variable = { famine_migration_food_reserve_consume_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_food_reserve_consumed_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_relief_granted_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_remaining_output = constant:famine_migration_food_reserve.zero }
	if = {
		limit = { famine_migration_food_reserve_release_request_is_valid = yes }
		famine_migration_refresh_food_reserve_capacity = yes
		set_temp_variable = { famine_migration_food_reserve_consume_amount = famine_migration_food_reserve_release_amount }
		clamp_temp_variable = {
			var = famine_migration_food_reserve_consume_amount
			min = constant:famine_migration_food_reserve.zero
			max = famine_migration_food_reserve_amount
		}
		round_temp_variable = famine_migration_food_reserve_consume_amount
		if = {
			limit = { check_variable = { var = famine_migration_food_reserve_consume_amount value = constant:famine_migration_food_reserve.zero compare = greater_than } }
			subtract_from_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_consume_amount }
			add_to_variable = { famine_migration_food_reserve_total_consumed = famine_migration_food_reserve_consume_amount }
			set_variable = { famine_migration_food_reserve_last_consumed = famine_migration_food_reserve_consume_amount }
			set_temp_variable = { famine_migration_food_reserve_relief_available = constant:famine_migration_food_reserve.relief_maximum }
			subtract_from_temp_variable = { famine_migration_food_reserve_relief_available = famine_migration_food_reserve_relief }

			}
			add_to_variable = { famine_migration_food_reserve_relief = famine_migration_food_reserve_relief_granted }
			clamp_variable = { var = famine_migration_food_reserve_relief min = constant:famine_migration_food_reserve.zero max = constant:famine_migration_food_reserve.relief_maximum }

	clear_variable = famine_migration_food_reserve_release_actor_proven

# Adds external/imported stock only up to the current state capacity. The
# accepted amount is returned so a caller cannot silently duplicate excess.
famine_migration_add_food_reserves = {
	set_temp_variable = { famine_migration_food_reserve_add_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_food_reserve_added_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_remaining_output = constant:famine_migration_food_reserve.zero }

		limit = { famine_migration_food_reserve_import_request_is_valid = yes }

		subtract_from_temp_variable = { famine_migration_food_reserve_import_available = famine_migration_food_reserve_amount }

			var = famine_migration_food_reserve_import_amount_applied

			max = famine_migration_food_reserve_import_available
		}
		round_temp_variable = famine_migration_food_reserve_import_amount_applied
		if = {
			limit = { check_variable = { var = famine_migration_food_reserve_import_amount_applied value = constant:famine_migration_food_reserve.zero compare = greater_than } }
			add_to_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_import_amount_applied }
			add_to_variable = { famine_migration_food_reserve_total_imported = famine_migration_food_reserve_import_amount_applied }

			set_temp_variable = { famine_migration_food_reserve_add_result = constant:famine_migration_route_result.valid }
		}
		set_temp_variable = { famine_migration_food_reserve_remaining_output = famine_migration_food_reserve_amount }
		set_temp_variable = { famine_migration_food_reserve_capacity_output = famine_migration_food_reserve_capacity }

	clear_variable = famine_migration_food_reserve_import_request_proven
	clear_variable = famine_migration_food_reserve_import_source_proven
	clear_variable = famine_migration_food_reserve_import_actor_proven
}

# Exact state-to-state reserve transfer. The accepted amount is debited once
# from the source and credited once to the destination; a non-zero residual
# triggers rollback before either cumulative transfer ledger is advanced.
famine_migration_transfer_food_reserves = {
	set_temp_variable = { famine_migration_food_reserve_transfer_result = constant:famine_migration_route_result.invalid }
	set_temp_variable = { famine_migration_food_reserve_transfer_source_debit_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_transfer_destination_credit_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_transfer_remaining_output = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_destination_available = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_destination_credit = constant:famine_migration_food_reserve.zero }
	set_temp_variable = { famine_migration_food_reserve_transfer_same_state = constant:famine_migration_runtime.zero }
	if = {
		limit = { famine_migration_food_reserve_transfer_request_is_valid = yes }
		famine_migration_refresh_food_reserve_capacity = yes
		save_event_target_as = famine_migration_food_reserve_source
		event_target:famine_migration_food_reserve_destination = {
			famine_migration_refresh_food_reserve_capacity = yes
			set_temp_variable = { famine_migration_food_reserve_destination_available = famine_migration_food_reserve_capacity }
			subtract_from_temp_variable = { famine_migration_food_reserve_destination_available = famine_migration_food_reserve_amount }
			clamp_temp_variable = { var = famine_migration_food_reserve_destination_available min = constant:famine_migration_food_reserve.zero }
			if = {
				limit = { state = event_target:famine_migration_food_reserve_source }
				set_temp_variable = { famine_migration_food_reserve_transfer_same_state = constant:famine_migration_runtime.one }
			}
		}
		if = {
			limit = { check_variable = { var = famine_migration_food_reserve_transfer_same_state value = constant:famine_migration_runtime.zero compare = equals } }
			set_temp_variable = { famine_migration_food_reserve_transfer_applied = famine_migration_food_reserve_transfer_amount }
			clamp_temp_variable = {
				var = famine_migration_food_reserve_transfer_applied
				min = constant:famine_migration_food_reserve.zero
				max = famine_migration_food_reserve_amount
			}
			clamp_temp_variable = {
				var = famine_migration_food_reserve_transfer_applied
				min = constant:famine_migration_food_reserve.zero
				max = famine_migration_food_reserve_destination_available
			}
			round_temp_variable = famine_migration_food_reserve_transfer_applied
			if = {
				limit = { check_variable = { var = famine_migration_food_reserve_transfer_applied value = constant:famine_migration_food_reserve.zero compare = greater_than } }
				set_temp_variable = { famine_migration_food_reserve_transfer_source_before = famine_migration_food_reserve_amount }
				subtract_from_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_transfer_applied }
				set_temp_variable = { famine_migration_food_reserve_transfer_source_debit = famine_migration_food_reserve_transfer_source_before }
				subtract_from_temp_variable = { famine_migration_food_reserve_transfer_source_debit = famine_migration_food_reserve_amount }
				event_target:famine_migration_food_reserve_destination = {
					set_temp_variable = { famine_migration_food_reserve_destination_before = famine_migration_food_reserve_amount }
					add_to_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_transfer_applied }
					set_temp_variable = { famine_migration_food_reserve_destination_credit = famine_migration_food_reserve_amount }
					subtract_from_temp_variable = { famine_migration_food_reserve_destination_credit = famine_migration_food_reserve_destination_before }
				}
				set_temp_variable = { famine_migration_food_reserve_transfer_residual = famine_migration_food_reserve_transfer_source_debit }
				subtract_from_temp_variable = { famine_migration_food_reserve_transfer_residual = famine_migration_food_reserve_destination_credit }
				if = {
					limit = { check_variable = { var = famine_migration_food_reserve_transfer_residual value = constant:famine_migration_food_reserve.zero compare = greater_than } }
					add_to_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_transfer_residual }
				}
				if = {
					limit = { check_variable = { var = famine_migration_food_reserve_transfer_residual value = constant:famine_migration_food_reserve.zero compare = less_than } }
					set_temp_variable = { famine_migration_food_reserve_transfer_excess = famine_migration_food_reserve_transfer_residual }
					multiply_temp_variable = { famine_migration_food_reserve_transfer_excess = constant:chaos_meter_deaths.negative_one }
					event_target:famine_migration_food_reserve_destination = {
						subtract_from_variable = { famine_migration_food_reserve_amount = famine_migration_food_reserve_transfer_excess }
					}
					subtract_from_temp_variable = { famine_migration_food_reserve_destination_credit = famine_migration_food_reserve_transfer_excess }
				}
				set_temp_variable = { famine_migration_food_reserve_transfer_source_debit = famine_migration_food_reserve_transfer_source_before }
				subtract_from_temp_variable = { famine_migration_food_reserve_transfer_source_debit = famine_migration_food_reserve_amount }
				set_temp_variable = { famine_migration_food_reserve_transfer_residual = famine_migration_food_reserve_transfer_source_debit }
				subtract_from_temp_variable = { famine_migration_food_reserve_transfer_residual = famine_migration_food_reserve_destination_credit }
				if = {
					limit = { check_variable = { var = famine_migration_food_reserve_transfer_residual value = constant:famine_migration_food_reserve.zero compare = equals } }
					add_to_variable = { famine_migration_food_reserve_total_transfer_out = famine_migration_food_reserve_transfer_source_debit }
					set_variable = { famine_migration_food_reserve_last_transfer_out = famine_migration_food_reserve_transfer_source_debit }
					event_target:famine_migration_food_reserve_destination = {
						add_to_variable = { famine_migration_food_reserve_total_transfer_in = famine_migration_food_reserve_destination_credit }
						set_variable = { famine_migration_food_reserve_last_transfer_in = famine_migration_food_reserve_destination_credit }
					}
					set_temp_variable = { famine_migration_food_reserve_transfer_source_debit_output = famine_migration_food_reserve_transfer_source_debit }
					set_temp_variable = { famine_migration_food_reserve_transfer_destination_credit_output = famine_migration_food_reserve_destination_credit }
					set_temp_variable = { famine_migration_food_reserve_transfer_result = constant:famine_migration_route_result.valid }
				}
			}
		}
		set_temp_variable = { famine_migration_food_reserve_transfer_remaining_output = famine_migration_food_reserve_amount }
	}
	clear_variable = famine_migration_food_reserve_transfer_amount
	clear_variable = famine_migration_food_reserve_transfer_request_proven
	clear_variable = famine_migration_food_reserve_transfer_route_proven
	clear_variable = famine_migration_food_reserve_transfer_actor_proven
}

# One mortality pulse is one call to the shared exact state-loss helper. The
# helper applies the protected floor, records one famine reason, and returns
# the actual applied result; this effect never performs a second debit.
famine_migration_apply_famine_mortality = {
	set_temp_variable = { famine_migration_mortality_result = constant:famine_migration_population.zero }
	if = {
		limit = {
			famine_migration_food_stage_requires_mortality = yes
			check_variable = { var = famine_migration_food_next_mortality_date value = global.date compare = less_than_or_equals }
		}
		famine_migration_initialize_food_state = yes
		set_temp_variable = { famine_migration_mortality_minimum_exposure = constant:famine_migration_mortality.acute_shortage_minimum_exposure_days }
		set_temp_variable = { famine_migration_mortality_interval = constant:famine_migration_mortality.acute_shortage_interval_days }
		set_temp_variable = { famine_migration_mortality_rate = constant:famine_migration_mortality.acute_shortage_rate }
		if = {
			limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.famine   compare = equals } }
			set_temp_variable = { famine_migration_mortality_minimum_exposure = constant:famine_migration_mortality.famine_minimum_exposure_days }
			set_temp_variable = { famine_migration_mortality_interval = constant:famine_migration_mortality.famine_interval_days }
			set_temp_variable = { famine_migration_mortality_rate = constant:famine_migration_mortality.famine_rate }
		}
		if = {
			limit = { check_variable = { var = famine_migration_food_stage value = constant:famine_migration_food_stage.catastrophic_famine   compare = equals } }
			set_temp_variable = { famine_migration_mortality_minimum_exposure = constant:famine_migration_mortality.catastrophic_minimum_exposure_days }
			set_temp_variable = { famine_migration_mortality_interval = constant:famine_migration_mortality.catastrophic_interval_days }
			set_temp_variable = { famine_migration_mortality_rate = constant:famine_migration_mortality.catastrophic_rate }
		}
		if = {
			limit = { check_variable = { var = famine_migration_food_exposure_days value = famine_migration_mortality_minimum_exposure compare = greater_than_or_equals } }
			set_temp_variable = { famine_migration_mortality_population = state_population_k }
			multiply_temp_variable = { famine_migration_mortality_population = constant:chaos_meter_deaths.people_per_k }
			round_temp_variable = famine_migration_mortality_population
			set_temp_variable = { famine_migration_mortality_floor = famine_migration_mortality_population }
			multiply_temp_variable = { famine_migration_mortality_floor = constant:famine_migration_mortality.protected_floor_share }
			round_temp_variable = famine_migration_mortality_floor
			if = {
				limit = { check_variable = { var = famine_migration_mortality_floor < constant:famine_migration_mortality.protected_floor_people  } }
				set_temp_variable = { famine_migration_mortality_floor = constant:famine_migration_mortality.protected_floor_people }
			}
			clamp_temp_variable = {
				var = famine_migration_mortality_floor
				min = constant:famine_migration_population.zero
				max = famine_migration_mortality_population
			}
			set_temp_variable = { famine_migration_mortality_available = famine_migration_mortality_population }
			subtract_from_temp_variable = { famine_migration_mortality_available = famine_migration_mortality_floor }
			clamp_temp_variable = { var = famine_migration_mortality_available min = constant:famine_migration_population.zero }

			set_temp_variable = { famine_migration_mortality_exposure_factor = famine_migration_food_exposure_days }
			multiply_temp_variable = { famine_migration_mortality_exposure_factor = constant:famine_migration_mortality.exposure_multiplier_per_day }
			add_to_temp_variable = { famine_migration_mortality_exposure_factor = constant:famine_migration_mortality.minimum_exposure_multiplier }
			clamp_temp_variable = {
				var = famine_migration_mortality_exposure_factor
				min = constant:famine_migration_mortality.minimum_exposure_multiplier
				max = constant:famine_migration_mortality.maximum_exposure_multiplier
			}

			set_temp_variable = { famine_migration_mortality_vulnerability_factor = constant:famine_migration_food_weight.component_factor_base }
		set_temp_variable = { famine_migration_mortality_component_factor = famine_migration_component_vulnerability }
		divide_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_food_component.component_scale }
		add_to_temp_variable = { famine_migration_mortality_vulnerability_factor = famine_migration_mortality_component_factor }
			set_temp_variable = { famine_migration_mortality_access_factor = constant:famine_migration_food_weight.component_factor_base }
			set_temp_variable = { famine_migration_mortality_component_factor = famine_migration_component_transport }
			multiply_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_mortality.access_transport_share }
			set_temp_variable = { famine_migration_mortality_need_factor = famine_migration_component_need }
			multiply_temp_variable = { famine_migration_mortality_need_factor = constant:famine_migration_mortality.access_need_share }
			add_to_temp_variable = { famine_migration_mortality_component_factor = famine_migration_mortality_need_factor }
			divide_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_food_component.component_scale }
			add_to_temp_variable = { famine_migration_mortality_access_factor = famine_migration_mortality_component_factor }
			set_temp_variable = { famine_migration_mortality_policy_factor = constant:famine_migration_food_weight.component_factor_base }
			set_temp_variable = { famine_migration_mortality_component_factor = famine_migration_component_extraction }
			divide_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_food_component.component_scale }
			add_to_temp_variable = { famine_migration_mortality_policy_factor = famine_migration_mortality_component_factor }
			set_temp_variable = { famine_migration_mortality_environment_factor = constant:famine_migration_food_weight.component_factor_base }
			set_temp_variable = { famine_migration_mortality_component_factor = famine_migration_component_environment }
			divide_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_food_component.component_scale }
			add_to_temp_variable = { famine_migration_mortality_environment_factor = famine_migration_mortality_component_factor }
			set_temp_variable = { famine_migration_mortality_governance_factor = constant:famine_migration_food_weight.component_factor_base }
			set_temp_variable = { famine_migration_mortality_component_factor = famine_migration_component_governance }
			divide_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_food_component.component_scale }
			add_to_temp_variable = { famine_migration_mortality_governance_factor = famine_migration_mortality_component_factor }
			set_temp_variable = { famine_migration_mortality_relief_factor = constant:famine_migration_food_weight.component_factor_base }
			set_temp_variable = { famine_migration_mortality_component_factor = famine_migration_component_relief }
			divide_temp_variable = { famine_migration_mortality_component_factor = constant:famine_migration_food_component.component_scale }
			subtract_from_temp_variable = { famine_migration_mortality_relief_factor = famine_migration_mortality_component_factor }
			clamp_temp_variable = {
				var = famine_migration_mortality_relief_factor
				min = constant:famine_migration_mortality.minimum_relief_factor
				max = constant:famine_migration_mortality.maximum_relief_factor
			}

			multiply_temp_variable = { famine_migration_mortality_request = famine_migration_mortality_policy_factor }
			multiply_temp_variable = { famine_migration_mortality_request = famine_migration_mortality_environment_factor }
			multiply_temp_variable = { famine_migration_mortality_request = famine_migration_mortality_governance_factor }
			multiply_temp_variable = { famine_migration_mortality_request = famine_migration_mortality_relief_factor }
			clamp_temp_variable = {
				var = famine_migration_mortality_request
				min = constant:famine_migration_population.zero
				max = famine_migration_mortality_available
			}
			round_temp_variable = famine_migration_mortality_request
			if = {
				limit = { check_variable = { var = famine_migration_mortality_request value = constant:famine_migration_mortality.minimum_pulse_people compare = greater_than_or_equals } }
				set_temp_variable = { state_civilian_population_loss_requested = famine_migration_mortality_request }
				set_temp_variable = { state_civilian_population_loss_minimum_remaining = famine_migration_mortality_floor }
				set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.famine }
				set_temp_variable = { state_civilian_population_loss_log_deaths = constant:chaos_meter_deaths.one }

				set_temp_variable = { state_civilian_population_loss_has_target_country = constant:chaos_meter_deaths.one }
				set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.one }

		famine_migration_reconcile_country_cohort_achievement = yes
		if = {
			limit = {
				has_country_flag = famine_migration_decision_resolution
				NOT = { has_variable = famine_migration_current_country_cohort_id }
				OR = {
					NOT = { has_variable = famine_migration_reception_load }
					NOT = { check_variable = { var = famine_migration_reception_load value = constant:famine_migration_population.zero compare = greater_than } }
				}
				NOT = {
					any_owned_state = {
						OR = {
							has_state_flag = famine_migration_food_security_active
							has_state_flag = famine_migration_displacement_active
						}
					}
				}
			}
			famine_migration_achievement_record_durable_outcome = yes
			set_country_flag = famine_migration_decision_dormant
			clr_country_flag = famine_migration_decision_resolution
		}
		famine_migration_retire_inactive_displacement_country = yes
	}
	else = {
		set_variable = { famine_migration_cleanup_requested = constant:famine_migration_runtime.one }
		famine_migration_cleanup_country_registration = yes
	}
}
