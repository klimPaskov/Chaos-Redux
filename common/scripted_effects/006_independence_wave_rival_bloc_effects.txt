# ============================================================================
# CHAOS REDUX - EVENT 006 RIVAL BLOC EFFECTS
# ============================================================================
# The post-expulsion rival bloc is a separate, generation-safe contract.  It
# shares the Event 006 network and former-host ledgers, but never reuses the
# main league member arrays.  Every public action has an explicit caller,
# value transaction, resource cost in the decision layer, and cleanup path.
# ============================================================================

independence_wave_rival_bloc_rebuild_regions = {
	clear_array = global.independence_wave_rival_bloc_region_entries
	set_variable = { global.independence_wave_rival_bloc_region_count = constant:independence_wave_rival_bloc.minimum }
	set_temp_variable = { independence_wave_rival_bloc_region_index = constant:independence_wave_plan.first_array_index }
	while_loop_effect = {
		limit = { check_variable = { var = independence_wave_rival_bloc_region_index value = global.independence_wave_rival_bloc_member_region_entries^num compare = less_than } }
		set_temp_variable = { independence_wave_rival_bloc_region_value = global.independence_wave_rival_bloc_member_region_entries^independence_wave_rival_bloc_region_index }
		if = {
			limit = { NOT = { is_in_array = { array = global.independence_wave_rival_bloc_region_entries value = independence_wave_rival_bloc_region_value } } }
			add_to_array = { array = global.independence_wave_rival_bloc_region_entries value = independence_wave_rival_bloc_region_value }
			add_to_variable = { global.independence_wave_rival_bloc_region_count = constant:independence_wave_plan.loop_increment }
		}
		add_to_temp_variable = { independence_wave_rival_bloc_region_index = constant:independence_wave_plan.loop_increment }
	}
}

independence_wave_rival_bloc_recalculate_confidence = {
	set_variable = { global.independence_wave_rival_bloc_member_count = global.independence_wave_rival_bloc_member_country_entries^num }
	set_temp_variable = { independence_wave_rival_bloc_confidence_total = constant:independence_wave_rival_bloc.minimum }
	set_temp_variable = { independence_wave_rival_bloc_confidence_index = constant:independence_wave_plan.first_array_index }
	while_loop_effect = {
		limit = { check_variable = { var = independence_wave_rival_bloc_confidence_index value = global.independence_wave_rival_bloc_member_confidence_entries^num compare = less_than } }
		add_to_temp_variable = { independence_wave_rival_bloc_confidence_total = global.independence_wave_rival_bloc_member_confidence_entries^independence_wave_rival_bloc_confidence_index }
		add_to_temp_variable = { independence_wave_rival_bloc_confidence_index = constant:independence_wave_plan.loop_increment }
	}
	if = {
		limit = { check_variable = { var = global.independence_wave_rival_bloc_member_count value = constant:independence_wave_rival_bloc.minimum compare = greater_than } }
		divide_temp_variable = { var = independence_wave_rival_bloc_confidence_total value = global.independence_wave_rival_bloc_member_count if_zero = 0 }
		set_variable = { global.independence_wave_rival_bloc_member_confidence = independence_wave_rival_bloc_confidence_total }
	}
	else = { set_variable = { global.independence_wave_rival_bloc_member_confidence = constant:independence_wave_rival_bloc.minimum } }
	independence_wave_rival_bloc_clamp_values = yes
}

# Reverse-prunes dead tags, reused tags with a new origin generation, stale
# contract rows, and countries that have lost the rival-member flag.
independence_wave_rival_bloc_reconcile_registry = {
	independence_wave_rival_bloc_initialize_runtime = yes
	if = {
		limit = { independence_wave_rival_bloc_member_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_rival_bloc_row = global.independence_wave_rival_bloc_member_country_entries^num }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_rival_bloc_row value = constant:independence_wave_rival_bloc.minimum compare = greater_than } }
			subtract_from_temp_variable = { independence_wave_rival_bloc_row = constant:independence_wave_plan.loop_increment }
			set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_plan.loop_increment }
			set_temp_variable = { independence_wave_rival_bloc_row_generation = global.independence_wave_rival_bloc_member_generation_entries^independence_wave_rival_bloc_row }
			set_temp_variable = { independence_wave_rival_bloc_row_contract_generation = global.independence_wave_rival_bloc_member_contract_generation_entries^independence_wave_rival_bloc_row }
			if = {
				limit = { NOT = { check_variable = { var = independence_wave_rival_bloc_row_contract_generation value = global.independence_wave_rival_bloc_contract_generation compare = equals } } }
				set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum }
			}
			var:global.independence_wave_rival_bloc_member_country_entries^independence_wave_rival_bloc_row = {
				if = { limit = { NOT = { exists = yes } } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum } } }
				if = { limit = { exists = yes NOT = { is_independence_wave_active_country = yes } } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum } } }
				if = { limit = { exists = yes NOT = { has_country_flag = independence_wave_network_member } } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum } } }
			if = { limit = { exists = yes has_country_flag = independence_wave_client_route_locked NOT = { check_variable = { global.independence_wave_rival_bloc_route = constant:independence_wave_rival_bloc_route.patron_balanced } } } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum } } }
				if = { limit = { exists = yes NOT = { has_country_flag = independence_wave_rival_bloc_member } } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum } } }
				if = { limit = { exists = yes NOT = { check_variable = { var = independence_wave_generation_id value = ROOT.independence_wave_rival_bloc_row_generation compare = equals } } } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_valid = constant:independence_wave_value.minimum } } }
			}
			if = { limit = { check_variable = { var = independence_wave_rival_bloc_row_valid value = constant:independence_wave_value.minimum compare = equals } } independence_wave_rival_bloc_remove_member_row = yes }
		}
	}
	else = { independence_wave_rival_bloc_clear_member_arrays = yes }
	independence_wave_rival_bloc_recalculate_confidence = yes
	independence_wave_rival_bloc_rebuild_regions = yes
}

independence_wave_rival_bloc_register_member = {
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_country_flag = independence_wave_network_member
			NOT = { has_country_flag = independence_wave_league_member }
			NOT = { has_country_flag = independence_wave_league_founder }
			OR = {
				NOT = { has_country_flag = independence_wave_client_route_locked }
				check_variable = { global.independence_wave_rival_bloc_route = constant:independence_wave_rival_bloc_route.patron_balanced }
			}
			NOT = { has_country_flag = independence_wave_rival_bloc_member }
			has_global_flag = independence_wave_rival_bloc_active
			check_variable = { var = global.independence_wave_rival_bloc_contract_generation value = constant:independence_wave_rival_bloc.minimum compare = greater_than }
			independence_wave_rival_bloc_member_arrays_are_aligned = yes
		}
		set_temp_variable = { independence_wave_rival_bloc_row = constant:independence_wave_plan.invalid_index }
		set_temp_variable = { independence_wave_rival_bloc_index = constant:independence_wave_plan.first_array_index }
		while_loop_effect = {
			limit = {
				check_variable = { var = independence_wave_rival_bloc_row value = constant:independence_wave_plan.invalid_index compare = equals }
				check_variable = { var = independence_wave_rival_bloc_index value = global.independence_wave_rival_bloc_member_country_entries^num compare = less_than }
			}
			var:global.independence_wave_rival_bloc_member_country_entries^independence_wave_rival_bloc_index = {
				if = { limit = { tag = ROOT } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row = independence_wave_rival_bloc_index } } }
			}
			add_to_temp_variable = { independence_wave_rival_bloc_index = constant:independence_wave_plan.loop_increment }
		}
		if = {
			limit = { check_variable = { var = independence_wave_rival_bloc_row value = constant:independence_wave_plan.invalid_index compare = equals } }
			set_temp_variable = { independence_wave_rival_bloc_row = global.independence_wave_rival_bloc_member_country_entries^num }
			add_to_array = { array = global.independence_wave_rival_bloc_member_country_entries value = THIS }
			add_to_array = { array = global.independence_wave_rival_bloc_member_generation_entries value = independence_wave_generation_id }
			add_to_array = { array = global.independence_wave_rival_bloc_member_contract_generation_entries value = global.independence_wave_rival_bloc_contract_generation }
			add_to_array = { array = global.independence_wave_rival_bloc_member_region_entries value = independence_wave_region }
			add_to_array = { array = global.independence_wave_rival_bloc_member_contribution_entries value = constant:independence_wave_rival_bloc.minimum }
			add_to_array = { array = global.independence_wave_rival_bloc_member_confidence_entries value = constant:independence_wave_rival_bloc.member_confidence_start }
			set_country_flag = independence_wave_rival_bloc_member
		}
		independence_wave_rival_bloc_recalculate_confidence = yes
		independence_wave_rival_bloc_rebuild_regions = yes
		if = { limit = { NOT = { has_event_target = independence_wave_rival_bloc_leader_target } } independence_wave_rival_bloc_select_leader = yes }
	}
}

independence_wave_rival_bloc_select_leader = {
	if = {
		limit = { has_event_target = independence_wave_rival_bloc_leader_target }
		event_target:independence_wave_rival_bloc_leader_target = { clr_country_flag = independence_wave_rival_bloc_leader }
	}
	clear_global_event_target = independence_wave_rival_bloc_leader_target
	clear_variable = global.independence_wave_rival_bloc_leader
	set_temp_variable = { independence_wave_rival_bloc_best_confidence = constant:independence_wave_rival_bloc.minimum }
	set_temp_variable = { independence_wave_rival_bloc_best_index = constant:independence_wave_plan.invalid_index }
	set_temp_variable = { independence_wave_rival_bloc_index = constant:independence_wave_plan.first_array_index }
	while_loop_effect = {
		limit = { check_variable = { var = independence_wave_rival_bloc_index value = global.independence_wave_rival_bloc_member_country_entries^num compare = less_than } }
		set_temp_variable = { independence_wave_rival_bloc_candidate_confidence = global.independence_wave_rival_bloc_member_confidence_entries^independence_wave_rival_bloc_index }
			if = {
				limit = {
				OR = {
					check_variable = { var = independence_wave_rival_bloc_candidate_confidence value = independence_wave_rival_bloc_best_confidence compare = greater_than }
					check_variable = { var = independence_wave_rival_bloc_best_index value = constant:independence_wave_plan.invalid_index compare = equals }
				}
			}
			var:global.independence_wave_rival_bloc_member_country_entries^independence_wave_rival_bloc_index = { save_event_target_as = independence_wave_rival_bloc_candidate_target }
			set_temp_variable = { independence_wave_rival_bloc_best_confidence = independence_wave_rival_bloc_candidate_confidence }
			set_temp_variable = { independence_wave_rival_bloc_best_index = independence_wave_rival_bloc_index }
		}
		add_to_temp_variable = { independence_wave_rival_bloc_index = constant:independence_wave_plan.loop_increment }
	}
	if = {
		limit = { has_event_target = independence_wave_rival_bloc_candidate_target }
		event_target:independence_wave_rival_bloc_candidate_target = {
			set_country_flag = independence_wave_rival_bloc_leader
			save_global_event_target_as = independence_wave_rival_bloc_leader_target
			set_variable = { global.independence_wave_rival_bloc_leader = THIS }
		}
	}
}

independence_wave_rival_bloc_set_leader = {
	if = {
		limit = { is_independence_wave_rival_bloc_member = yes }
		if = { limit = { has_event_target = independence_wave_rival_bloc_leader_target } event_target:independence_wave_rival_bloc_leader_target = { clr_country_flag = independence_wave_rival_bloc_leader } }
		clear_global_event_target = independence_wave_rival_bloc_leader_target
		set_country_flag = independence_wave_rival_bloc_leader
		set_variable = { global.independence_wave_rival_bloc_leader = THIS }
		save_global_event_target_as = independence_wave_rival_bloc_leader_target
	}
}

independence_wave_rival_bloc_unregister_member = {
	if = {
		limit = { has_country_flag = independence_wave_rival_bloc_member }
		save_event_target_as = independence_wave_rival_bloc_departing_member
		set_temp_variable = { independence_wave_rival_bloc_index = global.independence_wave_rival_bloc_member_country_entries^num }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_rival_bloc_index value = constant:independence_wave_rival_bloc.minimum compare = greater_than } }
			subtract_from_temp_variable = { independence_wave_rival_bloc_index = constant:independence_wave_plan.loop_increment }
			set_temp_variable = { independence_wave_rival_bloc_row_match = constant:independence_wave_value.minimum }
			var:global.independence_wave_rival_bloc_member_country_entries^independence_wave_rival_bloc_index = {
				if = { limit = { tag = event_target:independence_wave_rival_bloc_departing_member } ROOT = { set_temp_variable = { independence_wave_rival_bloc_row_match = constant:independence_wave_plan.loop_increment } } }
			}
			if = { limit = { check_variable = { var = independence_wave_rival_bloc_row_match value = constant:independence_wave_plan.loop_increment compare = equals } } independence_wave_rival_bloc_remove_member_row = yes }
		}
		set_temp_variable = { independence_wave_rival_bloc_was_leader = constant:independence_wave_value.minimum }
		if = { limit = { has_country_flag = independence_wave_rival_bloc_leader } set_temp_variable = { independence_wave_rival_bloc_was_leader = constant:independence_wave_plan.loop_increment } }
		clr_country_flag = independence_wave_rival_bloc_member
		clr_country_flag = independence_wave_rival_bloc_leader
		clr_country_flag = independence_wave_rival_bloc_host_front
		clr_country_flag = independence_wave_rival_bloc_invitation_pending
		clr_country_flag = independence_wave_rival_bloc_invitation_in_progress
		clear_variable = independence_wave_rival_bloc_inviter
		clear_variable = independence_wave_rival_bloc_invitation_contract_generation
		if = {
			limit = { check_variable = { var = independence_wave_rival_bloc_was_leader value = constant:independence_wave_plan.loop_increment compare = equals } }
			if = {
				limit = { has_event_target = independence_wave_rival_bloc_pending_invitation_target }
				event_target:independence_wave_rival_bloc_pending_invitation_target = {
					clr_country_flag = independence_wave_rival_bloc_invitation_pending
					clear_variable = independence_wave_rival_bloc_inviter
					clear_variable = independence_wave_rival_bloc_invitation_contract_generation
				}
				clr_global_flag = independence_wave_rival_bloc_invitation_open
				clear_global_event_target = independence_wave_rival_bloc_pending_invitation_target
			}
			clear_global_event_target = independence_wave_rival_bloc_leader_target
			clear_variable = global.independence_wave_rival_bloc_leader
			independence_wave_rival_bloc_select_leader = yes
		}
		independence_wave_rival_bloc_recalculate_confidence = yes
		independence_wave_rival_bloc_rebuild_regions = yes
		if = { limit = { check_variable = { global.independence_wave_rival_bloc_member_count = constant:independence_wave_rival_bloc.minimum } } independence_wave_rival_bloc_dissolve_contract = yes }
	}
}

			clr_country_flag = independence_wave_rival_bloc_host_front
			clr_country_flag = independence_wave_rival_bloc_patron_pressure
			clr_country_flag = independence_wave_rival_bloc_invitation_pending
			clear_variable = independence_wave_rival_bloc_inviter
			clear_variable = independence_wave_rival_bloc_invitation_contract_generation
		}
		independence_wave_rival_bloc_clear_member_arrays = yes
		clr_global_flag = independence_wave_rival_bloc_active
		clr_global_flag = independence_wave_rival_bloc_league_rivalry_active
		clr_global_flag = independence_wave_rival_bloc_activation_pending
		clear_variable = global.independence_wave_rival_bloc_contract_generation
		clear_variable = global.independence_wave_rival_bloc_route
		clear_variable = global.independence_wave_rival_bloc_cohesion
		clear_variable = global.independence_wave_rival_bloc_common_cause
		clear_variable = global.independence_wave_rival_bloc_patron_capture
		clear_variable = global.independence_wave_rival_bloc_shared_reserve
		clear_variable = global.independence_wave_rival_bloc_member_confidence
		clear_variable = global.independence_wave_rival_bloc_host_pressure

		clear_variable = global.independence_wave_rival_bloc_last_action_date
		clear_variable = global.independence_wave_rival_bloc_leader
		clear_global_event_target = independence_wave_rival_bloc_leader_target
		if = {
			limit = { check_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.rival_leagues } }
			if = {
				limit = { check_variable = { global.independence_wave_league_member_count value = constant:independence_wave_rival_bloc.minimum compare = greater_than } }
				independence_wave_clear_league_phase_flags = yes
				set_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.formal_league }
				set_global_flag = independence_wave_formal_league_active
			}
			else = {
				independence_wave_clear_league_phase_flags = yes
				set_variable = { global.independence_wave_league_phase = constant:independence_wave_league_phase.informal_network }
			}
		}
	}
}

# Called from origin cleanup even when the origin is not the last bloc member.
independence_wave_rival_bloc_cleanup_for_origin = {
	if = { limit = { has_country_flag = independence_wave_rival_bloc_member } independence_wave_rival_bloc_leave = yes }
	else = {
		if = {
			limit = {
				has_event_target = independence_wave_rival_bloc_pending_invitation_target
				event_target:independence_wave_rival_bloc_pending_invitation_target = { tag = ROOT }
			}
			event_target:independence_wave_rival_bloc_pending_invitation_target = {
				clr_country_flag = independence_wave_rival_bloc_invitation_pending
				clear_variable = independence_wave_rival_bloc_inviter
				clear_variable = independence_wave_rival_bloc_invitation_contract_generation
			}
			clr_global_flag = independence_wave_rival_bloc_invitation_open
			clear_global_event_target = independence_wave_rival_bloc_pending_invitation_target
		}
		clr_country_flag = independence_wave_rival_bloc_invitation_pending
		clear_variable = independence_wave_rival_bloc_inviter
		clear_variable = independence_wave_rival_bloc_invitation_contract_generation
	}
}

# The main league calls this after moving the phase to formal_league.  A copy
# array prevents member-row mutation from skipping the next rival member.
independence_wave_rival_bloc_reunify_into_league = {
	if = {
		limit = { has_global_flag = independence_wave_rival_bloc_active }
		clear_array = global.independence_wave_rival_bloc_reunification_entries
		for_each_scope_loop = { array = global.independence_wave_rival_bloc_member_country_entries add_to_array = { array = global.independence_wave_rival_bloc_reunification_entries value = THIS } }
		for_each_scope_loop = {
			array = global.independence_wave_rival_bloc_reunification_entries
			if = {
				limit = { is_independence_wave_active_country = yes has_country_flag = independence_wave_rival_bloc_member NOT = { has_country_flag = independence_wave_client_route_locked } }
				independence_wave_rival_bloc_unregister_member = yes
				independence_wave_register_league_member = yes
			}
		}
		clear_array = global.independence_wave_rival_bloc_reunification_entries
		independence_wave_rival_bloc_dissolve_contract = yes
	}
}
