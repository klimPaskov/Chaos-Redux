reset_white_peace_runtime_context = {
	set_variable = { global.white_peace_available = 0 }
	set_variable = { global.white_peace_context_ready = 0 }
	set_variable = { global.white_peace_active_war_count = 0 }
	set_variable = { global.white_peace_safe_minor_pair_count = 0 }
	set_variable = { global.white_peace_minor_only_war_count = 0 }
	set_variable = { global.white_peace_major_candidate_count = 0 }
	set_variable = { global.white_peace_major_locked_pair_count = 0 }
	set_variable = { global.white_peace_protected_war_actor_count = 0 }
	set_variable = { global.white_peace_recent_memory_actor_count = 0 }
	set_variable = { global.white_peace_recent_pair_memory_count = 0 }
	set_variable = { global.white_peace_environment_cap = 0 }
	set_variable = { global.white_peace_effective_cap = 0 }
	set_variable = { global.white_peace_effective_dynamic_cap = 0 }
	set_variable = { global.white_peace_repeatable_decay_multiplier = 1 }
	set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.no_active_war }
	set_variable = { global.white_peace_stage = constant:white_peace_event_log.stage_base }
	set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_base }
	set_temp_variable = { white_peace_pair_selected = 0 }
	set_temp_variable = { white_peace_branch = constant:white_peace_branch.single_minor }
	set_temp_variable = { white_peace_settlement_pair_cap = constant:white_peace_settlement_cap.base_pairs }
}

count_white_peace_minor_candidates = {
	set_variable = { global.white_peace_raw_active_war_country_count = 0 }
	set_variable = { global.white_peace_raw_safe_minor_pair_count = 0 }
	set_variable = { global.white_peace_raw_minor_only_pair_count = 0 }
	set_variable = { global.white_peace_protected_war_actor_count = 0 }
	set_variable = { global.white_peace_recent_memory_actor_count = 0 }
	set_variable = { global.white_peace_recent_pair_memory_count = 0 }

	every_country = {
		limit = {
			has_war = yes
			OR = {
				is_special_chaos_country = yes
				is_actual_nonhuman_country = yes
				has_country_flag = white_peace_protected_country
				has_country_flag = white_peace_protected_war_actor
			}
		}
		add_to_variable = { global.white_peace_protected_war_actor_count = 1 }
	}

	every_country = {
		limit = {
			has_war = yes
			OR = {
				has_country_flag = recent_white_peace_country
				has_country_flag = recent_major_white_peace_country
			}
		}
		add_to_variable = { global.white_peace_recent_memory_actor_count = 1 }
	}

	every_country = {
		limit = {
			can_country_be_white_peace_target = yes
			has_war = yes
		}
		every_enemy_country = {
			limit = {
				can_country_be_white_peace_target = yes
				has_war_with = PREV
				NOT = { tag = PREV }
				NOT = { is_in_faction_with = PREV }
				has_recent_white_peace_pair_with_prev = yes
			}
			add_to_variable = { global.white_peace_recent_pair_memory_count = 1 }
		}
	}

	every_country = {
		limit = {
			has_war = yes
			NOT = { is_special_chaos_country = yes }
			NOT = { is_actual_nonhuman_country = yes }
			NOT = { has_country_flag = white_peace_protected_country }
			NOT = { has_country_flag = white_peace_protected_war_actor }
		}
		add_to_variable = { global.white_peace_raw_active_war_country_count = 1 }
	}

	every_country = {
		limit = {
			can_country_be_white_peace_target = yes
			has_war = yes
		}
		every_enemy_country = {
			limit = { can_minor_pair_receive_white_peace = yes }
			add_to_variable = { global.white_peace_raw_safe_minor_pair_count = 1 }
			if = {
				limit = {
					has_war_with_major = no
					PREV = { has_war_with_major = no }
				}
				add_to_variable = { global.white_peace_raw_minor_only_pair_count = 1 }
			}
		}
	}

	set_temp_variable = { white_peace_count_proxy = global.white_peace_raw_active_war_country_count }
	divide_temp_variable = { white_peace_count_proxy = 2 }
	round_temp_variable = white_peace_count_proxy
	if = {
			limit = {
				check_variable = { global.white_peace_raw_active_war_country_count > 0 }
				check_variable = { white_peace_count_proxy < constant:white_peace_availability_min.active_wars }
			}
			set_temp_variable = { white_peace_count_proxy = constant:white_peace_availability_min.active_wars }
		}
	set_variable = { global.white_peace_active_war_count = white_peace_count_proxy }

	set_temp_variable = { white_peace_count_proxy = global.white_peace_raw_safe_minor_pair_count }
	divide_temp_variable = { white_peace_count_proxy = 2 }
	round_temp_variable = white_peace_count_proxy
	if = {
			limit = {
				check_variable = { global.white_peace_raw_safe_minor_pair_count > 0 }
				check_variable = { white_peace_count_proxy < constant:white_peace_branch_min_candidate.single_minor_pairs }
			}
			set_temp_variable = { white_peace_count_proxy = constant:white_peace_branch_min_candidate.single_minor_pairs }
		}
	set_variable = { global.white_peace_safe_minor_pair_count = white_peace_count_proxy }

	set_temp_variable = { white_peace_count_proxy = global.white_peace_raw_minor_only_pair_count }
	divide_temp_variable = { white_peace_count_proxy = 2 }
	round_temp_variable = white_peace_count_proxy
	if = {
			limit = {
				check_variable = { global.white_peace_raw_minor_only_pair_count > 0 }
				check_variable = { white_peace_count_proxy < constant:white_peace_branch_min_candidate.single_minor_pairs }
			}
			set_temp_variable = { white_peace_count_proxy = constant:white_peace_branch_min_candidate.single_minor_pairs }
		}
	set_variable = { global.white_peace_minor_only_war_count = white_peace_count_proxy }
}

determine_white_peace_stage = {
	set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_base }

	if = {
		limit = { has_global_flag = white_peace_stage_1_unlocked }
		set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_repeated_minor }
	}
	if = {
		limit = { has_global_flag = white_peace_stage_2_unlocked }
		set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_major_country }
	}
	if = {
		limit = { has_global_flag = white_peace_stage_3_unlocked }
		set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_broad_settlement }
	}
	if = {
		limit = {
			check_variable = { white_peace_stage < constant:white_peace_event_log.stage_repeated_minor }
			check_variable = {
				var = global.white_peace_active_war_count
				value = constant:white_peace_stage_pressure.repeated_minor_war_count
				compare = greater_than_or_equals
			}
			check_variable = {
				var = global.white_peace_safe_minor_pair_count
				value = constant:white_peace_stage_pressure.repeated_minor_pair_count
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_repeated_minor }
	}
	if = {
		limit = {
			check_variable = { white_peace_stage < constant:white_peace_event_log.stage_major_country }
			check_variable = {
				var = global.white_peace_active_war_count
				value = constant:white_peace_stage_pressure.major_war_count
				compare = greater_than_or_equals
			}
			check_variable = {
				var = global.white_peace_safe_minor_pair_count
				value = constant:white_peace_stage_pressure.major_pair_count
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_major_country }
	}
	if = {
		limit = {
			check_variable = { white_peace_stage < constant:white_peace_event_log.stage_broad_settlement }
			check_variable = {
				var = global.white_peace_active_war_count
				value = constant:white_peace_stage_pressure.broad_war_count
				compare = greater_than_or_equals
			}
			check_variable = {
				var = global.white_peace_safe_minor_pair_count
				value = constant:white_peace_stage_pressure.broad_pair_count
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { white_peace_stage = constant:white_peace_event_log.stage_broad_settlement }
	}

	set_variable = { global.white_peace_stage = white_peace_stage }
}

count_white_peace_major_candidates = {
	set_variable = { global.white_peace_raw_major_candidate_count = 0 }
	set_variable = { global.white_peace_raw_major_locked_pair_count = 0 }
	set_variable = { global.white_peace_major_locked_pair_count = 0 }
	if = {
		limit = { check_variable = { white_peace_stage > constant:white_peace_event_log.stage_repeated_minor } }
		every_country = {
			limit = {
				can_country_be_white_peace_target = yes
				has_war = yes
			}
			every_enemy_country = {
				limit = { can_major_pair_receive_white_peace = yes }
				add_to_variable = { global.white_peace_raw_major_candidate_count = 1 }
			}
		}
		set_temp_variable = { white_peace_count_proxy = global.white_peace_raw_major_candidate_count }
		divide_temp_variable = { white_peace_count_proxy = 2 }
		round_temp_variable = white_peace_count_proxy
		if = {
				limit = {
					check_variable = { global.white_peace_raw_major_candidate_count > 0 }
					check_variable = { white_peace_count_proxy < constant:white_peace_branch_min_candidate.major_pairs }
				}
				set_temp_variable = { white_peace_count_proxy = constant:white_peace_branch_min_candidate.major_pairs }
		}
		set_variable = { global.white_peace_major_candidate_count = white_peace_count_proxy }
	}
	else = {
		every_country = {
			limit = {
				can_country_be_white_peace_target = yes
				has_war = yes
			}
			every_enemy_country = {
				limit = { can_major_pair_receive_white_peace_if_stage_unlocked = yes }
				add_to_variable = { global.white_peace_raw_major_locked_pair_count = 1 }
			}
		}
		set_temp_variable = { white_peace_count_proxy = global.white_peace_raw_major_locked_pair_count }
		divide_temp_variable = { white_peace_count_proxy = 2 }
		round_temp_variable = white_peace_count_proxy
		if = {
			limit = {
				check_variable = { global.white_peace_raw_major_locked_pair_count > 0 }
				check_variable = { white_peace_count_proxy < constant:white_peace_branch_min_candidate.major_pairs }
			}
			set_temp_variable = { white_peace_count_proxy = constant:white_peace_branch_min_candidate.major_pairs }
		}
		set_variable = { global.white_peace_major_locked_pair_count = white_peace_count_proxy }
	}
}

calculate_white_peace_dynamic_cap = {
	reset_white_peace_runtime_context = yes
	count_white_peace_minor_candidates = yes
	determine_white_peace_stage = yes
	count_white_peace_major_candidates = yes

	if = {
		limit = {
			check_variable = {
				var = global.white_peace_active_war_count
				value = constant:white_peace_availability_min.active_wars
				compare = greater_than_or_equals
			}
			OR = {
				check_variable = {
					var = global.white_peace_safe_minor_pair_count
					value = constant:white_peace_branch_min_candidate.single_minor_pairs
					compare = greater_than_or_equals
				}
				check_variable = {
					var = global.white_peace_major_candidate_count
					value = constant:white_peace_branch_min_candidate.major_pairs
					compare = greater_than_or_equals
				}
			}
		}
		set_variable = { global.white_peace_available = 1 }
		set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.available }
		set_temp_variable = { white_peace_environment_cap = constant:white_peace_weight.base_cap }

		set_temp_variable = { white_peace_pressure_component = global.white_peace_active_war_count }
		multiply_temp_variable = { white_peace_pressure_component = constant:white_peace_weight.active_war_value }
		clamp_temp_variable = {
			var = white_peace_pressure_component
			min = 0
			max = constant:white_peace_weight.active_war_cap
		}
		add_to_temp_variable = { white_peace_environment_cap = white_peace_pressure_component }

		set_temp_variable = { white_peace_pressure_component = global.white_peace_safe_minor_pair_count }
		multiply_temp_variable = { white_peace_pressure_component = constant:white_peace_weight.safe_minor_pair_value }
		clamp_temp_variable = {
			var = white_peace_pressure_component
			min = 0
			max = constant:white_peace_weight.safe_minor_pair_cap
		}
		add_to_temp_variable = { white_peace_environment_cap = white_peace_pressure_component }

		set_temp_variable = { white_peace_pressure_component = global.white_peace_minor_only_war_count }
		multiply_temp_variable = { white_peace_pressure_component = constant:white_peace_weight.minor_only_war_value }
		clamp_temp_variable = {
			var = white_peace_pressure_component
			min = 0
			max = constant:white_peace_weight.minor_only_war_cap
		}
		add_to_temp_variable = { white_peace_environment_cap = white_peace_pressure_component }

		if = {
			limit = {
				check_variable = {
					var = global.white_peace_active_war_count
					value = constant:white_peace_stage_pressure.broad_war_count
					compare = greater_than_or_equals
				}
			}
			add_to_temp_variable = { white_peace_environment_cap = constant:white_peace_weight.clutter_bonus_12_wars }
		}
		else_if = {
			limit = {
				check_variable = {
					var = global.white_peace_active_war_count
					value = constant:white_peace_stage_pressure.major_war_count
					compare = greater_than_or_equals
				}
			}
			add_to_temp_variable = { white_peace_environment_cap = constant:white_peace_weight.clutter_bonus_8_wars }
		}
		else_if = {
			limit = {
				check_variable = {
					var = global.white_peace_active_war_count
					value = constant:white_peace_stage_pressure.repeated_minor_war_count
					compare = greater_than_or_equals
				}
			}
			add_to_temp_variable = { white_peace_environment_cap = constant:white_peace_weight.clutter_bonus_5_wars }
		}

		clamp_temp_variable = {
			var = white_peace_environment_cap
			min = constant:white_peace_weight.min_valid_cap
			max = constant:white_peace_weight.max_environment_cap
		}
		set_variable = { global.white_peace_environment_cap = white_peace_environment_cap }

		set_temp_variable = { white_peace_effective_cap = white_peace_environment_cap }
		set_temp_variable = { white_peace_repeatable_decay_multiplier = 1 }
		set_temp_variable = { white_peace_event_id = constant:white_peace_event_log.event_id }
		if = {
			limit = {
				check_variable = { white_peace_event_id < global.event_max_caps^num }
				check_variable = { global.default_event_weight > 0 }
			}
			set_temp_variable = { white_peace_repeatable_cap = global.event_max_caps^white_peace_event_id }
			set_temp_variable = { white_peace_repeatable_decay_multiplier = white_peace_repeatable_cap }
			divide_temp_variable = { white_peace_repeatable_decay_multiplier = global.default_event_weight }
			clamp_temp_variable = {
				var = white_peace_repeatable_decay_multiplier
				min = 0
				max = 1
			}
			multiply_temp_variable = { white_peace_effective_cap = white_peace_repeatable_decay_multiplier }
		}
		set_variable = { global.white_peace_repeatable_decay_multiplier = white_peace_repeatable_decay_multiplier }
		if = {
			limit = { check_variable = { white_peace_stage = constant:white_peace_event_log.stage_repeated_minor } }
			multiply_temp_variable = { white_peace_effective_cap = constant:white_peace_stage_multiplier.repeated_minor }
		}
		else_if = {
			limit = { check_variable = { white_peace_stage = constant:white_peace_event_log.stage_major_country } }
			multiply_temp_variable = { white_peace_effective_cap = constant:white_peace_stage_multiplier.major_country }
		}
		else_if = {
			limit = { check_variable = { white_peace_stage = constant:white_peace_event_log.stage_broad_settlement } }
			multiply_temp_variable = { white_peace_effective_cap = constant:white_peace_stage_multiplier.broad_settlement }
		}

		if = {
			limit = { has_global_flag = recent_broad_white_peace }
			multiply_temp_variable = { white_peace_effective_cap = constant:white_peace_weight.recent_global_penalty }
		}
		round_temp_variable = white_peace_effective_cap
		clamp_temp_variable = {
			var = white_peace_effective_cap
			min = constant:white_peace_weight.min_valid_cap
			max = constant:white_peace_weight.max_environment_cap
		}
		set_variable = { global.white_peace_effective_cap = white_peace_effective_cap }
		set_variable = { global.white_peace_effective_dynamic_cap = white_peace_effective_cap }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.white_peace_active_war_count
				value = constant:white_peace_availability_min.active_wars
				compare = greater_than_or_equals
			}
		}
		set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.no_safe_pair }
		if = {
			limit = {
				OR = {
					check_variable = { global.white_peace_recent_memory_actor_count > 0 }
					check_variable = { global.white_peace_recent_pair_memory_count > 0 }
				}
			}
			set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.recent_memory }
		}
		else_if = {
			limit = { check_variable = { global.white_peace_major_locked_pair_count > 0 } }
			set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.major_locked }
		}
	}
	else_if = {
		limit = { check_variable = { global.white_peace_protected_war_actor_count > 0 } }
		set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.protected_conflicts }
	}
}

select_white_peace_branch = {
	set_temp_variable = { white_peace_branch = constant:white_peace_branch.single_minor }
	set_temp_variable = { white_peace_settlement_pair_cap = constant:white_peace_settlement_cap.base_pairs }
	load_white_peace_branch_weight_variables = yes

	if = {
		limit = { check_variable = { white_peace_stage > constant:white_peace_event_log.stage_major_country } }
		random_list = {
			white_peace_branch_weight_stage_3_single = {
				modifier = {
					factor = 0
					check_variable = {
						var = global.white_peace_safe_minor_pair_count
						value = constant:white_peace_branch_min_candidate.single_minor_pairs
						compare = less_than
					}
				}
				set_temp_variable = { white_peace_branch = constant:white_peace_branch.single_minor }
			}
			white_peace_branch_weight_stage_3_multi = {
				modifier = {
					factor = 0
					check_variable = {
						var = global.white_peace_safe_minor_pair_count
						value = constant:white_peace_branch_min_candidate.multi_minor_pairs
						compare = less_than
					}
				}
				set_temp_variable = { white_peace_branch = constant:white_peace_branch.multi_minor }
				set_temp_variable = { white_peace_settlement_pair_cap = constant:white_peace_settlement_cap.stage_3_pairs }
			}
			white_peace_branch_weight_stage_3_major = {
				modifier = {
					factor = 0
					check_variable = {
						var = global.white_peace_major_candidate_count
						value = constant:white_peace_branch_min_candidate.major_pairs
						compare = less_than
					}
				}
				set_temp_variable = { white_peace_branch = constant:white_peace_branch.major_country }
			}
			white_peace_branch_weight_stage_3_broad = {
				modifier = {
					factor = 0
					OR = {
						check_variable = {
							var = global.white_peace_safe_minor_pair_count
							value = constant:white_peace_branch_min_candidate.broad_minor_pairs
							compare = less_than
						}
						has_global_flag = recent_broad_white_peace
					}
				}
				set_temp_variable = { white_peace_branch = constant:white_peace_branch.broad_settlement }
				set_temp_variable = { white_peace_settlement_pair_cap = constant:white_peace_settlement_cap.stage_3_pairs }
			}
		}
	}
	else_if = {
		limit = { check_variable = { white_peace_stage > constant:white_peace_event_log.stage_repeated_minor } }
		random_list = {
			white_peace_branch_weight_stage_2_single = {
				modifier = {
					factor = 0
					check_variable = {
						var = global.white_peace_safe_minor_pair_count
						value = constant:white_peace_branch_min_candidate.single_minor_pairs
						compare = less_than
					}
				}
				set_temp_variable = { white_peace_branch = constant:white_peace_branch.single_minor }
			}
			white_peace_branch_weight_stage_2_multi = {
				modifier = {
					factor = 0
					check_variable = {
						var = global.white_peace_safe_minor_pair_count
						value = constant:white_peace_branch_min_candidate.multi_minor_pairs

record_white_peace_evolution_if_needed = {
	set_temp_variable = { events_log_evolution_event_id = constant:white_peace_event_log.event_id }
	set_temp_variable = { events_log_evolution_type = constant:white_peace_event_log.evolution_type }
	set_temp_variable = { events_log_evolution_stage = white_peace_stage }
	set_temp_variable = { events_log_evolution_tier = white_peace_stage }
	if = {
			limit = {
				is_current_evolution_enabled = yes
				check_variable = { white_peace_stage > constant:white_peace_event_log.stage_base }
				OR = {
				AND = {
					check_variable = { white_peace_stage = constant:white_peace_event_log.stage_repeated_minor }
					NOT = { has_global_flag = white_peace_stage_1_recorded }
				}
				AND = {
					check_variable = { white_peace_stage = constant:white_peace_event_log.stage_major_country }
					NOT = { has_global_flag = white_peace_stage_2_recorded }
				}
				AND = {
					check_variable = { white_peace_stage = constant:white_peace_event_log.stage_broad_settlement }
					NOT = { has_global_flag = white_peace_stage_3_recorded }
				}
			}
		}
		record_events_log_evolution_entry = yes
		if = {
			limit = { check_variable = { white_peace_stage = constant:white_peace_event_log.stage_repeated_minor } }
			set_global_flag = white_peace_stage_1_recorded
			set_global_flag = white_peace_stage_1_unlocked
		}
		else_if = {
			limit = { check_variable = { white_peace_stage = constant:white_peace_event_log.stage_major_country } }
			set_global_flag = white_peace_stage_2_recorded
			set_global_flag = white_peace_stage_2_unlocked
		}
		else_if = {
			limit = { check_variable = { white_peace_stage = constant:white_peace_event_log.stage_broad_settlement } }
			set_global_flag = white_peace_stage_3_recorded
			set_global_flag = white_peace_stage_3_unlocked
		}
	}
}
