# Supported scope: COUNTRY coordinator. A committed ledger is immutable and
# this reset becomes a no-op once the ready flag exists.
fallout_reset_uncommitted_survival_ledger = {
	if = {
		limit = { NOT = { has_global_flag = fallout_survival_ledger_ready } }
		clr_global_flag = fallout_survival_ledger_identity_staged
		clr_global_flag = fallout_survival_ledger_building
		for_each_scope_loop = {
			array = global.fallout_survival_ledger_countries
			fallout_survival_clear_country_row = yes
		}
		clear_array = global.fallout_survival_ledger_countries
		clear_array = global.fallout_survival_ledger_country_generation_entries
		clear_array = global.fallout_survival_ledger_country_index_entries
		clear_array = global.fallout_survival_ledger_states

		clear_array = global.fallout_survival_ledger_state_index_entries
		clear_variable = global.fallout_survival_ledger_schema_version
		clear_variable = global.fallout_survival_ledger_generation
		clear_variable = global.fallout_survival_ledger_source_allocation_schema
		clear_variable = global.fallout_survival_ledger_source_allocation_generation
		clear_variable = global.fallout_survival_ledger_source_country_count
		clear_variable = global.fallout_survival_ledger_country_count
		clear_variable = global.fallout_survival_ledger_source_state_count
		clear_variable = global.fallout_survival_ledger_state_count
		clear_variable = global.fallout_survival_ledger_identity_staged_date
		clear_variable = global.fallout_survival_ledger_identity_staged_day
		clear_variable = global.fallout_survival_ledger_committed_date
		clear_variable = global.fallout_survival_ledger_committed_day
		every_country = { fallout_survival_clear_country_row = yes }
		every_possible_country = { fallout_survival_clear_country_row = yes }
		every_state = { fallout_survival_clear_state_row = yes }
	}
}

# Supported scope: COUNTRY coordinator. This consumes only the exact
# uncommitted identity-build error, then rebuilds the complete identity payload
# through the ordinary staging transaction. A failed retry owns the same sole
# error signature again.
fallout_recover_uncommitted_survival_ledger_identity_transaction = {
	if = {
		limit = { fallout_recoverable_survival_ledger_identity_error_is_active = yes }
		clr_global_flag = fallout_transition_error
		set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.none }
		set_variable = { global.fallout_transition_error_count = constant:fallout_runtime.zero }
		fallout_stage_survival_ledger_identity_transaction = yes
	}
}

# Supported scope: COUNTRY coordinator. This identity-first phase freezes
# all-and-only allocated country and state identities plus the producer
# generations required by the reviewed numerical transaction.
fallout_stage_survival_ledger_identity_transaction = {
	if = {
		limit = {
			NOT = { has_global_flag = fallout_survival_ledger_ready }
			NOT = { has_global_flag = fallout_transition_error }
			NOT = { has_global_flag = fallout_transition_survivor_allocation_applied }
			NOT = { has_global_flag = fallout_transition_player_continuation_applied }
			check_variable = {
				var = global.fallout_transition_phase
				value = constant:fallout_transition_phase.survivor_allocation
				compare = equals
			}
			fallout_player_planning_ledger_is_current = yes
			fallout_successor_allocation_is_current = yes
			NOT = { has_global_flag = fallout_player_continuation_commit_complete }
		}
		fallout_reset_uncommitted_survival_ledger = yes
		set_global_flag = fallout_survival_ledger_building
		set_variable = { global.fallout_survival_ledger_schema_version = constant:fallout_survival_ledger_schema.version }
		set_variable = { global.fallout_survival_ledger_generation = global.fallout_transition_generation }
		set_variable = { global.fallout_survival_ledger_source_allocation_schema = global.fallout_successor_allocation_schema_version }
		set_variable = { global.fallout_survival_ledger_source_allocation_generation = global.fallout_successor_allocation_generation }
		set_variable = { global.fallout_survival_ledger_source_country_count = global.fallout_successor_assignment_count }
		set_variable = { global.fallout_survival_ledger_source_state_count = global.fallout_successor_inventory_state_count }
		set_temp_variable = { fallout_survival_country_build_index = constant:fallout_runtime.zero }
		for_each_scope_loop = {
			array = global.fallout_successor_assigned_countries
			set_country_flag = fallout_survival_country_identity_staged
			set_variable = { fallout_survival_country_schema_version = constant:fallout_survival_ledger_schema.version }
			set_variable = { fallout_survival_country_generation = global.fallout_transition_generation }
			set_variable = { fallout_survival_country_index = fallout_survival_country_build_index }
			set_variable = { fallout_survival_country_source_allocation_schema = global.fallout_successor_allocation_schema_version }
			set_variable = { fallout_survival_country_source_allocation_generation = fallout_successor_assignment_generation }
			set_variable = { fallout_survival_country_state_count = num_owned_states }
			set_variable = { fallout_survival_country_source_region = fallout_region_id }
			set_variable = { fallout_survival_country_source_archetype = fallout_government_archetype }
			set_variable = { fallout_survival_country_source_memory = fallout_country_memory_id }
			clear_array = fallout_survival_country_resource_id_entries
			for_loop_effect = {
				start = constant:fallout_event_runtime.first_array_index
				end = constant:fallout_survival_resource.upper_bound
				value = fallout_survival_resource_build_index
				add_to_array = {
					array = fallout_survival_country_resource_id_entries
					value = fallout_survival_resource_build_index
				}
			}
			add_to_array = { array = global.fallout_survival_ledger_countries value = THIS }
			add_to_array = { array = global.fallout_survival_ledger_country_generation_entries value = global.fallout_transition_generation }
			add_to_array = { array = global.fallout_survival_ledger_country_index_entries value = fallout_survival_country_build_index }
			add_to_temp_variable = { fallout_survival_country_build_index = constant:fallout_runtime.one }
		}
		set_temp_variable = { fallout_survival_state_build_index = constant:fallout_runtime.zero }
		every_state = {
			set_state_flag = fallout_survival_state_identity_staged
			set_variable = { fallout_survival_state_schema_version = constant:fallout_survival_ledger_schema.version }
			set_variable = { fallout_survival_state_generation = global.fallout_transition_generation }
			set_variable = { fallout_survival_state_index = fallout_survival_state_build_index }
			set_variable = { fallout_survival_state_owner = OWNER }
			set_variable = { fallout_survival_state_source_snapshot_generation = fallout_pretransition_snapshot_generation }
			set_variable = { fallout_survival_state_source_air_winter_schema = fallout_pretransition_air_winter_source_schema_version }
			set_variable = { fallout_survival_state_source_air_winter_generation = fallout_pretransition_air_winter_source_generation }
			set_variable = { fallout_survival_state_source_air_winter_kind = fallout_pretransition_air_winter_source_kind }
			set_variable = { fallout_survival_state_source_grading_generation = fallout_state_grading_generation }
			set_variable = { fallout_survival_state_source_population_generation = fallout_population_loss_generation }
			set_variable = { fallout_survival_state_source_rewrite_generation = fallout_state_rewrite_generation }
			set_variable = { fallout_survival_state_source_supply_generation = fallout_supply_network_collapse_generation }
			clear_array = fallout_survival_state_resource_id_entries
			for_loop_effect = {
				start = constant:fallout_event_runtime.first_array_index
				end = constant:fallout_survival_resource.upper_bound
				value = fallout_survival_resource_build_index
				add_to_array = {
					array = fallout_survival_state_resource_id_entries
					value = fallout_survival_resource_build_index
				}
			}
			add_to_array = { array = global.fallout_survival_ledger_states value = THIS }
			add_to_array = { array = global.fallout_survival_ledger_state_generation_entries value = global.fallout_transition_generation }
			add_to_array = { array = global.fallout_survival_ledger_state_index_entries value = fallout_survival_state_build_index }
			add_to_temp_variable = { fallout_survival_state_build_index = constant:fallout_runtime.one }
		}
		set_variable = { global.fallout_survival_ledger_country_count = global.fallout_survival_ledger_countries^num }
		set_variable = { global.fallout_survival_ledger_state_count = global.fallout_survival_ledger_states^num }
		set_variable = { global.fallout_survival_ledger_identity_staged_date = global.date }
		set_variable = { global.fallout_survival_ledger_identity_staged_day = global.num_days }
		if = {
			limit = { fallout_survival_ledger_identity_payload_is_current = yes }
			set_global_flag = fallout_survival_ledger_identity_staged
			clr_global_flag = fallout_survival_ledger_building
		}
		else = {
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.survival_ledger_incomplete }
			add_to_variable = { global.fallout_transition_error_count = constant:fallout_runtime.one }
		}
	}
}

		limit = {
			fallout_survival_state_identity_row_is_current = yes
			fallout_physical_collapse_row_is_durable = yes
			check_variable = { var = fallout_survival_state_raw_numerator_entries^num value = constant:fallout_survival_resource.upper_bound compare = equals }
			check_variable = { var = fallout_survival_state_raw_denominator_entries^num value = constant:fallout_survival_resource.upper_bound compare = equals }

					OR = {
						NOT = { check_variable = { var = fallout_survival_state_raw_numerator_entries^fallout_survival_resource_build_index value = fallout_survival_expected_state_numerator compare = equals } }
						NOT = { check_variable = { var = fallout_survival_state_raw_denominator_entries^fallout_survival_resource_build_index value = fallout_survival_expected_state_denominator compare = equals } }
						NOT = { check_variable = { var = fallout_survival_state_equal_share_entries^fallout_survival_resource_build_index value = fallout_survival_expected_state_share compare = equals } }
					}
				}
				add_to_temp_variable = { fallout_survival_replay_mismatch_count = constant:fallout_event_runtime.one }
			}
		}
		fallout_state_calculate_supply_access_expected = yes
		set_variable = { fallout_state_supply_access_replay_initial = fallout_state_expected_supply_access }
		set_temp_variable = { fallout_state_expected_supply_impact = constant:fallout_state_supply_access.impact_base }

					NOT = { check_variable = { fallout_state_supply_access_schema_version = constant:fallout_survival_ledger_schema.version } }

	else = { add_to_temp_variable = { fallout_survival_replay_mismatch_count = constant:fallout_event_runtime.one } }

					divide_temp_variable = { fallout_survival_state_score = constant:fallout_survival_numeric.denominator }
					set_temp_variable = { fallout_survival_state_weighted_score = fallout_survival_state_score }
					multiply_temp_variable = { fallout_survival_state_weighted_score = fallout_survival_state_population_weight }
					add_to_temp_variable = { fallout_survival_expected_country_weight_sum = fallout_survival_state_population_weight }
					add_to_temp_variable = { fallout_survival_expected_country_weighted_sum = fallout_survival_state_weighted_score }
					if = {
						limit = { check_variable = { fallout_survival_state_score > fallout_survival_expected_country_hub_score } }
						set_temp_variable = { fallout_survival_expected_country_hub_score = fallout_survival_state_score }
					}
				}
			}
		}
	}
	set_temp_variable = { fallout_survival_expected_country_numerator = constant:fallout_survival_numeric.minimum }
	set_temp_variable = { fallout_survival_expected_country_denominator = constant:fallout_survival_numeric.minimum }
	set_temp_variable = { fallout_survival_expected_country_initial = constant:fallout_survival_numeric.minimum }
	if = {
		limit = { check_variable = { fallout_survival_resource_build_index > constant:fallout_survival_resource.none } }
		set_temp_variable = { fallout_survival_expected_country_denominator = constant:fallout_survival_numeric.denominator }
		if = {
			limit = {
				check_variable = { fallout_survival_expected_country_produced_count > constant:fallout_event_runtime.zero }
				check_variable = { fallout_survival_expected_country_weight_sum > constant:fallout_survival_numeric.minimum }
			}
			set_temp_variable = { fallout_survival_expected_country_coverage = fallout_survival_expected_country_weighted_sum }
			divide_temp_variable = { fallout_survival_expected_country_coverage = fallout_survival_expected_country_weight_sum }
			set_temp_variable = { fallout_survival_country_coverage_weight = constant:fallout_survival_country_blend.local_coverage }
			set_temp_variable = { fallout_survival_country_hub_weight = constant:fallout_survival_country_blend.local_hub }
			if = {
				limit = {
					OR = {
						check_variable = { fallout_survival_resource_build_index = constant:fallout_survival_resource.scrap }
						check_variable = { fallout_survival_resource_build_index = constant:fallout_survival_resource.fuel }
						check_variable = { fallout_survival_resource_build_index = constant:fallout_survival_resource.power }
					}
				}
				set_temp_variable = { fallout_survival_country_coverage_weight = constant:fallout_survival_country_blend.industrial_coverage }
				set_temp_variable = { fallout_survival_country_hub_weight = constant:fallout_survival_country_blend.industrial_hub }
			}
			else_if = {
				limit = { check_variable = { fallout_survival_resource_build_index = constant:fallout_survival_resource.recognition } }
				set_temp_variable = { fallout_survival_country_coverage_weight = constant:fallout_survival_country_blend.recognition_coverage }
				set_temp_variable = { fallout_survival_country_hub_weight = constant:fallout_survival_country_blend.recognition_hub }
			}
			set_temp_variable = { fallout_survival_expected_country_numerator = fallout_survival_expected_country_coverage }
			multiply_temp_variable = { fallout_survival_expected_country_numerator = fallout_survival_country_coverage_weight }
			set_temp_variable = { fallout_survival_country_hub_component = fallout_survival_expected_country_hub_score }
			multiply_temp_variable = { fallout_survival_country_hub_component = fallout_survival_country_hub_weight }
			add_to_temp_variable = { fallout_survival_expected_country_numerator = fallout_survival_country_hub_component }
			clamp_temp_variable = { var = fallout_survival_expected_country_numerator min = constant:fallout_survival_numeric.minimum max = constant:fallout_survival_numeric.maximum_numerator }
			set_temp_variable = { fallout_survival_expected_country_initial = fallout_survival_expected_country_numerator }
			divide_temp_variable = { fallout_survival_expected_country_initial = fallout_survival_expected_country_denominator }
			round_temp_variable = fallout_survival_expected_country_initial
			if = {
				limit = { check_variable = { fallout_survival_resource_build_index = constant:fallout_survival_resource.recognition } }
				fallout_survival_calculate_recognition_adjustment = yes
				add_to_temp_variable = { fallout_survival_expected_country_initial = fallout_survival_recognition_adjustment_value }
			}
			clamp_temp_variable = { var = fallout_survival_expected_country_initial min = constant:fallout_survival_numeric.minimum max = constant:fallout_survival_numeric.maximum }
		}
	}
}

	clear_variable = fallout_survival_country_replay_produced_state_count

	clear_array = fallout_survival_replay_weighted_score_sum_entries
	clear_array = fallout_survival_replay_hub_score_entries
	clear_variable = fallout_survival_cohesion_replay_initial
	set_temp_variable = { fallout_survival_replay_mismatch_count = constant:fallout_event_runtime.zero }

			check_variable = { var = fallout_survival_weighted_score_sum_entries^num value = constant:fallout_survival_resource.upper_bound compare = equals }
			check_variable = { var = fallout_survival_hub_score_entries^num value = constant:fallout_survival_resource.upper_bound compare = equals }

			start = constant:fallout_event_runtime.first_array_index
			end = constant:fallout_survival_resource.upper_bound
			value = fallout_survival_resource_build_index
			fallout_survival_calculate_country_resource_expected = yes
			add_to_array = { array = fallout_survival_replay_raw_numerator_entries value = fallout_survival_expected_country_numerator }
			add_to_array = { array = fallout_survival_replay_raw_denominator_entries value = fallout_survival_expected_country_denominator }
			add_to_array = { array = fallout_survival_replay_initial_entries value = fallout_survival_expected_country_initial }
			add_to_array = { array = fallout_survival_replay_population_weight_sum_entries value = fallout_survival_expected_country_weight_sum }
			add_to_array = { array = fallout_survival_replay_weighted_score_sum_entries value = fallout_survival_expected_country_weighted_sum }
			add_to_array = { array = fallout_survival_replay_hub_score_entries value = fallout_survival_expected_country_hub_score }
			if = {

					OR = {
						NOT = { check_variable = { var = fallout_survival_raw_numerator_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_numerator compare = equals } }
						NOT = { check_variable = { var = fallout_survival_raw_denominator_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_denominator compare = equals } }
						NOT = { check_variable = { var = fallout_survival_initial_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_initial compare = equals } }
						NOT = { check_variable = { var = fallout_survival_current_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_initial compare = equals } }
						NOT = { check_variable = { var = fallout_survival_population_weight_sum_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_weight_sum compare = equals } }
						NOT = { check_variable = { var = fallout_survival_weighted_score_sum_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_weighted_sum compare = equals } }
						NOT = { check_variable = { var = fallout_survival_hub_score_entries^fallout_survival_resource_build_index value = fallout_survival_expected_country_hub_score compare = equals } }
					}
				}

# Supported scope: COUNTRY coordinator. Sole numerical producer and sole setter
# of fallout_survival_ledger_ready. The ready flag is the final write.
fallout_commit_survival_ledger_numerical_transaction = {
	if = {
		limit = {
			NOT = { has_global_flag = fallout_transition_error }
			NOT = { has_global_flag = fallout_survival_ledger_ready }
			NOT = { has_global_flag = fallout_transition_survivor_allocation_applied }
			NOT = { has_global_flag = fallout_player_continuation_commit_complete }
			NOT = { has_global_flag = fallout_transition_player_continuation_applied }
			check_variable = { var = global.fallout_transition_phase value = constant:fallout_transition_phase.survivor_allocation compare = equals }
			fallout_player_planning_ledger_is_current = yes
			fallout_successor_allocation_is_current = yes
			fallout_survival_ledger_identity_is_current = yes
		}
		set_global_flag = fallout_survival_ledger_building
		fallout_reset_uncommitted_survival_numerical_payload = yes
		for_each_scope_loop = {
			array = global.fallout_survival_ledger_states
			fallout_survival_stage_state_numerical_row = yes
		}
		if = {
			limit = { fallout_survival_state_numerical_rows_are_precommit_current = yes }
			for_each_scope_loop = {
				array = global.fallout_survival_ledger_countries
				fallout_survival_stage_country_numerical_row = yes
			}
		}
		set_variable = { global.fallout_survival_ledger_committed_date = global.date }
		set_variable = { global.fallout_survival_ledger_committed_day = global.num_days }
		if = {
			limit = { fallout_survival_ledger_numerical_precommit_payload_is_current = yes }
			clr_global_flag = fallout_survival_ledger_building
			set_global_flag = fallout_survival_ledger_ready
		}
		else = {
			fallout_reset_uncommitted_survival_numerical_payload = yes
			clr_global_flag = fallout_survival_ledger_building
			set_global_flag = fallout_transition_error
			set_variable = { global.fallout_transition_error_code = constant:fallout_transition_error_code.survival_ledger_incomplete }
			set_variable = { global.fallout_transition_error_count = constant:fallout_event_runtime.one }
		}
	}
}

	}
