# ============================================================================
# CHAOS REDUX - EVENT 006 COUNTRY REGISTRY EFFECTS
# ============================================================================
# Narrow lifecycle wrappers for the canonical Event 006 origin marker. The
# helper is called only by Event 006 setup/cleanup; Event 012 must never call it
# and records its own Africa package flags instead.
# ============================================================================

# COUNTRY scope. Called by independence_wave_prepare_country_origin after the
# generation reset has validated Event 006 setup inputs.
independence_wave_registry_record_event6_origin = {
	set_country_flag = independence_wave_active_origin
	clr_country_flag = independence_wave_origin_ended
	set_variable = { liberation_origin = constant:liberation_origin.independence_wave }
}

# COUNTRY scope. Called by independence_wave_end_active_origin after package,
# focus, network, and ledger cleanup. It intentionally leaves the historical
# ended marker while clearing the live origin value.
independence_wave_registry_clear_event6_origin = {
	clr_country_flag = independence_wave_active_origin
	set_country_flag = independence_wave_origin_ended
	set_variable = { liberation_origin = constant:liberation_origin.none }
}
