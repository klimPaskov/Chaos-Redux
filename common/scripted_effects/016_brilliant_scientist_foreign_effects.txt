# ============================================================================
# CHAOS REDUX - EVENT 016 FOREIGN OPERATIONS EFFECTS
# ============================================================================
# The Evolution II international contest is initialized from the active host
# once per host tenure. Actors receive one bounded host target. Targeted timed
# decisions own scheduling; regular event targets carry actor and host through
# each response chain. Persistent parallel ledgers retain actor, host, family,
# operation, result, detection, and one-time reward history.
# ============================================================================

# ---------------------------------------------------------------------------
# Contest initialization and scoped lifecycle
# ---------------------------------------------------------------------------

# COUNTRY scope on the current host. This explicit lifecycle call performs one
# bounded world scan when Evolution II opens or the host changes. It is never
# called by a periodic on_action.
brilliant_scientist_foreign_initialize_contest = {
	if = {
		limit = {
			brilliant_scientist_is_current_host = yes
			has_country_flag = brilliant_scientist_evolution_ii_unlocked
			NOT = { has_country_flag = brilliant_scientist_kruger_confirmed_dead }
			NOT = { has_country_flag = brilliant_scientist_all_actions_locked_after_world_end }
			NOT = { has_global_flag = world_end }
		}
		set_variable = { brilliant_scientist_high_interest_foreign_actor_count = constant:brilliant_scientist_value.zero }
		every_country = {
			clear_array = brilliant_scientist_foreign_host_targets
		}
		every_country = {
			limit = { brilliant_scientist_is_valid_foreign_actor = yes }
			add_to_array = { array = brilliant_scientist_foreign_host_targets value = ROOT }
			set_country_flag = brilliant_scientist_foreign_contest_observer
			brilliant_scientist_refresh_foreign_interest = yes
			if = {
				limit = {
					check_variable = {
						var = brilliant_scientist_foreign_interest
						value = constant:brilliant_scientist_recognition_threshold.high_interest_score
						compare = greater_than_or_equals
					}
				}
				ROOT = { add_to_variable = { brilliant_scientist_high_interest_foreign_actor_count = constant:brilliant_scientist_value.one } }
			}
		}
		set_country_flag = brilliant_scientist_foreign_contest_initialized
		brilliant_scientist_refresh_international_recognition_score = yes
	}
}

# Targeted-decision ROOT actor, FROM host. Input temporary operation type.
brilliant_scientist_foreign_start_operation = {
	set_country_flag = brilliant_scientist_foreign_operation_live
	clr_country_flag = brilliant_scientist_foreign_operation_resolution_recorded
	clr_country_flag = brilliant_scientist_foreign_operation_compromised
	clr_country_flag = brilliant_scientist_foreign_gained_new_family_knowledge
	clr_country_flag = brilliant_scientist_foreign_transfer_received
	clr_country_flag = brilliant_scientist_foreign_transfer_race_prevented
	clr_country_flag = brilliant_scientist_foreign_defection_assistant_departed
	clr_country_flag = brilliant_scientist_foreign_extraction_captured_assistant
	clr_country_flag = brilliant_scientist_foreign_assassination_continuity_target
	clr_country_flag = brilliant_scientist_foreign_assassination_confirmed_death
	clr_country_flag = brilliant_scientist_foreign_assassination_injury
	set_variable = { brilliant_scientist_foreign_live_operation = brilliant_scientist_foreign_operation_type }
	set_variable = { brilliant_scientist_foreign_operation_started_date = global.date }
	set_variable = { brilliant_scientist_foreign_operation_host_id = FROM.id }
	add_to_variable = { brilliant_scientist_foreign_operations_started = constant:brilliant_scientist_foreign_consequence.operation_increment }
	FROM = {
		add_to_variable = { brilliant_scientist_foreign_incoming_operation_count = constant:brilliant_scientist_foreign_consequence.operation_increment }
		add_to_array = { array = brilliant_scientist_foreign_incoming_operation_actors value = ROOT }
		if = {
			limit = { check_variable = { ROOT.brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.assassination_attempt } }
			set_country_flag = brilliant_scientist_foreign_assassination_attempt_live
		}
	}
}

# COUNTRY scope on a host. Output temporary selected family and stage. The most
# advanced live family is the strategic target; equal stages preserve the
# lower stable family id. Input temporary require_undamaged may reject families
# already carrying sabotage damage.
brilliant_scientist_foreign_select_project_family = {
	set_temp_variable = { brilliant_scientist_foreign_selected_family = constant:brilliant_scientist_project_family.none }
	set_temp_variable = { brilliant_scientist_foreign_selected_stage = constant:brilliant_scientist_project_stage.none }
	set_temp_variable = { brilliant_scientist_foreign_project_index = constant:brilliant_scientist_value.zero }
	while = {
		limit = { check_variable = { brilliant_scientist_foreign_project_index < constant:brilliant_scientist_event.project_family_count } }
		set_temp_variable = { brilliant_scientist_foreign_candidate_family = brilliant_scientist_foreign_project_index }
		add_to_temp_variable = { brilliant_scientist_foreign_candidate_family = constant:brilliant_scientist_value.one }
		set_temp_variable = { brilliant_scientist_foreign_candidate_stage = brilliant_scientist_project_stage_entries^brilliant_scientist_foreign_project_index }
		set_temp_variable = { brilliant_scientist_foreign_candidate_valid = constant:brilliant_scientist_value.one }
		if = {
			limit = {
				check_variable = { brilliant_scientist_foreign_require_undamaged > constant:brilliant_scientist_value.zero }
				is_in_array = { array = brilliant_scientist_project_damaged_families value = brilliant_scientist_foreign_candidate_family }
			}
			set_temp_variable = { brilliant_scientist_foreign_candidate_valid = constant:brilliant_scientist_value.zero }
		}
		if = {
			limit = {
				check_variable = { brilliant_scientist_foreign_require_unstolen > constant:brilliant_scientist_value.zero }
				is_in_array = { array = brilliant_scientist_project_stolen_families value = brilliant_scientist_foreign_candidate_family }
			}
			set_temp_variable = { brilliant_scientist_foreign_candidate_valid = constant:brilliant_scientist_value.zero }
		}
		if = {
			limit = {
				check_variable = { brilliant_scientist_foreign_candidate_valid > constant:brilliant_scientist_value.zero }
				check_variable = {
					var = brilliant_scientist_foreign_candidate_stage
					value = brilliant_scientist_foreign_selected_stage
					compare = greater_than
				}
			}
			set_temp_variable = { brilliant_scientist_foreign_selected_family = brilliant_scientist_foreign_candidate_family }
			set_temp_variable = { brilliant_scientist_foreign_selected_stage = brilliant_scientist_foreign_candidate_stage }
		}
		add_to_temp_variable = { brilliant_scientist_foreign_project_index = constant:brilliant_scientist_value.one }
	}
}

# Targeted-decision ROOT actor, FROM host. Stores the exact selected family for
# the delayed resolution chain.
brilliant_scientist_foreign_prepare_project_target = {
	set_temp_variable = { brilliant_scientist_foreign_require_undamaged = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_foreign_require_unstolen = constant:brilliant_scientist_value.zero }
	if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.project_sabotage } }
		set_temp_variable = { brilliant_scientist_foreign_require_undamaged = constant:brilliant_scientist_value.one }
	}
	if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.archive_theft } }
		set_temp_variable = { brilliant_scientist_foreign_require_unstolen = constant:brilliant_scientist_value.one }
	}
	FROM = { brilliant_scientist_foreign_select_project_family = yes }
	set_variable = { brilliant_scientist_foreign_selected_project_family = brilliant_scientist_foreign_selected_family }
	set_variable = { brilliant_scientist_foreign_selected_project_stage = brilliant_scientist_foreign_selected_stage }
}

# COUNTRY scope on a foreign actor.
brilliant_scientist_foreign_initialize_knowledge_ledger = {
	if = {
		limit = { NOT = { has_country_flag = brilliant_scientist_foreign_knowledge_ledger_initialized } }
		clear_array = brilliant_scientist_foreign_project_knowledge_entries
		set_temp_variable = { brilliant_scientist_foreign_knowledge_index = constant:brilliant_scientist_value.zero }
		while = {
			limit = { check_variable = { brilliant_scientist_foreign_knowledge_index < constant:brilliant_scientist_event.project_family_count } }
			add_to_array = { array = brilliant_scientist_foreign_project_knowledge_entries value = constant:brilliant_scientist_project_stage.none }
			add_to_temp_variable = { brilliant_scientist_foreign_knowledge_index = constant:brilliant_scientist_value.one }
		}
		set_country_flag = brilliant_scientist_foreign_knowledge_ledger_initialized
	}
}

# COUNTRY scope on a foreign actor. Persistent selected family is the input.
# Foreign acquisition is capped at Theory and never copies the Event 016 host
# package, Kruger token, project armies, or later-stage ledgers.
brilliant_scientist_foreign_grant_selected_family_theory = {
	brilliant_scientist_foreign_initialize_knowledge_ledger = yes
	set_temp_variable = { brilliant_scientist_foreign_knowledge_index = brilliant_scientist_foreign_selected_project_family }
	subtract_from_temp_variable = { brilliant_scientist_foreign_knowledge_index = constant:brilliant_scientist_value.one }
	if = {
		limit = {
			check_variable = { brilliant_scientist_foreign_knowledge_index > constant:brilliant_scientist_value.invalid }
			check_variable = { brilliant_scientist_foreign_knowledge_index < constant:brilliant_scientist_event.project_family_count }
			check_variable = {
				var = brilliant_scientist_foreign_project_knowledge_entries^brilliant_scientist_foreign_knowledge_index
				value = constant:brilliant_scientist_foreign_limit.project_knowledge_stage
				compare = less_than
			}
		}
		set_variable = { brilliant_scientist_foreign_project_knowledge_entries^brilliant_scientist_foreign_knowledge_index = constant:brilliant_scientist_foreign_limit.project_knowledge_stage }
		add_to_variable = { brilliant_scientist_foreign_known_project_family_count = constant:brilliant_scientist_foreign_consequence.operation_increment }
		set_country_flag = brilliant_scientist_foreign_gained_new_family_knowledge
	}
}

# Any scope with regular actor and host event targets established.
brilliant_scientist_foreign_record_resolution = {
	if = {
		limit = {
			has_event_target = brilliant_scientist_foreign_operation_actor
			has_event_target = brilliant_scientist_foreign_operation_host
			event_target:brilliant_scientist_foreign_operation_actor = { NOT = { has_country_flag = brilliant_scientist_foreign_operation_resolution_recorded } }
		}
		event_target:brilliant_scientist_foreign_operation_actor = {
			set_country_flag = brilliant_scientist_foreign_operation_resolution_recorded
			add_to_array = { array = brilliant_scientist_foreign_operation_history_hosts value = event_target:brilliant_scientist_foreign_operation_host }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_types value = brilliant_scientist_foreign_live_operation }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_results value = brilliant_scientist_foreign_last_operation_result }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_detected value = brilliant_scientist_foreign_last_operation_detected }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_families value = brilliant_scientist_foreign_selected_project_family }
			add_to_variable = { brilliant_scientist_foreign_operations_resolved = constant:brilliant_scientist_foreign_consequence.operation_increment }
			if = {
				limit = { check_variable = { brilliant_scientist_foreign_last_operation_result = constant:brilliant_scientist_foreign_result.success } }
				add_to_variable = { brilliant_scientist_foreign_operations_succeeded = constant:brilliant_scientist_foreign_consequence.operation_increment }
				clr_country_flag = brilliant_scientist_foreign_prior_operation_failed
			}
			else_if = {
				limit = { check_variable = { brilliant_scientist_foreign_last_operation_result = constant:brilliant_scientist_foreign_result.partial } }
				add_to_variable = { brilliant_scientist_foreign_operations_partial = constant:brilliant_scientist_foreign_consequence.operation_increment }
			}
			else_if = {
				limit = { check_variable = { brilliant_scientist_foreign_last_operation_result = constant:brilliant_scientist_foreign_result.failure } }
				add_to_variable = { brilliant_scientist_foreign_operations_failed = constant:brilliant_scientist_foreign_consequence.operation_increment }
				set_country_flag = brilliant_scientist_foreign_prior_operation_failed
			}
		}
		event_target:brilliant_scientist_foreign_operation_host = {
			set_country_flag = brilliant_scientist_first_foreign_action_resolved
			add_to_array = { array = brilliant_scientist_foreign_operation_history_actors value = event_target:brilliant_scientist_foreign_operation_actor }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_types value = event_target:brilliant_scientist_foreign_operation_actor.brilliant_scientist_foreign_live_operation }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_results value = event_target:brilliant_scientist_foreign_operation_actor.brilliant_scientist_foreign_last_operation_result }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_detected value = event_target:brilliant_scientist_foreign_operation_actor.brilliant_scientist_foreign_last_operation_detected }
			add_to_array = { array = brilliant_scientist_foreign_operation_history_families value = event_target:brilliant_scientist_foreign_operation_actor.brilliant_scientist_foreign_selected_project_family }
			add_to_variable = { brilliant_scientist_foreign_operations_received = constant:brilliant_scientist_foreign_consequence.operation_increment }
			if = {
				limit = { check_variable = { event_target:brilliant_scientist_foreign_operation_actor.brilliant_scientist_foreign_last_operation_detected > constant:brilliant_scientist_value.zero } }
				add_to_variable = { brilliant_scientist_foreign_operations_detected = constant:brilliant_scientist_foreign_consequence.operation_increment }
			}
			brilliant_scientist_refresh_international_recognition_score = yes
			brilliant_scientist_try_fire_foreign_reaction = yes
		}
	}
}

# COUNTRY scope on the actor. The regular host target is the original target,
# including when a successful transfer has made it a former host.
brilliant_scientist_foreign_finish_operation = {
	if = {
		limit = { has_event_target = brilliant_scientist_foreign_operation_host }
		event_target:brilliant_scientist_foreign_operation_host = {
			if = {
				limit = { check_variable = { brilliant_scientist_foreign_incoming_operation_count > constant:brilliant_scientist_value.zero } }
				subtract_from_variable = { brilliant_scientist_foreign_incoming_operation_count = constant:brilliant_scientist_foreign_consequence.operation_increment }
			}
			remove_from_array = { array = brilliant_scientist_foreign_incoming_operation_actors value = ROOT }
			if = {
				limit = { check_variable = { ROOT.brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.assassination_attempt } }
				clr_country_flag = brilliant_scientist_foreign_assassination_attempt_live
			}
		}
		if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.observation } }
			add_to_array = { array = brilliant_scientist_foreign_observation_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.formal_invitation } }
			add_to_array = { array = brilliant_scientist_foreign_invitation_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.assistant_recruitment } }
			add_to_array = { array = brilliant_scientist_foreign_recruitment_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.archive_theft } }
			add_to_array = { array = brilliant_scientist_foreign_theft_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.project_sabotage } }
			add_to_array = { array = brilliant_scientist_foreign_sabotage_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.encourage_defection } }
			add_to_array = { array = brilliant_scientist_foreign_defection_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.extraction } }
			add_to_array = { array = brilliant_scientist_foreign_extraction_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.protection_offer } }
			add_to_array = { array = brilliant_scientist_foreign_protection_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.assassination_attempt } }
			add_to_array = { array = brilliant_scientist_foreign_assassination_resolved_targets value = event_target:brilliant_scientist_foreign_operation_host }
		}
	}
	clr_country_flag = brilliant_scientist_foreign_operation_live
	clr_country_flag = brilliant_scientist_foreign_operation_resolution_recorded
	clear_variable = brilliant_scientist_foreign_live_operation
	clear_variable = brilliant_scientist_foreign_operation_started_date
	clear_variable = brilliant_scientist_foreign_operation_host_id
	clear_variable = brilliant_scientist_foreign_selected_project_family
	clear_variable = brilliant_scientist_foreign_selected_project_stage
}

# Any scope with regular actor and host targets. Invalidated operations retain a
# cancellation ledger row and consume their per-host one-shot to prevent reroll
# exploits during transfer, annexation, or terminal cleanup.
brilliant_scientist_foreign_cancel_operation = {
	if = {
		limit = { has_event_target = brilliant_scientist_foreign_operation_actor }
		event_target:brilliant_scientist_foreign_operation_actor = {
			set_variable = { brilliant_scientist_foreign_last_operation_result = constant:brilliant_scientist_foreign_result.cancelled }
			set_variable = { brilliant_scientist_foreign_last_operation_detected = constant:brilliant_scientist_value.zero }
		}
		brilliant_scientist_foreign_record_resolution = yes
		event_target:brilliant_scientist_foreign_operation_actor = { brilliant_scientist_foreign_finish_operation = yes }
	}
}

# COUNTRY scope on the actor. Regular actor and host targets must exist. The
# result is written to persistent actor variables so events, ledgers, and later
# operations can inspect the exact score, roll, result band, and attribution.
brilliant_scientist_foreign_calculate_operation_outcome = {
	set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.observation_base }
	set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.observation_base }
	if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.assistant_recruitment } }
		set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.assistant_recruitment_base }
		set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.assistant_recruitment_base }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.archive_theft } }
		set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.archive_theft_base }
		set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.archive_theft_base }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.project_sabotage } }
		set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.project_sabotage_base }
		set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.project_sabotage_base }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.encourage_defection } }
		set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.encourage_defection_base }
		set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.encourage_defection_base }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.extraction } }
		set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.extraction_base }
		set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.extraction_base }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_foreign_live_operation = constant:brilliant_scientist_foreign_operation.assassination_attempt } }
		set_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.assassination_attempt_base }
		set_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.assassination_attempt_base }
	}

	if = {
		limit = { has_intelligence_agency = yes }
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.intelligence_agency }
		add_to_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.intelligence_agency }
	}
	if = {
		limit = {
			num_of_operatives > constant:brilliant_scientist_foreign_limit.minimum_operative_threshold
		}
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.operative }
		add_to_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.operative }
	}
	if = {
		limit = {
			num_of_operatives > constant:brilliant_scientist_foreign_limit.strong_operative_threshold
		}
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.strong_service }
		add_to_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.strong_service }
	}
	if = {
		limit = { is_major = yes }
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.major_power }
	}

	if = {
		limit = {
			has_intelligence_agency = yes
			network_strength = {
				target = event_target:brilliant_scientist_foreign_operation_host
				value > constant:brilliant_scientist_foreign_network.high
			}
		}
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.network_high }
		add_to_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.network_high }
	}
	else_if = {
		limit = {
			has_intelligence_agency = yes
			network_strength = {
				target = event_target:brilliant_scientist_foreign_operation_host
				value > constant:brilliant_scientist_foreign_network.medium
			}
		}
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.network_medium }
		add_to_temp_variable = { brilliant_scientist_foreign_detection_score = constant:brilliant_scientist_foreign_detection.network_medium }
	}
	else_if = {
		limit = {
			has_intelligence_agency = yes
			network_strength = {
				target = event_target:brilliant_scientist_foreign_operation_host
				value > constant:brilliant_scientist_foreign_network.low
			}
		}
		add_to_temp_variable = { brilliant_scientist_foreign_success_score = constant:brilliant_scientist_foreign_score.network_low }
