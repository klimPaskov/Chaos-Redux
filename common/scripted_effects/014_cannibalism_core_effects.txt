@CANNIBALISM_REINFECTION_PROTECTION_DAYS = 180

# ============================================================================
# CHAOS REDUX - EVENT 014 CANNIBALISM CORE EFFECTS
# ============================================================================
# Overview:
# - Selects the first wartime host and high-risk theater deterministically from
#   dynamic war, casualty, supply, stability, isolation, convoy, occupation,
#   manpower, and chaos factors.
# - Owns the canonical actor, node, recovery, and spread ledgers and the single
#   targeted pulse anchor. No broad periodic on_action is used.
# - Implements baseline country meters, state stages, exact finite population
#   consumption, local containment, external reinfection hooks, evolution
#   scheduling, global victory, world threat, and cleanup.
# - Warlord creation, unification, Wendigo convergence, country decisions, and
#   terminal packages build on this contract in their dedicated Event 014
#   effect files.
# ============================================================================

# ---------------------------------------------------------------------------
# Prefire context and deterministic target selection
# ---------------------------------------------------------------------------

cannibalism_clear_prefire_context = {
	clr_global_flag = cannibalism_prefire_pending_ready
	clr_global_flag = cannibalism_prefire_pending_evolution_i
	clr_global_flag = cannibalism_prefire_pending_evolution_ii
	clr_global_flag = cannibalism_prefire_pending_evolution_iii
	clear_variable = global.cannibalism_prefire_pending_generation
	clear_variable = global.cannibalism_prefire_pending_evolution
	clear_temp_array = cannibalism_host_scope_pool
	clear_temp_array = cannibalism_host_score_pool
	clear_temp_array = cannibalism_state_scope_pool
	clear_temp_array = cannibalism_state_score_pool
	set_temp_variable = { cannibalism_prefire_ready = 0 }
	set_temp_variable = { cannibalism_prefire_evolution = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_i_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_ii_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_iii_pending = 0 }
}

cannibalism_prepare_random_event_fire = {
	cannibalism_clear_prefire_context = yes
	if = {
		limit = { cannibalism_automatic_event_is_available = yes }
		cannibalism_select_first_host = yes
		if = {
			limit = { has_event_target = cannibalism_first_host }
			cannibalism_refresh_prefire_evolution_state = yes
			event_target:cannibalism_first_host = {
				cannibalism_select_highest_risk_state = yes
				if = {
					limit = { has_event_target = cannibalism_selected_state }
					event_target:cannibalism_selected_state = {
						save_event_target_as = cannibalism_initial_state
					}
					set_temp_variable = { cannibalism_prefire_ready = 1 }
				}
			}
		}
	}
	if = {
		limit = { check_variable = { cannibalism_prefire_ready < 1 } }
		cannibalism_clear_prefire_context = yes
	}
	else = {
		if = {
			limit = { NOT = { has_variable = global.cannibalism_prefire_generation } }
			set_variable = { global.cannibalism_prefire_generation = 0 }
		}
		if = {
			limit = { NOT = { has_variable = global.cannibalism_prefire_consumed_generation } }
			set_variable = { global.cannibalism_prefire_consumed_generation = 0 }
		}
		add_to_variable = { global.cannibalism_prefire_generation = 1 }
		set_temp_variable = { cannibalism_prefire_generation = global.cannibalism_prefire_generation }
		set_variable = { global.cannibalism_prefire_pending_generation = cannibalism_prefire_generation }
		set_variable = { global.cannibalism_prefire_pending_evolution = cannibalism_prefire_evolution }
		set_global_flag = cannibalism_prefire_pending_ready
		if = {
			limit = { check_variable = { cannibalism_prefire_evolution_i_enabled > 0 } }
			set_global_flag = cannibalism_prefire_pending_evolution_i
		}
		if = {
			limit = { check_variable = { cannibalism_prefire_evolution_ii_enabled > 0 } }
			set_global_flag = cannibalism_prefire_pending_evolution_ii
		}
		if = {
			limit = { check_variable = { cannibalism_prefire_evolution_iii_pending > 0 } }
			set_global_flag = cannibalism_prefire_pending_evolution_iii
		}
	}
}

cannibalism_refresh_prefire_evolution_state = {
	set_temp_variable = { cannibalism_prefire_evolution = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_i_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_ii_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_iii_pending = 0 }

	set_temp_variable = { events_log_evolution_event_id = constant:cannibalism_event.id }
	set_temp_variable = { events_log_evolution_type = constant:cannibalism_event.evolution_type }
	set_temp_variable = { events_log_evolution_tier = constant:cannibalism_event.evolution_i_tier }
	set_temp_variable = { events_log_evolution_stage = constant:cannibalism_event.evolution_i_stage }
	if = {
		limit = { is_current_evolution_enabled = yes }
		set_temp_variable = { cannibalism_prefire_evolution_i_enabled = 1 }
		set_temp_variable = { cannibalism_prefire_evolution = constant:cannibalism_event.evolution_i_stage }
	}

	set_temp_variable = { events_log_evolution_tier = constant:cannibalism_event.evolution_ii_tier }
	set_temp_variable = { events_log_evolution_stage = constant:cannibalism_event.evolution_ii_stage }
	if = {
		limit = { is_current_evolution_enabled = yes }
		set_temp_variable = { cannibalism_prefire_evolution_ii_enabled = 1 }
		set_temp_variable = { cannibalism_prefire_evolution = constant:cannibalism_event.evolution_ii_stage }
	}

	set_temp_variable = { events_log_evolution_tier = constant:cannibalism_event.evolution_iii_tier }
	set_temp_variable = { events_log_evolution_stage = constant:cannibalism_event.evolution_iii_stage }
	if = {
		limit = {
			is_current_evolution_enabled = yes
			check_variable = { cannibalism_prefire_evolution_ii_enabled > 0 }
			cannibalism_prefire_evolution_iii_capacity_is_available = yes
		}
		set_temp_variable = { cannibalism_prefire_evolution_iii_pending = 1 }
	}
}

cannibalism_score_current_host = {
	save_event_target_as = cannibalism_scored_host
	set_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.base }

	if = {
		limit = { is_ai = no }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.player }
	}
	if = {
		limit = { longest_war_length > constant:cannibalism_host_threshold.extreme_war_months }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.extreme_war }
	}
	else_if = {
		limit = { longest_war_length > constant:cannibalism_host_threshold.long_war_months }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.long_war }
	}
	else_if = {
		limit = { longest_war_length > constant:cannibalism_host_threshold.short_war_months }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.short_war }
	}

	if = {
		limit = { has_stability < constant:cannibalism_host_threshold.critical_stability }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.critical_stability }
	}
	else_if = {
		limit = { has_stability < constant:cannibalism_host_threshold.low_stability }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.low_stability }
	}
	else_if = {
		limit = { has_stability > constant:cannibalism_host_threshold.high_stability }
		subtract_from_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.high_stability_penalty }
	}
	if = {
		limit = { has_war_support < constant:cannibalism_host_threshold.low_war_support }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.low_war_support }
	}

	set_temp_variable = { cannibalism_host_casualty_ratio = 0 }
	if = {
		limit = { check_variable = { max_manpower_k > 0 } }
		set_temp_variable = { cannibalism_host_casualty_ratio = casualties_k }
		divide_temp_variable = { cannibalism_host_casualty_ratio = max_manpower_k }
	}
	if = {
		limit = {
			check_variable = {
				var = cannibalism_host_casualty_ratio
				value = constant:cannibalism_host_threshold.casualty_ratio_extreme
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.casualties_extreme }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cannibalism_host_casualty_ratio
				value = constant:cannibalism_host_threshold.casualty_ratio_high
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.casualties_high }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cannibalism_host_casualty_ratio
				value = constant:cannibalism_host_threshold.casualty_ratio_medium
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.casualties_medium }
	}

	set_temp_variable = { cannibalism_host_manpower_ratio = 0 }
	if = {
		limit = { check_variable = { max_manpower_k > 0 } }
		set_temp_variable = { cannibalism_host_manpower_ratio = manpower_k }
		divide_temp_variable = { cannibalism_host_manpower_ratio = max_manpower_k }
	}
	if = {
		limit = {
			check_variable = {
				var = cannibalism_host_manpower_ratio
				value = constant:cannibalism_host_threshold.manpower_ratio_critical
				compare = less_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.critical_manpower }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cannibalism_host_manpower_ratio
				value = constant:cannibalism_host_threshold.manpower_ratio_low
				compare = less_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.low_manpower }
	}

	if = {
		limit = { cannibalism_country_has_isolated_theater = yes }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.isolation }
	}
	if = {
		limit = { cannibalism_country_has_damaged_supply_theater = yes }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.damaged_supply }
	}
	if = {
		limit = { cannibalism_country_has_remote_army = yes }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.encircled_or_remote_army }
	}
	if = {
		limit = { cannibalism_country_has_convoy_pressure = yes }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.convoy_losses }
		if = {
			limit = { convoy_threat > constant:cannibalism_host_threshold.convoy_threat_high }
			add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.convoy_high }
		}
		else_if = {
			limit = { convoy_threat > constant:cannibalism_host_threshold.convoy_threat_medium }
			add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.convoy_medium }
		}
	}
	if = {
		limit = { cannibalism_country_has_external_hunger_pressure = yes }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_integration.host_shortage_score }
	}

	set_temp_variable = { cannibalism_host_occupation_count = 0 }
	every_controlled_state = {
		limit = {
			NOT = { is_core_of = event_target:cannibalism_scored_host }
		}
		add_to_temp_variable = { cannibalism_host_occupation_count = 1 }
	}
	if = {
		limit = { check_variable = { cannibalism_host_occupation_count > 0 } }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.occupied_state }
	}
	if = {
		limit = {
			check_variable = {
				var = cannibalism_host_occupation_count
				value = constant:cannibalism_host_threshold.large_occupation
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.occupied_state }
	}
	if = {
		limit = { num_divisions > constant:cannibalism_host_threshold.large_army }
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.large_army }
	}

	if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_final.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.chaos_final }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_4.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.chaos_tier_4 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_3.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.chaos_tier_3 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_2.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.chaos_tier_2 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_1.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_host_score = constant:cannibalism_host_weight.chaos_tier_1 }
	}

	round_temp_variable = cannibalism_host_score
	clamp_temp_variable = {
		var = cannibalism_host_score
		min = constant:cannibalism_host_weight.minimum
		max = constant:cannibalism_host_weight.maximum
	}
}

cannibalism_select_first_host = {
	clear_temp_array = cannibalism_host_scope_pool
	clear_temp_array = cannibalism_host_score_pool
	every_country = {
		limit = { cannibalism_can_be_origin = yes }
		cannibalism_score_current_host = yes
		add_to_temp_array = { cannibalism_host_scope_pool = THIS }
		add_to_temp_array = { cannibalism_host_score_pool = cannibalism_host_score }
	}
	if = {
		limit = { check_variable = { cannibalism_host_score_pool^num > 0 } }
		find_highest_in_array = {
			array = cannibalism_host_score_pool
			value = cannibalism_best_host_score
			index = cannibalism_best_host_index
		}
		set_temp_variable = { cannibalism_host_pool_iteration = 0 }
		set_temp_variable = { cannibalism_host_selected = 0 }
		for_each_scope_loop = {
			array = cannibalism_host_scope_pool
			if = {
				limit = {
					check_variable = { cannibalism_host_selected < 1 }
					check_variable = {
						var = cannibalism_host_pool_iteration
						value = cannibalism_best_host_index
						compare = equals
					}
				}
				save_event_target_as = cannibalism_first_host
				set_temp_variable = { cannibalism_host_selected = 1 }
			}
			add_to_temp_variable = { cannibalism_host_pool_iteration = 1 }
		}
	}
	clear_temp_array = cannibalism_host_scope_pool
	clear_temp_array = cannibalism_host_score_pool
}

cannibalism_score_current_state = {
	set_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.base }
	if = {
		limit = { cannibalism_state_has_army_presence = yes }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.army_presence }
	}
	if = {
		limit = { is_island_state = yes }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.island }
	}
	else_if = {
		limit = { NOT = { is_in_home_area = yes } }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.remote_from_capital }
	}
	if = {
		limit = { infrastructure < constant:cannibalism_state_threshold.critical_infrastructure }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.critical_infrastructure }
	}
	else_if = {
		limit = { infrastructure < constant:cannibalism_state_threshold.low_infrastructure }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.low_infrastructure }
	}
	if = {
		limit = { has_damaged_buildings = yes }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.damaged_port }
	}
	if = {
		limit = { naval_base > 0 }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.port }
	}
	if = {
		limit = { supply_node > 0 }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.supply_node }
	}
	if = {
		limit = { NOT = { is_core_of = event_target:cannibalism_state_host } }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.occupied }
	}
	if = {
		limit = { cannibalism_state_has_prison_or_camp = yes }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.prison_or_camp }
	}
	if = {
		limit = { cannibalism_state_has_external_hunger_pressure = yes }
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_integration.state_shortage_score }
	}
	if = {
		limit = {
			check_variable = {
				var = state_population_k
				value = constant:cannibalism_state_threshold.very_dense_population_k
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.very_dense_population }
	}
	else_if = {
		limit = {
			check_variable = {
				var = state_population_k
				value = constant:cannibalism_state_threshold.dense_population_k
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { cannibalism_state_score = constant:cannibalism_state_weight.dense_population }

	}
	round_temp_variable = cannibalism_state_score

		max = constant:cannibalism_state_weight.maximum

	}
	else = {
		for_each_scope_loop = {

			array = global.cannibalism_recovery_states

cannibalism_process_current_actor_pulse = {
	if = {
		limit = {
			cannibalism_country_can_continue_pulse = yes
			OR = {
				NOT = { has_variable = cannibalism_last_processed_pulse_sequence }
				check_variable = {
					var = cannibalism_last_processed_pulse_sequence
					value = global.cannibalism_pulse_sequence
					compare = less_than
				}
			}
		}
		set_variable = { cannibalism_last_processed_pulse_sequence = global.cannibalism_pulse_sequence }

		if = {
			limit = { has_war = yes }
			add_to_variable = { cannibalism_field_hunger = constant:cannibalism_delta.weekly_hunger_war }
		}
		else = {
			add_to_variable = { cannibalism_field_hunger = constant:cannibalism_delta.weekly_hunger_peace }
			add_to_variable = { cannibalism_frenzy = constant:cannibalism_consumption.frenzy_peace }
		}
		cannibalism_apply_external_pressure_to_current_actor = yes

		if = {
			limit = { has_country_flag = cannibalism_policy_open_emergency }
			add_to_variable = { cannibalism_command_integrity = constant:cannibalism_delta.weekly_integrity_open }
		}
		else_if = {
			limit = { has_country_flag = cannibalism_policy_concealment }
			add_to_variable = { cannibalism_command_integrity = constant:cannibalism_delta.weekly_integrity_conceal }
		}
		else_if = {
			limit = { has_country_flag = cannibalism_policy_exploitation }
			add_to_variable = { cannibalism_command_integrity = constant:cannibalism_delta.weekly_integrity_exploit }
			if = {
				limit = { has_variable = cannibalism_cult_cohesion }
				add_to_variable = { cannibalism_cult_cohesion = constant:cannibalism_delta.weekly_cult_exploit }
			}
		}

		if = {
			limit = { cannibalism_country_has_active_state_cell = yes }
			add_to_variable = { cannibalism_cell_strength = constant:cannibalism_delta.weekly_cell_unchecked }
		}
		else = {
			add_to_variable = { cannibalism_cell_strength = constant:cannibalism_delta.weekly_cell_recovery }
			if = {
				limit = {
					has_variable = cannibalism_cult_cohesion
					check_variable = {
						var = cannibalism_command_integrity
						value = constant:cannibalism_meter.active
						compare = greater_than_or_equals
					}
				}
				add_to_variable = { cannibalism_cult_cohesion = constant:cannibalism_delta.weekly_cult_recovery }
			}
		}

		every_controlled_state = {
			limit = {
				cannibalism_state_has_active_cell = yes
				cannibalism_state_has_any_larder_contamination = yes
			}
			cannibalism_process_current_state_contamination_against_cell = yes
		}

		if = {
			limit = { check_variable = { cannibalism_larder > 0 } }
			set_temp_variable = { cannibalism_larder_decay = constant:cannibalism_consumption.larder_weekly_decay_base }
			set_temp_variable = { cannibalism_larder_decay_component = num_of_controlled_states }
			multiply_temp_variable = { cannibalism_larder_decay_component = constant:cannibalism_consumption.larder_weekly_decay_per_state }
			add_to_temp_variable = { cannibalism_larder_decay = cannibalism_larder_decay_component }
			set_temp_variable = { cannibalism_larder_decay_component = num_divisions }
			multiply_temp_variable = { cannibalism_larder_decay_component = constant:cannibalism_consumption.larder_weekly_decay_per_division }
			add_to_temp_variable = { cannibalism_larder_decay = cannibalism_larder_decay_component }
			set_temp_variable = { cannibalism_larder_decay_negative = cannibalism_larder_decay }
			multiply_temp_variable = { cannibalism_larder_decay_negative = -1 }
			add_to_variable = { cannibalism_larder = cannibalism_larder_decay_negative }
		}
		if = {
			limit = {
				is_cannibalism_country = yes
				check_variable = {
					var = cannibalism_larder
					value = constant:cannibalism_consumption.larder_starving
					compare = less_than_or_equals
				}
			}
			add_to_variable = { cannibalism_frenzy = constant:cannibalism_consumption.frenzy_starvation }
		}

		clamp_variable = { var = cannibalism_field_hunger min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		clamp_variable = { var = cannibalism_command_integrity min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		clamp_variable = { var = cannibalism_cell_strength min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		if = {
			limit = { has_variable = cannibalism_cult_cohesion }
			clamp_variable = { var = cannibalism_cult_cohesion min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		}
		clamp_variable = { var = cannibalism_larder min = constant:cannibalism_consumption.larder_min max = constant:cannibalism_consumption.larder_max }
		clamp_variable = { var = cannibalism_frenzy min = constant:cannibalism_consumption.frenzy_min max = constant:cannibalism_consumption.frenzy_max }
		if = {
			limit = { has_variable = cannibalism_network_alignment }
			clamp_variable = { var = cannibalism_network_alignment min = constant:cannibalism_consumption.alignment_min max = constant:cannibalism_consumption.alignment_max }
		}
		cannibalism_refresh_country_penalty_variables = yes
		cannibalism_schedule_pending_evolution_reports = yes
		cannibalism_process_pending_evolution_reports = yes
		cannibalism_process_current_country_mission_objectives = yes
		cannibalism_process_current_country_objectives = yes
		if = {
			limit = { is_cannibalism_wendigo_hannibal_country = yes }
			cannibalism_process_wendigo_transformation_pulse = yes
		}
		cannibalism_try_enqueue_current_actor_border_spread = yes

		if = {
			limit = { cannibalism_country_can_begin_stabilization = yes }
			if = {
				limit = { NOT = { has_variable = cannibalism_local_victory_due_date } }
				set_variable = { cannibalism_local_victory_due_date = global.date }
				add_to_variable = { cannibalism_local_victory_due_date = constant:cannibalism_timing.stabilization_days }
			}
			else_if = {
				limit = {
					check_variable = {
						var = cannibalism_local_victory_due_date
						value = global.date
						compare = less_than_or_equals
					}
				}
				cannibalism_complete_current_country_local_victory = yes
			}
		}
		else = {
			clear_variable = cannibalism_local_victory_due_date
		}
	}
}

		limit = {
			has_state_flag = cannibalism_active_node
			has_variable = cannibalism_node_id
			has_variable = cannibalism_node_generation
			OR = {

		set_variable = { cannibalism_last_processed_pulse_sequence = global.cannibalism_pulse_sequence }
		set_temp_variable = { cannibalism_node_pulse_delta = 0 }
		set_temp_variable = { cannibalism_node_source_is_live = 0 }

		if = {
			limit = {
				has_variable = cannibalism_node_source_country
				has_variable = cannibalism_node_source_country_generation
				country_exists = var:cannibalism_node_source_country
			}
			set_temp_variable = { cannibalism_expected_source_generation = cannibalism_node_source_country_generation }
			var:cannibalism_node_source_country = {
				if = {
					limit = {
						is_cannibalism_active_country = yes
						check_variable = {
							var = cannibalism_actor_generation
							value = cannibalism_expected_source_generation
							compare = equals
						}
					}
					set_temp_variable = { cannibalism_node_source_is_live = 1 }
				}
			}
		}
		if = {
			limit = { check_variable = { cannibalism_node_source_is_live > 0 } }
			add_to_temp_variable = { cannibalism_node_pulse_delta = constant:cannibalism_delta.weekly_node_growth }
		}
		if = {
			limit = { cannibalism_state_is_isolated = yes }
			add_to_temp_variable = { cannibalism_node_pulse_delta = constant:cannibalism_delta.weekly_network_node }
		}
		if = {
			limit = {
				CONTROLLER = {
					is_cannibalism_active_ordinary_country = yes
					check_variable = {
						var = cannibalism_command_integrity
						value = constant:cannibalism_meter.active
						compare = greater_than_or_equals
					}
				}
			}

				check_variable = {
					var = cannibalism_node_strength

					compare = less_than_or_equals
				}
			}
			cannibalism_mark_current_state_for_recovery = yes
		}
		else_if = {
			limit = {
				check_variable = {
					var = global.cannibalism_global_stage
					value = constant:cannibalism_global_stage.network
					compare = greater_than_or_equals
				}

					compare = greater_than_or_equals

			set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.compromised_garrison }

		}

		if = {

			limit = { has_global_flag = cannibalism_prefire_pending_evolution_i }

		}
		if = {
			limit = { has_global_flag = cannibalism_prefire_pending_evolution_iii }
			set_temp_variable = { cannibalism_prefire_evolution_iii_pending = 1 }
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_prefire_ready > 0 }
			has_event_target = cannibalism_first_host
			has_event_target = cannibalism_initial_state
			check_variable = {
				var = cannibalism_prefire_generation
				value = global.cannibalism_prefire_consumed_generation
				compare = greater_than
			}
			event_target:cannibalism_first_host = {
				exists = yes
				cannibalism_can_be_origin = yes
				event_target:cannibalism_initial_state = {
					is_controlled_by = PREV
					cannibalism_state_is_valid_origin = yes
				}
			}
		}
		cannibalism_initialize_global_runtime = yes
		event_target:cannibalism_first_host = {
			cannibalism_initialize_current_country = yes
			if = {
				limit = { check_variable = { cannibalism_actor_register_result > 0 } }
				set_variable = { global.cannibalism_prefire_consumed_generation = cannibalism_prefire_generation }
				country_event = { id = chaosx.nr14.2 }
				set_temp_variable = { cannibalism_begin_result = 1 }
			}
		}
	}
	clr_global_flag = cannibalism_prefire_pending_ready
	clr_global_flag = cannibalism_prefire_pending_evolution_i
	clr_global_flag = cannibalism_prefire_pending_evolution_ii
	clr_global_flag = cannibalism_prefire_pending_evolution_iii
	clear_variable = global.cannibalism_prefire_pending_generation
	clear_variable = global.cannibalism_prefire_pending_evolution
}

	set_temp_variable = { cannibalism_population_loss_applied = 0 }
	set_temp_variable = { cannibalism_larder_gained = 0 }

	if = {
		limit = {
			has_event_target = cannibalism_consumption_actor
			event_target:cannibalism_consumption_actor = {
				exists = yes
				is_cannibalism_active_country = yes
			}
			check_variable = { cannibalism_consumption_request_id > 0 }
		}
		if = {
			limit = {
				has_variable = cannibalism_last_consumption_request_id
				check_variable = {
					var = cannibalism_last_consumption_request_id
					value = cannibalism_consumption_request_id
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.duplicate_request }
		}
		else_if = {
			limit = { NOT = { cannibalism_can_consume_current_state = yes } }
			set_temp_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.blocked_state }
		}
		else = {
			set_temp_variable = { cannibalism_consumption_diminishing_factor = constant:cannibalism_consumption.first_action_yield_factor }
			if = {
				limit = {
					has_variable = cannibalism_consumption_actions
					check_variable = { cannibalism_consumption_actions > constant:cannibalism_consumption.exhausted_action_prior_actions }
				}
				set_temp_variable = { cannibalism_consumption_diminishing_factor = constant:cannibalism_consumption.exhausted_action_yield_factor }
			}
			else_if = {
				limit = { check_variable = { cannibalism_consumption_actions > constant:cannibalism_consumption.fourth_action_prior_actions } }
				set_temp_variable = { cannibalism_consumption_diminishing_factor = constant:cannibalism_consumption.fourth_action_yield_factor }
			}
			else_if = {
				limit = { check_variable = { cannibalism_consumption_actions > constant:cannibalism_consumption.third_action_prior_actions } }
				set_temp_variable = { cannibalism_consumption_diminishing_factor = constant:cannibalism_consumption.third_action_yield_factor }
			}
			else_if = {
				limit = { check_variable = { cannibalism_consumption_actions > constant:cannibalism_consumption.second_action_prior_actions } }
				set_temp_variable = { cannibalism_consumption_diminishing_factor = constant:cannibalism_consumption.second_action_yield_factor }
			}

			set_temp_variable = { cannibalism_consumption_usable_factor = cannibalism_consumption_diminishing_factor }
			if = {
				limit = { cannibalism_state_has_bioweapon_contamination = yes }
				multiply_temp_variable = { cannibalism_consumption_usable_factor = constant:cannibalism_consumption.contamination_yield_factor }
			}
			if = {
				limit = { cannibalism_state_has_chemical_contamination = yes }
				multiply_temp_variable = { cannibalism_consumption_usable_factor = constant:cannibalism_integration.chemical_larder_yield_factor }
			}
			if = {
				limit = {
					has_variable = global.air_contamination_bp
					check_variable = {
						var = global.air_contamination_bp
						value = constant:air_contamination_threshold_bp.winter_75
						compare = greater_than_or_equals
					}
				}
				multiply_temp_variable = { cannibalism_consumption_usable_factor = constant:cannibalism_integration.severe_air_larder_yield_factor }
			}

			if = {
				limit = { check_variable = { cannibalism_consumption_exact_request_people > 0 } }
				set_temp_variable = { cannibalism_consumption_population_requested = cannibalism_consumption_exact_request_people }
			}
			else = {
				set_temp_variable = { cannibalism_consumption_population_available = state_population_k }
				multiply_temp_variable = { cannibalism_consumption_population_available = constant:cannibalism_consumption.population_people_per_k }
				set_temp_variable = { cannibalism_consumption_population_requested = cannibalism_consumption_population_available }
				multiply_temp_variable = { cannibalism_consumption_population_requested = cannibalism_consumption_base_percent }
				multiply_temp_variable = { cannibalism_consumption_population_requested = cannibalism_consumption_usable_factor }
				clamp_temp_variable = {
					var = cannibalism_consumption_population_requested
					min = 0
					max = cannibalism_consumption_hard_cap
				}
				round_temp_variable = cannibalism_consumption_population_requested
			}

			set_temp_variable = { state_civilian_population_loss_requested = cannibalism_consumption_population_requested }
			set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:cannibalism_consumption.minimum_remaining_people }
			set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.cannibalism_consumption }
			set_temp_variable = { state_civilian_population_loss_log_deaths = 1 }
			if = {
				limit = { check_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.prisoner_feeding } }
				set_temp_variable = { state_civilian_population_loss_log_deaths = 0 }
			}
			set_temp_variable = { state_civilian_population_loss_target_country = event_target:cannibalism_consumption_actor }
			set_temp_variable = { state_civilian_population_loss_has_target_country = 1 }
			set_temp_variable = { state_civilian_population_loss_applied = constant:chaos_meter_deaths.zero }
			set_temp_variable = { state_civilian_population_loss_result = constant:chaos_meter_deaths.zero }
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.one }
			apply_exact_state_civilian_population_loss = yes
			set_temp_variable = { cannibalism_population_loss_applied = state_civilian_population_loss_applied }
			if = {
				limit = {
					check_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.prisoner_feeding }
					check_variable = { cannibalism_population_loss_applied > 0 }
				}
				cannibalism_log_mixed_prisoner_feeding_deaths = yes
			}

			if = {
				limit = { check_variable = { cannibalism_population_loss_applied > 0 } }
				set_variable = { cannibalism_last_consumption_request_id = cannibalism_consumption_request_id }
				add_to_variable = { cannibalism_consumption_actions = 1 }
				add_to_variable = { global.cannibalism_consumption_sequence = 1 }
				set_variable = { cannibalism_last_consumption_sequence = global.cannibalism_consumption_sequence }
				add_to_variable = { global.cannibalism_consumed_population = cannibalism_population_loss_applied }
				set_variable = { global.cannibalism_consumed_population_k = global.cannibalism_consumed_population }
				divide_variable = { global.cannibalism_consumed_population_k = constant:cannibalism_consumption.population_people_per_k }

				set_temp_variable = { cannibalism_larder_gained = cannibalism_population_loss_applied }
				divide_temp_variable = { cannibalism_larder_gained = constant:cannibalism_consumption.population_people_per_k }
				multiply_temp_variable = { cannibalism_larder_gained = cannibalism_consumption_larder_factor }
				if = {
					limit = { check_variable = { cannibalism_consumption_exact_request_people > 0 } }
					multiply_temp_variable = { cannibalism_larder_gained = cannibalism_consumption_usable_factor }
				}
				event_target:cannibalism_consumption_actor = {
					if = {
						limit = {
							has_country_flag = cannibalism_larder_mobile
							has_variable = cannibalism_warlord_mobile_larder_yield_factor
						}
						multiply_temp_variable = { cannibalism_larder_gained = cannibalism_warlord_mobile_larder_yield_factor }
					}
					else_if = {
						limit = { has_variable = cannibalism_warlord_consumption_yield_factor }
						multiply_temp_variable = { cannibalism_larder_gained = cannibalism_warlord_consumption_yield_factor }
					}
				}
				clamp_temp_variable = {
					var = cannibalism_larder_gained
					min = 0
					max = cannibalism_consumption_larder_cap
				}

				set_temp_variable = { cannibalism_consumption_cooldown_days = constant:cannibalism_timing.consumption_cooldown_days }
				set_state_flag = { flag = cannibalism_consumption_cooldown days = cannibalism_consumption_cooldown_days }
				set_temp_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied }

				if = {
					limit = {
						check_variable = {
							var = cannibalism_consumption_actions
							value = constant:cannibalism_consumption.state_silent_actions
							compare = greater_than_or_equals
						}
						check_variable = { cannibalism_state_stage < constant:cannibalism_state_stage.silent_larder }
					}
					set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.silent_larder }
					cannibalism_set_current_state_stage = yes
				}
				else_if = {
					limit = {
						check_variable = {
							var = cannibalism_consumption_actions
							value = constant:cannibalism_consumption.state_feeding_actions
							compare = greater_than_or_equals
						}
						check_variable = { cannibalism_state_stage < constant:cannibalism_state_stage.feeding_state }
					}
					set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.feeding_state }
					cannibalism_set_current_state_stage = yes
				}
				else_if = {
					limit = {
						check_variable = {
							var = cannibalism_consumption_actions
							value = constant:cannibalism_consumption.state_hunting_actions
							compare = greater_than_or_equals
						}
						check_variable = { cannibalism_state_stage < constant:cannibalism_state_stage.hunting_ground }
					}
					set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.hunting_ground }
					cannibalism_set_current_state_stage = yes
				}

				event_target:cannibalism_consumption_actor = {
					if = {
						limit = {
							OR = {
								is_cannibalism_unified_country = yes
								is_cannibalism_wendigo_hannibal_country = yes
							}
						}
						add_to_variable = { cannibalism_larder = cannibalism_larder_gained }
						clamp_variable = { var = cannibalism_larder min = constant:cannibalism_consumption.larder_min max = constant:cannibalism_unification.global_larder_maximum }
						set_variable = { global.cannibalism_global_larder = cannibalism_larder }
					}
					else = {
						add_to_variable = { cannibalism_larder = cannibalism_larder_gained }
						clamp_variable = { var = cannibalism_larder min = constant:cannibalism_consumption.larder_min max = constant:cannibalism_consumption.larder_max }
					}
					add_to_variable = { cannibalism_consumed_population = cannibalism_population_loss_applied }
					set_variable = { cannibalism_consumed_population_k = cannibalism_consumed_population }
					divide_variable = { cannibalism_consumed_population_k = constant:cannibalism_consumption.population_people_per_k }
					add_to_variable = { cannibalism_frenzy = constant:cannibalism_consumption.frenzy_consumption }
					add_to_variable = { cannibalism_field_hunger = constant:cannibalism_delta.consumption_hunger }
					add_to_variable = { cannibalism_command_integrity = constant:cannibalism_delta.consumption_integrity }
					clamp_variable = { var = cannibalism_frenzy min = constant:cannibalism_consumption.frenzy_min max = constant:cannibalism_consumption.frenzy_max }
					clamp_variable = { var = cannibalism_field_hunger min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
					clamp_variable = { var = cannibalism_command_integrity min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
					cannibalism_refresh_country_penalty_variables = yes
				}
			}
			else = {
				set_temp_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.population_exhausted }
			}
		}
	}
}

	set_variable = { cannibalism_modifier_stability = cannibalism_field_hunger }

	multiply_variable = { cannibalism_modifier_stability = constant:cannibalism_country_penalty.hunger_stability_scale }
	set_variable = { cannibalism_modifier_component = cannibalism_cell_strength }
	divide_variable = { cannibalism_modifier_component = constant:cannibalism_meter.max }
	multiply_variable = { cannibalism_modifier_component = constant:cannibalism_country_penalty.cell_stability_scale }
	add_to_variable = { cannibalism_modifier_stability = cannibalism_modifier_component }

	set_variable = { cannibalism_modifier_war_support = cannibalism_field_hunger }
	divide_variable = { cannibalism_modifier_war_support = constant:cannibalism_meter.max }
	multiply_variable = { cannibalism_modifier_war_support = constant:cannibalism_country_penalty.hunger_war_support_scale }

	set_variable = { cannibalism_modifier_integrity_deficit = constant:cannibalism_meter.max }
	subtract_from_variable = { cannibalism_modifier_integrity_deficit = cannibalism_command_integrity }
	clamp_variable = {
		var = cannibalism_modifier_integrity_deficit

	multiply_variable = { cannibalism_modifier_supply_consumption = constant:cannibalism_country_penalty.hunger_supply_scale }

	set_variable = { cannibalism_modifier_planning = cannibalism_modifier_integrity_deficit }
	divide_variable = { cannibalism_modifier_planning = constant:cannibalism_meter.max }
	multiply_variable = { cannibalism_modifier_planning = constant:cannibalism_country_penalty.integrity_deficit_planning_scale }

	set_variable = { cannibalism_modifier_political_power = cannibalism_cell_strength }
	divide_variable = { cannibalism_modifier_political_power = constant:cannibalism_meter.max }
	multiply_variable = { cannibalism_modifier_political_power = constant:cannibalism_country_penalty.cell_political_power_scale }

	set_variable = { cannibalism_modifier_recruitable_population = 0 }
	if = {
		limit = { has_variable = cannibalism_cult_cohesion }
		set_variable = { cannibalism_modifier_recruitable_population = cannibalism_cult_cohesion }
		divide_variable = { cannibalism_modifier_recruitable_population = constant:cannibalism_meter.max }
		multiply_variable = { cannibalism_modifier_recruitable_population = constant:cannibalism_country_penalty.cult_recruitable_scale }
	}

	clamp_variable = { var = cannibalism_modifier_stability min = constant:cannibalism_country_penalty.stability_min max = constant:cannibalism_country_penalty.stability_max }
	clamp_variable = { var = cannibalism_modifier_war_support min = constant:cannibalism_country_penalty.war_support_min max = constant:cannibalism_country_penalty.war_support_max }
	clamp_variable = { var = cannibalism_modifier_army_organization min = constant:cannibalism_country_penalty.organization_min max = constant:cannibalism_country_penalty.organization_max }
	clamp_variable = { var = cannibalism_modifier_supply_consumption min = constant:cannibalism_country_penalty.supply_min max = constant:cannibalism_country_penalty.supply_max }
	clamp_variable = { var = cannibalism_modifier_planning min = constant:cannibalism_country_penalty.planning_min max = constant:cannibalism_country_penalty.planning_max }
	clamp_variable = { var = cannibalism_modifier_political_power min = constant:cannibalism_country_penalty.political_power_min max = constant:cannibalism_country_penalty.political_power_max }
	clamp_variable = { var = cannibalism_modifier_recruitable_population min = constant:cannibalism_country_penalty.recruitable_min max = constant:cannibalism_country_penalty.recruitable_max }
	clear_variable = cannibalism_modifier_component
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = cannibalism_frontline_crisis } } }
		add_dynamic_modifier = { modifier = cannibalism_frontline_crisis }
	}
}

cannibalism_clear_current_state_modifier = {
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_suspected_predation_state } } remove_dynamic_modifier = { modifier = cannibalism_suspected_predation_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_compromised_garrison_state } } remove_dynamic_modifier = { modifier = cannibalism_compromised_garrison_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_hunting_ground_state } } remove_dynamic_modifier = { modifier = cannibalism_hunting_ground_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_feeding_state } } remove_dynamic_modifier = { modifier = cannibalism_feeding_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_silent_larder_state } } remove_dynamic_modifier = { modifier = cannibalism_silent_larder_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_liberated_emergency_state } } remove_dynamic_modifier = { modifier = cannibalism_liberated_emergency_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_identification_recovery_state } } remove_dynamic_modifier = { modifier = cannibalism_identification_recovery_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_institutional_recovery_state } } remove_dynamic_modifier = { modifier = cannibalism_institutional_recovery_state } }
	if = { limit = { has_dynamic_modifier = { modifier = cannibalism_long_trauma_state } } remove_dynamic_modifier = { modifier = cannibalism_long_trauma_state } }
	clr_state_flag = cannibalism_suspected_predation
	clr_state_flag = cannibalism_compromised_garrison
	clr_state_flag = cannibalism_hunting_ground
	clr_state_flag = cannibalism_feeding_state
	clr_state_flag = cannibalism_silent_larder
	clr_state_flag = cannibalism_liberated_emergency
	clr_state_flag = cannibalism_identification_recovery
	clr_state_flag = cannibalism_institutional_recovery
	clr_state_flag = cannibalism_long_trauma
	clr_state_flag = cannibalism_state_stabilized
	set_variable = { cannibalism_state_modifier_supplies = constant:cannibalism_state_modifier.none }
	set_variable = { cannibalism_state_modifier_production = constant:cannibalism_state_modifier.none }
	set_variable = { cannibalism_state_modifier_resources = constant:cannibalism_state_modifier.none }
	set_variable = { cannibalism_state_modifier_manpower = constant:cannibalism_state_modifier.none }
	set_variable = { cannibalism_state_modifier_resistance = constant:cannibalism_state_modifier.none }
}

	cannibalism_achievement_record_current_state_as_former_feeding_state = yes
	cannibalism_clear_current_state_modifier = yes
	set_variable = { cannibalism_state_stage = cannibalism_new_state_stage }

		limit = { check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.feeding_state } }
		set_state_flag = cannibalism_feeding_state
		set_variable = { cannibalism_state_modifier_supplies = constant:cannibalism_state_modifier.feeding_supplies }
		set_variable = { cannibalism_state_modifier_production = constant:cannibalism_state_modifier.feeding_production }
		set_variable = { cannibalism_state_modifier_resources = constant:cannibalism_state_modifier.feeding_resources }
		set_variable = { cannibalism_state_modifier_manpower = constant:cannibalism_state_modifier.feeding_manpower }
		set_variable = { cannibalism_state_modifier_resistance = constant:cannibalism_state_modifier.feeding_resistance }
		add_dynamic_modifier = { modifier = cannibalism_feeding_state }
	}
	else_if = {
		limit = { check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.silent_larder } }
		set_state_flag = cannibalism_silent_larder
		set_variable = { cannibalism_state_modifier_supplies = constant:cannibalism_state_modifier.silent_supplies }
		set_variable = { cannibalism_state_modifier_production = constant:cannibalism_state_modifier.silent_production }
		set_variable = { cannibalism_state_modifier_resources = constant:cannibalism_state_modifier.silent_resources }
		set_variable = { cannibalism_state_modifier_manpower = constant:cannibalism_state_modifier.silent_manpower }
		set_variable = { cannibalism_state_modifier_resistance = constant:cannibalism_state_modifier.silent_resistance }
		add_dynamic_modifier = { modifier = cannibalism_silent_larder_state }
	}
	else_if = {
		limit = { check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.liberated_emergency } }
		set_state_flag = cannibalism_liberated_emergency
		set_variable = { cannibalism_state_modifier_supplies = constant:cannibalism_state_modifier.emergency_supplies }
		set_variable = { cannibalism_state_modifier_production = constant:cannibalism_state_modifier.emergency_production }
		set_variable = { cannibalism_state_modifier_manpower = constant:cannibalism_state_modifier.emergency_manpower }
		set_variable = { cannibalism_state_modifier_resistance = constant:cannibalism_state_modifier.emergency_resistance }
		add_dynamic_modifier = { modifier = cannibalism_liberated_emergency_state }
		cannibalism_achievement_begin_supply_recovery_tracking = yes
	}
	else_if = {
		limit = { check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.identification_recovery } }
		set_state_flag = cannibalism_identification_recovery
		set_variable = { cannibalism_state_modifier_supplies = constant:cannibalism_state_modifier.identification_supplies }
		set_variable = { cannibalism_state_modifier_production = constant:cannibalism_state_modifier.identification_production }

		add_dynamic_modifier = { modifier = cannibalism_identification_recovery_state }
	}
	else_if = {
		limit = { check_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.institutional_recovery } }
		set_state_flag = cannibalism_institutional_recovery
		set_variable = { cannibalism_state_modifier_supplies = constant:cannibalism_state_modifier.institutional_supplies }

		set_variable = { cannibalism_state_modifier_manpower = constant:cannibalism_state_modifier.institutional_manpower }
		set_variable = { cannibalism_state_modifier_resistance = constant:cannibalism_state_modifier.institutional_resistance }

	else_if = {

	}

cannibalism_begin_current_state_recovery = {
	set_temp_variable = { cannibalism_recovery_start_allowed = 0 }
	if = {
		limit = {
			has_state_flag = cannibalism_active_node
			cannibalism_current_node_identity_matches = yes
		}
		cannibalism_achievement_record_cell_suppression = yes
		cannibalism_retire_current_state_node = yes
		if = {
			limit = { check_variable = { cannibalism_node_retire_result > 0 } }
			set_temp_variable = { cannibalism_recovery_start_allowed = 1 }
		}
	}
	if = {
		limit = { check_variable = { cannibalism_recovery_start_allowed > 0 } }
		if = {
			limit = { NOT = { is_in_array = { array = global.cannibalism_recovery_states value = THIS } } }
			add_to_array = { array = global.cannibalism_recovery_states value = THIS }
		}
		set_state_flag = cannibalism_recovery_active
		set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.liberated_emergency }
		cannibalism_set_current_state_stage = yes
		set_variable = { cannibalism_recovery_next_date = global.date }
		add_to_variable = { cannibalism_recovery_next_date = constant:cannibalism_timing.liberated_emergency_days }
	}
}

cannibalism_advance_current_state_recovery = {
	if = {
		limit = {
			has_state_flag = cannibalism_recovery_active
			check_variable = {
				var = cannibalism_recovery_next_date
				value = global.date
				compare = less_than_or_equals
			}
		}
		cannibalism_achievement_check_supply_recovery_tracking = yes
		if = {
			limit = { check_variable = { cannibalism_state_stage = constant:cannibalism_state_stage.liberated_emergency } }
			set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.identification_recovery }
			cannibalism_set_current_state_stage = yes
			set_variable = { cannibalism_recovery_next_date = global.date }
			add_to_variable = { cannibalism_recovery_next_date = constant:cannibalism_timing.identification_days }
		}
		else_if = {
			limit = { check_variable = { cannibalism_state_stage = constant:cannibalism_state_stage.identification_recovery } }
			set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.institutional_recovery }
			cannibalism_set_current_state_stage = yes
			set_variable = { cannibalism_recovery_next_date = global.date }
			add_to_variable = { cannibalism_recovery_next_date = constant:cannibalism_timing.institutional_recovery_days }
		}
		else_if = {
			limit = { check_variable = { cannibalism_state_stage = constant:cannibalism_state_stage.institutional_recovery } }
			set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.long_trauma }
			cannibalism_set_current_state_stage = yes
			set_variable = { cannibalism_recovery_next_date = global.date }
			add_to_variable = { cannibalism_recovery_next_date = constant:cannibalism_timing.long_trauma_days }
		}
		else_if = {
			limit = { check_variable = { cannibalism_state_stage = constant:cannibalism_state_stage.long_trauma } }
			set_temp_variable = { cannibalism_new_state_stage = constant:cannibalism_state_stage.stabilized }
			cannibalism_set_current_state_stage = yes
			clr_state_flag = cannibalism_recovery_active
			clear_variable = cannibalism_recovery_next_date
		}
	}
}

cannibalism_apply_chaos_change = {
	if = {
		limit = {
			check_variable = {
				var = cannibalism_chaos_change
				value = 0
				compare = not_equals
			}
		}
		set_temp_variable = { chaos_change = cannibalism_chaos_change }
		set_temp_variable = { chaos_history_reason = cannibalism_chaos_reason }
		set_temp_variable = { chaos_history_reason_custom = 1 }
		save_event_target_as = chaos_history_actor
		add_chaos_meter_value = yes
	}
}
