# ============================================================================
# CHAOS REDUX - EVENT 025 PARTICIPANT PULSE AND ANALYSIS LIFECYCLE
# ============================================================================
# One outstanding country event per registered expedition while the race is
# live. After the race, material holders keep their own monthly watch pulse
# (see the holder effects); sealing one archive never closes another
# participant's analysis. These helpers iterate only the stored roster, and
# only when a country closes its analysis.

chaosx_nr25_schedule_participant_pulse = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_participant
			NOT = { has_country_flag = chaosx_nr25_pulse_pending }
			has_global_flag = chaosx_nr25_active
			NOT = { has_country_flag = chaosx_nr25_terminal }
		}
		set_country_flag = chaosx_nr25_pulse_pending
		country_event = { id = chaosx.nr25.280 days = constant:chaosx_nr25_timing.pulse_days }
	}
}

# COUNTRY scope. Opens the Evolution V aftermath for one material holder,
# whether or not it took part in the race, and hands it to the holder watch.
chaosx_nr25_open_holder_aftermath = {
	if = {
		limit = {
			has_global_flag = chaosx_nr25_evolution_technology_changes_users
			chaosx_nr25_holds_alien_material = yes
			NOT = { has_country_flag = chaosx_nr25_aftermath_closed }
		}
		if = {
			limit = { NOT = { has_country_flag = chaosx_nr25_dependence_started } }
			set_country_flag = chaosx_nr25_dependence_started
			set_variable = { chaosx_nr25_dependence = constant:chaosx_nr25_cost.dependence_opening }
		}
		set_global_flag = chaosx_nr25_aftermath_active
		set_country_flag = chaosx_nr25_aftermath_open
		set_country_flag = chaosx_nr25_evolution_technology_changes_users_holder
		clr_country_flag = chaosx_nr25_terminal
		chaosx_nr25_sync_outcome_idea = yes
		chaosx_nr25_enroll_holder = yes
	}
}

# COUNTRY scope. Both timed achievements require a continuous interval in
# their stated band. Leaving the band resets the interval; it cannot be
# banked across an accident or a period of unstable integration. The caller
# supplies its pulse length in the temporary chaosx_nr25_pulse_step.
chaosx_nr25_update_dependence_streaks = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_dependence_contained
			NOT = { has_country_flag = chaosx_nr25_dependence_accident }
			check_variable = { var = chaosx_nr25_dependence value = constant:chaosx_nr25_cost.dependence_contained_threshold compare = less_than }
		}
		add_to_variable = { chaosx_nr25_dependence_safe_days = chaosx_nr25_pulse_step }
	}
	else = { set_variable = { chaosx_nr25_dependence_safe_days = constant:chaosx_nr25_value.zero } }
	if = {
		limit = {
			has_country_flag = chaosx_nr25_dependence_started
			has_country_flag = chaosx_nr25_dependence_integrate_route
			NOT = { has_country_flag = chaosx_nr25_dependence_accident }
			check_variable = { var = chaosx_nr25_dependence value = constant:chaosx_nr25_cost.dependence_stable_threshold compare = greater_than_or_equals }
		}
		add_to_variable = { chaosx_nr25_dependence_stable_days = chaosx_nr25_pulse_step }
	}
	else = { set_variable = { chaosx_nr25_dependence_stable_days = constant:chaosx_nr25_value.zero } }
}

chaosx_nr25_close_country_analysis = {
	set_country_flag = chaosx_nr25_aftermath_closed
	clr_country_flag = chaosx_nr25_aftermath_open
	set_country_flag = chaosx_nr25_terminal
	set_country_flag = chaosx_nr25_board_closed
	set_variable = { chaosx_nr25_phase = constant:chaosx_nr25_phase.closed }
	if = {
		limit = {
			NOT = {
				any_of_scopes = {
					array = global.chaosx_nr25_participants
					exists = yes
					has_country_flag = chaosx_nr25_participant
					NOT = { has_country_flag = chaosx_nr25_analysis_sealed }
				}
			}
		}
		clr_global_flag = chaosx_nr25_aftermath_active
		set_global_flag = chaosx_nr25_closed
		set_variable = { global.chaosx_nr25_phase = constant:chaosx_nr25_race_state.closed }
	}
}

# COUNTRY scope. Every expedition records its first eligible recovery day once
# its own phase reaches final recovery, independent of any rival's progress.
# This date survives later deterioration and remains a deterministic tie key.
chaosx_nr25_track_final_readiness = {
	if = {
		limit = {
			has_global_flag = chaosx_nr25_active
			has_country_flag = chaosx_nr25_participant
			exists = yes
			NOT = { has_capitulated = yes }
			NOT = { is_subject = yes }
			NOT = { has_country_flag = chaosx_nr25_terminal }
			has_country_flag = chaosx_nr25_primary_site_confirmed
			has_country_flag = chaosx_nr25_outpost_established
			NOT = { has_country_flag = chaosx_nr25_emergency_weather }
			NOT = { has_country_flag = chaosx_nr25_crew_exhausted }
			check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.final_recovery compare = equals }
			check_variable = { var = chaosx_nr25_progress value = constant:chaosx_nr25_cost.progress_min compare = greater_than_or_equals }
			check_variable = { var = chaosx_nr25_logistics_readiness value = constant:chaosx_nr25_cost.readiness_min compare = greater_than_or_equals }
			check_variable = { var = chaosx_nr25_outpost_security value = constant:chaosx_nr25_cost.outpost_min compare = greater_than_or_equals }
			check_variable = { var = chaosx_nr25_access value = constant:chaosx_nr25_value.one compare = greater_than_or_equals }
			check_variable = { var = chaosx_nr25_final_ready_date value = constant:chaosx_nr25_value.zero compare = equals }
		}
		set_variable = { chaosx_nr25_final_ready_date = global.date }
	}
}

# COUNTRY scope (the country that opened the race). Every evolution whose own
# setting is enabled and whose Chaos gate is already met when the race opens
# changes the opening. Evolutions I to IV begin at once and log once through
# their normal activation; Evolution V stays a latent property that makes its
# later activation reach expeditions during the race. Disabled evolutions and
# unmet gates change nothing, and every baseline phase remains.
chaosx_nr25_apply_evolved_openings = {
	if = {
		limit = { chaosx_nr25_active_signal_evolution_enabled = yes }
		set_global_flag = chaosx_nr25_opening_active_signal
		chaosx_nr25_activate_evolution_active_signal = yes
	}
	if = {
		limit = { chaosx_nr25_something_survived_evolution_enabled = yes }
		set_global_flag = chaosx_nr25_opening_something_survived
		chaosx_nr25_activate_evolution_something_survived = yes
	}
	if = {
		limit = { chaosx_nr25_militarised_antarctica_evolution_enabled = yes }
		set_global_flag = chaosx_nr25_opening_militarised_antarctica
		chaosx_nr25_activate_evolution_militarised_antarctica = yes
	}
	if = {
		limit = { chaosx_nr25_wreck_breaking_apart_evolution_enabled = yes }
		set_global_flag = chaosx_nr25_opening_wreck_breaking_apart
		chaosx_nr25_activate_evolution_wreck_breaking_apart = yes
	}
	if = {
		limit = { chaosx_nr25_technology_changes_users_evolution_enabled = yes }
		set_global_flag = chaosx_nr25_opening_technology_changes_users
	}
}

# COUNTRY scope. Applies the evolved opening to one expedition at
# registration, after its baseline values are set.
chaosx_nr25_apply_opening_state = {
	if = {
		limit = { has_global_flag = chaosx_nr25_opening_active_signal }
		# The transmitter is already answering when the expedition arrives.
		set_variable = { chaosx_nr25_signal_intensity = constant:chaosx_nr25_cost.signal_opening_intensity }
		set_country_flag = chaosx_nr25_signal_profile_initialized
	}
	if = {
		limit = { has_global_flag = chaosx_nr25_opening_something_survived }
		# Tracks around the wreck are already in the first survey reports.
		set_country_flag = chaosx_nr25_survivor_tracks_reported
	}
	if = {
		limit = {
			has_global_flag = chaosx_nr25_opening_militarised_antarctica
			is_ai = yes
		}
		add_to_variable = { chaosx_nr25_logistics_readiness = constant:chaosx_nr25_opening.escort_readiness }
		set_country_flag = chaosx_nr25_security_zone
	}
	if = {
		limit = { has_global_flag = chaosx_nr25_opening_wreck_breaking_apart }
		set_variable = { chaosx_nr25_wreck_integrity = constant:chaosx_nr25_opening.wreck_start_integrity }
		set_country_flag = chaosx_nr25_fragment_field_active
	}
	if = {
		limit = { has_global_flag = chaosx_nr25_opening_technology_changes_users }
		add_to_variable = { chaosx_nr25_exposure_risk = constant:chaosx_nr25_opening.latent_exposure }
		set_country_flag = chaosx_nr25_latent_dependence
	}
}

# COUNTRY scope (any). When Evolution V activates during a race that opened
# with latent exposure, the exposed expeditions start their Dependence at once
# instead of waiting for settlement.
chaosx_nr25_start_latent_dependence = {
	if = {
		limit = { has_global_flag = chaosx_nr25_active }
		for_each_scope_loop = {
			array = global.chaosx_nr25_participants
			if = {
				limit = {
					exists = yes
					has_country_flag = chaosx_nr25_latent_dependence
					NOT = { has_country_flag = chaosx_nr25_dependence_started }
				}
				set_country_flag = chaosx_nr25_dependence_started
				set_variable = { chaosx_nr25_dependence = constant:chaosx_nr25_cost.dependence_opening }
			}
		}
	}
}

# COUNTRY scope (any). Closes the entry window once: AI allocation fills only
# the slots humans left, then every registered expedition moves from entry to
# its own mobilisation phase; the loop reads only the stored registry. Called
# by the delayed controller events and, once the stored close date has passed,
# by any expedition pulse, so the window cannot stay open forever.
chaosx_nr25_close_entry_window = {
	if = {
		limit = { has_global_flag = chaosx_nr25_entry_open }
		clr_global_flag = chaosx_nr25_entry_open
		chaosx_nr25_select_bounded_ai_roster = yes
		if = {
			limit = {
				check_variable = { var = global.chaosx_nr25_participant_count value = constant:chaosx_nr25_value.zero compare = greater_than }
			}
			set_variable = { global.chaosx_nr25_phase = constant:chaosx_nr25_race_state.active }
			for_each_scope_loop = {
				array = global.chaosx_nr25_participants
				if = {
					limit = {
						has_country_flag = chaosx_nr25_participant
						check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.entry compare = equals }
					}
					set_temp_variable = { chaosx_nr25_new_phase = constant:chaosx_nr25_phase.mobilization }
					chaosx_nr25_enter_phase = yes
					chaosx_nr25_advance_phase_if_ready = yes
					chaosx_nr25_prepare_board_slots = yes
					country_event = { id = chaosx.nr25.5 days = 1 }
				}
			}
		}
		else = {
			country_event = { id = chaosx.nr25.299 days = 1 }
		}
	}
}

# COUNTRY scope. Recalls an expedition that has neither entered a new phase
# nor opened an objective or final recovery attempt within the stall limit,
# and has none running. Its verified evidence is preserved, and the race
# closes through the normal path once no expedition remains.
chaosx_nr25_recall_if_stalled = {
	if = {
		limit = {
			has_global_flag = chaosx_nr25_active
			NOT = { has_global_flag = chaosx_nr25_entry_open }
			has_country_flag = chaosx_nr25_participant
			NOT = { has_country_flag = chaosx_nr25_terminal }
			NOT = { has_country_flag = chaosx_nr25_phase_mission_active }
			NOT = { has_country_flag = chaosx_nr25_final_mission_active }
			NOT = { has_country_flag = chaosx_nr25_evacuation_requested }
			has_variable = chaosx_nr25_stall_date
			check_variable = { var = global.date value = chaosx_nr25_stall_date compare = greater_than }
		}
		set_country_flag = chaosx_nr25_recalled_after_stall
		country_event = { id = chaosx.nr25.89 days = 1 }
		chaosx_nr25_withdraw_participant = yes
	}
}

# COUNTRY scope. Restarts the expedition's stall clock when it enters a phase
# or opens an objective or final recovery attempt. The earlier review date is
# when an AI expedition that still cannot pay for its next phase considers
# withdrawing.
chaosx_nr25_reset_stall_clock = {
	set_temp_variable = { chaosx_date_days = constant:chaosx_nr25_stall.phase_stall_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { chaosx_nr25_stall_date = global.date }
	add_to_variable = { chaosx_nr25_stall_date = chaosx_date_offset }
	set_temp_variable = { chaosx_date_days = constant:chaosx_nr25_ai.withdraw_unaffordable_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { chaosx_nr25_withdraw_review_date = global.date }
	add_to_variable = { chaosx_nr25_withdraw_review_date = chaosx_date_offset }
}
