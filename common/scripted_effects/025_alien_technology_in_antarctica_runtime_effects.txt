# COUNTRY scope. Builds at most three anonymous rival cards from the stored
# participant registry. The card IDs are private viewer-local variables; the
# board reveals only an evidence band and uses the ID for a later target
# resolution. This helper never scans the world. Inside the registry loop PREV
# is the participant being prepared, so registering an AI entrant from another
# country's event never writes that event country's cards.
# The viewer's chaosx_nr25_rival_page_offset skips that many valid rivals
# before the first card, so the next-rivals control pages through a roster
# larger than the column. A selected rival stays selected while its card is
# on another page.
chaosx_nr25_prepare_rival_cards = {
	set_temp_variable = { chaosx_nr25_previous_rival_target_id = chaosx_nr25_rival_target_id }
	set_temp_variable = { chaosx_nr25_rival_seen = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_candidate_count = constant:chaosx_nr25_value.zero }
	for_each_scope_loop = {
		array = global.chaosx_nr25_participants
		if = {
			limit = { chaosx_nr25_rival_card_candidate_valid = yes }
			PREV = { add_to_variable = { chaosx_nr25_rival_candidate_count = constant:chaosx_nr25_value.one } }
		}
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = chaosx_nr25_rival_page_offset }
				check_variable = { var = chaosx_nr25_rival_page_offset value = chaosx_nr25_rival_candidate_count compare = greater_than_or_equals }
			}
		}
		set_variable = { chaosx_nr25_rival_page_offset = constant:chaosx_nr25_value.zero }
	}
	set_variable = { chaosx_nr25_rival_card_count = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_1_id = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_2_id = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_3_id = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_1_state = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_2_state = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_nr25_rival_slot_3_state = constant:chaosx_nr25_value.zero }
	clr_country_flag = chaosx_nr25_rival_slot_1_visible
	clr_country_flag = chaosx_nr25_rival_slot_2_visible
	clr_country_flag = chaosx_nr25_rival_slot_3_visible
	clr_country_flag = chaosx_nr25_rival_slot_1_selected
	clr_country_flag = chaosx_nr25_rival_slot_2_selected
	clr_country_flag = chaosx_nr25_rival_slot_3_selected
	clr_country_flag = chaosx_nr25_rival_target_selected
	for_each_scope_loop = {
		array = global.chaosx_nr25_participants
		if = {
			limit = { chaosx_nr25_rival_card_candidate_valid = yes }
			add_to_temp_variable = { chaosx_nr25_rival_seen = constant:chaosx_nr25_value.one }
			if = {
				limit = { check_variable = { var = chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
				PREV = { set_country_flag = chaosx_nr25_rival_target_selected }
			}
		}
		if = {
			limit = {
				chaosx_nr25_rival_card_candidate_valid = yes
				check_variable = { var = chaosx_nr25_rival_seen value = PREV.chaosx_nr25_rival_page_offset compare = greater_than }
				check_variable = { var = PREV.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_board.rival_cards compare = less_than }
			}
			PREV = { add_to_variable = { chaosx_nr25_rival_card_count = constant:chaosx_nr25_value.one } }
			if = {
				limit = { check_variable = { var = PREV.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.one compare = equals } }
				set_variable = { PREV.chaosx_nr25_rival_slot_1_id = THIS.id }
				PREV = { set_country_flag = chaosx_nr25_rival_slot_1_visible }
				if = {
					limit = { check_variable = { var = chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
					PREV = { set_country_flag = chaosx_nr25_rival_slot_1_selected set_country_flag = chaosx_nr25_rival_target_selected }
				}
			}
			else_if = {
				limit = { check_variable = { var = PREV.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.two compare = equals } }
				set_variable = { PREV.chaosx_nr25_rival_slot_2_id = THIS.id }
				PREV = { set_country_flag = chaosx_nr25_rival_slot_2_visible }
				if = {
					limit = { check_variable = { var = chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
					PREV = { set_country_flag = chaosx_nr25_rival_slot_2_selected set_country_flag = chaosx_nr25_rival_target_selected }
				}
			}
			else_if = {
				limit = { check_variable = { var = PREV.chaosx_nr25_rival_card_count value = constant:chaosx_nr25_value.three compare = equals } }
				set_variable = { PREV.chaosx_nr25_rival_slot_3_id = THIS.id }
				PREV = { set_country_flag = chaosx_nr25_rival_slot_3_visible }
				if = {
					limit = { check_variable = { var = chaosx_nr25_previous_rival_target_id value = THIS.id compare = equals } }
					PREV = { set_country_flag = chaosx_nr25_rival_slot_3_selected set_country_flag = chaosx_nr25_rival_target_selected }
				}
			}
		}
	}
	if = {
		limit = { NOT = { has_country_flag = chaosx_nr25_rival_target_selected } }
		set_variable = { chaosx_nr25_rival_target_id = constant:chaosx_nr25_value.zero }
	}
}

# COUNTRY scope. Resolves a board-selected rival ID through the stored
# participant registry immediately before a hostile action. The board never
# stores a country scope in UI state, and a withdrawn target is cleared rather
# than silently retargeted.
chaosx_nr25_resolve_rival_target_id = {
	if = {
		limit = {
			has_variable = chaosx_nr25_rival_target_id
			check_variable = {
				var = chaosx_nr25_rival_target_id
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
		}
		# Rebuild from the durable ID even if an earlier action left a chain-local target.
		clear_variable = chaosx_nr25_rival_target
		for_each_scope_loop = {
			array = global.chaosx_nr25_participants
			if = {
				limit = {
					chaosx_nr25_rival_card_candidate_valid = yes
					check_variable = { var = PREV.chaosx_nr25_rival_target_id value = THIS.id compare = equals }
				}
				save_event_target_as = chaosx_nr25_rival_target_scope
				PREV = { set_variable = { chaosx_nr25_rival_target = PREV.id } }
			}
		}
	}
	if = {
		limit = {
			has_variable = chaosx_nr25_rival_target_id
			check_variable = {
				var = chaosx_nr25_rival_target_id
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
			NOT = { has_variable = chaosx_nr25_rival_target }
		}
		set_variable = { chaosx_nr25_rival_target_id = constant:chaosx_nr25_value.zero }
		clr_country_flag = chaosx_nr25_rival_target_selected
	}
}

# COUNTRY scope. Picks the rival for an untargeted response that acts on one
# rival: for a human, the rival selected on the Expedition Board while it is
# still valid; otherwise the valid rival with the most Expedition Progress.
# Saves it as the event target chaosx_nr25_picked_rival and sets the
# temporary chaosx_nr25_rival_picked to one; the caller sets that temporary
# to zero first.
chaosx_nr25_pick_rival_target = {
	if = {
		limit = {
			is_ai = no
			has_country_flag = chaosx_nr25_rival_target_selected
		}
		chaosx_nr25_resolve_rival_target_id = yes
		if = {
			limit = {
				has_variable = chaosx_nr25_rival_target
				has_event_target = chaosx_nr25_rival_target_scope
				event_target:chaosx_nr25_rival_target_scope = { chaosx_nr25_target_participant_valid = yes }
			}
			event_target:chaosx_nr25_rival_target_scope = { save_event_target_as = chaosx_nr25_picked_rival }
			set_temp_variable = { chaosx_nr25_rival_picked = constant:chaosx_nr25_value.one }
		}
	}
	if = {
		limit = { check_variable = { var = chaosx_nr25_rival_picked value = constant:chaosx_nr25_value.zero compare = equals } }
		set_temp_variable = { chaosx_nr25_rival_pick_best = constant:chaosx_nr25_value.zero }
		for_each_scope_loop = {
			array = global.chaosx_nr25_participants
			if = {
				limit = {
					chaosx_nr25_rival_card_candidate_valid = yes
					OR = {
						check_variable = { var = chaosx_nr25_rival_picked value = constant:chaosx_nr25_value.zero compare = equals }
						check_variable = { var = chaosx_nr25_progress value = chaosx_nr25_rival_pick_best compare = greater_than }
					}
				}
				set_temp_variable = { chaosx_nr25_rival_pick_best = chaosx_nr25_progress }
				set_temp_variable = { chaosx_nr25_rival_picked = constant:chaosx_nr25_value.one }
				save_event_target_as = chaosx_nr25_picked_rival
			}
		}
	}
}

# COUNTRY scope. Selects a bounded AI-major roster after all human invitations
# have been answered. The only non-major exception is the named KRG tag.
chaosx_nr25_select_bounded_ai_roster = {
	if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_value.zero compare = equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_zero_humans }
	}
	else_if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_value.one compare = equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_one_human }
	}
	else_if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_roster.human_count_three compare = less_than_or_equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_three_humans }
	}
	else_if = {
		limit = { check_variable = { var = global.chaosx_nr25_human_count value = constant:chaosx_nr25_roster.human_count_six compare = less_than_or_equals } }
		set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_six_humans }
	}
	else = { set_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_roster.ai_slots_many_humans } }

	while_loop_effect = {
		limit = {
			check_variable = { var = global.chaosx_nr25_ai_slots value = constant:chaosx_nr25_value.zero compare = greater_than }
			any_country = { chaosx_nr25_ai_candidate_available = yes }
		}
		random_country = {
			limit = { chaosx_nr25_ai_candidate_available = yes }
			chaosx_nr25_register_participant = yes
			set_temp_variable = { chaosx_nr25_action = constant:chaosx_nr25_action.authorize }
			chaosx_nr25_execute_action = yes
			subtract_from_variable = { global.chaosx_nr25_ai_slots = constant:chaosx_nr25_value.one }
		}
	}
}

}

# COUNTRY scope. Registers one expedition casualty incident in the shared
# Chaos deaths ledger under the dedicated Antarctic expedition reason, on the
# military side, and leaves a durable rescue-memory marker for Evolution III.
# The caller may set the temporary chaosx_nr25_casualty_amount to a
# chaosx_nr25_casualty band; otherwise the field band applies. The amount is
# cleared after use so a second incident in the same chain picks its own band,
# and chaosx_nr25_last_casualties keeps it for the report text.
chaosx_nr25_record_casualties = {
	if = {
		limit = { NOT = { check_variable = { var = chaosx_nr25_casualty_amount value = constant:chaosx_nr25_value.zero compare = greater_than } } }
		set_temp_variable = { chaosx_nr25_casualty_amount = constant:chaosx_nr25_casualty.field }
	}
	add_to_variable = { chaosx_nr25_casualties = chaosx_nr25_casualty_amount }
	set_variable = { chaosx_nr25_last_casualties = chaosx_nr25_casualty_amount }
	set_country_flag = chaosx_nr25_casualty_memory_ready
	set_temp_variable = { chaos_deaths_change = chaosx_nr25_casualty_amount }
	set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.antarctic_expedition }
	set_temp_variable = { chaos_deaths_is_civilian = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaos_deaths_target_country = THIS }
	set_temp_variable = { chaos_deaths_has_target_country = constant:chaosx_nr25_value.one }
	chaos_meter_register_deaths = yes
	set_temp_variable = { chaosx_nr25_casualty_amount = constant:chaosx_nr25_value.zero }
}

# COUNTRY scope. Moves this expedition into the phase stored in the temporary
# `chaosx_nr25_new_phase` value and keeps the public leading phase used by
# reports. It never changes another participant's phase.
# Gateway route model. Each gateway group is a short list of live staging
# ports (chaosx_nr25_gateway_state). An expedition sails through a group on the
# best band any of its ports offers it: direct when it holds a port, allied
# when the port holder is a faction partner, a subject or overlord, or friendly
# to it, long ocean otherwise, and chartered on chartered passage. A port held
# by a country at war with the expedition offers nothing; a group whose ports
# are all held by enemies is closed. Only the expedition's own country and the
# nine listed states are read, never the wider world.
@NR25_GATEWAY_FRIENDLY_OPINION = 50

# COUNTRY scope (the expedition). Rates the state in the temporary
# chaosx_nr25_gw_state for the expedition saved as chaosx_nr25_gw_actor and
# keeps the best band in chaosx_nr25_gw_best with its port holder in
# chaosx_nr25_gw_best_controller.
chaosx_nr25_rate_gateway_state = {
	var:chaosx_nr25_gw_state = {
		controller = {
			if = {
				limit = { tag = event_target:chaosx_nr25_gw_actor }
				set_temp_variable = { chaosx_nr25_gw_rating = constant:chaosx_nr25_gateway_band.direct }
			}
			else_if = {
				limit = { has_war_with = event_target:chaosx_nr25_gw_actor }
				set_temp_variable = { chaosx_nr25_gw_rating = constant:chaosx_nr25_gateway_band.closed }
			}
			else_if = {
				limit = {
					OR = {
						is_in_faction_with = event_target:chaosx_nr25_gw_actor
						is_subject_of = event_target:chaosx_nr25_gw_actor
						event_target:chaosx_nr25_gw_actor = { is_subject_of = PREV }
						has_opinion = { target = event_target:chaosx_nr25_gw_actor value > @NR25_GATEWAY_FRIENDLY_OPINION }
					}
				}
				set_temp_variable = { chaosx_nr25_gw_rating = constant:chaosx_nr25_gateway_band.allied }
			}
			else = { set_temp_variable = { chaosx_nr25_gw_rating = constant:chaosx_nr25_gateway_band.long_ocean } }
			if = {
				limit = { check_variable = { var = chaosx_nr25_gw_rating value = chaosx_nr25_gw_best compare = less_than } }
				set_temp_variable = { chaosx_nr25_gw_best = chaosx_nr25_gw_rating }
				set_temp_variable = { chaosx_nr25_gw_best_controller = THIS }
			}
		}
	}
}

# COUNTRY scope. Cuts the expedition's route when every port of its chosen
# gateway group is now held by a country at war with it. The expedition loses
# readiness, the route counts as collapsed once, the route-dependent
# objectives stop, and the expedition is told how to recover: an alternate
# landing or an emergency neutral charter.
chaosx_nr25_check_route_interruption = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_action_gateway_used
			NOT = { has_country_flag = chaosx_nr25_route_interrupted }
			OR = {
				AND = {
					check_variable = { var = chaosx_nr25_route value = constant:chaosx_nr25_route.south_america compare = equals }
					check_variable = { var = chaosx_nr25_gateway_band_south_america value = constant:chaosx_nr25_gateway_band.closed compare = equals }
				}
				AND = {
					check_variable = { var = chaosx_nr25_route value = constant:chaosx_nr25_route.africa compare = equals }
					check_variable = { var = chaosx_nr25_gateway_band_africa value = constant:chaosx_nr25_gateway_band.closed compare = equals }
				}
				AND = {
					check_variable = { var = chaosx_nr25_route value = constant:chaosx_nr25_route.tasmania compare = equals }
					check_variable = { var = chaosx_nr25_gateway_band_tasmania value = constant:chaosx_nr25_gateway_band.closed compare = equals }
				}
				AND = {
					check_variable = { var = chaosx_nr25_route value = constant:chaosx_nr25_route.new_zealand compare = equals }
					check_variable = { var = chaosx_nr25_gateway_band_new_zealand value = constant:chaosx_nr25_gateway_band.closed compare = equals }
				}
			}
		}
		set_country_flag = chaosx_nr25_route_interrupted
		subtract_from_variable = { chaosx_nr25_logistics_readiness = constant:chaosx_nr25_gateway.interruption_readiness_loss }
		add_to_variable = { chaosx_nr25_route_collapses = constant:chaosx_nr25_value.one }
		chaosx_nr25_report_route_interrupted = yes
	}
}

# COUNTRY scope. Stores an AI expedition's post-spend progress floors: the
# core reserve (higher at war) plus the cost of its current phase's gate
# action (in mobilisation, the coming gateway), so `chaosx_nr25_ai_keeps_progress_reserve` passes only when the
# country keeps the core reserve after paying for that action. Fuel is stored
# in thousands to match the engine's `fuel_k` reader. Refreshed on entry, on
# every phase change and on each participant pulse.
chaosx_nr25_refresh_ai_progress_floor = {
	if = {
		limit = { is_ai = yes }
		if = {
			limit = { has_war = yes }
			set_variable = { chaosx_nr25_ai_floor_pp = constant:chaosx_nr25_ai_floor.core_pp_war }
			set_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_ai_floor.core_fuel_war }
			set_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_ai_floor.core_support_war }
			set_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_ai_floor.core_convoys_war }
		}
		else = {
			set_variable = { chaosx_nr25_ai_floor_pp = constant:chaosx_nr25_ai_floor.core_pp_peace }
			set_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_ai_floor.core_fuel_peace }
			set_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_ai_floor.core_support_peace }
			set_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_ai_floor.core_convoys_peace }
		}
		if = {
			limit = { check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.entry compare = equals } }
			if = {
				limit = { chaosx_nr25_uses_chartered_passage = yes }
				add_to_variable = { chaosx_nr25_ai_floor_pp = constant:chaosx_nr25_cost.pp_charter_entry_required }
				add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_cost.fuel_charter_entry_required }
				add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_cost.support_charter_entry_required }
			}
			else = {
				add_to_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_cost.convoy_authorize_required }
				add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_cost.fuel_authorize_required }
				add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_cost.support_authorize_required }
			}
		}
		# Mobilisation saves for the gateway, the next gate to be paid, at the
		# best band open to the expedition.
		else_if = {
			limit = {
				OR = {
					check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.mobilization compare = equals }
					check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.crossing compare = equals }
				}
			}
			set_temp_variable = { chaosx_nr25_gw_best = constant:chaosx_nr25_gateway_band.closed }
			chaosx_nr25_read_best_gateway_band = yes
			if = {
				limit = { check_variable = { var = chaosx_nr25_gw_best value = constant:chaosx_nr25_gateway_band.direct compare = equals } }
				add_to_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_gateway.direct_convoys }
				add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_gateway.direct_fuel }
				add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_gateway.direct_support }
			}
			else_if = {
				limit = { check_variable = { var = chaosx_nr25_gw_best value = constant:chaosx_nr25_gateway_band.allied compare = equals } }
				add_to_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_gateway.allied_convoys }
				add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_gateway.allied_fuel }
				add_to_variable = { chaosx_nr25_ai_floor_pp = constant:chaosx_nr25_gateway.allied_pp }
			}
			else_if = {
				limit = { check_variable = { var = chaosx_nr25_gw_best value = constant:chaosx_nr25_gateway_band.chartered compare = equals } }
				add_to_variable = { chaosx_nr25_ai_floor_pp = constant:chaosx_nr25_gateway.chartered_pp }
				add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_gateway.chartered_fuel }
				add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_gateway.chartered_support }
			}
			else = {
				add_to_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_gateway.long_ocean_convoys }
				add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_gateway.long_ocean_fuel }
				add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_gateway.long_ocean_support }
			}
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.outpost compare = equals } }
			add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_cost.fuel_outpost_required }
			add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_cost.support_outpost_required }
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.survey compare = equals } }
			add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_cost.fuel_survey_required }
			add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_cost.support_survey_required }
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.final_recovery compare = equals } }
			if = {
				limit = { chaosx_nr25_uses_chartered_passage = yes }
				add_to_variable = { chaosx_nr25_ai_floor_pp = constant:chaosx_nr25_cost.pp_chartered_final_required }
			}
			else = { add_to_variable = { chaosx_nr25_ai_floor_convoys = constant:chaosx_nr25_cost.convoy_final_required } }
			add_to_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_cost.fuel_final_required }
			add_to_variable = { chaosx_nr25_ai_floor_support = constant:chaosx_nr25_cost.support_final_required }
		}
		divide_variable = { chaosx_nr25_ai_floor_fuel_k = constant:chaosx_nr25_ai_floor.fuel_per_k }
	}
}

# COUNTRY scope. Stores the due date of the objective that is being opened and
# schedules its completion check for the day before timeout. The caller sets
# the temporary `chaosx_nr25_phase_mission_days` duration. The stored duration
# is shown on the mission card; the guard date keeps a callback from an older,
# cancelled objective from completing a newer one early. Opening an objective
# restarts the expedition's stall clock.
chaosx_nr25_schedule_phase_mission_completion = {
	chaosx_nr25_reset_stall_clock = yes
	set_variable = { chaosx_nr25_phase_mission_duration = chaosx_nr25_phase_mission_days }
	clr_country_flag = chaosx_nr25_phase_mission_ready

	set_variable = { chaosx_nr25_phase_mission_ready_date = global.date }
	add_to_variable = { chaosx_nr25_phase_mission_ready_date = chaosx_date_offset }
	country_event = { id = chaosx.nr25.87 days = chaosx_nr25_phase_mission_ready_delay }

# COUNTRY scope. Opens the completion window for the running objective once its
# due date has arrived. Baseline objectives fail at timeout when the
# expedition cannot finish them; the wreck and containment objectives always
# settle through their own success, partial, and failure branches, and an
# evacuation is held up only by emergency weather.
chaosx_nr25_prepare_phase_mission_completion = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_phase_mission_active
			NOT = { has_country_flag = chaosx_nr25_phase_mission_ready }
			has_variable = chaosx_nr25_phase_mission_ready_date
			NOT = { check_variable = { var = global.date value = chaosx_nr25_phase_mission_ready_date compare = less_than } }

				OR = {
					check_variable = { var = chaosx_nr25_phase_mission_kind value = constant:chaosx_nr25_phase_mission.stabilize_wreck compare = equals }
					check_variable = { var = chaosx_nr25_phase_mission_kind value = constant:chaosx_nr25_phase_mission.contain_alien_systems compare = equals }
					AND = {
						check_variable = { var = chaosx_nr25_phase_mission_kind value = constant:chaosx_nr25_phase_mission.evacuate_expedition compare = equals }
						NOT = { has_country_flag = chaosx_nr25_emergency_weather }

						NOT = { check_variable = { var = chaosx_nr25_phase_mission_kind value = constant:chaosx_nr25_phase_mission.evacuate_expedition compare = equals } }
						chaosx_nr25_phase_mission_can_complete = yes
					}
				}
			}
			set_country_flag = chaosx_nr25_phase_mission_ready
		}
	}
}

# COUNTRY scope. The timed mission reaches its completion window on this
# event's date; the player must confirm the mission card to submit the result.
chaosx_nr25_prepare_final_mission_completion = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_final_mission_active
			NOT = { has_country_flag = chaosx_nr25_final_mission_ready }
			has_global_flag = chaosx_nr25_active
			NOT = { has_global_flag = chaosx_nr25_final_recovery_resolved }
			# Without these, the mission times out as a partial failure.

		}

	}
}

# COUNTRY scope. Completes the currently armed final recovery mission and
# submits its verified candidate to the deterministic same-day arbitration.
chaosx_nr25_complete_final_mission = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_final_mission_active
			has_country_flag = chaosx_nr25_final_mission_ready
			has_global_flag = chaosx_nr25_active
			NOT = { has_global_flag = chaosx_nr25_final_recovery_resolved }
			NOT = { chaosx_nr25_final_candidate_requirements_met = yes }
		}
		# Conditions changed after the window opened: settle as a failure.
		chaosx_nr25_fail_final_mission = yes
	}
	if = {
		limit = {
			has_country_flag = chaosx_nr25_final_mission_active
			has_country_flag = chaosx_nr25_final_mission_ready
			has_global_flag = chaosx_nr25_active
			NOT = { has_global_flag = chaosx_nr25_final_recovery_resolved }
		}
		chaosx_nr25_remove_final_missions = yes
		clr_country_flag = chaosx_nr25_final_mission_active
		clr_country_flag = chaosx_nr25_final_mission_ready
		set_country_flag = chaosx_nr25_final_mission_success
		clear_variable = chaosx_nr25_final_mission_kind
		chaosx_nr25_submit_final_candidate = yes
	}
}

# COUNTRY scope. Phase progression is per expedition. Each phase advances only
# this participant, after that phase's objective has resolved and the
# participant's own evidence gates hold, so a leading rival never moves or
# strands another expedition and no action can skip crossing or outpost setup.
# Entry becomes mobilisation when the entry window closes.
chaosx_nr25_advance_phase_if_ready = {
	if = {
		limit = {
			has_country_flag = chaosx_nr25_participant
			NOT = { has_country_flag = chaosx_nr25_terminal }
		}
		if = {
			limit = {
				check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.mobilization compare = equals }
				has_country_flag = chaosx_nr25_organize_mission_resolved
				check_variable = { var = chaosx_nr25_access value = constant:chaosx_nr25_value.one compare = greater_than_or_equals }
			}
			set_temp_variable = { chaosx_nr25_new_phase = constant:chaosx_nr25_phase.crossing }
			chaosx_nr25_enter_phase = yes
		}
		else_if = {
			limit = {
				check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.crossing compare = equals }
				has_country_flag = chaosx_nr25_gateway_mission_resolved
			}
			set_temp_variable = { chaosx_nr25_new_phase = constant:chaosx_nr25_phase.outpost }
			chaosx_nr25_enter_phase = yes
		}
		else_if = {
			limit = {
				check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.outpost compare = equals }
				has_country_flag = chaosx_nr25_outpost_mission_resolved
				has_country_flag = chaosx_nr25_outpost_established
				check_variable = { var = chaosx_nr25_outpost_security value = constant:chaosx_nr25_value.one compare = greater_than_or_equals }
			}
			set_temp_variable = { chaosx_nr25_new_phase = constant:chaosx_nr25_phase.survey }
			chaosx_nr25_enter_phase = yes
		}
		else_if = {
			limit = {
				check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.survey compare = equals }
				has_country_flag = chaosx_nr25_primary_site_confirmed
				check_variable = { var = chaosx_nr25_survey_evidence value = constant:chaosx_nr25_phase_gate.survey_evidence compare = greater_than_or_equals }
			}
			set_temp_variable = { chaosx_nr25_new_phase = constant:chaosx_nr25_phase.final_recovery }
			chaosx_nr25_enter_phase = yes
		}
	}
	chaosx_nr25_track_final_readiness = yes
}

# COUNTRY scope. Lexical comparator for candidates that become valid on the
# same day. The date window is handled by the caller; this helper only applies
# the accepted evidence ordering and ends with stable participant identifiers.
chaosx_nr25_candidate_is_better_same_day = {
	set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.zero }
	if = {
		limit = {
			check_variable = {
				var = chaosx_nr25_recovery_proof_survey
				value = global.chaosx_nr25_final_candidate_survey
				compare = greater_than
			}
		}
		set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
	}
	else_if = {
		limit = {
			check_variable = {
				var = chaosx_nr25_recovery_proof_survey
				value = global.chaosx_nr25_final_candidate_survey
				compare = equals
			}
		}
		if = {
			limit = { check_variable = { var = chaosx_nr25_recovery_proof_readiness value = global.chaosx_nr25_final_candidate_readiness compare = greater_than } }
			set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
		}
		else_if = {
			limit = { check_variable = { var = chaosx_nr25_recovery_proof_readiness value = global.chaosx_nr25_final_candidate_readiness compare = equals } }
			if = {
				limit = { check_variable = { var = chaosx_nr25_recovery_proof_outpost value = global.chaosx_nr25_final_candidate_outpost compare = greater_than } }
				set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
			}
			else_if = {
				limit = { check_variable = { var = chaosx_nr25_recovery_proof_outpost value = global.chaosx_nr25_final_candidate_outpost compare = equals } }
				if = {
					limit = { check_variable = { var = chaosx_nr25_recovery_proof_exposure value = global.chaosx_nr25_final_candidate_exposure compare = less_than } }
					set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
				}
				else_if = {
					limit = { check_variable = { var = chaosx_nr25_recovery_proof_exposure value = global.chaosx_nr25_final_candidate_exposure compare = equals } }
					if = {
						limit = { check_variable = { var = chaosx_nr25_recovery_proof_ready_date value = global.chaosx_nr25_final_candidate_ready_date compare = less_than } }
						set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
					}
					else_if = {
						limit = { check_variable = { var = chaosx_nr25_recovery_proof_ready_date value = global.chaosx_nr25_final_candidate_ready_date compare = equals } }
						if = {
							limit = { check_variable = { var = chaosx_nr25_entry_order value = global.chaosx_nr25_final_candidate_entry compare = less_than } }
							set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.one }
						}
						# Equal registry positions are impossible for valid participants.
						# No random seed is used as a hidden fifth tie-breaker.
					}
				}
			}
		}
	}
}

# COUNTRY scope. Only a successful, still-valid final mission may submit.
# The first day owns the recovery window; later-day receipts cannot displace
# it. A valid same-day receipt survives loss of the event's original actor.
chaosx_nr25_submit_final_candidate = {
	if = {
		limit = {
			chaosx_nr25_final_candidate_requirements_met = yes
			has_country_flag = chaosx_nr25_final_mission_success
			NOT = { has_country_flag = chaosx_nr25_final_mission_active }
			check_variable = { var = chaosx_nr25_final_cooldown value = constant:chaosx_nr25_value.zero compare = less_than_or_equals }
		}
		set_variable = { chaosx_nr25_final_cooldown = constant:chaosx_nr25_timing.final_cooldown_days }
		add_to_variable = { global.chaosx_nr25_recovery_order = constant:chaosx_nr25_value.one }
		if = {
			limit = { check_variable = { var = global.chaosx_nr25_final_resolution_pending value = constant:chaosx_nr25_value.zero compare = equals } }
			chaosx_nr25_capture_recovery_proof = yes
			set_variable = { global.chaosx_nr25_final_resolution_pending = constant:chaosx_nr25_value.one }
			set_variable = { global.chaosx_nr25_final_candidate_date = global.date }
			chaosx_nr25_store_final_candidate = yes
			country_event = { id = chaosx.nr25.84 days = 1 }
		}
		else_if = {
			limit = { check_variable = { var = global.chaosx_nr25_final_candidate_date value = global.date compare = equals } }
			chaosx_nr25_capture_recovery_proof = yes
			set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.zero }
			chaosx_nr25_candidate_is_better_same_day = yes
			if = {
				limit = { check_variable = { var = chaosx_nr25_candidate_is_better value = constant:chaosx_nr25_value.one compare = equals } }
				chaosx_nr25_store_final_candidate = yes
			}
			# This country supplies an independent resolver callback if the
			# first submitter ceases to exist before its scheduled event fires.
			country_event = { id = chaosx.nr25.84 days = 1 }
		}
	}
}

# COUNTRY scope. Revalidates only submitted receipts in the stored registry.
# This also recovers the next deterministic same-day finalist if the leading
# candidate is annexed, capitulates, becomes a subject, or withdraws.
chaosx_nr25_reselect_final_candidate = {
	clear_global_event_target = chaosx_nr25_final_candidate_country
	set_temp_variable = { chaosx_nr25_final_candidate_found = constant:chaosx_nr25_value.zero }
	for_each_scope_loop = {
		array = global.chaosx_nr25_participants
		if = {
			limit = {
				exists = yes
				NOT = { has_capitulated = yes }
				NOT = { is_subject = yes }
				has_country_flag = chaosx_nr25_participant
				NOT = { has_country_flag = chaosx_nr25_terminal }
				has_country_flag = chaosx_nr25_recovery_proof_submitted
				check_variable = { var = chaosx_nr25_recovery_proof_day value = global.chaosx_nr25_final_candidate_date compare = equals }
			}
			if = {
				limit = { check_variable = { var = chaosx_nr25_final_candidate_found value = constant:chaosx_nr25_value.zero compare = equals } }
				chaosx_nr25_store_final_candidate = yes
				set_temp_variable = { chaosx_nr25_final_candidate_found = constant:chaosx_nr25_value.one }
			}
			else = {
				set_temp_variable = { chaosx_nr25_candidate_is_better = constant:chaosx_nr25_value.zero }
				chaosx_nr25_candidate_is_better_same_day = yes
				if = {
					limit = { check_variable = { var = chaosx_nr25_candidate_is_better value = constant:chaosx_nr25_value.one compare = equals } }
					chaosx_nr25_store_final_candidate = yes
				}
			}
		}
	}
}

# COUNTRY scope. Settlement is applied while the registry-selected country is
# valid. Report acknowledgement never grants rewards or repeats settlement.
chaosx_nr25_resolve_final_candidate = {
	if = {
		limit = {
			has_global_flag = chaosx_nr25_active
			check_variable = { var = global.chaosx_nr25_final_resolution_pending value = constant:chaosx_nr25_value.one compare = equals }
			check_variable = { var = global.date value = global.chaosx_nr25_final_candidate_date compare = greater_than }
		}
		chaosx_nr25_reselect_final_candidate = yes
		if = {
			limit = { has_event_target = chaosx_nr25_final_candidate_country }
			set_global_flag = chaosx_nr25_final_recovery_resolved
			clr_global_flag = chaosx_nr25_active
			chaosx_nr25_clear_pending_evolutions = yes
			set_variable = { global.chaosx_nr25_phase = constant:chaosx_nr25_race_state.analysis }
			event_target:chaosx_nr25_final_candidate_country = {
				set_variable = { chaosx_nr25_phase = constant:chaosx_nr25_phase.analysis }
				chaosx_nr25_remove_phase_missions = yes
				chaosx_nr25_remove_final_missions = yes
				chaosx_nr25_clear_selection_state = yes
				set_variable = { global.chaosx_nr25_winner = THIS.id }
				set_country_flag = chaosx_nr25_recovery_winner
				set_country_flag = chaosx_nr25_terminal
				chaosx_nr25_record_winner_history = yes
				chaosx_nr25_apply_winner_settlement = yes
				country_event = { id = chaosx.nr25.81 days = 1 }
				news_event = { id = chaosx.news.26 days = 1 }
			}
			for_each_scope_loop = {
				array = global.chaosx_nr25_participants
				if = {
					limit = {
						exists = yes
						has_country_flag = chaosx_nr25_participant
						NOT = { has_country_flag = chaosx_nr25_recovery_winner }
						NOT = { has_country_flag = chaosx_nr25_loser_settlement_complete }
					}
					set_variable = { chaosx_nr25_phase = constant:chaosx_nr25_phase.analysis }
					chaosx_nr25_remove_phase_missions = yes
					chaosx_nr25_remove_final_missions = yes
					chaosx_nr25_clear_selection_state = yes
					set_country_flag = chaosx_nr25_terminal
					chaosx_nr25_apply_loser_settlement = yes
					country_event = { id = chaosx.nr25.82 days = 1 }
				}
			}
			clear_global_event_target = chaosx_nr25_final_candidate_country
			clear_global_event_target = chaosx_nr25_history_actor
		}
		else = {
			# The primary recovery proof is no longer valid. Preserve verified
			# losing evidence and close without inventing a replacement reward.
			clr_global_flag = chaosx_nr25_active
			set_global_flag = chaosx_nr25_closed_after_expeditions
			set_variable = { global.chaosx_nr25_phase = constant:chaosx_nr25_race_state.closed }
			chaosx_nr25_clear_pending_evolutions = yes
			for_each_scope_loop = {
				array = global.chaosx_nr25_participants
				if = {
					limit = { exists = yes has_country_flag = chaosx_nr25_participant }
					set_variable = { chaosx_nr25_phase = constant:chaosx_nr25_phase.closed }
					chaosx_nr25_remove_phase_missions = yes
					chaosx_nr25_remove_final_missions = yes
					chaosx_nr25_clear_selection_state = yes
					set_country_flag = chaosx_nr25_terminal
					if = {
						limit = { NOT = { has_country_flag = chaosx_nr25_loser_settlement_complete } }
						chaosx_nr25_apply_loser_settlement = yes
					}
					country_event = { id = chaosx.nr25.298 days = 1 }
				}
			}
			clear_global_event_target = chaosx_nr25_history_actor
		}
		set_variable = { global.chaosx_nr25_final_resolution_pending = constant:chaosx_nr25_value.zero }
	}
}

# COUNTRY scope. Applies one evolution-specific route. Evolution actions are
# secondary board actions and never remove the baseline phase actions, so a
# hazard can be managed without making final recovery impossible.
chaosx_nr25_apply_militarisation_target = {
	if = {
		limit = {
			has_variable = chaosx_nr25_rival_target has_event_target = chaosx_nr25_rival_target_scope
			event_target:chaosx_nr25_rival_target_scope = { chaosx_nr25_target_participant_valid = yes }
		}
		# Armed actions are public: the target knows who acted, and the actor
		# remembers its last armed action so the AI does not repeat it back to
		# back. A rescue is not hostile and changes neither.
