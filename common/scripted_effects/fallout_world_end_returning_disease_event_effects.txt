fallout_event_712_record_history = {
	set_temp_variable = { fallout_event_712_history_should_record = constant:fallout_event_runtime.zero }
	if = { limit = { AND = { check_variable = { fallout_event_712_history_stage = constant:fallout_event_runtime.zero } NOT = { has_country_flag = fallout_event_712_choice_history_recorded } } } set_temp_variable = { fallout_event_712_history_should_record = constant:fallout_event_runtime.one } }
	else_if = { limit = { AND = { check_variable = { fallout_event_712_history_stage = constant:fallout_event_runtime.one } NOT = { has_country_flag = fallout_event_712_result_history_recorded } } } set_temp_variable = { fallout_event_712_history_should_record = constant:fallout_event_runtime.one } }
	else_if = { limit = { AND = { check_variable = { fallout_event_712_history_stage = constant:fallout_event_runtime.two } NOT = { has_country_flag = fallout_event_712_callback_history_recorded } } } set_temp_variable = { fallout_event_712_history_should_record = constant:fallout_event_runtime.one } }
	if = {
		limit = { check_variable = { fallout_event_712_history_should_record = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.cancelled }
		if = {
			limit = { check_variable = { fallout_event_712_history_stage = constant:fallout_event_runtime.zero } }
			if = { limit = { check_variable = { fallout_event_712_selected_branch value = constant:fallout_event_712_branch.lock_down compare = equals } } set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.choice_lock_down } }
			else_if = { limit = { check_variable = { fallout_event_712_selected_branch value = constant:fallout_event_712_branch.targeted_response compare = equals } } set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.choice_targeted_response } }
			else_if = { limit = { check_variable = { fallout_event_712_selected_branch value = constant:fallout_event_712_branch.request_aid compare = equals } } set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.choice_request_aid } }
			else = { set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.choice_conceal_outbreak } }
		}
		else_if = {
			limit = { check_variable = { fallout_event_712_history_stage = constant:fallout_event_runtime.one } }
			if = {
				limit = { check_variable = { fallout_event_712_selected_branch value = constant:fallout_event_712_branch.lock_down compare = equals } }
				if = { limit = { check_variable = { fallout_event_712_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.lock_down_success } }

				else = { set_temp_variable = { fallout_event_712_history_payload = constant:fallout_event_712_log.lock_down_failure } }
			}
			else_if = {
