# Hawai'i retains David Kalakaua Kawananakoa as the vanilla ruling leader.
# Samuel Wilder King is an additive Event 006 territorial-delegate role and is
# deliberately not promoted, so the package never overwrites a living roster.
independence_wave_prepare_haw_origin_leadership = {
	independence_wave_apply_roster_checkpoint = yes
	if = {
		limit = {
			has_character = HAW_independence_wave_territorial_delegate
			NOT = { has_country_leader = { ruling_only = no character = HAW_independence_wave_territorial_delegate } }
		}
		add_country_leader_role = {
			character = HAW_independence_wave_territorial_delegate
			promote_leader = no
			country_leader = {
				ideology = centrism
				desc = HAW_independence_wave_territorial_delegate_desc
				expire = @pacific_character_expire
				traits = { }
			}
		}
	}
}

	if = {
		limit = { is_independence_wave_hbx_package = yes }
		independence_wave_remove_hbx_package_ideas = yes
		if = {
			limit = { has_stable_independence_wave_hbx_coastal_command = yes }
			add_ideas = hbx_californian_ports_and_works_board
			set_country_flag = independence_wave_hbx_coastal_command_stable
		}
		else = {
			add_ideas = hbx_divided_federal_arsenals
			clr_country_flag = independence_wave_hbx_coastal_command_stable
		}
	}
}
