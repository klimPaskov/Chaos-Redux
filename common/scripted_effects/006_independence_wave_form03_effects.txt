# Proposer country scope. Adds the one non-Event-006 founding delegation after
# the ordinary active-country pass and before the shared ledger recount.
independence_wave_form03_extend_member_ledger = {
	if = {
		limit = {
			check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.low_countries_federation }
			BEL = {
				is_independence_wave_form03_eligible_belgian_delegation = yes
				NOT = { is_in_array = { array = ROOT.independence_wave_formable_member_country_entries value = THIS } }
			}
		}
		BEL = { save_event_target_as = independence_wave_form03_belgian_delegation }
		6 = { save_event_target_as = independence_wave_form03_belgian_anchor }
		BEL = {
			independence_wave_formable_clear_frozen_consent_snapshot = yes
			independence_wave_formable_evaluate_candidate_consent = yes
			if = {
				limit = {
					check_variable = { independence_wave_formable_candidate_consent = constant:independence_wave_formable_consent.consenting_member }
					has_accepted_independence_wave_formable_invitation_from_root = yes
				}
				independence_wave_formable_freeze_consent_snapshot_from_root = yes
			}
		}
		add_to_array = { array = independence_wave_formable_member_country_entries value = event_target:independence_wave_form03_belgian_delegation }
		add_to_array = { array = independence_wave_formable_member_generation_entries value = constant:independence_wave_formable_tuning.no_entries }
		add_to_array = { array = independence_wave_formable_member_anchor_entries value = event_target:independence_wave_form03_belgian_anchor }
		add_to_array = { array = independence_wave_formable_member_consent_entries value = independence_wave_formable_candidate_consent }
	}
}

# Proposer country scope. The member arrays are already frozen by the congress.
# Only consenting AFX/AGX anchors are absorbed; a consenting BEL row receives
# membership flags and keeps every state, core, capital, tree, and identity.
independence_wave_formable_integration_adapter_3 = {
	if = {
		limit = {
			has_country_flag = independence_wave_formable_mutation_prevalidated
			has_country_flag = independence_wave_formable_identity_committed
			has_cosmetic_tag = LCX
			has_global_flag = independence_wave_form03_lcx_identity_in_use
			independence_wave_formable_member_arrays_are_aligned = yes
		}
		set_temp_variable = { independence_wave_form03_member_index = constant:independence_wave_formable_tuning.first_array_index }
		set_temp_variable = { independence_wave_form03_additional_member_processed = constant:independence_wave_formable_tuning.no_entries }
		while_loop_effect = {
			limit = {
				check_variable = { var = independence_wave_form03_member_index value = independence_wave_formable_member_country_entries^num compare = less_than }
			}
			set_temp_variable = { independence_wave_form03_member_consent = independence_wave_formable_member_consent_entries^independence_wave_form03_member_index }
			var:independence_wave_formable_member_country_entries^independence_wave_form03_member_index = {
				if = {
					limit = {
						check_variable = { independence_wave_form03_member_consent = constant:independence_wave_formable_consent.consenting_member }
						NOT = { tag = ROOT }
						original_tag = AFX
						owns_state = 34
						controls_state = 34
					}
					independence_wave_formable_end_consenting_member_origin = yes
					ROOT = {
						transfer_state = 34
						add_state_core = 34
						set_temp_variable = { independence_wave_form03_additional_member_processed = constant:independence_wave_formable_tuning.loop_increment }
					}
				}
				else_if = {
					limit = {
						check_variable = { independence_wave_form03_member_consent = constant:independence_wave_formable_consent.consenting_member }
						NOT = { tag = ROOT }
						original_tag = AGX
						owns_state = 36
						controls_state = 36
					}
					independence_wave_formable_end_consenting_member_origin = yes
					ROOT = {
						transfer_state = 36
						add_state_core = 36
						set_temp_variable = { independence_wave_form03_additional_member_processed = constant:independence_wave_formable_tuning.loop_increment }
					}
				}
				else_if = {
					limit = {
						check_variable = { independence_wave_form03_member_consent = constant:independence_wave_formable_consent.consenting_member }
						has_independence_wave_form03_belgian_declared_consent = yes
					}
					set_country_flag = independence_wave_form03_autonomous_member
					set_country_flag = independence_wave_form03_founding_member
					ROOT = {
						set_country_flag = independence_wave_form03_has_belgian_member
						set_temp_variable = { independence_wave_form03_additional_member_processed = constant:independence_wave_formable_tuning.loop_increment }
					}
				}
			}
			add_to_temp_variable = { independence_wave_form03_member_index = constant:independence_wave_formable_tuning.loop_increment }
		}
		if = {
			limit = {
				check_variable = {
					var = independence_wave_form03_additional_member_processed
					value = constant:independence_wave_formable_tuning.loop_increment
					compare = greater_than_or_equals
				}
			}
			if = {
				limit = { original_tag = AFX }
				add_state_core = 34
				set_capital = {
					state = 34
					remember_old_capital = no
				}
			}
			else_if = {
				limit = { original_tag = AGX }
				add_state_core = 36
				set_capital = {
					state = 36
					remember_old_capital = no
				}
			}
			clear_array = independence_wave_formable_integration_state_entries
			set_country_flag = independence_wave_formable_integration_committed
		}
	}
}

# BEL/HOL/LUX country scope. Joining records a sovereign confederal association
# only; it does not add the member to a faction, create a subject, move a state,
# change a cosmetic tag, or alter a focus tree.
independence_wave_form03_join_active_carrier_as_autonomous_member = {
	if = {
		limit = {
			NOT = { has_country_flag = independence_wave_form03_autonomous_member }
			has_independence_wave_form03_active_carrier_connection = yes
		}
		set_country_flag = independence_wave_form03_autonomous_member
		set_country_flag = independence_wave_form03_post_charter_member
		if = {
			limit = { original_tag = BEL has_cosmetic_tag = BEL_flanders }
			AFX = {
				if = {
					limit = { is_independence_wave_form03_active_carrier = yes }
					set_country_flag = independence_wave_form03_has_belgian_member
				}
			}
			AGX = {
				if = {
					limit = { is_independence_wave_form03_active_carrier = yes }
					set_country_flag = independence_wave_form03_has_belgian_member
				}
			}
		}
		else_if = {
			limit = { original_tag = HOL }
			AFX = {
				if = {
					limit = { is_independence_wave_form03_active_carrier = yes }
					set_country_flag = independence_wave_form03_has_dutch_member
				}
			}
			AGX = {
				if = {
					limit = { is_independence_wave_form03_active_carrier = yes }
					set_country_flag = independence_wave_form03_has_dutch_member
				}
			}
		}
		else_if = {
			limit = { original_tag = LUX }
			AFX = {
				if = {
					limit = { is_independence_wave_form03_active_carrier = yes }
					set_country_flag = independence_wave_form03_has_luxembourgish_member
				}
			}
			AGX = {
				if = {
					limit = { is_independence_wave_form03_active_carrier = yes }
					set_country_flag = independence_wave_form03_has_luxembourgish_member
				}
			}
		}
		independence_wave_form03_initialize_late_autonomous_member = yes
		AFX = {
			if = {
				limit = { has_independence_wave_form03_post_charter_progression = yes }
				independence_wave_form03_reconcile_member_mirrors = yes
			}
		}
		AGX = {
			if = {
				limit = { has_independence_wave_form03_post_charter_progression = yes }
				independence_wave_form03_reconcile_member_mirrors = yes
			}
		}
	}
}
