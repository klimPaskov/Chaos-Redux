# ============================================================================
# CHAOS REDUX - EVENT 042 EQUIPMENT FROM HEAVENS CONVENTIONAL REGISTRY
# ============================================================================
# Curated conventional family registry for Equipment from Heavens. Every row
# resolves to a complete usable item under the active DLC configuration:
# designer families create one neutral complete variant on the recipient and
# grant it by variant name, while non-designer configurations use complete
# legacy tokens. Bare chassis and empty airframes are never granted.
#
# Interface (COUNTRY scope, recipient):
# - equipment_from_heavens_load_family_profile
#   input temp efh_profile_family, efh_active_stage
#   output temp efh_profile_min_stage, efh_profile_max_tier, efh_profile_group,
#   efh_profile_qclass, efh_profile_designer, efh_profile_mismatch
# - equipment_from_heavens_resolve_current_tier
#   input temp efh_profile_family, output temp efh_current_tier
# - equipment_from_heavens_resolve_family_token
#   input temp efh_profile_family, efh_grant_tier; output country variables
#   efh_grant_token, efh_grant_name_token, efh_grant_family and temp
#   efh_grant_designer; designer rows also ensure the neutral variant exists.
# Tier 4 airframes, jets, and rocket interceptors are outside the registry
# because their engine and module validity is not proven across every DLC
# configuration; the highest safe tier per family is listed in each profile.
# ============================================================================

# The recipient's research only labels which tier is "current" for a family.
# It never changes quantity or removes a family.
equipment_from_heavens_resolve_current_tier = {
	set_temp_variable = { efh_current_tier = 1 }
	if = {
		limit = { check_variable = { efh_profile_family = constant:equipment_from_heavens_family.infantry } }
		if = {
			limit = { has_tech = advanced_infantry_weapons }
			set_temp_variable = { efh_current_tier = 3 }
		}
		else_if = {
			limit = { has_tech = improved_infantry_weapons }
			set_temp_variable = { efh_current_tier = 2 }
		}
	}
	else_if = {
		limit = { check_variable = { efh_profile_family = constant:equipment_from_heavens_family.artillery } }
		if = {
			limit = { has_tech = artillery4 }
			set_temp_variable = { efh_current_tier = 3 }
		}
		else_if = {
			limit = { has_tech = artillery1 }
			set_temp_variable = { efh_current_tier = 2 }
		}
	}
	else_if = {
		limit = { check_variable = { efh_profile_family = constant:equipment_from_heavens_family.anti_tank } }
		if = {
			limit = { has_tech = antitank5 }
			set_temp_variable = { efh_current_tier = 3 }
		}
		else_if = {
