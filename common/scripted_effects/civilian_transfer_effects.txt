# STATE scope. Railway damage is neutral physical route substrate shared by
# famine blockade evaluation and migration route/capacity consumers. Both
# mechanics refresh it only from their existing sparse registered-state jobs.
civilian_transfer_refresh_route_damage_projection = {
	set_temp_variable = { civilian_transfer_route_damage_changed = constant:humanitarian_runtime.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			check_variable = { var = damaged_building_level@rail_way value = constant:civilian_transfer_route_projection.damage_threshold compare = greater_than }
		}
		if = {
			limit = { NOT = { has_state_flag = civilian_transfer_route_damaged } }
			set_state_flag = civilian_transfer_route_damaged
			set_temp_variable = { civilian_transfer_route_damage_changed = constant:humanitarian_runtime.one }
		}
	}
	else_if = {
		limit = { has_state_flag = civilian_transfer_route_damaged }
		clr_state_flag = civilian_transfer_route_damaged
		set_temp_variable = { civilian_transfer_route_damage_changed = constant:humanitarian_runtime.one }
	}
	if = {
		limit = { check_variable = { var = civilian_transfer_route_damage_changed value = constant:humanitarian_runtime.one compare = equals } }
		migration_mark_reception_capacity_dirty_if_candidate = yes
	}
}

# Exact current-host proof for the physical debit. Every successful debit must
# carry a positive exact cohort id. The resolver finds exactly one aligned live
# row and proves this state is its persisted authoritative host.
civilian_transfer_prove_current_host = {
	# An exact physical debit has no safe proof without a positive cohort ID.
	# Generic caller scope is therefore rejected instead of being treated as an
	# authoritative host.
	set_temp_variable = { civilian_transfer_host_proof = constant:humanitarian_runtime.zero }
	set_temp_variable = { civilian_transfer_host_match_count = constant:humanitarian_runtime.zero }
	if = {
		limit = { check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than } }
		set_temp_variable = { civilian_transfer_host_proof = constant:humanitarian_runtime.zero }
		if = {
			limit = { migration_cohort_arrays_are_aligned = yes }
			set_temp_variable = { civilian_transfer_host_proof_index = global.migration_cohort_ids^num }
			while_loop_effect = {
				limit = { check_variable = { var = civilian_transfer_host_proof_index value = constant:humanitarian_runtime.zero compare = greater_than } }
				subtract_from_temp_variable = { civilian_transfer_host_proof_index = constant:humanitarian_runtime.array_index_increment }
				if = {
					limit = {
						check_variable = { var = global.migration_cohort_ids^civilian_transfer_host_proof_index value = migration_cohort_id_request compare = equals }
						OR = {
							check_variable = { var = global.migration_cohort_statuses^civilian_transfer_host_proof_index value = constant:migration_cohort_status.active compare = equals }
							check_variable = { var = global.migration_cohort_statuses^civilian_transfer_host_proof_index value = constant:migration_cohort_status.destination_bound compare = equals }
							check_variable = { var = global.migration_cohort_statuses^civilian_transfer_host_proof_index value = constant:migration_cohort_status.destination_bound_unsafe compare = equals }
						}
						check_variable = { var = global.migration_cohort_amounts^civilian_transfer_host_proof_index value = constant:humanitarian_population.zero compare = greater_than }
					}
					add_to_temp_variable = { civilian_transfer_host_match_count = constant:humanitarian_runtime.one }
					var:global.migration_cohort_hosts^civilian_transfer_host_proof_index = {
						if = {
							limit = { state = event_target:civilian_transfer_physical_origin }
							if = {
								limit = { check_variable = { var = global.migration_cohort_amounts^civilian_transfer_host_proof_index value = civilian_transfer_people compare = equals } }
								set_temp_variable = { civilian_transfer_host_proof = constant:humanitarian_runtime.one }
							}
						}
					}
				}
			}
		}
		if = {
			limit = { NOT = { check_variable = { var = civilian_transfer_host_match_count value = constant:humanitarian_runtime.one compare = equals } } }
			set_temp_variable = { civilian_transfer_host_proof = constant:humanitarian_runtime.zero }
		}
	}
}

# A successful exact debit settles only explicitly supplied paired obligations.
# Each paired ledger receives one identical settlement equal to the minimum of
# the measured debit and both ledger values; route deaths are never touched.
# A mismatched larger residual remains visible for repair. The projection
# recorder is called from this same success path so failed or zero transactions
# stay inert.
civilian_transfer_reconcile_obligations = {
	set_temp_variable = { civilian_transfer_flight_settled = constant:humanitarian_population.zero }
	set_temp_variable = { civilian_transfer_trapped_settled = constant:humanitarian_population.zero }
	if = {
		limit = { check_variable = { var = civilian_transfer_actual_origin_debit value = constant:humanitarian_population.zero compare = greater_than } }
		if = {
			limit = { check_variable = { var = civilian_transfer_flight_obligation_proven value = constant:humanitarian_runtime.one compare = equals } }
			set_temp_variable = { migration_flight_settlement_amount = civilian_transfer_actual_origin_debit }
			if = { limit = { NOT = { has_variable = migration_flight_pressure } } set_temp_variable = { migration_flight_settlement_amount = constant:humanitarian_population.zero } }
			if = { limit = { NOT = { has_variable = migration_state_flight_population } } set_temp_variable = { migration_flight_settlement_amount = constant:humanitarian_population.zero } }
			if = { limit = { has_variable = migration_flight_pressure check_variable = { var = migration_flight_pressure value = migration_flight_settlement_amount compare = less_than } } set_temp_variable = { migration_flight_settlement_amount = migration_flight_pressure } }
			if = { limit = { has_variable = migration_state_flight_population check_variable = { var = migration_state_flight_population value = migration_flight_settlement_amount compare = less_than } } set_temp_variable = { migration_flight_settlement_amount = migration_state_flight_population } }
			if = {
				limit = { check_variable = { var = migration_flight_settlement_amount value = constant:humanitarian_population.zero compare = greater_than } }
				# One paired settlement is applied to both ledgers. If either side
				# is short, the larger residual remains visible for repair rather
				# than allowing the pair to drift further.
				subtract_from_variable = { migration_flight_pressure = migration_flight_settlement_amount }
				subtract_from_variable = { migration_state_flight_population = migration_flight_settlement_amount }
			}
			set_temp_variable = { civilian_transfer_flight_settled = migration_flight_settlement_amount }
		}
		if = {
			limit = { check_variable = { var = civilian_transfer_trapped_obligation_proven value = constant:humanitarian_runtime.one compare = equals } }
			set_temp_variable = { migration_trapped_settlement_amount = civilian_transfer_actual_origin_debit }
			if = { limit = { NOT = { has_variable = migration_trapped_population } } set_temp_variable = { migration_trapped_settlement_amount = constant:humanitarian_population.zero } }
			if = { limit = { NOT = { has_variable = migration_state_trapped_population } } set_temp_variable = { migration_trapped_settlement_amount = constant:humanitarian_population.zero } }
			if = { limit = { has_variable = migration_trapped_population check_variable = { var = migration_trapped_population value = migration_trapped_settlement_amount compare = less_than } } set_temp_variable = { migration_trapped_settlement_amount = migration_trapped_population } }
			if = { limit = { has_variable = migration_state_trapped_population check_variable = { var = migration_state_trapped_population value = migration_trapped_settlement_amount compare = less_than } } set_temp_variable = { migration_trapped_settlement_amount = migration_state_trapped_population } }
			if = {
				limit = { check_variable = { var = migration_trapped_settlement_amount value = constant:humanitarian_population.zero compare = greater_than } }
				# The trapped pair follows the same minimum settlement rule. A
				# mismatched residual is retained for an explicit repair pass.
				subtract_from_variable = { migration_trapped_population = migration_trapped_settlement_amount }
				subtract_from_variable = { migration_state_trapped_population = migration_trapped_settlement_amount }
				migration_publish_reception_demand = yes
				famine_consume_migration_reception_demand = yes
			}
			set_temp_variable = { civilian_transfer_trapped_settled = migration_trapped_settlement_amount }
		}
	}
}

# Same-chain compensation for a failed post-debit finalization. Reception
# receipts are inverted, survivor population is removed from the destination,

# ledgers have not yet committed, so they require no inverse entry.

	set_temp_variable = { civilian_transfer_rollback_result = constant:civilian_transfer_route_result.invalid }
	set_temp_variable = { civilian_transfer_rollback_destination_reception = constant:humanitarian_population.zero }

# after source/destination reception and role-correct bind receipts match the
# measured amounts. Repeated calls fail because pending authority is consumed.

				# consumes the measured debit and pre-debit population exactly once

						}
					}
					migration_presentation_clear_transit_projection = yes
				}
				if = {
					limit = {
						OR = {
							check_variable = { var = civilian_transfer_finalize_role value = constant:migration_core_reconciliation.transfer_projection_role_voluntary_return compare = equals }
							check_variable = { var = civilian_transfer_finalize_role value = constant:migration_core_reconciliation.transfer_projection_role_forced_return compare = equals }
						}
					}
					migration_presentation_clear_return_readiness_projection = yes
				}
				# A positive survivor receipt is the only source for hosted transit or
				# resettlement/return presentation people. Route deaths and all-death
				# rows never create a destination presentation receipt.
				if = {
					limit = {
						has_event_target = civilian_transfer_route_destination
						check_variable = { var = civilian_transfer_survivor_credit value = constant:humanitarian_population.zero compare = greater_than }
						check_variable = { var = civilian_transfer_destination_population_after value = constant:humanitarian_population.zero compare = greater_than }
					}
					event_target:civilian_transfer_route_destination = {
						if = {
							limit = { check_variable = { var = civilian_transfer_finalize_role value = constant:migration_core_reconciliation.transfer_projection_role_transit compare = equals } }

							set_temp_variable = { migration_presentation_input_generation = global.civilian_transfer_generation }

						}
						if = {
