independence_wave_refresh_kar_lifecycle = {
	if = {
		limit = { is_independence_wave_kar_package = yes }
		remove_ideas = { kar_contested_forest_administration kar_forest_state_compact }
		if = {
			limit = { has_stable_independence_wave_kar_ledgers = yes }
			add_ideas = kar_forest_state_compact
			set_country_flag = independence_wave_kar_ledgers_stabilized
		}
		else = {
			add_ideas = kar_contested_forest_administration
			clr_country_flag = independence_wave_kar_ledgers_stabilized
		}
	}
	if = {
		limit = { is_independence_wave_cri_package = yes }
		remove_ideas = { cri_contested_peninsula_administration cri_return_state_compact }
		if = {
			limit = { has_stable_independence_wave_cri_ledgers = yes }
			add_ideas = cri_return_state_compact
			set_country_flag = independence_wave_cri_ledgers_stabilized
		}
		else = {
			add_ideas = cri_contested_peninsula_administration
			clr_country_flag = independence_wave_cri_ledgers_stabilized
		}
	}
}
