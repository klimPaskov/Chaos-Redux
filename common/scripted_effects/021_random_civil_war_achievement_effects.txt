# ============================================================================
# CHAOS REDUX - EVENT 021 HISTORICAL ACHIEVEMENT RECEIPTS
# ============================================================================
# Country-local history starts only at a committed opening. State-control and
# capitulation callbacks preserve failures between bounded country reviews.
# These helpers do not launch crises, select targets, or scan the world.
# ============================================================================

event021_inherit_achievement_disqualification = {
	if = {
		limit = {
			OR = {
				is_debug = yes
				has_country_flag = force_trigger_mode_enabled
				ROOT = { has_country_flag = force_trigger_mode_enabled }
				has_country_flag = random_civil_war_scenario_bypass_active
				has_global_flag = random_civil_war_manual_scenario_launch_in_progress
				AND = {
					has_event_target = random_civil_war_global_host_country
					event_target:random_civil_war_global_host_country = {
						has_country_flag = random_civil_war_achievement_disqualified
					}
				}
			}
		}
		set_country_flag = random_civil_war_achievement_disqualified
	}
}

# COUNTRY scope on the surviving ordinary claimant, before former-host cleanup.
# The caller binds random_civil_war_history_predecessor explicitly.
# Copy historical facts, never original-player status or an earned achievement.
event021_inherit_successor_achievement_history = {
	save_event_target_as = random_civil_war_history_successor
	if = {
		limit = {
			has_event_target = random_civil_war_history_predecessor
			event_target:random_civil_war_history_predecessor = {
				has_country_flag = random_civil_war_achievement_history_started
				has_variable = random_civil_war_crisis_id
				check_variable = { var = random_civil_war_crisis_id value = PREV.random_civil_war_crisis_id compare = equals }
			}
		}
		set_country_flag = random_civil_war_achievement_history_started
		set_variable = { random_civil_war_original_capital = event_target:random_civil_war_history_predecessor.random_civil_war_original_capital }
		set_variable = { random_civil_war_opening_state_count = event_target:random_civil_war_history_predecessor.random_civil_war_opening_state_count }
		clear_array = random_civil_war_opening_all_core_states
		event_target:random_civil_war_history_predecessor = {
			for_each_scope_loop = {
				array = random_civil_war_opening_all_core_states
				event_target:random_civil_war_history_successor = {
					add_to_array = { array = random_civil_war_opening_all_core_states value = PREV }
				}
			}
			if = {
				limit = { has_country_flag = random_civil_war_external_war_at_opening }
				event_target:random_civil_war_history_successor = { set_country_flag = random_civil_war_external_war_at_opening }
			}
			if = {
				limit = { has_country_flag = random_civil_war_external_war_capitulated }
				event_target:random_civil_war_history_successor = { set_country_flag = random_civil_war_external_war_capitulated }
			}
			if = {
				limit = { has_country_flag = random_civil_war_achievement_disqualified }
				event_target:random_civil_war_history_successor = { set_country_flag = random_civil_war_achievement_disqualified }
			}
			if = {
				limit = { has_country_flag = random_civil_war_harsh_settlement }
				event_target:random_civil_war_history_successor = { set_country_flag = random_civil_war_harsh_settlement }
			}
			if = {
				limit = { has_country_flag = random_civil_war_evolution_i_multifront }
				event_target:random_civil_war_history_successor = { set_country_flag = random_civil_war_evolution_i_multifront }
			}
		}
	}
}

event021_begin_achievement_history = {
	event021_inherit_achievement_disqualification = yes
	save_event_target_as = random_civil_war_history_country
	set_country_flag = random_civil_war_achievement_history_started
	# A committed recurrence cannot reuse the previous war's terminal receipts.
	# Completed achievement flags and long-lived recurrence memory stay intact.
	clr_country_flag = random_civil_war_settlement_recorded
	clr_country_flag = random_civil_war_reconstruction_complete
	clr_country_flag = random_civil_war_internal_fronts_resolved
	clr_country_flag = random_civil_war_government_victory
	# These are command/UI terminal receipts, not settlement history. A later
	# committed crisis must expose its actions and cannot consume an old offer.
	clr_country_flag = random_civil_war_cleanup_complete
	clr_country_flag = random_civil_war_cleanup_requested
	clr_country_flag = event021_settlement_pending
	clr_country_flag = event021_settlement_signed
	clr_country_flag = event021_settlement_resolved
	clr_country_flag = event021_settlement_talks_open
	clr_country_flag = random_civil_war_settlement_pending
	clr_country_flag = random_civil_war_settlement_phase
	clr_country_flag = random_civil_war_reconstruction_resolved
	clr_country_flag = random_civil_war_reconstruction_phase
	clr_country_flag = random_civil_war_decisive_capital_lost
	clr_country_flag = random_civil_war_external_war_capitulated
	clr_country_flag = random_civil_war_authority_ever_collapsed
	clr_country_flag = random_civil_war_replacement_capital_valid_at_opening
	clear_variable = random_civil_war_decisive_capital
	clear_variable = random_civil_war_minimum_authority
	# Settlement and reconstruction receipts are generation-local. Treaty
	# recurrence is recorded before this reset, so the durable agreement registry
	# retains the previous terms while the new crisis starts with empty inputs.
	clear_variable = random_civil_war_settlement_type
	clear_variable = random_civil_war_settlement_date
	clear_variable = random_civil_war_reconstruction_until
	clear_variable = random_civil_war_settlement_obligation
	clear_variable = random_civil_war_settlement_obligation_front_id
	clear_variable = random_civil_war_settlement_obligation_start_date
	clear_variable = random_civil_war_settlement_obligation_anchor_state
	clear_variable = random_civil_war_settlement_obligation_actor_id
	clear_variable = random_civil_war_settlement_terms_hold_until
	set_variable = { random_civil_war_settled_front_count = constant:random_civil_war_value.zero }
	clr_country_flag = random_civil_war_settlement_obligation_active
	clr_country_flag = random_civil_war_settlement_obligation_fulfilled
	clr_country_flag = random_civil_war_settlement_obligation_breached
	clr_country_flag = random_civil_war_settlement_obligation_failed
	clr_country_flag = random_civil_war_settlement_obligation_objective_met
	clr_country_flag = random_civil_war_settlement_obligation_objective_missed
	clr_country_flag = random_civil_war_settlement_terms_held
	capital_scope = {
		set_temp_variable = { event021_history_capital_scope = THIS }
		if = {
			limit = {
				is_owned_and_controlled_by = event_target:random_civil_war_history_country
				OR = {
					check_variable = { var = event021_history_capital_scope value = PREV.random_civil_war_original_capital compare = equals }
					AND = {
						state_population_k > constant:event021_parent_tuning.minimum_state_population_k
						check_variable = { infrastructure_level > constant:random_civil_war_value.zero }
					}
				}
			}
			save_event_target_as = random_civil_war_history_capital
			event_target:random_civil_war_history_country = {
				set_variable = { random_civil_war_decisive_capital = event_target:random_civil_war_history_capital }
			}
			if = {
				limit = { NOT = { check_variable = { var = event021_history_capital_scope value = PREV.random_civil_war_original_capital compare = equals } } }
				event_target:random_civil_war_history_country = { set_country_flag = random_civil_war_replacement_capital_valid_at_opening }
			}
		}
	}
	event021_update_authority_band = yes
	event021_record_capital_continuity = yes
}

event021_record_capital_continuity = {
	if = {
		limit = {
			has_country_flag = random_civil_war_active
			has_country_flag = random_civil_war_achievement_history_started
			NOT = { has_country_flag = random_civil_war_internal_fronts_resolved }
		}
		save_event_target_as = random_civil_war_history_country
		if = {
			limit = {
				OR = {
					NOT = { has_variable = random_civil_war_decisive_capital }
					var:random_civil_war_decisive_capital = {
						NOT = { is_owned_and_controlled_by = event_target:random_civil_war_history_country }
					}
				}
			}
			set_country_flag = random_civil_war_decisive_capital_lost
		}
	}
}

# Called only by on_capitulation: ROOT is the defeated country, FROM the winner.
# An Event 021 claimant shares the crisis id; other victors are external.
event021_record_external_capitulation = {
	if = {
		limit = {
			has_country_flag = random_civil_war_active
			has_country_flag = random_civil_war_external_war_at_opening
			has_country_flag = random_civil_war_achievement_history_started
			has_variable = random_civil_war_crisis_id
			NOT = { has_country_flag = random_civil_war_internal_fronts_resolved }
			NOT = {
				OR = {
					is_in_array = { array = random_civil_war_opposition_actors value = FROM }
					FROM = {
						has_variable = random_civil_war_crisis_id
						check_variable = { var = random_civil_war_crisis_id value = ROOT.random_civil_war_crisis_id compare = equals }
					}
				}
			}
		}
		set_country_flag = random_civil_war_external_war_capitulated
	}
}
