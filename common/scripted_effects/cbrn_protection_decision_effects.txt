cbrn_resolve_defective_reconditioned_batch = {
	if = {
		limit = {
			has_variable = cbrn_reconditioned_batch_size
			has_equipment = { gas_mask_equipment_reconditioned > constant:cbrn_math.zero }
		}
		set_temp_variable = { cbrn_defective_batch_loss = cbrn_reconditioned_batch_size }
		multiply_temp_variable = { cbrn_defective_batch_loss = constant:cbrn_protection_program_effect.defective_batch_loss_ratio }
		clamp_temp_variable = { var = cbrn_defective_batch_loss min = constant:cbrn_mask_scale.minimum_action_crates max = num_equipment@gas_mask_equipment_reconditioned }
		round_temp_variable = cbrn_defective_batch_loss
		set_temp_variable = { cbrn_defective_batch_debit = cbrn_defective_batch_loss }
		multiply_temp_variable = { cbrn_defective_batch_debit = -1 }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_reconditioned
			amount = cbrn_defective_batch_debit
		}
		add_to_variable = { cbrn_mask_replacement_demand = cbrn_defective_batch_loss }
		set_temp_variable = { cbrn_readiness_delta = constant:cbrn_protection_program_effect.defective_batch_readiness_loss }
		cbrn_modify_chemical_readiness = yes
		clear_variable = cbrn_reconditioned_batch_size
		cbrn_refresh_country_mask_snapshot = yes
	}
	else = {
		clear_variable = cbrn_reconditioned_batch_size
	}
}
