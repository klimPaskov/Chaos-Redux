# The effects behind the decisions, missions and event options of the
# campaign. Decisions only call these effects, so the AI, the player and the
# cleanup paths run the same code. Payment effects (malta_pay_*) live in
# 038_malta_crusaders_cost_effects.txt. All effects run in COUNTRY scope on
# the actor (MLT) unless stated otherwise.

# banners, regional campaign, settlement, the Eleventh Crusade, mission
# outcomes and achievement hooks.

# ---------------------------------------------------------------------------
# Demand answers
# ---------------------------------------------------------------------------

# Pays the price of the open demand in full and honours it.
malta_accept_open_demand = {
	if = {
		limit = { check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.treasury compare = equals } }
		malta_pay_demand_treasury = yes
	}
	else_if = {
		limit = { check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.command compare = equals } }
		malta_pay_demand_command = yes
	}
	else_if = {
		limit = { check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.levy compare = equals } }
		malta_pay_demand_levy = yes
	}
	else = {
		malta_pay_demand_headquarters = yes
	}
	set_variable = { malta_order_compromise_used^malta_demand_order = 0 }
	set_temp_variable = { malta_answer = 1 }
	malta_apply_demand_answer = yes
}

# Pays half and gives the order half of what it asked for.
malta_compromise_open_demand = {
	if = {
		limit = { check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.treasury compare = equals } }
		malta_pay_demand_treasury_half = yes
	}
	else_if = {
		limit = { check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.command compare = equals } }
		malta_pay_demand_command_half = yes
	}
	else_if = {
		limit = { check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.levy compare = equals } }
		malta_pay_demand_levy_half = yes
	}
	else = {
		malta_pay_demand_headquarters_half = yes
	}
	set_variable = { malta_order_compromise_used^malta_demand_order = 1 }
	set_temp_variable = { malta_answer = 2 }
	malta_apply_demand_answer = yes
}

# Turns the request down.
malta_refuse_open_demand = {
	set_temp_variable = { malta_answer = 3 }
	malta_apply_demand_answer = yes
}

# Concede: the grievance is met, at the price of political weight.
malta_incident_concede = {
	set_temp_variable = { malta_order = malta_incident_order }
	set_variable = { malta_order_grievance^malta_order = 0 }
	malta_raise_influence = yes
	set_temp_variable = { malta_change = constant:malta_incident_change.concede_cohesion }
	malta_change_cohesion = yes
	set_temp_variable = { malta_pay = constant:malta_incident_change.concede_political_power }
	add_political_power = malta_pay
	malta_record_incident = yes
	clr_country_flag = malta_incident_open
	malta_refresh_standing = yes
}

# Raises the grievance of every seated order by one.
malta_raise_grievance_of_all_seats = {
	for_loop_effect = {
		start = constant:malta_order.hospitallers
		end = constant:malta_council.order_count
		compare = less_than_or_equals
		value = malta_scan

			limit = { check_variable = { var = malta_order_seated^malta_scan value = 1 compare = equals } }
			set_temp_variable = { malta_order = malta_scan }
			malta_raise_grievance = yes
		}
	}
}

# STATE scope. Adds the owner of a registry state that Malta does not hold to
# the campaign target list once.
malta_collect_target = {
	if = {
		limit = {
			malta_state_is_usable = yes
			malta_state_is_open_for_claims = yes
			NOT = { is_owned_by = MLT }
		}
		owner = {
			if = {
				limit = {
					exists = yes
					is_special_chaos_country = no
					is_actual_nonhuman_country = no
					NOT = { malta_is_chartered_principality = yes }
					NOT = { is_in_array = { array = global.malta_campaign_targets value = THIS } }
				}
				add_to_array = { global.malta_campaign_targets = THIS }
			}
		}
	}
}

# Input: FROM is the target country. Claims, war and the one-regional-war rule.
malta_launch_regional_crusade = {
	malta_pay_crusade = yes
	save_event_target_as = malta_actor
	FROM = {
		malta_add_registry_claims = yes
		save_event_target_as = malta_regional_target
	}
	set_country_flag = malta_regional_crusade_active
	set_variable = { malta_regional_target = FROM }
	event_target:malta_actor = {
		declare_war_on = {
			target = event_target:malta_regional_target
			type = take_claimed_state
		}
	}
}

# Clears the regional crusade bookkeeping when the war is over.
malta_review_regional_crusade = {
	if = {
		limit = {
			has_country_flag = malta_regional_crusade_active
			NOT = { has_variable = malta_regional_target }
		}
		clr_country_flag = malta_regional_crusade_active
	}
	else_if = {
		limit = {
			has_country_flag = malta_regional_crusade_active
			has_variable = malta_regional_target
			OR = {
				var:malta_regional_target = { NOT = { exists = yes } }
				NOT = { has_war_with = var:malta_regional_target }
			}
		}
		clr_country_flag = malta_regional_crusade_active
		clear_variable = malta_regional_target
	}
}

# Input: FROM is the target country. Saves the landing state and country.
malta_select_landing = {
	set_country_flag = malta_eleventh_landing_chosen
	set_variable = { malta_landing_target = FROM }
	FROM = {
		random_owned_state = {
			limit = { malta_state_is_campaign_pool = yes }
			save_global_event_target_as = malta_landing_state
		}
	}
}

malta_launch_eleventh_crusade = {
	malta_pay_launch = yes
	save_event_target_as = malta_spawn_owner
	set_country_flag = malta_eleventh_launched
	set_variable = { malta_landing_hold_days = 0 }
	save_event_target_as = malta_actor
	var:malta_landing_target = {
		malta_add_registry_claims = yes
		save_event_target_as = malta_regional_target
	}
	if = {
		limit = { NOT = { has_war_with = event_target:malta_regional_target } }
		declare_war_on = {
			target = event_target:malta_regional_target
			type = take_claimed_state
		}
	}
	set_variable = { global.malta_spawn_xp = constant:malta_force.experience_factor }
	set_variable = { global.malta_spawn_equipment = constant:malta_force.equipment_factor }
	set_variable = { global.malta_spawn_manpower = constant:malta_force.manpower_factor }
	set_temp_variable = { malta_landing_force = malta_eleventh_failures }
	add_to_temp_variable = { malta_landing_force = malta_package_level }
	if = {
		limit = { has_country_flag = malta_naval_landings }
		add_to_temp_variable = { malta_landing_force = 1 }
	}
	if = {
		limit = { has_country_flag = malta_eleventh_vow }
		add_to_temp_variable = { malta_landing_force = 1 }
	}
	add_to_temp_variable = { malta_landing_force = constant:malta_eleventh_change.landing_force }
	clamp_temp_variable = { var = malta_landing_force min = 1 max = constant:malta_eleventh_change.landing_force_cap }
	set_variable = { global.malta_spawn_template = constant:malta_template.knight }
	set_temp_variable = { malta_spawn_count = malta_landing_force }
	event_target:malta_landing_state = {
		malta_spawn_landing_loop = yes
	}
	activate_mission = malta_hold_new_foothold_mission
}

# STATE scope. Landing divisions may appear on a hostile coast.
malta_spawn_landing_loop = {
	for_loop_effect = {
		start = 0
		end = malta_spawn_count
		value = malta_spawn_n
		meta_effect = {
			text = {
				create_unit = {
					division = "name = \"[DIV_NAME]\" division_template = \"[TEMPLATE_NAME]\" start_experience_factor = [XP_FACTOR] start_equipment_factor = [EQ_FACTOR] start_manpower_factor = [MP_FACTOR]"
					owner = event_target:malta_spawn_owner
					allow_spawning_on_enemy_provs = yes
				}
			}
			DIV_NAME = "[This.GetMaltaDivisionName]"
			TEMPLATE_NAME = "[This.GetMaltaTemplateName]"
			XP_FACTOR = "[?global.malta_spawn_xp|.2]"
			EQ_FACTOR = "[?global.malta_spawn_equipment|.2]"
			MP_FACTOR = "[?global.malta_spawn_manpower|.2]"
		}
	}
}
