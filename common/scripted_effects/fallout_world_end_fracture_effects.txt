# ============================================================================
# CHAOS REDUX - FALLOUT INDEPENDENCE WAVE FRACTURE CONTRACT
# ============================================================================
# This file owns the release-gated border fracture. It uses reserved
# Independence Wave country definitions only as graphical and map-color
# templates. It never loads their focus trees. The live conflict ledger is
# built before a template can be copied, and the original country tags are
# preserved by the separate map-return receipt.

# Supported scope: COUNTRY. Called on a reserved template scope.
fallout_fracture_record_template_status = {
	if = {
		limit = { exists = yes }
		add_to_array = { array = global.fallout_fracture_blocked_templates value = THIS }
		add_to_variable = { global.fallout_fracture_blocked_template_count = constant:fallout_fracture_contract.basic_floor_infantry }
	}
	else = {
		add_to_array = { array = global.fallout_fracture_safe_templates value = THIS }
		add_to_variable = { global.fallout_fracture_safe_template_count = constant:fallout_fracture_contract.basic_floor_infantry }
	}
}

# Supported scope: COUNTRY. Selects a random severely scarred source and one
# non-capital state. The random draw is deliberately after the seven-day
# blackout work and is generation/date-seeded by the engine.
fallout_fracture_select_source_and_state = {
	clear_event_target = fallout_fracture_source
	clear_event_target = fallout_fracture_state
	random_country = {
		limit = { fallout_fracture_source_is_eligible = yes }
		save_event_target_as = fallout_fracture_source
		random_owned_state = {
			limit = {
				NOT = { is_capital = yes }
				OR = {
					check_variable = {
						var = fallout_state_grade
						value = constant:fallout_state_grade.dead_city
						compare = greater_than_or_equals
					}
					check_variable = {
						var = fallout_state_grade
						value = constant:fallout_state_grade.wasteland
						compare = greater_than_or_equals
					}
				}
			}
			save_event_target_as = fallout_fracture_state
		}
	}
}

# Supported scope: COUNTRY. Shared output setup. The helper is called only
# after a concrete reserved tag has been copied.
fallout_fracture_finalize_dynamic_output = {
	event_target:fallout_fracture_output = {
		transfer_state = event_target:fallout_fracture_state
		add_state_core = event_target:fallout_fracture_state
		set_capital = {
			state = event_target:fallout_fracture_state
			remember_old_capital = no
		}
		set_popularities = {
			democratic = 0
			fascism = 0
			communism = 0
			neutrality = 100
		}
		set_politics = {
			ruling_party = neutrality
			elections_allowed = no
		}
		set_country_flag = fallout_fracture_dynamic_country
		set_country_flag = fallout_country_survives
		set_country_flag = fallout_country_classified
		set_country_flag = fallout_country_package_applied
		set_country_flag = fallout_focus_package_applied
		set_country_flag = fallout_archetype_package_applied
		set_country_flag = fallout_regional_package_applied
		set_country_flag = fallout_country_memory_package_applied
		set_variable = { fallout_country_package_generation = global.fallout_transition_generation }
		set_variable = { fallout_focus_package_generation = global.fallout_transition_generation }
		set_variable = { fallout_archetype_package_generation = global.fallout_transition_generation }
		set_variable = { fallout_regional_package_generation = global.fallout_transition_generation }
		set_variable = { fallout_country_memory_package_generation = global.fallout_transition_generation }
		set_variable = { fallout_fracture_output_generation = global.fallout_transition_generation }
		fallout_regress_country_technology = yes
	}
	event_target:fallout_fracture_state = {
		set_state_flag = fallout_fracture_state_transferred
		set_variable = { fallout_fracture_output_country = event_target:fallout_fracture_output }
		set_variable = { fallout_fracture_generation = global.fallout_transition_generation }
	}
	event_target:fallout_fracture_source = {
		set_country_flag = fallout_fracture_source_split
		set_variable = { fallout_fracture_generation = global.fallout_transition_generation }
	}
	add_to_array = { array = global.fallout_fracture_output_countries value = event_target:fallout_fracture_output }
	add_to_array = { array = global.fallout_fracture_output_states value = event_target:fallout_fracture_state }
	add_to_variable = { global.fallout_fracture_output_count = constant:fallout_fracture_contract.basic_floor_infantry }
}

# Each branch is deliberately explicit because create_dynamic_country requires
# a literal source template token. No branch loads the source focus tree.
fallout_fracture_materialize_ACX = {
	if = { limit = { is_in_array = { array = global.fallout_fracture_safe_templates value = ACX } NOT = { ACX = { exists = yes } } } event_target:fallout_fracture_source = { create_dynamic_country = { original_tag = ACX copy_tag = ACX reserve_dynamic_country = yes save_event_target_as = fallout_fracture_output } } }
	if = { limit = { has_event_target = fallout_fracture_output } fallout_fracture_finalize_dynamic_output = yes }
}

fallout_fracture_materialize_one_dynamic_country = {
	if = {
		limit = { has_event_target = fallout_fracture_source has_event_target = fallout_fracture_state }
		clear_event_target = fallout_fracture_output
		random_list = {
			100 = { fallout_fracture_materialize_ACX = yes }
			100 = { fallout_fracture_materialize_AFX = yes }
			100 = { fallout_fracture_materialize_AGX = yes }
			100 = { fallout_fracture_materialize_AJX = yes }
			100 = { fallout_fracture_materialize_ARX = yes }
			100 = { fallout_fracture_materialize_ASX = yes }
		}
		if = {
			limit = { NOT = { has_event_target = fallout_fracture_output } }
			if = { limit = { NOT = { has_event_target = fallout_fracture_output } } fallout_fracture_materialize_ACX = yes }
			if = { limit = { NOT = { has_event_target = fallout_fracture_output } } fallout_fracture_materialize_AFX = yes }
			if = { limit = { NOT = { has_event_target = fallout_fracture_output } } fallout_fracture_materialize_AGX = yes }
			if = { limit = { NOT = { has_event_target = fallout_fracture_output } } fallout_fracture_materialize_AJX = yes }
			if = { limit = { NOT = { has_event_target = fallout_fracture_output } } fallout_fracture_materialize_ARX = yes }
			if = { limit = { NOT = { has_event_target = fallout_fracture_output } } fallout_fracture_materialize_ASX = yes }
		}
	}
}

# Supported scope: COUNTRY. Release-gated entry point used by the map-return
# transaction. A missing proof gate records a blocker rather than silently
# falling back to static borders.
fallout_apply_fracture_contract = {
	if = {
		limit = { fallout_fracture_release_is_enabled = yes }
		fallout_build_fracture_conflict_ledger = yes
		if = {
			limit = { fallout_fracture_conflict_ledger_is_current = yes }
			fallout_fracture_select_source_and_state = yes
			fallout_fracture_materialize_one_dynamic_country = yes
			if = {
				limit = { has_event_target = fallout_fracture_output }
				set_global_flag = fallout_fracture_transition_applied
				set_variable = { global.fallout_fracture_transition_generation = global.fallout_transition_generation }
			}
			else = {
				set_global_flag = fallout_fracture_transition_blocked
				set_temp_variable = { fallout_transition_error_to_register = constant:fallout_transition_error_code.fracture_conflict_ledger_incomplete }
				fallout_register_transition_error_if_unowned = yes
				set_temp_variable = { fallout_transition_error_to_register = constant:fallout_transition_error_code.none }
			}
		}
	}
	else = { set_global_flag = fallout_fracture_transition_unproven }
}
