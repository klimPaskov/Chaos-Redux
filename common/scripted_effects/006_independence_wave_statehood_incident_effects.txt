# ============================================================================
# CHAOS REDUX - EVENT 006 STATEHOOD INCIDENT PULSE
# ============================================================================
# Overview:
# - `independence_wave_start_statehood_incidents` runs in the released country
#   when it resolves its founding posture and queues its first pulse.
# - Each pulse (hidden event chaosx.nr6.150) weighs every eligible incident in
#   six baseline categories (founding, recognition, security, government,
#   League, formable) and in the evolution category, then opens at most one
#   incident and queues the next pulse after a random interval.
# - Weights come from live country state: the founding phase raises founding
#   incidents, low recognition raises recognition incidents, war or a severe
#   host threat raises security incidents, an undecided government raises
#   government incidents, League leadership or crisis raises League incidents
#   and an open founding proposal raises formable incidents. A settled country
#   draws a quiet season more often, and a country in severe instability less
#   often.
# - Each incident fires once per generation. Opening one puts its whole
#   category on a cooldown stored as a date through
#   chaosx_convert_days_to_date_offset.
# - `independence_wave_clear_statehood_incident_runtime` clears every receipt;
#   the generation reset and origin end call it.
# No effect iterates countries or runs on a periodic world hook.
# ============================================================================

# COUNTRY scope: the released country after its founding posture resolves.
independence_wave_start_statehood_incidents = {
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_variable = independence_wave_generation_id
			NOT = { has_country_flag = independence_wave_statehood_incident_pulse_queued }
		}
		set_temp_variable = { independence_wave_statehood_incident_delay = constant:independence_wave_statehood_incident.first_delay_days }
		independence_wave_queue_statehood_incident_pulse = yes
	}
}

# COUNTRY scope. Input: temporary `independence_wave_statehood_incident_delay`
# in days. The due date sits one margin day before the event, so the event
# always arrives on or after it.
independence_wave_queue_statehood_incident_pulse = {
	set_country_flag = independence_wave_statehood_incident_pulse_queued
	set_variable = { independence_wave_statehood_incident_generation = independence_wave_generation_id }
	set_temp_variable = { chaosx_date_days = independence_wave_statehood_incident_delay }
	subtract_from_temp_variable = { chaosx_date_days = constant:independence_wave_statehood_incident.due_margin_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { independence_wave_statehood_incident_due_date = global.date }
	add_to_variable = { independence_wave_statehood_incident_due_date = chaosx_date_offset }
	country_event = { id = chaosx.nr6.150 days = independence_wave_statehood_incident_delay }
}
