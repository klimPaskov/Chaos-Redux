# Event 19 registry-management contract. Base zombies are the only zombie
# family exposed as trainable; this path authorizes their exact generated
# template and never unlocks a mutated or weaponized zombie battalion.
chaos_unit_family_provider_501_event19_evaluate_management = {
	set_temp_variable = { infantry_spawn_family_provider_can_train = 0 }
	set_temp_variable = { infantry_spawn_family_provider_can_spawn = 0 }
	set_temp_variable = { infantry_spawn_family_provider_can_sustain = 0 }
	set_temp_variable = { infantry_spawn_family_provider_uses_training = 1 }
	if = {
		limit = {
			NOT = { is_in_array = { array = infantry_spawn_trainable_family_ids value = constant:event19_zombie_family.family_id } }
			infantry_spawn_country_has_eligible_state = yes
			check_variable = { var = army_experience value = constant:event19_zombie_family.training_army_experience_cost compare = greater_than_or_equals }
			check_variable = { var = num_equipment@infantry_equipment value = constant:event19_zombie_family.training_infantry_equipment_cost compare = greater_than_or_equals }
			check_variable = { var = manpower value = constant:event19_zombie_family.training_manpower_cost compare = greater_than_or_equals }
		}
		set_temp_variable = { infantry_spawn_family_provider_can_train = 1 }
	}
	if = {
		limit = {
			check_variable = { infantry_spawn_family_live_division_count > 0 }
			check_variable = { var = num_equipment@infantry_equipment value = constant:event19_zombie_family.sustainment_infantry_equipment_cost compare = greater_than_or_equals }
			check_variable = { var = manpower value = constant:event19_zombie_family.sustainment_manpower_cost compare = greater_than_or_equals }
		}
		set_temp_variable = { infantry_spawn_family_provider_can_sustain = 1 }
	}
}

chaos_unit_family_provider_501_event19_pay_management_action = {
	set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
			check_variable = { infantry_spawn_family_provider_can_train > 0 }
		}
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_zombie_family.training_army_experience_cost }
		army_experience = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_zombie_family.training_infantry_equipment_cost }
		add_equipment_to_stockpile = { type = infantry_equipment amount = infantry_spawn_family_provider_negative_cost }
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_zombie_family.training_manpower_cost }
		add_manpower = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
	else_if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.sustainment }
			check_variable = { infantry_spawn_family_provider_can_sustain > 0 }
		}
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_zombie_family.sustainment_infantry_equipment_cost }
		add_equipment_to_stockpile = { type = infantry_equipment amount = infantry_spawn_family_provider_negative_cost }
		set_temp_variable = { infantry_spawn_family_provider_negative_cost = 0 }
		subtract_from_temp_variable = { infantry_spawn_family_provider_negative_cost = constant:event19_zombie_family.sustainment_manpower_cost }
		add_manpower = infantry_spawn_family_provider_negative_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
}

chaos_unit_family_provider_501_event19_refund_management_action = {
	if = {
		limit = {
			check_variable = { infantry_spawn_family_provider_payment_succeeded > 0 }
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
		}
		add_army_experience = constant:event19_zombie_family.training_army_experience_cost
		add_equipment_to_stockpile = { type = infantry_equipment amount = constant:event19_zombie_family.training_infantry_equipment_cost }
		add_manpower = constant:event19_zombie_family.training_manpower_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	}
	else_if = {
		limit = {
			check_variable = { infantry_spawn_family_provider_payment_succeeded > 0 }
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.sustainment }
		}
		add_equipment_to_stockpile = { type = infantry_equipment amount = constant:event19_zombie_family.sustainment_infantry_equipment_cost }
		add_manpower = constant:event19_zombie_family.sustainment_manpower_cost
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	}
}
