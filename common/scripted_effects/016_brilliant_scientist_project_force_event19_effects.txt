# ============================================================================
# CHAOS REDUX - EVENT 016 GENERIC EVENT 019 PROVIDERS
# ============================================================================
# Event 016 owns project history, native templates, equipment unlocks, and
# native force materialisation. This file is the external provider surface for
# Event 019: it registers eight generic family identities (504-510 and 522) only after
# the Event 016 runtime package is current, then supplies template, spawn,
# management, sustainment, and isolated-derivative callbacks.
#
# The provider rows use provider-neutral battalion and equipment identifiers.
# No Event 016 native spawn helper is called by this file.
# ============================================================================

# COUNTRY scope. Repeated calls are safe because the shared registry performs
# an exact idempotence check and rejects conflicting duplicate rows.
brilliant_scientist_register_event19_project_force_providers = {
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_cloning_active
				has_country_flag = chaosx_custom_technology_clone_operational
			}
		}
		chaos_unit_family_provider_504_register = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_robotics_active
				has_country_flag = chaosx_custom_technology_robot_operational
			}
		}
		chaos_unit_family_provider_505_register = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_paleogenetics_active
				has_country_flag = chaosx_custom_technology_paleogenetic_operational
			}
		}
		chaos_unit_family_provider_506_register = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_xenobiological_synthesis_active
				has_country_flag = chaosx_custom_technology_xenobiological_operational
			}
		}
		chaos_unit_family_provider_507_register = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_alien_arms_active
				has_country_flag = chaosx_custom_technology_exotic_operational
			}
		}
		chaos_unit_family_provider_508_register = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_teleportation_active
				has_country_flag = chaosx_custom_technology_portal_operational
			}
		}
		chaos_unit_family_provider_509_register = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_project_force_temporal_active
				has_country_flag = chaosx_custom_technology_temporal_operational
			}
		}
		chaos_unit_family_provider_510_register = yes
	}
	# Provider 522 is deliberately narrower than the neutral clone provider:
	# only the living German Mengele program with the completed cloning project,
	# public master-race claim, and explicit Aryan refinement may expose it.
	if = {
		limit = {
			germany_mengele_is_germany_scope = yes
			germany_mengele_program_active = yes
			has_country_flag = germany_mengele_cloning_project_completed
			has_country_flag = germany_master_race_claim_established
			has_tech = mengele_aryan_clone_refinement_tech
		}
		chaos_unit_family_provider_522_register = yes
	}
}
