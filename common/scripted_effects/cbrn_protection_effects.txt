# ============================================================================
# CHAOS REDUX - SHARED CBRN PROTECTION EFFECTS
# ============================================================================
# Overview:
# - Owns initialization and clamping for national CBRN capacity values.
# - Applies readiness changes only inside the current institutional cap.
# - Resolves respiratory, skin, antidote, decontamination, medical, and warning
#   layers into agent-class-specific casualty, disruption, and contamination
#   multipliers.
# - Does not infer equipment coverage. Stage 2 equipment/issue helpers must
#   supply all six protection layers and then call the calculator.
# ============================================================================

cbrn_set_use_policy = {
	set_temp_variable = { cbrn_policy_change_accepted = constant:cbrn_proof.missing }
	if = {
		limit = {
			has_variable = cbrn_requested_use_policy
			check_variable = {
				var = cbrn_requested_use_policy
				value = constant:cbrn_use_policy.defensive_preparation
				compare = greater_than_or_equals
			}
			check_variable = {
				var = cbrn_requested_use_policy
				value = constant:cbrn_use_policy.unrestricted_chaos_warfare
				compare = less_than_or_equals
			}
		}
		set_variable = { cbrn_use_policy = cbrn_requested_use_policy }
		set_temp_variable = { cbrn_policy_change_accepted = constant:cbrn_proof.supplied }
	}
}

cbrn_set_chemical_readiness_cap = {
	cbrn_initialize_country_data = yes
	if = {
		limit = { has_variable = cbrn_requested_readiness_cap }
		set_variable = { chemical_readiness_cap = cbrn_requested_readiness_cap }
		clamp_variable = { var = chemical_readiness_cap min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
		clamp_variable = { var = chemical_readiness min = constant:cbrn_meter.minimum max = chemical_readiness_cap }
	}
}

cbrn_modify_chemical_readiness = {
	cbrn_initialize_country_data = yes
	if = {
		limit = { has_variable = cbrn_readiness_delta }
		add_to_variable = { chemical_readiness = cbrn_readiness_delta }
		clamp_variable = { var = chemical_readiness min = constant:cbrn_meter.minimum max = chemical_readiness_cap }
	}
}

cbrn_calculate_action_protection = {
	cbrn_reset_protection_outputs = yes
	if = {
		limit = {
			cbrn_action_agent_class_is_chemical = yes
			has_variable = cbrn_protection_respiratory
			has_variable = cbrn_protection_skin
			has_variable = cbrn_protection_antidote
			has_variable = cbrn_protection_decontamination
			has_variable = cbrn_protection_medical
			has_variable = cbrn_protection_warning
		}
		set_temp_variable = { cbrn_protection_respiratory_local = cbrn_protection_respiratory }
		set_temp_variable = { cbrn_protection_skin_local = cbrn_protection_skin }
		set_temp_variable = { cbrn_protection_antidote_local = cbrn_protection_antidote }
		set_temp_variable = { cbrn_protection_decontamination_local = cbrn_protection_decontamination }
		set_temp_variable = { cbrn_protection_medical_local = cbrn_protection_medical }
		set_temp_variable = { cbrn_protection_warning_local = cbrn_protection_warning }

		clamp_temp_variable = { var = cbrn_protection_respiratory_local min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
		clamp_temp_variable = { var = cbrn_protection_skin_local min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
		clamp_temp_variable = { var = cbrn_protection_antidote_local min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
		clamp_temp_variable = { var = cbrn_protection_decontamination_local min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
		clamp_temp_variable = { var = cbrn_protection_medical_local min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }

		subtract_from_variable = { cbrn_civilian_mask_total_reconditioned = PREV.cbrn_civilian_mask_reconditioned_crates }
		subtract_from_variable = { cbrn_civilian_mask_total_basic = PREV.cbrn_civilian_mask_basic_crates }
		subtract_from_variable = { cbrn_civilian_mask_total_improved = PREV.cbrn_civilian_mask_improved_crates }
		subtract_from_variable = { cbrn_civilian_mask_total_advanced = PREV.cbrn_civilian_mask_advanced_crates }
		subtract_from_variable = { cbrn_civilian_mask_total_sealed = PREV.cbrn_civilian_mask_sealed_crates }
		subtract_from_variable = { cbrn_civilian_mask_total_crates = cbrn_transfer_old_total }
		clamp_variable = { var = cbrn_civilian_mask_total_reconditioned min = constant:cbrn_math.zero }
		clamp_variable = { var = cbrn_civilian_mask_total_basic min = constant:cbrn_math.zero }
		clamp_variable = { var = cbrn_civilian_mask_total_improved min = constant:cbrn_math.zero }
		clamp_variable = { var = cbrn_civilian_mask_total_advanced min = constant:cbrn_math.zero }
		clamp_variable = { var = cbrn_civilian_mask_total_sealed min = constant:cbrn_math.zero }
		clamp_variable = { var = cbrn_civilian_mask_total_crates min = constant:cbrn_math.zero }
	}

	multiply_variable = { cbrn_civilian_mask_reconditioned_crates = cbrn_transfer_survival_mult }
	multiply_variable = { cbrn_civilian_mask_basic_crates = cbrn_transfer_survival_mult }
	multiply_variable = { cbrn_civilian_mask_improved_crates = cbrn_transfer_survival_mult }
	multiply_variable = { cbrn_civilian_mask_advanced_crates = cbrn_transfer_survival_mult }
	multiply_variable = { cbrn_civilian_mask_sealed_crates = cbrn_transfer_survival_mult }
	multiply_variable = { cbrn_civilian_mask_fit_points = cbrn_transfer_survival_mult }
	subtract_from_variable = { cbrn_civilian_mask_filter_condition = constant:cbrn_mask_filters.standing_issue_condition_loss }
	clamp_variable = { var = cbrn_civilian_mask_filter_condition min = constant:cbrn_mask_filters.condition_minimum max = constant:cbrn_mask_filters.condition_maximum }
	cbrn_refresh_state_civilian_mask_snapshot = yes
	set_temp_variable = { cbrn_transfer_new_total = cbrn_civilian_mask_crates }
	set_temp_variable = { cbrn_transfer_lost_total = cbrn_transfer_old_total }
	subtract_from_temp_variable = { cbrn_transfer_lost_total = cbrn_transfer_new_total }

	event_target:cbrn_new_state_controller = {
		cbrn_initialize_country_data = yes
		add_to_variable = { cbrn_civilian_mask_total_reconditioned = PREV.cbrn_civilian_mask_reconditioned_crates }
		add_to_variable = { cbrn_civilian_mask_total_basic = PREV.cbrn_civilian_mask_basic_crates }
		add_to_variable = { cbrn_civilian_mask_total_improved = PREV.cbrn_civilian_mask_improved_crates }
		add_to_variable = { cbrn_civilian_mask_total_advanced = PREV.cbrn_civilian_mask_advanced_crates }
		add_to_variable = { cbrn_civilian_mask_total_sealed = PREV.cbrn_civilian_mask_sealed_crates }
		add_to_variable = { cbrn_civilian_mask_total_crates = cbrn_transfer_new_total }
		add_to_variable = { cbrn_mask_replacement_demand = cbrn_transfer_lost_total }
		cbrn_refresh_country_mask_snapshot = yes
		cbrn_start_protection_maintenance_job = yes
	}
}

cbrn_start_protection_maintenance_job = {
	if = {
		limit = { NOT = { has_country_flag = cbrn_protection_maintenance_active } }
		set_country_flag = cbrn_protection_maintenance_active
		set_temp_variable = { cbrn_protection_maintenance_delay = constant:cbrn_scheduler.protection_maintenance_days }
		country_event = { id = cbrn_protection.1 days = cbrn_protection_maintenance_delay }
	}
}
