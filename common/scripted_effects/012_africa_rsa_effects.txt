# COUNTRY scope on the original host before the civil war. Freeze every valid
# pre-war candidate from the accepted bounded Event 12 stores. Later roster
# refreshes cannot expand this ledger.
africa_rsa_snapshot_exile_patron_candidates = {
	clear_array = africa_rsa_prewar_exile_patron_candidates
	for_each_scope_loop = {
		array = africa_relationship_countries
		if = {
			limit = {
				africa_rsa_is_valid_exile_patron = yes
				event_target:africa_rsa_continental_coalition = { NOT = { is_in_array = { array = africa_rsa_prewar_exile_patron_candidates value = PREV } } }
			}
			event_target:africa_rsa_continental_coalition = { add_to_array = { array = africa_rsa_prewar_exile_patron_candidates value = PREV } }
		}
	}
	for_each_scope_loop = {
		array = africa_selected_targets
		if = {
			limit = {
				africa_rsa_is_valid_exile_patron = yes
				event_target:africa_rsa_continental_coalition = { NOT = { is_in_array = { array = africa_rsa_prewar_exile_patron_candidates value = PREV } } }
			}
			event_target:africa_rsa_continental_coalition = { add_to_array = { array = africa_rsa_prewar_exile_patron_candidates value = PREV } }
		}
	}
	for_each_scope_loop = {
		array = africa_frozen_first_contact_targets
		if = {
			limit = {
				africa_rsa_is_valid_exile_patron = yes
				event_target:africa_rsa_continental_coalition = { NOT = { is_in_array = { array = africa_rsa_prewar_exile_patron_candidates value = PREV } } }
			}
			event_target:africa_rsa_continental_coalition = { add_to_array = { array = africa_rsa_prewar_exile_patron_candidates value = PREV } }
		}
	}
}

# COUNTRY scope on the current Event 12 host. Selection is deterministic and
# bounded to the pre-war frozen ledger. Governments that explicitly supported
# the Coalition during the war take priority. No refreshed-roster substitution
# or world search is permitted.
africa_rsa_select_exile_patron = {
	clear_global_event_target = africa_rsa_exile_patron
	set_temp_variable = { africa_rsa_patron_selected = constant:africa_rsa_count.none }
	for_each_scope_loop = {
		array = africa_rsa_external_supporters
		if = {
			limit = {
				check_variable = { africa_rsa_patron_selected = constant:africa_rsa_count.none }
				africa_rsa_is_valid_exile_patron = yes
			}
			save_global_event_target_as = africa_rsa_exile_patron
			set_temp_variable = { africa_rsa_patron_selected = constant:africa_rsa_count.one }
		}
	}
	for_each_scope_loop = {
		array = africa_rsa_prewar_exile_patron_candidates
		if = {
			limit = {
				check_variable = { africa_rsa_patron_selected = constant:africa_rsa_count.none }
				africa_rsa_is_valid_exile_patron = yes
			}
			save_global_event_target_as = africa_rsa_exile_patron
			set_temp_variable = { africa_rsa_patron_selected = constant:africa_rsa_count.one }
		}
	}
}

# COUNTRY scope on original SAF.
africa_rsa_snapshot_prewar_context = {
	clear_global_event_target = africa_rsa_continental_coalition
	clear_global_event_target = africa_rsa_allied_union_government
	clear_global_event_target = africa_rsa_original_capital_state
	clear_global_event_target = africa_rsa_original_overlord
	clear_global_event_target = africa_rsa_original_faction_leader
	clear_global_event_target = africa_rsa_allied_faction_leader
	clear_global_event_target = africa_rsa_exile_patron
	clear_array = africa_rsa_inherited_enemies
	clear_array = africa_rsa_allied_interveners
	clear_array = africa_rsa_coalition_interveners
	clear_array = africa_rsa_external_supporters
	clear_array = africa_rsa_regional_responses
	clear_array = africa_rsa_prewar_exile_patron_candidates
	set_variable = { africa_rsa_external_support_count = constant:africa_rsa_count.none }

	save_global_event_target_as = africa_rsa_continental_coalition
	capital_scope = { save_global_event_target_as = africa_rsa_original_capital_state }
	if = {
		limit = { is_in_faction = yes }
		faction_leader = { save_global_event_target_as = africa_rsa_original_faction_leader }
		if = { limit = { is_faction_leader = yes } set_country_flag = africa_rsa_was_faction_leader }
		else = { clr_country_flag = africa_rsa_was_faction_leader }
	}

	africa_rsa_snapshot_autonomy = yes
	set_variable = { africa_rsa_prewar_democratic_support = party_popularity@democratic }
	set_variable = { africa_rsa_prewar_communist_support = party_popularity@communism }
	set_variable = { africa_rsa_prewar_fascist_support = party_popularity@fascism }
	set_variable = { africa_rsa_prewar_neutrality_support = party_popularity@neutrality }

	every_enemy_country = {
		event_target:africa_rsa_continental_coalition = {
			add_to_array = { array = africa_rsa_inherited_enemies value = PREV }
		}
	}
	africa_rsa_snapshot_exile_patron_candidates = yes
	africa_rsa_select_exile_patron = yes

	set_variable = { africa_rsa_allied_stockpile_ratio = constant:africa_rsa_ratio.allied_stockpile }
	set_variable = { africa_rsa_allied_army_ratio = constant:africa_rsa_ratio.allied_army }
	set_variable = { africa_rsa_allied_navy_ratio = constant:africa_rsa_ratio.allied_navy }
	set_variable = { africa_rsa_allied_air_ratio = constant:africa_rsa_ratio.allied_air }
	set_variable = { africa_rsa_first_proof_days = constant:africa_rsa_days.first_proof }
	set_variable = { africa_rsa_settlement_truce_days = constant:africa_rsa_days.settlement_truce }
	set_variable = { africa_rsa_exile_recovery_delay_days = constant:africa_rsa_days.exile_recovery_delay }
}

# COUNTRY scope on the original SAF host after the revolter exists.
africa_rsa_assign_allied_framework = {
	africa_rsa_restore_allied_autonomy = yes
	if = {
		limit = { has_country_flag = africa_rsa_was_faction_leader }
		add_to_faction = event_target:africa_rsa_allied_union_government
		event_target:africa_rsa_allied_union_government = {
			set_faction_leader = yes
			save_global_event_target_as = africa_rsa_allied_faction_leader
		}
		leave_faction = yes
	}
	else_if = {
		limit = { has_event_target = africa_rsa_original_faction_leader }
		event_target:africa_rsa_original_faction_leader = {
			add_to_faction = event_target:africa_rsa_allied_union_government
			save_global_event_target_as = africa_rsa_allied_faction_leader
		}
		leave_faction = yes
	}
}

# COUNTRY scope on the Continental Coalition. This only removes the wars the
# original country carried into the split, leaving the loyalist government and
# its old faction at war with those same enemies.
africa_rsa_end_inherited_coalition_wars = {
	for_each_scope_loop = {
		array = africa_rsa_inherited_enemies
		if = {
			limit = { exists = yes }
			event_target:africa_rsa_continental_coalition = {
				if = {
					limit = { has_war_with = PREV }
					white_peace = PREV
				}
				set_truce = { target = PREV days = africa_rsa_settlement_truce_days }
			}
		}
	}
}

# COUNTRY scope on the Continental Coalition. This bounded faction traversal
# captures relations created before on_war_relation_added could see the saved
# loyalist target inside start_civil_war.
africa_rsa_seed_initial_allied_intervention_ledger = {
	if = {
		limit = { has_event_target = africa_rsa_allied_faction_leader }
		event_target:africa_rsa_allied_faction_leader = {
			every_faction_member = {
				limit = {
					exists = yes
					has_war_with = event_target:africa_rsa_continental_coalition
					event_target:africa_rsa_allied_union_government = { NOT = { tag = PREV } }
				}
				event_target:africa_rsa_continental_coalition = {
					if = {
						limit = { NOT = { is_in_array = { array = africa_rsa_allied_interveners value = PREV } } }
						add_to_array = { array = africa_rsa_allied_interveners value = PREV }
					}
				}
			}
		}
	}
}

# COUNTRY scope on the Continental Coalition.
africa_rsa_prepare_first_proof = {
	africa_refresh_bounded_african_target_roster = yes
	africa_initialize_compound_first_proof_ledger = yes
	if = {
		limit = { has_country_flag = africa_first_proof_ledger_active }
		set_country_flag = africa_first_proof_active
		set_country_flag = africa_first_proof_pending
		activate_mission = africa_rsa_first_proof_mission
	}
}

# COUNTRY scope on the dynamic revolter inside start_civil_war. The original
# SAF remains the sole Event 12 host, so host identity markers are explicitly
# stripped even if the engine copies country state into the revolter.
africa_rsa_initialize_allied_union_government = {
	save_global_event_target_as = africa_rsa_allied_union_government
	set_country_flag = africa_rsa_allied_union_government
	set_cosmetic_tag = AFRICA_RSA_ALLIED_UNION_GOVERNMENT
	clr_country_flag = africa_unifier_host
	clr_country_flag = africa_host_commit_completed
	clr_country_flag = africa_original_host_preserved
	clr_country_flag = africa_protection_first_opening
	clr_country_flag = africa_first_proof_active
	clr_country_flag = africa_first_proof_pending
	clr_country_flag = africa_first_proof_ledger_active
	set_variable = { africa_rsa_settlement_truce_days = constant:africa_rsa_days.settlement_truce }
	set_variable = { africa_rsa_exile_recovery_delay_days = constant:africa_rsa_days.exile_recovery_delay }
	set_popularities = {
		democratic = event_target:africa_rsa_continental_coalition.africa_rsa_prewar_democratic_support
		communism = event_target:africa_rsa_continental_coalition.africa_rsa_prewar_communist_support
		fascism = event_target:africa_rsa_continental_coalition.africa_rsa_prewar_fascist_support
		neutrality = event_target:africa_rsa_continental_coalition.africa_rsa_prewar_neutrality_support
	}
}

# Public integration API. COUNTRY scope on the original, player-led SAF host.
# Callers must satisfy africa_rsa_allied_branch_can_start; the effect contains
# the same hard gate and performs no substitute split when it fails.
africa_rsa_start_allied_civil_war = {
	if = {
		limit = { africa_rsa_allied_branch_can_start = yes }
		africa_rsa_snapshot_prewar_context = yes
		set_global_flag = africa_rsa_branch_used
		set_global_flag = africa_rsa_civil_war_active
