independence_wave_refresh_yak_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_sakha_package = yes }
		remove_ideas = { yak_fragmented_arctic_mandate yak_sakha_arctic_compact }
		if = {
			limit = { has_stable_independence_wave_yak_compact = yes }
			add_ideas = yak_sakha_arctic_compact
