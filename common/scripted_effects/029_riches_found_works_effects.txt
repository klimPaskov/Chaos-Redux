# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND PHYSICAL WORKS
# ============================================================================
# STATE scope throughout; THIS is the mine state. Completed transport projects
# build real map buildings, and destructive raids and collapses damage them.
# The game repairs damaged buildings through ordinary construction; the mine's
# own damage is repaired through Repair and Restore the Shaft. A works-project
# marker records projects under construction so the AI builds one at a time.
# Building effects follow the vanilla add_building_construction and
# damage_building precedents. Their level, damage and repair-speed fields do
# not read script constants, so those values are file-scoped constants.
# ============================================================================

@riches_found_route_building_level = 1
@riches_found_raid_infrastructure_damage = 1
@riches_found_raid_naval_base_damage = 1
@riches_found_collapse_infrastructure_damage = 2
@riches_found_damage_repair_speed = -0.25

# Rail Spur completion: one level of state infrastructure below the cap.
riches_found_build_rail_spur = {
	if = {
		limit = { check_variable = { infrastructure_level < constant:riches_found_build.infrastructure_max } }
		add_building_construction = {
			type = infrastructure
			level = @riches_found_route_building_level
			instant_build = yes
		}
	}
}

# Open the Railhead resolution on a rail route: connect the mine to the
# current controller's capital over allied ground. Existing connections are
# reused; a previous controller's laid flag is never connection evidence.
# The mission verifies the same endpoint predicate before awarding success.
riches_found_lay_mine_railway = {
	if = {
		limit = {
			has_state_flag = riches_found_rail_route
			CONTROLLER = { exists = yes }
		}
		if = {
			limit = { NOT = { riches_found_state_has_capital_rail_connection = yes } }
			CONTROLLER = {
				if = {
					limit = {
						can_build_railway = {
							start_state = PREV
							target_state = THIS.capital
							build_only_on_allied = yes
						}
					}
					build_railway = {
						level = @riches_found_route_building_level
						start_state = PREV
						target_state = THIS.capital
						build_only_on_allied = yes
					}
				}
			}
		}
		if = {
			limit = { riches_found_state_has_capital_rail_connection = yes }
			set_state_flag = riches_found_railway_laid
		}
	}
}

# Port Route completion: one naval base level where the state already has a
# port, otherwise a new naval base on its coast.
riches_found_build_port_route = {
	if = {
		limit = { is_coastal = yes }
		if = {
			limit = { naval_base > constant:riches_found_value.zero }
			if = {
				limit = { naval_base < constant:riches_found_build.naval_base_max }
				add_building_construction = {
					type = naval_base
					level = @riches_found_route_building_level
					instant_build = yes
					province = {
						all_provinces = yes
						limit_to_coastal = yes
						limit_to_naval_base = yes
					}
				}
			}
		}
		else = {
			add_building_construction = {
				type = naval_base
				level = @riches_found_route_building_level
				instant_build = yes
				province = {
					all_provinces = yes
					limit_to_coastal = yes
				}
			}
		}
	}
}

# A destructive or lethal raid damages the port when it struck the sea route,
# and otherwise the state's infrastructure.
riches_found_damage_raided_works = {
	if = {
		limit = {
			check_variable = { riches_found_incident_kind = constant:riches_found_incident_kind.raid_port_route }
			naval_base > constant:riches_found_value.zero
		}
		damage_building = {
			type = naval_base
			damage = @riches_found_raid_naval_base_damage
			repair_speed_modifier = @riches_found_damage_repair_speed
		}
	}
	else_if = {
		limit = { check_variable = { infrastructure_level > constant:riches_found_value.zero } }
		damage_building = {
			type = infrastructure
			damage = @riches_found_raid_infrastructure_damage
			repair_speed_modifier = @riches_found_damage_repair_speed
		}
	}
}

# A collapse that closes the workings brings down the surface infrastructure
# around the shafts.
riches_found_damage_collapsed_works = {
	if = {
		limit = { check_variable = { infrastructure_level > constant:riches_found_value.zero } }
		damage_building = {
			type = infrastructure
			damage = @riches_found_collapse_infrastructure_damage
			repair_speed_modifier = @riches_found_damage_repair_speed
		}
	}
}

# STATE scope, ROOT the paying controller. A physical works project (Rail
# Spur, Port Route, Processing Works, Worker Housing, Shaft Reinforcement,
# Sanitation or Repair) has been paid for. The count lets the AI start one
# physical project at a time; a count left by an earlier controller is
# replaced.
riches_found_mark_works_project = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = riches_found_works_project_controller }
				NOT = { check_variable = { var = riches_found_works_project_controller value = ROOT compare = equals } }
			}
		}
		set_variable = { riches_found_works_project_count = constant:riches_found_value.zero }
		set_variable = { riches_found_works_project_controller = ROOT }
	}
	add_to_variable = { riches_found_works_project_count = constant:riches_found_value.one }
	set_state_flag = riches_found_works_project_active
}

# STATE scope, ROOT the paying controller. One of that controller's works
# projects has landed or been refunded; the marker clears with the last one.
# Control changes clear the marker through riches_found_clear_stale_targets.
riches_found_clear_works_project = {
	if = {
		limit = {
			has_variable = riches_found_works_project_controller
			check_variable = { var = riches_found_works_project_controller value = ROOT compare = equals }
		}
		subtract_from_variable = { riches_found_works_project_count = constant:riches_found_value.one }
		if = {
			limit = { check_variable = { var = riches_found_works_project_count value = constant:riches_found_value.zero compare = less_than_or_equals } }
			clr_state_flag = riches_found_works_project_active
			clear_variable = riches_found_works_project_controller
			clear_variable = riches_found_works_project_count
		}
	}
}
