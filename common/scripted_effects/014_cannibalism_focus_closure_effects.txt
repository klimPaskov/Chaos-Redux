cannibalism_wendigo_start_terminal_hunt = {
	set_temp_variable = { cannibalism_wendigo_terminal_hunt_start_result = 0 }
	if = {
		limit = {
			has_event_target = cannibalism_wendigo_decision_target
			cannibalism_wendigo_can_pay_terminal_hunt_launch_cost = yes
			event_target:cannibalism_wendigo_decision_target = { cannibalism_wendigo_is_valid_terminal_hunt_target = yes }
		}
		set_temp_variable = { cannibalism_focus_closure_larder_cost = constant:cannibalism_wendigo_terminal_hunt.launch_larder }
		cannibalism_pay_focus_closure_larder_cost = yes
		set_temp_variable = { cannibalism_current_cost = constant:cannibalism_wendigo_terminal_hunt.launch_command }
		cannibalism_pay_current_command_cost = yes
		set_temp_variable = { cannibalism_current_cost = constant:cannibalism_wendigo_terminal_hunt.launch_infantry }
		cannibalism_pay_current_infantry_equipment_cost = yes
		set_temp_variable = { cannibalism_current_cost = constant:cannibalism_wendigo_terminal_hunt.launch_support }
		cannibalism_pay_current_support_equipment_cost = yes

		event_target:cannibalism_wendigo_decision_target = {
			save_global_event_target_as = cannibalism_wendigo_terminal_hunt_target
			set_country_flag = cannibalism_wendigo_terminal_hunt_defender
			set_temp_variable = { cannibalism_wendigo_terminal_hunt_target_lock_days = constant:cannibalism_wendigo_terminal_hunt.target_lock_days }
			set_country_flag = { flag = cannibalism_wendigo_terminal_hunt_target_lock days = cannibalism_wendigo_terminal_hunt_target_lock_days }
		}
		set_country_flag = cannibalism_wendigo_terminal_hunt_active
		clr_country_flag = cannibalism_wendigo_terminal_hunt_press_cooldown
		clr_country_flag = cannibalism_wendigo_inherited_cell_hunt_pressure_applied
		set_variable = { cannibalism_wendigo_terminal_hunt_pressure = constant:cannibalism_wendigo_terminal_hunt.starting_pressure }
		set_variable = { cannibalism_wendigo_terminal_hunt_counterpressure = 0 }
		set_variable = { cannibalism_wendigo_terminal_hunt_start_date = global.date }
		set_variable = { cannibalism_wendigo_terminal_hunt_due_date = global.date }
		add_to_variable = { cannibalism_wendigo_terminal_hunt_due_date = constant:cannibalism_wendigo_terminal_hunt.mission_days }
		activate_mission = cannibalism_wendigo_terminal_hunt_mission
		add_ai_strategy = { type = conquer id = event_target:cannibalism_wendigo_terminal_hunt_target value = constant:cannibalism_wendigo_ai.prelock_target_high }
		add_ai_strategy = { type = antagonize id = event_target:cannibalism_wendigo_terminal_hunt_target value = constant:cannibalism_wendigo_ai.prelock_target_high }
		add_ai_strategy = { type = front_unit_request id = event_target:cannibalism_wendigo_terminal_hunt_target value = constant:cannibalism_wendigo_ai.prelock_target_high }
		set_temp_variable = { cannibalism_wendigo_terminal_hunt_start_result = 1 }
	}
}

		remove_mission = cannibalism_wendigo_terminal_hunt_mission
	}
	if = {
		limit = { has_event_target = cannibalism_wendigo_terminal_hunt_target }
		event_target:cannibalism_wendigo_terminal_hunt_target = {
			clr_country_flag = cannibalism_wendigo_terminal_hunt_defender

				limit = { ROOT = { check_variable = { cannibalism_wendigo_terminal_hunt_clear_target_lock > 0 } } }

	clr_country_flag = cannibalism_wendigo_terminal_hunt_active
	clr_country_flag = cannibalism_wendigo_terminal_hunt_press_cooldown

cannibalism_resolve_wendigo_terminal_hunt_success = {
	if = {
		limit = { cannibalism_wendigo_terminal_hunt_has_succeeded = yes }
		if = {
			limit = { NOT = { has_variable = cannibalism_wendigo_terminal_hunts_completed } }
			set_variable = { cannibalism_wendigo_terminal_hunts_completed = 0 }
		}
		add_to_variable = { cannibalism_wendigo_terminal_hunts_completed = 1 }
		add_to_variable = { cannibalism_wendigo_transformation_progress = constant:cannibalism_wendigo_terminal_hunt.success_progress }
		clamp_variable = {
			var = cannibalism_wendigo_transformation_progress
			min = constant:cannibalism_wendigo_countdown.progress_min
			max = constant:cannibalism_wendigo_countdown.progress_max
		}
		set_temp_variable = { cannibalism_wendigo_terminal_hunt_launch_cooldown_days = constant:cannibalism_wendigo_terminal_hunt.success_cooldown_days }
		set_country_flag = { flag = cannibalism_wendigo_terminal_hunt_launch_cooldown days = cannibalism_wendigo_terminal_hunt_launch_cooldown_days }
		set_temp_variable = { cannibalism_wendigo_terminal_hunt_remove_mission = 0 }
		set_temp_variable = { cannibalism_wendigo_terminal_hunt_clear_target_lock = 0 }
		cannibalism_clear_wendigo_terminal_hunt_runtime = yes
	}
}

cannibalism_clear_current_wendigo_enemy_death_receipt_epoch = {
	clear_variable = cannibalism_wendigo_enemy_casualties_snapshot
	clear_variable = cannibalism_wendigo_enemy_death_remainder
	clear_variable = cannibalism_wendigo_enemy_death_receipts_issued
}

cannibalism_initialize_wendigo_enemy_receipt_epoch_for_target = {
	if = {
		limit = {
			has_event_target = cannibalism_wendigo_receipt_actor
			has_event_target = cannibalism_wendigo_receipt_epoch_target
			event_target:cannibalism_wendigo_receipt_epoch_target = {
				exists = yes
				has_war_with = event_target:cannibalism_wendigo_receipt_actor
				cannibalism_target_country_is_ordinary_human_target = yes
			}
		}
		event_target:cannibalism_wendigo_receipt_epoch_target = {
			cannibalism_clear_current_wendigo_enemy_death_receipt_epoch = yes
			set_variable = { cannibalism_wendigo_enemy_casualties_snapshot = casualties }
			set_variable = { cannibalism_wendigo_enemy_death_remainder = 0 }
			set_variable = { cannibalism_wendigo_enemy_death_receipts_issued = 0 }
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = event_target:cannibalism_wendigo_receipt_actor.cannibalism_wendigo_enemy_death_receipt_tracked_countries
							value = THIS
						}
					}
				}
				event_target:cannibalism_wendigo_receipt_actor = {
					add_to_array = { cannibalism_wendigo_enemy_death_receipt_tracked_countries = event_target:cannibalism_wendigo_receipt_epoch_target }
				}
			}
		}
	}
}

cannibalism_prune_wendigo_enemy_death_receipt_registry = {
	save_event_target_as = cannibalism_wendigo_receipt_actor
	clear_temp_array = cannibalism_wendigo_active_receipt_enemy_countries
	for_each_scope_loop = {
		array = cannibalism_wendigo_enemy_death_receipt_tracked_countries
		if = {
			limit = {
				exists = yes
				has_war_with = event_target:cannibalism_wendigo_receipt_actor
			}
			add_to_temp_array = { cannibalism_wendigo_active_receipt_enemy_countries = THIS }
		}
		else = {
			cannibalism_clear_current_wendigo_enemy_death_receipt_epoch = yes
		}
	}
	resize_array = { cannibalism_wendigo_enemy_death_receipt_tracked_countries = 0 }
	for_each_scope_loop = {
		array = cannibalism_wendigo_active_receipt_enemy_countries
		event_target:cannibalism_wendigo_receipt_actor = {
			add_to_array = { cannibalism_wendigo_enemy_death_receipt_tracked_countries = PREV }
		}
	}
	clear_temp_array = cannibalism_wendigo_active_receipt_enemy_countries
}

cannibalism_clear_wendigo_enemy_death_receipt_registry = {
	for_each_scope_loop = {
		array = cannibalism_wendigo_enemy_death_receipt_tracked_countries
		cannibalism_clear_current_wendigo_enemy_death_receipt_epoch = yes
	}
	resize_array = { cannibalism_wendigo_enemy_death_receipt_tracked_countries = 0 }
	clear_temp_array = cannibalism_wendigo_active_receipt_enemy_countries
}

cannibalism_initialize_wendigo_enemy_death_receipts = {
	cannibalism_clear_wendigo_enemy_death_receipt_registry = yes
	set_country_flag = cannibalism_wendigo_enemy_death_receipts_open
	set_variable = { cannibalism_wendigo_enemy_death_receipts = 0 }
	save_event_target_as = cannibalism_wendigo_receipt_actor
	every_enemy_country = {
		save_event_target_as = cannibalism_wendigo_receipt_epoch_target
		event_target:cannibalism_wendigo_receipt_actor = { cannibalism_initialize_wendigo_enemy_receipt_epoch_for_target = yes }
	}
}

cannibalism_process_current_wendigo_enemy_death_delta = {
	set_temp_variable = { cannibalism_wendigo_enemy_current_casualties = casualties }
	if = {
		limit = { NOT = { has_variable = cannibalism_wendigo_enemy_casualties_snapshot } }
		set_variable = { cannibalism_wendigo_enemy_casualties_snapshot = cannibalism_wendigo_enemy_current_casualties }
		set_variable = { cannibalism_wendigo_enemy_death_remainder = 0 }
		set_variable = { cannibalism_wendigo_enemy_death_receipts_issued = 0 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = cannibalism_wendigo_enemy_current_casualties
				value = cannibalism_wendigo_enemy_casualties_snapshot
				compare = less_than
			}
		}
		set_variable = { cannibalism_wendigo_enemy_casualties_snapshot = cannibalism_wendigo_enemy_current_casualties }
		set_variable = { cannibalism_wendigo_enemy_death_remainder = 0 }
	}
	else = {
		set_temp_variable = { cannibalism_wendigo_enemy_new_casualties = cannibalism_wendigo_enemy_current_casualties }
		subtract_from_temp_variable = { cannibalism_wendigo_enemy_new_casualties = cannibalism_wendigo_enemy_casualties_snapshot }
		set_variable = { cannibalism_wendigo_enemy_casualties_snapshot = cannibalism_wendigo_enemy_current_casualties }
		if = {
			limit = { check_variable = { cannibalism_wendigo_enemy_new_casualties > 0 } }
			add_to_variable = { cannibalism_wendigo_enemy_death_remainder = cannibalism_wendigo_enemy_new_casualties }
		}
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_wendigo_enemy_death_remainder
					value = constant:cannibalism_wendigo_enemy_death_receipt.casualties_per_receipt
					compare = greater_than_or_equals
				}
				check_variable = {
					var = cannibalism_wendigo_enemy_death_receipts_issued
					value = constant:cannibalism_wendigo_enemy_death_receipt.per_enemy_cap
					compare = less_than
				}
				event_target:cannibalism_wendigo_receipt_actor = {
					check_variable = {
						var = cannibalism_wendigo_enemy_death_receipts
						value = constant:cannibalism_wendigo_enemy_death_receipt.pool_cap
						compare = less_than
					}
				}
			}
			subtract_from_variable = { cannibalism_wendigo_enemy_death_remainder = constant:cannibalism_wendigo_enemy_death_receipt.casualties_per_receipt }
			add_to_variable = { cannibalism_wendigo_enemy_death_receipts_issued = 1 }
			event_target:cannibalism_wendigo_receipt_actor = { add_to_variable = { cannibalism_wendigo_enemy_death_receipts = 1 } }
		}
	}
}

cannibalism_process_wendigo_enemy_death_receipts = {
	if = {
		limit = { cannibalism_wendigo_can_process_enemy_death_receipts = yes }
		save_event_target_as = cannibalism_wendigo_receipt_actor
		cannibalism_prune_wendigo_enemy_death_receipt_registry = yes
		every_enemy_country = {
			save_event_target_as = cannibalism_wendigo_receipt_epoch_target
			if = {
				limit = {
					is_in_array = {
						array = event_target:cannibalism_wendigo_receipt_actor.cannibalism_wendigo_enemy_death_receipt_tracked_countries
						value = THIS
					}
				}
				cannibalism_process_current_wendigo_enemy_death_delta = yes
			}
			else = {
				event_target:cannibalism_wendigo_receipt_actor = { cannibalism_initialize_wendigo_enemy_receipt_epoch_for_target = yes }
			}
		}
	}
}

cannibalism_muster_wendigo_pack_from_enemy_death_receipt_effect = {
	set_temp_variable = { cannibalism_wendigo_receipt_muster_result = 0 }
	set_temp_variable = { cannibalism_wendigo_requested_pack_batch = constant:cannibalism_wendigo_enemy_death_receipt.muster_pack_batch }
	if = {
		limit = {
			has_event_target = cannibalism_wendigo_decision_state
			cannibalism_wendigo_can_pay_enemy_death_receipt_muster = yes
			cannibalism_wendigo_requested_pack_batch_fits_capacity = yes
			event_target:cannibalism_wendigo_decision_state = { cannibalism_state_can_muster_wendigo_pack_from_enemy_death_receipt = yes }
		}
		save_event_target_as = cannibalism_wendigo_decision_actor
		event_target:cannibalism_wendigo_decision_state = { save_event_target_as = cannibalism_consumption_target_state }
		set_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_wendigo_enemy_death_receipt.muster_population_k }
		multiply_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_consumption.population_people_per_k }
		set_temp_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.recruitment }
		cannibalism_prepare_consumption_context = yes
		if = {
			limit = {
				check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied }
				check_variable = {
					var = cannibalism_population_loss_applied
					value = cannibalism_consumption_exact_request_people
					compare = equals
				}
			}
			subtract_from_variable = { cannibalism_wendigo_enemy_death_receipts = constant:cannibalism_wendigo_enemy_death_receipt.muster_receipt_cost }
			set_temp_variable = { cannibalism_focus_closure_larder_cost = constant:cannibalism_wendigo_enemy_death_receipt.muster_larder }

			cannibalism_pay_current_infantry_equipment_cost = yes
			set_temp_variable = { cannibalism_current_cost = constant:cannibalism_wendigo_enemy_death_receipt.muster_support }
			cannibalism_pay_current_support_equipment_cost = yes
			set_temp_variable = { cannibalism_wendigo_pack_manpower = cannibalism_population_loss_applied }
			multiply_temp_variable = { cannibalism_wendigo_pack_manpower = constant:cannibalism_wendigo_enemy_death_receipt.muster_manpower_factor }
			round_temp_variable = cannibalism_wendigo_pack_manpower

	set_country_flag = cannibalism_wendigo_captain_stage_bound

			}

			NOT = { has_trait = cannibalism_wendigo_bound_captain }

	}

cannibalism_wendigo_refresh_inherited_commander_stage = {
	if = {
		limit = {
			has_country_flag = cannibalism_wendigo_captain_stage_winter_hunt
			check_variable = {
				var = cannibalism_wendigo_winter_victories
				value = constant:cannibalism_wendigo_countdown.minimum_winter_victories
				compare = greater_than_or_equals
			}
		}
		every_army_leader = {
			limit = {
				OR = {
					has_trait = cannibalism_host_commander
					has_trait = cannibalism_bound_servant
				}
				NOT = { is_character = ZZZ_hannibal_wendigo }
			}
			if = { limit = { has_trait = cannibalism_wendigo_bound_captain } remove_unit_leader_trait = cannibalism_wendigo_bound_captain }
			if = { limit = { NOT = { has_trait = cannibalism_wendigo_winter_hunt_captain } } add_unit_leader_trait = cannibalism_wendigo_winter_hunt_captain }
		}
	}
}

cannibalism_activate_inherited_winter_cell_effect = {
	set_temp_variable = { cannibalism_wendigo_inherited_cell_result = 0 }
	if = {
		limit = {
			has_event_target = cannibalism_wendigo_decision_target
			cannibalism_wendigo_can_pay_inherited_cell_cost = yes
			event_target:cannibalism_wendigo_decision_target = { cannibalism_wendigo_is_valid_inherited_cell_target = yes }
		}
		set_temp_variable = { cannibalism_focus_closure_larder_cost = constant:cannibalism_wendigo_inherited_cell.larder }
		cannibalism_pay_focus_closure_larder_cost = yes
		set_temp_variable = { cannibalism_current_cost = constant:cannibalism_wendigo_inherited_cell.command }
		cannibalism_pay_current_command_cost = yes
		set_temp_variable = { cannibalism_current_cost = constant:cannibalism_wendigo_inherited_cell.support }
		cannibalism_pay_current_support_equipment_cost = yes
		set_temp_variable = { cannibalism_wendigo_inherited_cell_duration_days = constant:cannibalism_wendigo_inherited_cell.duration_days }
		set_temp_variable = { cannibalism_wendigo_inherited_cell_target_lock_days = constant:cannibalism_wendigo_inherited_cell.target_lock_days }
		event_target:cannibalism_wendigo_decision_target = {
			add_dynamic_modifier = { modifier = cannibalism_wendigo_inherited_cell_pressure days = cannibalism_wendigo_inherited_cell_duration_days }
			set_country_flag = { flag = cannibalism_wendigo_inherited_cell_target days = cannibalism_wendigo_inherited_cell_duration_days }
			set_country_flag = { flag = cannibalism_wendigo_inherited_cell_target_lock days = cannibalism_wendigo_inherited_cell_target_lock_days }
			if = {
				limit = { NOT = { is_in_array = { array = ROOT.cannibalism_wendigo_active_cell_targets value = THIS } } }
				ROOT = { add_to_array = { cannibalism_wendigo_active_cell_targets = PREV } }
			}
		}
		if = {
			limit = {
				has_country_flag = cannibalism_wendigo_terminal_hunt_active
				NOT = { has_country_flag = cannibalism_wendigo_inherited_cell_hunt_pressure_applied }
				has_event_target = cannibalism_wendigo_terminal_hunt_target
				event_target:cannibalism_wendigo_decision_target = { tag = event_target:cannibalism_wendigo_terminal_hunt_target }
			}
			add_to_variable = { cannibalism_wendigo_terminal_hunt_pressure = constant:cannibalism_wendigo_inherited_cell.hunt_pressure }
			clamp_variable = {
				var = cannibalism_wendigo_terminal_hunt_pressure
				min = 0
				max = constant:cannibalism_wendigo_terminal_hunt.pressure_goal
			}
			set_country_flag = cannibalism_wendigo_inherited_cell_hunt_pressure_applied
		}
		set_temp_variable = { cannibalism_wendigo_inherited_cell_result = 1 }
	}
}

	for_each_scope_loop = {
		array = cannibalism_wendigo_active_cell_targets
		if = {
			limit = { exists = yes }

			clr_country_flag = cannibalism_wendigo_inherited_cell_target_lock
		}
	}
	resize_array = { cannibalism_wendigo_active_cell_targets = 0 }
}
