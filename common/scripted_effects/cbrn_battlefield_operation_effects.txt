# ============================================================================
# CHAOS REDUX - NATIVE CHEMICAL LAND RAID RECEIPT HELPERS
# ============================================================================
# Overview:
# - Classifies the exact static agent and route passed by native land raids.
# - Supplies bounded weather, terrain, command, and release conditions to
#   the shared chemical action resolver.
# - Does not select an agent, debit equipment, schedule an operation, or
#   grant a second outcome after the native raid reservation.
# ============================================================================

cbrn_battlefield_set_action_agent_class_from_action = {
	set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.none }
	if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.chlorine compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.phosgene compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.choking }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.mustard compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.lewisite compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.blister }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.tabun compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.sarin compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.soman compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.nerve }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.malodor compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.behavioral compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.incapacitating }
	}
}

cbrn_battlefield_set_severity_from_action_route = {
	set_temp_variable = { cbrn_action_severity = constant:cbrn_operation_severity.none }
	if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.cylinder_release compare = equals } }
		set_temp_variable = { cbrn_action_severity = constant:cbrn_battlefield_operation_severity.cylinder_release }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.projector_barrage compare = equals } }
		set_temp_variable = { cbrn_action_severity = constant:cbrn_battlefield_operation_severity.projector_barrage }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals } }
		set_temp_variable = { cbrn_action_severity = constant:cbrn_battlefield_operation_severity.artillery_fire_plan }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.armored_delivery compare = equals } }
		set_temp_variable = { cbrn_action_severity = constant:cbrn_battlefield_operation_severity.armored_delivery }
	}
}
