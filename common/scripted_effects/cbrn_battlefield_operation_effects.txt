# ============================================================================
# CHAOS REDUX - EXACT-STATE CBRN BATTLEFIELD OPERATION EFFECTS
# ============================================================================
# Overview:
# - Implements the four non-air chemical delivery routes exposed by the
#   state-targeted timed decisions: cylinder release, projector barrage,
#   artillery fire plan, and armored delivery.
# - Every committed route saves one exact state, debits one shared chemical
#   payload lot and route equipment, records a bounded shortage receipt, and
#   later resolves through cbrn_prepare_chemical_action_record followed by
#   cbrn_dispatch_chemical_action_record.
# - The shared dispatcher owns protection, disruption, deaths, contamination,
#   medical saturation, evidence, attribution, Condemnation, Air Cleanliness,
#   history, and cleanup. This file only supplies the route receipt and its
#   commitment ledger.
# - No effect in this file estimates weather, searches for a target, scans all
#   countries, treats idle aircraft as activity, or refunds a committed payload.
# ============================================================================

cbrn_initialize_battlefield_operation_selection = {
	if = {
		limit = { NOT = { has_variable = cbrn_battlefield_selected_agent } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.chlorine }
	}
	if = {
		limit = {
			NOT = { has_tech = chlorine_gas }
			OR = { has_tech = phosgene has_tech = mustard_gas has_tech = lewisite has_tech = tabun has_tech = sarin has_tech = soman }
		}
		if = {
			limit = { has_tech = phosgene }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.phosgene }
		}
		else_if = {
			limit = { has_tech = mustard_gas }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.mustard }
		}
		else_if = {
			limit = { has_tech = lewisite }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.lewisite }
		}
		else_if = {
			limit = { has_tech = tabun }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.tabun }
		}
		else_if = {
			limit = { has_tech = sarin }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.sarin }
		}
		else_if = {
			limit = { has_tech = soman }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.soman }
		}
	}
	if = {
		limit = { NOT = { has_variable = cbrn_battlefield_last_result } }
		set_variable = { cbrn_battlefield_last_result = constant:cbrn_action_result.rejected }
	}
}

cbrn_battlefield_ai_select_agent_for_route = {
	if = {
		limit = { is_ai = yes }
		if = {
			limit = {
				check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals }
				has_variable = cbrn_shell_filling_agent
				cbrn_battlefield_shell_profile_agent_is_unlocked = yes
			}
			set_variable = { cbrn_battlefield_selected_agent = cbrn_shell_filling_agent }
		}
		else_if = {
			limit = { OR = { has_tech = soman is_special_project_completed = sp:sp_cw_soman_program } }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.soman }
		}
		else_if = {
			limit = { OR = { has_tech = sarin is_special_project_completed = sp:sp_cw_sarin_program } }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.sarin }
		}
		else_if = {
			limit = { has_tech = tabun }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.tabun }
		}
		else_if = {
			limit = { has_tech = lewisite }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.lewisite }
		}
		else_if = {
			limit = { has_tech = mustard_gas }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.mustard }
		}
		else_if = {
			limit = { has_tech = phosgene }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.phosgene }
		}
		else_if = {
			limit = { has_tech = chlorine_gas }
			set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.chlorine }
		}
	}
}

cbrn_battlefield_cycle_selected_agent = {
	if = {
		limit = { NOT = { has_variable = cbrn_battlefield_selected_agent } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.chlorine }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.chlorine compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.phosgene }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.phosgene compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.mustard }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.mustard compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.lewisite }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.lewisite compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.tabun }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.tabun compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.sarin }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.sarin compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.soman }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.soman compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.malodor }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_battlefield_selected_agent value = constant:cbrn_agent.malodor compare = equals } }
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.behavioral }
	}
	else = {
		set_variable = { cbrn_battlefield_selected_agent = constant:cbrn_agent.chlorine }
	}
}

cbrn_battlefield_set_action_agent_class_from_action = {
	set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.none }
	if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.chlorine compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.phosgene compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.choking }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.mustard compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.lewisite compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.blister }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.tabun compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.sarin compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.soman compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.nerve }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.malodor compare = equals }
				check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.behavioral compare = equals }
			}
		}
		set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.incapacitating }
	}
}

		limit = { check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals } }
		cbrn_battlefield_debit_ammunition = yes
		set_temp_variable = { cbrn_battlefield_ammunition_remaining = cbrn_battlefield_equipment_remaining }
	}

	if = {
		limit = { check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.projector_barrage compare = equals } }
		cbrn_battlefield_debit_projector = yes
		set_temp_variable = { cbrn_battlefield_projector_remaining = cbrn_battlefield_equipment_remaining }
	}
	if = {
		limit = { check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.armored_delivery compare = equals } }
		cbrn_battlefield_debit_tanks = yes
		set_temp_variable = { cbrn_battlefield_tank_remaining = cbrn_battlefield_equipment_remaining }
		cbrn_battlefield_debit_motorized = yes
		set_temp_variable = { cbrn_battlefield_motorized_remaining = cbrn_battlefield_equipment_remaining }
		set_temp_variable = { cbrn_battlefield_fuel_debit = cbrn_battlefield_fuel_requested }
		multiply_temp_variable = { cbrn_battlefield_fuel_debit = constant:cbrn_payload_math.negation_multiplier }
		add_fuel = cbrn_battlefield_fuel_debit
	}

	if = {
		limit = {
			check_variable = { var = cbrn_battlefield_masks_remaining value = constant:cbrn_math.zero compare = equals }
			check_variable = { var = cbrn_battlefield_decon_remaining value = constant:cbrn_math.zero compare = equals }
			check_variable = { var = cbrn_battlefield_instruments_remaining value = constant:cbrn_math.zero compare = equals }
			check_variable = { var = cbrn_battlefield_support_remaining value = constant:cbrn_math.zero compare = equals }
			OR = {
				NOT = { check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals } }
				check_variable = { var = cbrn_battlefield_ammunition_remaining value = constant:cbrn_math.zero compare = equals }
			}
			OR = {
				check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.cylinder_release compare = equals }
				check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals }
				check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.projector_barrage compare = equals }
				check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.armored_delivery compare = equals }
			}
			OR = {
				NOT = { check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.projector_barrage compare = equals } }
				check_variable = { var = cbrn_battlefield_projector_remaining value = constant:cbrn_math.zero compare = equals }
			}
			OR = {
				NOT = { check_variable = { var = cbrn_battlefield_requested_route value = constant:cbrn_delivery_route.armored_delivery compare = equals } }
				AND = {
					check_variable = { var = cbrn_battlefield_tank_remaining value = constant:cbrn_math.zero compare = equals }
					check_variable = { var = cbrn_battlefield_motorized_remaining value = constant:cbrn_math.zero compare = equals }
				}
			}
		}
		set_temp_variable = { cbrn_battlefield_equipment_payment_proof = constant:cbrn_proof.supplied }
	}
}

cbrn_battlefield_set_cooldown = {
	set_temp_variable = { cbrn_battlefield_cooldown_days = constant:cbrn_battlefield_operation_timing.cylinder_cooldown }

cbrn_battlefield_commit_operation = {
	set_country_flag = cbrn_battlefield_operation_in_progress
	save_event_target_as = cbrn_battlefield_actor_target
	set_variable = { cbrn_battlefield_active_route = cbrn_battlefield_requested_route }
	set_variable = { cbrn_battlefield_active_agent = cbrn_action_agent }
	set_variable = { cbrn_battlefield_active_payload = cbrn_action_payload_consumed }
	set_variable = { cbrn_battlefield_active_equipment_supply_ratio = cbrn_battlefield_equipment_supply_ratio }
	set_variable = { cbrn_battlefield_active_start_date = global.num_days }
	set_variable = { cbrn_battlefield_active_state = event_target:cbrn_battlefield_state_target }
	set_variable = { cbrn_battlefield_active_victim = event_target:cbrn_battlefield_victim_country }

	if = {
		limit = { NOT = { has_variable = cbrn_battlefield_attempt_count } }
		set_variable = { cbrn_battlefield_attempt_count = constant:cbrn_math.zero }
	}
	add_to_variable = { cbrn_battlefield_attempt_count = constant:cbrn_math.one }
	set_variable = { cbrn_battlefield_last_agent = cbrn_action_agent }
	set_variable = { cbrn_battlefield_last_state = event_target:cbrn_battlefield_state_target }
	set_variable = { cbrn_battlefield_last_victim = event_target:cbrn_battlefield_victim_country }

	event_target:cbrn_battlefield_state_target = {
		set_state_flag = cbrn_battlefield_operation_active
		set_variable = { cbrn_battlefield_actor = event_target:cbrn_battlefield_actor_target }
		set_variable = { cbrn_battlefield_victim = event_target:cbrn_battlefield_victim_country }
		set_variable = { cbrn_battlefield_agent = cbrn_action_agent }
		set_variable = { cbrn_battlefield_route = cbrn_battlefield_requested_route }
		set_variable = { cbrn_battlefield_payload_consumed = cbrn_action_payload_consumed }
		set_variable = { cbrn_battlefield_equipment_supply_ratio = cbrn_battlefield_equipment_supply_ratio }
		set_variable = { cbrn_battlefield_start_date = global.num_days }
	}
}

cbrn_battlefield_begin_operation = {
	FROM = {
		save_event_target_as = cbrn_battlefield_state_target
		CONTROLLER = { save_event_target_as = cbrn_battlefield_victim_country }
	}
	cbrn_battlefield_ai_select_agent_for_route = yes
	if = {
		limit = {
			has_event_target = cbrn_battlefield_state_target
			has_event_target = cbrn_battlefield_victim_country
			cbrn_battlefield_requested_operation_is_valid = yes
			cbrn_battlefield_requested_target_is_valid = yes
		}
		event_target:cbrn_battlefield_state_target = { save_event_target_as = cbrn_action_target_state }
		event_target:cbrn_battlefield_victim_country = { save_event_target_as = cbrn_action_victim_country }
		cbrn_reset_action_context = yes
		set_temp_variable = { cbrn_action_target_state_supplied = constant:cbrn_proof.supplied }
		set_temp_variable = { cbrn_action_weapon_class = constant:cbrn_weapon_class.chemical }
		set_temp_variable = { cbrn_action_delivery_route = cbrn_battlefield_requested_route }
		set_temp_variable = { cbrn_action_agent = cbrn_battlefield_selected_agent }
		cbrn_battlefield_set_action_agent_class_from_action = yes
		cbrn_battlefield_set_severity_from_action_route = yes
		if = {
			limit = {
				check_variable = { var = cbrn_action_agent_class value = constant:cbrn_agent_class.nerve compare = equals }
				has_tech = nerve_agent_suppression_formation
			}
			set_temp_variable = { cbrn_action_late_agent_route_authorized = constant:cbrn_proof.supplied }
		}
		cbrn_set_default_payload_requirement_for_action = yes
		cbrn_try_debit_action_payload = yes
		if = {
			limit = { check_variable = { var = cbrn_action_payload_consumed_proof value = constant:cbrn_proof.supplied compare = equals } }
			cbrn_battlefield_set_route_equipment_costs = yes
			cbrn_battlefield_pay_route_equipment = yes
			if = {
				limit = { check_variable = { var = cbrn_battlefield_equipment_payment_proof value = constant:cbrn_proof.supplied compare = equals } }
				cbrn_battlefield_set_route_commit_costs = yes
				cbrn_battlefield_commit_readiness_and_command_power = yes
				cbrn_battlefield_set_cooldown = yes
				cbrn_battlefield_commit_operation = yes
			}
		}
	}
}

cbrn_battlefield_clear_operation_state = {
	if = {
		limit = { has_variable = cbrn_battlefield_active_state }
		var:cbrn_battlefield_active_state = {
			clr_state_flag = cbrn_battlefield_operation_active
			clear_variable = cbrn_battlefield_actor
			clear_variable = cbrn_battlefield_victim
			clear_variable = cbrn_battlefield_agent
			clear_variable = cbrn_battlefield_route
			clear_variable = cbrn_battlefield_payload_consumed
			clear_variable = cbrn_battlefield_equipment_supply_ratio
			clear_variable = cbrn_battlefield_start_date
		}
	}
	else_if = {
		limit = { has_event_target = cbrn_battlefield_state_target }
		event_target:cbrn_battlefield_state_target = {
			clr_state_flag = cbrn_battlefield_operation_active
			clear_variable = cbrn_battlefield_actor
			clear_variable = cbrn_battlefield_victim
			clear_variable = cbrn_battlefield_agent
			clear_variable = cbrn_battlefield_route
			clear_variable = cbrn_battlefield_payload_consumed
			clear_variable = cbrn_battlefield_equipment_supply_ratio
			clear_variable = cbrn_battlefield_start_date
		}
	}
	clr_country_flag = cbrn_battlefield_operation_in_progress
	clear_variable = cbrn_battlefield_active_route
	clear_variable = cbrn_battlefield_active_agent
	clear_variable = cbrn_battlefield_active_payload
	clear_variable = cbrn_battlefield_active_equipment_supply_ratio
	clear_variable = cbrn_battlefield_active_start_date
	clear_variable = cbrn_battlefield_active_state
	clear_variable = cbrn_battlefield_active_victim
}

}

cbrn_battlefield_resolve_operation = {
	cbrn_reset_action_context = yes
	if = {
		limit = {
			has_variable = cbrn_battlefield_active_state
			has_variable = cbrn_battlefield_active_route
			has_variable = cbrn_battlefield_active_agent
			has_variable = cbrn_battlefield_active_payload
			cbrn_battlefield_current_version_condition_hook_verified = yes
		}
		var:cbrn_battlefield_active_state = {
			save_event_target_as = cbrn_action_target_state
			CONTROLLER = { save_event_target_as = cbrn_action_victim_country }
		}
		set_temp_variable = { cbrn_action_target_state_supplied = constant:cbrn_proof.supplied }
		set_temp_variable = { cbrn_action_weapon_class = constant:cbrn_weapon_class.chemical }
		set_temp_variable = { cbrn_action_delivery_route = cbrn_battlefield_active_route }
		set_temp_variable = { cbrn_action_agent = cbrn_battlefield_active_agent }
		set_temp_variable = { cbrn_action_payload_consumed = cbrn_battlefield_active_payload }
		set_temp_variable = { cbrn_action_payload_consumed_proof = constant:cbrn_proof.supplied }
		cbrn_battlefield_set_action_agent_class_from_action = yes
		cbrn_battlefield_set_severity_from_action_route = yes
		if = {
			limit = {
				check_variable = { var = cbrn_action_agent_class value = constant:cbrn_agent_class.nerve compare = equals }
				has_tech = nerve_agent_suppression_formation
			}
			set_temp_variable = { cbrn_action_late_agent_route_authorized = constant:cbrn_proof.supplied }
		}
		cbrn_resolve_action_target_protection = yes
		cbrn_battlefield_set_action_conditions_receipt = yes
		# The public prepare wrapper overwrites this with the attacker's actual
		# doctrine multiplier after validation; supply the positive validation
		# value here so battlefield routes do not fail before that overwrite.
		cbrn_set_doctrine_condemnation_mult_from_country = yes
		cbrn_prepare_chemical_action_record = yes
		if = {
			limit = { check_variable = { var = cbrn_action_result value = constant:cbrn_action_result.accepted compare = equals } }
			cbrn_dispatch_chemical_action_record = yes
			set_variable = { cbrn_battlefield_last_result = cbrn_action_result }
			set_variable = { cbrn_battlefield_last_disruption = cbrn_action_disruption }
			set_variable = { cbrn_battlefield_last_civilian_death_fraction = cbrn_action_civilian_death_fraction }
			set_variable = { cbrn_battlefield_last_contamination = cbrn_action_contamination_points }
			set_variable = { cbrn_battlefield_last_medical_saturation = cbrn_action_medical_saturation_points }
			set_variable = { cbrn_battlefield_last_evidence = cbrn_action_evidence_points }
		}
		else = {
			set_variable = { cbrn_battlefield_last_result = constant:cbrn_action_result.rejected }
			set_variable = { cbrn_battlefield_last_reject_reason = cbrn_action_reject_reason }
		}
	}
	else = {
		set_variable = { cbrn_battlefield_last_result = constant:cbrn_action_result.rejected }
	}
	cbrn_battlefield_clear_operation_state = yes
}
