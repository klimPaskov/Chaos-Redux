# ============================================================================
# CHAOS REDUX - EVENT 027 AI CHOICE SCORING
# ============================================================================
# Overview:
# - Scores every valid domain, Grand Doctrine, track, and registered
#   subdoctrine using live force, production, war, geography, theater,
#   strategy, continuity, completion, and owner-system signals. Domain scores
#   read country-wide signals; Grand Doctrine, track, and subdoctrine scores
#   read only the terms that differ between the candidates of one pool (force
#   fit, war fit, existing investment, completion, and continuity), and track
#   and subdoctrine scores are squared so a clearly better branch wins most
#   choices.
# - Uses bounded random selection from positive scored candidates, while
#   invalid or unsupported adapter rows remain at zero weight.
# - The chooser is called again after every successful choice, so adoption
#   immediately changes the next mastery pool and score pass.
# - Strategy factors use the native observable AI force/template, doctrine,
#   naval, air, and CBRN readiness signals exposed by the current ruleset.
# - Grand Doctrine and subdoctrine scores also mirror the vanilla doctrine
#   ai_will_do preferences (minor, major, and original-tag factors) through
#   the doctrine_research_ai_native_* constants, and each naval, air, and
#   Special Forces subdoctrine reads only its own ship, plane, or battalion.
# ============================================================================

# Read-only native progress check for the selected domain. Four ordinary
# tracks, two Special Forces tracks, or four owner-defined Chaos tracks are
# inspected without selecting a doctrine or granting mastery.
doctrine_research_ai_check_domain_completion = {
	set_temp_variable = { doctrine_research_ai_domain_completion_found = 0 }
	set_temp_variable = { doctrine_research_ai_completion_break = 0 }
	set_temp_variable = { doctrine_research_ai_completion_first_track = constant:doctrine_research_track.infantry }
	set_temp_variable = { doctrine_research_ai_completion_end_track = constant:doctrine_research_track.operations }
	if = {
		limit = { check_variable = { doctrine_research_selected_domain = constant:doctrine_research_domain.navy } }
		set_temp_variable = { doctrine_research_ai_completion_first_track = constant:doctrine_research_track.submarines }
		set_temp_variable = { doctrine_research_ai_completion_end_track = constant:doctrine_research_track.capital_ships }
	}
	else_if = {
		limit = { check_variable = { doctrine_research_selected_domain = constant:doctrine_research_domain.air } }
		set_temp_variable = { doctrine_research_ai_completion_first_track = constant:doctrine_research_track.fighter_aircraft }
		set_temp_variable = { doctrine_research_ai_completion_end_track = constant:doctrine_research_track.heavy_aircraft }
	}
	else_if = {
		limit = { check_variable = { doctrine_research_selected_domain = constant:doctrine_research_domain.special_forces } }
		set_temp_variable = { doctrine_research_ai_completion_first_track = constant:doctrine_research_track.special_forces_first }
		set_temp_variable = { doctrine_research_ai_completion_end_track = constant:doctrine_research_track.special_forces_second }
	}
	add_to_temp_variable = { doctrine_research_ai_completion_end_track = 1 }
	doctrine_research_capture_selected_domain_grand_doctrine = yes
	for_loop_effect = {
		start = doctrine_research_ai_completion_first_track
		end = doctrine_research_ai_completion_end_track
		value = doctrine_research_ai_completion_track
		break = doctrine_research_ai_completion_break
		set_variable = { doctrine_research_selected_track = doctrine_research_ai_completion_track }
		if = {
			limit = { doctrine_research_selected_grand_doctrine_is_active = yes doctrine_research_selected_track_has_valid_action = yes }
			doctrine_research_prepare_track_display = yes
			if = {
				limit = {
					check_variable = { doctrine_research_display_current_level > 0 }
					check_variable = { doctrine_research_display_completion = 0 }
					check_variable = { var = doctrine_research_display_next_level value = doctrine_research_display_maximum_level compare = greater_than_or_equals }
				}
				set_temp_variable = { doctrine_research_ai_domain_completion_found = 1 }
				set_temp_variable = { doctrine_research_ai_completion_break = 1 }
			}
		}
	}
}

# Country scope. Scores the selected Grand Doctrine from campaign evidence and
# the native vanilla ai_will_do preferences. Only terms that differ between the
# Grand Doctrines of one domain are read; country-wide terms such as divisions,
# factories, war, coast, faction, and domain readiness are equal for every
# candidate in the pool and belong to domain scoring instead.
# - Each Grand Doctrine adds the forces weight for the formations it serves.
#   Mobile Warfare adds it once for motorised or mechanised forces and again for
#   tanks, so a tank army prefers it over Superior Firepower. Superior
#   Firepower reads line or support artillery, anti-tank, or anti-air
#   (doctrine_research_ai_fields_fire_support), the family its bonuses serve.
# - Grand Battleplan starts from its native base, and Convoy Raiding and
#   Battlefield Support take the native factor for a country that is not a
#   major. The native zero for Mobile Warfare and Superior Firepower on a minor
#   is applied by the chooser, because it depends on whether another Army Grand
#   Doctrine is valid.
# - Mass Assault reads an infantry force without armour or motorised
#   templates, or militia and irregulars, so a mobile army is not counted as an
#   infantry-mass army.

# Country scope. Scores the selected track inside its domain's track pool.
# Only terms that differ between the tracks of one pool are read; country-wide
# terms such as divisions, factories, war, coast, faction, domain strategy, and
# domain readiness are equal for every track and belong to domain scoring.
# - Force fit (doctrine_research_ai_selected_track_has_force_fit): the country
#   fields the formations the track serves, or is at war for Army Operations.
# - Existing investment: the investment weight for every level the track's
#   active subdoctrine has already earned, so a developed branch is finished
#   before the AI drifts to an unrelated one.
# - Empty-track value: an empty track whose formations the country fields
#   would fill a missing capability.
# - Completion value: the track is one step from completing its branch. It
#   applies only together with force fit, so a near-complete branch the country
#   has no use for keeps only its investment value.
# - Continuity: the previous choice of this batch advanced the same track.
# The sum is squared before it enters the random_list. Every valid track keeps
# a positive weight, while a clearly better track wins most choices and close
# tracks stay close, which gives the bounded randomness of spec part 3.
doctrine_research_ai_score_selected_track = {
	doctrine_research_prepare_track_display = yes
	set_temp_variable = { doctrine_research_ai_candidate_score = constant:doctrine_research_ai_weight.base }
	set_temp_variable = { doctrine_research_ai_track_has_fit = 0 }
	if = {
		limit = { doctrine_research_ai_selected_track_has_force_fit = yes }
		set_temp_variable = { doctrine_research_ai_track_has_fit = 1 }
		add_to_temp_variable = { doctrine_research_ai_candidate_score = constant:doctrine_research_ai_weight.track_fit }
	}
	set_temp_variable = { doctrine_research_ai_track_investment = doctrine_research_display_current_level }
	multiply_temp_variable = { doctrine_research_ai_track_investment = constant:doctrine_research_ai_weight.investment }

# Country scope, called by doctrine_research_ai_score_selected_subdoctrine.
# Adds the selected subdoctrine's own fit to doctrine_research_ai_candidate_score,
# which already holds the base, investment, completion, and continuity terms,
# then squares the sum and applies the native preference. Called once per
# valid candidate with doctrine_research_selected_subdoctrine set to that
# candidate; it never changes which candidates are valid.
# - Naval, air, and Special Forces subdoctrines add the forces weight only for
#   their own ship, plane, or battalion type. Marines add the geography weight
#   for a coast and Mountaineers for mountains; no other Special Forces
#   subdoctrine reads geography.
# - The linear sum is squared, as in track scoring, so a fitting or invested
#   candidate clearly leads while equal candidates stay equal.
# - The vanilla ai_will_do factors (base, original tag, minor, major) are
#   mirrored from the doctrine_research_ai_native_* constants and multiply the
#   square. A candidate that vanilla would never pick keeps the base weight, so
#   every valid choice stays possible.
# - Army subdoctrines have no own fit term: their track's force fit is scored
#   when the track is chosen. Chaos Warfare tracks hold one subdoctrine each
#   and need no preference.
doctrine_research_ai_score_selected_subdoctrine_fit = {
	set_temp_variable = { doctrine_research_ai_native_factor = constant:doctrine_research_ai_native_base.neutral }
	if = {
		limit = { check_variable = { doctrine_research_selected_subdoctrine = constant:doctrine_research_subdoctrine.mobile_infantry } }
		if = { limit = { original_tag = GER } multiply_temp_variable = { doctrine_research_ai_native_factor = constant:doctrine_research_ai_native_country.mobile_infantry } }
		if = { limit = { is_major = no } multiply_temp_variable = { doctrine_research_ai_native_factor = constant:doctrine_research_ai_native_minor.mobile_infantry } }

# Country scope. Scores the selected subdoctrine inside its track's candidate
# pool. The caller sets doctrine_research_ai_candidate_score to the base weight
# before the call and reads it afterwards; doctrine_research_selected_domain,
# _grand_doctrine, _track, and _subdoctrine name the candidate. Only terms that
# differ between the candidates of one pool are read. Country-wide terms such
# as factories, war, coast, faction, and owner readiness, and the force fit
# that every subdoctrine of one track shares, belong to domain and track
# scoring.
# - Existing investment: the investment weight for every mastery level the
#   subdoctrine has already earned, including mastery banked while it was off
#   its track, so a partly mastered branch is resumed first.
# - Completion value: the subdoctrine is one level below its maximum, so one
#   choice completes the branch.
# - Continuity: the previous choice of this batch advanced the same
#   subdoctrine on the same track.
# - Own fit and native preference: doctrine_research_ai_score_selected_subdoctrine_fit
#   adds the subdoctrine's own ship, plane, or battalion term, squares the sum,
#   and applies the vanilla ai_will_do factors.
doctrine_research_ai_score_selected_subdoctrine = {
	set_temp_variable = { doctrine_research_transaction_observed_level = 0 }
	set_temp_variable = { doctrine_research_transaction_pre_level = 0 }
	doctrine_research_capture_transaction_level = yes
	set_temp_variable = { doctrine_research_ai_subdoc_investment = doctrine_research_transaction_observed_level }
	multiply_temp_variable = { doctrine_research_ai_subdoc_investment = constant:doctrine_research_ai_weight.investment }
	add_to_temp_variable = { doctrine_research_ai_candidate_score = doctrine_research_ai_subdoc_investment }
	set_temp_variable = { doctrine_research_ai_subdoc_maximum_level = constant:doctrine_research_event.max_mastery_level }
	if = {
		limit = { check_variable = { var = doctrine_research_selected_subdoctrine value = constant:doctrine_research_subdoctrine.peoples_war compare = equals } }
		set_temp_variable = { doctrine_research_ai_subdoc_maximum_level = constant:doctrine_research_event.peoples_war_max_mastery_level }
	}
	set_temp_variable = { doctrine_research_ai_subdoc_next_level = doctrine_research_transaction_observed_level }
	add_to_temp_variable = { doctrine_research_ai_subdoc_next_level = 1 }
	if = {
		limit = {
			check_variable = { var = doctrine_research_transaction_observed_level value = doctrine_research_ai_subdoc_maximum_level compare = less_than }
			check_variable = { var = doctrine_research_ai_subdoc_next_level value = doctrine_research_ai_subdoc_maximum_level compare = greater_than_or_equals }
		}
		add_to_temp_variable = { doctrine_research_ai_candidate_score = constant:doctrine_research_ai_weight.completion }
	}
	if = {
		limit = {
			check_variable = { var = doctrine_research_last_transaction_batch_id value = doctrine_research_active_batch_id compare = equals }
			check_variable = { var = doctrine_research_last_transaction_domain value = doctrine_research_selected_domain compare = equals }
			check_variable = { var = doctrine_research_last_transaction_grand_doctrine value = doctrine_research_selected_grand_doctrine compare = equals }
			check_variable = { var = doctrine_research_last_transaction_track value = doctrine_research_selected_track compare = equals }
			check_variable = { var = doctrine_research_last_transaction_subdoctrine value = doctrine_research_selected_subdoctrine compare = equals }
		}
		add_to_temp_variable = { doctrine_research_ai_candidate_score = constant:doctrine_research_ai_weight.continuity }
	}
	doctrine_research_ai_score_selected_subdoctrine_fit = yes
}
