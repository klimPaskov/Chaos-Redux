# ============================================================================
# CHAOS REDUX - EVENT 014 CXT TEST REGISTRATION
# ============================================================================
# Overview:
# - Registers one package-owned setup effect with the CXT dynamic harness.
# - That setup effect registers all Event 014 irregular infantry.
# - Registers their existing vanilla equipment dependencies for replenishment.
# - Is idempotent so startup and tag-scoped daily hooks can share one effect.
# - Builds one valid mixed infantry muster and one mobile rider formation.
# ============================================================================

# Caller-owned outputs, set by each caller before the call when it reads them
# afterwards: chaosx_test_country_extension_registration_added.
cannibalism_register_cxt_test_content = {
	set_temp_variable = {
		var = chaosx_test_country_registration_extension_effect
		value = token:chaosx_cxt_extension_event014_cannibalism
	}
	chaosx_test_country_register_extension_effect = yes
}

chaosx_cxt_extension_event014_cannibalism_apply = {
	if = {
		limit = { NOT = { has_global_flag = cannibalism_cxt_irregular_units_registered } }

		set_temp_variable = { var = chaosx_test_country_current_unit_family value = token:chaosx_cxt_extension_event014_cannibalism }
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_scavenger_warband }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_feast_guard }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_feast_cohort }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_bone_guard }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_bone_riders }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_island_reavers }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_siege_eaters }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_march_predation_column }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:cannibal_network_cadre }
		chaosx_test_country_register_frontline_subunit = yes

		# Event 014 uses these existing vanilla equipment definitions and does not
		# add another equipment archetype.
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:infantry_equipment_3 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:support_equipment_1 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:artillery_equipment_3 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:motorized_equipment_1 }
		chaosx_test_country_register_equipment = yes

		set_global_flag = cannibalism_cxt_irregular_units_registered
	}
	if = {
		limit = { NOT = { has_country_flag = chaosx_cxt_cannibal_formations_ready } }
		chaosx_cxt_event014_define_formations = yes
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_scavenger_warband } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_scavenger_warband } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_feast_guard } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_feast_guard } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_feast_cohort } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_feast_cohort } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_bone_guard } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_bone_guard } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_bone_riders } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_bone_riders } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_island_reavers } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_island_reavers } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_siege_eaters } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_siege_eaters } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_march_predation_column } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_march_predation_column } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_network_cadre } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:cannibal_network_cadre } }
		chaosx_cxt_event014_spawn_formations_initial = yes
		if = {
			limit = {
				AND = {
					is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_scavenger_warband }
					is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_bone_riders }
				}
			}
			set_country_flag = chaosx_cxt_cannibal_formations_ready
		}
	}
}

chaosx_cxt_event014_define_formations = {
	unlock_subunit = cannibal_scavenger_warband
	unlock_subunit = cannibal_feast_guard
	unlock_subunit = cannibal_feast_cohort
	unlock_subunit = cannibal_bone_guard
	unlock_subunit = cannibal_bone_riders

	unlock_subunit = cannibal_siege_eaters

	unlock_subunit = cannibal_network_cadre
	if = {
		limit = { NOT = { has_template = "CXT - Cannibal Irregular Muster" } }
		division_template = {
			name = "CXT - Cannibal Irregular Muster"
			is_locked = no
			force_allow_recruiting = yes

				cannibal_scavenger_warband = { x = 0 y = 0 }

				cannibal_feast_cohort = { x = 0 y = 2 }

				cannibal_siege_eaters = { x = 1 y = 1 }
				cannibal_march_predation_column = { x = 1 y = 2 }

			}
		}
	}
}

chaosx_cxt_event014_spawn_formations = {
	if = {
		limit = { has_template = "CXT - Cannibal Irregular Muster" }
		set_temp_variable = { var = chaosx_test_country_current_support_subunit value = 0 }
		set_temp_variable = { var = chaosx_test_country_current_family value = token:chaosx_cxt_extension_event014_cannibalism }
		set_temp_variable = { var = chaosx_test_country_family_spawn_succeeded value = 0 }
		set_variable = { global.chaosx_test_country_spawn_template = token:cxt_template_cannibal_irregular_muster }
		chaosx_test_country_spawn_family_template = yes
		if = {
			limit = { check_variable = { chaosx_test_country_family_spawn_succeeded > 0 } }
			if = { limit = { NOT = { is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_scavenger_warband } } } add_to_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_scavenger_warband } }
		}
	}
	if = {
		limit = { has_template = "CXT - Cannibal Bone Riders" }
		set_temp_variable = { var = chaosx_test_country_current_support_subunit value = 0 }
		set_temp_variable = { var = chaosx_test_country_current_family value = token:chaosx_cxt_extension_event014_cannibalism }
		set_temp_variable = { var = chaosx_test_country_family_spawn_succeeded value = 0 }
		set_variable = { global.chaosx_test_country_spawn_template = token:cxt_template_cannibal_bone_riders }
		chaosx_test_country_spawn_family_template = yes
		if = {
			limit = { check_variable = { chaosx_test_country_family_spawn_succeeded > 0 } }
			if = { limit = { NOT = { is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_bone_riders } } } add_to_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_bone_riders } }
		}
	}
}

		limit = { has_template = "CXT - Cannibal Irregular Muster" NOT = { is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_scavenger_warband } } }
		set_temp_variable = { var = chaosx_test_country_current_support_subunit value = 0 }
		set_temp_variable = { var = chaosx_test_country_current_family value = token:chaosx_cxt_extension_event014_cannibalism }
		set_temp_variable = { var = chaosx_test_country_family_spawn_succeeded value = 0 }
		set_variable = { global.chaosx_test_country_spawn_template = token:cxt_template_cannibal_irregular_muster }
		chaosx_test_country_spawn_family_template = yes
		if = {
			limit = { check_variable = { chaosx_test_country_family_spawn_succeeded > 0 } }
			add_to_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_scavenger_warband }
		}
	}
	if = {
		limit = { has_template = "CXT - Cannibal Bone Riders" NOT = { is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:cannibal_bone_riders } } }
		set_temp_variable = { var = chaosx_test_country_current_support_subunit value = 0 }
		set_temp_variable = { var = chaosx_test_country_current_family value = token:chaosx_cxt_extension_event014_cannibalism }
		set_temp_variable = { var = chaosx_test_country_family_spawn_succeeded value = 0 }

			limit = { check_variable = { chaosx_test_country_family_spawn_succeeded > 0 } }

		}
	}
}
