camp_rework_soviet_refresh_current_famine_state = {
	if = {
		limit = {
			has_variable = genocide_responsible_country
			var:genocide_responsible_country = { tag = ROOT }
			is_valid_camp_active_site_state = yes
			OR = {
				has_state_flag = genocide_soviet_famine_pressure_site
				has_state_flag = camp_rework_soviet_famine_pressure
				has_state_flag = camp_rework_soviet_industrial_pool
			}
			ROOT = { check_variable = { famine_pressure > 0 } }
		}
		set_state_flag = genocide_soviet_famine_pressure_site
		set_state_flag = camp_rework_soviet_famine_pressure
		set_variable = { sov_famine_state_pressure = ROOT.famine_pressure }
		clamp_variable = { var = sov_famine_state_pressure min = 0 max = constant:camp_rework_threshold.max_value }

		set_variable = { sov_famine_local_manpower_factor = sov_famine_state_pressure }
		multiply_variable = { sov_famine_local_manpower_factor = constant:camp_rework_soviet_state_modifier.local_manpower_per_pressure }
		clamp_variable = { var = sov_famine_local_manpower_factor min = constant:camp_rework_soviet_state_modifier.minimum_factor max = 0 }

		set_variable = { sov_famine_production_factor = sov_famine_state_pressure }
		multiply_variable = { sov_famine_production_factor = constant:camp_rework_soviet_state_modifier.production_per_pressure }
		clamp_variable = { var = sov_famine_production_factor min = constant:camp_rework_soviet_state_modifier.minimum_factor max = 0 }

		set_variable = { sov_famine_resources_factor = sov_famine_state_pressure }
		multiply_variable = { sov_famine_resources_factor = constant:camp_rework_soviet_state_modifier.resources_per_pressure }
		clamp_variable = { var = sov_famine_resources_factor min = constant:camp_rework_soviet_state_modifier.minimum_factor max = 0 }

		set_variable = { sov_famine_resistance_target = sov_famine_state_pressure }
		multiply_variable = { sov_famine_resistance_target = constant:camp_rework_soviet_state_modifier.resistance_per_pressure }
		clamp_variable = { var = sov_famine_resistance_target min = 0 max = constant:camp_rework_soviet_state_modifier.maximum_resistance }

		set_variable = { sov_famine_local_supplies_factor = sov_famine_state_pressure }
		multiply_variable = { sov_famine_local_supplies_factor = constant:camp_rework_soviet_state_modifier.local_supplies_per_pressure }
		clamp_variable = { var = sov_famine_local_supplies_factor min = constant:camp_rework_soviet_state_modifier.minimum_factor max = 0 }

		if = {
			limit = {
				ROOT = {
					check_variable = {
						var = famine_pressure
						value = constant:camp_rework_soviet.famine_pressure_critical
						compare = greater_than_or_equals
					}
				}
			}
			set_state_flag = camp_rework_soviet_famine_crisis
		}
		else = { clr_state_flag = camp_rework_soviet_famine_crisis }

		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = sov_famine_pressure_state } } }
			add_dynamic_modifier = { modifier = sov_famine_pressure_state }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		clr_state_flag = genocide_soviet_famine_pressure_site
		clr_state_flag = camp_rework_soviet_famine_pressure
		clr_state_flag = camp_rework_soviet_famine_crisis
		clear_variable = sov_famine_state_pressure
		clear_variable = sov_famine_local_manpower_factor
		clear_variable = sov_famine_production_factor
		clear_variable = sov_famine_resources_factor
		clear_variable = sov_famine_resistance_target
		clear_variable = sov_famine_local_supplies_factor
		if = {
			limit = { has_dynamic_modifier = { modifier = sov_famine_pressure_state } }
			remove_dynamic_modifier = { modifier = sov_famine_pressure_state }
		}
	}
}

camp_rework_soviet_refresh_famine_states = {
	for_each_scope_loop = {
		array = camp_active_site_states
		limit = {
			camp_rework_state_can_accept_responsibility_from_root = yes
			has_variable = genocide_responsible_country
			var:genocide_responsible_country = { tag = ROOT }
		}
		camp_rework_soviet_refresh_current_famine_state = yes
	}
}

# Delayed threshold events cannot rely on FROM retaining a state scope. Capture
# the first registered famine state deterministically and copy it to the event's
# dedicated country pointer before scheduling the event.
camp_rework_soviet_prepare_famine_event_state = {
	clear_variable = sov_famine_event_state_id
	for_each_scope_loop = {
		array = camp_active_site_states
		limit = {
			has_state_flag = genocide_soviet_famine_pressure_site
			camp_rework_state_is_actor_responsible_controlled = yes
			ROOT = { NOT = { has_variable = sov_famine_event_state_id } }
		}
		ROOT = { set_variable = { sov_famine_event_state_id = PREV.id } }
	}
}

camp_rework_soviet_update_famine = {
	camp_rework_soviet_initialize_country_values = yes
	set_temp_variable = { camp_soviet_monthly_famine_change = constant:camp_rework_soviet.monthly_famine_decay }
	if = {
		limit = { check_variable = { grain_extraction_burden > 0 } }
		add_to_temp_variable = { camp_soviet_monthly_famine_change = constant:camp_rework_soviet.monthly_famine_from_grain }
	}
	if = {
		limit = { check_variable = { forced_labor_quota > 0 } }
		add_to_temp_variable = { camp_soviet_monthly_famine_change = constant:camp_rework_soviet.monthly_famine_from_quota }
	}
	if = {
		limit = { camp_rework_soviet_paranoia_high = yes }
		add_to_temp_variable = { camp_soviet_monthly_famine_change = constant:camp_rework_soviet.monthly_famine_from_paranoia }
	}
	if = {
		limit = { stability < constant:camp_rework_threshold.low_stability }
		add_to_temp_variable = { camp_soviet_monthly_famine_change = constant:camp_rework_soviet.monthly_famine_from_low_stability }
	}
	if = {
		limit = { has_country_flag = sov_famine_relief_active }
		add_to_temp_variable = { camp_soviet_monthly_famine_change = constant:camp_rework_soviet.monthly_famine_decay }
	}
	add_to_variable = { famine_pressure = camp_soviet_monthly_famine_change }
	clamp_variable = { var = famine_pressure min = 0 max = constant:camp_rework_threshold.max_value }

	camp_rework_soviet_refresh_famine_states = yes

	if = {
		limit = {
			camp_rework_soviet_paranoia_medium = yes
			check_variable = { grain_extraction_burden > 0 }
			check_variable = { famine_pressure > 0 }
			NOT = { has_active_mission = sov_famine_pressure_cycle }
		}
		set_country_flag = camp_rework_soviet_famine_pressure_cycle_active
		activate_mission = sov_famine_pressure_cycle
	}
	if = {
		limit = {
			check_variable = {
				var = famine_pressure
				value = constant:camp_rework_threshold.band_medium
				compare = greater_than_or_equals
			}
			NOT = { has_country_flag = sov_famine_warning_fired }
		}
		set_country_flag = sov_famine_warning_fired
		camp_rework_soviet_prepare_famine_event_state = yes
		if = {
			limit = { has_variable = sov_famine_event_state_id }
			set_variable = { sov_famine_warning_state_id = sov_famine_event_state_id }
		}
		clear_variable = sov_famine_event_state_id
		set_temp_variable = { camp_rework_threshold_event_delay_days = constant:camp_rework_timing.threshold_event_delay_days }
		country_event = { id = soviet_gulag.1 days = camp_rework_threshold_event_delay_days }
	}
	if = {
		limit = {
			camp_rework_soviet_paranoia_high = yes
			check_variable = {
				var = famine_pressure
				value = constant:camp_rework_soviet.famine_pressure_critical
				compare = greater_than_or_equals
			}
			NOT = { has_country_flag = sov_famine_critical_fired }
		}
		set_country_flag = sov_famine_crisis_active
		set_country_flag = sov_famine_critical_fired
		camp_rework_soviet_prepare_famine_event_state = yes
		if = {
			limit = { has_variable = sov_famine_event_state_id }
			set_variable = { sov_famine_crisis_state_id = sov_famine_event_state_id }
		}
		clear_variable = sov_famine_event_state_id
		set_temp_variable = { camp_rework_threshold_event_delay_days = constant:camp_rework_timing.threshold_event_delay_days }
		country_event = { id = soviet_gulag.2 days = camp_rework_threshold_event_delay_days }
	}
	if = {
		limit = { check_variable = { famine_pressure = 0 } }
		clr_country_flag = sov_famine_crisis_active
		clr_country_flag = sov_famine_relief_active
	}
	camp_rework_soviet_refresh_values = yes
}

		add_to_variable = { grain_extraction_burden = constant:camp_rework_soviet.grain_relief }
		add_to_variable = { famine_pressure = constant:camp_rework_soviet.grain_relief }
		add_to_variable = { camp_administrator_corruption = constant:camp_rework_soviet.grain_relief }
		add_to_variable = { camp_policy_reform = constant:camp_rework_country_value.reform_major }
		clamp_variable = { var = forced_labor_quota min = 0 max = constant:camp_rework_threshold.max_value }
		clamp_variable = { var = grain_extraction_burden min = 0 max = constant:camp_rework_threshold.max_value }
		clamp_variable = { var = famine_pressure min = 0 max = constant:camp_rework_threshold.max_value }
