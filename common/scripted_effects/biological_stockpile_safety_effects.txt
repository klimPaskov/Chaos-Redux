# ============================================================================
# CHAOS REDUX - BIOLOGICAL STOCKPILE SAFETY SCRIPTED EFFECTS
# ============================================================================
# Inputs: country scope for all public helpers; the caller supplies the exact
#   arsenal state pointer/source, selected lifecycle agent, requested loss,
#   incident kind, and any private runtime sabotage/handling flags.
# Outputs: actual four-agent stockpile values and weights, exact-state
#   condition values, visible risk score/band, bounded loss amounts, incident
#   severity weights/selection, and a cleanup effect.
# Side effects: writes private country/state records, synchronizes the four risk
#   ideas, schedules only the designated country's monitor, applies exact
#   accident payload loss, and sends non-contained accidents into the shared
#   ordinary lifecycle. It never creates a delivery decision, raid, operation,
#   alternative target, deliberate-use record, or zombie state.
# ============================================================================

# Country scope. Captures the live national equipment values exactly. The
# equipment identifiers are the four existing biological payload models.
bio_stockpile_safety_capture_agent_stockpiles = {
	set_variable = { bio_stockpile_safety_anthrax_stockpile = num_equipment@anthrax_bomb_1 }
	set_variable = { bio_stockpile_safety_plague_stockpile = num_equipment@plague_bomb_1 }
	set_variable = { bio_stockpile_safety_tularemia_stockpile = num_equipment@tularemia_bomb_1 }
	set_variable = { bio_stockpile_safety_smallpox_stockpile = num_equipment@smallpox_bomb_1 }

	set_variable = { bio_stockpile_safety_total_stockpile = bio_stockpile_safety_anthrax_stockpile }
	add_to_variable = { bio_stockpile_safety_total_stockpile = bio_stockpile_safety_plague_stockpile }
	add_to_variable = { bio_stockpile_safety_total_stockpile = bio_stockpile_safety_tularemia_stockpile }
	add_to_variable = { bio_stockpile_safety_total_stockpile = bio_stockpile_safety_smallpox_stockpile }

	set_variable = { bio_stockpile_safety_weighted_stockpile = bio_stockpile_safety_anthrax_stockpile }
	multiply_variable = { bio_stockpile_safety_weighted_stockpile = constant:bio_stockpile_safety_agent_risk.anthrax }
	set_temp_variable = { bio_stockpile_safety_weighted_agent = bio_stockpile_safety_plague_stockpile }
	multiply_temp_variable = { bio_stockpile_safety_weighted_agent = constant:bio_stockpile_safety_agent_risk.plague }
	add_to_variable = { bio_stockpile_safety_weighted_stockpile = bio_stockpile_safety_weighted_agent }
	set_temp_variable = { bio_stockpile_safety_weighted_agent = bio_stockpile_safety_tularemia_stockpile }
	multiply_temp_variable = { bio_stockpile_safety_weighted_agent = constant:bio_stockpile_safety_agent_risk.tularemia }
	add_to_variable = { bio_stockpile_safety_weighted_stockpile = bio_stockpile_safety_weighted_agent }
	set_temp_variable = { bio_stockpile_safety_weighted_agent = bio_stockpile_safety_smallpox_stockpile }
	multiply_temp_variable = { bio_stockpile_safety_weighted_agent = constant:bio_stockpile_safety_agent_risk.smallpox }
	add_to_variable = { bio_stockpile_safety_weighted_stockpile = bio_stockpile_safety_weighted_agent }

	set_variable = { bio_stockpile_safety_stockpile_snapshot_status = constant:bio_lifecycle_proof.supplied }
}

# Country scope. Selection weights are the unmodified actual stockpile counts;
# the weighted stockpile above is only for visible risk scoring.
bio_stockpile_safety_prepare_agent_weights = {
	bio_stockpile_safety_capture_agent_stockpiles = yes
	set_variable = { bio_stockpile_safety_anthrax_weight = bio_stockpile_safety_anthrax_stockpile }
	set_variable = { bio_stockpile_safety_plague_weight = bio_stockpile_safety_plague_stockpile }
	set_variable = { bio_stockpile_safety_tularemia_weight = bio_stockpile_safety_tularemia_stockpile }
	set_variable = { bio_stockpile_safety_smallpox_weight = bio_stockpile_safety_smallpox_stockpile }
	set_variable = { bio_stockpile_safety_agent_weights_status = constant:bio_lifecycle_proof.supplied }
}

# Country scope. Reads only the one designated arsenal state and records exact
# building/infrastructure values on the country for the current calculation.
# The state flag is runtime-only and is cleared by the cleanup helper.
bio_stockpile_safety_capture_exact_arsenal_condition = {
	set_variable = { bio_stockpile_safety_condition_snapshot_status = constant:bio_lifecycle_proof.missing }
	set_variable = { bio_stockpile_safety_facility_level = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_facility_damaged_level = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_facility_operational_level = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_war_damage_active = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_recent_bombing_active = constant:bio_lifecycle_math.zero }

	if = {
		limit = { bio_stockpile_safety_arsenal_pointer_is_valid = yes }
		var:bio_stockpile_safety_arsenal_state = {
			set_temp_variable = { bio_stockpile_safety_facility_level_captured = building_level@biowarfare_facility }
			set_temp_variable = { bio_stockpile_safety_facility_damaged_level_captured = damaged_building_level@biowarfare_facility }
			set_temp_variable = { bio_stockpile_safety_facility_operational_level_captured = non_damaged_building_level@biowarfare_facility }

			if = {
				limit = { days_since_last_strategic_bombing < constant:bio_stockpile_safety_risk.recent_bombing_days }
				PREV = { set_variable = { bio_stockpile_safety_recent_bombing_active = constant:bio_lifecycle_math.one } }
			}

			set_state_flag = bio_stockpile_safety_condition_snapshot_active
			PREV = {
				set_variable = { bio_stockpile_safety_facility_level = bio_stockpile_safety_facility_level_captured }
				set_variable = { bio_stockpile_safety_facility_damaged_level = bio_stockpile_safety_facility_damaged_level_captured }
				set_variable = { bio_stockpile_safety_facility_operational_level = bio_stockpile_safety_facility_operational_level_captured }
				set_variable = { bio_stockpile_safety_condition_snapshot_status = constant:bio_lifecycle_proof.supplied }
			}
		}
	}

	if = {
		limit = {
			check_variable = {
				var = bio_stockpile_safety_facility_damaged_level
				value = constant:bio_lifecycle_math.zero
				compare = greater_than
			}
		}
		set_variable = { bio_stockpile_safety_war_damage_active = constant:bio_lifecycle_math.one }
	}
}

# Country scope. Reads explicit sabotage/handling evidence only when it is
# tied to the exact arsenal pointer. The private flags are caller-owned input
# flags on that exact state and are cleared only by
# bio_stockpile_safety_cleanup_runtime_context.
bio_stockpile_safety_capture_handling_state = {
	set_variable = { bio_stockpile_safety_handling_context_status = constant:bio_lifecycle_proof.missing }
	set_variable = { bio_stockpile_safety_sabotage_active = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_recent_handling_active_value = constant:bio_lifecycle_math.zero }

	if = {
		limit = { bio_stockpile_safety_arsenal_pointer_is_valid = yes }
		if = {
			limit = { bio_stockpile_safety_explicit_sabotage_is_active = yes }
			set_variable = { bio_stockpile_safety_sabotage_active = constant:bio_lifecycle_math.one }
			set_variable = { bio_stockpile_safety_recent_handling_active_value = constant:bio_lifecycle_math.one }
		}
		if = {
			limit = {
				var:bio_stockpile_safety_arsenal_state = {
					has_state_flag = bio_stockpile_safety_recent_handling_active
				}
			}
			set_variable = { bio_stockpile_safety_recent_handling_active_value = constant:bio_lifecycle_math.one }
		}

		if = {
			limit = {
				has_variable = bio_last_sabotage_start_date
				has_variable = bio_last_sabotage_state
			}
			set_temp_variable = { bio_stockpile_safety_handling_elapsed_days = global.num_days }
			subtract_from_temp_variable = { bio_stockpile_safety_handling_elapsed_days = bio_last_sabotage_start_date }
			if = {
				limit = {
					check_variable = {
						var = bio_stockpile_safety_handling_elapsed_days
						value = constant:bio_stockpile_safety_risk.recent_handling_days
						compare = less_than
					}
				}
				if = {
					limit = { var:bio_last_sabotage_state = { state = var:bio_stockpile_safety_arsenal_state } }
					set_variable = { bio_stockpile_safety_recent_handling_active_value = constant:bio_lifecycle_math.one }
				}
			}
		}
		set_variable = { bio_stockpile_safety_handling_context_status = constant:bio_lifecycle_proof.supplied }
	}
}

# Country scope. Refreshes the visible risk score and one of the four bands.
# It does not decide whether an incident is an ordinary accident; callers must
# set bio_stockpile_safety_incident_kind and use the dedicated immunity trigger.
bio_stockpile_safety_refresh_risk_band = {
	set_variable = { bio_stockpile_safety_risk_band = constant:bio_stockpile_safety_band.none }
	set_variable = { bio_stockpile_safety_risk_score = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_risk_context_status = constant:bio_lifecycle_proof.missing }
	set_variable = { bio_stockpile_safety_effective_safety = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_safety_tech_level = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_stockpile_risk_component = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_facility_risk_component = constant:bio_lifecycle_math.zero }

	bio_stockpile_safety_capture_agent_stockpiles = yes
	bio_stockpile_safety_capture_exact_arsenal_condition = yes
	bio_stockpile_safety_capture_handling_state = yes

	if = {
		limit = { bio_stockpile_safety_risk_context_is_valid = yes }
		set_variable = { bio_stockpile_safety_effective_safety = cbrn_biological_security }
		if = {
			limit = { has_tech = pathogen_handling_protocols }
			add_to_variable = { bio_stockpile_safety_effective_safety = constant:bio_lifecycle_response.pathogen_handling_biosecurity_add }
			add_to_variable = { bio_stockpile_safety_safety_tech_level = constant:bio_lifecycle_math.one }
		}
		if = {
			limit = { has_tech = sealed_containment_laboratories }
			add_to_variable = { bio_stockpile_safety_effective_safety = constant:bio_lifecycle_response.sealed_labs_biosecurity_add }
			add_to_variable = { bio_stockpile_safety_safety_tech_level = constant:bio_lifecycle_math.one }
		}
		if = {
			limit = { has_tech = fail_safe_containment_facilities }
			add_to_variable = { bio_stockpile_safety_effective_safety = constant:bio_lifecycle_response.fail_safe_biosecurity_add }
			add_to_variable = { bio_stockpile_safety_safety_tech_level = constant:bio_lifecycle_math.one }
		}
		if = {
			limit = { cbrn_designer_has_sealed_sample_chain = yes }
			add_to_variable = { bio_stockpile_safety_effective_safety = constant:cbrn_biosecurity_designer_effect.sealed_sample_safety_add }
		}
		clamp_variable = {
			var = bio_stockpile_safety_effective_safety
			min = constant:bio_lifecycle_math.zero
			max = constant:bio_lifecycle_math.hundred
		}
		if = {
			limit = {
				has_country_flag = bio_stockpile_safety_accident_history
				check_variable = {
					var = bio_stockpile_safety_effective_safety
					value = constant:cbrn_achievement_threshold.maximum_biosecurity
					compare = greater_than_or_equals
				}
			}
			set_country_flag = cbrn_achievement_safety_reformed_history
		}

		set_variable = { bio_stockpile_safety_stockpile_risk_component = bio_stockpile_safety_weighted_stockpile }
		divide_variable = { bio_stockpile_safety_stockpile_risk_component = constant:bio_stockpile_safety_risk.stockpile_score_scale }
		multiply_variable = { bio_stockpile_safety_stockpile_risk_component = constant:bio_lifecycle_math.hundred }
		set_variable = { bio_stockpile_safety_risk_score = constant:bio_lifecycle_math.hundred }
		subtract_from_variable = { bio_stockpile_safety_risk_score = bio_stockpile_safety_effective_safety }
		add_to_variable = { bio_stockpile_safety_risk_score = bio_stockpile_safety_stockpile_risk_component }

		set_variable = { bio_stockpile_safety_facility_risk_component = bio_stockpile_safety_facility_damaged_level }
		multiply_variable = { bio_stockpile_safety_facility_risk_component = constant:bio_stockpile_safety_risk.facility_damage_penalty_per_level }
		if = {
			limit = {
				check_variable = {
					var = bio_stockpile_safety_facility_operational_level
					value = constant:bio_stockpile_safety_risk.facility_operational_minimum_level
					compare = less_than
				}
			}
			add_to_variable = { bio_stockpile_safety_facility_risk_component = constant:bio_stockpile_safety_risk.facility_inoperable_penalty }
		}
		add_to_variable = { bio_stockpile_safety_risk_score = bio_stockpile_safety_facility_risk_component }

		if = {
			limit = {
				check_variable = {
					var = bio_stockpile_safety_war_damage_active
					value = constant:bio_lifecycle_math.one
					compare = equals
				}
			}
			add_to_variable = { bio_stockpile_safety_risk_score = constant:bio_stockpile_safety_risk.war_damage_penalty }
		}
		if = {
			limit = {
				check_variable = {
					var = bio_stockpile_safety_recent_bombing_active
					value = constant:bio_lifecycle_math.one
					compare = equals
				}
			}
			add_to_variable = { bio_stockpile_safety_risk_score = constant:bio_stockpile_safety_risk.recent_bombing_penalty }
		}
		if = {
			limit = {
				check_variable = {
					var = bio_stockpile_safety_sabotage_active
					value = constant:bio_lifecycle_math.one
					compare = equals
				}
			}
			add_to_variable = { bio_stockpile_safety_risk_score = constant:bio_stockpile_safety_risk.sabotage_penalty }
		}
		if = {
			limit = {
				check_variable = {
					var = bio_stockpile_safety_recent_handling_active_value
					value = constant:bio_lifecycle_math.one
					compare = equals
				}
			}
			add_to_variable = { bio_stockpile_safety_risk_score = constant:bio_stockpile_safety_risk.recent_handling_penalty }
		}
		if = {
			limit = { cbrn_designer_has_accelerated_cultivation = yes }
			add_to_variable = { bio_stockpile_safety_risk_score = constant:cbrn_biosecurity_designer_effect.accelerated_cultivation_accident_risk_add }
		}
		if = {
			limit = { cbrn_designer_has_strategic_dissemination_program = yes }
			add_to_variable = { bio_stockpile_safety_risk_score = constant:cbrn_biosecurity_designer_effect.strategic_dissemination_accident_risk_add }
		}
		clamp_variable = {
			var = bio_stockpile_safety_risk_score
			min = constant:bio_stockpile_safety_risk.score_minimum
			max = constant:bio_stockpile_safety_risk.score_maximum
		}

		# The player-facing band follows the accepted matrix directly. The score
		# remains a readable 0-100 summary, but it is not a substitute for the
		# explicit safety, stockpile, damage, bombing, and sabotage gates.
		if = {
			limit = { bio_stockpile_safety_meets_critical_conditions = yes }
			set_variable = { bio_stockpile_safety_risk_band = constant:bio_stockpile_safety_band.critical }
		}
		else_if = {
			limit = { bio_stockpile_safety_meets_dangerous_conditions = yes }
			set_variable = { bio_stockpile_safety_risk_band = constant:bio_stockpile_safety_band.dangerous }
		}
		else_if = {
			limit = { bio_stockpile_safety_meets_controlled_conditions = yes }
			set_variable = { bio_stockpile_safety_risk_band = constant:bio_stockpile_safety_band.controlled }
		}
		else = {
			set_variable = { bio_stockpile_safety_risk_band = constant:bio_stockpile_safety_band.strained }
		}
		set_variable = { bio_stockpile_safety_risk_context_status = constant:bio_lifecycle_proof.supplied }
	}
}

# Country scope. Produces caller-facing random_list weights for the five
# ordinary-accident severities. The effect deliberately does not enter the
# lifecycle; a caller maps the selected severity to its existing route record.
bio_stockpile_safety_prepare_incident_severity_inputs = {
	set_variable = { bio_stockpile_safety_contained_weight = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_lab_contamination_weight = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_local_outbreak_weight = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_major_domestic_outbreak_weight = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_international_exposure_weight = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_incident_selection_status = constant:bio_lifecycle_proof.missing }

	bio_stockpile_safety_refresh_risk_band = yes
	if = {
		limit = {
			bio_stockpile_safety_risk_band_is_valid = yes
			bio_stockpile_safety_incident_is_ordinary_accident = yes
		}
		if = {
			limit = { bio_stockpile_safety_is_controlled = yes }
			set_variable = { bio_stockpile_safety_contained_weight = constant:bio_stockpile_safety_incident_weight.controlled.contained }
			set_variable = { bio_stockpile_safety_lab_contamination_weight = constant:bio_stockpile_safety_incident_weight.controlled.lab_contamination }
			set_variable = { bio_stockpile_safety_local_outbreak_weight = constant:bio_stockpile_safety_incident_weight.controlled.local_outbreak }
			set_variable = { bio_stockpile_safety_major_domestic_outbreak_weight = constant:bio_stockpile_safety_incident_weight.controlled.major_domestic_outbreak }
			set_variable = { bio_stockpile_safety_international_exposure_weight = constant:bio_stockpile_safety_incident_weight.controlled.international_exposure }
		}
		else_if = {
			limit = { bio_stockpile_safety_is_strained = yes }
			set_variable = { bio_stockpile_safety_contained_weight = constant:bio_stockpile_safety_incident_weight.strained.contained }
			set_variable = { bio_stockpile_safety_lab_contamination_weight = constant:bio_stockpile_safety_incident_weight.strained.lab_contamination }
			set_variable = { bio_stockpile_safety_local_outbreak_weight = constant:bio_stockpile_safety_incident_weight.strained.local_outbreak }
			set_variable = { bio_stockpile_safety_major_domestic_outbreak_weight = constant:bio_stockpile_safety_incident_weight.strained.major_domestic_outbreak }
			set_variable = { bio_stockpile_safety_international_exposure_weight = constant:bio_stockpile_safety_incident_weight.strained.international_exposure }
		}
		else_if = {
			limit = { bio_stockpile_safety_is_dangerous = yes }
			set_variable = { bio_stockpile_safety_contained_weight = constant:bio_stockpile_safety_incident_weight.dangerous.contained }
			set_variable = { bio_stockpile_safety_lab_contamination_weight = constant:bio_stockpile_safety_incident_weight.dangerous.lab_contamination }
			set_variable = { bio_stockpile_safety_local_outbreak_weight = constant:bio_stockpile_safety_incident_weight.dangerous.local_outbreak }
			set_variable = { bio_stockpile_safety_major_domestic_outbreak_weight = constant:bio_stockpile_safety_incident_weight.dangerous.major_domestic_outbreak }
			set_variable = { bio_stockpile_safety_international_exposure_weight = constant:bio_stockpile_safety_incident_weight.dangerous.international_exposure }
		}
		else_if = {
			limit = { bio_stockpile_safety_is_critical = yes }
			set_variable = { bio_stockpile_safety_contained_weight = constant:bio_stockpile_safety_incident_weight.critical.contained }
			set_variable = { bio_stockpile_safety_lab_contamination_weight = constant:bio_stockpile_safety_incident_weight.critical.lab_contamination }
			set_variable = { bio_stockpile_safety_local_outbreak_weight = constant:bio_stockpile_safety_incident_weight.critical.local_outbreak }
			set_variable = { bio_stockpile_safety_major_domestic_outbreak_weight = constant:bio_stockpile_safety_incident_weight.critical.major_domestic_outbreak }
			set_variable = { bio_stockpile_safety_international_exposure_weight = constant:bio_stockpile_safety_incident_weight.critical.international_exposure }
		}
		set_variable = { bio_stockpile_safety_incident_selection_total_weight = bio_stockpile_safety_contained_weight }
		add_to_variable = { bio_stockpile_safety_incident_selection_total_weight = bio_stockpile_safety_lab_contamination_weight }
		add_to_variable = { bio_stockpile_safety_incident_selection_total_weight = bio_stockpile_safety_local_outbreak_weight }
		add_to_variable = { bio_stockpile_safety_incident_selection_total_weight = bio_stockpile_safety_major_domestic_outbreak_weight }
		add_to_variable = { bio_stockpile_safety_incident_selection_total_weight = bio_stockpile_safety_international_exposure_weight }
		set_variable = { bio_stockpile_safety_incident_selection_status = constant:bio_lifecycle_proof.supplied }
	}
}

# Country scope. Selects only the already-prepared ordinary-accident severity
# and stores the private severity enum; it does not dispatch any consequences.
bio_stockpile_safety_select_incident_severity = {
	bio_stockpile_safety_prepare_incident_severity_inputs = yes
	set_variable = { bio_stockpile_safety_selected_incident_severity = constant:bio_stockpile_safety_incident_severity.none }
	if = {
		limit = { bio_stockpile_safety_incident_selection_inputs_are_valid = yes }
		random_list = {
			var:bio_stockpile_safety_contained_weight = {
				set_variable = { bio_stockpile_safety_selected_incident_severity = constant:bio_stockpile_safety_incident_severity.contained }
			}
			var:bio_stockpile_safety_lab_contamination_weight = {
				set_variable = { bio_stockpile_safety_selected_incident_severity = constant:bio_stockpile_safety_incident_severity.lab_contamination }
			}
			var:bio_stockpile_safety_local_outbreak_weight = {
				set_variable = { bio_stockpile_safety_selected_incident_severity = constant:bio_stockpile_safety_incident_severity.local_outbreak }
			}
			var:bio_stockpile_safety_major_domestic_outbreak_weight = {
				set_variable = { bio_stockpile_safety_selected_incident_severity = constant:bio_stockpile_safety_incident_severity.major_domestic_outbreak }
			}
			var:bio_stockpile_safety_international_exposure_weight = {
				set_variable = { bio_stockpile_safety_selected_incident_severity = constant:bio_stockpile_safety_incident_severity.international_exposure }
			}
		}
	}
}

# Country scope. Prepares a positive and negative exact loss amount for the
# selected lifecycle agent. The caller must provide a positive requested loss;
# this helper rounds it, enforces a minimum of one, and caps it at live stock.
bio_stockpile_safety_prepare_stock_loss_amount = {
	set_variable = { bio_stockpile_safety_selected_agent_stockpile = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_stock_loss_amount = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_stock_loss_amount_negative = constant:bio_lifecycle_math.zero }
	set_variable = { bio_stockpile_safety_stock_loss_preparation_status = constant:bio_lifecycle_proof.missing }

	bio_stockpile_safety_capture_agent_stockpiles = yes
	if = {
		limit = { bio_stockpile_safety_stock_loss_request_is_valid = yes }
		if = {
			limit = { check_variable = { var = bio_stockpile_safety_selected_agent value = constant:bio_lifecycle_agent.anthrax compare = equals } }

				limit = { check_variable = { var = bio_stockpile_safety_selected_agent value = constant:bio_lifecycle_agent.tularemia compare = equals } }
				country_event = { id = chaosx_bioweapon.102 }
			}
			else_if = {
				limit = { check_variable = { var = bio_stockpile_safety_selected_agent value = constant:bio_lifecycle_agent.smallpox compare = equals } }
				country_event = { id = chaosx_bioweapon.203 }
			}
		}
	}
}

# Country scope. Schedules the private monitor only for a verified designated
# program. The flag prevents duplicate delayed events from multiple projects.
bio_stockpile_safety_start_monitor = {
	cbrn_initialize_country_data = yes
	if = {
		limit = {
			bio_stockpile_safety_arsenal_pointer_is_valid = yes
			NOT = { has_country_flag = bio_stockpile_safety_monitor_scheduled }
		}
		set_country_flag = bio_stockpile_safety_monitor_scheduled
		set_variable = { bio_stockpile_safety_monitor_due_day = global.num_days }
		add_to_variable = { bio_stockpile_safety_monitor_due_day = constant:bio_stockpile_safety_monitor.initial_delay_days }
		country_event = {
			id = cbrn_bio_safety.1
			days = constant:bio_stockpile_safety_monitor.initial_delay_days
		}
	}
}

# Country scope. Clears only a previously recorded exact designation. It never
# scans for a replacement state and never changes national payload stock.
bio_stockpile_safety_clear_arsenal_designation = {
	if = {
		limit = { has_variable = bio_stockpile_safety_arsenal_state }
		var:bio_stockpile_safety_arsenal_state = {
			if = {
				limit = {
					NOT = { has_state_flag = bio_captured_facility_unsecured }
					has_variable = bio_stockpile_safety_arsenal_actor
					var:bio_stockpile_safety_arsenal_actor = { tag = ROOT }
				}
				clr_state_flag = bio_stockpile_safety_arsenal_complex
				clear_variable = bio_stockpile_safety_arsenal_actor
				clear_variable = bio_stockpile_safety_arsenal_designated_date
			}
		}
	}
	clear_variable = bio_stockpile_safety_arsenal_state
	clear_variable = bio_stockpile_safety_arsenal_pointer_source
	clear_variable = bio_stockpile_safety_monitor_due_day
	clr_country_flag = bio_stockpile_safety_arsenal_designated
	clr_country_flag = bio_stockpile_safety_monitor_scheduled
	clr_country_flag = bio_stockpile_safety_monitor_suspended
	bio_stockpile_safety_remove_risk_ideas = yes
}
