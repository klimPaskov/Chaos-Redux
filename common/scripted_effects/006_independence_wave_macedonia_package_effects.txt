independence_wave_refresh_mac_lifecycle = {
	if = {
		limit = { is_independence_wave_mac_package = yes }
		remove_ideas = { mac_divided_vardar_authority mac_vardar_compact }
		if = {
			limit = { has_stable_independence_wave_mac_ledgers = yes }
			add_ideas = mac_vardar_compact
			set_country_flag = independence_wave_mac_compact_stabilized
		}
		else = {
			add_ideas = mac_divided_vardar_authority
			clr_country_flag = independence_wave_mac_compact_stabilized
		}
	}
}
