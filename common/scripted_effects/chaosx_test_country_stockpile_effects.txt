chaosx_test_country_register_flame_tank_variants = {
	if = {
		limit = { NOT = { has_country_flag = chaosx_test_flame_tank_variants_registered } }
		set_country_flag = chaosx_test_flame_tank_variants_registered
		create_equipment_variant = {
			name = "CXT Light Chemical Carrier"
			type = light_tank_chassis_3
			allow_without_tech = yes
			modules = {
				main_armament_slot = advanced_flamethrower
				turret_type_slot = tank_light_fixed_superstructure_turret
				suspension_type_slot = tank_christie_suspension
				armor_type_slot = tank_welded_armor
				engine_type_slot = tank_gasoline_engine
			}
		}
		create_equipment_variant = {
			name = "CXT Medium Chemical Carrier"
			type = medium_tank_chassis_3
			allow_without_tech = yes
			modules = {
				main_armament_slot = advanced_flamethrower
				turret_type_slot = tank_medium_fixed_superstructure_turret
				suspension_type_slot = tank_christie_suspension
				armor_type_slot = tank_welded_armor
				engine_type_slot = tank_gasoline_engine
			}
		}
		create_equipment_variant = {
			name = "CXT Heavy Chemical Carrier"
			type = heavy_tank_chassis_3
			allow_without_tech = yes
			modules = {
				main_armament_slot = advanced_flamethrower
				turret_type_slot = tank_heavy_fixed_superstructure_turret
				suspension_type_slot = tank_christie_suspension
				armor_type_slot = tank_welded_armor
				engine_type_slot = tank_gasoline_engine
			}
		}
	}
}
