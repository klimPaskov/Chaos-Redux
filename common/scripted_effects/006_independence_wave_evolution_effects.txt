# ============================================================================
# CHAOS REDUX - EVENT 006 INDEPENDENCE WAVE EVOLUTION EFFECTS
# ============================================================================
# Overview:
# - Owns the five canonical Event 006 evolution transitions and Event Log rows.
# - Freezes active evolution state before package weighting begins.
# - Delivers each transition only through the aligned Event 006 active registry.
# - Applies the frozen opening package to each released country idempotently.
# - Samples MTTH only when Event 006 itself is invoked; there is no periodic
#   country or world scan in this file.
# ============================================================================

# ---------------------------------------------------------------------------
# Event Log context and actor selection
# ---------------------------------------------------------------------------

# Input: temporary `independence_wave_evolution_stage`.
independence_wave_load_evolution_log_context = {
	set_temp_variable = { events_log_evolution_event_id = constant:independence_wave_evolution.event_id }
	set_temp_variable = { events_log_evolution_type = constant:independence_wave_evolution.evolution_type }
	set_temp_variable = { events_log_evolution_stage = independence_wave_evolution_stage }
	if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.replicable_independence_stage } }
		set_temp_variable = { events_log_evolution_tier = constant:independence_wave_evolution.replicable_independence_tier }
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.dormant_nations_stage } }
		set_temp_variable = { events_log_evolution_tier = constant:independence_wave_evolution.dormant_nations_tier }
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.armed_birth_stage } }
		set_temp_variable = { events_log_evolution_tier = constant:independence_wave_evolution.armed_birth_tier }
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.sovereign_congress_stage } }
		set_temp_variable = { events_log_evolution_tier = constant:independence_wave_evolution.sovereign_congress_tier }
	}
	else = {
		set_temp_variable = { events_log_evolution_tier = constant:independence_wave_evolution.open_sovereignty_tier }
	}
}

# Input: temporary `independence_wave_evolution_stage`. Called only after the
# canonical stage date exists and immediately before the Event Log write.
independence_wave_load_evolution_log_date = {
	if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.replicable_independence_stage } }
		set_temp_variable = { events_log_evolution_date_override = global.independence_wave_evolution_replicable_independence_date }
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.dormant_nations_stage } }
		set_temp_variable = { events_log_evolution_date_override = global.independence_wave_evolution_dormant_nations_date }
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.armed_birth_stage } }
		set_temp_variable = { events_log_evolution_date_override = global.independence_wave_evolution_armed_birth_date }
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.sovereign_congress_stage } }
		set_temp_variable = { events_log_evolution_date_override = global.independence_wave_evolution_sovereign_congress_date }
	}
	else = {
		set_temp_variable = { events_log_evolution_date_override = global.independence_wave_evolution_open_sovereignty_date }
	}
}

independence_wave_select_evolution_actor = {
	set_temp_variable = { events_log_evolution_has_actor = constant:independence_wave_value.minimum }
	if = {
		limit = { is_independence_wave_active_country = yes }
		save_event_target_as = events_log_evolution_actor
		set_temp_variable = { events_log_evolution_has_actor = constant:independence_wave_plan.loop_increment }
	}
	else_if = {
		limit = { independence_wave_evolution_registry_is_ready = yes }
		for_each_scope_loop = {
			array = global.independence_wave_active_countries
			limit = {
				check_variable = { events_log_evolution_has_actor = constant:independence_wave_value.minimum }
				is_independence_wave_active_country = yes
			}
			save_event_target_as = events_log_evolution_actor
			set_temp_variable = { events_log_evolution_has_actor = constant:independence_wave_plan.loop_increment }
		}
	}
}

# Input: temporary `independence_wave_evolution_stage`.
# A pre-fire transition has no Event 006 country yet, so its row is retained
# until the first frozen country is initialized and can be recorded as actor.
independence_wave_record_or_defer_current_evolution = {
	independence_wave_load_evolution_log_context = yes
	independence_wave_select_evolution_actor = yes
	if = {
		limit = { check_variable = { events_log_evolution_has_actor = constant:independence_wave_plan.loop_increment } }
		independence_wave_load_evolution_log_date = yes
		record_events_log_evolution_entry = yes
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.replicable_independence_stage } }
		set_global_flag = independence_wave_replicable_independence_log_pending
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.dormant_nations_stage } }
		set_global_flag = independence_wave_dormant_nations_log_pending
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.armed_birth_stage } }
		set_global_flag = independence_wave_armed_birth_log_pending
	}
	else_if = {
		limit = { check_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.sovereign_congress_stage } }
		set_global_flag = independence_wave_sovereign_congress_log_pending
	}
	else = {
		set_global_flag = independence_wave_open_sovereignty_log_pending
	}
}

# COUNTRY scope after origin registration.
independence_wave_record_pending_evolution_logs = {
	if = {
		limit = { is_independence_wave_active_country = yes has_global_flag = independence_wave_replicable_independence_log_pending }
		clr_global_flag = independence_wave_replicable_independence_log_pending
		set_temp_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.replicable_independence_stage }
		independence_wave_record_or_defer_current_evolution = yes
	}
	if = {
		limit = { is_independence_wave_active_country = yes has_global_flag = independence_wave_dormant_nations_log_pending }
		clr_global_flag = independence_wave_dormant_nations_log_pending
		set_temp_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.dormant_nations_stage }

	}
	if = {
		limit = { is_independence_wave_active_country = yes has_global_flag = independence_wave_armed_birth_log_pending }
		clr_global_flag = independence_wave_armed_birth_log_pending
		set_temp_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.armed_birth_stage }
		independence_wave_record_or_defer_current_evolution = yes
	}
	if = {
		limit = { is_independence_wave_active_country = yes has_global_flag = independence_wave_sovereign_congress_log_pending }
		clr_global_flag = independence_wave_sovereign_congress_log_pending
		set_temp_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.sovereign_congress_stage }
		independence_wave_record_or_defer_current_evolution = yes
	}
	if = {
		limit = { is_independence_wave_active_country = yes has_global_flag = independence_wave_open_sovereignty_log_pending }
		clr_global_flag = independence_wave_open_sovereignty_log_pending
		set_temp_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.open_sovereignty_stage }
		independence_wave_record_or_defer_current_evolution = yes
	}
}

independence_wave_apply_replicable_independence_to_country = {
	if = {
		limit = {
			is_independence_wave_active_country = yes
			NOT = { has_country_flag = independence_wave_replicable_independence_applied }
		}
		set_country_flag = independence_wave_replicable_independence_applied
		set_country_flag = independence_wave_unlock_foreign_service
		add_to_variable = { independence_wave_legitimacy = constant:independence_wave_evolution_country.replicable_legitimacy }
		add_to_variable = { independence_wave_recognition = constant:independence_wave_evolution_country.replicable_recognition }
		add_to_variable = { independence_wave_capacity = constant:independence_wave_evolution_country.replicable_capacity }
		add_to_variable = { independence_wave_network_standing = constant:independence_wave_evolution_country.replicable_network_standing }
		if = {
			limit = {
				OR = {
					AND = {
						has_variable = independence_wave_pending_evolution_preexisting_active_count
						check_variable = { var = independence_wave_pending_evolution_preexisting_active_count value = constant:independence_wave_value.minimum compare = greater_than }
					}
					AND = {
						NOT = { has_variable = independence_wave_pending_evolution_preexisting_active_count }
						check_variable = { var = global.independence_wave_active_country_count value = constant:independence_wave_plan.loop_increment compare = greater_than }
					}
				}
			}
			set_country_flag = independence_wave_prior_network_contact
			add_to_variable = { independence_wave_recognition = constant:independence_wave_evolution_country.replicable_prior_contact_recognition }
			add_to_variable = { independence_wave_network_standing = constant:independence_wave_evolution_country.replicable_prior_contact_network_standing }
		}
		clamp_variable = { var = independence_wave_network_standing min = constant:independence_wave_network.standing_minimum max = constant:independence_wave_network.standing_maximum }
		independence_wave_refresh_country_state = yes
	}
}

	if = {
		limit = {
			is_independence_wave_active_country = yes
			NOT = { has_country_flag = independence_wave_sovereign_congress_applied }
		}
		set_country_flag = independence_wave_sovereign_congress_applied
		set_country_flag = independence_wave_unlock_league_congress
		add_to_variable = { independence_wave_legitimacy = constant:independence_wave_evolution_country.congress_legitimacy }
		add_to_variable = { independence_wave_recognition = constant:independence_wave_evolution_country.congress_recognition }
		add_to_variable = { independence_wave_capacity = constant:independence_wave_evolution_country.congress_capacity }
		add_to_variable = { independence_wave_network_standing = constant:independence_wave_evolution_country.congress_network_standing }
		clr_country_flag = independence_wave_sovereign_congress_delegate
		clr_country_flag = independence_wave_sovereign_congress_observer
		clr_country_flag = independence_wave_sovereign_congress_rejected
		if = {
			limit = { has_country_flag = independence_wave_client_route_locked }
			set_country_flag = independence_wave_sovereign_congress_rejected
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { independence_wave_package_depth = constant:independence_wave_package_depth.signature }
					check_variable = { var = independence_wave_network_standing value = constant:independence_wave_network.invitation_gate compare = greater_than_or_equals }
				}
			}
			set_country_flag = independence_wave_sovereign_congress_delegate
		}
		else = { set_country_flag = independence_wave_sovereign_congress_observer }
		clamp_variable = { var = independence_wave_network_standing min = constant:independence_wave_network.standing_minimum max = constant:independence_wave_network.standing_maximum }
		independence_wave_refresh_country_state = yes
	}
}

}

	}
	independence_wave_seed_sovereign_congress = yes

		independence_wave_apply_open_sovereignty_to_country = yes
	}
	independence_wave_seed_open_sovereignty = yes
}

# Applies canonical transitions to countries born while stages were already
# active. The country effects are idempotent and do not re-record log rows.
independence_wave_sync_active_evolutions_to_registry = {
	if = { limit = { has_independence_wave_replicable_independence_evolution = yes } independence_wave_deliver_replicable_independence = yes }
	if = { limit = { has_independence_wave_dormant_nations_evolution = yes } independence_wave_deliver_dormant_nations = yes }
	if = { limit = { has_independence_wave_armed_birth_evolution = yes } independence_wave_deliver_armed_birth = yes }
	if = { limit = { has_independence_wave_sovereign_congress_evolution = yes } independence_wave_deliver_sovereign_congress = yes }
	if = { limit = { has_independence_wave_open_sovereignty_evolution = yes } independence_wave_deliver_open_sovereignty = yes }
}

		set_variable = { global.independence_wave_evolution_stage = constant:independence_wave_evolution.open_sovereignty_stage }
		set_variable = { global.independence_wave_evolution_open_sovereignty_date = global.date }
		independence_wave_deliver_open_sovereignty = yes
		set_temp_variable = { independence_wave_evolution_stage = constant:independence_wave_evolution.open_sovereignty_stage }
		independence_wave_record_or_defer_current_evolution = yes
		set_temp_variable = { independence_wave_evolution_activated = constant:independence_wave_plan.loop_increment }
	}
}

# Activates every tier-eligible, enabled stage before the first wave is frozen.

independence_wave_schedule_next_evolution_check = {
	if = {
		limit = { NOT = { independence_wave_all_evolutions_are_active = yes } }
		set_temp_variable = { independence_wave_evolution_delay_days = mtth:independence_wave_evolution_interval }
		clamp_temp_variable = {
			var = independence_wave_evolution_delay_days
			min = constant:independence_wave_evolution_mtth.minimum_days
			max = constant:independence_wave_evolution_mtth.maximum_days
		}
		set_variable = { global.independence_wave_next_evolution_date = global.date }
		add_to_variable = { global.independence_wave_next_evolution_date = independence_wave_evolution_delay_days }
	}
	else = { clear_variable = global.independence_wave_next_evolution_date }
}

# Called once by Event 006 before allocator state is frozen.
independence_wave_prepare_evolution_for_incident = {
	independence_wave_reconcile_active_registry = yes
	if = {
		limit = { independence_wave_evolution_registry_is_ready = yes }
		if = {
			limit = {
				check_variable = {
					var = global.independence_wave_active_countries^num
					value = constant:independence_wave_value.minimum
					compare = equals
				}
			}
			independence_wave_activate_prefire_evolutions = yes
		}
		else = {
			independence_wave_sync_active_evolutions_to_registry = yes
			if = {
				limit = { independence_wave_all_evolutions_are_active = yes }
				clear_variable = global.independence_wave_next_evolution_date
			}
			else_if = {
				limit = { NOT = { has_variable = global.independence_wave_next_evolution_date } }
				independence_wave_schedule_next_evolution_check = yes
			}
			else_if = {
				limit = {
					check_variable = {
						var = global.independence_wave_next_evolution_date
						value = global.date
						compare = less_than_or_equals
					}
				}
				independence_wave_try_next_active_evolution = yes
				independence_wave_schedule_next_evolution_check = yes
			}
		}
	}
}
