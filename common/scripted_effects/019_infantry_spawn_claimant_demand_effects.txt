			check_variable = { command_power > constant:infantry_spawn_generalissimo_cost.subordinate_command_command_power }
		}

	else_if = {
		limit = {

			check_variable = { political_power > constant:infantry_spawn_generalissimo_cost.parallel_command_political_power }
			check_variable = { command_power > constant:infantry_spawn_generalissimo_cost.parallel_command_command_power }
		}
		set_temp_variable = { infantry_spawn_selected_claimant_demand = constant:infantry_spawn_generalissimo_demand.parallel_command }
	}

		limit = {
			check_variable = {
				var = infantry_spawn_claimant_influence_entries^infantry_spawn_selected_claimant_index
				value = constant:infantry_spawn_generalissimo_threshold.parallel_command_influence
				compare = greater_than_or_equals

	}
	else_if = {
		limit = {
			check_variable = { army_experience > constant:infantry_spawn_generalissimo_cost.subordinate_command_army_experience }
			check_variable = { command_power > constant:infantry_spawn_generalissimo_cost.subordinate_command_command_power }
		}
		set_temp_variable = { infantry_spawn_selected_claimant_demand = constant:infantry_spawn_generalissimo_demand.subordinate_command }

	else_if = {
		limit = {
			check_variable = { political_power > constant:infantry_spawn_generalissimo_cost.parallel_command_political_power }
			check_variable = { command_power > constant:infantry_spawn_generalissimo_cost.parallel_command_command_power }
		}
		set_temp_variable = { infantry_spawn_selected_claimant_demand = constant:infantry_spawn_generalissimo_demand.parallel_command }
	}
}

	set_temp_variable = { infantry_spawn_claimant_demand_candidate_payable = 0 }

	}

# Restores the movable claimant selection from the immutable UID that opened
# the pending demand. A missing or mismatched owner is an invariant failure;
# no other claimant row is substituted.
infantry_spawn_quarantine_pending_claimant_demand = {
	infantry_spawn_mark_deferred_replay_invariant_failure = yes
	clr_country_flag = infantry_spawn_claimant_demand_pending
	clear_variable = infantry_spawn_claimant_demand_owner_uid
}

infantry_spawn_select_pending_claimant_demand_owner = {
	if = {
		limit = { has_country_flag = infantry_spawn_claimant_demand_pending }
		if = {
			limit = { has_variable = infantry_spawn_claimant_demand_owner_uid }
			set_temp_variable = { infantry_spawn_lookup_claimant_uid = infantry_spawn_claimant_demand_owner_uid }
			infantry_spawn_find_claimant_row = yes
			if = {
				limit = {
					check_variable = { infantry_spawn_claimant_row_index > constant:infantry_spawn_event.invalid_index }
					check_variable = {
						var = infantry_spawn_claimant_uid_entries^infantry_spawn_claimant_row_index
						value = infantry_spawn_claimant_demand_owner_uid
						compare = equals
					}
					check_variable = {
						var = infantry_spawn_claimant_status_entries^infantry_spawn_claimant_row_index
						value = constant:infantry_spawn_claimant_status.retired
						compare = less_than
					}
					check_variable = {
						var = infantry_spawn_claimant_demand_entries^infantry_spawn_claimant_row_index
						value = constant:infantry_spawn_claimant_demand.none
						compare = greater_than
					}
				}
				set_variable = { infantry_spawn_selected_claimant_index = infantry_spawn_claimant_row_index }
			}
			else = { infantry_spawn_quarantine_pending_claimant_demand = yes }
		}
		else = { infantry_spawn_quarantine_pending_claimant_demand = yes }
	}
}

infantry_spawn_issue_selected_claimant_demand = {
	if = {
		limit = {
			infantry_spawn_selected_claimant_index_is_valid = yes
			check_variable = { infantry_spawn_selected_claimant_demand > constant:infantry_spawn_claimant_demand.none }
		}
		infantry_spawn_selected_claimant_demand_candidate_is_payable = yes
		if = {
			limit = { check_variable = { infantry_spawn_claimant_demand_candidate_payable > 0 } }
			set_variable = { infantry_spawn_claimant_status_entries^infantry_spawn_selected_claimant_index = constant:infantry_spawn_claimant_status.demanding }
			set_variable = { infantry_spawn_claimant_demand_date_entries^infantry_spawn_selected_claimant_index = global.date }
			add_to_variable = { infantry_spawn_claimant_demand_date_entries^infantry_spawn_selected_claimant_index = constant:infantry_spawn_timer.claimant_demand_days }
			set_variable = { infantry_spawn_claimant_demand_owner_uid = infantry_spawn_claimant_uid_entries^infantry_spawn_selected_claimant_index }
			set_country_flag = infantry_spawn_claimant_demand_pending
			if = {
				limit = { is_ai = no }
				set_temp_variable = { infantry_spawn_claimant_localisation_index = infantry_spawn_selected_claimant_index }
				infantry_spawn_load_claimant_localisation_context = yes
				country_event = { id = chaosx.nr19.201 }
			}
			else = { infantry_spawn_ai_respond_to_selected_claimant_demand = yes }
		}
	}
}

infantry_spawn_maybe_issue_primary_claimant_demand = {
	if = {
		limit = { has_country_flag = infantry_spawn_claimant_demand_pending }
		infantry_spawn_select_pending_claimant_demand_owner = yes
	}
	else = { clear_variable = infantry_spawn_claimant_demand_owner_uid }
	if = {
		limit = {
			infantry_spawn_selected_claimant_index_is_valid = yes
			NOT = { has_country_flag = infantry_spawn_claimant_demand_pending }
			NOT = { has_country_flag = infantry_spawn_claimant_revolt_warning_active }
		}
		set_temp_variable = { infantry_spawn_claimant_demand_chance = constant:infantry_spawn_claimant_pulse.base_demand_chance }
		if = {
			limit = { infantry_spawn_control_is_low = yes }
			add_to_temp_variable = { infantry_spawn_claimant_demand_chance = constant:infantry_spawn_claimant_pulse.low_control_demand_bonus }
		}
		if = {
			limit = { infantry_spawn_congestion_is_severe = yes }
			add_to_temp_variable = { infantry_spawn_claimant_demand_chance = constant:infantry_spawn_claimant_pulse.severe_congestion_demand_bonus }
		}
		if = {
			limit = { has_war = yes }
			add_to_temp_variable = { infantry_spawn_claimant_demand_chance = constant:infantry_spawn_claimant_pulse.war_demand_bonus }
		}
		random = {
			chance = infantry_spawn_claimant_demand_chance
			infantry_spawn_select_primary_claimant_demand = yes
			infantry_spawn_issue_selected_claimant_demand = yes
		}
	}
	if = {
		limit = {
			has_country_flag = infantry_spawn_claimant_demand_pending
			infantry_spawn_selected_claimant_index_is_valid = yes
			check_variable = {
				var = global.date
				value = infantry_spawn_claimant_demand_date_entries^infantry_spawn_selected_claimant_index
				compare = greater_than_or_equals
			}
		}
		infantry_spawn_refuse_selected_claimant_demand = yes
	}
}

# Inputs: valid infantry_spawn_selected_claimant_index,
# infantry_spawn_claimant_debt_relief, and
# infantry_spawn_claimant_settlement_resource_profile. Pays only the selected
# claimant's exact matching equipment-family obligation rows in ledger order.
# Aggregate debt, lot debt, paid/outstanding amounts, and row status move
# together atomically.
infantry_spawn_settle_selected_claimant_equipment_obligations = {
	if = {
		limit = {
			infantry_spawn_selected_claimant_index_is_valid = yes
			check_variable = { infantry_spawn_claimant_debt_relief > 0 }
		}
		set_temp_variable = { infantry_spawn_claimant_settlement_remaining = infantry_spawn_claimant_debt_relief }
		set_temp_variable = { infantry_spawn_claimant_settlement_uid = infantry_spawn_claimant_uid_entries^infantry_spawn_selected_claimant_index }
		set_temp_variable = { infantry_spawn_claimant_settlement_lot_index = constant:infantry_spawn_event.first_index }
		while_loop_effect = {
			limit = {
				check_variable = { infantry_spawn_claimant_settlement_remaining > 0 }
				check_variable = { var = infantry_spawn_claimant_settlement_lot_index value = infantry_spawn_lot_uid_entries^num compare = less_than }
			}
			if = {
				limit = {
					check_variable = { var = infantry_spawn_lot_claimant_uid_entries^infantry_spawn_claimant_settlement_lot_index value = infantry_spawn_claimant_settlement_uid compare = equals }
					check_variable = { infantry_spawn_lot_outstanding_debt_entries^infantry_spawn_claimant_settlement_lot_index > 0 }
				}
				set_temp_variable = { infantry_spawn_claimant_settlement_lot_uid = infantry_spawn_lot_uid_entries^infantry_spawn_claimant_settlement_lot_index }
				set_temp_variable = { infantry_spawn_claimant_settlement_obligation_index = constant:infantry_spawn_event.first_index }
				while_loop_effect = {
					limit = {
						check_variable = { infantry_spawn_claimant_settlement_remaining > 0 }
						check_variable = { var = infantry_spawn_claimant_settlement_obligation_index value = infantry_spawn_obligation_uid_entries^num compare = less_than }
					}
					if = {
						limit = {
							check_variable = { var = infantry_spawn_obligation_lot_uid_entries^infantry_spawn_claimant_settlement_obligation_index value = infantry_spawn_claimant_settlement_lot_uid compare = equals }
							check_variable = { infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_claimant_settlement_obligation_index > 0 }
							check_variable = { var = infantry_spawn_obligation_resource_profile_entries^infantry_spawn_claimant_settlement_obligation_index value = infantry_spawn_claimant_settlement_resource_profile compare = equals }
							OR = {
								check_variable = { infantry_spawn_obligation_status_entries^infantry_spawn_claimant_settlement_obligation_index = constant:infantry_spawn_obligation_state.outstanding }
								check_variable = { infantry_spawn_obligation_status_entries^infantry_spawn_claimant_settlement_obligation_index = constant:infantry_spawn_obligation_state.servicing }
							}
						}
						set_temp_variable = { infantry_spawn_claimant_settlement_payment = infantry_spawn_obligation_outstanding_amount_entries^infantry_spawn_claimant_settlement_obligation_index }
						if = {
							limit = { check_variable = { var = infantry_spawn_claimant_settlement_remaining value = infantry_spawn_claimant_settlement_payment compare = less_than } }
							set_temp_variable = { infantry_spawn_claimant_settlement_payment = infantry_spawn_claimant_settlement_remaining }
						}
						set_temp_variable = { infantry_spawn_obligation_payment_index = infantry_spawn_claimant_settlement_obligation_index }
						set_temp_variable = { infantry_spawn_obligation_payment_lot_index = infantry_spawn_claimant_settlement_lot_index }
						set_temp_variable = { infantry_spawn_obligation_payment_amount = infantry_spawn_claimant_settlement_payment }
						set_temp_variable = { infantry_spawn_obligation_payment_is_material = 1 }
						infantry_spawn_apply_exact_obligation_payment = yes
						if = {
							limit = { check_variable = { infantry_spawn_obligation_payment_applied > 0 } }
							subtract_from_temp_variable = { infantry_spawn_claimant_settlement_remaining = infantry_spawn_claimant_settlement_payment }
						}
						else = { set_temp_variable = { infantry_spawn_claimant_settlement_remaining = 0 } }
					}
					add_to_temp_variable = { infantry_spawn_claimant_settlement_obligation_index = 1 }
				}
			}
			add_to_temp_variable = { infantry_spawn_claimant_settlement_lot_index = 1 }
		}
		infantry_spawn_close_resolved_generations = yes
	}
}

# COUNTRY scope. Materializes one exact Evolution III registry lot at the
# claimant's recorded headquarters and attaches it only after the registry's
# transaction flag confirms the formation exists.
infantry_spawn_generate_selected_claimant_requested_lot = {
	set_temp_variable = { infantry_spawn_claimant_request_succeeded = 0 }
	infantry_spawn_prepare_request_generation_context = yes
	if = {
		limit = {
			check_variable = { infantry_spawn_request_context_ready > 0 }
			infantry_spawn_selected_claimant_has_safe_request_origin = yes
		}
		var:infantry_spawn_claimant_headquarters_state_entries^infantry_spawn_selected_claimant_index = {
			save_event_target_as = infantry_spawn_current_origin_state
		}
		infantry_spawn_prepare_selected_claimant_request_inputs = yes
		infantry_spawn_unit_registry_draw_requested_lot = yes
		if = {
			limit = {
				is_in_array = {
					array = infantry_spawn_unit_registry_outcome_tag_ids
					value = constant:infantry_spawn_unit_registry_outcome_tag.draw_complete
				}
			}
			infantry_spawn_unit_registry_materialize_drawn_lot = yes
		}
		if = {
			limit = { has_country_flag = infantry_spawn_unit_registry_materialization_succeeded }
			set_temp_variable = { infantry_spawn_claimant_best_lot_index = infantry_spawn_unit_registry_materialized_lot_index }
			infantry_spawn_assign_lot_index_to_selected_claimant = yes
			add_to_variable = { infantry_spawn_request_count_generation = 1 }
			if = {
				limit = { infantry_spawn_contributes_to_ordinary_evolution_history = yes }
				add_to_variable = { global.infantry_spawn_total_requests = 1 }
			}
			set_temp_variable = { infantry_spawn_claimant_request_succeeded = 1 }
		}
	}
}

infantry_spawn_accept_selected_claimant_demand = {
	if = {
		limit = { infantry_spawn_can_accept_selected_claimant_demand = yes }
		set_temp_variable = { infantry_spawn_claimant_demand_resolution_succeeded = 1 }
		set_temp_variable = { infantry_spawn_current_claimant_demand = infantry_spawn_claimant_demand_entries^infantry_spawn_selected_claimant_index }
		if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_claimant_demand.formal_appointment } }
			infantry_spawn_apply_formal_appointment_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_claimant_demand.equipment_share } }
			infantry_spawn_apply_equipment_share_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_claimant_demand.autonomous_district } }
			infantry_spawn_apply_autonomous_district_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_claimant_demand.another_formation } }
			set_temp_variable = { infantry_spawn_claimant_demand_resolution_succeeded = 0 }
			infantry_spawn_apply_another_formation_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_claimant_demand.political_seat } }
			infantry_spawn_apply_political_seat_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_claimant_demand.emergency_powers } }
			infantry_spawn_apply_emergency_powers_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_generalissimo_demand.subordinate_command } }
			infantry_spawn_apply_subordinate_command_demand = yes
		}
		else_if = {
			limit = { check_variable = { infantry_spawn_current_claimant_demand = constant:infantry_spawn_generalissimo_demand.parallel_command } }
			infantry_spawn_apply_parallel_command_demand = yes
		}
		if = {
			limit = { check_variable = { infantry_spawn_claimant_demand_resolution_succeeded > 0 } }
			add_to_variable = { infantry_spawn_claimant_accepted_demand_count_entries^infantry_spawn_selected_claimant_index = 1 }
			infantry_spawn_finish_selected_claimant_demand = yes
		}
	}
}

infantry_spawn_ai_respond_to_selected_claimant_demand = {
	if = {
		limit = { infantry_spawn_can_accept_selected_claimant_demand = yes }
		set_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_claimant_demand_ai.accept_base }
		if = {
			limit = { has_war = yes }
			add_to_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_claimant_demand_ai.accept_war_bonus }
		}
		if = {
			limit = { infantry_spawn_control_is_low = yes }
			add_to_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_claimant_demand_ai.accept_low_control_bonus }
		}
		if = {
			limit = { infantry_spawn_control_is_strong = yes }
			subtract_from_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_claimant_demand_ai.accept_strong_control_penalty }
		}
		if = {
			limit = { infantry_spawn_selected_claimant_is_event67_generalissimo = yes }
			add_to_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_generalissimo_ai.exact_claimant_accept_bonus }
		}
		if = {
			limit = {
				check_variable = {
					var = infantry_spawn_claimant_demand_entries^infantry_spawn_selected_claimant_index
					value = constant:infantry_spawn_generalissimo_demand.subordinate_command
					compare = equals
				}
			}
			add_to_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_generalissimo_ai.subordinate_command_accept_bonus }
		}
		if = {
			limit = {
				check_variable = {
					var = infantry_spawn_claimant_demand_entries^infantry_spawn_selected_claimant_index
					value = constant:infantry_spawn_generalissimo_demand.parallel_command
					compare = equals
				}
			}
			subtract_from_temp_variable = { infantry_spawn_ai_accept_claimant_weight = constant:infantry_spawn_generalissimo_ai.parallel_command_accept_penalty }
		}
		clamp_temp_variable = {
			var = infantry_spawn_ai_accept_claimant_weight
			min = constant:infantry_spawn_claimant_demand_ai.accept_minimum
			max = constant:infantry_spawn_claimant_demand_ai.accept_maximum
		}
		random = {
			chance = infantry_spawn_ai_accept_claimant_weight
			infantry_spawn_accept_selected_claimant_demand = yes
		}
		if = {
			limit = { has_country_flag = infantry_spawn_claimant_demand_pending }
			infantry_spawn_refuse_selected_claimant_demand = yes
		}
	}
	else = { infantry_spawn_refuse_selected_claimant_demand = yes }
}
