# COUNTRY scope on the host after the owning opening/playbook logic proves one
# of the stable start profiles. The caller sets africa_achievement_host_profile_id.
africa_achievement_record_host_profile = {
	if = {
		limit = { check_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.voluntary_league } }
		set_global_flag = africa_achievement_voluntary_league_route
	}
	else_if = {
		limit = { check_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.monarchical_or_restored } }
		set_global_flag = africa_achievement_origin_monarchical_or_restored
	}
	else_if = {
		limit = { check_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.socialist_or_communist } }
		set_global_flag = africa_achievement_origin_socialist_or_communist
	}
	else_if = {
		limit = { check_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.military_or_authoritarian } }
		set_global_flag = africa_achievement_origin_military_or_authoritarian
	}
	else_if = {
		limit = { check_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.coastal } }
		set_global_flag = africa_achievement_origin_coastal_host
	}
}

# COUNTRY scope on the normally selected host immediately after the lifetime
# ledgers are initialised. These are immutable launch facts: later elections,
# coups, or cosmetic identities cannot manufacture a different eligible start.
africa_achievement_record_initial_host_profiles = {
	if = {
		limit = { has_government = communism }
		set_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.socialist_or_communist }
		africa_achievement_record_host_profile = yes
	}
	if = {
		limit = {
			OR = {
				has_government = fascism
				has_government = neutrality
			}
		}
		set_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.military_or_authoritarian }
		africa_achievement_record_host_profile = yes
	}
	if = {
		limit = {
			OR = {
				check_variable = { africa_host_playbook = constant:africa_host_playbook.ethiopia }
				check_variable = { africa_host_playbook = constant:africa_host_playbook.egypt }
				check_variable = { africa_host_playbook = constant:africa_host_playbook.morocco }
				check_variable = { africa_host_playbook = constant:africa_host_playbook.basutoland }
				check_variable = { africa_host_playbook = constant:africa_host_playbook.swaziland }
				check_variable = { africa_host_playbook = constant:africa_host_playbook.zanzibar }
			}
		}
		set_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.monarchical_or_restored }
		africa_achievement_record_host_profile = yes
	}
	if = {
		limit = {
			OR = {
				capital_scope = { is_coastal = yes }
				check_variable = { africa_regional_overlay = constant:africa_overlay.swahili_indian_ocean }
				check_variable = { africa_regional_overlay = constant:africa_overlay.madagascar_islands }
			}
		}
		set_variable = { africa_achievement_host_profile_id = constant:africa_achievement_host_profile.coastal }
		africa_achievement_record_host_profile = yes
	}
	clear_variable = africa_achievement_host_profile_id
}

# Any authoritative settlement scope. The caller sets
# africa_achievement_milestone_id only after the named live result is final.
africa_achievement_record_milestone = {
	if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.nile_overlaps_settled } }
		set_global_flag = africa_achievement_nile_overlaps_settled
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.nile_corridor_operational } }
		set_global_flag = africa_achievement_nile_corridor_operational
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.savanna_overlaps_settled } }
		set_global_flag = africa_achievement_savanna_overlap_settled
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.savanna_courts_same_order } }
		set_global_flag = africa_achievement_savanna_courts_same_order
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.nonhuman_great_power_war_won } }
		set_global_flag = africa_achievement_nonhuman_great_power_war_won
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.priority_member_alternative_recognised } }
		set_global_flag = africa_achievement_recognised_african_confederation
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.stoneborn_constitutional_status } }
		set_global_flag = africa_achievement_stoneborn_constitutional_status
		set_temp_variable = { africa_achievement_restoration_identity_id = constant:africa_achievement_restoration_identity.stoneborn }
		africa_achievement_record_restoration_identity = yes
	}
	else_if = {
		limit = { check_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.weather_war_won } }
		set_global_flag = africa_achievement_weather_war_won
	}
	africa_achievement_refresh_host_duration_windows = yes
}

# COUNTRY scope on a priority member after the bounded rival-bloc victory
# transaction. The member's refusal and player-origin flags are required before
# the host receives the recognised-confederation milestone; a relationship or
# cosmetic snapshot alone is never sufficient.
africa_achievement_record_recognised_rival_confederation = {
	if = {
		limit = {
			has_country_flag = africa_priority_member_player_origin_validated
			has_country_flag = africa_priority_member_league_offer_refused
			has_country_flag = africa_priority_member_rival_bloc_victory
			africa_relationship_is_rival_bloc = yes
			has_event_target = africa_host
		}
		event_target:africa_host = {
			set_global_flag = africa_achievement_member_refusal_owner_ready
			set_temp_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.priority_member_alternative_recognised }
			africa_achievement_record_milestone = yes
		}
	}
}

# COUNTRY scope on the host after both priority-member corridor settlements
# are live. Aksum's heritage consent and Harar's diversified guarantees are
# the two existing owner outcomes; this helper only joins those outcomes and
# never infers a corridor from a package name or regional overlay.
africa_achievement_record_horn_corridor_operational = {
	if = {
		limit = { NOT = { has_global_flag = africa_achievement_horn_corridors_operational } }
		set_temp_variable = { africa_achievement_horn_aksum_ready = constant:africa_value.zero }
		set_temp_variable = { africa_achievement_horn_harar_ready = constant:africa_value.zero }
		for_each_scope_loop = {
			array = africa_relationship_countries
			if = {
				limit = {
					exists = yes
					NOT = { has_capitulated = yes }
					africa_member_host_generation_is_current = yes
					has_country_flag = africa_priority_member_politics_settled
				}
				if = {
					limit = {
						africa_priority_member_package_is_aksum = yes
						has_country_flag = africa_priority_aksum_heritage_consent_separated
					}
					add_to_temp_variable = { africa_achievement_horn_aksum_ready = constant:africa_value.one }
				}
				if = {
					limit = {
						africa_priority_member_package_is_harar = yes
						has_country_flag = africa_priority_harar_corridor_guarantees_diversified
					}
					add_to_temp_variable = { africa_achievement_horn_harar_ready = constant:africa_value.one }
				}
			}
		}
		if = {
			limit = {
				check_variable = { var = africa_achievement_horn_aksum_ready value = constant:africa_value.one compare = greater_than_or_equals }
				check_variable = { var = africa_achievement_horn_harar_ready value = constant:africa_value.one compare = greater_than_or_equals }
			}
			set_global_flag = africa_achievement_horn_corridors_operational
		}
		africa_achievement_refresh_survival_windows = yes
	}
}

# COUNTRY scope on an action target immediately after the immutable active
# action record is created and before the mission begins.
africa_achievement_record_action_start = {
	if = {
		limit = {
			check_variable = { africa_active_action_id = constant:africa_action.open_aid_corridor }
			africa_is_valid_relationship_target = yes
			africa_member_host_generation_is_current = yes
			africa_relationship_is_cooperative = yes
			surrender_progress > constant:africa_achievement_ratio.last_convoy_surrender_progress
			has_war = yes
			NOT = { has_capitulated = yes }
		}
		set_country_flag = africa_achievement_last_convoy_candidate
		set_country_flag = africa_achievement_last_convoy_war_required
		set_variable = { global.africa_achievement_last_convoy_target_id = THIS.id }
		set_variable = { global.africa_achievement_last_convoy_start_day = global.num_days }
		set_variable = { global.africa_achievement_last_convoy_surrender_snapshot = surrender_progress }
		clr_global_flag = africa_achievement_last_convoy_completed
	}
	if = {
		limit = { check_variable = { africa_active_action_id = constant:africa_action.monitor_rival_bloc } }
		set_country_flag = africa_achievement_rival_crisis_started
		set_variable = { africa_achievement_rival_crisis_start_day = global.num_days }
	}
	if = {
		limit = { check_variable = { africa_active_action_id = constant:africa_action.prepare_member_exit_terms } }
		set_country_flag = africa_achievement_peaceful_exit_started
		set_variable = { africa_achievement_peaceful_exit_start_day = global.num_days }
		set_variable = { global.africa_achievement_peaceful_exit_target_id = THIS.id }
	}
}

			if = { limit = { has_country_flag = africa_action_offer_development_charter_full } add_to_temp_variable = { africa_achievement_member_clause_count = constant:africa_value.one } }

# HOST scope. Rebuild the current-generation medium-confidence regional proof
# from the relationship registry. This is a live ledger observation, not an
# action-count proxy: a region enters only when a cooperative member carrying
# that overlay is still alive and at least medium confidence.
africa_achievement_refresh_medium_confidence_regions = {
	clear_array = global.africa_achievement_medium_confidence_regions
	for_each_scope_loop = {
		array = africa_relationship_countries
		if = {
			limit = {
				exists = yes
				NOT = { has_capitulated = yes }
				NOT = { has_country_flag = africa_achievement_exclude_from_live_region_proof }
				africa_member_host_generation_is_current = yes
				has_variable = africa_regional_overlay
				check_variable = { africa_regional_overlay > constant:africa_overlay.none }
				NOT = { check_variable = { africa_regional_overlay > constant:africa_overlay.madagascar_islands } }
				has_variable = africa_member_confidence
				check_variable = {
					var = africa_member_confidence
					value = constant:africa_measure.medium
					compare = greater_than_or_equals
				}
				OR = {
					africa_relationship_is_protected = yes
					africa_relationship_is_associate = yes
					africa_relationship_is_chartered = yes
					africa_relationship_is_autonomous_federal = yes
					africa_relationship_is_integrated_region = yes
				}
			}
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = global.africa_achievement_medium_confidence_regions
							value = africa_regional_overlay
						}
					}
				}
				add_to_array = {
					array = global.africa_achievement_medium_confidence_regions
					value = africa_regional_overlay
				}
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = global.africa_achievement_medium_confidence_regions^num
				value = constant:africa_achievement_count.continental_regions
				compare = greater_than_or_equals
			}
		}
		set_global_flag = africa_achievement_all_regions_medium_confidence
	}
	else = {
		clr_global_flag = africa_achievement_all_regions_medium_confidence
	}
}

	set_temp_variable = { africa_achievement_previous_represented_regions = global.africa_achievement_represented_regions^num }

# reset, not a disconnected-component disqualifier; the next bounded rebuild
# can start a fresh corridor clock after the missing physical mode returns.

	if = {
		limit = { NOT = { has_global_flag = africa_achievement_corridor_access_completed } }
		set_variable = { global.africa_achievement_corridor_access_start_day = constant:africa_value.invalid }
		set_variable = { global.africa_achievement_corridor_access_deadline_day = constant:africa_achievement_days.unset_deadline }
	}
}

# HOST scope during a full achievement reset. Clear the state-local pending
# witness before clearing its global array so a recycled field cannot suppress
# a later Event 018 nationalisation or make an empty ledger look cured.
africa_achievement_clear_pending_forced_resource_fields = {
	for_each_scope_loop = {
		array = global.africa_achievement_pending_forced_resource_fields
		if = {
			limit = { exists = yes }
			clr_state_flag = africa_achievement_forced_resource_pending
		}
	}
	clear_array = global.africa_achievement_pending_forced_resource_fields
	clr_global_flag = africa_achievement_forced_resource_seizure
}

# HOST scope after an exact Event 018 compensation settlement. The DQ follows
# the pending field ledger: it remains while any nationalised field lacks its
# own compensation settlement and clears only when the last entry is removed.
# No unrelated settlement path calls this refresh.
africa_achievement_refresh_forced_resource_seizure_ledger = {
	if = {
		limit = {
			check_variable = {
				var = global.africa_achievement_pending_forced_resource_fields^num
				value = constant:africa_value.zero
				compare = greater_than
			}
		}
		set_global_flag = africa_achievement_forced_resource_seizure
	}
	else = {
		clr_global_flag = africa_achievement_forced_resource_seizure
	}
}
