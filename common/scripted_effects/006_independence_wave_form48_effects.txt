# Member-country scope with ROOT as the HBX/PFX carrier. Directional ownership
# flags ensure cleanup removes only relations created by this transaction.
independence_wave_form48_install_autonomous_member = {
	save_event_target_as = independence_wave_form48_autonomous_member_target
	set_country_flag = independence_wave_form48_autonomous_member
	set_variable = { independence_wave_form48_autonomous_family = ROOT.independence_wave_formable_family }
	set_variable = { independence_wave_form48_autonomous_carrier = ROOT }
	set_variable = { independence_wave_form48_autonomous_carrier_generation = ROOT.independence_wave_generation_id }
	if = {
		limit = { NOT = { gives_military_access_to = ROOT } }
		give_military_access = ROOT
		set_country_flag = independence_wave_form48_member_granted_access_to_carrier
	}
	if = {
		limit = { NOT = { has_guaranteed = ROOT } }
		give_guarantee = ROOT
		set_country_flag = independence_wave_form48_member_guaranteed_carrier
	}
	ROOT = {
		if = {
			limit = { NOT = { gives_military_access_to = event_target:independence_wave_form48_autonomous_member_target } }
			give_military_access = event_target:independence_wave_form48_autonomous_member_target
			event_target:independence_wave_form48_autonomous_member_target = { set_country_flag = independence_wave_form48_carrier_granted_access_to_member }
		}
		if = {
			limit = { NOT = { has_guaranteed = event_target:independence_wave_form48_autonomous_member_target } }
			give_guarantee = event_target:independence_wave_form48_autonomous_member_target
			event_target:independence_wave_form48_autonomous_member_target = { set_country_flag = independence_wave_form48_carrier_guaranteed_member }
		}
		set_country_flag = independence_wave_form48_has_autonomous_members
	}
}

independence_wave_formable_integration_adapter_48 = {
	if = {
		limit = {
			has_country_flag = independence_wave_formable_mutation_prevalidated
			has_country_flag = independence_wave_formable_identity_committed
			has_cosmetic_tag = PFX
			independence_wave_formable_member_arrays_are_aligned = yes
		}
		set_temp_variable = { independence_wave_form48_member_index = constant:independence_wave_formable_tuning.first_array_index }
		set_temp_variable = { independence_wave_form48_additional_members_processed = constant:independence_wave_formable_tuning.no_entries }
		set_temp_variable = { independence_wave_form48_required_additional_members = independence_wave_formable_minimum_members }
		subtract_from_temp_variable = { independence_wave_form48_required_additional_members = constant:independence_wave_formable_tuning.loop_increment }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_form48_member_index value = independence_wave_formable_member_country_entries^num compare = less_than } }
			set_temp_variable = { independence_wave_form48_member_consent = independence_wave_formable_member_consent_entries^independence_wave_form48_member_index }
			var:independence_wave_formable_member_country_entries^independence_wave_form48_member_index = {
				if = {
					limit = {
						NOT = { tag = ROOT }
						is_independence_wave_form48_eligible_member = yes
						OR = { original_tag = HAW original_tag = FSM }
						check_variable = { independence_wave_form48_member_consent = constant:independence_wave_formable_consent.consenting_member }
					}
					independence_wave_form48_install_autonomous_member = yes
					add_to_temp_variable = { independence_wave_form48_additional_members_processed = constant:independence_wave_formable_tuning.loop_increment }
				}
			}
			add_to_temp_variable = { independence_wave_form48_member_index = constant:independence_wave_formable_tuning.loop_increment }
		}
		if = {
			limit = {
				check_variable = { var = independence_wave_form48_additional_members_processed value = independence_wave_form48_required_additional_members compare = equals }
				HAW = {
					has_country_flag = independence_wave_form48_autonomous_member
					has_variable = independence_wave_form48_autonomous_carrier
					has_variable = independence_wave_form48_autonomous_carrier_generation
					has_variable = independence_wave_form48_autonomous_family
					var:independence_wave_form48_autonomous_carrier = { tag = ROOT }
					check_variable = { var = independence_wave_form48_autonomous_carrier_generation value = ROOT.independence_wave_generation_id compare = equals }
					check_variable = { var = independence_wave_form48_autonomous_family value = ROOT.independence_wave_formable_family compare = equals }
				}
				FSM = {
					has_country_flag = independence_wave_form48_autonomous_member
					has_variable = independence_wave_form48_autonomous_carrier
					has_variable = independence_wave_form48_autonomous_carrier_generation
					has_variable = independence_wave_form48_autonomous_family
					var:independence_wave_form48_autonomous_carrier = { tag = ROOT }
					check_variable = { var = independence_wave_form48_autonomous_carrier_generation value = ROOT.independence_wave_generation_id compare = equals }
					check_variable = { var = independence_wave_form48_autonomous_family value = ROOT.independence_wave_formable_family compare = equals }
				}
			}
			set_country_flag = independence_wave_form48_all_autonomous_members_bound
			set_country_flag = independence_wave_formable_integration_committed
		}
	}
}
