# ============================================================================
# CHAOS REDUX - EVENT 020 SHARED RESPONSE EFFECTS
# ============================================================================
# Bounded state and country actions consumed by the shared disease category.
# Every state action stores its owner and action id, pays equipment/fuel at a
# population-scaled rate, occupies response capacity, and resolves into a
# distinct disease, logistics, recovery, knowledge, or anti-rat result.
# ============================================================================

# STATE scope; ROOT is the responding country. The action id is stored on ROOT
# as black_plague_shared_action_pending by the decision complete effect.
black_plague_begin_shared_state_action = {
	if = {
	limit = {
			OR = {
				black_plague_shared_state_can_start_action = yes
				black_plague_shared_royal_node_state_can_start_action = yes
				black_plague_shared_crown_target_is_valid = yes
			}
			ROOT = { has_variable = black_plague_shared_action_pending }
		}
		set_temp_variable = { black_plague_shared_action_scale = constant:black_plague_shared_scale.minimum }
		if = {
			limit = {
				check_variable = {
					var = state_population_k
					value = constant:black_plague_shared_scale.metropolitan_population_k
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { black_plague_shared_action_scale = constant:black_plague_shared_scale.metropolitan }
		}
		else_if = {
			limit = {
				check_variable = {
					var = state_population_k
					value = constant:black_plague_shared_scale.large_population_k
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { black_plague_shared_action_scale = constant:black_plague_shared_scale.large }
		}
		set_variable = { black_plague_shared_action_scale = black_plague_shared_action_scale }
		set_variable = { black_plague_shared_action = ROOT.black_plague_shared_action_pending }
		set_variable = { black_plague_shared_action_owner = ROOT }
		set_state_flag = black_plague_shared_action_active
		set_variable = { black_plague_shared_action_burden = constant:black_plague_shared_burden.medium }
		if = {
			limit = {
				OR = {
					check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.seal_all_transport compare = equals }
					check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.rebuild_hospitals_transport compare = equals }
					check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.liberate_and_quarantine compare = equals }
				}
			}
			set_variable = { black_plague_shared_action_burden = constant:black_plague_shared_burden.heavy }
		}
		if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.strike_the_crown compare = equals } }
			set_variable = { black_plague_shared_action_burden = constant:black_plague_response_burden.crown_strike }
		}
		if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.seal_royal_burrows compare = equals } }
			set_variable = { black_plague_shared_action_burden = constant:black_plague_shared_result.royal_seal_response_burden }
		}
		ROOT = {
			if = {
				limit = { check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.strike_royal_node compare = equals } }
				black_plague_reset_response_payment = yes
				set_temp_variable = { black_plague_response_payment_support = constant:black_plague_shared_cost.country_support_equipment }
				set_temp_variable = { black_plague_response_payment_motorized = constant:black_plague_shared_cost.country_motorized_equipment }
				set_temp_variable = { black_plague_response_payment_fuel = constant:black_plague_shared_cost.country_fuel }
				set_temp_variable = { black_plague_response_payment_infantry = constant:black_plague_shared_cost.country_infantry_equipment }
				set_temp_variable = { black_plague_response_payment_manpower = constant:black_plague_shared_cost.country_manpower }
				set_temp_variable = { black_plague_response_payment_command = constant:black_plague_shared_cost.country_command_power }
				black_plague_pay_response_cost = yes
			}
			else_if = {
				limit = { check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.seal_royal_burrows compare = equals } }
				black_plague_reset_response_payment = yes
				set_temp_variable = { black_plague_response_payment_support = constant:black_plague_shared_result.royal_seal_support_equipment }
				set_temp_variable = { black_plague_response_payment_motorized = constant:black_plague_shared_result.royal_seal_motorized_equipment }
				set_temp_variable = { black_plague_response_payment_infantry = constant:black_plague_shared_result.royal_seal_infantry_equipment }
				set_temp_variable = { black_plague_response_payment_manpower = constant:black_plague_shared_result.royal_seal_manpower }
				set_temp_variable = { black_plague_response_payment_fuel = constant:black_plague_shared_result.royal_seal_fuel }
				set_temp_variable = { black_plague_response_payment_command = constant:black_plague_shared_result.royal_seal_command_power }
				black_plague_pay_response_cost = yes
			}
			else_if = {
				limit = { check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.strike_the_crown compare = equals } }
				black_plague_reset_response_payment = yes
				set_temp_variable = { black_plague_response_payment_support = constant:black_plague_shared_result.crown_strike_support_equipment }
				set_temp_variable = { black_plague_response_payment_motorized = constant:black_plague_shared_result.crown_strike_motorized_equipment }
				set_temp_variable = { black_plague_response_payment_infantry = constant:black_plague_shared_result.crown_strike_infantry_equipment }
				set_temp_variable = { black_plague_response_payment_manpower = constant:black_plague_shared_result.crown_strike_manpower }
				set_temp_variable = { black_plague_response_payment_fuel = constant:black_plague_shared_result.crown_strike_fuel }
				set_temp_variable = { black_plague_response_payment_command = constant:black_plague_shared_result.crown_strike_command_power }
				black_plague_pay_response_cost = yes
			}
			else = {
				set_variable = { black_plague_shared_payment_support = PREV.black_plague_shared_action_scale }
				multiply_variable = { black_plague_shared_payment_support = constant:black_plague_shared_cost.standard_support_equipment }
				set_variable = { black_plague_shared_payment_motorized = PREV.black_plague_shared_action_scale }
				multiply_variable = { black_plague_shared_payment_motorized = constant:black_plague_shared_cost.standard_motorized_equipment }
				set_variable = { black_plague_shared_payment_fuel = PREV.black_plague_shared_action_scale }
				multiply_variable = { black_plague_shared_payment_fuel = constant:black_plague_shared_cost.standard_fuel }
				set_temp_variable = { equipment_stockpile_removal_amount = var:black_plague_shared_payment_support }
				remove_support_equipment_from_stockpile = yes
				set_temp_variable = { equipment_stockpile_removal_amount = var:black_plague_shared_payment_motorized }
				remove_motorized_equipment_from_stockpile = yes
				set_temp_variable = { fuel_stockpile_removal_amount = black_plague_shared_payment_fuel }
				remove_fuel_from_stockpile = yes
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.port_inspections compare = equals }
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.relief_corridor compare = equals }
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.evacuate_threatened_perimeter compare = equals }
					}
				}
				set_temp_variable = { equipment_stockpile_removal_amount = constant:black_plague_shared_cost.standard_trains }
				remove_trains_from_stockpile = yes
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.restrict_troop_routes compare = equals }
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.seal_all_transport compare = equals }
					}
				}
				black_plague_reset_response_payment = yes
				if = {
					limit = { check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.restrict_troop_routes compare = equals } }
					set_temp_variable = { black_plague_response_payment_command = constant:black_plague_shared_cost.standard_command_power }
				}
				else = {
					set_temp_variable = { black_plague_response_payment_command = constant:black_plague_shared_cost.country_command_power }
				}
				black_plague_pay_response_cost = yes
			}
			clear_variable = black_plague_shared_payment_support
			clear_variable = black_plague_shared_payment_motorized
			clear_variable = black_plague_shared_payment_fuel
			black_plague_register_current_response_state = yes
			black_plague_recompute_country_response_capacity = yes
			clear_variable = black_plague_shared_action_pending
		}
	}
}

# STATE scope; ROOT is the action owner. This single resolution effect keeps
# each shared action distinct while avoiding duplicate decision categories.
black_plague_finish_shared_state_action = {
	if = {
		limit = {
			has_state_flag = black_plague_shared_action_active
			has_variable = black_plague_shared_action_owner
			check_variable = { var = black_plague_shared_action_owner value = ROOT compare = equals }
		}
		set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.sentinel_surveillance }
		if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.sentinel_surveillance compare = equals } }
			ROOT = {
				set_temp_variable = { black_plague_shared_sentinel_days = constant:black_plague_shared_duration.sentinel_surveillance }
				set_country_flag = { flag = black_plague_shared_sentinel_surveillance_active days = black_plague_shared_sentinel_days }
				subtract_from_variable = { black_plague_incoming_exposure = constant:black_plague_shared_result.preparedness_gain }
			}
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.sentinel_surveillance }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.inspect_transport_hubs compare = equals } }
			set_state_flag = black_plague_shared_transport_inspections_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.inspect_transport_hubs }
			subtract_from_variable = { black_plague_incoming_exposure = constant:black_plague_shared_result.transport_spread_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.stronger_prevention_law compare = equals } }
			ROOT = {
				set_country_flag = black_plague_shared_prevention_law_active
				subtract_from_variable = { black_plague_incoming_exposure = constant:black_plague_shared_result.preparedness_gain }
				add_stability = constant:black_plague_shared_cost.stability_loss
			}
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.stronger_prevention_law }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.close_threatened_corridor compare = equals } }
			set_state_flag = black_plague_shared_threatened_corridor_active

		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.port_inspections compare = equals } }
			set_state_flag = black_plague_shared_port_inspections_active

			subtract_from_variable = { black_plague_spread_pressure = constant:black_plague_shared_result.port_spread_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.restrict_troop_routes compare = equals } }
			set_state_flag = black_plague_shared_troop_restrictions_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.restrict_troop_routes }
			subtract_from_variable = { black_plague_spread_pressure = constant:black_plague_shared_result.troop_spread_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.preposition_field_hospitals compare = equals } }
			set_state_flag = black_plague_shared_field_hospitals_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.preposition_field_hospitals }
			add_to_variable = { black_plague_treatment_coverage = constant:black_plague_shared_result.field_hospital_treatment }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.suspend_civilian_travel compare = equals } }
			set_state_flag = black_plague_shared_civilian_travel_suspended
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.suspend_civilian_travel }
			subtract_from_variable = { black_plague_spread_pressure = constant:black_plague_shared_result.civilian_travel_spread_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.relief_corridor compare = equals } }
			set_state_flag = black_plague_shared_relief_corridor_active

			add_to_variable = { black_plague_treatment_coverage = constant:black_plague_shared_result.relief_treatment }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.burial_sanitation compare = equals } }
			set_state_flag = black_plague_shared_burial_sanitation_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.burial_sanitation }
			subtract_from_variable = { black_plague_load = constant:black_plague_shared_result.burial_load_reduction }
			subtract_from_variable = { black_plague_mortality_pressure = constant:black_plague_shared_result.burial_mortality_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.seal_all_transport compare = equals } }
			set_state_flag = black_plague_shared_transport_sealed
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.seal_all_transport }
			subtract_from_variable = { black_plague_spread_pressure = constant:black_plague_shared_result.seal_transport_spread_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.evacuate_threatened_perimeter compare = equals } }

			set_state_flag = black_plague_shared_foreign_medical_mission_active

			ROOT = {

		}

			add_to_variable = { black_plague_containment = constant:black_plague_shared_result.containment_gain }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.controlled_reopening compare = equals } }
			set_state_flag = black_plague_shared_controlled_reopening_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.controlled_reopening }
			subtract_from_variable = { black_plague_containment = constant:black_plague_shared_result.reopening_containment_loss }
			clr_state_flag = black_plague_quarantine_monitoring
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.trace_residual_clusters compare = equals } }
			set_state_flag = black_plague_shared_residual_tracing_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.trace_residual_clusters }
			subtract_from_variable = { black_plague_load = constant:black_plague_shared_result.residual_load_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.rotate_exhausted_personnel compare = equals } }
			set_state_flag = black_plague_shared_personnel_rotation_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.rotate_exhausted_personnel }
			add_to_variable = { black_plague_containment = constant:black_plague_shared_result.personnel_containment_gain }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.restore_local_administration compare = equals } }
			set_state_flag = black_plague_shared_administration_restored

		else_if = {

			set_state_flag = black_plague_shared_armored_clearance_complete
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.armored_clearance }

		}

			set_state_flag = black_plague_shared_air_reconnaissance_complete
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.air_reconnaissance_sweep }
			subtract_from_variable = { black_plague_incoming_exposure = constant:black_plague_shared_result.evacuation_exposure_reduction }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.liberate_and_quarantine compare = equals } }
			set_state_flag = black_plague_shared_liberation_quarantine_active
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.liberate_and_quarantine }
			add_to_variable = { black_plague_containment = constant:black_plague_shared_result.anti_rat_containment }
			set_temp_variable = { black_plague_exposure_provenance = constant:black_plague_provenance.rat_occupation }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.seal_royal_burrows compare = equals } }
			set_temp_variable = { black_plague_shared_cooldown_days = constant:black_plague_shared_duration.seal_royal_burrows }

# COUNTRY scope. Knowledge actions retain the shared program and use real
# equipment/fuel costs; they never create hidden political-power stores.
black_plague_begin_shared_country_action = {
	if = {
		limit = {
			black_plague_shared_country_knowledge_visible = yes
			has_variable = black_plague_shared_action_pending
		}
		set_variable = { black_plague_shared_country_action = black_plague_shared_action_pending }
		set_country_flag = black_plague_shared_country_action_active
		set_temp_variable = { equipment_stockpile_removal_amount = constant:black_plague_shared_cost.country_support_equipment }
		remove_support_equipment_from_stockpile = yes
		set_temp_variable = { equipment_stockpile_removal_amount = constant:black_plague_shared_cost.country_motorized_equipment }
		remove_motorized_equipment_from_stockpile = yes
		set_temp_variable = { fuel_stockpile_removal_amount = constant:black_plague_shared_cost.country_fuel }
		remove_fuel_from_stockpile = yes
		black_plague_register_response_country = yes
		clear_variable = black_plague_shared_action_pending
	}
}
