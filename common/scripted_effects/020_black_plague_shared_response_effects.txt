# ============================================================================
# CHAOS REDUX - EVENT 020 SHARED RESPONSE EFFECTS
# ============================================================================
# Bounded state and country actions consumed by the shared disease category.
# Every state action stores its owner and action id, pays equipment/fuel at a
# population-scaled rate, occupies response capacity, and resolves into a
# distinct disease, logistics, recovery, knowledge, or anti-rat result.
# ============================================================================

# STATE scope; ROOT is the responding country. The action id is stored on ROOT
# as black_plague_shared_action_pending by the decision complete effect.
black_plague_begin_shared_state_action = {
	if = {
	limit = {
			black_plague_shared_state_can_start_action = yes
			ROOT = { has_variable = black_plague_shared_action_pending }
		}
		set_temp_variable = { black_plague_shared_action_scale = constant:black_plague_shared_scale.minimum }
		if = {
			limit = {
				check_variable = {
					var = state_population_k
					value = constant:black_plague_shared_scale.metropolitan_population_k
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { black_plague_shared_action_scale = constant:black_plague_shared_scale.metropolitan }
		}
		else_if = {
			limit = {
				check_variable = {
					var = state_population_k
					value = constant:black_plague_shared_scale.large_population_k
					compare = greater_than_or_equals
				}
			}
			set_temp_variable = { black_plague_shared_action_scale = constant:black_plague_shared_scale.large }
		}
		set_variable = { black_plague_shared_action_scale = black_plague_shared_action_scale }
		set_variable = { black_plague_shared_action = ROOT.black_plague_shared_action_pending }
		set_variable = { black_plague_shared_action_owner = ROOT }
		set_state_flag = black_plague_shared_action_active
		set_variable = { black_plague_shared_action_burden = constant:black_plague_shared_burden.medium }
		if = {
			limit = {
				OR = {
					check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.seal_all_transport compare = equals }
					check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.rebuild_hospitals_transport compare = equals }
					check_variable = { var = black_plague_shared_action value = constant:black_plague_shared_action.liberate_and_quarantine compare = equals }
				}
			}
			set_variable = { black_plague_shared_action_burden = constant:black_plague_shared_burden.heavy }
		}
		ROOT = {
			set_variable = { black_plague_shared_payment_support = PREV.black_plague_shared_action_scale }
			multiply_variable = { black_plague_shared_payment_support = constant:black_plague_shared_cost.standard_support_equipment }
			set_variable = { black_plague_shared_payment_motorized = PREV.black_plague_shared_action_scale }
			multiply_variable = { black_plague_shared_payment_motorized = constant:black_plague_shared_cost.standard_motorized_equipment }
			set_variable = { black_plague_shared_payment_fuel = PREV.black_plague_shared_action_scale }
			multiply_variable = { black_plague_shared_payment_fuel = constant:black_plague_shared_cost.standard_fuel }
			remove_equipment_to_stockpile = { type = support_equipment amount = var:black_plague_shared_payment_support }
			remove_equipment_to_stockpile = { type = motorized_equipment amount = var:black_plague_shared_payment_motorized }
			subtract_from_fuel = black_plague_shared_payment_fuel
			if = {
				limit = {
					OR = {
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.port_inspections compare = equals }
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.relief_corridor compare = equals }
						check_variable = { var = PREV.black_plague_shared_action value = constant:black_plague_shared_action.evacuate_threatened_perimeter compare = equals }
					}
				}
				remove_equipment_to_stockpile = { type = train_equipment amount = constant:black_plague_shared_cost.standard_trains }
			}
			clear_variable = black_plague_shared_payment_support
			clear_variable = black_plague_shared_payment_motorized
			clear_variable = black_plague_shared_payment_fuel
			black_plague_register_current_response_state = yes
			black_plague_recompute_country_response_capacity = yes
			clear_variable = black_plague_shared_action_pending
		}
	}
}

# STATE scope; ROOT is the action owner. This single resolution effect keeps
# each shared action distinct while avoiding duplicate decision categories.

			has_state_flag = black_plague_shared_action_active

				set_temp_variable = { black_plague_shared_sentinel_days = constant:black_plague_shared_duration.sentinel_surveillance }

				subtract_from_variable = { black_plague_incoming_exposure = constant:black_plague_shared_result.preparedness_gain }

# STATE scope. Cancels a pending shared action without applying its outcome.
# ROOT is the action owner, so capacity and the decision's pending variable are
# cleared atomically before the next board refresh.
black_plague_cancel_shared_state_action = {
	if = {
		limit = {
			has_state_flag = black_plague_shared_action_active
			has_variable = black_plague_shared_action_owner
			check_variable = { var = black_plague_shared_action_owner value = ROOT compare = equals }
		}
		clr_state_flag = black_plague_shared_action_active
		clear_variable = black_plague_shared_action
		clear_variable = black_plague_shared_action_owner
		clear_variable = black_plague_shared_action_scale
		clear_variable = black_plague_shared_action_burden
		set_temp_variable = { black_plague_shared_cancel_cooldown_days = constant:black_plague_shared_duration.close_threatened_corridor }
		set_state_flag = { flag = black_plague_shared_action_cooldown days = black_plague_shared_cancel_cooldown_days }
		ROOT = {
			clear_variable = black_plague_shared_action_pending
			black_plague_recompute_country_response_capacity = yes
		}
	}
}

		add_to_temp_variable = { black_plague_response_containment_gain = constant:black_plague_shared_result.administration_recovery_gain }
	}
	if = {
		limit = { has_state_flag = black_plague_shared_hospitals_transport_rebuilt }
		add_to_temp_variable = { black_plague_response_treatment_gain = constant:black_plague_shared_result.rebuild_treatment_gain }

	if = {
		limit = { has_state_flag = black_plague_shared_households_supported }
		add_to_temp_variable = { black_plague_response_mortality_reduction = constant:black_plague_shared_result.household_mortality_reduction }

# COUNTRY scope. Knowledge actions retain the shared program and use real
# equipment/fuel costs; they never create hidden political-power stores.
black_plague_begin_shared_country_action = {
	if = {
		limit = {
			black_plague_shared_country_knowledge_visible = yes
			has_variable = black_plague_shared_action_pending
		}
		set_variable = { black_plague_shared_country_action = black_plague_shared_action_pending }
		set_country_flag = black_plague_shared_country_action_active
		remove_equipment_to_stockpile = { type = support_equipment amount = constant:black_plague_shared_cost.country_support_equipment }
		remove_equipment_to_stockpile = { type = motorized_equipment amount = constant:black_plague_shared_cost.country_motorized_equipment }
		subtract_from_fuel = constant:black_plague_shared_cost.country_fuel
		black_plague_register_response_country = yes
		clear_variable = black_plague_shared_action_pending
	}
}

black_plague_finish_shared_country_action = {
	if = {
		limit = { has_country_flag = black_plague_shared_country_action_active }
		if = {
			limit = { check_variable = { var = black_plague_shared_country_action value = constant:black_plague_shared_action.publish_findings compare = equals } }
			set_country_flag = black_plague_shared_findings_published
			set_temp_variable = { black_plague_countermeasure_gain = constant:black_plague_shared_result.findings_progress }
			black_plague_record_countermeasure_findings = yes
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_country_action value = constant:black_plague_shared_action.restricted_alliance_exchange compare = equals } }
			set_temp_variable = { black_plague_alliance_exchange_days = constant:black_plague_shared_duration.restricted_alliance_exchange }
			set_country_flag = { flag = black_plague_shared_alliance_exchange_active days = black_plague_alliance_exchange_days }
			set_temp_variable = { black_plague_countermeasure_gain = constant:black_plague_shared_result.alliance_progress }
			black_plague_record_countermeasure_findings = yes
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_country_action value = constant:black_plague_shared_action.hoard_protocol compare = equals } }
			set_country_flag = black_plague_shared_protocol_hoarded
			add_stability = constant:black_plague_shared_cost.stability_loss
			set_temp_variable = { black_plague_countermeasure_gain = constant:black_plague_shared_result.hoard_progress }
			black_plague_record_countermeasure_findings = yes
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_country_action value = constant:black_plague_shared_action.steal_foreign_progress compare = equals } }
			set_temp_variable = { black_plague_countermeasure_gain = constant:black_plague_shared_result.steal_progress }
			black_plague_record_countermeasure_findings = yes
			set_temp_variable = { black_plague_steal_cooldown_days = constant:black_plague_shared_duration.steal_foreign_progress }
			set_country_flag = { flag = black_plague_shared_steal_progress_cooldown days = black_plague_steal_cooldown_days }
		}
		else_if = {
			limit = { check_variable = { var = black_plague_shared_country_action value = constant:black_plague_shared_action.international_medical_mission compare = equals } }
			set_temp_variable = { black_plague_international_mission_days = constant:black_plague_shared_duration.international_medical_mission }
			set_country_flag = { flag = black_plague_shared_international_mission_active days = black_plague_international_mission_days }
			set_temp_variable = { black_plague_countermeasure_gain = constant:black_plague_shared_result.international_progress }
			black_plague_record_countermeasure_findings = yes
		}
		clear_variable = black_plague_shared_country_action
		clr_country_flag = black_plague_shared_country_action_active
		black_plague_recompute_country_response_capacity = yes
	}
}

# COUNTRY scope. A knowledge project that loses its parent program is simply
# cancelled; its paid material is not refunded, and the response burden is
# released for the next decision.
black_plague_cancel_shared_country_action = {
	if = {
		limit = { has_country_flag = black_plague_shared_country_action_active }
		clr_country_flag = black_plague_shared_country_action_active
		clear_variable = black_plague_shared_country_action
		clear_variable = black_plague_shared_action_pending
		black_plague_recompute_country_response_capacity = yes
	}
}

# COUNTRY scope. Terminal cleanup is deliberately explicit: shared response
# flags are state-scoped and must not survive a cured episode or a world-end
# cleanup pass. Existing four-lane response helpers are reused where ownership
# is still valid, then all shared flags are removed from every live record.
black_plague_clear_country_response_runtime = {
	for_each_scope_loop = {
		array = global.black_plague_response_states
		black_plague_clear_shared_state_flags = yes
		if = {
			limit = {
				has_variable = black_plague_response_civil_lane_owner
				check_variable = { var = black_plague_response_civil_lane_owner value = ROOT compare = equals }
			}
			black_plague_clear_current_state_civil_lane = yes
		}
		if = {
			limit = {
				has_variable = black_plague_response_quarantine_lane_owner
				check_variable = { var = black_plague_response_quarantine_lane_owner value = ROOT compare = equals }
			}
			black_plague_clear_current_state_quarantine_lane = yes
		}
		if = {
			limit = {
				has_variable = black_plague_response_cordon_lane_owner
				check_variable = { var = black_plague_response_cordon_lane_owner value = ROOT compare = equals }
			}
			black_plague_clear_current_state_cordon_lane = yes
		}
		if = {
			limit = {
				has_variable = black_plague_response_treatment_lane_owner
				check_variable = { var = black_plague_response_treatment_lane_owner value = ROOT compare = equals }
			}
			black_plague_clear_current_state_treatment_lane = yes
		}
		if = {
			limit = {
				has_variable = black_plague_response_persistent_owner
				check_variable = { var = black_plague_response_persistent_owner value = ROOT compare = equals }
			}
			black_plague_clear_current_state_persistent_measures = yes
		}
	}
	clr_country_flag = black_plague_shared_country_action_active
	clr_country_flag = black_plague_shared_findings_published
	clr_country_flag = black_plague_shared_alliance_exchange_active
	clr_country_flag = black_plague_shared_protocol_hoarded
	clr_country_flag = black_plague_shared_steal_progress_cooldown
	clr_country_flag = black_plague_shared_international_mission_active
	clear_variable = black_plague_shared_country_action
	clear_variable = black_plague_shared_action_pending
	black_plague_recompute_country_response_capacity = yes
}
