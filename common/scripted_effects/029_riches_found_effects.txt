# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND SCRIPTED EFFECTS
# ============================================================================
# State-bound mine infrastructure for Event 029. A registered state owns the
# durable mine record, four public values, hidden risk values, lifecycle flags,
# contract/mission history, evolution history, and one state presentation
# modifier. The bounded global mine array is the only registry consumer; no
# recurring world scan is introduced.
#
# Consumer contract:
# - COUNTRY callers use the discovery, selection, and controller aggregate
#   helpers.
# - STATE callers use record, value, lifecycle, cleanup, and evolution helpers.
# - Temporary input values are explicitly named in comments beside helpers.
# - Every mutating state helper refreshes presentation, contribution, and
#   evolution eligibility when a durable value or lifecycle fact changes.
# ============================================================================

# ---------------------------------------------------------------------------
# Registry and discovery preparation
# ---------------------------------------------------------------------------

# COUNTRY scope. Idempotently creates the bounded Event 029 registry. The
# startup hook calls this through one random existing country only so the
# global variables have a valid country scope without a world iteration.
riches_found_initialize_registry = {
	if = {
		limit = { NOT = { has_variable = global.riches_found_registry_version } }
		set_variable = { global.riches_found_registry_version = constant:riches_found_event.registry_version }
		set_variable = { global.riches_found_next_mine_sequence = constant:riches_found_value.one }
		clear_array = global.riches_found_registered_mines
	}
	if = {
		limit = { NOT = { has_variable = global.riches_found_next_mine_sequence } }
		set_variable = { global.riches_found_next_mine_sequence = constant:riches_found_value.one }
	}
}

# COUNTRY scope. Adds this eligible country to the shared temporary pool using
# temporary pool. Event 029 chooses uniformly among eligible ordinary
# countries; crisis load remains a gate and does not become a second weight.
riches_found_add_current_country_to_prefire_pool = {
	add_to_temp_array = { riches_found_country_prefire_pool = THIS }
}

# COUNTRY scope. Selects the state after the country-level pressure roll has
# chosen the exact owner.
riches_found_prepare_selected_discovery_country = {
	save_event_target_as = riches_found_prefire_owner
	if = {
		limit = {
			any_owned_state = {
				is_owned_and_controlled_by = PREV
				riches_found_state_is_preferred_target = yes
			}
		}
		random_list = {
			constant:riches_found_state_selection.strong_weight = {
				random_owned_controlled_state = {
					limit = { riches_found_state_is_preferred_target = yes }
					riches_found_save_prefire_state = yes
				}
			}
			constant:riches_found_state_selection.broad_weight = {
				random_owned_controlled_state = {
					limit = { riches_found_is_valid_discovery_state = yes }
					riches_found_save_prefire_state = yes
				}
			}
		}
	}
	else = {
		random_owned_controlled_state = {
			limit = { riches_found_is_valid_discovery_state = yes }
			riches_found_save_prefire_state = yes
		}
	}
	if = {
		limit = { has_event_target = riches_found_prepared_mine_state }
		set_country_flag = riches_found_discovery_pending
		set_variable = { riches_found_discovery_prepared_date = global.date }
	}
}

# COUNTRY scope. Selects one eligible ordinary country with load-aware tickets,
# then one eligible state in that country. No value or grant changes here.
riches_found_prepare_random_event_fire = {
	clear_temp_array = riches_found_country_prefire_pool
	every_country = {
		limit = { riches_found_is_eligible_discovery_country = yes }
		riches_found_add_current_country_to_prefire_pool = yes
	}
	if = {
		limit = { check_variable = { riches_found_country_prefire_pool^num > constant:riches_found_value.zero } }
		random_scope_in_array = {
			array = riches_found_country_prefire_pool
			riches_found_prepare_selected_discovery_country = yes
		}
	}
	clear_temp_array = riches_found_country_prefire_pool
	if = {
		limit = {
			has_event_target = riches_found_prefire_owner
			has_event_target = riches_found_prefire_state
		}
		set_temp_variable = { riches_found_prefire_ready = constant:riches_found_value.one }
	}
}

# STATE scope. Saves both state targets used by the shared repeatable-event
# dispatcher. Keeping the two names in one helper prevents a branch from
# exposing a mine to Event 029 without also satisfying the owner/state pair.
riches_found_save_prefire_state = {
	save_event_target_as = riches_found_prepared_mine_state
	save_event_target_as = riches_found_prefire_state
}

# COUNTRY scope. Commits the prepared event target, initializes exactly one
# state record, and pays the one-time 1,000 political-power discovery grant.
# Input: regular event target riches_found_prepared_mine_state. Side effects:
# state registration/initialization and country flag riches_found_discovery_grant_paid.
riches_found_commit_discovery = {
	if = {
		limit = {
			has_event_target = riches_found_prepared_mine_state
			event_target:riches_found_prepared_mine_state = {
				NOT = { has_state_flag = riches_found_discovery_grant_paid }
			}
		}
		save_event_target_as = riches_found_discovery_actor
		event_target:riches_found_prepared_mine_state = {
			riches_found_initialize_mine_record = yes
		}
		if = {
			limit = { event_target:riches_found_prepared_mine_state = { riches_found_mine_is_registered = yes } }
			add_political_power = constant:riches_found_event.discovery_political_power
			event_target:riches_found_prepared_mine_state = { set_state_flag = riches_found_discovery_grant_paid }
			clr_country_flag = riches_found_discovery_pending
			event_target:riches_found_prepared_mine_state = {
				set_variable = { riches_found_discoverer = event_target:riches_found_discovery_actor }
				riches_found_record_achievement_discovery = yes
			}
		}
	}
}

# STATE scope. Registers THIS in the bounded persistent mine array and assigns
# a monotonic history sequence exactly once. The state flag and sequence are
# both required for later registry membership checks.
riches_found_bind_history_sequence = {
	if = {
		limit = { NOT = { has_variable = global.riches_found_next_mine_sequence } }
		set_variable = { global.riches_found_next_mine_sequence = constant:riches_found_value.one }
	}
	if = {
		limit = {
			NOT = { has_variable = riches_found_mine_sequence }
			NOT = {
				check_variable = {
					var = global.riches_found_registered_mines^num
					value = constant:riches_found_event.registry_cap
					compare = greater_than_or_equals
				}
			}
		}
		set_variable = { riches_found_mine_sequence = global.riches_found_next_mine_sequence }
		add_to_variable = { global.riches_found_next_mine_sequence = constant:riches_found_value.one }
	}
	if = {
		limit = {
			has_variable = riches_found_mine_sequence
			NOT = {
				is_in_array = {
					array = global.riches_found_registered_mines
					value = THIS
				}
			}
			NOT = {
				check_variable = {
					var = global.riches_found_registered_mines^num
					value = constant:riches_found_event.registry_cap
					compare = greater_than_or_equals
				}
			}
		}
		add_to_array = { array = global.riches_found_registered_mines value = THIS }
		set_state_flag = riches_found_mine_registered
	}
}

# STATE scope. Initializes one physical mine record without changing map
# resources. The parent event owns discovery wording and event-log wiring.
riches_found_initialize_mine_record = {
	if = {
		limit = { NOT = { has_state_flag = riches_found_mine_initialized } }
		riches_found_bind_history_sequence = yes
	}
	if = {
		limit = {
			has_state_flag = riches_found_mine_registered
			NOT = { has_state_flag = riches_found_mine_initialized }
		}
		save_event_target_as = riches_found_mine_state
		if = {
			limit = { has_event_target = riches_found_discovery_actor }
			set_variable = { riches_found_discoverer = event_target:riches_found_discovery_actor }
		}
		else = {
			if = { limit = { OWNER = { exists = yes } } set_variable = { riches_found_discoverer = OWNER } }
		}

		random_list = {
			constant:riches_found_profile_weight.alluvial = { set_variable = { riches_found_mine_profile = constant:riches_found_profile.alluvial } }
			constant:riches_found_profile_weight.deep_vein = { set_variable = { riches_found_mine_profile = constant:riches_found_profile.deep_vein } }
			constant:riches_found_profile_weight.gemstone = { set_variable = { riches_found_mine_profile = constant:riches_found_profile.gemstone } }
			constant:riches_found_profile_weight.platinum = { set_variable = { riches_found_mine_profile = constant:riches_found_profile.platinum } }
		}
		set_variable = { riches_found_mine_quality_factor = constant:riches_found_scaling.quality_alluvial }
		if = { limit = { check_variable = { var = riches_found_mine_profile value = constant:riches_found_profile.deep_vein compare = equals } } set_variable = { riches_found_mine_quality_factor = constant:riches_found_scaling.quality_deep_vein } }
		else_if = { limit = { check_variable = { var = riches_found_mine_profile value = constant:riches_found_profile.gemstone compare = equals } } set_variable = { riches_found_mine_quality_factor = constant:riches_found_scaling.quality_gemstone } }
		else_if = { limit = { check_variable = { var = riches_found_mine_profile value = constant:riches_found_profile.platinum compare = equals } } set_variable = { riches_found_mine_quality_factor = constant:riches_found_scaling.quality_platinum } }

		set_variable = { riches_found_extraction_pressure = constant:riches_found_value.public_initial_pressure }
		set_variable = { riches_found_mine_development = constant:riches_found_value.public_initial_development }
		set_variable = { riches_found_local_order = constant:riches_found_value.public_initial_order }
		set_variable = { riches_found_revenue_legitimacy = constant:riches_found_value.public_initial_legitimacy }
		set_variable = { riches_found_concession_exposure = constant:riches_found_value.public_initial_concession_exposure }
		set_variable = { riches_found_illicit_capture = constant:riches_found_value.public_initial_illicit_capture }
		set_variable = { riches_found_deep_excavation = constant:riches_found_value.public_initial_deep_excavation }
		set_variable = { riches_found_private_security_autonomy = constant:riches_found_value.public_initial_security_autonomy }
		set_variable = { riches_found_foreign_claim_pressure = constant:riches_found_value.public_initial_foreign_claim_pressure }
		set_variable = { riches_found_gold_disease_pressure = constant:riches_found_value.public_initial_gold_disease }
		set_variable = { riches_found_supernatural_pressure = constant:riches_found_value.public_initial_supernatural_pressure }
		set_variable = { riches_found_sealed_depth_integrity = constant:riches_found_value.public_initial_sealed_integrity }
		set_variable = { riches_found_collapse_pressure = constant:riches_found_value.public_initial_collapse_pressure }
		set_variable = { riches_found_chaos_exposure = constant:riches_found_value.public_initial_chaos_exposure }

		set_variable = { riches_found_operating_state = constant:riches_found_operating_state.discovery }
		set_variable = { riches_found_phase = constant:riches_found_phase.discovery_claims }
		set_variable = { riches_found_policy = constant:riches_found_policy.unassigned }
		set_variable = { riches_found_recorded_owner = OWNER }
		set_variable = { riches_found_recorded_controller = CONTROLLER }
		set_variable = { riches_found_discovery_date = global.date }
		set_variable = { riches_found_last_review_date = global.date }
		set_variable = { riches_found_evolution_track = constant:riches_found_value.zero }
		set_variable = { riches_found_achievement_transfer_count = constant:riches_found_value.zero }
		set_variable = { riches_found_achievement_settlement_count = constant:riches_found_value.zero }
		set_variable = { riches_found_achievement_pay_train_count = constant:riches_found_value.zero }
		set_variable = { riches_found_achievement_mission_success_count = constant:riches_found_value.zero }
		riches_found_initialize_incident_clock = yes
		set_state_flag = riches_found_mine_active
		set_state_flag = riches_found_mine_state
		set_state_flag = riches_found_mine_initialized
		riches_found_register_achievement_hooks = yes
		riches_found_refresh_mine_presentation = yes
		riches_found_refresh_mine_contribution = yes
		riches_found_refresh_evolution_gates = yes
		riches_found_refresh_state_achievement_timers = yes
		if = { limit = { CONTROLLER = { exists = yes } } CONTROLLER = { riches_found_refresh_controller_aggregate = yes } }
	}
}

# STATE scope. Input: temporary `riches_found_report_kind` from the shared
# report-id constants. One pending report is allowed per mine, and the hidden
# state callback below delivers it to the current controller after one day.
# Reports are state-bound, so a transfer cannot duplicate or redirect a report
# through a stale country target.
riches_found_queue_report = {
	if = {
		limit = {
			riches_found_mine_is_registered = yes
			CONTROLLER = { exists = yes }
			NOT = { has_state_flag = riches_found_report_pending }
			has_variable = riches_found_report_kind
		}
		set_state_flag = riches_found_report_pending
		set_state_flag = riches_found_report_scheduled
		set_temp_variable = { riches_found_report_delay_days = constant:riches_found_value.one }
		state_event = { id = chaosx.nr29.910 days = riches_found_report_delay_days trigger_for = controller }
	}
}

# STATE scope. Delivers the one queued report to the current controller. The
# branch is intentionally explicit so every report picture remains a stable,
# auditable event id instead of a free-form dynamic event target.
riches_found_run_queued_report = {
	clr_state_flag = riches_found_report_scheduled
	if = {
		limit = { has_state_flag = riches_found_report_pending has_variable = riches_found_report_kind }
		if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.rush compare = equals } } state_event = { id = chaosx.nr29.10 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.concession compare = equals } } state_event = { id = chaosx.nr29.20 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.raid compare = equals } } state_event = { id = chaosx.nr29.30 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.collapse compare = equals } } state_event = { id = chaosx.nr29.40 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.resource_curse compare = equals } } state_event = { id = chaosx.nr29.101 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.gold_disease compare = equals } } state_event = { id = chaosx.nr29.102 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.opened_depths compare = equals } } state_event = { id = chaosx.nr29.103 trigger_for = controller } }
		else_if = { limit = { check_variable = { var = riches_found_report_kind value = constant:riches_found_report.gilded_sovereignty compare = equals } } state_event = { id = chaosx.nr29.111 trigger_for = controller } }

# STATE scope. Rebuilds compatibility markers from current geography,
# infrastructure, deployed forces, and durable Event 029 choices. These flags
# are derived facts only; they never replace the concrete predicates used by
# decisions and missions.
riches_found_refresh_mine_capabilities = {

# COUNTRY scope. One scheduled event owns the controller pulse. The guard flag
# prevents repeated decisions or transfers from queueing duplicate pulses.
riches_found_schedule_controller_pulse = {
	if = {
		limit = {
			has_country_flag = riches_found_controller_aggregate_active
			NOT = { has_country_flag = riches_found_controller_pulse_scheduled }
		}
		set_country_flag = riches_found_controller_pulse_scheduled
		country_event = { id = chaosx.nr29.900 days = constant:riches_found_duration.controller_pulse_days }
	}
}

# COUNTRY scope. Event 029's only recurring work is bounded by the persistent
# registry and filtered to the current controller. Each mine gets one local
# value refresh and one MTTH-backed evolution check; no whole-world action is
# introduced.
riches_found_run_controller_pulse = {
	clr_country_flag = riches_found_controller_pulse_scheduled
	if = {
		limit = {
			has_country_flag = riches_found_controller_aggregate_active
			has_variable = global.riches_found_registered_mines^num
		}
		for_each_scope_loop = {
			array = global.riches_found_registered_mines
			if = {
				limit = {
					riches_found_mine_is_registered = yes
					is_controlled_by = ROOT
					riches_found_mine_is_active = yes
				}
				riches_found_refresh_contract_lifecycle = yes
				riches_found_run_due_incident_review = yes
				set_temp_variable = { riches_found_pulse_pressure_delta = constant:riches_found_scaling.pulse_pressure_add }
				if = { limit = { ROOT = { has_war = yes } } add_to_temp_variable = { riches_found_pulse_pressure_delta = constant:riches_found_scaling.pulse_war_pressure_add } }
				if = { limit = { riches_found_mine_is_severe_pressure = yes } add_to_temp_variable = { riches_found_pulse_pressure_delta = constant:riches_found_scaling.pulse_severe_pressure_add } }
				add_to_variable = { riches_found_extraction_pressure = riches_found_pulse_pressure_delta }
				set_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_collapse_pressure_add }
				if = { limit = { riches_found_mine_is_local_order_crisis = yes } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_low_order_collapse_add } }
				if = { limit = { check_variable = { var = riches_found_sealed_depth_integrity value = constant:riches_found_threshold.sealed_integrity_gate compare = less_than } } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_low_integrity_collapse_add } }
				if = { limit = { check_variable = { var = riches_found_chaos_exposure value = constant:riches_found_value.zero compare = greater_than } } add_to_temp_variable = { riches_found_pulse_collapse_delta = constant:riches_found_scaling.pulse_chaos_collapse_add } }
				add_to_variable = { riches_found_collapse_pressure = riches_found_pulse_collapse_delta }
				if = { limit = { riches_found_mine_is_high_pressure = yes } set_state_flag = riches_found_collapse_warning }
				if = {
					limit = {
						OR = {
							riches_found_mine_is_severe_pressure = yes
							riches_found_mine_is_local_order_crisis = yes
							riches_found_mine_is_legitimacy_crisis = yes
						}
					}
					set_state_flag = riches_found_crisis_active
				}
				if = { limit = { riches_found_mine_is_severe_pressure = yes } set_state_flag = riches_found_raid_crisis }
				if = { limit = { riches_found_mine_is_local_order_crisis = yes } set_state_flag = riches_found_crime_crisis }
				if = { limit = { riches_found_mine_is_legitimacy_crisis = yes } set_state_flag = riches_found_administration_compromised }
				riches_found_clamp_mine_values = yes
				riches_found_refresh_active_mission_progress = yes
				if = {
					limit = {
						riches_found_mine_is_collapse_risk = yes
						riches_found_mine_is_severe_pressure = yes
						check_variable = { var = riches_found_sealed_depth_integrity value = constant:riches_found_threshold.sealed_integrity_gate compare = less_than }
					}
					riches_found_collapse_mine = yes
				}
				riches_found_refresh_mine_presentation = yes
				riches_found_refresh_mine_contribution = yes
				riches_found_refresh_evolution_gates = yes
				riches_found_refresh_state_achievement_timers = yes
				if = {
					limit = { NOT = { riches_found_evolution_i_gate = yes } }
					clr_state_flag = riches_found_evolution_i_pending
				}
				if = {
					limit = { NOT = { riches_found_evolution_ii_gate = yes } }
					clr_state_flag = riches_found_evolution_ii_pending
				}
				if = {
					limit = { NOT = { riches_found_evolution_iii_gate = yes } }
					clr_state_flag = riches_found_evolution_iii_pending
				}
				if = {
					limit = { NOT = { riches_found_gilded_sovereignty_gate = yes } }
					clr_state_flag = riches_found_gilded_sovereignty_pending
				}
				if = {
					limit = { NOT = { riches_found_bottomless_account_gate = yes } }
					clr_state_flag = riches_found_bottomless_account_pending
				}
				if = {
					limit = { riches_found_evolution_i_gate = yes NOT = { has_state_flag = riches_found_evolution_i_pending } }
					set_state_flag = riches_found_evolution_i_pending
					set_temp_variable = { riches_found_evolution_delay_days = mtth:riches_found_evolution_i_interval }
					clamp_temp_variable = { var = riches_found_evolution_delay_days min = constant:riches_found_duration.evolution_review_floor_days max = constant:riches_found_duration.evolution_review_ceiling_days }
					state_event = { id = chaosx.nr29.901 days = riches_found_evolution_delay_days trigger_for = controller }
				}
				else_if = {
					limit = { riches_found_evolution_ii_gate = yes NOT = { has_state_flag = riches_found_evolution_ii_pending } }
					set_state_flag = riches_found_evolution_ii_pending
					set_temp_variable = { riches_found_evolution_delay_days = mtth:riches_found_evolution_ii_interval }
					clamp_temp_variable = { var = riches_found_evolution_delay_days min = constant:riches_found_duration.evolution_review_floor_days max = constant:riches_found_duration.evolution_review_ceiling_days }
					state_event = { id = chaosx.nr29.902 days = riches_found_evolution_delay_days trigger_for = controller }
				}
				else_if = {
					limit = { riches_found_evolution_iii_gate = yes NOT = { has_state_flag = riches_found_evolution_iii_pending } }
					set_state_flag = riches_found_evolution_iii_pending
					set_temp_variable = { riches_found_evolution_delay_days = mtth:riches_found_evolution_iii_interval }
					clamp_temp_variable = { var = riches_found_evolution_delay_days min = constant:riches_found_duration.evolution_review_floor_days max = constant:riches_found_duration.evolution_review_ceiling_days }
					state_event = { id = chaosx.nr29.903 days = riches_found_evolution_delay_days trigger_for = controller }
				}
				if = {
					limit = { riches_found_gilded_sovereignty_gate = yes NOT = { has_state_flag = riches_found_gilded_sovereignty_pending } }
					set_state_flag = riches_found_gilded_sovereignty_pending
					set_temp_variable = { riches_found_late_outcome_delay_days = constant:riches_found_duration.late_outcome_review_days }
					clamp_temp_variable = { var = riches_found_late_outcome_delay_days min = constant:riches_found_duration.late_outcome_review_floor_days max = constant:riches_found_duration.late_outcome_review_ceiling_days }
					state_event = { id = chaosx.nr29.904 days = riches_found_late_outcome_delay_days trigger_for = controller }
				}
				if = {
					limit = { riches_found_bottomless_account_gate = yes NOT = { has_state_flag = riches_found_bottomless_account_pending } }
					set_state_flag = riches_found_bottomless_account_pending
					set_temp_variable = { riches_found_late_outcome_delay_days = constant:riches_found_duration.late_outcome_review_days }
					clamp_temp_variable = { var = riches_found_late_outcome_delay_days min = constant:riches_found_duration.late_outcome_review_floor_days max = constant:riches_found_duration.late_outcome_review_ceiling_days }
					state_event = { id = chaosx.nr29.905 days = riches_found_late_outcome_delay_days trigger_for = controller }
				}
			}
		}
	}
	if = { limit = { has_country_flag = riches_found_controller_aggregate_active } riches_found_schedule_controller_pulse = yes }
}

# STATE scope. Input: regular event target riches_found_contract_partner_target
# optional temporary riches_found_contract_type, and optional temporary
# riches_found_contract_exclusive_request (0 or 1).
riches_found_begin_contract = {
	if = {
		limit = {
			riches_found_contract_can_open = yes
			has_event_target = riches_found_contract_partner_target
			event_target:riches_found_contract_partner_target = { exists = yes }
		}
		set_variable = { riches_found_contract_partner = event_target:riches_found_contract_partner_target }
		if = {
			limit = { has_variable = riches_found_input_contract_type }
			set_variable = { riches_found_contract_type = riches_found_input_contract_type }
		}
		set_variable = { riches_found_contract_stage = constant:riches_found_contract_stage.active }
		set_variable = { riches_found_contract_started_date = global.date }
		set_variable = { riches_found_contract_review_date = global.date }
		add_to_variable = { riches_found_contract_review_date = constant:riches_found_duration.contract_review_days }
		set_state_flag = riches_found_contract_active
		set_state_flag = riches_found_concession_active
		clr_state_flag = riches_found_contract_expired
		clr_state_flag = riches_found_contract_dispute
		clr_state_flag = riches_found_contract_renewal_due
		clr_state_flag = riches_found_contract_transfer_needs_renegotiation
		clear_variable = riches_found_contract_renewal_due_date
		clear_variable = riches_found_contract_expiry_date
		if = {
			limit = { check_variable = { var = riches_found_contract_exclusive_request value = constant:riches_found_value.one compare = greater_than } }
			set_state_flag = riches_found_exclusive_contract
			set_state_flag = riches_found_achievement_exclusive_concession_history
		}
		riches_found_refresh_mine_presentation = yes
		riches_found_refresh_mine_contribution = yes
	}
}

# STATE scope. Records one compatible term amendment on the existing bounded
# non-exclusive contract. The partner pointer and review cadence remain owned
# by the mine; no signing grant, infrastructure award, or second contract is
# created.
riches_found_extend_limited_contract = {
	if = {
		limit = {
			riches_found_contract_is_active_and_valid = yes
			NOT = { has_state_flag = riches_found_exclusive_contract }
			NOT = { has_state_flag = riches_found_contract_pending_review }
			NOT = { has_state_flag = riches_found_contract_renewal_due }
		}
		add_to_variable = { riches_found_contract_amendment_count = constant:riches_found_value.one }
		set_variable = { riches_found_contract_last_amendment_date = global.date }
		set_variable = { riches_found_contract_stage = constant:riches_found_contract_stage.active }
		set_state_flag = riches_found_contract_active
		set_state_flag = riches_found_concession_active
		set_state_flag = riches_found_contract_amended
		riches_found_clamp_mine_values = yes
		riches_found_reconcile_mine_phase = yes
		riches_found_refresh_mine_state = yes
	}
}

# STATE scope. Re-evaluates one preserved contract after control or ownership
# change. Invalid partners are voided; valid partners resume as active.
riches_found_review_contract = {
	if = {
		limit = { has_state_flag = riches_found_contract_pending_review }
		if = {
			limit = {
				has_variable = riches_found_contract_partner
				check_variable = { var = riches_found_contract_partner value = CONTROLLER compare = not_equals }
				var:riches_found_contract_partner = {
					exists = yes
					NOT = { is_special_chaos_country = yes }
					NOT = { is_actual_nonhuman_country = yes }
					has_capitulated = no
				}
			}
			set_variable = { riches_found_contract_stage = constant:riches_found_contract_stage.active }
			set_state_flag = riches_found_contract_active
			clr_state_flag = riches_found_contract_pending_review
			set_state_flag = riches_found_contract_transfer_needs_renegotiation
			riches_found_mark_contract_renewal_due = yes
		}
		else = {
			set_state_flag = riches_found_contract_dispute
			riches_found_expire_contract = yes
		}
	}
}

# STATE scope. Opens one renewal window for the current contract. The deadline
# is stored on the mine, so transfer and repeated controller pulses cannot
# create parallel renewals or extend the same cycle.
riches_found_mark_contract_renewal_due = {
	if = {
		limit = {
			has_state_flag = riches_found_contract_active
			NOT = { has_state_flag = riches_found_contract_renewal_due }
		}
		set_state_flag = riches_found_contract_renewal_due
		set_variable = { riches_found_contract_renewal_due_date = global.date }
		set_variable = { riches_found_contract_expiry_date = global.date }
		add_to_variable = { riches_found_contract_expiry_date = constant:riches_found_duration.contract_renewal_grace_days }
	}
}

# STATE scope. Completes the single active renewal cycle without rebuilding
# physical infrastructure or awarding one-time contract benefits again.
riches_found_renew_contract = {
	if = {
		limit = {
			riches_found_contract_is_active_and_valid = yes
			has_state_flag = riches_found_contract_renewal_due
		}
		clr_state_flag = riches_found_contract_renewal_due
		clr_state_flag = riches_found_contract_dispute
		clr_state_flag = riches_found_contract_transfer_needs_renegotiation
		clear_variable = riches_found_contract_renewal_due_date
		clear_variable = riches_found_contract_expiry_date
		set_variable = { riches_found_contract_review_date = global.date }
		add_to_variable = { riches_found_contract_review_date = constant:riches_found_duration.contract_review_days }
		set_variable = { riches_found_contract_last_renewal_date = global.date }
		add_to_variable = { riches_found_contract_renewal_count = constant:riches_found_value.one }
		set_state_flag = riches_found_contract_renewed
		add_to_variable = { riches_found_extraction_pressure = constant:riches_found_scaling.contract_renewal_pressure_add }
		add_to_variable = { riches_found_revenue_legitimacy = constant:riches_found_scaling.contract_renewal_legitimacy_add }
		riches_found_record_achievement_contract_renegotiated = yes
		riches_found_clamp_mine_values = yes
		riches_found_refresh_mine_presentation = yes
		riches_found_refresh_mine_contribution = yes
		riches_found_refresh_state_achievement_timers = yes
	}
}

# STATE scope. Resolves an invalid or missed renewal once. It removes all live
# rights and pointers, preserves physical works, and leaves one bounded dispute
# fact for the security/foreign-interference routes.
riches_found_expire_contract = {
	if = {
		limit = {
			OR = {
				has_state_flag = riches_found_contract_active
				has_state_flag = riches_found_contract_pending_review
				has_state_flag = riches_found_contract_renewal_due
			}
		}
		if = { limit = { has_state_flag = riches_found_private_guards } set_state_flag = riches_found_private_guard_contract_lapsed }
		riches_found_clear_contract_state = yes
		set_variable = { riches_found_contract_stage = constant:riches_found_contract_stage.expired }
		set_variable = { riches_found_last_contract_end_date = global.date }
		set_state_flag = riches_found_contract_expired
		set_state_flag = riches_found_contract_expired_history
		set_state_flag = riches_found_contract_dispute
		add_to_variable = { riches_found_extraction_pressure = constant:riches_found_scaling.contract_expiry_pressure_add }
		add_to_variable = { riches_found_revenue_legitimacy = constant:riches_found_scaling.contract_expiry_legitimacy_add }
		add_to_variable = { riches_found_foreign_claim_pressure = constant:riches_found_scaling.contract_dispute_claim_pressure_add }
		riches_found_clamp_mine_values = yes
		riches_found_refresh_mine_presentation = yes
		riches_found_refresh_mine_contribution = yes
		riches_found_refresh_state_achievement_timers = yes
	}
}

# STATE scope. Called only from the bounded current-controller pulse. It opens
# the renewal window on the stored review date and expires a missed or invalid
# contract without scanning unrelated countries or states.
riches_found_refresh_contract_lifecycle = {
	if = {
		limit = { has_state_flag = riches_found_contract_active }
		if = {
			limit = { NOT = { riches_found_contract_is_active_and_valid = yes } }
			set_state_flag = riches_found_contract_dispute
			riches_found_expire_contract = yes
		}
		else = {
			if = {
				limit = {
					has_variable = riches_found_contract_review_date
					check_variable = { var = global.date value = riches_found_contract_review_date compare = greater_than_or_equals }
				}
				riches_found_mark_contract_renewal_due = yes
			}
			if = {
				limit = {
					has_state_flag = riches_found_contract_renewal_due
					has_variable = riches_found_contract_expiry_date
					check_variable = { var = global.date value = riches_found_contract_expiry_date compare = greater_than_or_equals }
				}
				riches_found_expire_contract = yes
			}
		}
	}
}

# STATE scope. Inputs: temporary riches_found_input_mission_kind and
# riches_found_input_mission_duration_days. State size, disorder, damage,
# transport, and a supplied force move the deadline inside the accepted band
# for the requested objective.
riches_found_calculate_mission_duration = {
	set_variable = { riches_found_mission_duration_days = riches_found_input_mission_duration_days }
	if = {
		limit = { check_variable = { state_population_k > constant:riches_found_threshold.preferred_state_population_thousands } }
		add_to_variable = { riches_found_mission_duration_days = constant:riches_found_duration.mission_population_delay_days }
	}
	if = {
		limit = { riches_found_mine_is_local_order_crisis = yes }
		add_to_variable = { riches_found_mission_duration_days = constant:riches_found_duration.mission_disorder_delay_days }
	}
	if = {
		limit = { has_state_flag = riches_found_mine_damaged }
		add_to_variable = { riches_found_mission_duration_days = constant:riches_found_duration.mission_damage_delay_days }
	}
	if = {
		limit = { riches_found_state_has_route_capacity = yes }
		add_to_variable = { riches_found_mission_duration_days = constant:riches_found_duration.mission_route_reduction_days }
	}
	if = {
		limit = { riches_found_state_has_supplied_force = yes }
		add_to_variable = { riches_found_mission_duration_days = constant:riches_found_duration.mission_supply_reduction_days }
	}
	if = {
		limit = { check_variable = { var = riches_found_input_mission_kind value = constant:riches_found_mission_kind.secure_rush compare = equals } }
		clamp_variable = { var = riches_found_mission_duration_days min = constant:riches_found_duration.secure_rush_min_days max = constant:riches_found_duration.secure_rush_max_days }
	}
	else_if = {
		limit = { check_variable = { var = riches_found_input_mission_kind value = constant:riches_found_mission_kind.open_railhead compare = equals } }
		clamp_variable = { var = riches_found_mission_duration_days min = constant:riches_found_duration.open_railhead_min_days max = constant:riches_found_duration.open_railhead_max_days }
	}

# STATE scope. The bounded controller pulse advances only active objectives on
# the current mine. Time-based proofs require the original route/security
# conditions to remain true and cannot be inherited from before mission start.
riches_found_refresh_active_mission_progress = {
	if = {
		limit = {
			has_state_flag = riches_found_mission_protect_pay_train_subject
			has_state_flag = riches_found_mission_protect_pay_train_progress
			has_state_flag = riches_found_pay_train_risk_at_start
			has_state_flag = riches_found_pay_train_route_active
			NOT = { has_state_flag = riches_found_pay_train_route_lost }
			NOT = { has_state_flag = riches_found_raid_robbery_succeeded }
			riches_found_state_has_supplied_force = yes
			check_variable = { var = global.date value = riches_found_mission_progress_date compare = greater_than_or_equals }
		}
		set_state_flag = riches_found_pay_train_safe_passage_proven
	}
	if = {
		limit = {
			has_state_flag = riches_found_mission_restore_shaft_subject
			has_state_flag = riches_found_mission_restore_shaft_progress
			has_state_flag = riches_found_repair_started
			riches_found_state_has_route_capacity = yes
			NOT = { has_state_flag = riches_found_severe_collapse }
			check_variable = { var = global.date value = riches_found_mission_progress_date compare = greater_than_or_equals }
		}
		set_state_flag = riches_found_shaft_restored
		set_state_flag = riches_found_shafts_reinforced
		clr_state_flag = riches_found_mine_damaged
		clr_state_flag = riches_found_mine_collapsed
	}
	if = {
		limit = {
			has_state_flag = riches_found_mission_hold_mine_subject
			has_state_flag = riches_found_mission_hold_mine_progress
			has_state_flag = riches_found_hold_started_during_war
			NOT = { has_state_flag = riches_found_occupation_active }
			riches_found_state_has_transport_route = yes
			riches_found_state_has_security_presence = yes
			OR = {
				check_variable = { var = global.date value = riches_found_mission_progress_date compare = greater_than_or_equals }
				CONTROLLER = { has_war = no }
			}
		}
		set_state_flag = riches_found_mine_hold_proven
	}
	if = {
		limit = {
			has_state_flag = riches_found_mission_evacuate_workforce_subject
			has_state_flag = riches_found_mission_evacuate_workforce_progress
			has_state_flag = riches_found_demons_evacuation_authorized
			has_state_flag = riches_found_demons_evacuation_ready
			has_state_flag = riches_found_demons_evacuation_route_active
			NOT = { has_state_flag = riches_found_demons_district_evacuated }
			check_variable = { var = global.date value = riches_found_mission_progress_date compare = greater_than_or_equals }
		}
		set_state_flag = riches_found_demons_evacuation_ready
	}
}

# COUNTRY scope. Recounts the currently active mission types from controlled
# Event 029 states. Each mission type is deliberately counted once because
# the decision surface forbids a second subject of the same type while one is
# active. This is bounded to the current controller's states and never scans
# the world.
riches_found_recount_active_mission_slots = {
	every_controlled_state = {
		limit = {
			riches_found_mine_is_registered = yes
			has_state_flag = riches_found_mission_active
			NOT = {
				OR = {
					has_state_flag = riches_found_mission_secure_rush_subject
					has_state_flag = riches_found_mission_open_railhead_subject
					has_state_flag = riches_found_mission_protect_pay_train_subject
					has_state_flag = riches_found_mission_break_claim_war_subject
					has_state_flag = riches_found_mission_restore_shaft_subject
					has_state_flag = riches_found_mission_hold_mine_subject
					has_state_flag = riches_found_mission_public_settlement_subject
					has_state_flag = riches_found_mission_audit_concession_subject
					has_state_flag = riches_found_mission_evacuate_workforce_subject
				}
			}
		}
		set_variable = { riches_found_last_mission_result = constant:riches_found_mission_stage.cancelled }
		set_variable = { riches_found_mission_stage = constant:riches_found_mission_stage.none }
		clear_variable = riches_found_mission_kind
		clear_variable = riches_found_mission_start_date
		clear_variable = riches_found_mission_deadline
		clear_variable = riches_found_mission_duration_days
		clear_variable = riches_found_mission_progress_date
		clr_state_flag = riches_found_mission_active
		clr_state_flag = riches_found_mission_cleanup_pending
	}
	set_variable = { riches_found_active_mission_count = constant:riches_found_value.zero }
	if = { limit = { any_controlled_state = { has_state_flag = riches_found_mission_secure_rush_subject } } add_to_variable = { riches_found_active_mission_count = constant:riches_found_value.one } }
	if = { limit = { any_controlled_state = { has_state_flag = riches_found_mission_open_railhead_subject } } add_to_variable = { riches_found_active_mission_count = constant:riches_found_value.one } }
	if = { limit = { any_controlled_state = { has_state_flag = riches_found_mission_protect_pay_train_subject } } add_to_variable = { riches_found_active_mission_count = constant:riches_found_value.one } }
	if = { limit = { any_controlled_state = { has_state_flag = riches_found_mission_break_claim_war_subject } } add_to_variable = { riches_found_active_mission_count = constant:riches_found_value.one } }
	if = { limit = { any_controlled_state = { has_state_flag = riches_found_mission_restore_shaft_subject } } add_to_variable = { riches_found_active_mission_count = constant:riches_found_value.one } }
	if = { limit = { any_controlled_state = { has_state_flag = riches_found_mission_hold_mine_subject } } add_to_variable = { riches_found_active_mission_count = constant:riches_found_value.one } }

# STATE scope. Called by `on_state_control_changed` after it saves regular
# targets riches_found_changed_state, riches_found_old_controller, and
# riches_found_new_controller. The old and new controller aggregates are each
# rebuilt once, while the mine record and all historical values persist.
riches_found_handle_state_control_change = {
	if = {
		limit = { riches_found_mine_is_registered = yes }
		riches_found_clear_stale_targets = yes
		if = {
			limit = {
				has_event_target = riches_found_old_controller
				has_event_target = riches_found_new_controller
			}
			add_to_variable = { riches_found_achievement_transfer_count = constant:riches_found_value.one }
			riches_found_record_achievement_transfer = yes
		}
		set_state_flag = riches_found_control_disrupted
		if = {
			limit = { has_state_flag = riches_found_contract_active }
			set_state_flag = riches_found_contract_pending_review
			set_variable = { riches_found_contract_stage = constant:riches_found_contract_stage.suspended }
		}
		riches_found_clear_mission_state = yes
		if = { limit = { CONTROLLER = { exists = yes } } set_variable = { riches_found_recorded_controller = CONTROLLER } else = { clear_variable = riches_found_recorded_controller } }
		riches_found_handle_occupation_change = yes
		riches_found_review_contract = yes
		riches_found_refresh_mine_presentation = yes
		riches_found_refresh_mine_contribution = yes
		if = { limit = { has_state_flag = riches_found_account_closure_finalized } set_state_flag = riches_found_achievement_account_closure_invalidated }
		if = { limit = { has_variable = riches_found_last_shift_controller } set_state_flag = riches_found_achievement_last_shift_invalidated }
		riches_found_refresh_state_achievement_timers = yes

		if = {
			limit = { has_event_target = riches_found_old_controller }
			event_target:riches_found_old_controller = {
				if = {
					limit = {
						has_variable = riches_found_selected_mine
						check_variable = { var = riches_found_selected_mine value = event_target:riches_found_changed_state compare = equals }
					}
					clear_variable = riches_found_selected_mine
				}
				riches_found_refresh_controller_aggregate = yes
				riches_found_recount_active_mission_slots = yes
			}
		}
		if = {
			limit = { has_event_target = riches_found_new_controller }
			event_target:riches_found_new_controller = {
				if = { limit = { NOT = { riches_found_country_has_selected_mine = yes } } set_variable = { riches_found_selected_mine = event_target:riches_found_changed_state } }
				riches_found_refresh_controller_aggregate = yes
				riches_found_recount_active_mission_slots = yes
			}
		}
	}
}

# STATE scope. Input is a real successful pay/shipment mission, supplied by
# the owning mission consumer after objective validation.
riches_found_record_achievement_pay_train = {
	if = {
		limit = {
			has_state_flag = riches_found_pay_train_instance_active
			NOT = { has_state_flag = riches_found_pay_train_instance_counted }
		}
		set_state_flag = riches_found_pay_train_instance_counted
		add_to_variable = { riches_found_achievement_pay_train_count = constant:riches_found_value.one }
		set_state_flag = riches_found_achievement_pay_train_recorded
		if = {
			limit = {
				check_variable = {
					var = riches_found_achievement_pay_train_count
					value = constant:riches_found_achievement_limit.pay_train_successes_required
					compare = greater_than_or_equals
				}
			}
			set_state_flag = riches_found_achievement_pay_train_sequence_valid
			set_state_flag = riches_found_achievement_pay_train_ready
		}
	}
}
