# Event 021 date-unit conversion.
# ANY scope. Caller initializes event021_date_offset with a duration in days.
# In-place output is an encoded date offset. Fractional hours round up so a
# due-date comparison cannot resolve before the requested duration has passed.
event021_days_to_date_offset = {
	multiply_temp_variable = { event021_date_offset = constant:event021_calendar.hours_per_day }
	set_temp_variable = { event021_date_unrounded_hours = event021_date_offset }
	round_temp_variable = event021_date_offset
	if = {
		limit = { check_variable = { var = event021_date_offset value = event021_date_unrounded_hours compare = less_than } }
		add_to_temp_variable = { event021_date_offset = constant:event021_calendar.hour_step }
	}
	divide_temp_variable = { event021_date_offset = constant:event021_calendar.hours_per_date_unit }
}
