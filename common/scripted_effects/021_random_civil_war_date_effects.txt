# Event 021 date-unit conversion.
# ANY scope. Caller initializes event021_date_offset with a duration in days.
# In-place output is an encoded date offset. The shared
# chaosx_convert_days_to_date_offset helper does the arithmetic and rounds
# fractional hours up, so a due-date comparison cannot resolve before the
# requested duration has passed. Its output is created here before the call
# so the value survives the helper's return.
event021_days_to_date_offset = {
	set_temp_variable = { chaosx_date_days = event021_date_offset }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_temp_variable = { event021_date_offset = chaosx_date_offset }
}
