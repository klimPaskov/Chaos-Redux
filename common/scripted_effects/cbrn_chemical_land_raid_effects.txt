# ============================================================================
# CHAOS REDUX - NATIVE CHEMICAL LAND RAID OUTCOME ADAPTER
# ============================================================================
# A hidden immediate country event supplies actor-country ROOT, exact actor,
# state and victim event targets, agent, route, engine outcome, and the Phantom
# Mist variant signal. Native essential_equipment is the sole payload debit.
# This adapter never spends or refunds stock.
# ============================================================================

cbrn_land_raid_set_payload_requirement = {
	set_temp_variable = { cbrn_action_payload_required = constant:cbrn_math.zero }
	if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.cylinder_release compare = equals } }
		set_temp_variable = { cbrn_action_payload_required = constant:cbrn_payload_route_cost.cylinder_release }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.projector_barrage compare = equals } }
		set_temp_variable = { cbrn_action_payload_required = constant:cbrn_payload_route_cost.projector_barrage }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals } }
		set_temp_variable = { cbrn_action_payload_required = constant:cbrn_payload_route_cost.artillery_fire_plan }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.armored_delivery compare = equals } }
		set_temp_variable = { cbrn_action_payload_required = constant:cbrn_payload_route_cost.armored_delivery }
	}
}

cbrn_land_raid_apply_engine_outcome = {
	if = {
		limit = { check_variable = { var = cbrn_raid_engine_outcome value = constant:cbrn_chemical_raid_engine_outcome.limited_success compare = equals } }
		multiply_temp_variable = { cbrn_action_release_efficiency_mult = constant:cbrn_chemical_raid_outcome.partial_dose_max }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_raid_engine_outcome value = constant:cbrn_chemical_raid_engine_outcome.critical_success compare = equals } }
		multiply_temp_variable = { cbrn_action_release_efficiency_mult = constant:cbrn_chemical_raid_outcome.catastrophic_dose_min }
	}
}

cbrn_land_raid_apply_phantom_mist_result = {
	if = {
		limit = {
			check_variable = { var = cbrn_raid_phantom_mist value = constant:cbrn_math.one compare = equals }
			check_variable = { var = cbrn_action_agent value = constant:cbrn_agent.malodor compare = equals }
			cbrn_project_has_phantom_mist = yes
		}
		multiply_temp_variable = { cbrn_action_military_death_fraction = constant:cbrn_project_effect.phantom_mist_casualty_multiplier }
		multiply_temp_variable = { cbrn_action_civilian_death_fraction = constant:cbrn_project_effect.phantom_mist_casualty_multiplier }
		multiply_temp_variable = { cbrn_action_disruption = constant:cbrn_project_effect.phantom_mist_disruption_multiplier }
		clamp_temp_variable = { var = cbrn_action_disruption min = constant:cbrn_math.zero max = constant:cbrn_action_output_bounds.disruption_max }
	}
}

				limit = {
					cbrn_chemical_land_raid_policy_allows_exact_target = yes
