cannibalism_scenario_select_source_country = {
	set_temp_variable = { cannibalism_scenario_source_found = 0 }
	if = {
		limit = { cannibalism_scenario_valid_preflight_source_country = yes }
		save_event_target_as = cannibalism_scenario_source_country
		set_temp_variable = { cannibalism_scenario_source_found = 1 }
	}
	else = {
		every_country = {
			limit = {
				check_variable = { cannibalism_scenario_source_found < 1 }
				is_ai = yes
				cannibalism_scenario_valid_preflight_source_country = yes
			}
			save_event_target_as = cannibalism_scenario_source_country
			set_temp_variable = { cannibalism_scenario_source_found = 1 }
		}
	}
}

cannibalism_scenario_build_manual_preflight_plan = {
	set_temp_variable = { cannibalism_scenario_preflight_result = 0 }
	set_temp_variable = { cannibalism_scenario_use_preflight_plan = 0 }
	clear_temp_array = cannibalism_scenario_planned_actor_countries
	clear_temp_array = cannibalism_scenario_planned_island_states
	clear_temp_array = cannibalism_scenario_planned_siege_states
	clear_temp_array = cannibalism_scenario_planned_march_states
	clear_temp_array = cannibalism_scenario_planned_warlord_slots
	set_temp_variable = { cannibalism_scenario_planned_actor_count = 0 }
	cannibalism_scenario_select_source_country = yes
	if = {
		limit = {
			cannibalism_scenario_can_launch = yes
			check_variable = { cannibalism_scenario_source_found > 0 }
			has_event_target = cannibalism_scenario_source_country
		}
		event_target:cannibalism_scenario_source_country = {
			add_to_temp_array = { cannibalism_scenario_planned_actor_countries = THIS }
		}
		add_to_temp_variable = { cannibalism_scenario_planned_actor_count = 1 }
		every_country = {
			limit = {
				check_variable = { cannibalism_scenario_planned_actor_count < cannibalism_scenario_target_countries }
				is_ai = yes
				cannibalism_scenario_valid_preflight_additional_actor = yes
				NOT = { tag = event_target:cannibalism_scenario_source_country }
			}
			add_to_temp_array = { cannibalism_scenario_planned_actor_countries = THIS }
			add_to_temp_variable = { cannibalism_scenario_planned_actor_count = 1 }
		}
	}

	if = {
		limit = {
			check_variable = {
				var = cannibalism_scenario_planned_actor_count
				value = cannibalism_scenario_target_countries
				compare = equals
			}
		}
		cannibalism_scenario_prepare_preflight_requirements = yes
		set_temp_variable = { cannibalism_scenario_planned_island_count = 0 }
		set_temp_variable = { cannibalism_scenario_planned_island_warlord_count = 0 }
		set_temp_variable = { cannibalism_scenario_planned_siege_count = 0 }
		set_temp_variable = { cannibalism_scenario_planned_march_count = 0 }

		every_state = {
			limit = {
				check_variable = { cannibalism_scenario_planned_island_warlord_count < cannibalism_scenario_required_island_warlord_states }
				cannibalism_scenario_preflight_island_warlord_state = yes
				NOT = { is_in_array = { array = cannibalism_scenario_planned_island_states value = THIS } }
			}
			add_to_temp_array = { cannibalism_scenario_planned_island_states = THIS }
			add_to_temp_variable = { cannibalism_scenario_planned_island_count = 1 }
			add_to_temp_variable = { cannibalism_scenario_planned_island_warlord_count = 1 }
		}
		every_state = {
			limit = {
				check_variable = { cannibalism_scenario_planned_island_count < cannibalism_scenario_required_island_states }
				cannibalism_scenario_preflight_island_node_state = yes
				NOT = { is_in_array = { array = cannibalism_scenario_planned_island_states value = THIS } }
			}
			add_to_temp_array = { cannibalism_scenario_planned_island_states = THIS }
			add_to_temp_variable = { cannibalism_scenario_planned_island_count = 1 }
		}
		every_state = {
			limit = {
				check_variable = { cannibalism_scenario_planned_siege_count < cannibalism_scenario_required_siege_states }
				cannibalism_scenario_preflight_siege_warlord_state = yes
				NOT = { is_in_array = { array = cannibalism_scenario_planned_island_states value = THIS } }
				NOT = { is_in_array = { array = cannibalism_scenario_planned_siege_states value = THIS } }
			}
			add_to_temp_array = { cannibalism_scenario_planned_siege_states = THIS }
			add_to_temp_variable = { cannibalism_scenario_planned_siege_count = 1 }
		}
		every_state = {
			limit = {
				check_variable = { cannibalism_scenario_planned_march_count < cannibalism_scenario_required_march_states }
				cannibalism_scenario_preflight_march_warlord_state = yes
				NOT = { is_in_array = { array = cannibalism_scenario_planned_island_states value = THIS } }
				NOT = { is_in_array = { array = cannibalism_scenario_planned_siege_states value = THIS } }
				NOT = { is_in_array = { array = cannibalism_scenario_planned_march_states value = THIS } }
			}
			add_to_temp_array = { cannibalism_scenario_planned_march_states = THIS }
			add_to_temp_variable = { cannibalism_scenario_planned_march_count = 1 }
		}
		cannibalism_scenario_plan_available_warlord_slots = yes

		if = {
			limit = {
				check_variable = { var = cannibalism_scenario_planned_island_count value = cannibalism_scenario_required_island_states compare = equals }
				check_variable = { var = cannibalism_scenario_planned_island_warlord_count value = cannibalism_scenario_required_island_warlord_states compare = equals }
				check_variable = { var = cannibalism_scenario_planned_siege_count value = cannibalism_scenario_required_siege_states compare = equals }
				check_variable = { var = cannibalism_scenario_planned_march_count value = cannibalism_scenario_required_march_states compare = equals }
				check_variable = { var = cannibalism_scenario_planned_warlord_slot_count value = cannibalism_scenario_required_warlord_slots compare = equals }
				OR = {
					NOT = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_convergence } }
					AND = {
						NOT = { has_global_flag = cannibalism_convergence_active }
						NOT = { has_global_flag = cannibalism_reveal_complete }
						NOT = { has_global_flag = cannibalism_convergence_broken }
					}
				}
			}
			set_temp_variable = { cannibalism_scenario_preflight_result = 1 }
			set_temp_variable = { cannibalism_scenario_use_preflight_plan = 1 }
		}
	}
}

			}

cannibalism_scenario_setup_silent_islands = {
	set_temp_variable = { cannibalism_scenario_setup_result = 0 }
	set_temp_variable = { cannibalism_scenario_target_countries = constant:cannibalism_scenario_scale.low_countries }
	cannibalism_scenario_initialize_required_actors = yes
	set_temp_variable = { cannibalism_scenario_island_states_seeded = 0 }
	set_temp_variable = { cannibalism_scenario_warlords_created = 0 }
	set_temp_variable = { cannibalism_scenario_warlord_formation_required = 0 }
	if = {
		limit = { check_variable = { cannibalism_scenario_actor_setup_result > 0 } }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_scenario_island_states_seeded
					value = cannibalism_scenario_target_states
					compare = less_than
				}
				check_variable = { cannibalism_scenario_setup_failed < 1 }
			}
			cannibalism_scenario_select_island_state = yes
			if = {
				limit = {
					check_variable = { cannibalism_scenario_state_found > 0 }
					has_event_target = cannibalism_scenario_selected_state
				}
				event_target:cannibalism_scenario_source_country = { save_event_target_as = cannibalism_node_source_actor }
				event_target:cannibalism_scenario_selected_state = { cannibalism_scenario_seed_current_state = yes }
				if = {
					limit = { check_variable = { cannibalism_scenario_node_setup_result > 0 } }
					add_to_temp_variable = { cannibalism_scenario_island_states_seeded = 1 }
					if = {
						limit = {
							check_variable = {
								var = cannibalism_scenario_warlords_created
								value = cannibalism_scenario_target_silent_hosts
								compare = less_than
							}
						}
						cannibalism_scenario_form_selected_warlord = yes
					}
					event_target:cannibalism_scenario_selected_state = { clr_state_flag = cannibalism_scenario_reserved_state }
				}
				else = { set_temp_variable = { cannibalism_scenario_setup_failed = 1 } }
			}
			else = { set_temp_variable = { cannibalism_scenario_setup_failed = 1 } }
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_scenario_setup_failed < 1 }
			check_variable = {
				var = cannibalism_scenario_island_states_seeded
				value = cannibalism_scenario_target_states
				compare = equals
			}
		}
		set_temp_variable = { cannibalism_scenario_setup_result = 1 }
	}
}

# convergence transactions without marking the campaign as a manual scenario.

# reusable slots exist, so the opening creates a real three-warlord network,

# ---------------------------------------------------------------------------
# Public wrapper called by the shared triggerable-scenario dispatcher
# ---------------------------------------------------------------------------

trigger_cannibalism_scenario = {
	set_temp_variable = { cannibalism_scenario_setup_failed = 0 }
	set_temp_variable = { cannibalism_scenario_setup_result = 0 }
	cannibalism_scenario_prepare_scale = yes
	cannibalism_scenario_build_manual_preflight_plan = yes
	if = {
		limit = {
			check_variable = { cannibalism_scenario_preflight_result > 0 }
			has_event_target = cannibalism_scenario_source_country
		}
		clr_country_flag = cannibalism_manual_scenario_setup_failed
		set_global_flag = cannibalism_scenario_launch_active
		cannibalism_scenario_prepare_runtime = yes
		if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_discipline_collapse } }
			cannibalism_scenario_setup_discipline_collapse = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_ritual_cells } }
			cannibalism_scenario_setup_ritual_cells = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_silent_islands } }
			cannibalism_scenario_setup_silent_islands = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_warlord_states } }
			cannibalism_scenario_setup_warlord_states = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_convergence } }
			cannibalism_scenario_setup_convergence = yes
		}
	}

	if = {
		limit = { check_variable = { cannibalism_scenario_setup_result > 0 } }
		set_country_flag = cannibalism_manual_scenario_launcher
		set_global_flag = cannibalism_manual_scenario_ever_launched
		set_variable = { global.cannibalism_scenario_last_type = cannibalism_scenario_selected_type }
		set_variable = { global.cannibalism_scenario_last_intensity = cannibalism_scenario_selected_intensity }
		add_to_variable = { global.cannibalism_scenario_launch_count = 1 }
		cannibalism_compact_actor_registry = yes
		cannibalism_compact_node_registry = yes
		cannibalism_rebuild_runtime_counts = yes
		cannibalism_refresh_world_threat_source = yes
		cannibalism_schedule_next_pulse = yes
	}
	else = {
		set_country_flag = cannibalism_manual_scenario_setup_failed
	}

	if = {
		limit = { check_variable = { cannibalism_scenario_preflight_result > 0 } }
		every_state = {
			limit = {
				OR = {
					has_state_flag = cannibalism_scenario_reserved_state
					has_state_flag = cannibalism_scenario_march_column
				}
			}
			clr_state_flag = cannibalism_scenario_reserved_state
			clr_state_flag = cannibalism_scenario_march_column
		}
		clr_global_flag = cannibalism_scenario_launch_active
	}
	clear_temp_array = cannibalism_scenario_planned_actor_countries
	clear_temp_array = cannibalism_scenario_planned_island_states
	clear_temp_array = cannibalism_scenario_planned_siege_states
	clear_temp_array = cannibalism_scenario_planned_march_states
	clear_temp_array = cannibalism_scenario_planned_warlord_slots
	set_temp_variable = { cannibalism_prefire_evolution = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_i_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_ii_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_iii_pending = 0 }
}
