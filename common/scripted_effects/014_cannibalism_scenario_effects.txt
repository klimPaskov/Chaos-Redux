cannibalism_scenario_select_source_country = {
	set_temp_variable = { cannibalism_scenario_source_found = 0 }
	if = {
		limit = { cannibalism_scenario_valid_source_country = yes }
		save_event_target_as = cannibalism_scenario_source_country
		set_temp_variable = { cannibalism_scenario_source_found = 1 }
	}
	else = {
		every_country = {
			limit = {
				check_variable = { cannibalism_scenario_source_found < 1 }
				is_ai = yes
				cannibalism_scenario_valid_source_country = yes
			}
			save_event_target_as = cannibalism_scenario_source_country
			set_temp_variable = { cannibalism_scenario_source_found = 1 }
		}
	}
}

cannibalism_scenario_apply_scale_to_current_actor = {
	set_variable = { cannibalism_field_hunger = cannibalism_scenario_field_hunger }
	set_variable = { cannibalism_command_integrity = cannibalism_scenario_command_integrity }
	set_variable = { cannibalism_cell_strength = cannibalism_scenario_cell_strength }
	if = {
		limit = { check_variable = { cannibalism_prefire_evolution > 0 } }
		set_variable = { cannibalism_cult_cohesion = cannibalism_scenario_cult_cohesion }
	}
	if = {
		limit = { check_variable = { cannibalism_prefire_evolution > 1 } }
		set_variable = { cannibalism_network_alignment = cannibalism_scenario_network_alignment }
		set_country_flag = cannibalism_network_actor
	}
	cannibalism_clamp_current_country_meters = yes
	cannibalism_refresh_country_penalty_variables = yes

	if = {
		limit = {
			is_ai = yes
			NOT = { has_country_flag = cannibalism_opening_policy_chosen }
			check_variable = { cannibalism_scenario_selected_intensity = constant:triggerable_scenario_intensity.low }
		}
		cannibalism_choose_open_emergency_policy = yes
	}
	else_if = {
		limit = {
			is_ai = yes
			NOT = { has_country_flag = cannibalism_opening_policy_chosen }
			check_variable = { cannibalism_scenario_selected_intensity = constant:triggerable_scenario_intensity.medium }
		}
		cannibalism_choose_concealment_policy = yes
	}
	else_if = {
		limit = {
			is_ai = yes
			NOT = { has_country_flag = cannibalism_opening_policy_chosen }
		}
		cannibalism_choose_exploitation_policy = yes
	}
}

cannibalism_scenario_initialize_current_country = {
	set_temp_variable = { cannibalism_scenario_actor_initialization_result = 0 }
	set_temp_variable = { cannibalism_scenario_actor_was_new = 0 }
	if = {
		limit = { is_cannibalism_active_ordinary_country = yes }
		save_event_target_as = cannibalism_current_actor
		save_event_target_as = cannibalism_node_source_actor
		set_temp_variable = { cannibalism_scenario_actor_initialization_result = 1 }
	}
	else_if = {
		limit = { cannibalism_scenario_valid_new_actor_country = yes }
		clr_country_flag = cannibalism_local_victory
		clr_country_flag = cannibalism_external_reinfection_protected
		clr_country_flag = cannibalism_country_cleanup_complete
		cannibalism_select_highest_risk_state = yes
		if = {
			limit = {
				check_variable = { cannibalism_state_selected > 0 }
				has_event_target = cannibalism_selected_state
			}
			event_target:cannibalism_selected_state = { save_event_target_as = cannibalism_initial_state }
			cannibalism_initialize_current_country = yes
			if = {
				limit = { check_variable = { cannibalism_actor_register_result > 0 } }
				set_temp_variable = { cannibalism_scenario_actor_initialization_result = 1 }
				set_temp_variable = { cannibalism_scenario_actor_was_new = 1 }
			}
		}
	}
	if = {
		limit = { check_variable = { cannibalism_scenario_actor_initialization_result > 0 } }
		set_country_flag = cannibalism_manual_scenario_actor
		cannibalism_scenario_apply_scale_to_current_actor = yes
	}
}

cannibalism_scenario_authorize_requested_evolutions = {
	if = {
		limit = {
			check_variable = { cannibalism_prefire_evolution_i_enabled > 0 }
			NOT = { has_global_flag = cannibalism_evolution_i_active }
		}
		set_temp_variable = { cannibalism_evolution_transition_authorized = 1 }
		cannibalism_open_evolution_i = yes
	}
	if = {
		limit = {
			check_variable = { cannibalism_prefire_evolution_ii_enabled > 0 }
			NOT = { has_global_flag = cannibalism_evolution_ii_active }
		}
		set_temp_variable = { cannibalism_evolution_transition_authorized = 1 }
		cannibalism_open_evolution_ii = yes
	}

}

cannibalism_scenario_initialize_required_actors = {
	set_temp_variable = { cannibalism_scenario_actor_count = 0 }
	set_temp_variable = { cannibalism_scenario_nodes_created = 0 }
	set_temp_variable = { cannibalism_scenario_source_was_new = 0 }
	event_target:cannibalism_scenario_source_country = {
		cannibalism_scenario_initialize_current_country = yes
		if = {
			limit = { check_variable = { cannibalism_scenario_actor_initialization_result > 0 } }
			add_to_temp_variable = { cannibalism_scenario_actor_count = 1 }
			if = {
				limit = { check_variable = { cannibalism_scenario_actor_was_new > 0 } }
				set_temp_variable = { cannibalism_scenario_source_was_new = 1 }
				cannibalism_scenario_record_new_actor_nodes = yes
			}
		}
	}
	if = {
		limit = { check_variable = { cannibalism_scenario_source_was_new > 0 } }
		event_target:cannibalism_scenario_source_country = {
			if = { limit = { is_ai = no } country_event = { id = chaosx.nr14.2 days = 1 } }
		}
	}
	cannibalism_scenario_authorize_requested_evolutions = yes

	every_country = {
		limit = {
			check_variable = {
				var = cannibalism_scenario_actor_count
				value = cannibalism_scenario_target_countries
				compare = less_than
			}
			is_ai = yes
			cannibalism_scenario_valid_source_country = yes
			NOT = { tag = event_target:cannibalism_scenario_source_country }
		}
		cannibalism_scenario_initialize_current_country = yes
		if = {
			limit = { check_variable = { cannibalism_scenario_actor_initialization_result > 0 } }
			add_to_temp_variable = { cannibalism_scenario_actor_count = 1 }
			if = {
				limit = { check_variable = { cannibalism_scenario_actor_was_new > 0 } }
				cannibalism_scenario_record_new_actor_nodes = yes
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				var = cannibalism_scenario_actor_count
				value = cannibalism_scenario_target_countries
				compare = equals
			}
		}
		set_temp_variable = { cannibalism_scenario_actor_setup_result = 1 }
	}
	else = {
		set_temp_variable = { cannibalism_scenario_actor_setup_result = 0 }
	}
}

cannibalism_scenario_seed_current_state = {
	set_temp_variable = { cannibalism_scenario_node_setup_result = 0 }
	if = {
		limit = {
			NOT = { cannibalism_state_has_active_cell = yes }
			has_event_target = cannibalism_scenario_source_country
		}
		set_temp_variable = { cannibalism_new_node_stage = cannibalism_scenario_node_stage }
		cannibalism_initialize_current_state_as_opening_node = yes
		if = {
			limit = { cannibalism_state_has_active_cell = yes }
			add_to_temp_variable = { cannibalism_scenario_nodes_created = 1 }
		}
	}
	if = {
		limit = { cannibalism_state_has_active_cell = yes }
		set_variable = { cannibalism_node_strength = constant:cannibalism_scenario_scale.mature_node_strength }
		set_temp_variable = { cannibalism_new_state_stage = cannibalism_scenario_node_stage }
		cannibalism_set_current_state_stage = yes
		if = {
			limit = { check_variable = { cannibalism_prefire_evolution > 1 } }
			set_state_flag = cannibalism_network_node
		}
		if = {
			limit = { check_variable = { cannibalism_scenario_mark_external_route > 0 } }
			set_state_flag = cannibalism_external_route_node
		}
		if = { limit = { is_island_state = yes } set_state_flag = cannibalism_commune_state }
		set_temp_variable = { cannibalism_scenario_node_setup_result = 1 }
	}
}

cannibalism_scenario_seed_additional_source_states = {
	event_target:cannibalism_scenario_source_country = {
		save_event_target_as = cannibalism_current_actor
		save_event_target_as = cannibalism_node_source_actor
	}
	while_loop_effect = {
		limit = {
			check_variable = {
				var = cannibalism_scenario_nodes_created
				value = cannibalism_scenario_target_states
				compare = less_than
			}
			check_variable = { cannibalism_scenario_setup_failed < 1 }
		}
		event_target:cannibalism_scenario_source_country = { cannibalism_select_highest_risk_state = yes }
		if = {
			limit = {
				check_variable = { cannibalism_state_selected > 0 }
				has_event_target = cannibalism_selected_state
			}
			event_target:cannibalism_selected_state = { cannibalism_scenario_seed_current_state = yes }
			if = {
				limit = { check_variable = { cannibalism_scenario_node_setup_result < 1 } }
				set_temp_variable = { cannibalism_scenario_setup_failed = 1 }
			}
		}
		else = {
			set_temp_variable = { cannibalism_scenario_setup_failed = 1 }
		}
	}
}

cannibalism_scenario_form_selected_warlord = {
	set_temp_variable = { cannibalism_scenario_warlord_formation_result = 0 }
	set_temp_variable = { cannibalism_warlord_creation_result = 0 }
	if = {
		limit = {
			check_variable = { cannibalism_scenario_setup_failed < 1 }
			check_variable = { cannibalism_scenario_state_found > 0 }
			has_event_target = cannibalism_scenario_selected_state
		}
		event_target:cannibalism_scenario_source_country = {
			save_event_target_as = cannibalism_current_actor
			save_event_target_as = cannibalism_node_source_actor
		}
		event_target:cannibalism_scenario_selected_state = {
			CONTROLLER = { save_event_target_as = cannibalism_state_host }
			cannibalism_scenario_seed_current_state = yes
			if = {
				limit = { check_variable = { cannibalism_scenario_node_setup_result > 0 } }
				cannibalism_prepare_warlord_creation_context = yes
				cannibalism_allocate_warlord_slot_for_current_state = yes
			}
		}
		if = {
			limit = { check_variable = { cannibalism_warlord_creation_result > 0 } }
			set_temp_variable = { cannibalism_scenario_warlord_formation_result = 1 }
			add_to_temp_variable = { cannibalism_scenario_warlords_created = 1 }
			event_target:cannibalism_warlord_new_country = {
				set_country_flag = cannibalism_manual_scenario_country
				set_variable = { cannibalism_network_alignment = cannibalism_scenario_network_alignment }
				cannibalism_clamp_current_country_meters = yes
				cannibalism_refresh_country_penalty_variables = yes
			}
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_scenario_warlord_formation_result < 1 }
			check_variable = { cannibalism_scenario_warlord_formation_required > 0 }
		}
		set_temp_variable = { cannibalism_scenario_setup_failed = 1 }
	}
	if = {
		limit = { has_event_target = cannibalism_scenario_selected_state }
		event_target:cannibalism_scenario_selected_state = {
			clr_state_flag = cannibalism_scenario_reserved_state

cannibalism_scenario_setup_silent_islands = {
	set_temp_variable = { cannibalism_scenario_setup_result = 0 }
	set_temp_variable = { cannibalism_scenario_target_countries = constant:cannibalism_scenario_scale.low_countries }
	cannibalism_scenario_initialize_required_actors = yes
	set_temp_variable = { cannibalism_scenario_island_states_seeded = 0 }
	set_temp_variable = { cannibalism_scenario_warlords_created = 0 }
	set_temp_variable = { cannibalism_scenario_warlord_formation_required = 0 }
	if = {
		limit = { check_variable = { cannibalism_scenario_actor_setup_result > 0 } }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = cannibalism_scenario_island_states_seeded
					value = cannibalism_scenario_target_states
					compare = less_than
				}
				check_variable = { cannibalism_scenario_setup_failed < 1 }
			}
			cannibalism_scenario_select_island_state = yes
			if = {
				limit = {
					check_variable = { cannibalism_scenario_state_found > 0 }
					has_event_target = cannibalism_scenario_selected_state
				}
				event_target:cannibalism_scenario_source_country = { save_event_target_as = cannibalism_node_source_actor }
				event_target:cannibalism_scenario_selected_state = { cannibalism_scenario_seed_current_state = yes }
				if = {
					limit = { check_variable = { cannibalism_scenario_node_setup_result > 0 } }
					add_to_temp_variable = { cannibalism_scenario_island_states_seeded = 1 }
					if = {
						limit = {
							check_variable = {
								var = cannibalism_scenario_warlords_created
								value = cannibalism_scenario_target_silent_hosts
								compare = less_than
							}
						}
						cannibalism_scenario_form_selected_warlord = yes
					}
					event_target:cannibalism_scenario_selected_state = { clr_state_flag = cannibalism_scenario_reserved_state }
				}
				else = { set_temp_variable = { cannibalism_scenario_setup_failed = 1 } }
			}
			else = { set_temp_variable = { cannibalism_scenario_setup_failed = 1 } }
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_scenario_setup_failed < 1 }
			check_variable = {
				var = cannibalism_scenario_island_states_seeded
				value = cannibalism_scenario_target_states
				compare = equals
			}
		}
		set_temp_variable = { cannibalism_scenario_setup_result = 1 }
	}
}

trigger_cannibalism_scenario = {
	set_temp_variable = { cannibalism_scenario_setup_failed = 0 }
	set_temp_variable = { cannibalism_scenario_setup_result = 0 }
	clr_country_flag = cannibalism_manual_scenario_setup_failed
	set_global_flag = cannibalism_scenario_launch_active
	set_country_flag = cannibalism_manual_scenario_launcher
	cannibalism_scenario_prepare_scale = yes
	cannibalism_scenario_select_source_country = yes
	if = {
		limit = {
			cannibalism_scenario_can_launch = yes
			check_variable = { cannibalism_scenario_source_found > 0 }
			has_event_target = cannibalism_scenario_source_country
		}
		cannibalism_scenario_prepare_runtime = yes
		if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_discipline_collapse } }
			cannibalism_scenario_setup_discipline_collapse = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_ritual_cells } }
			cannibalism_scenario_setup_ritual_cells = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_silent_islands } }
			cannibalism_scenario_setup_silent_islands = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_warlord_states } }
			cannibalism_scenario_setup_warlord_states = yes
		}
		else_if = {
			limit = { check_variable = { cannibalism_scenario_selected_type = constant:cannibalism_scenario.type_convergence } }
			cannibalism_scenario_setup_convergence = yes
		}
	}

	if = {
		limit = { check_variable = { cannibalism_scenario_setup_result > 0 } }
		set_global_flag = cannibalism_manual_scenario_ever_launched
		set_variable = { global.cannibalism_scenario_last_type = cannibalism_scenario_selected_type }
		set_variable = { global.cannibalism_scenario_last_intensity = cannibalism_scenario_selected_intensity }
		add_to_variable = { global.cannibalism_scenario_launch_count = 1 }
		cannibalism_compact_actor_registry = yes
		cannibalism_compact_node_registry = yes
		cannibalism_rebuild_runtime_counts = yes
		cannibalism_refresh_world_threat_source = yes
		cannibalism_schedule_next_pulse = yes
	}
	else = {
		set_country_flag = cannibalism_manual_scenario_setup_failed
	}

	every_state = {
		limit = {
			OR = {
				has_state_flag = cannibalism_scenario_reserved_state
				has_state_flag = cannibalism_scenario_march_column
				has_state_flag = cannibalism_scenario_prison_compound
			}
		}
		clr_state_flag = cannibalism_scenario_reserved_state
		clr_state_flag = cannibalism_scenario_march_column
		clr_state_flag = cannibalism_scenario_prison_compound
	}
	clr_global_flag = cannibalism_scenario_launch_active
	set_temp_variable = { cannibalism_prefire_evolution = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_i_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_ii_enabled = 0 }
	set_temp_variable = { cannibalism_prefire_evolution_iii_pending = 0 }
}
