# ============================================================================
# CHAOS REDUX - EVENT 024 VIDEO GAME IN SWEDEN EFFECTS
# ============================================================================
# This file owns the single Swedish simulation program. The host pointer is
# global because the event can outlive a tag change, while all operating state
# remains on the country that was selected when the fire-once event began.
# Reliance, the staged idea, decision phase, incidents, recovery, and
# achievement memory are deliberately kept together so every exit path uses
# the same cleanup contract.
# Removal guards check the current country modifier or the global target before
# cleanup so fresh initialization and repeated exits share the same safe path.
# ============================================================================

video_game_in_sweden_add_current_country_to_prefire_pool = {
	set_temp_variable = { individual_crisis_candidate_base_weight = constant:individual_crisis_targeting.one }
	# Both outputs must survive the nested curve and load-trigger calls.
	set_temp_variable = { individual_crisis_candidate_adjusted_weight = constant:individual_crisis_targeting.zero }
	set_temp_variable = { individual_crisis_active_load = constant:individual_crisis_targeting.zero }
	adjust_individual_crisis_candidate_ticket_weight = yes
	set_temp_variable = { video_game_in_sweden_prefire_pool_remaining = individual_crisis_candidate_adjusted_weight }
	while_loop_effect = {
		limit = { check_variable = { video_game_in_sweden_prefire_pool_remaining > constant:individual_crisis_targeting.zero } }
		add_to_temp_array = { video_game_in_sweden_prefire_host_pool = THIS }
		subtract_from_temp_variable = { video_game_in_sweden_prefire_pool_remaining = constant:individual_crisis_targeting.one }
	}
}

video_game_in_sweden_select_stockholm_controller_host = {
	clear_temp_array = video_game_in_sweden_prefire_host_pool
	every_country_with_original_tag = {
		original_tag_to_check = SWE
		limit = {
			owns_state = constant:video_game_in_sweden_event.stockholm_state
			controls_state = constant:video_game_in_sweden_event.stockholm_state
			video_game_in_sweden_can_receive_program = yes
		}
		video_game_in_sweden_add_current_country_to_prefire_pool = yes
	}
	if = {
		limit = { check_variable = { video_game_in_sweden_prefire_host_pool^num > constant:individual_crisis_targeting.zero } }
		random_scope_in_array = {
			array = video_game_in_sweden_prefire_host_pool
			save_global_event_target_as = video_game_in_sweden_host
		}
	}
	clear_temp_array = video_game_in_sweden_prefire_host_pool
}

video_game_in_sweden_prepare_random_event_fire = {
	set_temp_variable = { video_game_in_sweden_prefire_ready = 0 }
	# After the fire-once opening, a forced or manual re-trigger prepares
	# nothing. It releases only a host that never started the programme and
	# leaves a running cabinet's host in place.
	if = {
		limit = { is_in_array = { global.fired_events = constant:video_game_in_sweden_event.id } }
		if = {
			limit = {
				has_event_target = video_game_in_sweden_host
				NOT = {
					event_target:video_game_in_sweden_host = {
						has_country_flag = video_game_in_sweden_host_selected
					}
				}
			}
			clear_global_event_target = video_game_in_sweden_host
		}
	}
	if = {
		limit = {
			NOT = { is_in_array = { global.fired_events = constant:video_game_in_sweden_event.id } }
			has_event_target = video_game_in_sweden_host
			event_target:video_game_in_sweden_host = {
				video_game_in_sweden_can_receive_program = yes
			}
		}
		set_temp_variable = { video_game_in_sweden_prefire_ready = 1 }
	}
	if = {
		limit = {
			check_variable = {
				var = video_game_in_sweden_prefire_ready
				value = 1
				compare = less_than
			}
			NOT = { is_in_array = { global.fired_events = constant:video_game_in_sweden_event.id } }
		}
		if = {
			limit = { has_event_target = video_game_in_sweden_host }
			clear_global_event_target = video_game_in_sweden_host
		}
		if = {
			limit = {
				any_country_with_original_tag = {
					original_tag_to_check = SWE
					owns_state = constant:video_game_in_sweden_event.stockholm_state
					controls_state = constant:video_game_in_sweden_event.stockholm_state
					video_game_in_sweden_can_receive_program = yes
				}
			}
			video_game_in_sweden_select_stockholm_controller_host = yes
		}
		else_if = {
			limit = {
				any_country_with_original_tag = {
					original_tag_to_check = SWE
					owns_state = constant:video_game_in_sweden_event.stockholm_state
					video_game_in_sweden_can_receive_program = yes
				}
			}
			video_game_in_sweden_select_stockholm_owner_host = yes
		}
		else_if = {
			limit = { video_game_in_sweden_has_any_valid_host = yes }
			set_variable = { global.video_game_in_sweden_best_core_share = 0 }
			every_country_with_original_tag = {
				original_tag_to_check = SWE
				limit = { video_game_in_sweden_can_receive_program = yes }
				set_variable = { video_game_in_sweden_candidate_core_share = 0 }
				every_controlled_state = {
					limit = { is_core_of = SWE }
					PREV = {
						add_to_variable = { video_game_in_sweden_candidate_core_share = 1 }
					}
				}
				if = {
					limit = {
						check_variable = {
							var = video_game_in_sweden_candidate_core_share
							value = global.video_game_in_sweden_best_core_share
							compare = greater_than
						}
					}
					save_global_event_target_as = video_game_in_sweden_host
					set_variable = {
						global.video_game_in_sweden_best_core_share = video_game_in_sweden_candidate_core_share
					}
				}
				clear_variable = video_game_in_sweden_candidate_core_share
			}
			clear_variable = global.video_game_in_sweden_best_core_share
		}
	}
	if = {
		limit = {
			NOT = { is_in_array = { global.fired_events = constant:video_game_in_sweden_event.id } }
			has_event_target = video_game_in_sweden_host
			event_target:video_game_in_sweden_host = {
				video_game_in_sweden_is_valid_host = yes
			}
		}
		event_target:video_game_in_sweden_host = {
			set_country_flag = video_game_in_sweden_prefire_opening_pending
			# A cabinet opened by hand from the event settings or the Event
			# Log, or through a forced trigger, cannot earn the achievements.
			# Before the fire-once opening the only earlier mark comes from a
			# manual preparation that never opened the cabinet, so an ordinary
			# dispatch clears it.
			if = {
				limit = {
					OR = {
						check_variable = { events_log_manual_event_detail_trigger > constant:video_game_in_sweden_value.zero }
						ROOT = { has_country_flag = force_trigger_mode_enabled }
						ROOT = { has_variable = temp_bypass_checks }
					}
				}
				set_country_flag = video_game_in_sweden_debug_invalid
			}
			else = {
				clr_country_flag = video_game_in_sweden_debug_invalid
			}
		}
		set_temp_variable = { video_game_in_sweden_prefire_ready = 1 }
	}
}

video_game_in_sweden_change_reliance = {
	if = {
		limit = { NOT = { has_variable = simulation_reliance } }
		set_variable = { simulation_reliance = 0 }
	}
	set_temp_variable = { video_game_in_sweden_reliance_next = simulation_reliance }
	add_to_temp_variable = {
		video_game_in_sweden_reliance_next = video_game_in_sweden_reliance_delta
	}
	clamp_temp_variable = {
		var = video_game_in_sweden_reliance_next
		min = 0
		max = constant:video_game_in_sweden_value.maximum
	}
	# A formal auxiliary doctrine debriefs every session against field reports,
	# so during the staff-culture stage gains stop at its ceiling.
	if = {
		limit = {
			has_country_flag = video_game_in_sweden_formal_auxiliary_doctrine
			has_country_flag = video_game_in_sweden_evolution_i_active
			check_variable = { video_game_in_sweden_reliance_next > constant:video_game_in_sweden_reliance.auxiliary_reliance_cap }
		}
		set_temp_variable = { video_game_in_sweden_reliance_ceiling = constant:video_game_in_sweden_reliance.auxiliary_reliance_cap }
		if = {
			limit = { check_variable = { simulation_reliance > constant:video_game_in_sweden_reliance.auxiliary_reliance_cap } }
			set_temp_variable = { video_game_in_sweden_reliance_ceiling = simulation_reliance }
		}
		clamp_temp_variable = {
			var = video_game_in_sweden_reliance_next
			max = video_game_in_sweden_reliance_ceiling
		}
	}
	# With the next evolution switched off, the current stage keeps a clean
	# conclusion: gains stop just below the gate of the switched-off stage
	# (Doctrine for the baseline and Evolution I, Worldview for Evolution II).
	# Losses are never affected, and a value already above the ceiling holds.
	if = {
		limit = { video_game_in_sweden_next_evolution_disabled = yes }
		set_temp_variable = { video_game_in_sweden_reliance_ceiling = constant:video_game_in_sweden_reliance.habit_max }
		if = {
			limit = {
				check_variable = {
					var = video_game_in_sweden_phase
					value = constant:video_game_in_sweden_value.phase_evolution_ii
					compare = equals
				}
			}
			set_temp_variable = { video_game_in_sweden_reliance_ceiling = constant:video_game_in_sweden_reliance.doctrine_max }
		}
		if = {
			limit = { check_variable = { simulation_reliance > video_game_in_sweden_reliance_ceiling } }
			set_temp_variable = { video_game_in_sweden_reliance_ceiling = simulation_reliance }
		}
		clamp_temp_variable = {
			var = video_game_in_sweden_reliance_next
			max = video_game_in_sweden_reliance_ceiling
		}
	}
	set_variable = { simulation_reliance = video_game_in_sweden_reliance_next }
	# Consumed arithmetic input. Zero prevents accidental reuse within this chain.
	set_temp_variable = { video_game_in_sweden_reliance_delta = 0 }
	video_game_in_sweden_refresh_reliance_state = yes
	video_game_in_sweden_update_achievement_tracking = yes
	video_game_in_sweden_refresh_program_package = yes
}

# COUNTRY scope. Lowers Reliance to the caller's band ceiling and never raises
# it. Input: caller-owned temp video_game_in_sweden_reliance_cap.
video_game_in_sweden_cap_reliance = {
	if = {
		limit = { NOT = { has_variable = simulation_reliance } }
		set_variable = { simulation_reliance = 0 }
	}
	if = {
		limit = {
			check_variable = {
				var = simulation_reliance
				value = video_game_in_sweden_reliance_cap
				compare = greater_than
			}
		}
		set_variable = { simulation_reliance = video_game_in_sweden_reliance_cap }
	}
	video_game_in_sweden_refresh_reliance_state = yes
	video_game_in_sweden_update_achievement_tracking = yes
	video_game_in_sweden_refresh_program_package = yes
}

video_game_in_sweden_refresh_reliance_state = {
	clr_country_flag = video_game_in_sweden_reliance_instrument
	clr_country_flag = video_game_in_sweden_reliance_habit
	clr_country_flag = video_game_in_sweden_reliance_doctrine
	clr_country_flag = video_game_in_sweden_reliance_worldview
	if = {
		limit = {
			check_variable = {
				var = simulation_reliance
				value = constant:video_game_in_sweden_reliance.instrument_max
				compare = less_than_or_equals
			}
		}
		set_country_flag = video_game_in_sweden_reliance_instrument
		set_variable = { video_game_in_sweden_reliance_state = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = simulation_reliance
				value = constant:video_game_in_sweden_reliance.habit_max
				compare = less_than_or_equals
			}
		}
		set_country_flag = video_game_in_sweden_reliance_habit
		set_variable = { video_game_in_sweden_reliance_state = 2 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = simulation_reliance
				value = constant:video_game_in_sweden_reliance.doctrine_max
				compare = less_than_or_equals
			}
		}
		set_country_flag = video_game_in_sweden_reliance_doctrine
		set_variable = { video_game_in_sweden_reliance_state = 3 }
	}
	else = {
		set_country_flag = video_game_in_sweden_reliance_worldview
		set_variable = { video_game_in_sweden_reliance_state = 4 }
	}
}

video_game_in_sweden_roll_rulebook_commander = {
	if = {
		limit = {
			NOT = { has_country_flag = video_game_in_sweden_rulebook_commander_granted }
			check_variable = { video_game_in_sweden_commander_chance > constant:video_game_in_sweden_value.zero }
			any_unit_leader = {
				is_army_leader = yes
				is_assigned = yes
				num_units > constant:video_game_in_sweden_value.zero
			}
		}
		set_temp_variable = { video_game_in_sweden_commander_miss = constant:video_game_in_sweden_value.maximum }
		subtract_from_temp_variable = { video_game_in_sweden_commander_miss = video_game_in_sweden_commander_chance }
		clamp_temp_variable = {
			var = video_game_in_sweden_commander_miss
			min = constant:video_game_in_sweden_value.zero
			max = constant:video_game_in_sweden_value.maximum
		}
		random_list = {
			video_game_in_sweden_commander_chance = {
				video_game_in_sweden_assign_rulebook_commander = yes
			}
			video_game_in_sweden_commander_miss = {
			}
		}
	}
}

video_game_in_sweden_assign_rulebook_commander = {
	if = {
		limit = {
			NOT = { has_country_flag = video_game_in_sweden_rulebook_commander_assigned }
			NOT = { has_country_flag = video_game_in_sweden_rulebook_commander_granted }
		}
		if = {
			limit = {
				any_unit_leader = {
					is_army_leader = yes
					is_field_marshal = yes
					is_assigned = yes
					num_units > constant:video_game_in_sweden_value.zero
					NOT = { has_trait = video_game_in_sweden_rulebook_commander }
				}
			}
			random_unit_leader = {
				limit = {
					is_army_leader = yes
					is_field_marshal = yes
					is_assigned = yes
					num_units > constant:video_game_in_sweden_value.zero
					NOT = { has_trait = video_game_in_sweden_rulebook_commander }
				}
				add_unit_leader_trait = video_game_in_sweden_rulebook_commander
				save_global_event_target_as = video_game_in_sweden_rulebook_commander
			}
		}
		else_if = {
			limit = {
				any_unit_leader = {
					is_army_leader = yes
					is_field_marshal = no
					is_assigned = yes
					num_units > constant:video_game_in_sweden_value.zero
					NOT = { has_trait = video_game_in_sweden_rulebook_commander }
				}
			}
			random_unit_leader = {
				limit = {
					is_army_leader = yes
					is_field_marshal = no
					is_assigned = yes
					num_units > constant:video_game_in_sweden_value.zero
					NOT = { has_trait = video_game_in_sweden_rulebook_commander }
				}
				add_unit_leader_trait = video_game_in_sweden_rulebook_commander
				save_global_event_target_as = video_game_in_sweden_rulebook_commander
			}
		}
		if = {
			limit = { has_event_target = video_game_in_sweden_rulebook_commander }
			set_country_flag = video_game_in_sweden_rulebook_commander_assigned
			set_country_flag = video_game_in_sweden_rulebook_commander_granted
		}
	}
}

video_game_in_sweden_validate_commander = {
	if = {
		limit = {
			has_event_target = video_game_in_sweden_rulebook_commander
			event_target:video_game_in_sweden_rulebook_commander = {
				scope_exists = yes
				has_trait = video_game_in_sweden_rulebook_commander
			}
		}
		event_target:video_game_in_sweden_rulebook_commander = {
			remove_unit_leader_trait = video_game_in_sweden_rulebook_commander
		}
		every_unit_leader = {
			if = {
				limit = { has_trait = video_game_in_sweden_rulebook_commander }
				remove_unit_leader_trait = video_game_in_sweden_rulebook_commander
			}
		}
		clear_global_event_target = video_game_in_sweden_rulebook_commander
	}
	else = {
		every_unit_leader = {
			if = {
				limit = { has_trait = video_game_in_sweden_rulebook_commander }
				remove_unit_leader_trait = video_game_in_sweden_rulebook_commander
			}
		}
		if = {
			limit = { has_event_target = video_game_in_sweden_rulebook_commander }
			clear_global_event_target = video_game_in_sweden_rulebook_commander
		}
		clr_country_flag = video_game_in_sweden_rulebook_commander_assigned
	}
}

# COUNTRY scope. Evolution III opens with one incident chosen by the campaign:
# production without transport in any state, a diplomatic category error when
# Sweden has foreign business, and the perfect offensive or casualties as
# counters at war. It skips the cooldown but never a pending incident or a
# severe crisis, and its penalty is priced like any other incident.
video_game_in_sweden_queue_opening_incident = {
	if = {
		limit = {
			check_variable = { var = video_game_in_sweden_current_incident value = constant:video_game_in_sweden_value.zero compare = equals }
			NOT = { has_country_flag = video_game_in_sweden_opening_incident_queued }
			NOT = { video_game_in_sweden_is_incident_pause_crisis = yes }
		}
		set_country_flag = video_game_in_sweden_opening_incident_queued
		set_temp_variable = { video_game_in_sweden_opening_production_weight = constant:video_game_in_sweden_incident.opening_production_weight }
		set_temp_variable = { video_game_in_sweden_opening_diplomatic_weight = constant:video_game_in_sweden_value.zero }
		set_temp_variable = { video_game_in_sweden_opening_offensive_weight = constant:video_game_in_sweden_value.zero }
		set_temp_variable = { video_game_in_sweden_opening_casualties_weight = constant:video_game_in_sweden_value.zero }
		if = {
			limit = {
				OR = {
					has_country_flag = video_game_in_sweden_diplomacy_active
					is_in_faction = yes
					is_subject = yes
					has_war = yes
				}
			}
			set_temp_variable = { video_game_in_sweden_opening_diplomatic_weight = constant:video_game_in_sweden_incident.opening_diplomatic_weight }
		}
		if = {
			limit = { has_war = yes }
			set_temp_variable = { video_game_in_sweden_opening_offensive_weight = constant:video_game_in_sweden_incident.opening_offensive_weight }
			set_temp_variable = { video_game_in_sweden_opening_casualties_weight = constant:video_game_in_sweden_incident.opening_casualties_weight }
		}
		random_list = {
			video_game_in_sweden_opening_production_weight = {
				set_variable = { video_game_in_sweden_current_incident = constant:video_game_in_sweden_value.incident_production }
			}
			video_game_in_sweden_opening_diplomatic_weight = {
				set_variable = { video_game_in_sweden_current_incident = constant:video_game_in_sweden_value.incident_diplomatic }

			video_game_in_sweden_opening_offensive_weight = {
				set_variable = { video_game_in_sweden_current_incident = constant:video_game_in_sweden_value.incident_offensive }
			}
			video_game_in_sweden_opening_casualties_weight = {
				set_variable = { video_game_in_sweden_current_incident = constant:video_game_in_sweden_value.incident_casualties }
			}
		}
		video_game_in_sweden_prepare_incident_severity = yes
		country_event = { id = chaosx.nr24.30 days = constant:video_game_in_sweden_timing.evolution_iii_opening_incident_days }
	}
}

# COUNTRY scope. Monthly incident check. A due incident fires with a chance
# that rises with Reliance and officer competition and falls with debriefed
# doctrine, a red team and field validation. Incidents pause during a severe
# crisis, restriction enforcement and recovery.
video_game_in_sweden_roll_incident = {
	if = {
		limit = {
			check_variable = { var = video_game_in_sweden_program_days value = video_game_in_sweden_incident_ready_day compare = greater_than_or_equals }
			video_game_in_sweden_has_live_program_package = yes
			NOT = { video_game_in_sweden_is_incident_pause_crisis = yes }
			check_variable = { var = video_game_in_sweden_current_incident value = 0 compare = equals }
		}
		# Once every currently valid family has appeared, begin a fresh cycle.
		if = {
			limit = { NOT = { video_game_in_sweden_has_unseen_incident_family = yes } }
			video_game_in_sweden_reset_incident_cycle = yes
		}
		if = {
			limit = { video_game_in_sweden_has_unseen_incident_family = yes }
			video_game_in_sweden_compute_incident_chance = yes
			random_list = {
				video_game_in_sweden_incident_fire_weight = {
					video_game_in_sweden_select_incident_family = yes
				}
				video_game_in_sweden_incident_skip_weight = {
				}
			}
		}
		if = {
			limit = { check_variable = { var = video_game_in_sweden_current_incident value = 0 compare = greater_than } }
			video_game_in_sweden_prepare_incident_severity = yes

		}
	}
}
