# COUNTRY scope. Advances the selected field's presentation-only workboard
# page. The page is field-local so a player managing several deposits returns
# to the last view used for each one.
resources_found_cycle_selected_field_workboard = {
	if = {
		limit = { resources_found_decision_has_selected_field = yes }
		var:resources_found_selected_field = {
			add_to_variable = { resources_found_field_workboard_page = constant:resources_found_value.one }
			if = {
				limit = { check_variable = { resources_found_field_workboard_page > constant:resources_found_workboard_page.operations } }
				set_variable = { resources_found_field_workboard_page = constant:resources_found_workboard_page.administration }
			}
		}
	}
}

# COUNTRY scope. Maps the calculator's real profile/family/base-duration inputs
# to the stable cost and duration rows shown before selection. If an unsupported
# combination reaches the calculator, no slot is assigned and payment fails.

	clear_variable = resources_found_active_project_cost_preview_slot
	clear_variable = resources_found_active_project_duration_preview_slot
	if = {
		limit = {
			OR = {
				check_variable = { resources_found_active_project_family = constant:resources_found_decision_family.field }
				check_variable = { resources_found_active_project_family = constant:resources_found_decision_family.trade }
				check_variable = { resources_found_active_project_family = constant:resources_found_decision_family.containment }
			}
		}
		if = { limit = { check_variable = { resources_found_active_project_profile = constant:resources_found_decision_profile.administration } } set_variable = { resources_found_active_project_cost_preview_slot = constant:resources_found_decision_preview_slot.administration_field } }
		else_if = { limit = { check_variable = { resources_found_active_project_profile = constant:resources_found_decision_profile.survey } } set_variable = { resources_found_active_project_cost_preview_slot = constant:resources_found_decision_preview_slot.survey_field } }
		else_if = { limit = { check_variable = { resources_found_active_project_profile = constant:resources_found_decision_profile.industry } } set_variable = { resources_found_active_project_cost_preview_slot = constant:resources_found_decision_preview_slot.industry_field } }

		limit = { has_war = yes }
		set_country_flag = resources_found_decision_preview_war_context

	clear_variable = resources_found_decision_preview_excavation_depth
	clear_variable = resources_found_decision_preview_breach_pressure
	clear_variable = resources_found_decision_preview_subsurface_disturbance
	clear_variable = resources_found_decision_preview_workforce_safety
	clr_country_flag = resources_found_decision_preview_contract_context
	clr_country_flag = resources_found_decision_preview_poor_infrastructure_context
	clr_country_flag = resources_found_decision_preview_strong_infrastructure_context
	if = {
		limit = { resources_found_country_has_valid_selected_field = yes }
		set_variable = { resources_found_decision_preview_field = resources_found_selected_field }
		var:resources_found_selected_field = {
			set_variable = { ROOT.resources_found_decision_preview_excavation_depth = resources_found_excavation_depth }
			set_variable = { ROOT.resources_found_decision_preview_breach_pressure = resources_found_breach_pressure }

			set_variable = { ROOT.resources_found_decision_preview_workforce_safety = resources_found_workforce_safety }
			if = {
				limit = { resources_found_contract_is_active_and_valid = yes }
				ROOT = { set_country_flag = resources_found_decision_preview_contract_context }

			if = {
				limit = { NOT = { infrastructure > constant:resources_found_country_scale.poor_infrastructure_maximum } }
				ROOT = { set_country_flag = resources_found_decision_preview_poor_infrastructure_context }
			}

	}
	set_country_flag = resources_found_decision_cost_preview_ready

# COUNTRY scope for a field owner completing a review or renegotiation. The

	if = {
		limit = {
			resources_found_country_has_valid_selected_field = yes
			var:resources_found_selected_field = {
				has_variable = resources_found_contract_partner
				var:resources_found_contract_partner = { resources_found_decision_is_valid_foreign_partner = yes }
			}
		}
		var:resources_found_selected_field = {
			var:resources_found_contract_partner = { save_event_target_as = resources_found_contract_partner_target }
		}
		resources_found_grant_selected_field_resource_rights = yes
		# Event 12 achievement owner: a restored private concession in an African
		# field is recorded only at the real Event 18 rights-grant source.  This
		# deliberately does not infer a concession from an Event 12 action proxy.
		var:resources_found_selected_field = {
			if = {
				limit = {
					is_on_continent = africa
					has_state_flag = resources_found_foreign_concession_granted
					owner = {
						OR = {
							africa_is_current_host = yes
							AND = {
								africa_member_host_generation_is_current = yes
								has_event_target = africa_host
							}
						}
					}
				}
				owner = {
					if = {
						limit = { africa_is_current_host = yes }
						africa_achievement_record_private_resource_concession = yes
					}
					else_if = {
						limit = {
							africa_member_host_generation_is_current = yes
							has_event_target = africa_host
						}
						event_target:africa_host = {
							africa_achievement_record_private_resource_concession = yes
						}
					}
				}
			}
		}
		var:resources_found_selected_field = {
			if = {
				limit = { has_state_flag = resources_found_long_term_export_contract }
				resources_found_begin_contract_century_qualification = yes
			}
		}
	}
}

			clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_value.zero max = constant:resources_found_foreign_interest_score.resource_deficit_cap }
			multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.resource_deficit_per_unit }
			add_to_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_component }
		}
		set_temp_variable = { resources_found_foreign_interest_component = resource_imported@chromium }
		clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_value.zero max = constant:resources_found_foreign_interest_score.resource_import_cap }
		multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.resource_import_per_unit }
		add_to_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_component }
		set_temp_variable = { resources_found_foreign_interest_component = resource_consumed@chromium }
		clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_value.zero max = constant:resources_found_foreign_interest_score.resource_demand_cap }
		multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.resource_demand_per_unit }
		add_to_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_component }
		set_temp_variable = { resources_found_foreign_interest_component = resource_produced@chromium }
		subtract_from_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.resource_abundant_extraction_floor }
		if = {
			limit = { check_variable = { resources_found_foreign_interest_component > constant:resources_found_value.zero } }
			clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_value.zero max = constant:resources_found_foreign_interest_score.resource_abundant_extraction_cap }
			multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.resource_abundant_extraction_penalty_per_unit }
			subtract_from_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_component }
		}
	}

	# Route score uses the strongest available route once, so a neighboring
	# faction member does not receive several copies of the same access value.
	if = {
		limit = { is_neighbor_of = ROOT }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.route_neighbor }
	}
	else_if = {
		limit = { is_in_faction_with = ROOT }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.route_faction }
	}
	else_if = {
		limit = {
			OR = {
				has_military_access_to = ROOT
				ROOT = { has_military_access_to = PREV }
			}
		}
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.route_military_access }
	}
	else_if = {
		limit = { is_on_same_continent_as = event_target:resources_found_foreign_interest_field_target }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.route_same_continent }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_owned_and_controlled_by = PREV
				is_coastal = yes
			}
			OR = {
				event_target:resources_found_foreign_interest_field_target = { is_coastal = yes }
				AND = {
					event_target:resources_found_foreign_interest_field_target = { is_in_home_area = yes }
					event_target:resources_found_foreign_interest_owner_target = {
						any_owned_state = {
							is_owned_and_controlled_by = PREV
							is_coastal = yes
						}
					}
				}
			}
		}
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.route_maritime }
	}
	else = {
		subtract_from_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.route_inaccessible_penalty }
	}

	set_temp_variable = { resources_found_foreign_interest_component = opinion@ROOT }
	clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_foreign_interest_score.opinion_minimum max = constant:resources_found_foreign_interest_score.opinion_maximum }
	multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.opinion_per_point }
	add_to_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_component }

	if = {
		limit = { event_target:resources_found_foreign_interest_field_target = { is_core_of = PREV } }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.field_core }
	}
	else_if = {
		limit = { event_target:resources_found_foreign_interest_field_target = { is_claimed_by = PREV } }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.field_claim }
	}

	if = {
		limit = {
			has_event_target = resources_found_foreign_interest_existing_partner_target
			event_target:resources_found_foreign_interest_existing_partner_target = { tag = PREV }
		}
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.existing_partner }
	}
	else_if = {
		limit = {
			has_event_target = resources_found_foreign_interest_existing_partner_target
			has_war_with = event_target:resources_found_foreign_interest_existing_partner_target
		}
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.hostile_partner_rivalry }
	}

	if = {
		limit = { has_war = yes }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.candidate_at_war }
	}
	if = {
		limit = { is_major = yes }
		add_to_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.candidate_major }
	}
	set_temp_variable = { resources_found_foreign_interest_component = num_of_factories }
	clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_value.zero max = constant:resources_found_foreign_interest_score.candidate_factory_cap }
	multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.candidate_factory_per_factory }
	add_to_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_component }
	set_temp_variable = { resources_found_foreign_interest_component = surrender_progress }
	if = {
		limit = {
			check_variable = {
				var = resources_found_foreign_interest_component
				value = constant:resources_found_foreign_interest_score.candidate_overextension_threshold
				compare = greater_than
			}
		}
		subtract_from_temp_variable = { resources_found_foreign_interest_score = constant:resources_found_foreign_interest_score.candidate_overextension_penalty }
	}
}

# COUNTRY scope for the owner after inviting bids. This is the only full-country
# scan in the bid workflow. It runs once when that project completes, keeps the
# highest qualifying candidate in a regular event target, and then preserves the
# previous actor flag, owner pointer, field pointer, and cost-preview refresh.
resources_found_select_foreign_interest_actor = {
	if = {
		limit = {
			resources_found_country_has_valid_selected_field = yes
			var:resources_found_selected_field = {
				resources_found_is_active_field = yes
				NOT = { has_state_flag = resources_found_field_closing }
				NOT = { has_state_flag = resources_found_field_converted_to_cave_origin }
				NOT = { has_state_flag = resources_found_field_transaction_lock }
			}
		}
		save_event_target_as = resources_found_foreign_interest_owner_target
		var:resources_found_selected_field = { save_event_target_as = resources_found_foreign_interest_field_target }

		set_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_value.zero }
		set_temp_variable = { resources_found_foreign_interest_owner_factories = num_of_factories }
		set_temp_variable = { resources_found_foreign_interest_owner_stability = stability }
		set_temp_variable = { resources_found_foreign_interest_owner_surrender = surrender_progress }

		event_target:resources_found_foreign_interest_field_target = {
			set_temp_variable = { resources_found_foreign_interest_component = resources_found_total_added }
			clamp_temp_variable = { var = resources_found_foreign_interest_component min = constant:resources_found_value.zero max = constant:resources_found_foreign_interest_score.field_total_cap }
			multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.field_total_per_unit }
			add_to_temp_variable = { resources_found_foreign_interest_field_score = resources_found_foreign_interest_component }
			set_temp_variable = { resources_found_foreign_interest_component = resources_found_distinct_resource_count }
			multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.field_resource_type }
			add_to_temp_variable = { resources_found_foreign_interest_field_score = resources_found_foreign_interest_component }
			set_temp_variable = { resources_found_foreign_interest_component = resources_found_developed_yield }
			multiply_temp_variable = { resources_found_foreign_interest_component = constant:resources_found_foreign_interest_score.field_developed_yield_per_point }
			add_to_temp_variable = { resources_found_foreign_interest_field_score = resources_found_foreign_interest_component }
			if = {
				limit = { has_state_flag = resources_found_globally_significant_field }
				add_to_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.field_global_significance }
			}
			if = {
				limit = { has_state_flag = resources_found_field_suspended }
				subtract_from_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.field_suspension_penalty }
			}
			if = {
				limit = { has_state_flag = resources_found_commission_active }
				subtract_from_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.field_commission_penalty }
			}

			if = {
				limit = {
					has_variable = resources_found_concession_partner
					var:resources_found_concession_partner = { resources_found_decision_is_valid_foreign_partner = yes }
				}
				var:resources_found_concession_partner = { save_event_target_as = resources_found_foreign_interest_existing_partner_target }
			}
			else_if = {
				limit = {
					has_variable = resources_found_contract_partner
					var:resources_found_contract_partner = { resources_found_decision_is_valid_foreign_partner = yes }
				}
				var:resources_found_contract_partner = { save_event_target_as = resources_found_foreign_interest_existing_partner_target }
			}
		}

		if = {
			limit = { NOT = { is_major = yes } }
			add_to_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.owner_minor_weakness }
		}
		if = {
			limit = {
				check_variable = { resources_found_foreign_interest_owner_factories < constant:resources_found_foreign_interest_score.owner_factory_weakness_threshold }
			}
			add_to_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.owner_factory_weakness }
		}
		if = {
			limit = {
				check_variable = { resources_found_foreign_interest_owner_stability < constant:resources_found_foreign_interest_score.owner_stability_weakness_threshold }
			}
			add_to_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.owner_instability }
		}
		if = {
			limit = {
				check_variable = { resources_found_foreign_interest_owner_surrender > constant:resources_found_foreign_interest_score.owner_surrender_weakness_threshold }
			}
			add_to_temp_variable = { resources_found_foreign_interest_field_score = constant:resources_found_foreign_interest_score.owner_surrender_weakness }
		}

		set_temp_variable = { resources_found_foreign_interest_best_score = constant:resources_found_foreign_interest_score.minimum_qualifying }
		every_country = {
			limit = {
				resources_found_decision_is_valid_foreign_partner = yes
				NOT = { has_capitulated = yes }
			}
			set_temp_variable = { resources_found_foreign_interest_score = resources_found_foreign_interest_field_score }
			resources_found_score_foreign_interest_candidate = yes
			if = {
				limit = {
					check_variable = {
						var = resources_found_foreign_interest_score
						value = constant:resources_found_foreign_interest_score.minimum_qualifying
						compare = greater_than_or_equals
					}
					OR = {
						NOT = { has_event_target = resources_found_foreign_interest_best_candidate_target }
						check_variable = {
							var = resources_found_foreign_interest_score
							value = resources_found_foreign_interest_best_score
							compare = greater_than
						}
					}
				}
				set_temp_variable = { resources_found_foreign_interest_best_score = resources_found_foreign_interest_score }
				save_event_target_as = resources_found_foreign_interest_best_candidate_target
			}
		}

		if = {
			limit = { has_event_target = resources_found_foreign_interest_best_candidate_target }
			event_target:resources_found_foreign_interest_best_candidate_target = {
				set_country_flag = resources_found_foreign_field_actor
				set_variable = { resources_found_foreign_interest_owner = event_target:resources_found_foreign_interest_owner_target }
				set_variable = { resources_found_foreign_interest_field = event_target:resources_found_foreign_interest_field_target }
				resources_found_refresh_decision_cost_previews = yes
			}
		}
	}
}
