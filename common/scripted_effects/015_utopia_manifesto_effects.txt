utopia_manifesto_prepare_candidate_weight = {
	set_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.base }

	if = {
		limit = { utopia_manifesto_candidate_is_weak_industrially = yes }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.weak_factory }
	}
	if = {
		limit = { utopia_manifesto_candidate_is_small = yes }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.small_state }
	}
	if = {
		limit = { utopia_manifesto_candidate_has_low_research = yes }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.low_research }
	}
	if = {
		limit = { utopia_manifesto_candidate_has_weak_infrastructure = yes }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.weak_infrastructure }
	}
	if = {
		limit = { utopia_manifesto_candidate_capital_is_secure = yes }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.secure_capital }
	}
	if = {
		limit = { is_subject = no }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.independent }
	}
	else = {
		subtract_from_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.subject_penalty }
	}
	if = {
		limit = { is_in_faction = no }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.no_faction }
	}
	if = {
		limit = { utopia_manifesto_candidate_is_coastal_or_island = yes }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.coastal_or_island }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_inland_island }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.inland_island }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_migration_pressure }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.migration_pressure }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_housing_pressure }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.housing_pressure }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_trade_pressure }
		add_to_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.trade_pressure }
	}
	if = {
		limit = { has_defensive_war = yes }
		subtract_from_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.defensive_war_penalty }
	}
	if = {
		limit = { num_occupied_states > constant:utopia_manifesto_math.zero }
		subtract_from_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.occupation_penalty }
	}
	if = {
		limit = { num_subjects > constant:utopia_manifesto_math.zero }
		subtract_from_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.subject_network_penalty }
	}
	if = {
		limit = {
			any_country = {
				is_major = yes
				has_guaranteed = PREV
			}
		}
		subtract_from_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.major_guarantor_penalty }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_recent_industrial_growth }
		subtract_from_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_score.recent_industrial_growth_penalty }
	}

	round_temp_variable = utopia_manifesto_candidate_weight
	clamp_temp_variable = {
		var = utopia_manifesto_candidate_weight
		min = constant:utopia_manifesto_candidate_limits.ticket_minimum
		max = constant:utopia_manifesto_candidate_limits.ticket_maximum
	}
}

utopia_manifesto_add_current_country_to_candidate_pool = {
	set_temp_variable = { utopia_manifesto_candidate_weight = constant:utopia_manifesto_candidate_limits.ticket_minimum }
	utopia_manifesto_prepare_candidate_weight = yes
	for_loop_effect = {
		start = constant:utopia_manifesto_narrative_count.zero
		end = utopia_manifesto_candidate_weight
		value = utopia_manifesto_candidate_ticket_index
		if = {
			limit = { check_variable = { utopia_manifesto_candidate_class = constant:utopia_manifesto_candidate_class.human_generic } }
			add_to_temp_array = { utopia_manifesto_class_one_candidates = THIS }
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_candidate_class = constant:utopia_manifesto_candidate_class.ai_generic } }
			add_to_temp_array = { utopia_manifesto_class_two_candidates = THIS }
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_candidate_class = constant:utopia_manifesto_candidate_class.approved_registry } }
			add_to_temp_array = { utopia_manifesto_class_three_candidates = THIS }
		}
	}
}

utopia_manifesto_prepare_random_event_fire = {
	set_temp_variable = { utopia_manifesto_prefire_candidate_class = constant:utopia_manifesto_candidate_class.none }
	clear_temp_array = utopia_manifesto_class_one_candidates
	clear_temp_array = utopia_manifesto_class_two_candidates
	clear_temp_array = utopia_manifesto_class_three_candidates

	every_country = {
		if = {
			limit = { utopia_manifesto_is_class_one_candidate = yes }
			set_temp_variable = { utopia_manifesto_candidate_class = constant:utopia_manifesto_candidate_class.human_generic }
			utopia_manifesto_add_current_country_to_candidate_pool = yes
		}
		else_if = {
			limit = { utopia_manifesto_is_class_two_candidate = yes }
			set_temp_variable = { utopia_manifesto_candidate_class = constant:utopia_manifesto_candidate_class.ai_generic }
			utopia_manifesto_add_current_country_to_candidate_pool = yes
		}
		else_if = {
			limit = { utopia_manifesto_is_class_three_candidate = yes }
			set_temp_variable = { utopia_manifesto_candidate_class = constant:utopia_manifesto_candidate_class.approved_registry }
			utopia_manifesto_add_current_country_to_candidate_pool = yes
		}
	}

	if = {
		limit = { check_variable = { utopia_manifesto_class_one_candidates^num > constant:utopia_manifesto_math.zero } }
		set_temp_variable = { utopia_manifesto_prefire_candidate_class = constant:utopia_manifesto_candidate_class.human_generic }
		random_scope_in_array = {
			array = utopia_manifesto_class_one_candidates
			save_event_target_as = utopia_manifesto_prefire_actor
			save_global_event_target_as = utopia_manifesto_latest_actor
			utopia_manifesto_evaluate_prefire_evolutions = yes
		}
	}
	else_if = {
		limit = { check_variable = { utopia_manifesto_class_two_candidates^num > constant:utopia_manifesto_math.zero } }
		set_temp_variable = { utopia_manifesto_prefire_candidate_class = constant:utopia_manifesto_candidate_class.ai_generic }
		random_scope_in_array = {
			array = utopia_manifesto_class_two_candidates
			save_event_target_as = utopia_manifesto_prefire_actor
			save_global_event_target_as = utopia_manifesto_latest_actor
			utopia_manifesto_evaluate_prefire_evolutions = yes
		}
	}
	else_if = {
		limit = { check_variable = { utopia_manifesto_class_three_candidates^num > constant:utopia_manifesto_math.zero } }
		set_temp_variable = { utopia_manifesto_prefire_candidate_class = constant:utopia_manifesto_candidate_class.approved_registry }
		random_scope_in_array = {
			array = utopia_manifesto_class_three_candidates
			save_event_target_as = utopia_manifesto_prefire_actor
			save_global_event_target_as = utopia_manifesto_latest_actor
			utopia_manifesto_evaluate_prefire_evolutions = yes
		}
	}

	clear_temp_array = utopia_manifesto_class_one_candidates
	clear_temp_array = utopia_manifesto_class_two_candidates
	clear_temp_array = utopia_manifesto_class_three_candidates
}

utopia_manifesto_clear_candidate_unavailability_reason_flags = {
	clr_country_flag = utopia_manifesto_unavailable_already_handled
	clr_country_flag = utopia_manifesto_unavailable_terminal_identity
	clr_country_flag = utopia_manifesto_unavailable_nonstandard_country
	clr_country_flag = utopia_manifesto_unavailable_major
	clr_country_flag = utopia_manifesto_unavailable_non_generic_tree
	clr_country_flag = utopia_manifesto_unavailable_active_event_package
	clr_country_flag = utopia_manifesto_unavailable_protected_package
	clr_country_flag = utopia_manifesto_unavailable_mature_tree
	clr_country_flag = utopia_manifesto_unavailable_civil_war
	clr_country_flag = utopia_manifesto_unavailable_offensive_war
	clr_country_flag = utopia_manifesto_unavailable_near_capitulation
	clr_country_flag = utopia_manifesto_unavailable_insecure_capital
	clr_country_flag = utopia_manifesto_unavailable_faction_leader
	clr_country_flag = utopia_manifesto_unavailable_industrial_power
	clr_country_flag = utopia_manifesto_unavailable_subject_empire
	clr_country_flag = utopia_manifesto_unavailable_extensive_occupation
	clr_country_flag = utopia_manifesto_unavailable_unsafe_subject
	clr_country_flag = utopia_manifesto_candidate_available
}

utopia_manifesto_refresh_candidate_unavailability_reason = {
	utopia_manifesto_clear_candidate_unavailability_reason_flags = yes
	if = {
		limit = {
			OR = {
				has_country_flag = utopia_manifesto_accepted
				has_country_flag = utopia_manifesto_rejected
				has_country_flag = utopia_manifesto_focus_tree_loaded
			}
		}
		set_country_flag = utopia_manifesto_unavailable_already_handled
	}
	else_if = {
		limit = { utopia_manifesto_candidate_has_terminal_identity = yes }
		set_country_flag = utopia_manifesto_unavailable_terminal_identity
	}
	else_if = {
		limit = { NOT = { uses_normal_civilian_systems = yes } }
		set_country_flag = utopia_manifesto_unavailable_nonstandard_country
	}
	else_if = {
		limit = { is_major = yes }
		set_country_flag = utopia_manifesto_unavailable_major
	}
	else_if = {
		limit = {
			NOT = {
				OR = {
					has_focus_tree = generic_focus
					utopia_manifesto_candidate_has_approved_light_tree = yes
				}
			}
		}
		set_country_flag = utopia_manifesto_unavailable_non_generic_tree
	}
	else_if = {
		limit = { utopia_manifesto_candidate_has_active_event_package = yes }
		set_country_flag = utopia_manifesto_unavailable_active_event_package
	}
	else_if = {
		limit = { utopia_manifesto_candidate_has_protected_package = yes }
		set_country_flag = utopia_manifesto_unavailable_protected_package
	}
	else_if = {
		limit = { utopia_manifesto_candidate_has_mature_generic_tree = yes }
		set_country_flag = utopia_manifesto_unavailable_mature_tree
	}
	else_if = {
		limit = { has_civil_war = yes }
		set_country_flag = utopia_manifesto_unavailable_civil_war
	}
	else_if = {
		limit = { has_offensive_war = yes }
		set_country_flag = utopia_manifesto_unavailable_offensive_war
	}
	else_if = {
		limit = { NOT = { surrender_progress < constant:utopia_manifesto_candidate_thresholds.maximum_surrender_progress } }
		set_country_flag = utopia_manifesto_unavailable_near_capitulation
	}
	else_if = {
		limit = { NOT = { utopia_manifesto_candidate_capital_is_secure = yes } }
		set_country_flag = utopia_manifesto_unavailable_insecure_capital
	}
	else_if = {
		limit = { utopia_manifesto_candidate_is_dominant_faction_leader = yes }
		set_country_flag = utopia_manifesto_unavailable_faction_leader
	}
	else_if = {
		limit = { num_of_factories > constant:utopia_manifesto_candidate_limits.maximum_total_factories }
		set_country_flag = utopia_manifesto_unavailable_industrial_power
	}
	else_if = {
		limit = { NOT = { num_subjects < constant:utopia_manifesto_candidate_limits.extensive_subject_count } }
		set_country_flag = utopia_manifesto_unavailable_subject_empire
	}
	else_if = {
		limit = { NOT = { num_occupied_states < constant:utopia_manifesto_candidate_limits.extensive_occupation_count } }
		set_country_flag = utopia_manifesto_unavailable_extensive_occupation
	}
	else_if = {
		limit = {
			is_subject = yes
			NOT = { utopia_manifesto_subject_candidate_is_safe = yes }
		}
		set_country_flag = utopia_manifesto_unavailable_unsafe_subject
	}
	else = {
		set_country_flag = utopia_manifesto_candidate_available
	}
}

utopia_manifesto_accept_manifesto = {
	if = {
		limit = { utopia_manifesto_manual_event_is_available = yes }
		set_country_flag = utopia_manifesto_accepted
		clr_country_flag = utopia_manifesto_rejected
		clr_country_flag = utopia_manifesto_kernel_disabled
		set_country_flag = utopia_manifesto_ledger_visible
		set_variable = { utopia_manifesto_route = constant:utopia_manifesto_route.unresolved }
		utopia_manifesto_initialize_ledger = yes
		utopia_manifesto_initialize_callings = yes
		utopia_manifesto_clear_league_runtime = yes
		utopia_manifesto_prepare_island_variant = yes
		utopia_manifesto_refresh_dynamic_costs = yes
		load_focus_tree = { tree = utopia_manifesto_tree keep_completed = no }
		set_country_flag = utopia_manifesto_focus_tree_loaded
		set_temp_variable = { utopia_manifesto_recent_transformation_duration = constant:utopia_manifesto_durations.recent_transformation_days }
		set_country_flag = {
			flag = utopia_manifesto_transformation_recent
			days = utopia_manifesto_recent_transformation_duration
		}
		utopia_manifesto_consume_prefire_evolution_state = yes
		utopia_manifesto_evaluate_active_evolutions = yes
	}
}

utopia_manifesto_reject_manifesto = {
	set_country_flag = utopia_manifesto_rejected
	clr_country_flag = utopia_manifesto_accepted
	utopia_manifesto_clear_all_runtime_state = yes
}

utopia_manifesto_initialize_ledger = {
	utopia_manifesto_clear_ledger_breakdown = yes
	set_country_flag = utopia_manifesto_ledger_visible
	set_variable = { utopia_need = constant:utopia_manifesto_ledger_defaults.need }
	set_variable = { utopia_plenty = constant:utopia_manifesto_ledger_defaults.plenty }
	set_variable = { utopia_concord = constant:utopia_manifesto_ledger_defaults.concord }
	set_variable = { utopia_assignment = constant:utopia_manifesto_ledger_defaults.assignment }
	set_variable = { utopia_reserve_score = constant:utopia_manifesto_ledger_defaults.reserve }
	set_variable = { utopia_manifesto_need_from_base = constant:utopia_manifesto_ledger_defaults.need }
	set_variable = { utopia_manifesto_plenty_from_base = constant:utopia_manifesto_ledger_defaults.plenty }
	set_variable = { utopia_manifesto_concord_from_base = constant:utopia_manifesto_ledger_defaults.concord }
	set_variable = { utopia_manifesto_assignment_from_base = constant:utopia_manifesto_ledger_defaults.assignment }

	if = {
		limit = { utopia_manifesto_candidate_is_weak_industrially = yes }
		set_variable = { utopia_manifesto_need_from_industry = constant:utopia_manifesto_ledger_contribution.weak_industry_need }
		set_variable = { utopia_manifesto_plenty_from_industry = constant:utopia_manifesto_ledger_contribution.weak_industry_plenty }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_industry }
		add_to_variable = { utopia_plenty = utopia_manifesto_plenty_from_industry }
	}
	else = {
		set_variable = { utopia_manifesto_need_from_industry = constant:utopia_manifesto_ledger_contribution.strong_industry_need }
		set_variable = { utopia_manifesto_plenty_from_industry = constant:utopia_manifesto_ledger_contribution.strong_industry_plenty }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_industry }
		add_to_variable = { utopia_plenty = utopia_manifesto_plenty_from_industry }
	}
	if = {
		limit = { utopia_manifesto_candidate_has_weak_infrastructure = yes }
		set_variable = { utopia_manifesto_need_from_infrastructure = constant:utopia_manifesto_ledger_contribution.poor_infrastructure_need }
		set_variable = { utopia_manifesto_plenty_from_infrastructure = constant:utopia_manifesto_ledger_contribution.poor_infrastructure_plenty }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_infrastructure }
		add_to_variable = { utopia_plenty = utopia_manifesto_plenty_from_infrastructure }
	}
	if = {
		limit = { has_war = yes }
		set_variable = { utopia_manifesto_need_from_war = constant:utopia_manifesto_ledger_contribution.war_need }
		set_variable = { utopia_manifesto_concord_from_war = constant:utopia_manifesto_ledger_contribution.war_concord }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_war }
		add_to_variable = { utopia_concord = utopia_manifesto_concord_from_war }
	}
	if = {
		limit = { num_occupied_states > constant:utopia_manifesto_math.zero }
		set_variable = { utopia_manifesto_need_from_occupation = constant:utopia_manifesto_ledger_contribution.occupation_need }
		set_variable = { utopia_manifesto_concord_from_occupation = constant:utopia_manifesto_ledger_contribution.occupation_concord }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_occupation }
		add_to_variable = { utopia_concord = utopia_manifesto_concord_from_occupation }
	}
	if = {
		limit = { utopia_manifesto_candidate_capital_is_secure = yes }
		set_variable = { utopia_manifesto_need_from_capital = constant:utopia_manifesto_ledger_contribution.secure_capital_need }
		set_variable = { utopia_manifesto_concord_from_capital = constant:utopia_manifesto_ledger_contribution.secure_capital_concord }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_capital }
		add_to_variable = { utopia_concord = utopia_manifesto_concord_from_capital }
	}
	if = {
		limit = { has_government = democratic }
		set_variable = { utopia_manifesto_concord_from_institutions = constant:utopia_manifesto_ledger_contribution.institutional_concord }
		set_variable = { utopia_manifesto_assignment_from_institutions = constant:utopia_manifesto_ledger_contribution.institutional_assignment }
		add_to_variable = { utopia_concord = utopia_manifesto_concord_from_institutions }
		add_to_variable = { utopia_assignment = utopia_manifesto_assignment_from_institutions }
	}
	else_if = {
		limit = {
			OR = {
				has_government = fascism
				has_government = communism
			}
		}
		set_variable = { utopia_manifesto_concord_from_institutions = constant:utopia_manifesto_ledger_contribution.authoritarian_concord }
		set_variable = { utopia_manifesto_assignment_from_institutions = constant:utopia_manifesto_ledger_contribution.authoritarian_assignment }
		add_to_variable = { utopia_concord = utopia_manifesto_concord_from_institutions }
		add_to_variable = { utopia_assignment = utopia_manifesto_assignment_from_institutions }
	}
	if = {
		limit = { is_subject = yes }
		set_variable = { utopia_manifesto_need_from_subject_status = constant:utopia_manifesto_ledger_contribution.subject_need }
		set_variable = { utopia_manifesto_concord_from_subject_status = constant:utopia_manifesto_ledger_contribution.subject_concord }
		add_to_variable = { utopia_need = utopia_manifesto_need_from_subject_status }
		add_to_variable = { utopia_concord = utopia_manifesto_concord_from_subject_status }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_migration_pressure }
		add_to_variable = { utopia_manifesto_need_from_pressure = constant:utopia_manifesto_ledger_contribution.migration_need }
		add_to_variable = { utopia_manifesto_concord_from_pressure = constant:utopia_manifesto_ledger_contribution.migration_concord }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_housing_pressure }
		add_to_variable = { utopia_manifesto_need_from_pressure = constant:utopia_manifesto_ledger_contribution.housing_need }
	}
	if = {
		limit = { has_country_flag = utopia_manifesto_trade_pressure }
		add_to_variable = { utopia_manifesto_need_from_pressure = constant:utopia_manifesto_ledger_contribution.trade_need }
		add_to_variable = { utopia_manifesto_plenty_from_pressure = constant:utopia_manifesto_ledger_contribution.trade_plenty }
	}
	if = { limit = { has_variable = utopia_manifesto_need_from_pressure } add_to_variable = { utopia_need = utopia_manifesto_need_from_pressure } }

					check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.joint_administration }

		limit = { utopia_manifesto_active_need_case_is_valid = yes }
		set_country_flag = utopia_manifesto_case_settlement_accepted
		if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.purchase } }
			utopia_manifesto_apply_purchase_settlement = yes
			utopia_manifesto_start_stewardship_from_active_case = yes
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.long_supply_contract } }
			utopia_manifesto_start_long_supply_contract_from_active_case = yes
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.lease } }
			utopia_manifesto_apply_lease_settlement = yes
			if = {
				limit = { check_variable = { utopia_manifesto_case_type = constant:utopia_manifesto_case_type.island_project_lease } }
				utopia_manifesto_start_island_project_lease_from_active_case = yes
			}
			else = {
				utopia_manifesto_start_stewardship_from_active_case = yes
			}
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.settlement_agreement } }
			utopia_manifesto_start_settlement_agreement_from_active_case = yes
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.joint_administration } }
			utopia_manifesto_apply_joint_administration_settlement = yes
			utopia_manifesto_start_stewardship_from_active_case = yes
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.association } }
			utopia_manifesto_start_association_duties_from_active_case = yes
		}
		else_if = {
			limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.ultimatum } }
			set_country_flag = utopia_manifesto_coercive_conduct_used
			set_country_flag = utopia_manifesto_conduct_ultimatum_settlement
			utopia_manifesto_apply_purchase_settlement = yes
			utopia_manifesto_start_stewardship_from_active_case = yes
		}
	}
}

utopia_manifesto_refuse_case_settlement = {

	}
}

utopia_manifesto_enforce_active_need_case = {
	if = {
		limit = {
			utopia_manifesto_active_need_case_is_valid = yes
			NOT = { has_country_flag = utopia_manifesto_case_wargoal_granted }
		}
		set_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.unilateral_enforcement }
		if = {
			limit = {
				check_variable = {
					var = utopia_manifesto_case_integrity
					value = constant:utopia_manifesto_case_limits.stewardship_minimum_integrity
					compare = less_than
				}
			}
			set_country_flag = utopia_manifesto_unjustified_enforced_case
		}
		for_each_scope_loop = {
			array = utopia_manifesto_active_case_targets
			save_event_target_as = utopia_manifesto_case_wargoal_target
			ROOT = {
				if = {
					limit = {
						NOT = {
							has_wargoal_against = {
								target = event_target:utopia_manifesto_case_wargoal_target
								type = utopia_manifesto_necessary_ground_take_state
							}
						}
					}
					create_wargoal = {
						type = utopia_manifesto_necessary_ground_take_state
						target = event_target:utopia_manifesto_case_wargoal_target
						generator = { has_state_flag = utopia_manifesto_active_case_state }
					}
				}
			}
		}
		set_country_flag = utopia_manifesto_case_enforced
		set_country_flag = utopia_manifesto_case_wargoal_granted
	}
}

utopia_manifesto_record_external_case_completion = {
	utopia_manifesto_record_achievement_peaceful_case_resolution = yes
	set_country_flag = utopia_manifesto_first_external_case_complete
	set_country_flag = utopia_manifesto_first_external_case_resolved
	set_variable = { utopia_manifesto_first_external_case_method = utopia_manifesto_case_method }
	if = {
		limit = { check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.association } }
		set_country_flag = utopia_manifesto_first_associate_recognized
		set_country_flag = utopia_manifesto_consensual_partner_proven
		for_each_scope_loop = {
			array = utopia_manifesto_active_case_targets
			if = {
				limit = { NOT = { is_in_array = { array = ROOT.utopia_manifesto_recognized_associates value = THIS } } }
				ROOT = { add_to_array = { array = utopia_manifesto_recognized_associates value = PREV } }
			}
			if = {
				limit = { NOT = { is_in_array = { array = ROOT.utopia_manifesto_recognized_external_partners value = THIS } } }
				ROOT = { add_to_array = { array = utopia_manifesto_recognized_external_partners value = PREV } }
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.purchase }
				check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.long_supply_contract }
				check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.lease }
				check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.joint_administration }
				check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.settlement_agreement }
			}
		}
		set_country_flag = utopia_manifesto_consensual_partner_proven
		for_each_scope_loop = {
			array = utopia_manifesto_active_case_targets
			if = {
				limit = { NOT = { is_in_array = { array = ROOT.utopia_manifesto_recognized_compacts value = THIS } } }
				ROOT = { add_to_array = { array = utopia_manifesto_recognized_compacts value = PREV } }
			}
			if = {
				limit = { NOT = { is_in_array = { array = ROOT.utopia_manifesto_recognized_external_partners value = THIS } } }
				ROOT = { add_to_array = { array = utopia_manifesto_recognized_external_partners value = PREV } }
			}
		}
	}
	if = { limit = { NOT = { has_variable = utopia_manifesto_external_case_count } } set_variable = { utopia_manifesto_external_case_count = constant:utopia_manifesto_math.zero } }
	add_to_variable = { utopia_manifesto_external_case_count = constant:utopia_manifesto_math.one }
	utopia_manifesto_refresh_league_state = yes
}

utopia_manifesto_restore_active_case_state_to_target = {
	for_each_scope_loop = {
		array = utopia_manifesto_active_case_targets
		save_event_target_as = utopia_manifesto_case_return_target
	}
	utopia_manifesto_clear_temporary_stewardship_state_modifiers = yes
	if = {
		limit = { has_event_target = utopia_manifesto_case_return_target }
		event_target:utopia_manifesto_case_return_target = {
			if = {
				limit = { has_country_flag = utopia_manifesto_case_created_access_to_founder }
				diplomatic_relation = { country = ROOT relation = military_access active = no }
				clr_country_flag = utopia_manifesto_case_created_access_to_founder
			}
		}
		if = {
			limit = { has_country_flag = utopia_manifesto_case_created_guarantee_to_target }
			diplomatic_relation = { country = event_target:utopia_manifesto_case_return_target relation = guarantee active = no }
			clr_country_flag = utopia_manifesto_case_created_guarantee_to_target
		}
		if = {
			limit = { has_country_flag = utopia_manifesto_case_created_reciprocal_access }
			diplomatic_relation = { country = event_target:utopia_manifesto_case_return_target relation = military_access active = no }
			clr_country_flag = utopia_manifesto_case_created_reciprocal_access
		}
		for_each_scope_loop = {
			array = utopia_manifesto_active_case_states
			if = {
				limit = {
					OR = {
						check_variable = { ROOT.utopia_manifesto_case_method = constant:utopia_manifesto_case_method.purchase }
						check_variable = { ROOT.utopia_manifesto_case_method = constant:utopia_manifesto_case_method.ultimatum }
						check_variable = { ROOT.utopia_manifesto_case_method = constant:utopia_manifesto_case_method.unilateral_enforcement }
					}
					is_owned_by = ROOT
				}
				transfer_state_to = event_target:utopia_manifesto_case_return_target
			}
			else_if = {
				limit = {
					OR = {
						check_variable = { ROOT.utopia_manifesto_case_method = constant:utopia_manifesto_case_method.lease }
						check_variable = { ROOT.utopia_manifesto_case_method = constant:utopia_manifesto_case_method.joint_administration }
					}
					owner = { has_country_flag = utopia_manifesto_active_case_target }
				}
				set_state_controller_to = event_target:utopia_manifesto_case_return_target
			}
			else_if = {
				limit = { NOT = { owner = { has_country_flag = utopia_manifesto_active_case_target } } }
				ROOT = { set_country_flag = utopia_manifesto_case_ownership_reconciliation_required }
			}
			clr_state_flag = utopia_manifesto_purchase_settlement_state
		}
	}
}

	if = {

		clamp_variable = {
			var = utopia_manifesto_league_cohesion
			min = constant:utopia_manifesto_league_limits.cohesion_minimum
			max = constant:utopia_manifesto_league_limits.cohesion_maximum
		}
	}
	set_variable = { utopia_manifesto_league_member_count = utopia_manifesto_league_members^num }
	set_variable = { utopia_manifesto_league_aid_count = utopia_manifesto_league_aid_targets^num }
	set_variable = { utopia_manifesto_league_reserve_count = utopia_manifesto_league_reserve_targets^num }
	set_variable = { utopia_manifesto_league_defense_count = utopia_manifesto_league_defense_targets^num }
	set_variable = { utopia_manifesto_league_refusal_count = utopia_manifesto_league_refusals^num }
	set_variable = { utopia_manifesto_league_exit_count = utopia_manifesto_league_exits^num }
	set_variable = { utopia_manifesto_league_shared_action_count = utopia_manifesto_league_aid_count }
	add_to_variable = { utopia_manifesto_league_shared_action_count = utopia_manifesto_league_reserve_count }
	add_to_variable = { utopia_manifesto_league_shared_action_count = utopia_manifesto_league_defense_count }
	clear_array = utopia_manifesto_external_network_members
	for_each_scope_loop = {
		array = utopia_manifesto_league_members
		ROOT = {
			add_to_array = {
				array = utopia_manifesto_external_network_members
				value = PREV
			}
		}
	}
	for_each_scope_loop = {
		array = utopia_manifesto_recognized_external_partners
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = ROOT.utopia_manifesto_external_network_members
						value = THIS
					}
				}
			}
			ROOT = {
				add_to_array = {
					array = utopia_manifesto_external_network_members
					value = PREV
				}
			}
		}
	}
	set_variable = { utopia_manifesto_external_network_count = utopia_manifesto_external_network_members^num }
	if = {
		limit = {
			check_variable = {
				var = utopia_manifesto_external_network_count
				value = constant:utopia_manifesto_league_limits.minimum_formation_network
				compare = greater_than_or_equals
			}
		}
		set_country_flag = utopia_manifesto_first_associate_recognized
	}
	if = {
		limit = {
