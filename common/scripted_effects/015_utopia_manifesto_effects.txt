utopia_manifesto_prepare_random_event_fire = {
	set_temp_variable = { utopia_manifesto_context_ready = 0 }
	utopia_manifesto_select_target = yes
	if = {
		limit = { check_variable = { utopia_manifesto_target_selected > 0 } }
		event_target:utopia_manifesto_target = {
			save_event_target_as = utopia_manifesto_latest_actor
		}
		set_temp_variable = { utopia_manifesto_context_ready = 1 }
	}
}

utopia_manifesto_select_target = {
	clear_temp_array = utopia_manifesto_target_pool
	set_temp_variable = { utopia_manifesto_target_pool_count = 0 }
	set_temp_variable = { utopia_manifesto_target_selected = 0 }
	if = {
		limit = { utopia_manifesto_has_dispatchable_target_available = yes }
		every_country = {
			limit = {
				OR = {
					is_valid_utopia_manifesto_player_target = yes
					is_valid_utopia_manifesto_automatic_target = yes
				}
			}
			utopia_manifesto_add_current_country_to_target_pool = yes
		}
	}
	if = {
		limit = { check_variable = { utopia_manifesto_target_pool_count > 0 } }
		random_scope_in_array = {
			array = utopia_manifesto_target_pool
			save_event_target_as = utopia_manifesto_target
			set_temp_variable = { utopia_manifesto_target_selected = 1 }
		}
	}
	clear_temp_array = utopia_manifesto_target_pool
	set_temp_variable = { utopia_manifesto_target_pool_count = 0 }
	set_temp_variable = { utopia_manifesto_target_pool_weight_remaining = 0 }
	set_temp_variable = { utopia_manifesto_target_weight = 0 }
}

utopia_manifesto_prepare_target_weight = {
	set_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.base }
	if = {
		limit = { is_ai = no }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.player_bonus }
	}
	if = {
		limit = { num_of_controlled_states < constant:utopia_manifesto_target_threshold.one_state_max }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.one_state_bonus }
	}
	else_if = {
		limit = { num_of_controlled_states < constant:utopia_manifesto_target_threshold.two_state_max }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.two_state_bonus }
	}
	else_if = {
		limit = { num_of_controlled_states < constant:utopia_manifesto_target_threshold.three_state_max }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.three_state_bonus }
	}
	if = {
		limit = { num_of_factories < constant:utopia_manifesto_target_threshold.small_factory_max }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.small_factory_bonus }
	}
	if = {
		limit = { is_subject = no }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.independent_bonus }
	}
	else = {
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.light_subject_bonus }
	}
	if = {
		limit = { utopia_manifesto_country_has_coast = yes }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.coastal_bonus }
	}
	if = {
		limit = { utopia_manifesto_country_island_capital = yes }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.island_bonus }
	}
	if = {
		limit = { utopia_manifesto_country_landlocked = yes }
		add_to_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.landlocked_floor_bonus }
	}
	if = {
		limit = { has_war = yes }
		subtract_from_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.at_war_penalty }
	}
	if = {
		limit = { has_stability < constant:utopia_manifesto_stability_threshold.low }
		subtract_from_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.low_stability_penalty }
	}
	if = {
		limit = {
			is_subject = yes
			has_war = yes
		}
		subtract_from_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.subject_at_war_penalty }
	}
	if = {
		limit = { num_of_controlled_states > constant:utopia_manifesto_target_threshold.strong_state_penalty_min }
		subtract_from_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.strong_state_penalty }
	}
	if = {
		limit = { num_of_factories > constant:utopia_manifesto_target_threshold.strong_factory_penalty_min }
		subtract_from_temp_variable = { utopia_manifesto_target_weight = constant:utopia_manifesto_target_weight.strong_factory_penalty }
	}
	round_temp_variable = utopia_manifesto_target_weight
	clamp_temp_variable = {
		var = utopia_manifesto_target_weight
		min = constant:utopia_manifesto_target_weight.min
		max = constant:utopia_manifesto_target_weight.max
	}
}

utopia_manifesto_add_current_country_to_target_pool = {
	utopia_manifesto_prepare_target_weight = yes
	set_temp_variable = { utopia_manifesto_target_pool_weight_remaining = utopia_manifesto_target_weight }
	while_loop_effect = {
		limit = { check_variable = { utopia_manifesto_target_pool_weight_remaining > 0 } }
		add_to_temp_array = { utopia_manifesto_target_pool = THIS }
		add_to_temp_variable = { utopia_manifesto_target_pool_count = 1 }
		subtract_from_temp_variable = { utopia_manifesto_target_pool_weight_remaining = 1 }
	}
	set_temp_variable = { utopia_manifesto_target_pool_weight_remaining = 0 }
	set_temp_variable = { utopia_manifesto_target_weight = 0 }
}

utopia_manifesto_initialize_ledger = {
	set_variable = { utopia_need = constant:utopia_manifesto_ledger_default.need }
	set_variable = { utopia_consent = constant:utopia_manifesto_ledger_default.consent }
	set_variable = { utopia_surplus = constant:utopia_manifesto_ledger_default.surplus }
	set_variable = { utopia_overreach = constant:utopia_manifesto_ledger_default.overreach }
	set_variable = { utopia_vocation_balance = constant:utopia_manifesto_ledger_default.vocation_balance }
	set_variable = { utopia_foreign_suspicion = constant:utopia_manifesto_ledger_default.foreign_suspicion }
	set_variable = { utopia_league_confidence = constant:utopia_manifesto_ledger_default.league_confidence }
	set_variable = { utopia_agriculture_share = constant:utopia_manifesto_ledger_default.agriculture_share }
	set_variable = { utopia_useful_arts_share = constant:utopia_manifesto_ledger_default.useful_arts_share }
	set_variable = { utopia_learning_share = constant:utopia_manifesto_ledger_default.learning_share }
	set_variable = { utopia_civic_service_share = constant:utopia_manifesto_ledger_default.civic_service_share }
	set_variable = { utopia_defense_share = constant:utopia_manifesto_ledger_default.defense_share }
	if = {
		limit = { has_war = yes }
		add_to_variable = { utopia_need = constant:utopia_manifesto_ledger_delta.war_need }
		add_to_variable = { utopia_foreign_suspicion = constant:utopia_manifesto_ledger_delta.military_suspicion }
	}
	if = {
		limit = { is_subject = yes }
		add_to_variable = { utopia_need = constant:utopia_manifesto_ledger_delta.subject_need }
	}
	if = {
		limit = { has_stability < constant:utopia_manifesto_stability_threshold.low }
		add_to_variable = { utopia_need = constant:utopia_manifesto_ledger_delta.low_stability_need }
		subtract_from_variable = { utopia_consent = constant:utopia_manifesto_ledger_delta.low_stability_consent_loss }
	}
	if = {
		limit = { has_stability > constant:utopia_manifesto_stability_threshold.high }
		add_to_variable = { utopia_consent = constant:utopia_manifesto_ledger_delta.high_stability_consent }
	}
	if = {
		limit = { utopia_manifesto_country_has_coast = yes }
		add_to_variable = { utopia_surplus = constant:utopia_manifesto_ledger_delta.coastal_surplus }
	}
	if = {
		limit = { utopia_manifesto_country_landlocked = yes }
		add_to_variable = { utopia_need = constant:utopia_manifesto_ledger_delta.landlocked_need }
	}
	if = {
		limit = { num_of_factories > constant:utopia_manifesto_target_threshold.initial_surplus_factory_step_1 }
		add_to_variable = { utopia_surplus = constant:utopia_manifesto_ledger_delta.factory_surplus_step }
	}
	if = {
		limit = { num_of_factories > constant:utopia_manifesto_target_threshold.initial_surplus_factory_step_2 }
		add_to_variable = { utopia_surplus = constant:utopia_manifesto_ledger_delta.factory_surplus_step }
	}
	utopia_manifesto_refresh_ledger = yes
}

		set_state_flag = utopia_manifesto_compensated_settlement
		owner = {

			set_country_flag = utopia_manifesto_boundary_compensation_accepted_from_root
		}
		ROOT = {
			set_country_flag = utopia_manifesto_boundary_arbitration_claim_created_this_mission
			add_to_variable = { utopia_manifesto_arbitration_compensations = 1 }
			set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.small }
			multiply_temp_variable = { utopia_manifesto_ledger_delta = -1 }
			utopia_manifesto_add_surplus = yes
			set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.small }
			utopia_manifesto_add_consent = yes
		}
	}
	else_if = {
		limit = {
			ROOT = {
				utopia_manifesto_overreach_safe = yes
				OR = {
					utopia_manifesto_consent_stable = yes
					has_completed_focus = utopia_commonwealth_observers
					has_completed_focus = utopia_no_secret_empire
				}
			}
		}
		add_claim_by = ROOT
		set_state_flag = utopia_manifesto_needful_land_claimed
		set_state_flag = utopia_manifesto_arbitrated_claim
		set_state_flag = utopia_manifesto_guaranteed_settlement_charter
		owner = {
			add_opinion_modifier = { target = ROOT modifier = utopia_manifesto_boundary_guarantee }
			set_country_flag = utopia_manifesto_boundary_guarantee_charter_from_root
			ROOT = {
				if = {
					limit = { NOT = { has_guaranteed = PREV } }
					give_guarantee = PREV
				}
			}
		}
		ROOT = {
			set_country_flag = utopia_manifesto_boundary_arbitration_claim_created_this_mission
			add_to_variable = { utopia_manifesto_arbitration_guarantees = 1 }
			set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.small }
			utopia_manifesto_add_league_confidence = yes
			set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.small }
			multiply_temp_variable = { utopia_manifesto_ledger_delta = -1 }
			utopia_manifesto_add_foreign_suspicion = yes
		}
	}
	else = {
		set_state_flag = utopia_manifesto_boundary_public_refusal
		owner = {
			if = {
				limit = { NOT = { has_country_flag = utopia_manifesto_boundary_arbitration_refused_root } }
				add_opinion_modifier = { target = ROOT modifier = utopia_manifesto_boundary_refusal }
			}
			set_country_flag = utopia_manifesto_boundary_arbitration_refused_root
		}
		ROOT = {
			set_country_flag = utopia_manifesto_failed_arbitration
			add_to_variable = { utopia_manifesto_arbitration_refusals = 1 }
		}
	}
}

		add_to_array = {
			array = ROOT.utopia_manifesto_league_aid_targets
		}
	}
	activate_mission = mission_utopia_league_aid_corridor
}

utopia_manifesto_resolve_league_aid_mission = {
	if = {
		limit = {
			utopia_manifesto_league_aid_objective_ready = yes
		}
		for_each_scope_loop = {
			array = utopia_manifesto_league_aid_targets
			if = {
				limit = { utopia_manifesto_league_target_valid = yes }
				add_equipment_to_stockpile = { type = support_equipment_1 amount = constant:utopia_manifesto_decision_cost.league_aid_support_equipment }
				set_country_flag = utopia_manifesto_league_aid_from_root
				utopia_manifesto_set_relationship_league_member_from_prev = yes
				if = {
					limit = {
						NOT = {
							is_in_array = {
								array = ROOT.utopia_manifesto_league_member_countries
								value = THIS
							}
						}
					}
					add_to_array = {
						array = ROOT.utopia_manifesto_league_member_countries
					}
					ROOT = { add_to_variable = { utopia_manifesto_league_member_count = 1 } }
				}
			}
		}
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.medium }
		utopia_manifesto_add_consent = yes
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.small }
		multiply_temp_variable = { utopia_manifesto_ledger_delta = -1 }
		utopia_manifesto_add_foreign_suspicion = yes
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.medium }
		utopia_manifesto_add_league_confidence = yes
	}
	else = {
		set_country_flag = utopia_manifesto_league_aid_failed
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.medium }
		utopia_manifesto_add_foreign_suspicion = yes
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.medium }
		multiply_temp_variable = { utopia_manifesto_ledger_delta = -1 }
		utopia_manifesto_add_league_confidence = yes
	}
	clr_country_flag = utopia_manifesto_league_aid_mission_active
	utopia_manifesto_clear_league_aid_targets = yes
	utopia_manifesto_maybe_check_achievements = yes
}

utopia_manifesto_clear_league_aid_targets = {
	clear_array = utopia_manifesto_league_aid_targets
	clr_country_flag = utopia_manifesto_league_aid_mission_active
}

utopia_manifesto_start_renunciation_vote_mission = {

	activate_mission = mission_utopia_renunciation_vote
}

utopia_manifesto_resolve_renunciation_vote_mission = {
	clr_country_flag = utopia_manifesto_renunciation_vote_active
	if = {
		limit = {
			utopia_manifesto_consent_stable = yes
			NOT = { utopia_manifesto_overreach_high = yes }
		}
		clr_country_flag = utopia_manifesto_marked_bounds_active
		set_country_flag = achievement_utopia_renounced_bounds_ready
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.large }
		multiply_temp_variable = { utopia_manifesto_ledger_delta = -1 }
		utopia_manifesto_add_overreach = yes
	}
	else = {
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.medium }
		utopia_manifesto_add_need = yes
		set_temp_variable = { utopia_manifesto_ledger_delta = constant:utopia_manifesto_ledger_delta.small }
		multiply_temp_variable = { utopia_manifesto_ledger_delta = -1 }
		utopia_manifesto_add_consent = yes
	}
	utopia_manifesto_maybe_check_achievements = yes
}

		}
