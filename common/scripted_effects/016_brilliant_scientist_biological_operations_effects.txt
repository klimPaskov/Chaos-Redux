# COUNTRY scope. Inputs: brilliant_scientist_biological_agent_request and
# brilliant_scientist_biological_payload_multiplier. Output:
# brilliant_scientist_biological_payload_required.
brilliant_scientist_calculate_biological_payload_requirement = {
	set_temp_variable = { brilliant_scientist_biological_payload_required = 0 }
	if = {
		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.anthrax } }
		set_temp_variable = { brilliant_scientist_biological_payload_required = constant:brilliant_scientist_biological_payload.anthrax }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.plague } }
		set_temp_variable = { brilliant_scientist_biological_payload_required = constant:brilliant_scientist_biological_payload.plague }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.tularemia } }
		set_temp_variable = { brilliant_scientist_biological_payload_required = constant:brilliant_scientist_biological_payload.tularemia }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.smallpox } }
		set_temp_variable = { brilliant_scientist_biological_payload_required = constant:brilliant_scientist_biological_payload.smallpox }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.weaponized_zombies } }
		set_temp_variable = { brilliant_scientist_biological_payload_required = constant:brilliant_scientist_biological_payload.weaponized_zombies }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.black_plague } }
		set_temp_variable = { brilliant_scientist_biological_payload_required = constant:brilliant_scientist_biological_payload.black_plague }
	}
	if = {
		limit = { check_variable = { brilliant_scientist_biological_payload_multiplier > 0 } }
		multiply_temp_variable = { brilliant_scientist_biological_payload_required = brilliant_scientist_biological_payload_multiplier }
	}
}

# COUNTRY scope. Inputs: brilliant_scientist_biological_agent_request and
# brilliant_scientist_biological_payload_amount. Positive amounts add/refund;
# negative amounts debit. No numeric quantity ledger is created.

		add_equipment_to_stockpile = { type = tularemia_bomb_1 amount = brilliant_scientist_biological_payload_amount }

		add_equipment_to_stockpile = { type = smallpox_bomb_1 amount = brilliant_scientist_biological_payload_amount }

		limit = { check_variable = { brilliant_scientist_biological_agent_request = constant:brilliant_scientist_biological_agent.black_plague } }

}

# Private decision-start callback, COUNTRY scope. Input
# brilliant_scientist_biological_production_multiplier_request. The decision
# checks and occupies the factory budget; this callback rechecks consumable
# inputs without counting the decision's already-activated modifier twice.
brilliant_scientist_begin_biological_production = {
	if = {
		limit = {
			OR = {
				AND = {
					check_variable = { brilliant_scientist_biological_production_multiplier_request = constant:brilliant_scientist_biological_production.single_multiplier }
					brilliant_scientist_can_begin_single_biological_production = yes
				}
				AND = {
					check_variable = { brilliant_scientist_biological_production_multiplier_request = constant:brilliant_scientist_biological_production.triple_multiplier }
					brilliant_scientist_can_begin_triple_biological_production = yes
				}
			}
		}
		set_country_flag = brilliant_scientist_biological_production_active
		set_variable = { brilliant_scientist_biological_production_agent = brilliant_scientist_selected_biological_agent }
		set_variable = { brilliant_scientist_biological_production_multiplier = brilliant_scientist_biological_production_multiplier_request }
		set_temp_variable = { brilliant_scientist_biological_agent_request = brilliant_scientist_biological_production_agent }
		set_temp_variable = { brilliant_scientist_biological_payload_multiplier = brilliant_scientist_biological_production_multiplier }
		brilliant_scientist_calculate_biological_payload_requirement = yes
		set_variable = { brilliant_scientist_biological_production_output = brilliant_scientist_biological_payload_required }
		if = {
			limit = { check_variable = { brilliant_scientist_biological_production_multiplier = constant:brilliant_scientist_biological_production.single_multiplier } }
			add_equipment_to_stockpile = { type = support_equipment amount = constant:brilliant_scientist_biological_production.single_support_equipment_spend }
			add_manpower = constant:brilliant_scientist_biological_production.single_manpower_spend
		}
		else = {
			add_equipment_to_stockpile = { type = support_equipment amount = constant:brilliant_scientist_biological_production.triple_support_equipment_spend }
			add_manpower = constant:brilliant_scientist_biological_production.triple_manpower_spend
		}
	}
}

brilliant_scientist_complete_biological_production = {
	if = {
		limit = {
			has_country_flag = brilliant_scientist_biological_production_active
			brilliant_scientist_biological_operations_country_is_active = yes
			brilliant_scientist_selected_biological_agent_is_unlocked = yes
		}
		set_temp_variable = { brilliant_scientist_biological_agent_request = brilliant_scientist_biological_production_agent }
		set_temp_variable = { brilliant_scientist_biological_payload_amount = brilliant_scientist_biological_production_output }
		brilliant_scientist_change_biological_payload_stockpile = yes
		set_country_flag = brilliant_scientist_biological_production_history
		add_to_variable = { brilliant_scientist_biological_payload_lots_produced = brilliant_scientist_biological_production_multiplier }
		brilliant_scientist_clear_biological_production_transaction = yes
	}
	else = { brilliant_scientist_clear_biological_production_transaction = yes }
}

brilliant_scientist_clear_biological_production_transaction = {

	clear_variable = brilliant_scientist_biological_production_agent
	clear_variable = brilliant_scientist_biological_production_multiplier
	clear_variable = brilliant_scientist_biological_production_output
}

# COUNTRY scope with FROM = selected state. Input route request.
brilliant_scientist_begin_biological_deployment = {
	if = {
		limit = {
			OR = {
				AND = {
					check_variable = { brilliant_scientist_biological_route_request = constant:brilliant_scientist_biological_route.battlefield }
					brilliant_scientist_biological_battlefield_deployment_is_available = yes
					FROM = { brilliant_scientist_biological_battlefield_target_is_valid = yes }
				}
				AND = {
					check_variable = { brilliant_scientist_biological_route_request = constant:brilliant_scientist_biological_route.covert }
					brilliant_scientist_biological_covert_deployment_is_available = yes
					FROM = { brilliant_scientist_biological_covert_target_is_valid = yes }
				}
			}
		}
		set_country_flag = brilliant_scientist_biological_deployment_pending
		set_country_flag = brilliant_scientist_biological_deployment_payload_debited
		set_variable = { brilliant_scientist_biological_deployment_agent = brilliant_scientist_selected_biological_agent }
		set_variable = { brilliant_scientist_biological_deployment_route = brilliant_scientist_biological_route_request }
		set_variable = { brilliant_scientist_biological_deployment_target_state = FROM }
		set_variable = { brilliant_scientist_biological_deployment_transport_amount = constant:brilliant_scientist_value.zero }
		FROM = { CONTROLLER = { ROOT = { set_variable = { brilliant_scientist_biological_deployment_victim = PREV } } } }
		set_temp_variable = { brilliant_scientist_biological_agent_request = brilliant_scientist_biological_deployment_agent }
		if = {
			limit = { check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield } }
			set_temp_variable = { brilliant_scientist_biological_payload_multiplier = constant:brilliant_scientist_biological_deployment.battlefield_payload_multiplier }
			add_command_power = constant:brilliant_scientist_biological_deployment.battlefield_command_power_spend
			if = {
				limit = { FROM = { NOT = { brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes } } }
				set_variable = { brilliant_scientist_biological_deployment_transport_amount = constant:brilliant_scientist_biological_deployment.portal_transport_equipment }
				add_equipment_to_stockpile = { type = teleportation_equipment_1 amount = constant:brilliant_scientist_biological_deployment.portal_transport_equipment_spend }
			}
		}
		else = {
			set_temp_variable = { brilliant_scientist_biological_payload_multiplier = constant:brilliant_scientist_biological_deployment.covert_payload_multiplier }
			add_command_power = constant:brilliant_scientist_biological_deployment.covert_command_power_spend
		}
		brilliant_scientist_calculate_biological_payload_requirement = yes
		set_variable = { brilliant_scientist_biological_deployment_payload_amount = brilliant_scientist_biological_payload_required }
		set_temp_variable = { brilliant_scientist_biological_payload_amount = brilliant_scientist_biological_payload_required }
		multiply_temp_variable = { brilliant_scientist_biological_payload_amount = -1 }
		brilliant_scientist_change_biological_payload_stockpile = yes
	}
}

# COUNTRY scope on the annexing country, FROM = annexed actor. Reserved cargo
# is absent from the native stockpile transfer. Capture it once, then clear the
# former actor's receipt before any delayed decision callback can run.
brilliant_scientist_seize_annexed_biological_reservation = {
	if = {
		limit = {
			exists = yes
			FROM = {
				has_country_flag = brilliant_scientist_biological_deployment_pending
				has_country_flag = brilliant_scientist_biological_deployment_payload_debited
			}
		}
		FROM = {
			set_temp_variable = { brilliant_scientist_biological_agent_request = brilliant_scientist_biological_deployment_agent }
			set_temp_variable = { brilliant_scientist_biological_payload_amount = brilliant_scientist_biological_deployment_payload_amount }
			set_temp_variable = { brilliant_scientist_biological_captured_transport = brilliant_scientist_biological_deployment_transport_amount }
			brilliant_scientist_clear_biological_deployment_transaction = yes
			set_country_flag = brilliant_scientist_biological_reservation_seized_history
		}
		brilliant_scientist_change_biological_payload_stockpile = yes
		if = {
			limit = { check_variable = { brilliant_scientist_biological_captured_transport > constant:brilliant_scientist_value.zero } }
			add_equipment_to_stockpile = { type = teleportation_equipment_1 amount = brilliant_scientist_biological_captured_transport }
		}
		set_country_flag = brilliant_scientist_biological_reservation_capture_history
	}
	FROM = {
		brilliant_scientist_clear_biological_production_transaction = yes
		clr_country_flag = brilliant_scientist_biological_staging_directive_active
	}
}

# COUNTRY scope. Inputs: release target state, release route, and accident flag.
brilliant_scientist_dispatch_standard_biological_release = {
	bio_lifecycle_reset_seed_record = yes
	if = {
		limit = { check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.anthrax } }
		set_temp_variable = { bio_seed_agent = constant:bio_lifecycle_agent.anthrax }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.tularemia } }
		set_temp_variable = { bio_seed_agent = constant:bio_lifecycle_agent.tularemia }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.smallpox } }
		set_temp_variable = { bio_seed_agent = constant:bio_lifecycle_agent.smallpox }
	}
	else = { set_temp_variable = { bio_seed_agent = constant:bio_lifecycle_agent.plague } }
	if = {
		limit = { check_variable = { brilliant_scientist_biological_release_is_accident > 0 } }
		set_temp_variable = { bio_seed_route = constant:bio_lifecycle_route.laboratory_accident }
		set_temp_variable = { bio_seed_source = constant:bio_lifecycle_source.accident }
		set_temp_variable = { bio_seed_result = constant:bio_lifecycle_result.accident }
	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield } }
		set_temp_variable = { bio_seed_route = constant:bio_lifecycle_route.battlefield_dissemination }
		set_temp_variable = { bio_seed_source = constant:bio_lifecycle_source.deliberate }
		set_temp_variable = { bio_seed_result = constant:bio_lifecycle_result.success }
	}
	else = {
		set_temp_variable = { bio_seed_route = constant:bio_lifecycle_route.operative_release }
		set_temp_variable = { bio_seed_source = constant:bio_lifecycle_source.deliberate }
		set_temp_variable = { bio_seed_result = constant:bio_lifecycle_result.catastrophic }
	}
	set_temp_variable = { bio_seed_actor_proof = constant:bio_lifecycle_proof.supplied }
	set_temp_variable = { bio_seed_victim_proof = constant:bio_lifecycle_proof.supplied }
	set_temp_variable = { bio_seed_payload_debit_proof = constant:bio_lifecycle_proof.supplied }
	set_temp_variable = { bio_seed_use_history_proof = constant:bio_lifecycle_proof.supplied }
	set_temp_variable = { bio_seed_payload_required = brilliant_scientist_biological_deployment_payload_amount }
	set_temp_variable = { bio_seed_payload_consumed = brilliant_scientist_biological_deployment_payload_amount }
	save_event_target_as = bio_seed_actor
	var:brilliant_scientist_biological_release_target_state = {
		CONTROLLER = { save_event_target_as = bio_seed_victim }
		bio_lifecycle_dispatch_seed = yes
	}
	set_temp_variable = { brilliant_scientist_biological_release_dispatched = bio_seed_dispatch_status }
}

brilliant_scientist_dispatch_weaponized_zombie_release = {
	set_temp_variable = { actor_country = THIS }
	set_temp_variable = { target_state = var:brilliant_scientist_biological_release_target_state }
	if = {
		limit = { check_variable = { brilliant_scientist_biological_release_is_accident > constant:brilliant_scientist_value.zero } }
		set_temp_variable = { victim_country = THIS }
	}
	else = { set_temp_variable = { victim_country = var:brilliant_scientist_biological_deployment_victim } }
	if = {
		limit = { check_variable = { brilliant_scientist_biological_release_is_accident = constant:brilliant_scientist_value.zero } }
		apply_weaponized_zombie_strike_consequences = yes
	}
	var:brilliant_scientist_biological_release_target_state = {
		PREV = { spawn_weaponized_zombie_outbreak_from_creator = yes }
		if = {
			limit = { OWNER = { tag = var:weaponized_zombie_new_country } }
			set_temp_variable = { brilliant_scientist_biological_release_dispatched = constant:bio_lifecycle_proof.supplied }
		}
	}
}

brilliant_scientist_dispatch_black_plague_release = {
	brilliant_scientist_dispatch_standard_biological_release = yes
	if = {
		limit = { check_variable = { brilliant_scientist_biological_release_dispatched = constant:bio_lifecycle_proof.supplied } }
		var:brilliant_scientist_biological_release_target_state = {
			if = {
				limit = { check_variable = { ROOT.brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield } }
				set_temp_variable = { black_plague_exposure_amount = constant:brilliant_scientist_biological_deployment.black_plague_battlefield_exposure }
			}
			else = { set_temp_variable = { black_plague_exposure_amount = constant:brilliant_scientist_biological_deployment.black_plague_covert_exposure } }
			set_temp_variable = { black_plague_exposure_route = constant:black_plague_route.biological_strike }
			set_temp_variable = { black_plague_exposure_provenance = constant:black_plague_provenance.weaponized }
			black_plague_apply_exposure = yes
			set_state_flag = black_plague_weaponized_provenance
		}
		set_country_flag = black_plague_weaponization_delivery_history
	}
}

brilliant_scientist_apply_selected_biological_release = {
	set_temp_variable = { brilliant_scientist_biological_release_dispatched = constant:bio_lifecycle_proof.missing }

	}
	else_if = {
		limit = { check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.black_plague } }
		brilliant_scientist_dispatch_black_plague_release = yes
	}
	else = { brilliant_scientist_dispatch_standard_biological_release = yes }
}

brilliant_scientist_settle_biological_deployment = {
	if = {
		limit = { has_country_flag = brilliant_scientist_biological_deployment_pending }
		set_country_flag = brilliant_scientist_biological_deployment_history
		set_variable = { brilliant_scientist_biological_last_deployment_agent = brilliant_scientist_biological_deployment_agent }
		set_variable = { brilliant_scientist_biological_last_deployment_route = brilliant_scientist_biological_deployment_route }
		add_to_variable = { brilliant_scientist_biological_deployment_count = 1 }
		if = {
			limit = { check_variable = { brilliant_scientist_biological_deployment_transport_amount > constant:brilliant_scientist_value.zero } }
			set_country_flag = brilliant_scientist_biological_portal_deployment_history
			var:brilliant_scientist_biological_deployment_target_state = { set_state_flag = brilliant_scientist_biological_portal_target_history }
		}
		brilliant_scientist_clear_biological_deployment_transaction = yes
	}
}

brilliant_scientist_resolve_biological_deployment = {
	if = {
		limit = { brilliant_scientist_biological_deployment_transaction_is_valid = yes }
		if = {
			limit = { check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield } }
