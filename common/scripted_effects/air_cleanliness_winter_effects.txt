# ============================================================================
# CHAOS REDUX - AIR CLEANLINESS WINTER SCRIPTED EFFECTS
# ============================================================================
# Deterministic seven-phase state lifecycle for Air Cleanliness winter damage.
# Host integration owns the existing monthly state iteration and must call:
#   air_winter_begin_monthly_cycle = yes       (COUNTRY/host scope)
#   air_winter_update_state = yes              (once per STATE)
#   air_winter_finalize_monthly_cycle = yes    (COUNTRY/host scope)
# Every normal lifecycle entry point is date-idempotent. This file creates no
# on_action and selects no random states. Its dedicated Fallout shutdown helper
# performs one bounded cleanup sweep after the consequence lock and never calls
# the Fallout request coordinator.
# ============================================================================

# ---------------------------------------------------------------------------
# Global begin/finalize contract
# ---------------------------------------------------------------------------

# Supported scope: COUNTRY (the existing host country)
air_winter_initialize_global = {
	if = { limit = { NOT = { has_variable = global.air_winter_pressure } } set_variable = { global.air_winter_pressure = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_recovery_pressure } } set_variable = { global.air_winter_recovery_pressure = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_valid_state_count } } set_variable = { global.air_winter_valid_state_count = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_active_state_count } } set_variable = { global.air_winter_active_state_count = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_severe_state_count } } set_variable = { global.air_winter_severe_state_count = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_terminal_state_count } } set_variable = { global.air_winter_terminal_state_count = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_unclassified_state_count } } set_variable = { global.air_winter_unclassified_state_count = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_population_exposed } } set_variable = { global.air_winter_population_exposed = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_peak_phase } } set_variable = { global.air_winter_peak_phase = constant:air_winter_phase.clear } }
	if = { limit = { NOT = { has_variable = global.air_winter_cycle_id } } set_variable = { global.air_winter_cycle_id = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_cycle_contamination_bp_snapshot } } set_variable = { global.air_winter_cycle_contamination_bp_snapshot = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_cycle_air_delta_bp_snapshot } } set_variable = { global.air_winter_cycle_air_delta_bp_snapshot = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_cycle_chemical_states_snapshot } } set_variable = { global.air_winter_cycle_chemical_states_snapshot = constant:air_winter_runtime.zero } }
	if = { limit = { NOT = { has_variable = global.air_winter_cycle_fallout_states_snapshot } } set_variable = { global.air_winter_cycle_fallout_states_snapshot = constant:air_winter_runtime.zero } }
}

# Supported scope: COUNTRY (the existing host country)
air_winter_begin_monthly_cycle = {
	air_winter_initialize_global = yes
	air_winter_prepare_regional_visual_cycle = yes

	if = {
		limit = {
			OR = {
				NOT = { has_variable = global.air_winter_last_prepare_date }
				NOT = {
					check_variable = {
						var = global.air_winter_last_prepare_date
						value = global.date
						compare = equals
					}
				}
			}
		}

		set_variable = { global.air_winter_last_prepare_date = global.date }
		add_to_variable = { global.air_winter_cycle_id = constant:air_winter_runtime.one }
		set_global_flag = air_winter_cycle_open
		air_winter_event_prepare_candidate_cycle = yes
		air_winter_response_prepare_evacuation_cache_cycle = yes

		set_variable = { global.air_winter_cycle_contamination_bp_snapshot = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_cycle_air_delta_bp_snapshot = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_cycle_chemical_states_snapshot = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_cycle_fallout_states_snapshot = constant:air_winter_runtime.zero }
		if = {
			limit = { has_variable = global.air_contamination_bp }
			set_variable = { global.air_winter_cycle_contamination_bp_snapshot = global.air_contamination_bp }
		}
		if = {
			limit = { has_variable = global.air_contamination_last_delta_bp }
			set_variable = { global.air_winter_cycle_air_delta_bp_snapshot = global.air_contamination_last_delta_bp }
		}
		if = {
			limit = { has_variable = global.air_contamination_chem_states }
			set_variable = { global.air_winter_cycle_chemical_states_snapshot = global.air_contamination_chem_states }
		}
		if = {
			limit = { has_variable = global.air_contamination_fallout_states }
			set_variable = { global.air_winter_cycle_fallout_states_snapshot = global.air_contamination_fallout_states }
		}

		set_variable = { global.air_winter_valid_state_count = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_active_state_count = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_severe_state_count = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_terminal_state_count = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_unclassified_state_count = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_population_exposed = constant:air_winter_runtime.zero }
		set_variable = { global.air_winter_peak_phase = constant:air_winter_phase.clear }

		if = {
			limit = { air_winter_system_enabled = yes }
			set_variable = { global.air_winter_pressure = global.air_winter_cycle_contamination_bp_snapshot }
			divide_variable = { global.air_winter_pressure = constant:air_winter_global.pressure_bp_divisor }

			if = {
				limit = { check_variable = { global.air_winter_cycle_fallout_states_snapshot > constant:air_winter_runtime.zero } }
				set_temp_variable = { air_winter_global_fallout_pressure = global.air_winter_cycle_fallout_states_snapshot }
				multiply_temp_variable = { air_winter_global_fallout_pressure = constant:air_winter_global.fallout_state_pressure }
				add_to_variable = { global.air_winter_pressure = air_winter_global_fallout_pressure }
			}
			if = {
				limit = { check_variable = { global.air_winter_cycle_chemical_states_snapshot > constant:air_winter_runtime.zero } }
				set_temp_variable = { air_winter_global_chemical_pressure = global.air_winter_cycle_chemical_states_snapshot }
				multiply_temp_variable = { air_winter_global_chemical_pressure = constant:air_winter_global.chemical_state_pressure }
				add_to_variable = { global.air_winter_pressure = air_winter_global_chemical_pressure }
			}
			if = {
				limit = { has_global_flag = air_contamination_threshold_75 }
				add_to_variable = { global.air_winter_pressure = constant:air_winter_global.legacy_winter_severe_pressure }
			}
			else_if = {
				limit = { has_global_flag = air_contamination_threshold_50 }
				add_to_variable = { global.air_winter_pressure = constant:air_winter_global.legacy_winter_mild_pressure }
			}
			if = {
				limit = { has_global_flag = air_contamination_final_silence_locked }
				add_to_variable = { global.air_winter_pressure = constant:air_winter_global.terminal_lock_pressure }
			}
			if = {
				limit = { has_variable = global.air_winter_global_mitigation }
				subtract_from_variable = { global.air_winter_pressure = global.air_winter_global_mitigation }
			}
			clamp_variable = {
				var = global.air_winter_pressure
				min = constant:air_winter_runtime.zero
				max = constant:air_winter_global.maximum_pressure
			}

			set_variable = { global.air_winter_recovery_pressure = constant:air_winter_global.base_recovery_pressure }
			if = {
				limit = {
					check_variable = {
						var = global.air_winter_pressure
						value = constant:air_winter_phase_entry.phase_1
						compare = less_than
					}
				}
				add_to_variable = { global.air_winter_recovery_pressure = constant:air_winter_global.clear_air_recovery_bonus }
			}
			if = {
				limit = {
					check_variable = { global.air_winter_cycle_air_delta_bp_snapshot < constant:air_winter_runtime.zero }
				}
				set_temp_variable = { air_winter_negative_air_delta = global.air_winter_cycle_air_delta_bp_snapshot }
				multiply_temp_variable = { air_winter_negative_air_delta = constant:air_winter_runtime.negative_one }
				divide_temp_variable = { air_winter_negative_air_delta = constant:air_winter_global.negative_delta_bp_divisor }
				add_to_variable = { global.air_winter_recovery_pressure = air_winter_negative_air_delta }
			}
			if = {
				limit = { has_variable = global.air_winter_global_recovery_bonus }
				add_to_variable = { global.air_winter_recovery_pressure = global.air_winter_global_recovery_bonus }
			}
			clamp_variable = {
				var = global.air_winter_recovery_pressure
				min = constant:air_winter_runtime.zero
				max = constant:air_winter_global.maximum_pressure
			}
		}
		else = {
			set_variable = { global.air_winter_pressure = constant:air_winter_runtime.zero }
			set_variable = { global.air_winter_recovery_pressure = constant:air_winter_runtime.zero }
		}
	}
}

# Supported scope: COUNTRY (the existing host country)
air_winter_finalize_monthly_cycle = {
	if = {
		limit = {
			has_global_flag = air_winter_cycle_open
			has_variable = global.air_winter_cycle_id
			has_variable = global.air_winter_last_prepare_date
			check_variable = {
				var = global.air_winter_last_prepare_date
				value = global.date
				compare = equals
			}
			OR = {
				NOT = { has_variable = global.air_winter_last_finalize_cycle_id }
				NOT = {
					check_variable = {
						var = global.air_winter_last_finalize_cycle_id
						value = global.air_winter_cycle_id
						compare = equals
					}
				}
			}
		}
		set_variable = { global.air_winter_last_finalize_date = global.date }
		set_variable = { global.air_winter_last_finalize_cycle_id = global.air_winter_cycle_id }
		set_variable = { global.air_winter_severe_state_share = constant:air_winter_runtime.zero }

		if = {
			limit = { check_variable = { global.air_winter_valid_state_count > constant:air_winter_runtime.zero } }
			set_variable = { global.air_winter_severe_state_share = global.air_winter_severe_state_count }
			multiply_variable = { global.air_winter_severe_state_share = constant:air_winter_runtime.percentage_base }
			divide_variable = { global.air_winter_severe_state_share = global.air_winter_valid_state_count }
		}

		# The persistent registry is bounded by countries observed in the existing
		# state pass. Each row finalizes once for this cycle or removes stale burden.
		for_each_scope_loop = {
			array = global.air_winter_registered_countries
			air_winter_finalize_country_air_operations = yes
		}

		if = {
			limit = { check_variable = { global.air_winter_active_state_count > constant:air_winter_runtime.zero } }
			set_global_flag = air_winter_active
		}
		else = { clr_global_flag = air_winter_active }

		if = {
			limit = { check_variable = { global.air_winter_terminal_state_count > constant:air_winter_runtime.zero } }
			set_global_flag = air_winter_terminal_states_present
		}
		else = { clr_global_flag = air_winter_terminal_states_present }

		if = {
			limit = {
				check_variable = {
					var = global.air_winter_severe_state_share
					value = constant:air_winter_global.severe_share_entry_percent
					compare = greater_than_or_equals
				}
			}
			set_global_flag = air_winter_severe
		}
		else_if = {
			limit = {
				check_variable = {
					var = global.air_winter_severe_state_share
					value = constant:air_winter_global.severe_share_exit_percent
					compare = less_than
				}
			}
			clr_global_flag = air_winter_severe
		}

		if = {
			limit = { check_variable = { global.air_winter_unclassified_state_count > constant:air_winter_runtime.zero } }
			set_global_flag = air_winter_has_unclassified_states
		}
		else = { clr_global_flag = air_winter_has_unclassified_states }

		clr_global_flag = air_winter_reset_states_on_next_cycle
		clr_global_flag = air_winter_cycle_open
	}
}

# Supported scope: STATE. The authoritative typed state-ID ledger may write
# air_winter_presentation_class directly, then call this validator. A missing or
# invalid value remains visibly unclassified. No geographic fallback is used.
air_winter_refresh_presentation_class = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = air_winter_presentation_class }
				check_variable = { air_winter_presentation_class < constant:air_winter_presentation_class.unclassified }
				check_variable = { air_winter_presentation_class > constant:air_winter_presentation_class.maximum }
			}
		}
		set_variable = { air_winter_presentation_class = constant:air_winter_presentation_class.unclassified }
	}
	if = {
		limit = { air_winter_presentation_is_unclassified = yes }
		set_state_flag = air_winter_presentation_unclassified
	}
	else = { clr_state_flag = air_winter_presentation_unclassified }
}

# Supported scope: STATE. Seeds the climate-class water baseline once. The flag
# prevents a later ledger refresh from overwriting player-earned water security.
air_winter_initialize_presentation_water_profile = {
	if = {
		limit = {
			NOT = { has_state_flag = air_winter_water_profile_initialized }
			NOT = { air_winter_presentation_is_unclassified = yes }
		}
		if = { limit = { air_winter_presentation_is_boreal = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.boreal_continental } }
		else_if = { limit = { air_winter_presentation_is_maritime = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.temperate_maritime } }
		else_if = { limit = { air_winter_presentation_is_mediterranean = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.mediterranean } }
		else_if = { limit = { air_winter_presentation_is_arid = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.desert_arid_plateau } }
		else_if = { limit = { air_winter_presentation_is_tropical = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.tropical_coast_monsoon } }
		else_if = { limit = { air_winter_presentation_is_equatorial = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.equatorial_rainforest } }
		else_if = { limit = { air_winter_presentation_is_highland = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.mountain_highland } }
		else_if = { limit = { air_winter_presentation_is_oceanic = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.island_oceanic } }
		else_if = { limit = { air_winter_presentation_is_polar = yes } set_variable = { air_winter_water_security = constant:air_winter_presentation_water.polar_subpolar } }
		set_state_flag = air_winter_water_profile_initialized
	}
}

# Supported scope: STATE
# Input: temp variable air_winter_requested_presentation_class (0-9)
air_winter_set_presentation_class = {
	if = {
		limit = {
			NOT = { check_variable = { air_winter_requested_presentation_class < constant:air_winter_presentation_class.unclassified } }
			NOT = { check_variable = { air_winter_requested_presentation_class > constant:air_winter_presentation_class.maximum } }
		}
		set_variable = { air_winter_presentation_class = air_winter_requested_presentation_class }
		set_variable = { air_winter_presentation_ledger_version = constant:air_winter_presentation_ledger.schema_version }
	}
	else = {
		set_variable = { air_winter_presentation_class = constant:air_winter_presentation_class.unclassified }
		clear_variable = air_winter_presentation_ledger_version
	}
	air_winter_refresh_presentation_class = yes
	air_winter_initialize_presentation_water_profile = yes
}

# Supported scope: STATE. Reconciles the reviewed typed state-ID ledger only
# when its schema version is missing/stale or the stored enum is invalid. All
# nine memberships are evaluated so a missing or conflicting row resolves to
# the visible unclassified state instead of a geographic fallback.
air_winter_refresh_presentation_class_from_ledger = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = air_winter_presentation_ledger_version }
				NOT = { check_variable = { air_winter_presentation_ledger_version = constant:air_winter_presentation_ledger.schema_version } }
				NOT = { has_variable = air_winter_presentation_class }
				check_variable = { air_winter_presentation_class < constant:air_winter_presentation_class.unclassified }
				check_variable = { air_winter_presentation_class > constant:air_winter_presentation_class.maximum }
			}
		}

		set_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.zero }
		set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.unclassified }

		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.boreal_continental
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.boreal_continental }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.temperate_maritime
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.temperate_maritime }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.mediterranean
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.mediterranean }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.desert_arid_plateau
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.desert_arid_plateau }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.tropical_coast_monsoon
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.tropical_coast_monsoon }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.equatorial_rainforest
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.equatorial_rainforest }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.mountain_highland
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.mountain_highland }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.island_oceanic
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.island_oceanic }
		}
		if = {
			limit = {
				any_state_of = {
					target = constant:air_winter_presentation_states.polar_subpolar
					state = PREV
				}
			}
			add_to_temp_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.polar_subpolar }
		}

		if = {
			limit = { NOT = { check_variable = { air_winter_presentation_match_count = constant:air_winter_runtime.one } } }
			set_temp_variable = { air_winter_requested_presentation_class = constant:air_winter_presentation_class.unclassified }
		}
		air_winter_set_presentation_class = yes
	}
	else = {
		air_winter_refresh_presentation_class = yes
	}
}

# Supported scope: any. Each call opens a distinct producer generation so a
# same-date Fallout retry cannot authenticate a stale state receipt.
air_winter_begin_fallout_snapshot_production = {
	# Active-transition retries keep the country air burden cleared. Prelock
	# validation does not remove it before the Fallout request actually commits.
	if = {
		limit = { has_global_flag = fallout_transition_active }
		air_winter_clear_all_country_air_operations_modifiers = yes
	}
	if = {
		limit = { NOT = { has_variable = global.air_winter_fallout_snapshot_producer_generation } }
		set_variable = { global.air_winter_fallout_snapshot_producer_generation = constant:air_winter_runtime.zero }
	}
	add_to_variable = { global.air_winter_fallout_snapshot_producer_generation = constant:air_winter_runtime.one }
	set_variable = { global.air_winter_fallout_snapshot_producer_schema_version = constant:air_winter_fallout_snapshot_schema.version }
}

# Supported scope: STATE
air_winter_record_original_category = {
	if = {
		limit = { NOT = { has_state_flag = air_winter_original_category_recorded } }
		if = { limit = { has_state_category = megalopolis } set_variable = { air_winter_original_category = constant:air_winter_state_category.megalopolis } set_state_flag = air_winter_original_category_recorded }
		else_if = { limit = { has_state_category = metropolis } set_variable = { air_winter_original_category = constant:air_winter_state_category.metropolis } set_state_flag = air_winter_original_category_recorded }

# Supported scope: STATE
air_winter_initialize_state = {
	if = {
		limit = { NOT = { has_state_flag = air_winter_initialized } }
		air_winter_record_original_category = yes
		air_winter_refresh_presentation_class = yes

		set_variable = { air_winter_phase = constant:air_winter_phase.clear }

		set_variable = { air_winter_recovery_months = constant:air_winter_runtime.zero }
		set_variable = { air_winter_recovery = constant:air_winter_ledger.minimum }
		set_variable = { air_winter_exposure = constant:air_winter_runtime.zero }
		set_variable = { air_winter_expected_exposure_delta = constant:air_winter_runtime.zero }
		set_variable = { air_winter_adaptation = constant:air_winter_default.adaptation }
		set_variable = { air_winter_food_reserve = constant:air_winter_default.food_reserve }
		set_variable = { air_winter_shelter_capacity = constant:air_winter_default.shelter_capacity }
		set_variable = { air_winter_reclamation = constant:air_winter_default.reclamation }
		if = { limit = { NOT = { has_variable = air_winter_water_security } } set_variable = { air_winter_water_security = constant:air_winter_default.water_security } }
		set_variable = { air_winter_refugee_pressure = constant:air_winter_default.refugee_pressure }
		set_variable = { air_winter_disease_pressure = constant:air_winter_runtime.zero }
		set_variable = { air_winter_survival_value = constant:air_winter_survival.minimum }
		set_variable = { air_winter_expected_monthly_death_percent = constant:air_winter_runtime.zero }
		set_variable = { air_winter_local_supply_factor = constant:air_winter_runtime.zero }
		set_variable = { air_winter_building_damage_pressure = constant:air_winter_runtime.zero }

		set_variable = { air_winter_category_cooldown = constant:air_winter_runtime.zero }

		set_variable = { air_winter_airbase_closure_months = constant:air_winter_runtime.zero }
		set_variable = { air_winter_population_loss_memory = constant:air_winter_runtime.zero }

		if = {
			limit = { OR = { has_state_category = pastoral has_state_category = rural } }
			add_to_variable = { air_winter_food_reserve = constant:air_winter_default.rural_food_bonus }
		}
		else_if = {
			limit = { OR = { has_state_category = town has_state_category = large_town } }
			add_to_variable = { air_winter_food_reserve = constant:air_winter_default.town_food_bonus }
			add_to_variable = { air_winter_shelter_capacity = constant:air_winter_default.town_shelter_bonus }
		}
		else_if = {
			limit = { has_state_category = city }
			add_to_variable = { air_winter_food_reserve = constant:air_winter_default.city_food_bonus }
			add_to_variable = { air_winter_shelter_capacity = constant:air_winter_default.city_shelter_bonus }
			add_to_variable = { air_winter_reclamation = constant:air_winter_default.urban_reclamation_bonus }

		}

				NOT = { infrastructure < constant:air_winter_pressure.low_infrastructure_level }
			}

			add_to_variable = { air_winter_reclamation = constant:air_winter_default.infrastructure_reclamation_bonus }
		}

			limit = { bunker > constant:air_winter_runtime.zero }
			add_to_variable = { air_winter_shelter_capacity = constant:air_winter_default.bunker_shelter_bonus }
		}
		if = {
			limit = { supply_node > constant:air_winter_runtime.zero }
			add_to_variable = { air_winter_shelter_capacity = constant:air_winter_default.supply_node_shelter_bonus }
			add_to_variable = { air_winter_reclamation = constant:air_winter_default.supply_node_reclamation_bonus }
		}
		air_winter_initialize_presentation_water_profile = yes

		clamp_variable = { var = air_winter_adaptation min = constant:air_winter_ledger.minimum max = constant:air_winter_ledger.maximum }
		clamp_variable = { var = air_winter_food_reserve min = constant:air_winter_ledger.minimum max = constant:air_winter_ledger.maximum }
		clamp_variable = { var = air_winter_shelter_capacity min = constant:air_winter_ledger.minimum max = constant:air_winter_ledger.maximum }
		clamp_variable = { var = air_winter_reclamation min = constant:air_winter_ledger.minimum max = constant:air_winter_ledger.maximum }
		set_state_flag = air_winter_initialized
	}
}

# Supported scope: STATE
air_winter_calculate_state_pressure = {
	set_variable = { air_winter_pressure = global.air_winter_pressure }
	set_variable = { air_winter_recovery_pressure = global.air_winter_recovery_pressure }

	if = {
		limit = { has_dynamic_modifier = { modifier = nuclear_fallout_state } }
		add_to_variable = { air_winter_pressure = constant:air_winter_pressure.nuclear_fallout_base }
		if = {
			limit = { has_variable = nuclear_fallout_intensity }
			set_temp_variable = { air_winter_local_fallout_pressure = nuclear_fallout_intensity }
			multiply_temp_variable = { air_winter_local_fallout_pressure = constant:air_winter_pressure.nuclear_fallout_per_intensity }
			add_to_variable = { air_winter_pressure = air_winter_local_fallout_pressure }
		}
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = chem_state_contamination } }
		add_to_variable = { air_winter_pressure = constant:air_winter_pressure.chemical_contamination }
	}

	if = { limit = { has_state_category = megalopolis } add_to_variable = { air_winter_pressure = constant:air_winter_pressure.urban_megalopolis } }
	else_if = { limit = { has_state_category = metropolis } add_to_variable = { air_winter_pressure = constant:air_winter_pressure.urban_metropolis } }
	else_if = { limit = { has_state_category = large_city } add_to_variable = { air_winter_pressure = constant:air_winter_pressure.urban_large_city } }
	else_if = { limit = { has_state_category = city } add_to_variable = { air_winter_pressure = constant:air_winter_pressure.urban_city } }

	if = {
		limit = { infrastructure < constant:air_winter_pressure.low_infrastructure_level }
		add_to_variable = { air_winter_pressure = constant:air_winter_pressure.low_infrastructure }
	}
	if = {
		limit = { NOT = { is_owned_and_controlled_by = OWNER } }
		add_to_variable = { air_winter_pressure = constant:air_winter_pressure.occupation_disruption }
	}
	if = {
		limit = { any_neighbor_state = { air_winter_was_severe_at_cycle_start = yes } }
		add_to_variable = { air_winter_pressure = constant:air_winter_pressure.severe_neighbor }
	}
	if = {
		limit = { has_state_flag = air_winter_relief_route }
		subtract_from_variable = { air_winter_pressure = constant:air_winter_pressure.relief_route }
	}
	if = {
		limit = { has_state_flag = air_winter_sampling_stations }
		subtract_from_variable = { air_winter_pressure = constant:air_winter_response_result.sampling_pressure_reduction }
	}

	set_temp_variable = { air_winter_adaptation_resistance = air_winter_adaptation }
	multiply_temp_variable = { air_winter_adaptation_resistance = constant:air_winter_pressure.adaptation_resistance }
	subtract_from_variable = { air_winter_pressure = air_winter_adaptation_resistance }
	set_temp_variable = { air_winter_food_resistance = air_winter_food_reserve }
	multiply_temp_variable = { air_winter_food_resistance = constant:air_winter_pressure.food_resistance }
	subtract_from_variable = { air_winter_pressure = air_winter_food_resistance }
	set_temp_variable = { air_winter_shelter_resistance = air_winter_shelter_capacity }
	multiply_temp_variable = { air_winter_shelter_resistance = constant:air_winter_pressure.shelter_resistance }
	subtract_from_variable = { air_winter_pressure = air_winter_shelter_resistance }
	set_temp_variable = { air_winter_reclamation_resistance = air_winter_reclamation }
	multiply_temp_variable = { air_winter_reclamation_resistance = constant:air_winter_pressure.reclamation_resistance }
	subtract_from_variable = { air_winter_pressure = air_winter_reclamation_resistance }

	if = {
		limit = { has_variable = air_winter_pressure_modifier }
		add_to_variable = { air_winter_pressure = air_winter_pressure_modifier }
	}
	if = {
		limit = { has_variable = air_winter_recovery_bonus }
		add_to_variable = { air_winter_recovery_pressure = air_winter_recovery_bonus }
	}

	clamp_variable = {
		var = air_winter_pressure
		min = constant:air_winter_runtime.zero
		max = constant:air_winter_global.maximum_pressure
	}
	clamp_variable = {
		var = air_winter_recovery_pressure
		min = constant:air_winter_runtime.zero
		max = constant:air_winter_global.maximum_pressure
	}
	set_variable = { air_winter_recovery = air_winter_recovery_pressure }
	clamp_variable = {
		var = air_winter_recovery
		min = constant:air_winter_ledger.minimum
		max = constant:air_winter_ledger.maximum
	}
}

# Supported scope: STATE
air_winter_calculate_target_phase = {
	set_variable = { air_winter_target_phase = constant:air_winter_phase.clear }

	if = {
		limit = { check_variable = { var = air_winter_pressure value = constant:air_winter_phase_entry.phase_1 compare = greater_than_or_equals } }
		set_variable = { air_winter_target_phase = constant:air_winter_phase.dimming }
	}
	if = {
		limit = { check_variable = { var = air_winter_pressure value = constant:air_winter_phase_entry.phase_2 compare = greater_than_or_equals } }
		set_variable = { air_winter_target_phase = constant:air_winter_phase.crop_shock }
	}
	if = {
		limit = { check_variable = { var = air_winter_pressure value = constant:air_winter_phase_entry.phase_3 compare = greater_than_or_equals } }
		set_variable = { air_winter_target_phase = constant:air_winter_phase.hard_freeze }
	}
	if = {
		limit = { check_variable = { var = air_winter_pressure value = constant:air_winter_phase_entry.phase_4 compare = greater_than_or_equals } }
		set_variable = { air_winter_target_phase = constant:air_winter_phase.black_harvest }
	}
	if = {
		limit = { check_variable = { var = air_winter_pressure value = constant:air_winter_phase_entry.phase_5 compare = greater_than_or_equals } }
		set_variable = { air_winter_target_phase = constant:air_winter_phase.ash_winter }
	}
	if = {
		limit = { check_variable = { var = air_winter_pressure value = constant:air_winter_phase_entry.phase_6 compare = greater_than_or_equals } }
		set_variable = { air_winter_target_phase = constant:air_winter_phase.terminal_winter }
	}

	# Retention thresholds are lower than entry thresholds. They hold the current
	# phase without allowing pressure to skip more than one actual transition.
	if = {
		limit = {
			air_winter_is_phase_1 = yes
			check_variable = { var = air_winter_pressure value = constant:air_winter_phase_retention.phase_1 compare = greater_than_or_equals }
			check_variable = { air_winter_target_phase < constant:air_winter_phase.dimming }
		}
		set_variable = { air_winter_target_phase = constant:air_winter_phase.dimming }
	}
	else_if = {
		limit = {
			air_winter_is_phase_2 = yes
			check_variable = { var = air_winter_pressure value = constant:air_winter_phase_retention.phase_2 compare = greater_than_or_equals }
			check_variable = { air_winter_target_phase < constant:air_winter_phase.crop_shock }
		}
		set_variable = { air_winter_target_phase = constant:air_winter_phase.crop_shock }
	}
	else_if = {
		limit = {
			air_winter_is_phase_3 = yes
			check_variable = { var = air_winter_pressure value = constant:air_winter_phase_retention.phase_3 compare = greater_than_or_equals }
			check_variable = { air_winter_target_phase < constant:air_winter_phase.hard_freeze }
		}
		set_variable = { air_winter_target_phase = constant:air_winter_phase.hard_freeze }
	}
	else_if = {
		limit = {
			air_winter_is_phase_4 = yes
			check_variable = { var = air_winter_pressure value = constant:air_winter_phase_retention.phase_4 compare = greater_than_or_equals }
			check_variable = { air_winter_target_phase < constant:air_winter_phase.black_harvest }
		}
		set_variable = { air_winter_target_phase = constant:air_winter_phase.black_harvest }
	}
	else_if = {
		limit = {
			air_winter_is_phase_5 = yes
			check_variable = { var = air_winter_pressure value = constant:air_winter_phase_retention.phase_5 compare = greater_than_or_equals }
			check_variable = { air_winter_target_phase < constant:air_winter_phase.ash_winter }
		}
		set_variable = { air_winter_target_phase = constant:air_winter_phase.ash_winter }
	}
	else_if = {
		limit = {
			air_winter_is_phase_6 = yes
			check_variable = { var = air_winter_pressure value = constant:air_winter_phase_retention.phase_6 compare = greater_than_or_equals }
			check_variable = { air_winter_target_phase < constant:air_winter_phase.terminal_winter }
		}
		set_variable = { air_winter_target_phase = constant:air_winter_phase.terminal_winter }
	}

	if = {
		limit = {
			has_variable = air_winter_minimum_phase
			check_variable = {
				var = air_winter_target_phase
				value = air_winter_minimum_phase
				compare = less_than
			}
		}
		set_variable = { air_winter_target_phase = air_winter_minimum_phase }
	}
	clamp_variable = { var = air_winter_target_phase min = constant:air_winter_phase.clear max = constant:air_winter_phase.maximum }

	set_variable = { air_winter_trend = constant:air_winter_trend.stable }
	if = {
		limit = { check_variable = { var = air_winter_target_phase value = air_winter_phase compare = greater_than } }
		set_variable = { air_winter_trend = constant:air_winter_trend.worsening }
	}
	else_if = {
		limit = { check_variable = { var = air_winter_target_phase value = air_winter_phase compare = less_than } }
		set_variable = { air_winter_trend = constant:air_winter_trend.recovering }
	}
}

air_winter_advance_phase = {
	set_variable = { air_winter_previous_phase = air_winter_phase }
	add_to_variable = { air_winter_phase_months = constant:air_winter_runtime.one }
	clr_state_flag = air_winter_phase_changed_this_tick

	if = {
		limit = {
			air_winter_trend_is_worsening = yes
			OR = {
				check_variable = {
					var = air_winter_phase_months
					value = constant:air_winter_transition.minimum_escalation_months
					compare = greater_than_or_equals
				}
				has_state_flag = air_winter_emergency_escalation
			}
		}
		add_to_variable = { air_winter_phase = constant:air_winter_runtime.one }
		clamp_variable = { var = air_winter_phase min = constant:air_winter_phase.clear max = constant:air_winter_phase.maximum }
		set_variable = { air_winter_phase_months = constant:air_winter_runtime.zero }
		set_variable = { air_winter_recovery_months = constant:air_winter_runtime.zero }
		set_state_flag = air_winter_phase_changed_this_tick
		clr_state_flag = air_winter_emergency_escalation
	}
	else_if = {
		limit = { air_winter_trend_is_recovering = yes }
		set_temp_variable = { air_winter_recovery_gap = air_winter_recovery_pressure }
		subtract_from_temp_variable = { air_winter_recovery_gap = air_winter_pressure }
		if = {
			limit = {
				check_variable = {
					var = air_winter_recovery_gap
					value = constant:air_winter_transition.recovery_margin
					compare = greater_than_or_equals
				}
			}
			add_to_variable = { air_winter_recovery_months = constant:air_winter_runtime.one }
		}
		else = { set_variable = { air_winter_recovery_months = constant:air_winter_runtime.zero } }

		if = {
			limit = {
				check_variable = {
					var = air_winter_recovery_months
					value = constant:air_winter_transition.recovery_months_required
					compare = greater_than_or_equals
				}
				check_variable = {
					var = air_winter_phase_months
					value = constant:air_winter_transition.minimum_phase_months_for_recovery
					compare = greater_than_or_equals
				}
			}
			subtract_from_variable = { air_winter_phase = constant:air_winter_runtime.one }
			clamp_variable = { var = air_winter_phase min = constant:air_winter_phase.clear max = constant:air_winter_phase.maximum }
			set_variable = { air_winter_phase_months = constant:air_winter_runtime.zero }
			set_variable = { air_winter_recovery_months = constant:air_winter_runtime.zero }
			set_state_flag = air_winter_phase_changed_this_tick
		}
	}
	else = { set_variable = { air_winter_recovery_months = constant:air_winter_runtime.zero } }

	if = {
		limit = {
			check_variable = { air_winter_previous_phase = constant:air_winter_phase.clear }
			check_variable = { air_winter_phase = constant:air_winter_phase.dimming }
		}
		set_variable = { air_winter_peak_phase_memory = air_winter_phase }
	}
	else_if = {
		limit = {
			check_variable = {
				var = air_winter_phase
				value = air_winter_peak_phase_memory
				compare = greater_than
			}
		}
		set_variable = { air_winter_peak_phase_memory = air_winter_phase }
	}

	if = {
		limit = { has_state_flag = air_winter_phase_changed_this_tick }
		set_state_flag = air_winter_recent_phase_change
		set_variable = { air_winter_recent_phase_change_months = constant:air_winter_transition.recent_phase_change_months }
		set_variable = { air_winter_last_phase_change_date = global.date }
	}
}

air_winter_update_building_damage = {
	clr_state_flag = air_winter_building_damage_applied_this_tick
	set_temp_variable = { air_winter_building_pressure_delta = constant:air_winter_runtime.zero }
	if = { limit = { air_winter_is_phase_3 = yes } set_temp_variable = { air_winter_building_pressure_delta = constant:air_winter_building_damage.phase_3_pressure } }
	else_if = { limit = { air_winter_is_phase_4 = yes } set_temp_variable = { air_winter_building_pressure_delta = constant:air_winter_building_damage.phase_4_pressure } }
	else_if = { limit = { air_winter_is_phase_5 = yes } set_temp_variable = { air_winter_building_pressure_delta = constant:air_winter_building_damage.phase_5_pressure } }
	else_if = { limit = { air_winter_is_phase_6 = yes } set_temp_variable = { air_winter_building_pressure_delta = constant:air_winter_building_damage.phase_6_pressure } }

	set_temp_variable = { air_winter_building_adaptation = air_winter_adaptation }
	multiply_temp_variable = { air_winter_building_adaptation = constant:air_winter_building_damage.adaptation_reduction }
	set_temp_variable = { air_winter_building_reclamation = air_winter_reclamation }
	multiply_temp_variable = { air_winter_building_reclamation = constant:air_winter_building_damage.reclamation_reduction }

	if = {
		limit = { check_variable = { air_winter_building_pressure_delta > constant:air_winter_runtime.zero } }
		subtract_from_temp_variable = { air_winter_building_pressure_delta = air_winter_building_adaptation }
		subtract_from_temp_variable = { air_winter_building_pressure_delta = air_winter_building_reclamation }
		clamp_temp_variable = { var = air_winter_building_pressure_delta min = constant:air_winter_runtime.zero }
		add_to_variable = { air_winter_building_damage_pressure = air_winter_building_pressure_delta }
	}
	else = {
		set_temp_variable = { air_winter_building_recovery = air_winter_building_reclamation }
		add_to_temp_variable = { air_winter_building_recovery = air_winter_building_adaptation }
		subtract_from_variable = { air_winter_building_damage_pressure = air_winter_building_recovery }
	}
	clamp_variable = {
		var = air_winter_building_damage_pressure
		min = constant:air_winter_runtime.zero
		max = constant:air_winter_building_damage.maximum_pressure
	}

	if = {
		limit = { air_winter_building_damage_is_ready = yes }
		set_temp_variable = { air_winter_building_damage_amount = constant:air_winter_building_damage.damage_phase_3 }
		if = { limit = { air_winter_is_phase_4 = yes } set_temp_variable = { air_winter_building_damage_amount = constant:air_winter_building_damage.damage_phase_4 } }
		else_if = { limit = { air_winter_is_phase_5 = yes } set_temp_variable = { air_winter_building_damage_amount = constant:air_winter_building_damage.damage_phase_5 } }
		else_if = { limit = { air_winter_is_phase_6 = yes } set_temp_variable = { air_winter_building_damage_amount = constant:air_winter_building_damage.damage_phase_6 } }

		set_temp_variable = { air_winter_building_family_attempts = constant:air_winter_runtime.zero }
		while_loop_effect = {
			limit = {
				check_variable = { air_winter_building_family_attempts < constant:air_winter_building_damage.family_count }
				NOT = { has_state_flag = air_winter_building_damage_applied_this_tick }
			}
			air_winter_try_damage_current_building_family = yes
			add_to_temp_variable = { air_winter_building_family_attempts = constant:air_winter_runtime.one }
		}

		if = {
			limit = { has_state_flag = air_winter_building_damage_applied_this_tick }
			subtract_from_variable = { air_winter_building_damage_pressure = constant:air_winter_building_damage.threshold }
			clamp_variable = { var = air_winter_building_damage_pressure min = constant:air_winter_runtime.zero }
			set_variable = { air_winter_building_damage_cooldown = constant:air_winter_building_damage.cooldown_months }
			set_state_flag = air_winter_recent_building_loss
			set_variable = { air_winter_recent_building_loss_months = constant:air_winter_building_damage.recent_loss_months }
			set_variable = { air_winter_last_building_loss_date = global.date }
		}
	}
}

# Supported scope: COUNTRY. This receipt makes a repeated finalize for the same
# monthly cycle inert. A country with no current contribution loses the modifier.
air_winter_finalize_country_air_operations = {
	air_winter_prepare_country_air_operations_cycle = yes
	if = {
		limit = {
			has_variable = global.air_winter_cycle_id
			OR = {
				NOT = { has_variable = air_winter_air_operations_last_finalize_cycle_id }
				NOT = {
					check_variable = {
						var = air_winter_air_operations_last_finalize_cycle_id
						value = global.air_winter_cycle_id
						compare = equals
					}
				}
			}
		}
		set_variable = { air_winter_air_operations_last_finalize_cycle_id = global.air_winter_cycle_id }
		set_variable = { air_winter_controlled_airfield_phase_average = constant:air_winter_runtime.zero }
		set_variable = { air_winter_air_operations_burden = constant:air_winter_runtime.zero }
		set_variable = { air_winter_air_mission_efficiency = constant:air_winter_runtime.zero }
		set_variable = { air_winter_air_detection = constant:air_winter_runtime.zero }
		set_variable = { air_winter_air_accidents_factor = constant:air_winter_runtime.zero }
		set_variable = { air_winter_air_weather_penalty = constant:air_winter_runtime.zero }

		if = {
			limit = {
				exists = yes
				air_winter_system_enabled = yes
				check_variable = { air_winter_controlled_airfield_state_count > constant:air_winter_runtime.zero }
			}
			set_variable = { air_winter_controlled_airfield_phase_average = air_winter_controlled_airfield_phase_sum }
			divide_variable = { air_winter_controlled_airfield_phase_average = air_winter_controlled_airfield_state_count }
			set_variable = { air_winter_air_operations_burden = air_winter_controlled_airfield_phase_average }
			divide_variable = { air_winter_air_operations_burden = constant:air_winter_phase.maximum }
			clamp_variable = {
				var = air_winter_air_operations_burden
				min = constant:air_winter_runtime.zero
				max = constant:air_winter_runtime.one
			}

			set_variable = { air_winter_air_mission_efficiency = air_winter_air_operations_burden }
			multiply_variable = { air_winter_air_mission_efficiency = constant:air_winter_air_operations.full_burden_mission_efficiency }
			set_variable = { air_winter_air_detection = air_winter_air_operations_burden }
			multiply_variable = { air_winter_air_detection = constant:air_winter_air_operations.full_burden_detection }
			set_variable = { air_winter_air_accidents_factor = air_winter_air_operations_burden }
			multiply_variable = { air_winter_air_accidents_factor = constant:air_winter_air_operations.full_burden_accidents }
			set_variable = { air_winter_air_weather_penalty = air_winter_air_operations_burden }
			multiply_variable = { air_winter_air_weather_penalty = constant:air_winter_air_operations.full_burden_weather_penalty }
		}

		if = {
			limit = { check_variable = { air_winter_air_operations_burden > constant:air_winter_runtime.zero } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = air_winter_air_operations_country } } }
				add_dynamic_modifier = { modifier = air_winter_air_operations_country }
			}
			force_update_dynamic_modifier = yes
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = air_winter_air_operations_country } }
				remove_dynamic_modifier = { modifier = air_winter_air_operations_country }
			}
		}
	}
}

# Supported scope: COUNTRY. Clears only the calling country's Air Winter
# reception and event memory. This helper deliberately performs no iterator.
air_winter_reset_country = {
	air_winter_clear_country_air_operations_modifier = yes
	if = {
		limit = { has_variable = air_winter_response_active_project_state }
		var:air_winter_response_active_project_state = { air_winter_response_cancel_state_project = yes }
	}
	if = {
		limit = { has_variable = air_winter_response_priority_state }
		var:air_winter_response_priority_state = {
			clr_state_flag = air_winter_response_priority_target
			clear_variable = air_winter_response_priority_owner
		}
	}
	for_each_scope_loop = {
		array = air_winter_reception_states
		clr_state_flag = air_winter_response_reception_target
		clear_variable = air_winter_response_reception_owner
	}
	clear_array = air_winter_reception_states
	clear_variable = air_winter_response_active_project_state
	clear_variable = air_winter_response_priority_state
	clr_country_flag = air_winter_response_project_active
	clr_country_flag = air_winter_atmospheric_office
	air_winter_response_clear_evacuation_cache = yes
	air_winter_event_clear_country_memory = yes
}

# Supported scope: COUNTRY (the existing host country). Registered countries
# are reset through a bounded array. State cleanup is deferred to the next
# existing monthly state pass, so this helper adds no world iterator.
air_winter_reset_global = {
	air_winter_reset_country = yes
	for_each_scope_loop = {
		array = global.air_winter_registered_countries
		# Country scopes remain valid while a tag is absent from the map. Clearing
		# every registered entry prevents stale memory from returning on release.
		air_winter_reset_country = yes
	}
	clear_array = global.air_winter_registered_countries
	air_winter_destroy_normal_map_proof_entity = yes
	set_global_flag = air_winter_reset_states_on_next_cycle
