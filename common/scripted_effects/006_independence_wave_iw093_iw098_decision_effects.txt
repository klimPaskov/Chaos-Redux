# CHAOS REDUX - EVENT 006 IW-093 / IW-098 DECISION EFFECTS
# ============================================================================
# Paid project ledgers and lifecycle helpers for the Asante and Sokoto
# signature decision categories. These helpers never create units, grant
# equipment, set content attestation, or commit FORM-24/FORM-25. Decisions
# provide the requested cost bundle in temporary variables before opening a
# ledger. Failure and cancellation close the ledger without a refund.

independence_wave_iw093_begin_paid_decision_transaction = {
	set_temp_variable = { independence_wave_iw093_decision_transaction_result = constant:independence_wave_iw093.zero }
	if = {
		limit = {
			is_independence_wave_iw093_country = yes
			NOT = { has_country_flag = independence_wave_iw093_decision_transaction_active }
			OR = {
				check_variable = { independence_wave_iw093_decision_command_power_cost = constant:independence_wave_iw093.zero }
				command_power > independence_wave_iw093_decision_command_power_cost
			}
			OR = {
				check_variable = { independence_wave_iw093_decision_infantry_equipment_cost = constant:independence_wave_iw093.zero }
				has_equipment = { infantry_equipment > independence_wave_iw093_decision_infantry_equipment_cost }
			}
			OR = {
				check_variable = { independence_wave_iw093_decision_train_equipment_cost = constant:independence_wave_iw093.zero }
				has_equipment = { train_equipment > independence_wave_iw093_decision_train_equipment_cost }
			}
			OR = {
				check_variable = { independence_wave_iw093_decision_convoy_cost = constant:independence_wave_iw093.zero }
				has_equipment = { convoy > independence_wave_iw093_decision_convoy_cost }
			}
		}
		if = {
			limit = { independence_wave_iw093_decision_command_power_cost > constant:independence_wave_iw093.zero }
			set_temp_variable = { independence_wave_iw093_negative_decision_cost = independence_wave_iw093_decision_command_power_cost }
			multiply_temp_variable = { independence_wave_iw093_negative_decision_cost = constant:independence_wave_iw093.negative_one }
			add_command_power = independence_wave_iw093_negative_decision_cost
		}
		if = {
			limit = { independence_wave_iw093_decision_infantry_equipment_cost > constant:independence_wave_iw093.zero }
			set_temp_variable = { independence_wave_iw093_negative_decision_cost = independence_wave_iw093_decision_infantry_equipment_cost }
			multiply_temp_variable = { independence_wave_iw093_negative_decision_cost = constant:independence_wave_iw093.negative_one }
			add_equipment_to_stockpile = { type = infantry_equipment amount = independence_wave_iw093_negative_decision_cost }
		}
		if = {
			limit = { independence_wave_iw093_decision_train_equipment_cost > constant:independence_wave_iw093.zero }
			set_temp_variable = { independence_wave_iw093_negative_decision_cost = independence_wave_iw093_decision_train_equipment_cost }
			multiply_temp_variable = { independence_wave_iw093_negative_decision_cost = constant:independence_wave_iw093.negative_one }
			add_equipment_to_stockpile = { type = train_equipment amount = independence_wave_iw093_negative_decision_cost }
		}
		if = {
			limit = { independence_wave_iw093_decision_convoy_cost > constant:independence_wave_iw093.zero }
			set_temp_variable = { independence_wave_iw093_negative_decision_cost = independence_wave_iw093_decision_convoy_cost }
			multiply_temp_variable = { independence_wave_iw093_negative_decision_cost = constant:independence_wave_iw093.negative_one }
			add_equipment_to_stockpile = { type = convoy amount = independence_wave_iw093_negative_decision_cost }
		}
		set_variable = { independence_wave_iw093_paid_command_power_cost = independence_wave_iw093_decision_command_power_cost }
		set_variable = { independence_wave_iw093_paid_infantry_equipment_cost = independence_wave_iw093_decision_infantry_equipment_cost }
		set_variable = { independence_wave_iw093_paid_train_equipment_cost = independence_wave_iw093_decision_train_equipment_cost }
		set_variable = { independence_wave_iw093_paid_convoy_cost = independence_wave_iw093_decision_convoy_cost }
		set_country_flag = independence_wave_iw093_decision_transaction_active
		set_temp_variable = { independence_wave_iw093_decision_transaction_result = constant:independence_wave_iw093.one }
	}
}

independence_wave_iw093_refresh_project_idea = {
	if = {
		limit = {
			has_country_flag = independence_wave_iw093_cocoa_depot_project_complete
			has_country_flag = independence_wave_iw093_kumasi_rail_project_complete
			has_idea = independence_wave_iw093_unsettled_restoration_idea
		}
		swap_ideas = {
			remove_idea = independence_wave_iw093_unsettled_restoration_idea
			add_idea = independence_wave_iw093_cocoa_rail_compact_idea
		}
	}
}

independence_wave_iw093_resolve_cocoa_depot_project = {
	if = {
		limit = {
			is_independence_wave_iw093_country = yes
			has_country_flag = independence_wave_iw093_cocoa_depot_project_active
			274 = { is_owned_by = ROOT is_controlled_by = ROOT }
		}
		independence_wave_iw093_close_paid_decision_transaction = yes
		if = {
			limit = { independence_wave_iw093_decision_transaction_result > constant:independence_wave_iw093.zero }
			set_country_flag = independence_wave_iw093_cocoa_depot_project_complete
			add_to_variable = { iw093_cocoa_rail_throughput = constant:independence_wave_iw093.focus_standard_gain }
			add_to_variable = { iw093_confederated_authority = constant:independence_wave_iw093.focus_small_gain }
			274 = { add_building_construction = { type = industrial_complex level = constant:independence_wave_decision_effect.construction_factory instant_build = yes } }
			independence_wave_clamp_iw093_values = yes
			independence_wave_iw093_refresh_project_idea = yes
		}
	}
	else = {
		independence_wave_iw093_close_paid_decision_transaction = yes
		if = {
			limit = { is_independence_wave_iw093_country = yes }
			set_country_flag = independence_wave_iw093_cocoa_depot_project_failed
			subtract_from_variable = { iw093_cocoa_rail_throughput = constant:independence_wave_iw093.focus_small_gain }
			independence_wave_clamp_iw093_values = yes
		}
	}
	clr_country_flag = independence_wave_iw093_cocoa_depot_project_active
}
