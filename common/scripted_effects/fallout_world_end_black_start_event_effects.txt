# ============================================================================
# CHAOS REDUX - FALLOUT BLACK START EFFECTS
# ============================================================================
# Dormant country-level recovery transaction. The chain freezes Power, Scrap,
# Cohesion, Recognition, Reclamation, and durable grid ledgers, then resolves
# one grid-governance policy through a 45-day result, a 270-day callback,
# Deaths-backed failure, Event Log memory, dynamic modifiers, and cleanup.
# ============================================================================

fallout_event_310_apply_failure_deaths = {
	every_owned_state = {
		if = {
			limit = { check_variable = { state_population_k value = constant:fallout_event_runtime.zero compare = greater_than } }
			set_temp_variable = { fallout_event_310_state_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_310_state_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_310_state_death_request = constant:fallout_event_310_threshold.result_failure_death_percent }
			round_temp_variable = fallout_event_310_state_death_request
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_requested = fallout_event_310_state_death_request }
			set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:fallout_event_310_threshold.minimum_remaining_people }
			set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.fallout_aftermath }
			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }
			apply_exact_state_civilian_population_loss = yes
			clear_temp_variable = fallout_event_310_state_death_request
		}
	}
}

	if = {
		limit = { NOT = { has_country_flag = fallout_event_310_cost_paid } }

			limit = { check_variable = { fallout_event_310_selected_branch = constant:fallout_event_310_branch.centralized_grid } }
			subtract_from_variable = { fallout_survival_current_entries^6 = constant:fallout_event_310_cost.central_power }
			subtract_from_variable = { fallout_survival_current_entries^4 = constant:fallout_event_310_cost.central_scrap }
		}
		else_if = {
			limit = { check_variable = { fallout_event_310_selected_branch = constant:fallout_event_310_branch.local_cooperatives } }

			subtract_from_variable = { fallout_survival_current_entries^1 = constant:fallout_event_310_cost.cooperative_food }
		}

			limit = { check_variable = { fallout_event_310_selected_branch = constant:fallout_event_310_branch.military_grid } }
			subtract_from_variable = { fallout_survival_current_entries^6 = constant:fallout_event_310_cost.military_power }

}
