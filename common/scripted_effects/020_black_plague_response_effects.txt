# STATE scope. Input/output: black_plague_response_quote_working.
black_plague_response_ceil_working_quote = {
	set_temp_variable = { black_plague_response_quote_rounded = black_plague_response_quote_working }
	round_temp_variable = black_plague_response_quote_rounded
	if = {
		limit = {
			check_variable = {
				var = black_plague_response_quote_rounded
				value = black_plague_response_quote_working
				compare = less_than
			}
		}
		add_to_temp_variable = { black_plague_response_quote_rounded = constant:black_plague_response_value.one }
	}
	set_temp_variable = { black_plague_response_quote_working = black_plague_response_quote_rounded }
}

# COUNTRY scope. Medical reserve is a capped logistical stock rather than
# political power. Industry expands storage and production capacity; the
# production milestone raises the output of each priced batch.
black_plague_recompute_medical_reserve_capacity = {
	if = {
		limit = { NOT = { has_variable = black_plague_medical_reserve } }
		set_variable = { black_plague_medical_reserve = constant:black_plague_response_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = black_plague_countermeasure_progress } }
		set_variable = { black_plague_countermeasure_progress = constant:black_plague_response_value.zero }
	}
	set_variable = { black_plague_medical_reserve_capacity = constant:black_plague_medical_reserve.base_capacity }
	set_temp_variable = { black_plague_medical_reserve_industry_capacity = num_of_civilian_factories }
	multiply_temp_variable = { black_plague_medical_reserve_industry_capacity = constant:black_plague_medical_reserve.capacity_per_civilian_factory }
	clamp_temp_variable = {
		var = black_plague_medical_reserve_industry_capacity
		min = constant:black_plague_response_value.zero
		max = constant:black_plague_medical_reserve.capacity_industry_contribution_cap
	}
	add_to_variable = { black_plague_medical_reserve_capacity = black_plague_medical_reserve_industry_capacity }
	set_temp_variable = { black_plague_medical_reserve_knowledge_capacity = black_plague_countermeasure_progress }
	multiply_temp_variable = { black_plague_medical_reserve_knowledge_capacity = constant:black_plague_medical_reserve.capacity_per_countermeasure_point }
	add_to_variable = { black_plague_medical_reserve_capacity = black_plague_medical_reserve_knowledge_capacity }
	clamp_variable = {
		var = black_plague_medical_reserve_capacity
		min = constant:black_plague_medical_reserve.minimum_capacity
		max = constant:black_plague_medical_reserve.maximum_capacity
	}
	clamp_variable = {
		var = black_plague_medical_reserve
		min = constant:black_plague_response_value.zero
		max = constant:black_plague_medical_reserve.maximum_capacity
	}
	set_variable = { black_plague_medical_reserve_room = black_plague_medical_reserve_capacity }
	subtract_from_variable = { black_plague_medical_reserve_room = black_plague_medical_reserve }
	clamp_variable = {
		var = black_plague_medical_reserve_room
		min = constant:black_plague_response_value.zero

# multiplied once at each earned milestone, then subtracted without unary
# variable negation.
black_plague_apply_current_state_response_deltas = {
	if = {
		limit = { has_state_flag = black_plague_doctor_wu_foreign_treatment }
		add_to_temp_variable = { black_plague_response_treatment_gain = constant:black_plague_response_result.doctor_wu_treatment }
		add_to_temp_variable = { black_plague_response_mortality_reduction = constant:black_plague_response_result.doctor_wu_mortality }
		add_to_temp_variable = { black_plague_response_containment_gain = constant:black_plague_response_result.doctor_wu_containment }
	}
	if = {
		limit = {
			ROOT = {
				OR = {
					check_variable = {
						var = black_plague_countermeasure_progress
						value = constant:black_plague_response_threshold.countermeasure_cleanup_bonus
						compare = greater_than_or_equals
					}
					has_country_flag = black_plague_accelerated_cleanup_protocol
				}
			}
		}
		multiply_temp_variable = { black_plague_response_infestation_reduction = constant:black_plague_response_result.cleanup_reduction_factor }
		multiply_temp_variable = { black_plague_response_load_reduction = constant:black_plague_response_result.cleanup_reduction_factor }
		multiply_temp_variable = { black_plague_response_relapse_reduction = constant:black_plague_response_result.cleanup_reduction_factor }
	}
	if = {
		limit = {
			ROOT = {
				check_variable = {
					var = black_plague_countermeasure_progress
					value = constant:black_plague_response_threshold.countermeasure_spread_bonus
					compare = greater_than_or_equals
				}
			}
		}
		multiply_temp_variable = { black_plague_response_spread_reduction = constant:black_plague_response_result.spread_reduction_factor }
		multiply_temp_variable = { black_plague_response_exposure_reduction = constant:black_plague_response_result.spread_reduction_factor }
	}
	if = {
		limit = { check_variable = { black_plague_response_infestation_reduction > constant:black_plague_response_value.zero } }
		subtract_from_variable = { black_plague_rat_infestation = black_plague_response_infestation_reduction }
	}
	if = {
		limit = { check_variable = { black_plague_response_load_reduction > constant:black_plague_response_value.zero } }
		subtract_from_variable = { black_plague_load = black_plague_response_load_reduction }
	}
	if = {
		limit = { check_variable = { black_plague_response_spread_reduction > constant:black_plague_response_value.zero } }
		subtract_from_variable = { black_plague_spread_pressure = black_plague_response_spread_reduction }
	}
	if = {
		limit = { check_variable = { black_plague_response_mortality_reduction > constant:black_plague_response_value.zero } }
		subtract_from_variable = { black_plague_mortality_pressure = black_plague_response_mortality_reduction }
	}
	if = {
		limit = { check_variable = { black_plague_response_exposure_reduction > constant:black_plague_response_value.zero } }
		subtract_from_variable = { black_plague_incoming_exposure = black_plague_response_exposure_reduction }
	}
	if = {
		limit = { check_variable = { black_plague_response_relapse_reduction > constant:black_plague_response_value.zero } }
		subtract_from_variable = { black_plague_relapse_risk = black_plague_response_relapse_reduction }
	}
	add_to_variable = { black_plague_containment = black_plague_response_containment_gain }
	add_to_variable = { black_plague_treatment_coverage = black_plague_response_treatment_gain }
	add_to_variable = { black_plague_rat_infestation = black_plague_response_infestation_gain }
	add_to_variable = { black_plague_mortality_pressure = black_plague_response_mortality_gain }
	add_to_variable = { black_plague_relapse_risk = black_plague_response_relapse_gain }
	black_plague_ensure_current_state_values = yes
	black_plague_refresh_current_state_modifier = yes
	set_global_flag = black_plague_mapmode_dirty
}

black_plague_reconcile_current_state_cleanup_proof = {
	if = {
		limit = { black_plague_response_cleanup_requirements_are_met = yes }
		set_state_flag = black_plague_cleanup_complete
	}
	else = { clr_state_flag = black_plague_cleanup_complete }
}

black_plague_add_current_state_cleanup_attempt = {
	add_to_variable = { black_plague_cleanup_attempts = constant:black_plague_response_value.one }
	black_plague_reconcile_current_state_cleanup_proof = yes
}
