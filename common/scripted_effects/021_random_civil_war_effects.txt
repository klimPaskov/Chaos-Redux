# ============================================================================
# CHAOS REDUX - EVENT 021 RANDOM CIVIL WAR EFFECTS
# ============================================================================
# Reusable, bounded scripted-system contracts for Event 021. The parent event,
# decision, and scenario files own narrative text, country creation, state
# transfer, wars, ideas, AI, and shared cluster/event-log registration. These
# effects own the state that those surfaces share: hidden Fracture Pressure,
# visible State Authority, target scoring, plan reservations, front records,
# evolution scheduling, exposure, queue/cap bookkeeping, settlement, and
# cleanup.
#
# Public input convention:
# - normal variables named `random_civil_war_*` are persistent inputs or
#   outputs on the current scope;
# - temporary variables with the same prefix are one-call inputs/outputs;
# - event targets `random_civil_war_host_country`,
#   `random_civil_war_current_state`, `random_civil_war_anchor_state`, and
#   `random_civil_war_current_actor` are short-lived parent call-frame targets;
# - global arrays store bounded registries and are changed only by explicit
#   register/unregister/rollback helpers.
#
# The only world-facing review helper is `event021_global_review_batch`. It
# opens a fixed review budget. The parent must feed countries through
# `event021_review_current_country` from an existing bounded hook. No default
# any_country daily/monthly loop is defined here.
#
# Event 006 adapter boundary: `event021_prepare_event6_admission` calls only
# the existing static admission/preflight predicates after bridging the package
# id. It never calls `independence_wave_prepare_country_origin`, never marks an
# Event 006 origin/evolution/league state, and never applies an Event 006
# package dispatcher. The parent owns the package-specific execution step.
# ============================================================================

# -----------------------------------------------------------------------------
# Core state: Fracture Pressure and State Authority
# -----------------------------------------------------------------------------

# COUNTRY scope. Idempotently creates the persistent state used by all other
# helpers, then refreshes derived pressure and authority bands. Call after a
# country is created, restored, annexed, or otherwise changes crisis inputs.
event021_initialize_country_state = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_fracture_pressure } }
		set_variable = { random_civil_war_fracture_pressure = constant:random_civil_war_pressure.base }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_state_authority } }
		set_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.default }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_pressure_modifier } }
		set_variable = { random_civil_war_pressure_modifier = constant:random_civil_war_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_generation } }
		set_variable = { random_civil_war_generation = constant:random_civil_war_value.zero }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_settled_front_count } }
		set_variable = { random_civil_war_settled_front_count = constant:random_civil_war_value.zero }
	}
	event021_refresh_country_state = yes
}

# COUNTRY scope. Recomputes hidden Fracture Pressure from live country values
# and parent-supplied bounded component flags/counters. It intentionally does
# not treat diversity alone as pressure: a route, grievance, organization,
# territorial access, or institutional access must still be supplied by the
# target planner.
event021_refresh_country_state = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_fracture_pressure } }
		set_variable = { random_civil_war_fracture_pressure = constant:random_civil_war_pressure.base }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_state_authority } }
		set_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.default }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_pressure_modifier } }
		set_variable = { random_civil_war_pressure_modifier = constant:random_civil_war_value.zero }
	}
	set_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.base }
	if = {
		limit = { has_stability < constant:random_civil_war_pressure.stability_low }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.political_grievance }
	}
	if = {
		limit = { has_stability > constant:random_civil_war_pressure.stability_high }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.cohesive_relief }
	}
	if = {
		limit = { has_war_support < constant:random_civil_war_pressure.war_support_low }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.political_grievance }
	}
	if = {
		limit = { has_war_support > constant:random_civil_war_pressure.war_support_high }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.cohesive_relief }
	}
	if = {
		limit = { has_war = yes }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.active_war }
	}
	if = {
		limit = { has_country_flag = random_civil_war_political_competition_active }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.political_grievance }
	}
	if = {
		limit = { has_country_flag = random_civil_war_military_loyalty_low }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.military_disloyalty }
	}
	if = {
		limit = { has_country_flag = random_civil_war_regional_grievance_active }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.regional_identity }
	}
	if = {
		limit = { has_country_flag = random_civil_war_state_capacity_low }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.state_capacity_loss }
	}
	if = {
		limit = { has_country_flag = random_civil_war_nearby_conflict_exposure }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.nearby_conflict }
	}
	if = {
		limit = { has_country_flag = random_civil_war_recent_civil_war_memory }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.event_memory }
	}
	if = {
		limit = { has_country_flag = random_civil_war_contested_territory }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.territorial_loss }
	}
	if = {
		limit = { has_country_flag = random_civil_war_key_state_lost }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.territorial_loss }
	}
	if = {
		limit = { has_country_flag = random_civil_war_settlement_relief_active }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.settlement_relief }
	}
	if = {
		limit = { has_country_flag = random_civil_war_reconstruction_active }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.reconstruction_relief }
	}
	if = {
		limit = { has_country_flag = random_civil_war_authority_failing }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.authority_failing }
	}
	if = {
		limit = { has_country_flag = random_civil_war_authority_collapse }
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.authority_collapse }
	}
	set_temp_variable = { event021_pressure_chaos_tier = constant:chaos_meter_tier_id.tier_0 }
	get_current_chaos_tier_value = yes
	if = {
		limit = {
			check_variable = {
				var = current_chaos_tier_value
				value = constant:chaos_meter_tier_id.tier_1
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { event021_pressure_recalculated = constant:random_civil_war_pressure.chaos_tier }
	}
	if = {
		limit = { has_variable = random_civil_war_contested_state_count }
		set_temp_variable = { event021_pressure_state_component = random_civil_war_contested_state_count }
		multiply_temp_variable = { event021_pressure_state_component = constant:random_civil_war_pressure.territorial_loss }
		add_to_temp_variable = { event021_pressure_recalculated = event021_pressure_state_component }
	}
	if = {
		limit = { has_variable = random_civil_war_pressure_modifier }
		add_to_temp_variable = { event021_pressure_recalculated = random_civil_war_pressure_modifier }
	}
	clamp_temp_variable = {
		var = event021_pressure_recalculated
		min = constant:random_civil_war_pressure.minimum
		max = constant:random_civil_war_pressure.maximum
	}
	set_variable = { random_civil_war_fracture_pressure = event021_pressure_recalculated }
	set_variable = { random_civil_war_pressure_snapshot_date = global.date }
	if = {
		limit = {
			has_country_flag = random_civil_war_recent_civil_war_memory
			has_variable = random_civil_war_memory_until
			check_variable = { var = global.date value = random_civil_war_memory_until compare = greater_than_or_equals }
		}
		clr_country_flag = random_civil_war_recent_civil_war_memory
	}
	if = {
		limit = {
			has_country_flag = random_civil_war_recent_target
			has_variable = random_civil_war_recent_target_until
			check_variable = { var = global.date value = random_civil_war_recent_target_until compare = greater_than_or_equals }
		}
		clr_country_flag = random_civil_war_recent_target
	}
	event021_update_pressure_band = yes
	event021_update_authority_band = yes
}

# COUNTRY scope. `random_civil_war_pending_pressure_delta` is an optional
# normal-variable input. If absent, the helper applies the centralized
# reconstruction relief. The input is consumed into the persistent modifier so
# a decision cannot lose its one-shot change on a later bounded refresh.
event021_reduce_pressure = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_pending_pressure_delta } }
		set_variable = { random_civil_war_pending_pressure_delta = constant:random_civil_war_pressure.reconstruction_relief }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_pressure_modifier } }
		set_variable = { random_civil_war_pressure_modifier = constant:random_civil_war_value.zero }
	}
	add_to_variable = { random_civil_war_pressure_modifier = random_civil_war_pending_pressure_delta }
	clear_variable = random_civil_war_pending_pressure_delta
	event021_refresh_country_state = yes
}

# COUNTRY scope. `random_civil_war_pending_authority_delta` is an optional
# normal-variable input. Without one, the helper applies the centralized
# administrative restoration delta. Condition flags add the loss or recovery
# associated with the current capital, supply, and settlement state.
event021_add_authority = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_pending_authority_delta } }
		set_variable = { random_civil_war_pending_authority_delta = constant:random_civil_war_authority.admin_restored }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_state_authority } }
		set_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.default }
	}
	add_to_variable = { random_civil_war_state_authority = random_civil_war_pending_authority_delta }
	if = {
		limit = { has_country_flag = random_civil_war_capital_lost }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.capital_lost }
	}
	if = {
		limit = { has_country_flag = random_civil_war_capital_secured }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.capital_secured }
	}
	if = {
		limit = { has_country_flag = random_civil_war_supply_lost }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.supply_lost }
	}
	if = {
		limit = { has_country_flag = random_civil_war_supply_secured }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.supply_secured }
	}
	if = {
		limit = { has_country_flag = random_civil_war_admin_restored }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.admin_restored }
	}
	if = {
		limit = { has_country_flag = random_civil_war_negotiation_active }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.negotiation }
	}
	if = {
		limit = { has_country_flag = random_civil_war_repression_active }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.repression }
	}
	if = {
		limit = { has_country_flag = random_civil_war_foreign_intervention_active }
		add_to_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.foreign_intervention }
	}
	clamp_variable = {
		var = random_civil_war_state_authority
		min = constant:random_civil_war_authority.minimum
		max = constant:random_civil_war_authority.maximum
	}
	clear_variable = random_civil_war_pending_authority_delta
	event021_update_authority_band = yes
}

# COUNTRY scope. Refreshes the visible four-band State Authority ladder.
event021_update_authority_band = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_state_authority } }
		set_variable = { random_civil_war_state_authority = constant:random_civil_war_authority.default }
	}
	clamp_variable = {
		var = random_civil_war_state_authority
		min = constant:random_civil_war_authority.minimum
		max = constant:random_civil_war_authority.maximum
	}
	if = {
		limit = {
			has_country_flag = random_civil_war_active
			has_country_flag = random_civil_war_achievement_history_started
			NOT = { has_country_flag = random_civil_war_internal_fronts_resolved }
			OR = {
				NOT = { has_variable = random_civil_war_minimum_authority }
				check_variable = { var = random_civil_war_state_authority value = random_civil_war_minimum_authority compare = less_than }
			}
		}
		set_variable = { random_civil_war_minimum_authority = random_civil_war_state_authority }
	}
	set_variable = { random_civil_war_authority_band = constant:random_civil_war_authority_band.collapse }
	clr_country_flag = random_civil_war_authority_cohesive
	clr_country_flag = random_civil_war_authority_contested
	clr_country_flag = random_civil_war_authority_failing
	clr_country_flag = random_civil_war_authority_collapse
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_state_authority
				value = constant:random_civil_war_authority.collapse_maximum
				compare = greater_than
			}
		}
		set_variable = { random_civil_war_authority_band = constant:random_civil_war_authority_band.failing }
		set_country_flag = random_civil_war_authority_failing
	}
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_state_authority
				value = constant:random_civil_war_authority.failing_maximum
				compare = greater_than
			}
		}
		set_variable = { random_civil_war_authority_band = constant:random_civil_war_authority_band.contested }
		clr_country_flag = random_civil_war_authority_failing
		set_country_flag = random_civil_war_authority_contested
	}
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_state_authority
				value = constant:random_civil_war_authority.contested_maximum
				compare = greater_than
			}
		}
		set_variable = { random_civil_war_authority_band = constant:random_civil_war_authority_band.cohesive }
		clr_country_flag = random_civil_war_authority_contested
		set_country_flag = random_civil_war_authority_cohesive
	}
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_state_authority
				value = constant:random_civil_war_authority.collapse_maximum
				compare = less_than_or_equals
			}
		}
		set_variable = { random_civil_war_authority_band = constant:random_civil_war_authority_band.collapse }
		clr_country_flag = random_civil_war_authority_failing
		set_country_flag = random_civil_war_authority_collapse
		if = {
			limit = {
				has_country_flag = random_civil_war_active
				NOT = { has_country_flag = random_civil_war_internal_fronts_resolved }
			}
			set_country_flag = random_civil_war_authority_ever_collapsed
		}
	}
}

# COUNTRY scope. Refreshes the hidden pressure band used by the bounded global
# review. The band is a derived value and never starts a crisis by itself.
event021_update_pressure_band = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_fracture_pressure } }
		set_variable = { random_civil_war_fracture_pressure = constant:random_civil_war_pressure.base }
	}
	clamp_variable = {
		var = random_civil_war_fracture_pressure
		min = constant:random_civil_war_pressure.minimum
		max = constant:random_civil_war_pressure.maximum
	}
	set_variable = { random_civil_war_pressure_band = constant:random_civil_war_pressure_band.stable }
	clr_country_flag = random_civil_war_pressure_stable
	clr_country_flag = random_civil_war_pressure_exposed
	clr_country_flag = random_civil_war_pressure_fractured
	clr_country_flag = random_civil_war_pressure_critical
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_fracture_pressure
				value = constant:random_civil_war_pressure.exposed_threshold
				compare = greater_than_or_equals
			}
		}
		set_variable = { random_civil_war_pressure_band = constant:random_civil_war_pressure_band.exposed }
		set_country_flag = random_civil_war_pressure_exposed
	}
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_fracture_pressure
				value = constant:random_civil_war_pressure.fractured_threshold
				compare = greater_than_or_equals
			}
		}
		set_variable = { random_civil_war_pressure_band = constant:random_civil_war_pressure_band.fractured }
		clr_country_flag = random_civil_war_pressure_exposed
		set_country_flag = random_civil_war_pressure_fractured
	}
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_fracture_pressure
				value = constant:random_civil_war_pressure.critical_threshold
				compare = greater_than_or_equals
			}
		}
		set_variable = { random_civil_war_pressure_band = constant:random_civil_war_pressure_band.critical }
		clr_country_flag = random_civil_war_pressure_fractured
		set_country_flag = random_civil_war_pressure_critical
	}
	if = {
		limit = {
			check_variable = {
				var = random_civil_war_fracture_pressure
				value = constant:random_civil_war_pressure.exposed_threshold
				compare = less_than
			}
		}
		set_country_flag = random_civil_war_pressure_stable
	}
}

# COUNTRY scope. Produces temporary `event021_target_weight` and persistent
# `random_civil_war_target_weight`. Call once for each candidate selected by a
# bounded parent pool. A zero result is fail-closed and never creates an actor.
event021_random_civil_war_prepare_target = {
	event021_initialize_country_state = yes
	set_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.base }
	if = {
		limit = { event021_country_can_be_target = yes }
		set_temp_variable = { event021_pressure_component = random_civil_war_fracture_pressure }
		multiply_temp_variable = { event021_pressure_component = constant:random_civil_war_target_weight.pressure_per_point }
		add_to_temp_variable = { event021_target_weight = event021_pressure_component }
		if = {
			limit = { random_civil_war_authority_is_failing = yes }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.authority_failing }
		}
		if = {
			limit = { random_civil_war_authority_is_collapsed = yes }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.authority_collapse }
		}
		if = {
			limit = { random_civil_war_has_valid_actor_route = yes }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.valid_actor_route }
		}
		if = {
			limit = {
				is_major = yes
				random_civil_war_major_severity_allowed = yes
			}
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.major_stage }
		}
		if = {
			limit = { has_country_flag = random_civil_war_nearby_conflict_exposure }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.nearby_exposure }
		}
		if = {
			limit = { has_country_flag = random_civil_war_recent_civil_war_memory }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.recent_memory }
		}
		if = {
			limit = { is_subject = yes }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.subject }
		}
		if = {
			limit = { has_country_flag = random_civil_war_player_target }
			add_to_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.player }
		}
		if = {
			limit = { is_major = yes }
			multiply_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.major_target_factor }
		}
	}
	else = {
		set_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.minimum }
	}
	if = {
		limit = { has_country_flag = random_civil_war_target_reserved }
		set_temp_variable = { event021_target_weight = constant:random_civil_war_target_weight.minimum }
	}
	clamp_temp_variable = {
		var = event021_target_weight
		min = constant:random_civil_war_target_weight.minimum
		max = constant:random_civil_war_target_weight.maximum
	}
	round_temp_variable = event021_target_weight
	set_variable = { random_civil_war_target_weight = event021_target_weight }
	if = {
		limit = {
			check_variable = {
				var = event021_target_weight
				value = constant:random_civil_war_target_weight.minimum
				compare = greater_than
			}
		}
		set_country_flag = random_civil_war_target_pool_candidate
		save_event_target_as = random_civil_war_target_country
	}
	else = { clr_country_flag = random_civil_war_target_pool_candidate }
}

# COUNTRY scope. Produces the four dynamic route weights as temporary values
# for the parent selector. The selector may use a random_list or its own
# bounded pool; this helper does not invent an actor when all routes are zero.
event021_prepare_archetype_weights = {
	set_temp_variable = { event021_ideological_weight = constant:random_civil_war_target_weight.minimum }
	set_temp_variable = { event021_legal_weight = constant:random_civil_war_target_weight.minimum }
	set_temp_variable = { event021_regional_weight = constant:random_civil_war_target_weight.minimum }
	set_temp_variable = { event021_command_weight = constant:random_civil_war_target_weight.minimum }
	set_temp_variable = { event021_event6_weight = constant:random_civil_war_target_weight.minimum }
	set_temp_variable = { event021_same_tag_weight = constant:random_civil_war_target_weight.minimum }
	if = {
		limit = { random_civil_war_ideological_route_valid = yes }
		set_temp_variable = { event021_ideological_weight = constant:random_civil_war_target_weight.valid_actor_route }
	}
	if = {
		limit = { random_civil_war_legal_route_valid = yes }

# COUNTRY scope. Opens the single bounded opening transaction used to reserve
# connected states and anchors. A second plan cannot overwrite the first one.
# The parent calls `event021_reserve_state` once per selected state.
event021_begin_opening_transaction = {
	if = {
		limit = { NOT = { has_global_flag = random_civil_war_plan_active } }
		set_global_flag = random_civil_war_plan_active
		set_global_flag = random_civil_war_plan_seed_pending
		set_global_flag = random_civil_war_planning_lock
		clear_array = global.random_civil_war_plan_states
		clear_array = global.random_civil_war_reserved_states

		save_event_target_as = random_civil_war_host_country
	}
	else = { set_global_flag = random_civil_war_plan_rejected }
}

# STATE scope through the `random_civil_war_current_state` event target. Adds a
# state to both the temporary plan and the global collision reservation. The
# seed flag is consumed after the first state; later states need adjacency or

event021_reserve_state = {

				add_to_variable = { global.random_civil_war_plan_state_count = constant:random_civil_war_value.one }
				set_variable = { global.random_civil_war_last_reserved_state = THIS }
			}
		}
		clr_global_flag = random_civil_war_plan_seed_pending
	}
}

# COUNTRY scope. Records the caller's already-proven force and stockpile plan.
# Optional normal inputs are `random_civil_war_requested_force_divisions`,
# `random_civil_war_requested_force_manpower_k`, and the three equipment pools.
# When absent, force output is derived from the current country's divisions,
# factories, and manpower instead of using a fixed half-stockpile split.
event021_prepare_force_package = {
	if = {
		limit = { NOT = { has_variable = random_civil_war_requested_force_divisions } }
		set_variable = { random_civil_war_requested_force_divisions = num_divisions }
		multiply_variable = { random_civil_war_requested_force_divisions = constant:random_civil_war_force.actor_division_ratio }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_requested_force_manpower_k } }
		set_variable = { random_civil_war_requested_force_manpower_k = manpower_k }
		multiply_variable = { random_civil_war_requested_force_manpower_k = constant:random_civil_war_force.actor_manpower_ratio }
	}
	clamp_variable = {
		var = random_civil_war_requested_force_divisions
		min = constant:random_civil_war_capacity.minimum_force_divisions
		max = constant:random_civil_war_capacity.maximum_force_divisions
	}
	if = {
		limit = {
			OR = {
				has_country_flag = random_civil_war_one_state_route
				has_country_flag = random_civil_war_all_island_route_ready
			}
		}
		set_variable = { random_civil_war_requested_force_divisions = constant:random_civil_war_force.one_state_division_count }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_requested_force_equipment } }
		set_variable = { random_civil_war_requested_force_equipment = num_of_military_factories }
		multiply_variable = { random_civil_war_requested_force_equipment = constant:random_civil_war_force.actor_equipment_ratio }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_requested_force_support_equipment } }
		set_variable = { random_civil_war_requested_force_support_equipment = constant:random_civil_war_force.minimum_support_equipment }
	}
	if = {
		limit = { NOT = { has_variable = random_civil_war_requested_force_artillery_equipment } }
		set_variable = { random_civil_war_requested_force_artillery_equipment = constant:random_civil_war_force.minimum_artillery_equipment }
	}
	if = {
		limit = {
			OR = {
				has_manpower < constant:random_civil_war_force.minimum_manpower
				NOT = { num_divisions > constant:random_civil_war_value.zero }
			}
		}
		set_country_flag = random_civil_war_force_package_invalid
	}
	else = { set_country_flag = random_civil_war_force_package_ready }
}

# COUNTRY scope. Registers one stable front row. Inputs are normal
# `random_civil_war_front_id` and the `random_civil_war_anchor_state` event
# target. Parallel fronts are represented by aligned global arrays. The
# metadata row is written at the same commit boundary as the ID, actor, and
# anchor so lifecycle, settlement, successor, and event-detail consumers never
# have to infer a front from a partially populated registry.
event021_register_front = {
	if = {
		limit = {
			has_variable = random_civil_war_front_id
			NOT = { random_civil_war_front_id_is_registered = yes }
			has_event_target = random_civil_war_anchor_state
		}
		add_to_array = { array = global.random_civil_war_front_ids value = random_civil_war_front_id }
		add_to_array = { array = global.random_civil_war_front_actors value = THIS }
		event_target:random_civil_war_anchor_state = { add_to_array = { array = global.random_civil_war_front_states } }
		# Keep all front metadata aligned with the three core rows above. The
		# current actor carries the durable receipts; the crisis host and anchor
		# are promoted to scope rows for consumers that need a stable pointer.
		save_event_target_as = event021_front_metadata_host
		if = {
			limit = { has_variable = random_civil_war_crisis_host_scope }
			var:random_civil_war_crisis_host_scope = { save_event_target_as = event021_front_metadata_host }
		}
		add_to_array = { array = global.random_civil_war_front_hosts value = event_target:event021_front_metadata_host }
		event_target:random_civil_war_anchor_state = { add_to_array = { array = global.random_civil_war_front_capitals } }
		set_temp_variable = { event021_front_metadata_generation = constant:random_civil_war_value.zero }
		if = {
			limit = { has_variable = random_civil_war_generation }
			set_temp_variable = { event021_front_metadata_generation = random_civil_war_generation }
		}
		add_to_array = { array = global.random_civil_war_front_generations value = event021_front_metadata_generation }
		set_temp_variable = { event021_front_metadata_crisis_id = constant:random_civil_war_value.zero }
		if = {
			limit = { has_variable = random_civil_war_crisis_id }
			set_temp_variable = { event021_front_metadata_crisis_id = random_civil_war_crisis_id }
		}
		add_to_array = { array = global.random_civil_war_front_crisis_ids value = event021_front_metadata_crisis_id }
		set_temp_variable = { event021_front_metadata_archetype = constant:random_civil_war_value.zero }
		if = {
			limit = { has_variable = random_civil_war_archetype }
			set_temp_variable = { event021_front_metadata_archetype = random_civil_war_archetype }
		}
		add_to_array = { array = global.random_civil_war_front_archetypes value = event021_front_metadata_archetype }
		set_temp_variable = { event021_front_metadata_route_source = constant:random_civil_war_value.zero }
		if = {
			limit = { has_variable = random_civil_war_evidence_last_source }
			set_temp_variable = { event021_front_metadata_route_source = random_civil_war_evidence_last_source }
		}
		add_to_array = { array = global.random_civil_war_front_route_sources value = event021_front_metadata_route_source }
		set_temp_variable = { event021_front_metadata_package_id = constant:random_civil_war_value.zero }
		if = {
			limit = { has_variable = random_civil_war_event6_package_id }
			set_temp_variable = { event021_front_metadata_package_id = random_civil_war_event6_package_id }
		}
		add_to_array = { array = global.random_civil_war_front_package_ids value = event021_front_metadata_package_id }
		set_temp_variable = { event021_front_metadata_region_count = constant:random_civil_war_value.one }
		if = {
			limit = { has_variable = random_civil_war_opening_state_count }
			set_temp_variable = { event021_front_metadata_region_count = random_civil_war_opening_state_count }
		}
		add_to_array = { array = global.random_civil_war_front_region_counts value = event021_front_metadata_region_count }
		set_temp_variable = { event021_front_metadata_goal = constant:random_civil_war_value.zero }
		if = {
			limit = { has_variable = random_civil_war_front_goal }
			set_temp_variable = { event021_front_metadata_goal = random_civil_war_front_goal }
		}
		add_to_array = { array = global.random_civil_war_front_goals value = event021_front_metadata_goal }
		add_to_array = { array = global.random_civil_war_front_statuses value = constant:random_civil_war_front_status.active }
		set_temp_variable = { event021_front_metadata_settlement = constant:random_civil_war_value.zero }

			limit = { has_variable = random_civil_war_settlement_type }
			set_temp_variable = { event021_front_metadata_settlement = random_civil_war_settlement_type }
		}
		add_to_array = { array = global.random_civil_war_front_settlements value = event021_front_metadata_settlement }
		add_to_array = { array = global.random_civil_war_front_registered_dates value = global.date }
		set_country_flag = random_civil_war_front_registered
		set_variable = { random_civil_war_front_registered_date = global.date }
		if = {
			limit = { NOT = { has_global_flag = random_civil_war_capacity_initialized } }

		add_to_variable = { global.random_civil_war_active_front_count = constant:random_civil_war_value.one }
	}

# COUNTRY scope. Removes the current front row and every aligned metadata row,
# then updates the bounded front cap counter. The parent should call this once
# per actor/front, not from a world-wide periodic loop.
event021_unregister_front = {
	if = {
		limit = { has_country_flag = random_civil_war_front_registered }
		if = {
			limit = { has_variable = random_civil_war_front_id }
			set_temp_variable = { event021_front_unregister_index = global.random_civil_war_front_ids^num }
			set_temp_variable = { event021_front_unregister_found = constant:random_civil_war_value.zero }
			while_loop_effect = {
				limit = {
					check_variable = {
						var = event021_front_unregister_index
						value = constant:random_civil_war_value.zero
						compare = greater_than
					}
					check_variable = {
						var = event021_front_unregister_found
						value = constant:random_civil_war_value.zero
						compare = equals
					}
				}
				subtract_from_temp_variable = { event021_front_unregister_index = constant:random_civil_war_value.one }
				if = {
					limit = {
						check_variable = {
							var = global.random_civil_war_front_ids^event021_front_unregister_index
							value = random_civil_war_front_id
							compare = equals
						}
					}
					set_temp_variable = { event021_front_unregister_found = constant:random_civil_war_value.one }
				}
			}
			if = {
				limit = {
					check_variable = {
						var = event021_front_unregister_found
						value = constant:random_civil_war_value.one
						compare = equals
					}
				}
				remove_from_array = { array = global.random_civil_war_front_ids index = event021_front_unregister_index }
				if = { limit = { check_variable = { var = global.random_civil_war_front_actors^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_actors index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_states^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_states index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_hosts^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_hosts index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_capitals^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_capitals index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_generations^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_generations index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_crisis_ids^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_crisis_ids index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_archetypes^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_archetypes index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_route_sources^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_route_sources index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_package_ids^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_package_ids index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_region_counts^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_region_counts index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_goals^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_goals index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_statuses^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_statuses index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_settlements^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_settlements index = event021_front_unregister_index } }
				if = { limit = { check_variable = { var = global.random_civil_war_front_registered_dates^num value = event021_front_unregister_index compare = greater_than } } remove_from_array = { array = global.random_civil_war_front_registered_dates index = event021_front_unregister_index } }

					event_target:random_civil_war_anchor_state = { remove_from_array = { array = global.random_civil_war_front_states value = THIS } }

			}
		}
		else = {
			remove_from_array = { array = global.random_civil_war_front_actors value = THIS }
			if = {
				limit = { has_event_target = random_civil_war_anchor_state }
				event_target:random_civil_war_anchor_state = { remove_from_array = { array = global.random_civil_war_front_states value = THIS } }
			}
		}
		clr_country_flag = random_civil_war_front_registered

					var = global.random_civil_war_critical_queue_budget

				}
			}
		}

		clear_variable = random_civil_war_critical_exit_reason
		clear_variable = random_civil_war_critical_exit_date
		clear_variable = random_civil_war_critical_last_attempt_date
		add_to_array = { array = global.random_civil_war_critical_queue value = THIS }
		if = {
			limit = { NOT = { has_global_flag = random_civil_war_queue_initialized } }
			set_variable = { global.random_civil_war_critical_queue_count = constant:random_civil_war_value.zero }
			set_global_flag = random_civil_war_queue_initialized
		}
		add_to_variable = { global.random_civil_war_critical_queue_count = constant:random_civil_war_value.one }
		if = {
			limit = { has_global_flag = random_civil_war_review_batch_active }
			subtract_from_variable = { global.random_civil_war_critical_queue_budget = constant:random_civil_war_value.one }
		}
	}
}

# COUNTRY scope. Computes a recurrence score from unresolved pressure,
# post-war memory, settlement quality, and persistent network flags. It marks
# recurrence eligibility but never immediately starts another crisis.
event021_prepare_recurrence = {
	set_variable = { random_civil_war_recurrence_score = constant:random_civil_war_value.zero }
	if = {
		limit = { has_variable = random_civil_war_fracture_pressure }
		set_variable = { random_civil_war_recurrence_score = random_civil_war_fracture_pressure }
	}
	if = {
		limit = { has_country_flag = random_civil_war_recent_civil_war_memory }
		add_to_variable = { random_civil_war_recurrence_score = constant:random_civil_war_settlement_tuning.postwar_memory_pressure }
	}
	if = {
		limit = { has_country_flag = random_civil_war_failed_settlement }
		add_to_variable = { random_civil_war_recurrence_score = constant:random_civil_war_settlement_tuning.failed_settlement_pressure }
	}
	if = {
		limit = { has_country_flag = random_civil_war_authority_collapse }
		add_to_variable = { random_civil_war_recurrence_score = constant:random_civil_war_pressure.authority_collapse }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = random_civil_war_settlement_government_victory
				has_country_flag = random_civil_war_settlement_coalition
			}
		}
		add_to_variable = { random_civil_war_recurrence_score = constant:random_civil_war_settlement_tuning.settlement_recurrence_relief }
	}
	clamp_variable = {
		var = random_civil_war_recurrence_score
		min = constant:random_civil_war_pressure.minimum
		max = constant:random_civil_war_pressure.maximum
	}
	if = {
		limit = { random_civil_war_recurrence_valid = yes }
		set_country_flag = random_civil_war_recurrence_eligible
		set_variable = { random_civil_war_recurrence_review_date = global.date }
		add_to_variable = { random_civil_war_recurrence_review_date = constant:random_civil_war_timing.recurrence_minimum_days }
	}
	else = { clr_country_flag = random_civil_war_recurrence_eligible }
}
