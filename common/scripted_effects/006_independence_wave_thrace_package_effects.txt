independence_wave_refresh_bax_lifecycle = {
	if = {
		limit = { is_independence_wave_bax_package = yes }
		remove_ideas = { bax_divided_thrace_authority bax_thrace_compact }
		if = {
			limit = { has_stable_independence_wave_bax_ledgers = yes }
			add_ideas = bax_thrace_compact

		}

}
