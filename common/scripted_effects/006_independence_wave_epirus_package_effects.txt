independence_wave_refresh_bbx_lifecycle = {
	if = {
		limit = { is_independence_wave_bbx_package = yes }
		remove_ideas = { bbx_divided_epirus_authority bbx_epirus_compact }
		if = {
			limit = { has_stable_independence_wave_bbx_ledgers = yes }
			add_ideas = bbx_epirus_compact
			set_country_flag = independence_wave_bbx_compact_stabilized

}
