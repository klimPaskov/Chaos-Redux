independence_wave_refresh_mnt_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_mnt_package = yes }
		remove_ideas = { mnt_divided_mountain_authority mnt_mountain_state_compact }
		if = {
			limit = { has_stable_independence_wave_mnt_compact = yes }
			add_ideas = mnt_mountain_state_compact
			set_country_flag = independence_wave_mnt_compact_stabilized
		}
		else = {
			add_ideas = mnt_divided_mountain_authority
			clr_country_flag = independence_wave_mnt_compact_stabilized
		}
	}
}

# Inputs: temporary `independence_wave_mnt_cohesion_delta` and
# `independence_wave_mnt_crown_delta`.
independence_wave_change_mnt_compact_values = {
	if = {
		limit = { is_independence_wave_mnt_package = yes }
		add_to_variable = { independence_wave_mnt_mountain_cohesion = independence_wave_mnt_cohesion_delta }
		add_to_variable = { independence_wave_mnt_crown_legitimacy = independence_wave_mnt_crown_delta }
		clamp_variable = {
			var = independence_wave_mnt_mountain_cohesion
			min = constant:independence_wave_montenegro_pressure.minimum
			max = constant:independence_wave_montenegro_pressure.maximum
		}
		clamp_variable = {
			var = independence_wave_mnt_crown_legitimacy
			min = constant:independence_wave_montenegro_pressure.minimum
			max = constant:independence_wave_montenegro_pressure.maximum
		}
		independence_wave_refresh_mnt_compact_lifecycle = yes
	}
}
