# ============================================================================
# CHAOS REDUX - EVENT 016 FIRST-PROTOTYPE REPORT EFFECTS
# ============================================================================
# A first Prototype in each project family produces one ordinary report. The
# report does not grant a second project reward: it only records the family,
# presents a public or classified governance choice, and adjusts the existing
# Directorate causal values through the shared bounded effects.
# ============================================================================

# COUNTRY scope. Inputs are the temporary project family and requested stage
# supplied by the project output bridge.
brilliant_scientist_dispatch_breakthrough_report = {
	if = {
		limit = {
			check_variable = { brilliant_scientist_requested_project_stage = constant:brilliant_scientist_project_stage.prototype }
			NOT = { brilliant_scientist_breakthrough_report_already_recorded = yes }
			NOT = { is_in_array = { array = brilliant_scientist_breakthrough_reported_families value = brilliant_scientist_project_family } }
			NOT = { is_in_array = { array = brilliant_scientist_breakthrough_pending_families value = brilliant_scientist_project_family } }
			OR = {
				brilliant_scientist_is_current_host = yes
				brilliant_scientist_is_active_kruger_state = yes
			}
		}
		if = {
			limit = { has_country_flag = brilliant_scientist_breakthrough_report_active }
			add_to_array = { array = brilliant_scientist_breakthrough_pending_families value = brilliant_scientist_project_family }
			brilliant_scientist_mark_breakthrough_family_pending_on_kruger = yes
		}
		else = {
			add_to_array = { array = brilliant_scientist_breakthrough_reported_families value = brilliant_scientist_project_family }
			brilliant_scientist_mark_breakthrough_family_pending_on_kruger = yes
			set_variable = { brilliant_scientist_last_breakthrough_project_family = brilliant_scientist_project_family }
			set_variable = { brilliant_scientist_last_breakthrough_project_stage = brilliant_scientist_requested_project_stage }
			set_country_flag = brilliant_scientist_breakthrough_report_active
			country_event = { id = chaosx.nr16.6 days = constant:brilliant_scientist_directorate_timing.breakthrough_report_spacing_days }
		}
	}
}

# COUNTRY scope. Starts the next queued first-Prototype report after the
# current public/classified decision has committed. The queue is finite and
# family-based, so two native project completions in one synchronization pass
# cannot silently discard the later report.
brilliant_scientist_dispatch_next_breakthrough_report = {
	if = {
		limit = {
			NOT = { has_country_flag = brilliant_scientist_breakthrough_report_active }
			OR = {
				brilliant_scientist_is_current_host = yes
				brilliant_scientist_is_active_kruger_state = yes
			}
		}
		set_temp_variable = { brilliant_scientist_next_breakthrough_found = constant:brilliant_scientist_value.zero }
		set_temp_variable = { brilliant_scientist_next_breakthrough_index = constant:brilliant_scientist_value.zero }
		set_temp_variable = { brilliant_scientist_next_breakthrough_family = constant:brilliant_scientist_project_family.none }
		for_each_loop = {
			array = brilliant_scientist_breakthrough_pending_families
			value = brilliant_scientist_next_breakthrough_family_entry
			index = brilliant_scientist_next_breakthrough_index_entry
			if = {
				limit = { check_variable = { brilliant_scientist_next_breakthrough_found = constant:brilliant_scientist_value.zero } }
				set_temp_variable = { brilliant_scientist_next_breakthrough_family = brilliant_scientist_next_breakthrough_family_entry }
				set_temp_variable = { brilliant_scientist_next_breakthrough_index = brilliant_scientist_next_breakthrough_index_entry }
				set_temp_variable = { brilliant_scientist_next_breakthrough_found = constant:brilliant_scientist_value.one }
			}
		}
		if = {
			limit = { check_variable = { brilliant_scientist_next_breakthrough_found = constant:brilliant_scientist_value.one } }
			remove_from_array = { array = brilliant_scientist_breakthrough_pending_families index = brilliant_scientist_next_breakthrough_index }
			add_to_array = { array = brilliant_scientist_breakthrough_reported_families value = brilliant_scientist_next_breakthrough_family }
			set_temp_variable = { brilliant_scientist_project_family = brilliant_scientist_next_breakthrough_family }
			brilliant_scientist_mark_breakthrough_family_pending_on_kruger = yes
			set_variable = { brilliant_scientist_last_breakthrough_project_family = brilliant_scientist_next_breakthrough_family }
			set_variable = { brilliant_scientist_last_breakthrough_project_stage = constant:brilliant_scientist_project_stage.prototype }
			set_country_flag = brilliant_scientist_breakthrough_report_active
			country_event = { id = chaosx.nr16.6 days = constant:brilliant_scientist_directorate_timing.breakthrough_report_spacing_days }
		}
	}
}

# COUNTRY scope. Scales the current report delta by the number of already
# resolved family reports. The reservation and queue never call this helper.

	if = {
