cannibalism_start_restore_supply_corridor = {
	if = {
		limit = {
			cannibalism_containment_decisions_visible = yes
			cannibalism_has_valid_primary_theater = yes
			cannibalism_can_pay_logistics_cost = yes
			NOT = { has_country_flag = cannibalism_logistics_mission_active }
		}
		cannibalism_pay_logistics_cost = yes
		set_country_flag = cannibalism_logistics_mission_active
		set_variable = { cannibalism_logistics_hold_days = 0 }
		set_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_decision_timing.logistics_hold_base_days }
		var:cannibalism_primary_theater_state = {
			if = {
				limit = { is_island_state = yes }
				ROOT = { add_to_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_decision_timing.logistics_island_add_days } }
			}
			else_if = {
				limit = { is_coastal = yes }
				ROOT = { add_to_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_decision_timing.logistics_coastal_add_days } }
			}
			if = {
				limit = { infrastructure < constant:cannibalism_decision_timing.logistics_low_infrastructure_threshold }
				ROOT = { add_to_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_decision_timing.logistics_low_infrastructure_add_days } }
			}
		}
		if = {
			limit = { check_variable = { cannibalism_field_hunger > constant:cannibalism_meter.severe } }
			add_to_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_decision_timing.logistics_severe_hunger_add_days }
		}
		if = {
			limit = {
				has_variable = cannibalism_primary_theater_state
				var:cannibalism_primary_theater_state = { cannibalism_state_has_external_hunger_pressure = yes }
			}
			add_to_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_integration.logistics_shortage_add_days }
		}
		if = {
			limit = {
				has_variable = cannibalism_primary_theater_state
				var:cannibalism_primary_theater_state = { cannibalism_state_has_severe_external_hunger_pressure = yes }
			}
			add_to_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_integration.logistics_severe_shortage_add_days }
		}
		if = {
			limit = { has_country_flag = cannibalism_policy_open_emergency }
			subtract_from_variable = { cannibalism_logistics_required_hold_days = constant:cannibalism_decision_timing.logistics_open_response_reduction_days }
		}
		clamp_variable = { var = cannibalism_logistics_required_hold_days min = constant:cannibalism_decision_timing.logistics_minimum_days max = constant:cannibalism_decision_timing.logistics_maximum_days }
		set_variable = { cannibalism_logistics_mission_duration = cannibalism_logistics_required_hold_days }
		add_to_variable = { cannibalism_logistics_mission_duration = constant:cannibalism_decision_timing.logistics_timeout_buffer_days }
		set_variable = { cannibalism_logistics_division_floor = 0 }
		add_dynamic_modifier = { modifier = cannibalism_emergency_logistics_burden }
		activate_mission = cannibalism_restore_supply_corridor_mission
	}
}

cannibalism_start_rotate_compromised_formations = {
	if = {
		limit = {
			cannibalism_containment_decisions_visible = yes
			cannibalism_has_valid_primary_theater = yes
			cannibalism_can_pay_rotation_cost = yes
			NOT = { has_country_flag = cannibalism_rotation_mission_active }
		}
		cannibalism_pay_rotation_cost = yes
		set_country_flag = cannibalism_rotation_mission_active
		set_variable = { cannibalism_rotation_hold_days = 0 }
		set_variable = { cannibalism_rotation_required_hold_days = constant:cannibalism_decision_timing.rotation_hold_base_days }
		if = {
			limit = { has_war = yes }
			add_to_variable = { cannibalism_rotation_required_hold_days = constant:cannibalism_decision_timing.rotation_war_add_days }
		}
		if = {
			limit = { check_variable = { cannibalism_cell_strength > constant:cannibalism_meter.severe } }
			add_to_variable = { cannibalism_rotation_required_hold_days = constant:cannibalism_decision_timing.rotation_severe_cell_add_days }
		}
		if = {
			limit = { check_variable = { cannibalism_command_integrity > constant:cannibalism_meter.active } }
			subtract_from_variable = { cannibalism_rotation_required_hold_days = constant:cannibalism_decision_timing.rotation_high_integrity_reduction_days }
		}
		clamp_variable = { var = cannibalism_rotation_required_hold_days min = constant:cannibalism_decision_timing.rotation_minimum_days max = constant:cannibalism_decision_timing.rotation_maximum_days }
		set_variable = { cannibalism_rotation_mission_duration = cannibalism_rotation_required_hold_days }
		add_to_variable = { cannibalism_rotation_mission_duration = constant:cannibalism_decision_timing.rotation_timeout_buffer_days }
		set_variable = { cannibalism_rotation_division_floor = constant:cannibalism_decision_result.rotation_division_base }
		if = {
			limit = { num_divisions > constant:cannibalism_decision_result.rotation_medium_army_threshold }
			set_variable = { cannibalism_rotation_division_floor = constant:cannibalism_decision_result.rotation_division_medium }
		}
		if = {
			limit = { num_divisions > constant:cannibalism_decision_result.rotation_large_army_threshold }
			set_variable = { cannibalism_rotation_division_floor = constant:cannibalism_decision_result.rotation_division_large }
		}
		subtract_from_variable = { cannibalism_rotation_division_floor = constant:cannibalism_decision_result.objective_comparison_offset }
		add_dynamic_modifier = { modifier = cannibalism_formation_screening_burden }
		activate_mission = cannibalism_rotate_compromised_formations_mission
	}
}

cannibalism_process_current_country_mission_objectives = {
	if = {
		limit = { has_country_flag = cannibalism_logistics_mission_active }
		if = {
			limit = { cannibalism_logistics_objective_secured = yes }
			add_to_variable = { cannibalism_logistics_hold_days = global.cannibalism_scheduled_pulse_interval_days }
		}
		else = { set_variable = { cannibalism_logistics_hold_days = 0 } }
	}
	if = {
		limit = { has_country_flag = cannibalism_rotation_mission_active }
		if = {
			limit = { cannibalism_rotation_objective_secured = yes }
			add_to_variable = { cannibalism_rotation_hold_days = global.cannibalism_scheduled_pulse_interval_days }
		}
		else = { set_variable = { cannibalism_rotation_hold_days = 0 } }
	}
}

	add_to_variable = { cannibalism_command_integrity = constant:cannibalism_delta.logistics_success_integrity }

		limit = { has_variable = cannibalism_primary_theater_state }

			cannibalism_apply_shared_relief_to_current_state = yes
			if = {
				limit = { has_variable = cannibalism_node_strength }
				add_to_variable = { cannibalism_node_strength = constant:cannibalism_decision_result.logistics_node_suppression }
