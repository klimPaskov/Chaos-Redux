independence_wave_refresh_fer_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_fer_package = yes }
		remove_ideas = { fer_fragmented_railway_mandate fer_railway_compact }
		if = {
			limit = { has_stable_independence_wave_fer_compact = yes }
			add_ideas = fer_railway_compact
			set_country_flag = independence_wave_fer_compact_stabilized
		}
		else = {
			add_ideas = fer_fragmented_railway_mandate
			clr_country_flag = independence_wave_fer_compact_stabilized
		}
	}
}

# Inputs are temporary `independence_wave_fer_congress_delta` and
# `independence_wave_fer_railway_delta`. Values are clamped to the package
# pressure ladder so repeated focus/decision calls cannot overflow the ledger.
independence_wave_change_fer_compact_values = {
	if = {
		limit = { is_independence_wave_fer_package = yes }
		add_to_variable = { independence_wave_fer_congress_cohesion = independence_wave_fer_congress_delta }
		add_to_variable = { independence_wave_fer_rail_security = independence_wave_fer_railway_delta }
		clamp_variable = {
			var = independence_wave_fer_congress_cohesion
			min = constant:independence_wave_fer_pressure.minimum
			max = constant:independence_wave_fer_pressure.maximum
		}
		clamp_variable = {
			var = independence_wave_fer_rail_security
			min = constant:independence_wave_fer_pressure.minimum
			max = constant:independence_wave_fer_pressure.maximum
		}
		independence_wave_refresh_fer_compact_lifecycle = yes
		independence_wave_refresh_country_state = yes
	}
}
