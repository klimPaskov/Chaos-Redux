# ============================================================================
# CHAOS REDUX - EVENT 016 NATIVE RAID EFFECTS
# ============================================================================
# Persistent bookkeeping, portal landings, factory extraction, and target
# damage for the portal facility raid.
# Native raid reservation, cancellation, expiry, outcome selection, equipment
# collection, and raid history remain owned by the HOI4 raid engine. These
# effects resolve the Event 016 consequences after the native outcome has been
# selected: the breach can seize one province, create a Portal Raider cadre,
# and extract factories to the raider country as off-map industry.
# ============================================================================

@CR_SC_PORTAL_RAID_LIGHT_DAMAGE = 1
@CR_SC_PORTAL_RAID_HEAVY_DAMAGE = 2

# COUNTRY scope. Called once by each native portal-raid outcome.
brilliant_scientist_record_portal_raid_attempt = {
	set_country_flag = brilliant_scientist_portal_raid_attempted
	add_to_variable = { brilliant_scientist_portal_raid_attempts = constant:brilliant_scientist_value.one }
}

# COUNTRY scope. The flags are persistent history for later decisions,
# intelligence, and aftermath reactions; they do not grant a second payload.
brilliant_scientist_record_portal_raid_limited_success = {
	brilliant_scientist_record_portal_raid_attempt = yes
	set_country_flag = brilliant_scientist_portal_raid_limited_success_ever
	add_to_variable = { brilliant_scientist_portal_raid_successes = constant:brilliant_scientist_value.one }
}

brilliant_scientist_record_portal_raid_success = {
	brilliant_scientist_record_portal_raid_attempt = yes
	set_country_flag = brilliant_scientist_portal_raid_success_ever
	add_to_variable = { brilliant_scientist_portal_raid_successes = constant:brilliant_scientist_value.one }
}

brilliant_scientist_record_portal_raid_critical_success = {
	brilliant_scientist_record_portal_raid_success = yes
	set_country_flag = brilliant_scientist_portal_raid_critical_success_ever
}

# STATE scope. The successful portal operation is a real one-province landing,
# not a damage-only report. The actor becomes controller of the selected
# province while the target country retains the surrounding state, and the
# existing locked Quantum Transit Raiders template supplies the spawned cadre.
# This leaves a persistent state flag for later spread, containment, and
# aftermath systems without adding a whole-world polling on_action.
brilliant_scientist_portal_raid_establish_beachhead = {
	if = {
		limit = {
			var:ROOT.actor_country = { exists = yes }
			var:ROOT.target_province = { exists = yes }
		}
		set_state_flag = brilliant_scientist_portal_beachhead_active
		set_state_flag = brilliant_scientist_portal_raid_breach_recorded
		var:ROOT.actor_country = {
			set_province_controller = var:ROOT.target_province
			set_country_flag = brilliant_scientist_portal_beachhead_established_ever
			add_to_variable = { brilliant_scientist_portal_beachheads = constant:brilliant_scientist_value.one }
		}
		var:ROOT.target_province = {
			create_unit = {
				division = "name = \"Portal Breach Cadre\" division_template = \"Quantum Transit Raiders\" start_experience_factor = 0.50 start_equipment_factor = 0.40 start_manpower_factor = 0.75"
				owner = var:ROOT.actor_country
				allow_spawning_on_enemy_provs = yes
				count = 1
			}
		}
	}
}

# STATE scope. Extracts exactly one factory level from the selected target and
# returns it to the raiding country as an off-map factory. Off-map industry is
# intentional here: the factory is portable loot and cannot fail because the
# raider has no free construction slot at the moment of the raid. The priority
# keeps the outcome legible and lets a critical success call this effect twice.
brilliant_scientist_portal_raid_steal_factory = {
	if = {
		limit = { var:ROOT.actor_country = { exists = yes } }
		if = {
			limit = { arms_factory > 0 }
			remove_building = { type = arms_factory level = 1 }
			set_state_flag = brilliant_scientist_portal_factory_extracted
			var:ROOT.actor_country = {
				add_offsite_building = { type = arms_factory level = 1 }
				set_country_flag = brilliant_scientist_portal_factory_extraction_ever
				add_to_variable = { brilliant_scientist_portal_factories_stolen = constant:brilliant_scientist_value.one }
			}
		}
		else_if = {
			limit = { industrial_complex > 0 }
			remove_building = { type = industrial_complex level = 1 }
			set_state_flag = brilliant_scientist_portal_factory_extracted
			var:ROOT.actor_country = {
				add_offsite_building = { type = industrial_complex level = 1 }
				set_country_flag = brilliant_scientist_portal_factory_extraction_ever
				add_to_variable = { brilliant_scientist_portal_factories_stolen = constant:brilliant_scientist_value.one }
			}
		}
		else_if = {
			limit = { dockyard > 0 }
			remove_building = { type = dockyard level = 1 }
			set_state_flag = brilliant_scientist_portal_factory_extracted
			var:ROOT.actor_country = {
				add_offsite_building = { type = dockyard level = 1 }
				set_country_flag = brilliant_scientist_portal_factory_extraction_ever
				add_to_variable = { brilliant_scientist_portal_factories_stolen = constant:brilliant_scientist_value.one }
			}
		}
	}
}

# STATE scope. Damages the highest-value surviving strategic facility in the
# selected state. State history is retained even if the building is later
# repaired, so foreign reaction and intelligence decisions can distinguish a
# portal breach from an ordinary land battle.
brilliant_scientist_portal_raid_damage_target_light = {
	set_state_flag = brilliant_scientist_portal_raid_targeted
	if = {
		limit = { nuclear_reactor > 0 }
		damage_building = { type = nuclear_reactor damage = @CR_SC_PORTAL_RAID_LIGHT_DAMAGE }
	}
	else_if = {
		limit = { commercial_nuclear_reactor > 0 }
		damage_building = { type = commercial_nuclear_reactor damage = @CR_SC_PORTAL_RAID_LIGHT_DAMAGE }
	}
	else_if = {
		limit = { rocket_site > 0 }
		damage_building = { type = rocket_site damage = @CR_SC_PORTAL_RAID_LIGHT_DAMAGE }
	}
	else_if = {
		limit = { arms_factory > 0 }
		damage_building = { type = arms_factory damage = @CR_SC_PORTAL_RAID_LIGHT_DAMAGE }
	}
	else_if = {
		limit = { industrial_complex > 0 }
		damage_building = { type = industrial_complex damage = @CR_SC_PORTAL_RAID_LIGHT_DAMAGE }
	}
	else_if = {
		limit = { dockyard > 0 }
		damage_building = { type = dockyard damage = @CR_SC_PORTAL_RAID_LIGHT_DAMAGE }
	}
}
