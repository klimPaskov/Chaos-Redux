# ============================================================================
# CHAOS REDUX - EVENT 039 MURDER MYSTERY INTEGRATION EFFECTS
# ============================================================================
# Overview:
# - This file is the owner-facing bridge between the bounded Event 039 runtime
#   and visible events, decisions, countries, operations, evolutions, and the
#   terminal campaign.
# - It performs only validated transactions. The runtime registry remains the
#   authority for safe targets, cells, split rows, derivatives, and terminal
#   sectors; this file supplies the concrete country and event side effects.
# - The automatic host probe is a one-shot event-pool selection. No effect in
#   this file installs a recurring world scan.

# ---------------------------------------------------------------------------
# Event-pool host selection and intelligence scaling
# ---------------------------------------------------------------------------

murder_mystery_add_current_country_to_prefire_pool = {
	set_temp_variable = { individual_crisis_candidate_base_weight = constant:murder_mystery_value.one }
	adjust_individual_crisis_candidate_ticket_weight = yes
	set_temp_variable = { murder_mystery_prefire_pool_remaining = individual_crisis_candidate_adjusted_weight }
	while_loop_effect = {
		limit = { check_variable = { murder_mystery_prefire_pool_remaining > constant:murder_mystery_value.zero } }
		add_to_temp_array = { murder_mystery_prefire_host_pool = THIS }
		subtract_from_temp_variable = { murder_mystery_prefire_pool_remaining = constant:murder_mystery_value.one }
	}
}

# Input: temp murder_mystery_scenario_host_selection, one for a manual
# scenario launch (no individual-crisis load cap), zero for the event pool.
murder_mystery_select_weighted_prefire_host = {
	clear_temp_array = murder_mystery_prefire_host_pool
	every_country = {
		limit = {
			OR = {
				AND = {
					check_variable = { var = murder_mystery_scenario_host_selection value = constant:murder_mystery_value.zero compare = equals }
					murder_mystery_prefire_host_is_valid = yes
				}
				AND = {
					check_variable = { var = murder_mystery_scenario_host_selection value = constant:murder_mystery_value.one compare = equals }
					murder_mystery_scenario_host_is_valid = yes
				}
			}
		}
		murder_mystery_add_current_country_to_prefire_pool = yes
	}
	if = {
		limit = { check_variable = { murder_mystery_prefire_host_pool^num > constant:murder_mystery_value.zero } }
		random_scope_in_array = {
			array = murder_mystery_prefire_host_pool
			save_global_event_target_as = murder_mystery_prefire_host
		}
	}
	clear_temp_array = murder_mystery_prefire_host_pool
}

murder_mystery_prepare_random_event_fire = {
	set_temp_variable = { murder_mystery_prefire_ready = constant:murder_mystery_value.zero }
	if = {
		limit = { has_event_target = murder_mystery_prefire_host }
		clear_global_event_target = murder_mystery_prefire_host
	}
	set_temp_variable = { murder_mystery_scenario_host_selection = constant:murder_mystery_value.zero }
	murder_mystery_select_weighted_prefire_host = yes
	if = {
		limit = { murder_mystery_prefire_host_is_current = yes }
		set_temp_variable = { murder_mystery_prefire_ready = constant:murder_mystery_value.one }
	}
}

# Shared event-pool reason for Events Log rows. The caller resets
# event_pool_candidate_unavailability_reason to none before the call.
murder_mystery_evaluate_event_pool_unavailability = {
	if = {
		limit = {
			OR = {
				has_global_flag = murder_mystery_runtime_active
				has_global_flag = murder_mystery_scenario_launch_in_progress
			}
		}
		set_temp_variable = { event_pool_candidate_unavailability_reason = constant:event_system_event_unavailability_reason.murder_mystery_active }
	}
	else_if = {
		limit = { has_global_flag = murder_mystery_runtime_resolved }
		set_temp_variable = { event_pool_candidate_unavailability_reason = constant:event_system_event_unavailability_reason.murder_mystery_resolved }
	}
	else_if = {
		limit = { has_global_flag = world_end }
		set_temp_variable = { event_pool_candidate_unavailability_reason = constant:event_system_event_unavailability_reason.murder_mystery_world_end }
	}
	else_if = {
		limit = { NOT = { murder_mystery_has_any_eligible_host = yes } }
		set_temp_variable = { event_pool_candidate_unavailability_reason = constant:event_system_event_unavailability_reason.murder_mystery_host }
	}
}

# Manual Assassin Network launch. The opening runs at once on the chosen host,
# so a launch is never left waiting on a delayed event that could be
# rejected: it either commits and applies the intensity setup, or fails and
# leaves the runtime dormant with the launch-failed receipt.
murder_mystery_triggerable_scenario_launch = {
	if = {
		limit = { murder_mystery_triggerable_scenario_can_launch = yes }
		clr_global_flag = murder_mystery_scenario_launch_failed
		if = {
			limit = { has_event_target = murder_mystery_prefire_host }
			clear_global_event_target = murder_mystery_prefire_host
		}
		set_variable = { global.murder_mystery_scenario_launch_intensity = triggerable_scenarios_intensity }
		set_global_flag = murder_mystery_scenario_launch_in_progress
		set_temp_variable = { murder_mystery_scenario_host_selection = constant:murder_mystery_value.one }
		murder_mystery_select_weighted_prefire_host = yes
		if = {
			limit = { has_event_target = murder_mystery_prefire_host }
			event_target:murder_mystery_prefire_host = { murder_mystery_run_opening = yes }
		}
		clr_global_flag = murder_mystery_scenario_launch_in_progress
		clear_variable = global.murder_mystery_scenario_launch_intensity
		if = {
			limit = { NOT = { has_global_flag = murder_mystery_scenario_launched } }
			set_global_flag = murder_mystery_scenario_launch_failed
		}
	}
}

# COUNTRY scope: the chosen host. Shared by the event-pool opening
# (chaosx.nr39.1) and the manual scenario launch.
murder_mystery_run_opening = {
	if = {
		limit = {
			NOT = { has_global_flag = murder_mystery_scenario_launched }
			NOT = { has_global_flag = murder_mystery_scenario_launch_in_progress }
		}
		set_global_flag = murder_mystery_natural_run
	}
	murder_mystery_prepare_opening_package = yes
	murder_mystery_commit_opening_transaction = yes
	if = {
		limit = { NOT = { has_global_flag = murder_mystery_opening_transaction_committed } }
		murder_mystery_abort_opening_transaction = yes
	}
	if = {
		limit = {
			has_global_flag = murder_mystery_opening_transaction_committed
			has_global_flag = murder_mystery_scenario_launch_in_progress
		}
		set_global_flag = murder_mystery_scenario_launched
		murder_mystery_apply_scenario_setup = yes
	}
	clr_global_flag = murder_mystery_scenario_launch_in_progress
	clear_variable = global.murder_mystery_scenario_launch_intensity
}

# Scenario setup commits each missing stage up to the requested intensity in
# order. A stage disabled in Event Details is a ceiling: it and every later
# stage stay uncommitted, and each committed stage records one history row.
murder_mystery_commit_scenario_evolution_stage = {
	if = {
		limit = {
			has_global_flag = murder_mystery_scenario_launched
			has_variable = murder_mystery_scenario_stage_input
			check_variable = { var = murder_mystery_scenario_stage_input value = constant:murder_mystery_evolution.stage_i compare = greater_than_or_equals }
			check_variable = { var = murder_mystery_scenario_stage_input value = constant:murder_mystery_evolution.stage_v compare = less_than }
			check_variable = { var = murder_mystery_scenario_stage_input value = global.murder_mystery_evolution_stage compare = greater_than }
		}
		set_temp_variable = { murder_mystery_evolution_stage_check = global.murder_mystery_evolution_stage }
		add_to_temp_variable = { murder_mystery_evolution_stage_check = constant:murder_mystery_identity.loop_increment }
		while_loop_effect = {
			limit = {
				check_variable = { var = murder_mystery_evolution_stage_check value = murder_mystery_scenario_stage_input compare = less_than_or_equals }
				murder_mystery_evolution_stage_check_is_enabled = yes
			}
			set_variable = { global.murder_mystery_evolution_stage = murder_mystery_evolution_stage_check }
			if = {
				limit = { check_variable = { var = murder_mystery_evolution_stage_check value = constant:murder_mystery_evolution.stage_i compare = equals } }
				set_global_flag = murder_mystery_evolution_i_committed
				set_global_flag = murder_mystery_murder_cult_revealed
				set_variable = { global.murder_mystery_lifecycle = constant:murder_mystery_lifecycle.murder_cult }
			}
			else_if = {
				limit = { check_variable = { var = murder_mystery_evolution_stage_check value = constant:murder_mystery_evolution.stage_ii compare = equals } }
				set_global_flag = murder_mystery_evolution_ii_committed
				set_global_flag = murder_mystery_international_network_open
				set_variable = { global.murder_mystery_active_cell_cap = constant:murder_mystery_cell.mature_cap }
				set_variable = { global.murder_mystery_lifecycle = constant:murder_mystery_lifecycle.international_network }
			}
			else_if = {
				limit = { check_variable = { var = murder_mystery_evolution_stage_check value = constant:murder_mystery_evolution.stage_iii compare = equals } }
				set_global_flag = murder_mystery_evolution_iii_committed
				set_global_flag = murder_mystery_territorial_movement_open
				set_variable = { global.murder_mystery_active_cell_cap = constant:murder_mystery_cell.stage_iii_cap }
				set_variable = { global.murder_mystery_lifecycle = constant:murder_mystery_lifecycle.territorial_movement }
			}
			else_if = {
				limit = { check_variable = { var = murder_mystery_evolution_stage_check value = constant:murder_mystery_evolution.stage_iv compare = equals } }
				set_global_flag = murder_mystery_evolution_iv_committed
				set_global_flag = murder_mystery_veiled_compact_open
				set_variable = { global.murder_mystery_active_cell_cap = constant:murder_mystery_cell.stage_iv_cap }
				set_variable = { global.murder_mystery_lifecycle = constant:murder_mystery_lifecycle.brotherhood_system }
			}
			set_temp_variable = { murder_mystery_log_stage = murder_mystery_evolution_stage_check }
			murder_mystery_record_scenario_evolution_history = yes
			add_to_temp_variable = { murder_mystery_evolution_stage_check = constant:murder_mystery_identity.loop_increment }
		}
	}
}

# COUNTRY scope. Input: temp murder_mystery_log_stage.
murder_mystery_record_scenario_evolution_history = {
	if = {
		limit = {
			has_global_flag = murder_mystery_scenario_launched
			has_variable = global.murder_mystery_scenario_evolution_history_count
			check_variable = { var = global.murder_mystery_scenario_evolution_history_count value = constant:murder_mystery_registry.maximum_evolution_milestones compare = less_than }
		}
		add_to_variable = { global.murder_mystery_scenario_evolution_history_count = constant:murder_mystery_value.one }
		set_variable = { global.murder_mystery_scenario_last_evolution = murder_mystery_log_stage }
		murder_mystery_record_evolution_log_entry = yes
	}
}

# COUNTRY actor scope. Input: temp murder_mystery_log_stage (stage_i to stage_v).
# Records one Events Log evolution row under Event 39's own evolution type,
# with the stage's Chaos tier, so Event Details and the history views resolve
# the Murder Mystery titles and descriptions.
murder_mystery_record_evolution_log_entry = {
	set_temp_variable = { events_log_evolution_event_id = constant:murder_mystery_identity.event_id }
	set_temp_variable = { events_log_evolution_type = constant:murder_mystery_evolution_log.evolution_type }
	set_temp_variable = { events_log_evolution_stage = murder_mystery_log_stage }
	set_temp_variable = { events_log_evolution_tier = constant:murder_mystery_evolution_log.tier_i }
	if = { limit = { check_variable = { var = murder_mystery_log_stage value = constant:murder_mystery_evolution_log.stage_ii compare = equals } } set_temp_variable = { events_log_evolution_tier = constant:murder_mystery_evolution_log.tier_ii } }
	if = { limit = { check_variable = { var = murder_mystery_log_stage value = constant:murder_mystery_evolution_log.stage_iii compare = equals } } set_temp_variable = { events_log_evolution_tier = constant:murder_mystery_evolution_log.tier_iii } }
	if = { limit = { check_variable = { var = murder_mystery_log_stage value = constant:murder_mystery_evolution_log.stage_iv compare = equals } } set_temp_variable = { events_log_evolution_tier = constant:murder_mystery_evolution_log.tier_iv } }
	if = { limit = { check_variable = { var = murder_mystery_log_stage value = constant:murder_mystery_evolution_log.stage_v compare = equals } } set_temp_variable = { events_log_evolution_tier = constant:murder_mystery_evolution_log.tier_v } }
	set_temp_variable = { events_log_evolution_has_actor = constant:murder_mystery_value.one }
	set_temp_variable = { events_log_evolution_actor = THIS.id }
	record_events_log_evolution_entry = yes
}

murder_mystery_apply_scenario_setup = {
	if = {
		limit = {
			has_global_flag = murder_mystery_scenario_launched
			has_variable = global.murder_mystery_scenario_launch_intensity
			has_country_flag = murder_mystery_investigation_open
		}
		set_variable = { murder_mystery_case_progress = constant:murder_mystery_scenario.low_case_progress }
		set_variable = { murder_mystery_network_reach = constant:murder_mystery_scenario.low_network_reach }
		set_variable = { murder_mystery_brotherhood_cohesion = constant:murder_mystery_scenario.low_cohesion }
		set_variable = { global.murder_mystery_scenario_cell_target = constant:murder_mystery_scenario.initial_cell_count_low }
		if = {
			limit = { check_variable = { var = global.murder_mystery_scenario_launch_intensity value = constant:triggerable_scenario_intensity.medium compare = greater_than_or_equals } }
			set_variable = { murder_mystery_case_progress = constant:murder_mystery_scenario.medium_case_progress }
			set_variable = { murder_mystery_network_reach = constant:murder_mystery_scenario.medium_network_reach }
			set_variable = { murder_mystery_brotherhood_cohesion = constant:murder_mystery_scenario.medium_cohesion }
			set_variable = { global.murder_mystery_scenario_cell_target = constant:murder_mystery_scenario.initial_cell_count_medium }
			set_variable = { murder_mystery_scenario_stage_input = constant:murder_mystery_evolution.stage_ii }
			murder_mystery_commit_scenario_evolution_stage = yes
		}
		else = {
			set_variable = { murder_mystery_scenario_stage_input = constant:murder_mystery_evolution.stage_i }
			murder_mystery_commit_scenario_evolution_stage = yes
		}
		if = {
			limit = { check_variable = { var = global.murder_mystery_scenario_launch_intensity value = constant:triggerable_scenario_intensity.high compare = greater_than_or_equals } }
			set_variable = { murder_mystery_case_progress = constant:murder_mystery_scenario.high_case_progress }
			set_variable = { murder_mystery_network_reach = constant:murder_mystery_scenario.high_network_reach }
			set_variable = { murder_mystery_brotherhood_cohesion = constant:murder_mystery_scenario.high_cohesion }
			set_variable = { global.murder_mystery_scenario_cell_target = constant:murder_mystery_scenario.initial_cell_count_high }
		}
		if = {
			limit = { check_variable = { var = global.murder_mystery_scenario_launch_intensity value = constant:triggerable_scenario_intensity.maximum compare = equals } }
			set_variable = { murder_mystery_case_progress = constant:murder_mystery_scenario.maximum_case_progress }
			set_variable = { murder_mystery_network_reach = constant:murder_mystery_scenario.maximum_network_reach }
			set_variable = { murder_mystery_brotherhood_cohesion = constant:murder_mystery_scenario.maximum_cohesion }
			set_variable = { global.murder_mystery_scenario_cell_target = constant:murder_mystery_scenario.initial_cell_count_maximum }
		}
		murder_mystery_clamp_country_values = yes
		# Each seeding attempt picks one bounded route; the attempt budget stops
		# the loop when the map offers too few candidates.
		set_variable = { global.murder_mystery_scenario_cells_seeded = constant:murder_mystery_value.zero }
		set_temp_variable = { murder_mystery_scenario_attempts_left = global.murder_mystery_scenario_cell_target }
		multiply_temp_variable = { murder_mystery_scenario_attempts_left = constant:murder_mystery_seeding.maximum_attempts }
		while_loop_effect = {
			limit = {
				check_variable = { var = global.murder_mystery_scenario_cells_seeded value = global.murder_mystery_scenario_cell_target compare = less_than }
				check_variable = { var = murder_mystery_scenario_attempts_left value = constant:murder_mystery_value.zero compare = greater_than }
				murder_mystery_cell_registry_has_capacity = yes
			}
			subtract_from_temp_variable = { murder_mystery_scenario_attempts_left = constant:murder_mystery_value.one }
			clr_global_flag = murder_mystery_cell_registration_succeeded
			murder_mystery_select_foreign_cell_candidate = yes
			if = {
				limit = { has_event_target = murder_mystery_selected_cell }
				murder_mystery_seed_bounded_cell = yes
				if = {
					limit = { has_global_flag = murder_mystery_cell_registration_succeeded }
					add_to_variable = { global.murder_mystery_scenario_cells_seeded = constant:murder_mystery_value.one }
				}
				murder_mystery_clear_selected_cell_context = yes
			}
		}
		# High intensity opens the territorial split. Evolution III and IV are
		# recorded only after the Assassin State commits.
		if = {
			limit = {
				check_variable = { var = global.murder_mystery_scenario_launch_intensity value = constant:triggerable_scenario_intensity.high compare = greater_than_or_equals }
				check_variable = { var = global.murder_mystery_evolution_stage value = constant:murder_mystery_evolution.stage_ii compare = equals }
				NOT = { has_global_flag = events_log_disabled_evolution_39_39_3 }
			}
			set_variable = { murder_mystery_requested_split_share = constant:murder_mystery_scenario.scenario_high_split_share }
			murder_mystery_select_assassin_state_cluster = yes
			murder_mystery_create_assassin_state_candidate = yes
			murder_mystery_commit_assassin_state_creation = yes
			if = {
				limit = { has_global_flag = murder_mystery_assassin_state_commit_succeeded }
				set_variable = { murder_mystery_scenario_stage_input = constant:murder_mystery_evolution.stage_iii }
				murder_mystery_commit_scenario_evolution_stage = yes
				if = {
					limit = { check_variable = { var = global.murder_mystery_scenario_launch_intensity value = constant:triggerable_scenario_intensity.maximum compare = equals } }
					set_variable = { murder_mystery_scenario_stage_input = constant:murder_mystery_evolution.stage_iv }
					murder_mystery_commit_scenario_evolution_stage = yes
				}
			}
		}
		if = {
			limit = {
				check_variable = { var = global.murder_mystery_scenario_launch_intensity value = constant:triggerable_scenario_intensity.maximum compare = equals }
				has_global_flag = murder_mystery_evolution_iv_committed
				has_global_flag = murder_mystery_assassin_state_exists
			}
			set_variable = { global.murder_mystery_scenario_derivatives_target = constant:murder_mystery_scenario.scenario_maximum_derivatives }
			set_variable = { global.murder_mystery_scenario_derivatives_created = constant:murder_mystery_value.zero }
			set_temp_variable = { murder_mystery_scenario_revolt_rows = constant:murder_mystery_scenario.scenario_maximum_derivatives }
			murder_mystery_prime_scenario_revolt_cells = yes
			set_temp_variable = { murder_mystery_scenario_attempts_left = global.murder_mystery_scenario_derivatives_target }
			multiply_temp_variable = { murder_mystery_scenario_attempts_left = constant:murder_mystery_seeding.maximum_attempts }
			while_loop_effect = {
				limit = {
					check_variable = { var = global.murder_mystery_scenario_derivatives_created value = global.murder_mystery_scenario_derivatives_target compare = less_than }
					check_variable = { var = murder_mystery_scenario_attempts_left value = constant:murder_mystery_value.zero compare = greater_than }
				}
				subtract_from_temp_variable = { murder_mystery_scenario_attempts_left = constant:murder_mystery_value.one }
				clr_global_flag = murder_mystery_derivative_commit_succeeded
				set_temp_variable = { murder_mystery_revolt_minimum_readiness = constant:murder_mystery_value.zero }
				murder_mystery_select_revolt_cell = yes
				if = {
					limit = { has_event_target = murder_mystery_selected_cell }
					murder_mystery_start_derivative_creation = yes
					if = {
						limit = { has_global_flag = murder_mystery_derivative_commit_succeeded }
						add_to_variable = { global.murder_mystery_scenario_derivatives_created = constant:murder_mystery_value.one }
					}
					murder_mystery_clear_selected_cell_context = yes
				}
			}
		}
		clear_variable = murder_mystery_scenario_stage_input
		clear_variable = murder_mystery_requested_split_share
	}
}

murder_mystery_calculate_intelligence_scaling = {
	set_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.base }
	set_temp_variable = { murder_mystery_stability_score = stability }
	multiply_temp_variable = { murder_mystery_stability_score = constant:murder_mystery_intelligence.stability_factor }
	add_to_variable = { murder_mystery_intelligence_score = murder_mystery_stability_score }
	if = {
		limit = { has_country_flag = political_control_high }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.political_control_factor }
	}
	if = {
		limit = { has_country_flag = political_control_medium }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_value.ten }
	}
	if = {
		limit = { has_country_flag = murder_mystery_investigation_command_ready }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.command_factor }
	}
	if = {
		limit = { has_tech = basic_encryption }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.encryption_factor }
	}
	if = {
		limit = { has_tech = radio_detection }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.radar_factor }
	}
	if = {
		limit = { has_tech = radio }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.radio_factor }
	}
	if = {
		limit = { has_tech = tech_military_police }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.military_intel_factor }
	}
	if = {
		limit = { has_intelligence_agency = yes }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.agency_factor }
	}
	if = {
		limit = { num_of_factories > constant:murder_mystery_value.ten }
		add_to_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.industry_factor }
	}
	if = {
		limit = { has_war = yes }
		subtract_from_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.war_penalty }
	}
	if = {
		limit = { has_country_flag = occupation_active }
		subtract_from_variable = { murder_mystery_intelligence_score = constant:murder_mystery_intelligence.occupation_penalty }
	}
	clamp_variable = {
		var = murder_mystery_intelligence_score
		min = constant:murder_mystery_intelligence.minimum
		max = constant:murder_mystery_intelligence.maximum
	}
}

murder_mystery_apply_mapped_chaos = {
	if = {
		limit = {
			has_variable = murder_mystery_mapped_chaos_delta

murder_mystery_select_assassin_state_cluster = {
	clr_global_flag = murder_mystery_split_connectivity_proven
	if = {
		limit = { has_event_target = murder_mystery_original_host }
		event_target:murder_mystery_original_host = {
			every_owned_state = {
				limit = { has_state_flag = murder_mystery_split_cluster_member }
				clr_state_flag = murder_mystery_split_cluster_member
			}
		}
	}
	clear_array = global.murder_mystery_split_state_entries
	set_variable = { global.murder_mystery_split_state_count = constant:murder_mystery_value.zero }
	set_temp_variable = { murder_mystery_split_state_limit = constant:murder_mystery_value.zero }
	if = {
		limit = { has_event_target = murder_mystery_original_host }
		event_target:murder_mystery_original_host = {
			set_temp_variable = { murder_mystery_split_state_limit = num_of_controlled_states }
		}
	}
	if = {
		limit = { NOT = { has_variable = murder_mystery_requested_split_share } }
		set_variable = { murder_mystery_requested_split_share = constant:murder_mystery_split.minimum_share }
	}
	multiply_temp_variable = { murder_mystery_split_state_limit = murder_mystery_requested_split_share }
	divide_temp_variable = { murder_mystery_split_state_limit = constant:murder_mystery_value.hundred }
	clamp_temp_variable = { var = murder_mystery_split_state_limit min = constant:murder_mystery_split.minimum_cluster_states max = constant:murder_mystery_split.maximum_share }
	event_target:murder_mystery_original_host = {
		capital_scope = {
			random_neighbor_state = {
				limit = {
					impassable = no
					is_owned_and_controlled_by = event_target:murder_mystery_original_host
					is_capital = no
				}
				save_global_event_target_as = murder_mystery_split_anchor_state
				set_global_flag = murder_mystery_split_connectivity_proven
			}
		}
	}
	if = {
		limit = { has_event_target = murder_mystery_split_anchor_state }
		event_target:murder_mystery_split_anchor_state = {
			set_state_flag = murder_mystery_split_cluster_member
			save_event_target_as = murder_mystery_current_split_candidate
			murder_mystery_register_split_state = yes
		}
	}
	set_variable = { global.murder_mystery_split_expansion_round = constant:murder_mystery_value.zero }
	while_loop_effect = {
		limit = {
			check_variable = { var = global.murder_mystery_split_state_entries^num value = murder_mystery_split_state_limit compare = less_than }
			check_variable = { var = global.murder_mystery_split_expansion_round value = constant:murder_mystery_registry.maximum_split_state_rows compare = less_than }
		}
		add_to_variable = { global.murder_mystery_split_expansion_round = constant:murder_mystery_identity.loop_increment }
		event_target:murder_mystery_original_host = {
			every_owned_state = {
				limit = {
					has_state_flag = murder_mystery_split_cluster_member
					is_owned_and_controlled_by = event_target:murder_mystery_original_host
				}
				every_neighbor_state = {
					limit = {
						impassable = no
						is_owned_and_controlled_by = event_target:murder_mystery_original_host
						is_capital = no
						NOT = { has_state_flag = murder_mystery_split_cluster_member }
						check_variable = { var = global.murder_mystery_split_state_entries^num value = murder_mystery_split_state_limit compare = less_than }
					}
					set_state_flag = murder_mystery_split_cluster_member
					save_event_target_as = murder_mystery_current_split_candidate
					murder_mystery_register_split_state = yes
				}
			}
		}
	}
	if = {
		limit = { has_event_target = murder_mystery_original_host }
		event_target:murder_mystery_original_host = {
			set_variable = { murder_mystery_split_total_state_count = num_of_controlled_states }
			set_variable = { murder_mystery_host_remnant_state_count = murder_mystery_split_total_state_count }
			subtract_from_variable = { murder_mystery_host_remnant_state_count = global.murder_mystery_split_state_entries^num }
			set_variable = { murder_mystery_actual_split_share = global.murder_mystery_split_state_entries^num }
			multiply_variable = { murder_mystery_actual_split_share = constant:murder_mystery_value.hundred }
			divide_variable = { murder_mystery_actual_split_share = murder_mystery_split_total_state_count }
			murder_mystery_validate_split_state_registry = yes
		}
	}
}

	if = {

# Any scope. Input: event target murder_mystery_selected_cell (validated).
# One transaction turns the cell's revolt into a derivative country: a
# connected cluster (anchor plus bounded neighbours, never the capital) moves
# from the cell's government to a new dynamic country, which becomes a subject
# of the central state, receives its starting forces and goes to war with the
# government it rose against. Any failed postcondition returns every moved
# state to that government and leaves no landless derivative behind.
murder_mystery_start_derivative_creation = {
	clr_global_flag = murder_mystery_derivative_commit_succeeded
	if = {
		limit = {
			has_event_target = murder_mystery_assassin_state
			NOT = { has_global_flag = murder_mystery_derivative_transaction_active }
			murder_mystery_derivative_registry_has_capacity = yes
			murder_mystery_assassin_state_has_derivative_anchor = yes
			murder_mystery_selected_cell_is_valid = yes
			event_target:murder_mystery_selected_cell = { murder_mystery_country_can_lose_revolt_cluster = yes }
		}
		if = { limit = { has_event_target = murder_mystery_derivative_anchor_state } clear_global_event_target = murder_mystery_derivative_anchor_state }
		if = { limit = { has_event_target = murder_mystery_derivative_country_candidate } clear_global_event_target = murder_mystery_derivative_country_candidate }
		clear_array = global.murder_mystery_derivative_transfer_states
		event_target:murder_mystery_selected_cell = {
			random_owned_controlled_state = {
				limit = {
					impassable = no
					is_capital = no
				}
				save_global_event_target_as = murder_mystery_derivative_anchor_state
			}
		}
		if = {
			limit = { has_event_target = murder_mystery_derivative_anchor_state }
			create_dynamic_country = {
				original_tag = AXT
				save_global_event_target_as = murder_mystery_derivative_country_candidate
				set_country_flag = murder_mystery_derivative_carrier_validated
				set_country_flag = murder_mystery_derivative_country
				set_country_flag = murder_mystery_cell_administration_open
				set_country_flag = murder_mystery_nonterminal_play_enabled
				murder_mystery_initialize_country_runtime = yes
				set_variable = { murder_mystery_case_progress = constant:murder_mystery_value.twenty }
				set_variable = { murder_mystery_network_reach = constant:murder_mystery_value.forty }
				set_variable = { murder_mystery_brotherhood_cohesion = constant:murder_mystery_value.fifty }
				add_ideas = murder_mystery_foreign_derivative_administration
				add_manpower = constant:murder_mystery_assassin.derivative_manpower
				add_equipment_to_stockpile = { type = assassin_operations_kit_1 amount = constant:murder_mystery_assassin.derivative_operations_kits }
			}
			# The revolt follows the central state's route.
			if = {
				limit = { event_target:murder_mystery_assassin_state = { has_country_flag = murder_mystery_hidden_hand_route } }
				event_target:murder_mystery_derivative_country_candidate = {
					set_country_flag = murder_mystery_hidden_hand_route
					create_country_leader = {
						name = MURDER_MYSTERY_HIDDEN_HAND_ROUTE_LEADER_NAME
						desc = MURDER_MYSTERY_HIDDEN_HAND_ROUTE_LEADER_DESC
						picture = GFX_portrait_039_hidden_hand
						ideology = despotism
					}
				}
			}
			else_if = {
				limit = { event_target:murder_mystery_assassin_state = { has_country_flag = murder_mystery_necessary_mask_route } }
				event_target:murder_mystery_derivative_country_candidate = {
					set_country_flag = murder_mystery_necessary_mask_route
					create_country_leader = {
						name = MURDER_MYSTERY_NECESSARY_MASK_ROUTE_LEADER_NAME
						desc = MURDER_MYSTERY_NECESSARY_MASK_ROUTE_LEADER_DESC
						picture = GFX_portrait_039_necessary_mask
						ideology = despotism
					}
				}
			}
			else = {
				event_target:murder_mystery_derivative_country_candidate = {
					set_country_flag = murder_mystery_cells_without_masters_route
					set_country_flag = murder_mystery_derivative_autonomous
					create_country_leader = {
						name = MURDER_MYSTERY_CELLS_WITHOUT_MASTERS_NAME
						desc = MURDER_MYSTERY_CELLS_WITHOUT_MASTERS_DESC
						picture = GFX_portrait_039_cells_without_masters
						ideology = despotism
					}
				}
			}
			event_target:murder_mystery_derivative_anchor_state = {
				add_to_array = { array = global.murder_mystery_derivative_transfer_states value = THIS }
				transfer_state_to = event_target:murder_mystery_derivative_country_candidate
			}
			set_temp_variable = { murder_mystery_derivative_extra_states = constant:murder_mystery_derivative.cluster_extra_states }
			event_target:murder_mystery_derivative_anchor_state = {
				every_neighbor_state = {
					limit = {
						impassable = no
						is_capital = no
						is_owned_and_controlled_by = event_target:murder_mystery_selected_cell
						check_variable = { var = murder_mystery_derivative_extra_states value = constant:murder_mystery_value.zero compare = greater_than }
					}
					add_to_array = { array = global.murder_mystery_derivative_transfer_states value = THIS }
					transfer_state_to = event_target:murder_mystery_derivative_country_candidate
					subtract_from_temp_variable = { murder_mystery_derivative_extra_states = constant:murder_mystery_value.one }
				}
			}
			event_target:murder_mystery_derivative_country_candidate = {
				set_capital = { state = event_target:murder_mystery_derivative_anchor_state remember_old_capital = no }
				every_owned_state = { add_core_of = event_target:murder_mystery_derivative_country_candidate }
				if = {
					limit = { murder_mystery_host_has_live_state = yes }
					set_country_flag = murder_mystery_derivative_state_postcondition_passed
				}
			}
			if = {
				limit = {
					event_target:murder_mystery_derivative_country_candidate = { has_country_flag = murder_mystery_derivative_state_postcondition_passed }
					event_target:murder_mystery_selected_cell = { murder_mystery_host_has_live_state = yes }
				}
				event_target:murder_mystery_assassin_state = {
					puppet = {
						target = event_target:murder_mystery_derivative_country_candidate
						end_wars = yes
						end_civil_wars = yes
					}
				}
				event_target:murder_mystery_derivative_country_candidate = {
					if = {
						limit = { is_subject_of = event_target:murder_mystery_assassin_state }
						set_country_flag = murder_mystery_derivative_subject_bound
					}
				}
				if = {
					limit = { event_target:murder_mystery_derivative_country_candidate = { has_country_flag = murder_mystery_derivative_subject_bound } }
					set_global_flag = murder_mystery_derivative_subject_postcondition_passed
				}
				set_variable = { murder_mystery_derivative_stage_input = constant:murder_mystery_derivative.minimum_stage }
				set_global_flag = murder_mystery_derivative_postcondition_passed
				murder_mystery_begin_derivative_transaction = yes
				if = {
					limit = { has_global_flag = murder_mystery_derivative_transaction_active }
					murder_mystery_commit_derivative_transaction = yes
				}
				clear_variable = murder_mystery_derivative_stage_input
			}
			if = {
				limit = { has_global_flag = murder_mystery_derivative_commit_succeeded }
				event_target:murder_mystery_selected_cell = {
					set_country_flag = murder_mystery_derivative_source_consumed
					event_target:murder_mystery_derivative_country_candidate = { set_variable = { murder_mystery_derivative_parent_government = PREV } }
					set_temp_variable = { murder_mystery_cell_retire_reason = constant:murder_mystery_defeat.retire_reason_derivative }
					murder_mystery_retire_cell_row = yes
				}
				event_target:murder_mystery_derivative_country_candidate = {
					murder_mystery_setup_derivative_forces = yes
					declare_war_on = {
						target = event_target:murder_mystery_selected_cell
						type = annex_everything
					}
					news_event = { id = chaosx.news.4404 }
				}
				set_temp_variable = { murder_mystery_mapped_chaos_delta = constant:murder_mystery_chaos.derivative_revolt }
				set_temp_variable = { murder_mystery_mapped_chaos_reason = constant:chaos_meter_history_reason.special.murder_mystery_network }
				murder_mystery_apply_mapped_chaos = yes
				clear_array = global.murder_mystery_derivative_transfer_states
				clear_global_event_target = murder_mystery_derivative_country_candidate
			}
			else = {
				murder_mystery_revert_derivative_creation = yes
			}
		}
	}
}

						NOT = { has_country_flag = murder_mystery_derivative_country }
						NOT = { has_country_flag = murder_mystery_terminal_exempt }

					}

					set_country_flag = murder_mystery_terminal_target_not_derivative_proof
					set_country_flag = murder_mystery_terminal_target_not_protected_proof

					clr_global_flag = murder_mystery_terminal_target_registration_succeeded
					set_variable = { murder_mystery_terminal_registration_sector = global.murder_mystery_terminal_sector_cursor }

					if = {
						limit = { has_country_flag = murder_mystery_terminal_target_safe_callback }
						murder_mystery_register_terminal_target = yes
					}

						limit = { has_global_flag = murder_mystery_terminal_target_registration_succeeded }
						add_to_variable = { global.murder_mystery_terminal_sector_cursor = constant:murder_mystery_value.one }

							limit = { check_variable = { var = global.murder_mystery_terminal_sector_cursor value = constant:murder_mystery_terminal.maximum_sector compare = greater_than } }
