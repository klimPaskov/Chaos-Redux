random_faction_clamp_current_country_values = {
	if = { limit = { NOT = { has_variable = random_faction_pressure_score } } set_variable = { random_faction_pressure_score = 0 } }
	if = { limit = { NOT = { has_variable = random_faction_neutrality_resilience } } set_variable = { random_faction_neutrality_resilience = constant:random_faction_pressure_value.neutrality_resilience } }
	clamp_variable = { var = random_faction_pressure_score min = constant:random_faction_pressure_value.resilience_floor max = constant:random_faction_pressure_value.resilience_ceiling }
	clamp_variable = { var = random_faction_neutrality_resilience min = constant:random_faction_pressure_value.resilience_floor max = constant:random_faction_pressure_value.resilience_ceiling }
}

random_faction_prepare_candidate_weight = {
	set_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.base }
	if = { limit = { NOT = { is_major = yes } } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.minor_bonus } }
	if = { limit = { any_neighbor_country = { is_in_faction = yes } } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.bordering_faction_bonus } }
	if = {
		limit = {
			OR = {
				has_war = yes
				any_neighbor_country = { has_war = yes }
			}
		}
		add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.war_adjacent_bonus }
	}
	if = { limit = { has_stability < constant:random_faction_resistance_threshold.low_stability } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.low_stability_bonus } }
	if = { limit = { has_war_support < constant:random_faction_resistance_threshold.low_war_support } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.low_war_support_bonus } }
	if = { limit = { has_country_flag = random_faction_under_pressure } subtract_from_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.pressure_repeat_penalty } }
	if = { limit = { has_country_flag = random_faction_selection_cooldown } subtract_from_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.cooldown_penalty } }
	if = {
		limit = {
			has_variable = random_faction_neutrality_resilience
			check_variable = { var = random_faction_neutrality_resilience value = constant:random_faction_ai_threshold.low_neutrality_resilience compare = less_than }
		}
		add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.low_resilience_bonus }
	}
	if = {
		limit = {
			has_variable = random_faction_neutrality_resilience
			check_variable = { var = random_faction_neutrality_resilience value = constant:random_faction_ai_threshold.high_neutrality_resilience compare = greater_than }
		}
		subtract_from_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.high_resilience_penalty }
	}
	round_temp_variable = random_faction_candidate_weight
	clamp_temp_variable = {
		var = random_faction_candidate_weight
		min = constant:random_faction_selection_weight.min
		max = constant:random_faction_selection_weight.max
	}
}

random_faction_add_current_country_to_target_pool = {
	random_faction_prepare_candidate_weight = yes
	set_temp_variable = { random_faction_candidate_weight_remaining = random_faction_candidate_weight }
	while_loop_effect = {
		limit = { check_variable = { random_faction_candidate_weight_remaining > 0 } }
		add_to_temp_array = { random_faction_target_pool = THIS }
		ROOT = { add_to_temp_variable = { random_faction_target_pool_count = 1 } }
		subtract_from_temp_variable = { random_faction_candidate_weight_remaining = 1 }
	}
	set_temp_variable = { random_faction_candidate_weight = 0 }
	set_temp_variable = { random_faction_candidate_weight_remaining = 0 }
}

random_faction_prepare_runtime_context = {
	random_faction_refresh_runtime_arrays = yes
	set_temp_variable = { random_faction_context_ready = 0 }
	clear_temp_array = random_faction_target_pool
	set_temp_variable = { random_faction_target_pool_count = 0 }

	if = {
		limit = { check_variable = { random_faction_context_ready < 1 } }
		every_country = {
			limit = {
				OR = {
					ROOT = { NOT = { has_country_flag = random_faction_exclude_dispatch_requester } }
					NOT = { tag = ROOT }
				}
				OR = {
					is_random_faction_eligible_country = yes
					AND = {
						has_global_flag = random_faction_evolution_ii_unlocked
						is_random_faction_wartime_candidate = yes
					}
				}
			}
			random_faction_add_current_country_to_target_pool = yes
		}
	}
	if = {
		limit = {
			check_variable = { random_faction_context_ready < 1 }
			check_variable = { random_faction_target_pool_count > 0 }
		}
		random_scope_in_array = {
			array = random_faction_target_pool
			limit = {
				OR = {
					ROOT = { NOT = { has_country_flag = random_faction_exclude_dispatch_requester } }
					NOT = { tag = ROOT }
				}
				OR = {
					is_random_faction_eligible_country = yes
					AND = {
						has_global_flag = random_faction_evolution_ii_unlocked
						is_random_faction_wartime_candidate = yes
					}
				}
			}
			save_event_target_as = random_faction_target_country
		}
	}
	clear_temp_array = random_faction_target_pool
	if = {
		limit = {
			check_variable = { random_faction_context_ready < 1 }
			has_event_target = random_faction_target_country
			event_target:random_faction_target_country = {
				OR = {
					ROOT = { NOT = { has_country_flag = random_faction_exclude_dispatch_requester } }
					NOT = { tag = ROOT }
				}
				OR = {
					is_random_faction_eligible_country = yes
					AND = {
						has_global_flag = random_faction_evolution_ii_unlocked
						is_random_faction_wartime_candidate = yes
					}
				}
			}
		}
		event_target:random_faction_target_country = {
			random_faction_collect_faction_options = yes
			if = {
				limit = { check_variable = { random_faction_option_count > 0 } }
				ROOT = { set_temp_variable = { random_faction_context_ready = 1 } }
			}
		}
	}
}

random_faction_add_current_leader_to_option_pool = {
	set_temp_variable = { random_faction_leader_weight = constant:random_faction_ai_weight.medium }
	if = { limit = { any_country = { is_in_faction_with = PREV is_neighbor_of = PREV.PREV } } add_to_temp_variable = { random_faction_leader_weight = constant:random_faction_ai_weight.high } }
	if = { limit = { has_war = yes } add_to_temp_variable = { random_faction_leader_weight = constant:random_faction_ai_weight.low } }
	clamp_temp_variable = { var = random_faction_leader_weight min = constant:random_faction_ai_weight.low max = constant:random_faction_ai_weight.urgent }
	while_loop_effect = {
		limit = { check_variable = { random_faction_leader_weight > 0 } }
		add_to_temp_array = { random_faction_option_pool = THIS }
		ROOT = { add_to_temp_variable = { random_faction_option_pool_count = 1 } }
		subtract_from_temp_variable = { random_faction_leader_weight = 1 }
	}
}

random_faction_collect_faction_options = {
	set_variable = { random_faction_option_count = 0 }
	set_temp_variable = { random_faction_option_pool_count = 0 }
	clear_temp_array = random_faction_option_pool
	clear_temp_array = random_faction_selected_options
	if = {
		limit = {
			OR = {
				is_random_faction_eligible_country = yes
				AND = {
					has_global_flag = random_faction_evolution_ii_unlocked
					is_random_faction_wartime_candidate = yes
				}
			}
		}
		every_country = {
			limit = { random_faction_is_valid_faction_leader_for_prev_target = yes }
			random_faction_add_current_leader_to_option_pool = yes
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_pool_count > 0 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = { random_faction_is_valid_faction_leader_for_prev_target = yes }
			save_event_target_as = random_faction_option_1_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			PREV = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 0 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = {
				random_faction_is_valid_faction_leader_for_prev_target = yes
				NOT = { is_in_array = { array = random_faction_selected_options value = THIS } }
			}
			save_event_target_as = random_faction_option_2_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			PREV = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 1 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = {
				random_faction_is_valid_faction_leader_for_prev_target = yes
				NOT = { is_in_array = { array = random_faction_selected_options value = THIS } }
			}
			save_event_target_as = random_faction_option_3_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			PREV = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 2 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = {
				random_faction_is_valid_faction_leader_for_prev_target = yes
				NOT = { is_in_array = { array = random_faction_selected_options value = THIS } }
			}
			save_event_target_as = random_faction_option_4_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			PREV = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = {
			check_variable = {
				var = random_faction_option_count
				value = constant:random_faction_requirement.option_cap
				compare = equals
			}
		}
		set_country_flag = random_faction_four_doors_choice
	}
	clear_temp_array = random_faction_selected_options
	clear_temp_array = random_faction_option_pool
	set_temp_variable = { random_faction_option_pool_count = 0 }
}

random_faction_join_selected_faction = {
	if = {
		limit = {
			has_event_target = random_faction_selected_leader
			event_target:random_faction_selected_leader = { can_random_faction_join_faction = yes }
		}
		random_faction_reset_current_country_decision_cycle = yes
		clr_country_flag = random_faction_frontier_join_context
		if = {
			limit = {
				has_global_flag = random_faction_evolution_ii_unlocked
				OR = {
					has_war = yes
					any_neighbor_country = {
						is_in_faction = yes
						event_target:random_faction_selected_leader = { has_war_with = PREV }
					}
				}
			}
			set_country_flag = random_faction_frontier_join_context
		}
		set_temp_variable = { random_faction_current_duration = constant:random_faction_days.join_guard }
		set_country_flag = { flag = random_faction_join_in_progress days = random_faction_current_duration }
		event_target:random_faction_selected_leader = {
			add_to_faction = ROOT
		}
		if = {
			limit = { is_in_faction_with = event_target:random_faction_selected_leader }
			random_faction_remove_current_target_from_not_everyone_candidates = yes
			random_faction_retire_current_alignment_pressure = yes
			set_variable = { random_faction_chosen_leader = event_target:random_faction_selected_leader }
			if = {
				limit = { has_country_flag = random_faction_history_result_pending }
				set_variable = { random_faction_pending_history_leader = event_target:random_faction_selected_leader }
				random_faction_finalize_history_result = yes
			}
			set_variable = { random_faction_region_anchor = THIS }
			random_faction_apply_alignment_shock = yes
			random_faction_prepare_stabilize_alignment_cost = yes
			event_target:random_faction_selected_leader = {
				if = { limit = { NOT = { is_in_array = { array = global.random_faction_faction_leaders value = THIS } } } add_to_array = { array = global.random_faction_faction_leaders value = THIS } }
				add_to_variable = { random_faction_recent_member_count = 1 }
				add_to_variable = { random_faction_faction_cohesion_strain = constant:random_faction_pressure_value.leader_cohesion_strain }
				clamp_variable = { var = random_faction_faction_cohesion_strain min = constant:random_faction_pressure_value.resilience_floor max = constant:random_faction_pressure_value.resilience_ceiling }
				set_variable = { random_faction_last_aligned_member = ROOT }
				set_variable = { random_faction_last_region_anchor = ROOT }
				if = {
					limit = { NOT = { has_country_flag = random_faction_rival_reaction_cooldown } }
					set_temp_variable = { random_faction_current_duration = constant:random_faction_days.leader_reaction }
					set_country_flag = { flag = random_faction_rival_reaction_cooldown days = random_faction_current_duration }
					country_event = { id = chaosx.nr17.40 }
				}
			}
			random_faction_apply_regional_pressure = yes
			random_faction_check_evolution_unlocks = yes
			random_faction_schedule_neighbor_followup = yes
			random_faction_check_join_achievements = yes
			news_event = { id = chaosx.news.4 days = constant:random_faction_days.news_report }
			set_temp_variable = { random_faction_current_duration = 0 }
		}
		else = {
			clr_country_flag = random_faction_join_in_progress
			random_faction_reopen_dead_faction_file = yes
		}
	}
	else = {
		random_faction_reopen_dead_faction_file = yes
	}
}

random_faction_add_current_country_to_region_bucket = {
	ROOT = {
		if = {
			limit = { NOT = { is_in_array = { array = random_faction_current_region_targets value = PREV } } }
			add_to_array = { array = random_faction_current_region_targets value = PREV }
		}
	}
	set_variable = { random_faction_pressure_source_country = ROOT }
	set_variable = { random_faction_pressure_anchor_country = ROOT }
	set_variable = { random_faction_region_anchor = ROOT }
	if = {
		limit = { ROOT = { has_variable = random_faction_region_id } }
		set_variable = { random_faction_region_id = ROOT.random_faction_region_id }
	}
	set_temp_variable = { random_faction_current_duration = constant:random_faction_days.border_pressure }
	set_country_flag = { flag = random_faction_regional_bucket_target days = random_faction_current_duration }

random_faction_build_region_bucket_if_empty = {
	if = {
		limit = {
			NOT = {
				any_of_scopes = {
					array = random_faction_current_region_targets
					random_faction_country_base_valid = yes
				}
			}
		}
		every_neighbor_country = {
			limit = { is_random_faction_pressure_neighbor = yes }
			random_faction_add_current_country_to_region_bucket = yes
		}
	}
	if = {
		limit = {
			NOT = {
				any_of_scopes = {
					array = random_faction_current_region_targets
					random_faction_country_base_valid = yes
				}
			}
		}
		every_country = {
			random_select_amount = @random_faction_maritime_reach_targets
			limit = {
				is_random_faction_pressure_neighbor = yes
				NOT = { tag = ROOT }
				is_on_same_continent_as = ROOT
				OR = {
					AND = {
						has_event_target = random_faction_selected_leader
						any_neighbor_country = { is_in_faction_with = event_target:random_faction_selected_leader }
					}
					AND = {
						any_controlled_state = { is_coastal = yes }
						ROOT = { any_controlled_state = { is_coastal = yes } }
						has_event_target = random_faction_selected_leader
						event_target:random_faction_selected_leader = {
							OR = {
								any_controlled_state = { is_coastal = yes }
								any_country = {
									is_in_faction_with = PREV
									any_controlled_state = { is_coastal = yes }
								}
							}
						}
					}
				}
			}
			random_faction_add_current_country_to_region_bucket = yes
		}
	}
}

random_faction_prepare_pressure_choice = {
	if = {
		limit = {
			random_faction_pressured_neutral_decisions_visible = yes
			random_faction_pressure_source_is_valid = yes
		}
		random_faction_track_current_selection = yes
		random_faction_collect_faction_options = yes
		# The pressured-neutral response exposes only its source and one rival;
		# the hidden four-leader scan must not qualify Four Doors.
		clr_country_flag = random_faction_four_doors_choice
		if = {
			limit = {
				random_faction_option_1_valid = yes
				NOT = { check_variable = { random_faction_pressure_source_leader = event_target:random_faction_option_1_leader } }
			}
			event_target:random_faction_option_1_leader = { save_event_target_as = random_faction_pressure_rival_leader }
		}
		else_if = {
			limit = {
				random_faction_option_2_valid = yes
				NOT = { check_variable = { random_faction_pressure_source_leader = event_target:random_faction_option_2_leader } }
			}
			event_target:random_faction_option_2_leader = { save_event_target_as = random_faction_pressure_rival_leader }
		}
		else_if = {
			limit = {
				random_faction_option_3_valid = yes
				NOT = { check_variable = { random_faction_pressure_source_leader = event_target:random_faction_option_3_leader } }
			}
			event_target:random_faction_option_3_leader = { save_event_target_as = random_faction_pressure_rival_leader }
		}
		else_if = {
			limit = {
				random_faction_option_4_valid = yes
				NOT = { check_variable = { random_faction_pressure_source_leader = event_target:random_faction_option_4_leader } }
			}
			event_target:random_faction_option_4_leader = { save_event_target_as = random_faction_pressure_rival_leader }
		}
		var:random_faction_pressure_source_leader = { save_event_target_as = random_faction_option_1_leader }
		set_variable = { random_faction_option_count = 1 }
		if = {
			limit = {
				has_event_target = random_faction_pressure_rival_leader
				event_target:random_faction_pressure_rival_leader = { can_random_faction_join_faction = yes }
			}
			event_target:random_faction_pressure_rival_leader = { save_event_target_as = random_faction_option_2_leader }
			set_variable = { random_faction_option_count = 2 }
		}
	}
}

random_faction_run_evo3_cascade = {
	if = {
		limit = {
			has_global_flag = random_faction_evolution_iii_unlocked
			NOT = { has_country_flag = random_faction_evo3_cascade_lock }
		}
		random_faction_set_current_region_id = yes
		random_faction_build_region_bucket_if_empty = yes
		clear_temp_array = random_faction_evo3_candidate_targets
		for_each_scope_loop = {
			array = random_faction_current_region_targets
			if = {
				limit = {
					is_random_faction_cascade_candidate = yes
					NOT = { is_in_array = { array = random_faction_evo3_candidate_targets value = THIS } }
				}
				add_to_temp_array = { random_faction_evo3_candidate_targets = THIS }
			}
		}
		set_variable = { random_faction_evo3_candidate_count = random_faction_evo3_candidate_targets^num }
		if = {
			limit = {
				check_variable = {
					var = random_faction_evo3_candidate_count
					value = constant:random_faction_requirement.evo3_min_region_candidates
					compare = greater_than_or_equals
				}
				random_faction_current_region_cascade_available = yes
			}
			random_faction_lock_current_region_cascade = yes
			set_temp_variable = { random_faction_current_duration = constant:random_faction_days.evo3_region_lock }
			set_country_flag = { flag = random_faction_evo3_cascade_lock days = random_faction_current_duration }
			set_variable = { random_faction_evo3_cascade_count = 0 }
			set_variable = { random_faction_evo3_response_budget = 0 }
			set_variable = { random_faction_evo3_budget_remaining = random_faction_evo3_candidate_count }
			while_loop_effect = {
				limit = { check_variable = { random_faction_evo3_budget_remaining > 0 } }
				add_to_variable = { random_faction_evo3_response_budget = 1 }
				subtract_from_variable = { random_faction_evo3_budget_remaining = constant:random_faction_requirement.evo3_region_share_divisor }
			}
			clamp_variable = {
				var = random_faction_evo3_response_budget
				min = constant:random_faction_requirement.evo3_survivor_reserve
				max = constant:random_faction_requirement.evo3_max_followups
			}
			set_variable = { random_faction_evo3_candidate_limit = random_faction_evo3_candidate_count }
			subtract_from_variable = { random_faction_evo3_candidate_limit = constant:random_faction_requirement.evo3_survivor_reserve }
			if = {
				limit = {
					check_variable = {
						var = random_faction_evo3_response_budget
						value = random_faction_evo3_candidate_limit
						compare = greater_than
					}
				}
				set_variable = { random_faction_evo3_response_budget = random_faction_evo3_candidate_limit }
			}
			set_variable = { random_faction_evo3_forced_choice_budget = constant:random_faction_requirement.evo3_min_followups }
			if = {
				limit = {
					check_variable = {
						var = random_faction_evo3_forced_choice_budget
						value = random_faction_evo3_response_budget
						compare = greater_than
					}
				}
				set_variable = { random_faction_evo3_forced_choice_budget = random_faction_evo3_response_budget }
			}
			while_loop_effect = {
				limit = { random_faction_region_can_cascade = yes }
				random_scope_in_array = {
					array = random_faction_evo3_candidate_targets
					limit = { is_random_faction_cascade_candidate = yes }
					set_temp_variable = { random_faction_current_duration = constant:random_faction_days.recent_alignment }
					set_country_flag = { flag = random_faction_evo3_handled days = random_faction_current_duration }
					ROOT = { add_to_variable = { random_faction_evo3_cascade_count = 1 } }
					if = {
						limit = {
							ROOT = {
								check_variable = {
									var = random_faction_evo3_cascade_count
									value = random_faction_evo3_forced_choice_budget
									compare = less_than_or_equals
								}
							}
						}
						random_faction_dispatch_current_country_choice = yes
					}
					else = {
						set_temp_variable = { random_faction_evo3_join_weight = constant:random_faction_evo3_weight.join_base }
						set_temp_variable = { random_faction_evo3_resist_weight = constant:random_faction_evo3_weight.resist_base }
						if = {
							limit = { random_faction_target_has_low_resilience = yes }
							add_to_temp_variable = { random_faction_evo3_join_weight = constant:random_faction_evo3_weight.low_resilience_shift }
							subtract_from_temp_variable = { random_faction_evo3_resist_weight = constant:random_faction_evo3_weight.low_resilience_shift }
						}
						if = {
							limit = { random_faction_target_has_high_resilience = yes }
							subtract_from_temp_variable = { random_faction_evo3_join_weight = constant:random_faction_evo3_weight.high_resilience_shift }
							add_to_temp_variable = { random_faction_evo3_resist_weight = constant:random_faction_evo3_weight.high_resilience_shift }
						}
						if = {
							limit = { has_war = yes }
							add_to_temp_variable = { random_faction_evo3_join_weight = constant:random_faction_evo3_weight.wartime_shift }
							subtract_from_temp_variable = { random_faction_evo3_resist_weight = constant:random_faction_evo3_weight.wartime_shift }
						}
						if = {
							limit = { random_faction_target_has_source_faction_neighbor = yes }
							add_to_temp_variable = { random_faction_evo3_join_weight = constant:random_faction_evo3_weight.source_neighbor_shift }
							subtract_from_temp_variable = { random_faction_evo3_resist_weight = constant:random_faction_evo3_weight.source_neighbor_shift }
						}
						if = {
							limit = { random_faction_high_chaos = yes }
							add_to_temp_variable = { random_faction_evo3_join_weight = constant:random_faction_evo3_weight.high_chaos_shift }
							subtract_from_temp_variable = { random_faction_evo3_resist_weight = constant:random_faction_evo3_weight.high_chaos_shift }
						}
						clamp_temp_variable = { var = random_faction_evo3_join_weight min = constant:random_faction_evo3_weight.minimum max = constant:random_faction_evo3_weight.maximum }
						clamp_temp_variable = { var = random_faction_evo3_resist_weight min = constant:random_faction_evo3_weight.minimum max = constant:random_faction_evo3_weight.maximum }
						random_list = {
							random_faction_evo3_join_weight = { random_faction_dispatch_current_country_choice = yes }
							random_faction_evo3_resist_weight = { random_faction_pressure_resist_alignment = yes }
						}
						set_temp_variable = { random_faction_evo3_join_weight = 0 }
						set_temp_variable = { random_faction_evo3_resist_weight = 0 }
					}
				}
			}
			if = {
				limit = { random_faction_current_region_has_neutral_survivor = yes }
				random_faction_start_not_everyone_candidate = yes
			}
			if = {
				limit = {
					check_variable = {
						var = random_faction_evo3_cascade_count
						value = constant:random_faction_requirement.regional_report_followups
						compare = greater_than_or_equals
					}

random_faction_finalize_history_result = {
	set_temp_variable = { random_faction_history_result_bound = 0 }
	if = {
		limit = {
			has_variable = random_faction_pending_history_sequence
			has_variable = random_faction_pending_history_leader
			country_exists = var:random_faction_pending_history_leader
			var:random_faction_pending_history_leader = { exists = yes }
		}
		sanitize_events_log_history_actor_entries = yes
		set_temp_variable = { random_faction_history_result_index = 0 }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = random_faction_history_result_index
					value = global.events_log_history_sequence_entries^num
					compare = less_than
				}
				check_variable = { random_faction_history_result_bound < 1 }
			}
			if = {
				limit = {
					check_variable = {
						var = global.events_log_history_sequence_entries^random_faction_history_result_index
						value = random_faction_pending_history_sequence
						compare = equals
					}
					check_variable = {
						var = global.events_log_history_event_id_entries^random_faction_history_result_index
						value = constant:random_faction_event.id
						compare = equals
					}
				}
				set_variable = { global.events_log_history_secondary_actor_entries^random_faction_history_result_index = random_faction_pending_history_leader }
				set_variable = { global.events_log_history_has_secondary_actor_entries^random_faction_history_result_index = 1 }
				set_temp_variable = { random_faction_history_result_bound = 1 }
			}
			add_to_temp_variable = { random_faction_history_result_index = 1 }
		}
	}
	if = {
		limit = { check_variable = { random_faction_history_result_bound > 0 } }
		clr_country_flag = random_faction_history_result_pending
		clear_variable = random_faction_pending_history_sequence
		clear_variable = random_faction_pending_history_leader
		every_country = {
			limit = {
				is_ai = no
				has_country_flag = events_log_window_open
			}
			if = { limit = { has_country_flag = events_log_history_tab } rebuild_events_log_history_view = yes }
			if = { limit = { has_country_flag = events_log_event_details_open } events_log_rebuild_open_event_details_view = yes }
		}
	}
}

			array = random_faction_liaison_web_candidate_targets
			if = {
				limit = {
					random_faction_country_base_valid = yes
					NOT = { is_subject = yes }
					NOT = { has_capitulated = yes }
					event_target:random_faction_supported_leader = {
						is_random_faction_allowed_faction_leader = yes
						NOT = { has_war_with = PREV }
					}
				}
				ROOT = { add_to_temp_variable = { random_faction_liaison_web_candidate_count = 1 } }
			}
		}
		if = {
			limit = {
				check_variable = {
					var = random_faction_liaison_web_candidate_count
					value = constant:random_faction_requirement.liaison_web_targets
					compare = greater_than_or_equals
				}
			}
			set_country_flag = random_faction_achievement_liaison_web
		}
		random_faction_unregister_liaison_web_candidate = yes
		set_temp_variable = { random_faction_liaison_web_candidate_count = 0 }
	}
	else = {
		random_faction_unregister_liaison_web_candidate = yes
	}
}

random_faction_check_crowded_border_achievement = {
	every_country = {
		limit = {
			random_faction_country_base_valid = yes
			NOT = { is_subject = yes }
			NOT = { is_in_faction = yes }
			NOT = { is_major = yes }
			OR = {
				has_country_flag = random_faction_regional_bucket_target
				random_faction_has_active_pressure = yes
				is_in_array = { array = global.random_faction_pressured_neutrals value = THIS }
			}
			any_neighbor_country = { is_in_faction = yes }
		}
		save_event_target_as = random_faction_crowded_border_candidate
		set_temp_variable = { random_faction_neighbor_faction_count = 0 }
		every_country = {
			limit = {
				is_random_faction_allowed_faction_leader = yes
				any_country = {
					is_in_faction_with = PREV
					is_neighbor_of = event_target:random_faction_crowded_border_candidate
				}
			}
			add_to_temp_variable = { random_faction_neighbor_faction_count = 1 }
		}
		if = {
			limit = {
				check_variable = {
					var = random_faction_neighbor_faction_count
					value = constant:random_faction_requirement.crowded_border_factions
					compare = greater_than_or_equals
				}
			}
			set_country_flag = random_faction_achievement_crowded_border
			set_global_flag = random_faction_achievement_crowded_border
		}
	}
}
