random_faction_clamp_current_country_values = {
	if = { limit = { NOT = { has_variable = random_faction_pressure_score } } set_variable = { random_faction_pressure_score = 0 } }
	if = { limit = { NOT = { has_variable = random_faction_neutrality_resilience } } set_variable = { random_faction_neutrality_resilience = constant:random_faction_pressure_value.neutrality_resilience } }
	clamp_variable = { var = random_faction_pressure_score min = constant:random_faction_pressure_value.resilience_floor max = constant:random_faction_pressure_value.resilience_ceiling }
	clamp_variable = { var = random_faction_neutrality_resilience min = constant:random_faction_pressure_value.resilience_floor max = constant:random_faction_pressure_value.resilience_ceiling }
}

random_faction_prepare_candidate_weight = {
	set_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.base }
	if = { limit = { NOT = { is_major = yes } } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.minor_bonus } }
	if = { limit = { any_neighbor_country = { is_in_faction = yes } } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.bordering_faction_bonus } }
	if = { limit = { has_war = yes } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.war_adjacent_bonus } }
	if = { limit = { has_stability < constant:random_faction_resistance_threshold.low_stability } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.low_stability_bonus } }
	if = { limit = { has_war_support < constant:random_faction_resistance_threshold.low_war_support } add_to_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.low_war_support_bonus } }
	if = { limit = { has_country_flag = random_faction_under_pressure } subtract_from_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.pressure_repeat_penalty } }
	if = { limit = { has_country_flag = random_faction_selection_cooldown } subtract_from_temp_variable = { random_faction_candidate_weight = constant:random_faction_selection_weight.cooldown_penalty } }
	round_temp_variable = random_faction_candidate_weight
	clamp_temp_variable = {
		var = random_faction_candidate_weight
		min = constant:random_faction_selection_weight.min
		max = constant:random_faction_selection_weight.max
	}
}

random_faction_add_current_country_to_target_pool = {
	random_faction_prepare_candidate_weight = yes
	set_temp_variable = { random_faction_candidate_weight_remaining = random_faction_candidate_weight }
	while_loop_effect = {
		limit = { check_variable = { random_faction_candidate_weight_remaining > 0 } }
		add_to_temp_array = { random_faction_target_pool = THIS }
		ROOT = { add_to_temp_variable = { random_faction_target_pool_count = 1 } }
		subtract_from_temp_variable = { random_faction_candidate_weight_remaining = 1 }
	}
	set_temp_variable = { random_faction_candidate_weight = 0 }
	set_temp_variable = { random_faction_candidate_weight_remaining = 0 }
}

random_faction_prepare_runtime_context = {
	random_faction_refresh_runtime_arrays = yes
	set_temp_variable = { random_faction_context_ready = 0 }
	clear_temp_array = random_faction_target_pool
	set_temp_variable = { random_faction_target_pool_count = 0 }

	if = {
		limit = {
			has_country_flag = random_faction_prefer_current_dispatch_target
			OR = {
				is_random_faction_eligible_country = yes
				AND = {
					has_global_flag = random_faction_evolution_ii_unlocked
					is_random_faction_wartime_candidate = yes
				}
			}
		}
		save_event_target_as = random_faction_target_country
		random_faction_collect_faction_options = yes
		if = {
			limit = { check_variable = { random_faction_option_count > 0 } }
			set_temp_variable = { random_faction_context_ready = 1 }
		}
	}

	if = {
		limit = { check_variable = { random_faction_context_ready < 1 } }
		every_country = {
			limit = {
				OR = {
					is_random_faction_eligible_country = yes
					AND = {
						has_global_flag = random_faction_evolution_ii_unlocked
						is_random_faction_wartime_candidate = yes
					}
				}
			}
			random_faction_add_current_country_to_target_pool = yes
		}
	}
	if = {
		limit = {
			check_variable = { random_faction_context_ready < 1 }
			check_variable = { random_faction_target_pool_count > 0 }
		}
		random_scope_in_array = {
			array = random_faction_target_pool
			limit = {
				OR = {
					is_random_faction_eligible_country = yes
					AND = {
						has_global_flag = random_faction_evolution_ii_unlocked
						is_random_faction_wartime_candidate = yes
					}
				}
			}
			save_event_target_as = random_faction_target_country
		}
	}
	clear_temp_array = random_faction_target_pool
	if = {
		limit = {
			check_variable = { random_faction_context_ready < 1 }
			has_event_target = random_faction_target_country
			event_target:random_faction_target_country = {
				OR = {
					is_random_faction_eligible_country = yes
					AND = {
						has_global_flag = random_faction_evolution_ii_unlocked
						is_random_faction_wartime_candidate = yes
					}
				}
			}
		}
		event_target:random_faction_target_country = {
			random_faction_collect_faction_options = yes
			if = {
				limit = { check_variable = { random_faction_option_count > 0 } }
				ROOT = { set_temp_variable = { random_faction_context_ready = 1 } }
			}
		}
	}
}

random_faction_add_current_leader_to_option_pool = {
	set_temp_variable = { random_faction_leader_weight = constant:random_faction_ai_weight.medium }
	if = { limit = { any_country = { is_in_faction_with = PREV is_neighbor_of = ROOT } } add_to_temp_variable = { random_faction_leader_weight = constant:random_faction_ai_weight.high } }
	if = { limit = { has_war = yes } add_to_temp_variable = { random_faction_leader_weight = constant:random_faction_ai_weight.low } }
	clamp_temp_variable = { var = random_faction_leader_weight min = constant:random_faction_ai_weight.low max = constant:random_faction_ai_weight.urgent }
	while_loop_effect = {
		limit = { check_variable = { random_faction_leader_weight > 0 } }
		add_to_temp_array = { random_faction_option_pool = THIS }
		ROOT = { add_to_temp_variable = { random_faction_option_pool_count = 1 } }
		subtract_from_temp_variable = { random_faction_leader_weight = 1 }
	}
}

random_faction_collect_faction_options = {
	set_variable = { random_faction_option_count = 0 }
	set_temp_variable = { random_faction_option_pool_count = 0 }
	clear_temp_array = random_faction_option_pool
	clear_temp_array = random_faction_selected_options
	if = {
		limit = {
			OR = {
				is_random_faction_eligible_country = yes
				AND = {
					has_global_flag = random_faction_evolution_ii_unlocked
					is_random_faction_wartime_candidate = yes
				}
			}
		}
		every_country = {
			limit = { random_faction_is_valid_faction_leader_for_root_target = yes }
			random_faction_add_current_leader_to_option_pool = yes
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_pool_count > 0 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = { random_faction_is_valid_faction_leader_for_root_target = yes }
			save_event_target_as = random_faction_option_1_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			ROOT = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 0 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = {
				random_faction_is_valid_faction_leader_for_root_target = yes
				NOT = { is_in_array = { array = random_faction_selected_options value = THIS } }
			}
			save_event_target_as = random_faction_option_2_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			ROOT = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 1 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = {
				random_faction_is_valid_faction_leader_for_root_target = yes
				NOT = { is_in_array = { array = random_faction_selected_options value = THIS } }
			}
			save_event_target_as = random_faction_option_3_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			ROOT = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 2 } }
		random_scope_in_array = {
			array = random_faction_option_pool
			limit = {
				random_faction_is_valid_faction_leader_for_root_target = yes
				NOT = { is_in_array = { array = random_faction_selected_options value = THIS } }
			}
			save_event_target_as = random_faction_option_4_leader
			add_to_temp_array = { random_faction_selected_options = THIS }
			ROOT = { add_to_variable = { random_faction_option_count = 1 } }
		}
	}
	if = {
		limit = { check_variable = { random_faction_option_count > 3 } }
		set_country_flag = random_faction_four_doors_choice
	}
	clear_temp_array = random_faction_selected_options
	clear_temp_array = random_faction_option_pool
	set_temp_variable = { random_faction_option_pool_count = 0 }
}

random_faction_apply_pressure_to_current_region_target = {
	if = {
		limit = { is_random_faction_pressure_neighbor = yes }
		set_country_flag = { flag = random_faction_under_pressure days = @random_faction_border_pressure_days }
		set_country_flag = { flag = random_faction_pressure_cooldown days = @random_faction_border_pressure_days }
		add_timed_idea = { idea = random_faction_border_pressure days = @random_faction_border_pressure_days }

random_faction_build_region_bucket_if_empty = {
	if = {
		limit = {
			NOT = {
				any_of_scopes = {
					array = random_faction_current_region_targets
					random_faction_country_base_valid = yes
				}
			}
		}
		every_neighbor_country = {
			limit = { is_random_faction_pressure_neighbor = yes }
			random_faction_add_current_country_to_region_bucket = yes
		}
	}
	if = {
		limit = {
			NOT = {
				any_of_scopes = {
					array = random_faction_current_region_targets
					random_faction_country_base_valid = yes
				}
			}
		}
		every_country = {
			random_select_amount = @random_faction_regional_fallback_targets
			limit = {
				is_random_faction_pressure_neighbor = yes
				NOT = { tag = ROOT }
				is_on_same_continent_as = ROOT
				OR = {
					AND = {
						has_event_target = random_faction_selected_leader
						any_neighbor_country = { is_in_faction_with = event_target:random_faction_selected_leader }
					}
					AND = {
						any_controlled_state = { is_coastal = yes }
						ROOT = { any_controlled_state = { is_coastal = yes } }
					}
				}
			}
			random_faction_add_current_country_to_region_bucket = yes
		}
	}
}

random_faction_apply_regional_pressure = {
	add_to_variable = { global.random_faction_fire_count = 1 }
	clear_array = random_faction_current_region_targets
	every_neighbor_country = {
		limit = { is_random_faction_pressure_neighbor = yes }
		random_faction_apply_pressure_to_current_region_target = yes
	}
	if = {
		limit = {
			NOT = {
				any_of_scopes = {
					array = random_faction_current_region_targets
					random_faction_country_base_valid = yes
				}
			}
		}
		every_country = {
			random_select_amount = @random_faction_regional_fallback_targets
			limit = {
				is_random_faction_pressure_neighbor = yes
				NOT = { tag = ROOT }
				is_on_same_continent_as = ROOT
				AND = {
					any_controlled_state = { is_coastal = yes }
					ROOT = { any_controlled_state = { is_coastal = yes } }
				}
			}
			random_faction_apply_pressure_to_current_region_target = yes
		}
	}
	if = {
		limit = {
			has_global_flag = random_faction_evolution_i_unlocked
			any_of_scopes = {
				array = random_faction_current_region_targets
				random_faction_pressured_neutral_decisions_visible = yes
			}
		}
		random_scope_in_array = {
			array = random_faction_current_region_targets
			limit = { random_faction_pressured_neutral_decisions_visible = yes }
			country_event = { id = chaosx.nr17.30 days = @random_faction_evo1_followup_min_days random_days = @random_faction_evo1_followup_random_days }
		}
	}
	if = {
		limit = { has_global_flag = random_faction_evolution_ii_unlocked }
		country_event = { id = chaosx.nr17.60 days = @random_faction_evo2_pressure_min_days random_days = @random_faction_evo2_pressure_random_days }
	}
	if = {
		limit = { has_global_flag = random_faction_evolution_iii_unlocked }
		country_event = { id = chaosx.nr17.70 days = @random_faction_evo3_cascade_min_days random_days = @random_faction_evo3_cascade_random_days }
	}
}

random_faction_run_evo1_regional_bloc_race = {
	if = {
		limit = { has_global_flag = random_faction_evolution_i_unlocked }
		if = {
			limit = { random_faction_pressured_neutral_decisions_visible = yes }
			random_faction_prepare_pressure_choice = yes
			if = {
				limit = { is_ai = no }
				country_event = { id = chaosx.nr17.30 }
			}
			else = {
				random_list = {
					35 = { random_faction_pressure_copy_bloc = yes }
					25 = { random_faction_pressure_counter_align = yes }
					40 = { random_faction_pressure_resist_alignment = yes }
				}
			}
		}
		else = {
			random_faction_build_region_bucket_if_empty = yes
			random_scope_in_array = {
				array = random_faction_current_region_targets
				limit = { random_faction_pressured_neutral_decisions_visible = yes }
				country_event = { id = chaosx.nr17.30 }
			}
		}
	}
}

random_faction_run_evo2_pressure = {
	if = {
		limit = { has_global_flag = random_faction_evolution_ii_unlocked }
		random_faction_build_region_bucket_if_empty = yes
		if = {
			limit = {
				any_of_scopes = {
					array = random_faction_current_region_targets
					is_random_faction_wartime_candidate = yes
					NOT = { has_country_flag = random_faction_under_pressure }
				}
			}
			for_each_scope_loop = {
				array = random_faction_current_region_targets
				random_faction_apply_wartime_pressure_to_current_region_target = yes
			}
		}
		else = {
			every_neighbor_country = {
				limit = {
					is_random_faction_wartime_candidate = yes
					NOT = { has_country_flag = random_faction_under_pressure }
				}
				random_faction_apply_wartime_pressure_to_current_region_target = yes
			}
		}
	}
}

random_faction_run_evo3_cascade = {
	if = {
		limit = {
			has_global_flag = random_faction_evolution_iii_unlocked
			NOT = { has_country_flag = random_faction_evo3_cascade_lock }
		}
		random_faction_build_region_bucket_if_empty = yes
		set_country_flag = { flag = random_faction_evo3_cascade_lock days = @random_faction_evo3_cascade_random_days }
		set_variable = { global.random_faction_evo3_cascade_count = 0 }
		while_loop_effect = {
			limit = { random_faction_region_can_cascade = yes }
			random_scope_in_array = {
				array = random_faction_current_region_targets
				limit = {
					is_random_faction_pressure_neighbor = yes
					NOT = { has_country_flag = random_faction_evo3_handled }
				}
				set_country_flag = { flag = random_faction_evo3_handled days = @random_faction_selection_cooldown_days }
				add_to_variable = { global.random_faction_evo3_cascade_count = 1 }
				if = {
					limit = {
						check_variable = {
							var = global.random_faction_evo3_cascade_count
							value = constant:random_faction_requirement.evo3_min_followups
							compare = less_than_or_equals
						}
					}
					random_faction_dispatch_current_country_choice = yes
				}
				else = {
					random_list = {
						35 = { random_faction_dispatch_current_country_choice = yes }
						65 = { random_faction_pressure_resist_alignment = yes }
					}
				}
			}
		}
		if = {
			limit = { random_faction_current_region_has_neutral_survivor = yes }
			set_global_flag = random_faction_not_everyone_candidate
			set_country_flag = { flag = random_faction_not_everyone_region_candidate days = @random_faction_achievement_frontier_days }
			country_event = { id = chaosx.nr17.86 days = @random_faction_achievement_frontier_days }
		}
	}
}

random_faction_record_current_evolution = {
	set_temp_variable = { events_log_evolution_event_id = constant:random_faction_event.id }
	set_temp_variable = { events_log_evolution_type = constant:random_faction_event.evolution_type }
	set_temp_variable = { events_log_evolution_has_actor = 0 }
	if = {
		limit = {
			has_event_target = random_faction_target_country
			event_target:random_faction_target_country = { exists = yes }
		}
		event_target:random_faction_target_country = { save_event_target_as = events_log_evolution_actor }
		set_temp_variable = { events_log_evolution_has_actor = 1 }
	}
	if = { limit = { is_current_evolution_enabled = yes } record_events_log_evolution_entry = yes }
}

random_faction_check_evolution_unlocks = {
	if = {
		limit = {
			NOT = { has_global_flag = random_faction_evolution_i_unlocked }
			check_variable = { var = global.random_faction_fire_count value = 2 compare = greater_than_or_equals }
		}
		set_temp_variable = { events_log_evolution_stage = constant:random_faction_evolution_stage.regional_bloc_race }
		set_temp_variable = { events_log_evolution_tier = constant:chaos_meter_tier_id.tier_1 }
		if = {
			limit = { is_current_evolution_enabled = yes }
			set_global_flag = random_faction_evolution_i_unlocked
			random_faction_record_current_evolution = yes
			country_event = { id = chaosx.nr17.50 days = @random_faction_evo1_followup_min_days random_days = @random_faction_evo1_followup_random_days }
		}
	}
	if = {
		limit = {
			has_global_flag = random_faction_evolution_i_unlocked
			NOT = { has_global_flag = random_faction_evolution_ii_unlocked }
			OR = {
				has_war = yes
				any_neighbor_country = { has_war = yes }
			}
		}
		set_temp_variable = { events_log_evolution_stage = constant:random_faction_evolution_stage.pressured_neutrality }
		set_temp_variable = { events_log_evolution_tier = constant:chaos_meter_tier_id.tier_2 }
		if = {
			limit = { is_current_evolution_enabled = yes }
			set_global_flag = random_faction_evolution_ii_unlocked
			random_faction_record_current_evolution = yes
			country_event = { id = chaosx.nr17.60 days = @random_faction_evo2_pressure_min_days random_days = @random_faction_evo2_pressure_random_days }
		}
	}
	if = {
		limit = {
			has_global_flag = random_faction_evolution_ii_unlocked
			NOT = { has_global_flag = random_faction_evolution_iii_unlocked }
			check_variable = { var = global.random_faction_pressure_count value = 3 compare = greater_than_or_equals }
		}
		set_temp_variable = { events_log_evolution_stage = constant:random_faction_evolution_stage.neutrality_collapse }
		set_temp_variable = { events_log_evolution_tier = constant:chaos_meter_tier_id.tier_3 }
		if = {
			limit = { is_current_evolution_enabled = yes }
			set_global_flag = random_faction_evolution_iii_unlocked
			random_faction_record_current_evolution = yes
			country_event = { id = chaosx.nr17.70 days = @random_faction_evo3_cascade_min_days random_days = @random_faction_evo3_cascade_random_days }
		}
	}
}

random_faction_refresh_runtime_arrays = {
	clear_temp_array = random_faction_live_leaders
	for_each_scope_loop = {
		array = global.random_faction_faction_leaders
		if = {
			limit = { is_random_faction_allowed_faction_leader = yes }
			random_faction_refresh_supported_minor_count = yes
			add_to_temp_array = { random_faction_live_leaders = THIS }
		}
		else = { random_faction_clear_current_leader_pressure_state = yes }
	}
	resize_array = { global.random_faction_faction_leaders = 0 }
	for_each_scope_loop = {
		array = random_faction_live_leaders
		add_to_array = { array = global.random_faction_faction_leaders value = THIS }
	}
	clear_temp_array = random_faction_live_leaders

	clear_temp_array = random_faction_live_pressure_targets
	for_each_scope_loop = {
		array = global.random_faction_pressure_targets
		if = {
			limit = {
				random_faction_country_base_valid = yes
				OR = {
					random_faction_has_active_pressure = yes
					has_country_flag = random_faction_recent_alignment
					is_in_faction = yes
				}
			}
			add_to_temp_array = { random_faction_live_pressure_targets = THIS }
		}
		else = { random_faction_clear_current_country_pressure_state = yes }
	}
	resize_array = { global.random_faction_pressure_targets = 0 }
	for_each_scope_loop = {
		array = random_faction_live_pressure_targets
		add_to_array = { array = global.random_faction_pressure_targets value = THIS }
	}
	clear_temp_array = random_faction_live_pressure_targets
}

random_faction_cleanup_after_country_lifecycle_change = {

	}

}

}

random_faction_refresh_supported_minor_count = {
	save_event_target_as = random_faction_supported_leader
	set_variable = { random_faction_supported_minors = 0 }
	clear_temp_array = random_faction_live_supported_minor_targets
	for_each_scope_loop = {
		array = random_faction_supported_minor_targets
		if = {
			limit = {
				random_faction_country_base_valid = yes
				NOT = { is_subject = yes }
				NOT = { has_capitulated = yes }
				event_target:random_faction_supported_leader = {
					is_random_faction_allowed_faction_leader = yes
					NOT = { has_war_with = PREV }
				}
			}
			event_target:random_faction_supported_leader = {
				if = {
					limit = { NOT = { is_in_array = { array = random_faction_live_supported_minor_targets value = PREV } } }
					add_to_temp_array = { random_faction_live_supported_minor_targets = PREV }
				}
				add_to_variable = { random_faction_supported_minors = 1 }
			}
		}
	}
	resize_array = { random_faction_supported_minor_targets = 0 }
	for_each_scope_loop = {
		array = random_faction_live_supported_minor_targets
		add_to_array = { array = random_faction_supported_minor_targets value = THIS }
	}
	clear_temp_array = random_faction_live_supported_minor_targets
}

random_faction_start_liaison_web_candidate = {
	random_faction_refresh_supported_minor_count = yes
	if = {
		limit = {
			NOT = { has_country_flag = random_faction_achievement_liaison_web }
			NOT = { has_country_flag = random_faction_liaison_web_candidate }
			check_variable = {
				var = random_faction_supported_minors
				value = constant:random_faction_requirement.liaison_web_targets
				compare = greater_than_or_equals
			}
		}
		clear_array = random_faction_liaison_web_candidate_targets
		for_each_scope_loop = {
			array = random_faction_supported_minor_targets
			add_to_array = { array = random_faction_liaison_web_candidate_targets value = THIS }
		}
		set_country_flag = { flag = random_faction_liaison_web_candidate days = @random_faction_liaison_mission_days }
		country_event = { id = chaosx.nr17.83 days = @random_faction_liaison_mission_days }
	}
}

random_faction_check_liaison_web_candidate_achievement = {

		}
		save_event_target_as = random_faction_supported_leader
		set_temp_variable = { random_faction_liaison_web_candidate_count = 0 }
		for_each_scope_loop = {
			array = random_faction_liaison_web_candidate_targets
			if = {
				limit = {
					random_faction_country_base_valid = yes
					NOT = { is_subject = yes }
					NOT = { has_capitulated = yes }
					event_target:random_faction_supported_leader = {
						is_random_faction_allowed_faction_leader = yes
						NOT = { has_war_with = PREV }
					}
				}
				ROOT = { add_to_temp_variable = { random_faction_liaison_web_candidate_count = 1 } }
			}
		}
		if = {
			limit = {
				check_variable = {
					var = random_faction_liaison_web_candidate_count
					value = constant:random_faction_requirement.liaison_web_targets
					compare = greater_than_or_equals
				}
			}
			set_country_flag = random_faction_achievement_liaison_web
		}
		clr_country_flag = random_faction_liaison_web_candidate
		clear_array = random_faction_liaison_web_candidate_targets
		set_temp_variable = { random_faction_liaison_web_candidate_count = 0 }
	}
}

random_faction_check_crowded_border_achievement = {
	every_country = {
		limit = {
			NOT = { is_in_faction = yes }
			NOT = { is_major = yes }
			any_neighbor_country = { is_in_faction = yes }
		}
		save_event_target_as = random_faction_crowded_border_candidate
		set_temp_variable = { random_faction_neighbor_faction_count = 0 }
		every_country = {
			limit = {
				is_random_faction_allowed_faction_leader = yes
				any_country = {
					is_in_faction_with = PREV
					is_neighbor_of = event_target:random_faction_crowded_border_candidate
				}
			}
			add_to_temp_variable = { random_faction_neighbor_faction_count = 1 }
		}
		if = {
			limit = {
				check_variable = {
					var = random_faction_neighbor_faction_count
					value = constant:random_faction_requirement.crowded_border_factions
					compare = greater_than_or_equals
				}
			}
			set_country_flag = random_faction_achievement_crowded_border
			set_global_flag = random_faction_achievement_crowded_border
		}
	}
}
