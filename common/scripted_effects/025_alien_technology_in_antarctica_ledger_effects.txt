# ============================================================================
# CHAOS REDUX - EVENT 025 / EVENT 036 SHARED ALIEN RECOVERY LEDGER
# ============================================================================
# Overview:
# - This file owns the narrow, reusable country-scoped ledger bridge shared by
#   the Antarctic Event 025 expedition and the legacy Event 036 spacecraft.
# - Public effects are resolve_event025, resolve_event036, grant_loser_result,
#   assign_fragment, transfer_physical_custody, and destroy_physical_custody.
#   Contracts and parent wiring are in the paired ledger markdown document.
# - Event 025 has separate historical-owner and physical-core targets. Fragment slots one
#   through six have one persistent global country target each, plus role data.
# - Arrays global.chaosx_nr25_fragment_owners and
#   global.chaosx_nr25_fragment_status are projections of those six slots and
#   are rebuilt only when this ledger is touched; no world scan is performed.
# - Event 025 and Event 036 result calls are idempotent country claims. Every
#   external reward request carries a source, result, overlap, and duplicate
#   guard before calling the parent-owned Event 016 API by its exact name.
# - This file deliberately creates no technology, equipment, Event 016 history,
#   Kruger or Directorate state, facility, unit, or recurring on-action scan.
#
# Public input contract:
# - Before assign_fragment, the caller supplies temporary
#   chaosx_alien_recovery_fragment_id and chaosx_alien_recovery_fragment_role.
#   IDs and roles are bounded to the six configured slots.
# - Before resolve_event025, the caller may supply temporary
#   chaosx_alien_recovery_overlap: zero means no exact collision, one means an
#   exact aircraft-family collision, and two means an exact propulsion collision.
#
# Parent Event 016 API contract:
# - The parent integration must define
#   chaosx_grant_external_alien_recovery_reward = yes.
# - Result, source_event, field, tier, and policy request values are normal
#   country variables. Overlap, upgraded, and reward_applied are temporary.
#   Public temporary copies use an _output suffix to avoid country shadowing.
# - An exact overlap asks the API to exclude only the overlapping family while
#   another base family remains. Event 025 still allows a base reward; an
#   Event 036 conversion allows only an upgrade or the Integration capstone.
# - The API maps each granted family or upgrade into one broad public field.
#   Only the flight, propulsion and weapons group (aircraft or propulsion
#   codes) can overlap between Event 025 and Event 036.
# - The bridge records those outputs in country variables and flags. It never
#   substitutes another Event 016 API when the named API is absent.
#
# Exact public effects:
# - chaosx_alien_recovery_ledger_resolve_event025
# - chaosx_alien_recovery_ledger_resolve_event036
# - chaosx_alien_recovery_ledger_grant_loser_result
# - chaosx_alien_recovery_ledger_assign_fragment
# Internal helpers include cleanup_invalid_targets, project_fragment_arrays,
# record_api_result, and the six explicit assign_fragment_slot effects.
# ============================================================================

	set_temp_variable = { chaosx_alien_recovery_pending_output = constant:chaosx_nr25_value.zero }

	set_temp_variable = { chaosx_alien_recovery_result_tier_output = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_alien_recovery_result_overlap_output = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_alien_recovery_result_source_event_output = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_alien_recovery_fragment_assigned = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_alien_recovery_reward_applied = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_alien_recovery_result = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_alien_recovery_result_field = constant:chaosx_nr25_value.zero }

	clr_country_flag = chaosx_alien_recovery_reward_pending
}

# COUNTRY scope. Sets the external API request defaults. The API may refine the
# result, field, tier, overlap, and applied marker, but it receives a complete
# request even when the optional field and tier are unknown.
chaosx_alien_recovery_ledger_prepare_api_request = {
	# The current Event 016 owner API stores its result as a normal variable;
	# keep the request there so the same bridge works before and after API wiring.
	set_variable = { chaosx_external_alien_recovery_result = chaosx_external_alien_recovery_requested_result }
	set_temp_variable = { chaosx_external_alien_recovery_overlap = chaosx_external_alien_recovery_requested_overlap }
	set_variable = { chaosx_external_alien_recovery_source_event = chaosx_external_alien_recovery_requested_source_event }
	# Reset the normal-variable bridge on every request so a later Event 036
	# overlap cannot inherit Event 025's previous field or tier.
	set_variable = { chaosx_external_alien_recovery_field = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_external_alien_recovery_tier = constant:chaosx_nr25_value.zero }
	# Clear a stale random-helper success receipt before any owner branch.
	set_temp_variable = { chaosx_custom_technology_random_grant_applied = constant:chaosx_nr25_value.zero }
	# Always bind provenance to this transaction rather than a preceding grant.
	set_temp_variable = { chaosx_custom_technology_source = chaosx_external_alien_recovery_requested_source_event }
	if = {
		limit = {
			check_variable = {
				var = chaosx_external_alien_recovery_requested_field
				value = constant:chaosx_nr25_alien_field.none
				compare = greater_than
			}
		}
		set_variable = { chaosx_external_alien_recovery_field = chaosx_external_alien_recovery_requested_field }
	}
	set_temp_variable = { chaosx_external_alien_recovery_upgraded = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_external_alien_recovery_upgrade_applied = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_external_alien_recovery_reward_applied = constant:chaosx_nr25_value.zero }
	set_variable = { chaosx_external_alien_recovery_allow_base = chaosx_external_alien_recovery_requested_allow_base }
	set_variable = { chaosx_external_alien_recovery_duplicate_guard = constant:chaosx_nr25_value.one }

	if = {
		limit = {
			check_variable = {
				var = chaosx_external_alien_recovery_requested_overlap
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
		}
		set_country_flag = chaosx_external_alien_recovery_exclude_overlap
	}
	else = {
		clr_country_flag = chaosx_external_alien_recovery_exclude_overlap
	}
}

# COUNTRY scope. Validates the API's bounded overlap output and copies its
# result into the shared public variables. A missing result is represented by
# the requested result, while reward_applied remains zero and pending remains
# visible so a parent integration failure cannot look like a grant.
chaosx_alien_recovery_ledger_record_api_result = {
	if = {
		limit = {
			check_variable = {
				var = chaosx_external_alien_recovery_result
				value = constant:chaosx_nr25_result.none
				compare = equals
			}
		}
		set_variable = { chaosx_external_alien_recovery_result = chaosx_external_alien_recovery_requested_result }
	}

	if = {
		limit = {
			OR = {
				check_variable = {
					var = chaosx_external_alien_recovery_overlap
					value = constant:chaosx_nr25_value.zero
					compare = less_than
				}
				check_variable = {
					var = chaosx_external_alien_recovery_overlap
					value = constant:chaosx_nr25_value.two
					compare = greater_than
				}
			}
		}
		set_temp_variable = { chaosx_external_alien_recovery_overlap = constant:chaosx_nr25_value.zero }
	}

	set_variable = { chaosx_alien_recovery_result = THIS.chaosx_external_alien_recovery_result }
	set_variable = { chaosx_alien_recovery_result_field = THIS.chaosx_external_alien_recovery_field }
	set_variable = { chaosx_alien_recovery_result_tier = THIS.chaosx_external_alien_recovery_tier }
	set_variable = { chaosx_alien_recovery_result_overlap = chaosx_external_alien_recovery_overlap }
	set_variable = { chaosx_alien_recovery_result_source_event = THIS.chaosx_external_alien_recovery_source_event }
	set_variable = { chaosx_alien_recovery_reward_applied = chaosx_external_alien_recovery_reward_applied }

	set_temp_variable = { chaosx_alien_recovery_reward_applied_output = chaosx_external_alien_recovery_reward_applied }
	set_temp_variable = { chaosx_alien_recovery_result_output = THIS.chaosx_external_alien_recovery_result }
	set_temp_variable = { chaosx_alien_recovery_result_field_output = THIS.chaosx_external_alien_recovery_field }
	set_temp_variable = { chaosx_alien_recovery_result_tier_output = THIS.chaosx_external_alien_recovery_tier }
	set_temp_variable = { chaosx_alien_recovery_result_overlap_output = chaosx_external_alien_recovery_overlap }
	set_temp_variable = { chaosx_alien_recovery_result_source_event_output = THIS.chaosx_external_alien_recovery_source_event }

	if = {
		limit = {
			check_variable = {
				var = chaosx_external_alien_recovery_reward_applied
				value = constant:chaosx_nr25_value.one
				compare = greater_than_or_equals
			}
		}
		set_country_flag = chaosx_alien_recovery_reward_applied_flag
		clr_country_flag = chaosx_alien_recovery_reward_pending
		set_temp_variable = { chaosx_alien_recovery_pending_output = constant:chaosx_nr25_value.zero }
	}
	else = {
		clr_country_flag = chaosx_alien_recovery_reward_applied_flag
		set_country_flag = chaosx_alien_recovery_reward_pending
		set_temp_variable = { chaosx_alien_recovery_pending_output = constant:chaosx_nr25_value.one }
	}

	if = {
		limit = {
			has_country_flag = chaosx_alien_recovery_reward_applied_flag
			check_variable = { var = chaosx_external_alien_recovery_overlap value = constant:chaosx_nr25_value.zero compare = greater_than }
			check_variable = {
				var = chaosx_external_alien_recovery_upgraded
				value = constant:chaosx_nr25_value.one
				compare = greater_than_or_equals
			}
		}
		set_country_flag = chaosx_alien_recovery_overlap_conversion_used
	}
}

# COUNTRY scope. Public Event 025 owner resolver. The current country becomes
# the only persistent Event 025 owner before the API call, making a duplicate
# callback safe even if the parent API returns no applied marker.
chaosx_alien_recovery_ledger_resolve_event025 = {
	chaosx_alien_recovery_ledger_initialize_public_outputs = yes
	chaosx_alien_recovery_ledger_cleanup_invalid_targets = yes

	if = {
		limit = { has_country_flag = chaosx_alien_recovery_event025_recorded }
		chaosx_alien_recovery_ledger_expose_event025_result = yes
	}
	else_if = {
		limit = { chaosx_alien_recovery_ledger_event025_can_resolve = yes }
		set_temp_variable = { chaosx_alien_recovery_ledger_overlap = constant:chaosx_nr25_value.zero }
		if = {
			limit = {
				has_variable = chaosx_alien_recovery_overlap
				check_variable = {
					var = chaosx_alien_recovery_overlap
					value = constant:chaosx_nr25_value.one
					compare = greater_than_or_equals
				}
				check_variable = {
					var = chaosx_alien_recovery_overlap
					value = constant:chaosx_nr25_value.two
					compare = less_than_or_equals
				}
			}
			set_temp_variable = { chaosx_alien_recovery_ledger_overlap = chaosx_alien_recovery_overlap }
		}

		save_global_event_target_as = chaosx_alien_recovery_event025_owner
		set_global_flag = chaosx_alien_recovery_event025_owner_recorded
		set_global_flag = chaosx_alien_recovery_event025_claimed
		set_country_flag = chaosx_alien_recovery_event025_recorded
		set_country_flag = chaosx_alien_recovery_event025_reward_pending
		chaosx_alien_recovery_ledger_initialize_core_custody = yes
		set_variable = { chaosx_alien_recovery_event025_overlap = chaosx_alien_recovery_ledger_overlap }
		# The current Event 016 API reads this scoped normal-variable bridge;
		# the public temporary input remains untouched for the caller.
		set_variable = { chaosx_alien_recovery_overlap = chaosx_alien_recovery_ledger_overlap }

		if = {
			limit = { NOT = { has_global_flag = chaosx_alien_recovery_world_event025_count_initialized } }
			set_variable = { global.chaosx_alien_recovery_world_event025_count = constant:chaosx_nr25_value.zero }
			set_global_flag = chaosx_alien_recovery_world_event025_count_initialized
		}
		add_to_variable = { global.chaosx_alien_recovery_world_event025_count = constant:chaosx_nr25_value.one }

		set_temp_variable = { chaosx_external_alien_recovery_requested_overlap = chaosx_alien_recovery_ledger_overlap }
		set_temp_variable = { chaosx_external_alien_recovery_requested_source_event = constant:chaosx_nr25_identity.event_id }
		# A prior Event 036 recovery still allows a base reward: the API excludes
		# only the overlapping family, then falls back to an upgrade and finally
		# the Integration capstone. The API chooses the public field.
		set_temp_variable = { chaosx_external_alien_recovery_requested_allow_base = constant:chaosx_nr25_value.one }
		set_temp_variable = { chaosx_external_alien_recovery_requested_field = constant:chaosx_nr25_alien_field.none }
		set_temp_variable = { chaosx_external_alien_recovery_requested_result = constant:chaosx_nr25_result.base_operational }
		chaosx_alien_recovery_ledger_prepare_api_request = yes
		chaosx_grant_external_alien_recovery_reward = yes
		chaosx_alien_recovery_ledger_record_api_result = yes
		clear_variable = chaosx_alien_recovery_overlap

		set_variable = { chaosx_alien_recovery_event025_result = chaosx_alien_recovery_result }
		set_variable = { chaosx_alien_recovery_event025_field = chaosx_alien_recovery_result_field }
		set_variable = { chaosx_alien_recovery_event025_tier = chaosx_alien_recovery_result_tier }
		set_variable = { chaosx_alien_recovery_event025_overlap = chaosx_alien_recovery_result_overlap }
		if = {
			limit = {
				check_variable = {
					var = chaosx_alien_recovery_reward_applied
					value = constant:chaosx_nr25_value.one
					compare = greater_than_or_equals
				}
			}
			set_country_flag = chaosx_alien_recovery_event025_reward_applied
			clr_country_flag = chaosx_alien_recovery_event025_reward_pending
		}
		else = {
			clr_country_flag = chaosx_alien_recovery_event025_reward_applied
			set_country_flag = chaosx_alien_recovery_event025_reward_pending
		}
		clr_country_flag = chaosx_alien_recovery_event025_overlap_aircraft
		clr_country_flag = chaosx_alien_recovery_event025_overlap_propulsion
		if = {
			limit = {
				check_variable = {
					var = chaosx_alien_recovery_event025_overlap
					value = constant:chaosx_nr25_value.one
					compare = equals
				}
			}
			set_country_flag = chaosx_alien_recovery_event025_overlap_aircraft
		}
		else_if = {
			limit = {
				check_variable = {
					var = chaosx_alien_recovery_event025_overlap
					value = constant:chaosx_nr25_value.two
					compare = equals
				}
			}
			set_country_flag = chaosx_alien_recovery_event025_overlap_propulsion
		}
	}
}

# COUNTRY scope. Public Event 036 resolver. It preserves the legacy caller
# when there is no exact Event 025 overlap and requests the parent API only for
# a later exact aircraft or propulsion conversion.
chaosx_alien_recovery_ledger_resolve_event036 = {
	chaosx_alien_recovery_ledger_initialize_public_outputs = yes
	chaosx_alien_recovery_ledger_cleanup_invalid_targets = yes

	if = {
		limit = { has_country_flag = chaosx_alien_recovery_event036_recorded }
		chaosx_alien_recovery_ledger_expose_event036_result = yes
	}
	else_if = {
		limit = { chaosx_alien_recovery_ledger_event036_can_resolve = yes }
		set_country_flag = chaosx_alien_recovery_event036_recorded
		# Event 036's current legacy result is an aircraft-family recovery, but
		# the bridge accepts an explicit field for future propulsion-family
		# callers. Exact overlap is resolved only when both sides fall in the
		# flight, propulsion and weapons field.
		set_variable = { chaosx_alien_recovery_event036_field = constant:chaosx_nr25_alien_field.aircraft }
		if = {
			limit = {
				has_variable = chaosx_alien_recovery_event036_requested_field
				check_variable = { var = chaosx_alien_recovery_event036_requested_field value = constant:chaosx_nr25_alien_field.none compare = greater_than }
			}
			set_variable = { chaosx_alien_recovery_event036_field = chaosx_alien_recovery_event036_requested_field }

# COUNTRY scope. Rebuilds physical fragment markers/counts from six targets.
# Uses a distinct loop name so release and transfer loops cannot be clobbered.
chaosx_alien_recovery_ledger_refresh_country_fragments = {
	set_variable = { chaosx_alien_recovery_ledger_fragment_count = constant:chaosx_nr25_value.zero }
	for_loop_effect = {
		start = constant:chaosx_nr25_value.one
		end = constant:chaosx_nr25_cost.fragments_total
		compare = less_than_or_equals
		value = chaosx_alien_recovery_refresh_slot
		meta_effect = {
			text = {
				clr_country_flag = chaosx_alien_recovery_fragment_[SLOT]_owner
				clear_variable = chaosx_alien_recovery_fragment_[SLOT]_role
				if = {
					limit = {
						has_event_target = chaosx_alien_recovery_fragment_owner_[SLOT]
						has_global_flag = chaosx_alien_recovery_fragment_[SLOT]_assigned
						event_target:chaosx_alien_recovery_fragment_owner_[SLOT] = { tag = PREV }
					}
					set_country_flag = chaosx_alien_recovery_fragment_[SLOT]_owner
					set_variable = { chaosx_alien_recovery_fragment_[SLOT]_role = global.chaosx_alien_recovery_fragment_[SLOT]_role }
					add_to_variable = { chaosx_alien_recovery_ledger_fragment_count = constant:chaosx_nr25_value.one }
				}
			}
			SLOT = "[?chaosx_alien_recovery_refresh_slot|.0]"
		}
	}
	set_variable = { chaosx_nr25_fragment_count = THIS.chaosx_alien_recovery_ledger_fragment_count }
}

# COUNTRY scope. Transfers every physical asset held by the sender, retaining
# the recipient's existing fragments. Input: selected normal country ID,
# matching regular custody_recipient target, and recipient consent flag.
# Outputs: temporary custody_applied, core_moved, fragments_moved (zero default).
# No cost, dependence, diplomacy, technology, or historical reward mutation.
chaosx_alien_recovery_ledger_transfer_physical_custody = {
	set_temp_variable = { chaosx_alien_recovery_custody_applied = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_alien_recovery_core_moved = constant:chaosx_nr25_value.zero }
	set_temp_variable = { chaosx_alien_recovery_fragments_moved = constant:chaosx_nr25_value.zero }
	if = {
		limit = { chaosx_alien_recovery_ledger_custody_transfer_is_valid = yes }
		save_event_target_as = chaosx_alien_recovery_custody_sender
		if = {
			limit = { chaosx_alien_recovery_ledger_is_core_custodian = yes }
			clr_country_flag = chaosx_alien_recovery_core_holder
			event_target:chaosx_alien_recovery_custody_recipient = {
				save_global_event_target_as = chaosx_alien_recovery_core_custodian
				set_country_flag = chaosx_alien_recovery_core_holder
			}
			set_temp_variable = { chaosx_alien_recovery_core_moved = constant:chaosx_nr25_value.one }
		}
		for_loop_effect = {
			start = constant:chaosx_nr25_value.one
			end = constant:chaosx_nr25_cost.fragments_total
			compare = less_than_or_equals
			value = chaosx_alien_recovery_transfer_slot
			meta_effect = {
				text = {
					if = {
						limit = {
							has_event_target = chaosx_alien_recovery_fragment_owner_[SLOT]
							has_global_flag = chaosx_alien_recovery_fragment_[SLOT]_assigned
							event_target:chaosx_alien_recovery_fragment_owner_[SLOT] = { tag = event_target:chaosx_alien_recovery_custody_sender }
						}
						set_temp_variable = { chaosx_alien_recovery_fragment_role = global.chaosx_alien_recovery_fragment_[SLOT]_role }
						event_target:chaosx_alien_recovery_custody_recipient = { chaosx_alien_recovery_ledger_assign_fragment_slot_[SLOT] = yes }
						add_to_temp_variable = { chaosx_alien_recovery_fragments_moved = constant:chaosx_nr25_value.one }
					}
				}
				SLOT = "[?chaosx_alien_recovery_transfer_slot|.0]"
			}
		}
		chaosx_alien_recovery_ledger_refresh_country_fragments = yes
		event_target:chaosx_alien_recovery_custody_recipient = {
			chaosx_alien_recovery_ledger_refresh_country_fragments = yes
			clr_country_flag = chaosx_alien_recovery_custody_consent_@PREV
		}
		chaosx_alien_recovery_ledger_project_fragment_arrays = yes
		set_temp_variable = { chaosx_alien_recovery_custody_applied = constant:chaosx_nr25_value.one }
	}
	clear_variable = chaosx_alien_recovery_custody_recipient_id
}

# COUNTRY scope (the seizing country). Forcible single-slot transfer used by
# an Evolution III outpost seizure: moves the lowest fragment slot held by the
# event target chaosx_nr25_rival_target_scope to this country and keeps its
# role. The recovered core never moves this way. Outputs: temporary
# chaosx_alien_recovery_fragment_seized, one when a slot moved, and
# chaosx_alien_recovery_seized_slot, the slot that moved; the caller sets
# both and chaosx_alien_recovery_fragment_assigned to zero first.
chaosx_alien_recovery_ledger_seize_fragment = {
	if = {
		limit = {
			exists = yes
			is_subject = no
			has_capitulated = no
			has_event_target = chaosx_nr25_rival_target_scope
			event_target:chaosx_nr25_rival_target_scope = { NOT = { tag = PREV } }
		}
		chaosx_alien_recovery_ledger_cleanup_invalid_targets = yes
		set_temp_variable = { chaosx_alien_recovery_fragment_role = constant:chaosx_nr25_value.zero }
		for_loop_effect = {
			start = constant:chaosx_nr25_value.one
			end = constant:chaosx_nr25_cost.fragments_total
			compare = less_than_or_equals
			value = chaosx_alien_recovery_seize_slot
			if = {
				limit = { check_variable = { var = chaosx_alien_recovery_fragment_seized value = constant:chaosx_nr25_value.zero compare = equals } }
				meta_effect = {
					text = {
						if = {
							limit = {
								has_event_target = chaosx_alien_recovery_fragment_owner_[SLOT]
								has_global_flag = chaosx_alien_recovery_fragment_[SLOT]_assigned
								event_target:chaosx_alien_recovery_fragment_owner_[SLOT] = { tag = event_target:chaosx_nr25_rival_target_scope }
							}
							set_temp_variable = { chaosx_alien_recovery_fragment_role = global.chaosx_alien_recovery_fragment_[SLOT]_role }
							chaosx_alien_recovery_ledger_assign_fragment_slot_[SLOT] = yes
							set_temp_variable = { chaosx_alien_recovery_fragment_seized = constant:chaosx_nr25_value.one }
							set_temp_variable = { chaosx_alien_recovery_seized_slot = chaosx_alien_recovery_seize_slot }
						}
					}
					SLOT = "[?chaosx_alien_recovery_seize_slot|.0]"
				}
			}
		}
		if = {
			limit = { check_variable = { var = chaosx_alien_recovery_fragment_seized value = constant:chaosx_nr25_value.one compare = equals } }
			chaosx_alien_recovery_ledger_refresh_country_fragments = yes

		}
	}
