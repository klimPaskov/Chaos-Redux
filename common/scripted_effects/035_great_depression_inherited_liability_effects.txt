								check_variable = { var = great_depression_center_project_work_days value = PREV.great_depression_inherited_liability_checkpoint_work_days_entries^inherited_row compare = greater_than }
							}
						}
						NOT = {
							AND = {
								has_variable = great_depression_center_project_failed_authorization_id
								check_variable = { var = great_depression_center_project_failed_authorization_id value = great_depression_center_project_completed_authorization_id compare = equals }
							}
						}
						check_variable = { var = great_depression_center_project_completed_authorization_id value = PREV.great_depression_inherited_liability_completed_id_entries^inherited_row compare = greater_than }
						check_variable = { var = great_depression_center_project_completed_authorization_id value = PREV.great_depression_inherited_liability_authorization_id_entries^inherited_row compare = greater_than_or_equals }
						OR = { has_state_flag = great_depression_center_reopened has_state_flag = great_depression_center_restructured }
						PREV = {
							has_variable = great_depression_inherited_resolution_authorization_id
							check_variable = { var = great_depression_inherited_liability_authorization_id_entries^inherited_row value = constant:great_depression_api.zero compare = greater_than_or_equals }
							check_variable = { var = great_depression_inherited_liability_completed_id_entries^inherited_row value = constant:great_depression_api.zero compare = greater_than_or_equals }
							check_variable = { var = great_depression_inherited_resolution_outcome value = constant:great_depression_center_stage.reopened compare = equals }
						}
						NOT = { check_variable = { var = great_depression_center_project_type value = constant:great_depression_center_project.protection compare = equals } }
						check_variable = { var = great_depression_center_project_type value = constant:great_depression_center_project.none compare = greater_than }
						check_variable = { var = great_depression_center_project_type value = constant:great_depression_center_project.protection compare = less_than }
					}
					set_temp_variable = { inherited_resolution = @inherited_paid_resolution }
					set_temp_variable = { inherited_resolution_authorization = great_depression_center_project_completed_authorization_id }
				}
				else_if = {
					limit = {
						check_variable = { var = inherited_row_eligible value = constant:great_depression_api.one compare = equals }
						is_controlled_by = PREV
						check_variable = { var = great_depression_center_episode_id value = PREV.great_depression_active_episode_id compare = equals }
						PREV = { has_country_flag = great_depression_inherited_disposition_paid }
					}
					if = {
						limit = {
							has_state_flag = great_depression_center_liquidated
							PREV = { check_variable = { var = great_depression_inherited_resolution_outcome value = constant:great_depression_center_stage.liquidated compare = equals } }
						}
						set_temp_variable = { inherited_resolution = @inherited_liquidated_resolution }
					}
					else_if = {
						limit = {
							has_state_flag = great_depression_center_abandoned
							PREV = { check_variable = { var = great_depression_inherited_resolution_outcome value = constant:great_depression_center_stage.abandoned compare = equals } }
						}
						set_temp_variable = { inherited_resolution = @inherited_abandoned_resolution }
					}
				}
				else_if = {
					limit = {
						NOT = { is_owned_by = PREV }
						PREV = {
							has_country_flag = great_depression_inherited_permanent_transfer_proven
							check_variable = { var = great_depression_inherited_resolution_outcome value = constant:great_depression_region_control.transferred compare = equals }
						}
					}
					# Native owner change, never merely controller change.
					set_temp_variable = { inherited_resolution = @inherited_transferred_resolution }
				}
			}
			if = {
				limit = { NOT = { check_variable = { var = inherited_resolution value = @inherited_unresolved compare = equals } } }
				set_variable = { great_depression_inherited_liability_outcome_entries^inherited_row = inherited_resolution }
				set_variable = { great_depression_inherited_liability_resolution_day_entries^inherited_row = global.num_days }
				set_variable = { great_depression_inherited_liability_resolution_authorization_id_entries^inherited_row = inherited_resolution_authorization }
				set_country_flag = great_depression_inherited_resolution_applied
			}
		}
	}
	clear_variable = great_depression_inherited_resolution_row
	clear_variable = great_depression_inherited_resolution_episode_id
	clear_variable = great_depression_inherited_resolution_event34_episode_id
	clear_variable = great_depression_inherited_resolution_receipt_id
	clear_variable = great_depression_inherited_resolution_state_id
	clear_variable = great_depression_inherited_resolution_outcome
	clear_variable = great_depression_inherited_resolution_authorization_id
	clr_country_flag = great_depression_inherited_disposition_paid
	clr_country_flag = great_depression_inherited_permanent_transfer_proven
}

# COUNTRY THIS. Native state callbacks supply state/outcome and optional paid
# authorization. Choose the latest exact-state row without native ROOT.
great_depression_dispatch_inherited_state_resolution = {
	clr_country_flag = great_depression_inherited_resolution_applied
	great_depression_check_inherited_liability_ledger = yes
	set_temp_variable = { inherited_dispatch_found = constant:great_depression_api.zero }
	if = {
		limit = {
			has_country_flag = great_depression_inherited_ledger_valid
			great_depression_active_country_is_valid = yes
			has_variable = great_depression_inherited_resolution_state_id
			has_variable = great_depression_inherited_resolution_outcome
			is_in_array = { array = global.states value = great_depression_inherited_resolution_state_id }
		}
		set_temp_variable = { inherited_dispatch_state = great_depression_inherited_resolution_state_id }
		set_temp_variable = { inherited_dispatch_row = great_depression_inherited_region_receipt_id_entries^num }
		while_loop_effect = {
			limit = {
				check_variable = { var = inherited_dispatch_row value = constant:great_depression_api.zero compare = greater_than }
				check_variable = { var = inherited_dispatch_found value = constant:great_depression_api.zero compare = equals }
			}
			subtract_from_temp_variable = { inherited_dispatch_row = constant:great_depression_api.one }
			if = {
				limit = {
					check_variable = { var = great_depression_inherited_region_episode_id_entries^inherited_dispatch_row value = great_depression_active_episode_id compare = equals }
					check_variable = { var = great_depression_inherited_region_state_id_entries^inherited_dispatch_row value = inherited_dispatch_state compare = equals }
				}
				set_temp_variable = { inherited_dispatch_found = constant:great_depression_api.one }
				set_variable = { great_depression_inherited_resolution_row = inherited_dispatch_row }
				set_variable = { great_depression_inherited_resolution_episode_id = great_depression_active_episode_id }
				set_variable = { great_depression_inherited_resolution_event34_episode_id = great_depression_inherited_region_event34_episode_id_entries^inherited_dispatch_row }
				set_variable = { great_depression_inherited_resolution_receipt_id = great_depression_inherited_region_receipt_id_entries^inherited_dispatch_row }
				great_depression_resolve_inherited_liability = yes
			}
		}
	}
	# A missing row must not leave transaction authority for a later call.
	clear_variable = great_depression_inherited_resolution_state_id
	clear_variable = great_depression_inherited_resolution_outcome
	clear_variable = great_depression_inherited_resolution_authorization_id
	clr_country_flag = great_depression_inherited_disposition_paid
	clr_country_flag = great_depression_inherited_permanent_transfer_proven
}
