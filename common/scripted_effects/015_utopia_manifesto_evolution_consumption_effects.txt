# ============================================================================
# CHAOS REDUX - EVENT 015 EVOLUTION CONSUMPTION EFFECTS
# ============================================================================
# Idempotent interpretation setup plus a single bounded policy-obligation
# lifecycle shared by all fifteen evolution options. Active event options and
# explicit prepared prefire choices enter through the same setup dispatcher.
# Ordinary setting-enabled prefire snapshots remain staged for later delivery.
# ============================================================================

# Input: temporary utopia_manifesto_evolution_choice_input.
# Output: temporary utopia_manifesto_evolution_choice_applied (0/1).
# Each stage accepts exactly one interpretation. Repeating the same prepared
# choice is idempotent; a later conflicting choice cannot replace the first.
utopia_manifesto_apply_prepared_evolution_choice = {
	set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.zero }

	if = {
		limit = {
			check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.glosses_open_schools }
			OR = {
				has_country_flag = utopia_manifesto_interpretation_schools_open
				NOT = { has_country_flag = utopia_manifesto_evolution_glosses_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_glosses_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_glosses_choice_setup
			set_country_flag = utopia_manifesto_interpretation_schools_open
			set_country_flag = utopia_manifesto_open_interpretation
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.medium }
			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_narrative_delta.negative_small }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}
	if = {
		limit = {
			check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.glosses_chartered_schools }
			OR = {
				has_country_flag = utopia_manifesto_interpretation_schools_chartered
				NOT = { has_country_flag = utopia_manifesto_evolution_glosses_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_glosses_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_glosses_choice_setup
			set_country_flag = utopia_manifesto_interpretation_schools_chartered
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.small }
			utopia_manifesto_apply_prepared_ledger_delta = yes

	}
	if = {
		limit = {
			check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.glosses_censored_margins }
			OR = {
				has_country_flag = utopia_manifesto_conduct_margins_censored
				NOT = { has_country_flag = utopia_manifesto_evolution_glosses_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_glosses_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_glosses_choice_setup
			set_country_flag = utopia_manifesto_conduct_margins_censored
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.negative_medium }
			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_narrative_delta.medium }

			OR = {

			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_shores_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_shores_choice_setup
			set_country_flag = utopia_manifesto_necessary_shores_domestic_alternative
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.medium }
			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_narrative_delta.negative_small }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}

				NOT = { has_country_flag = utopia_manifesto_evolution_shores_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_shores_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_shores_choice_setup
			set_country_flag = utopia_manifesto_necessary_shores_emergency_powers
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.negative_medium }

	}

	if = {

			OR = {

		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_cities_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_cities_choice_setup
			set_country_flag = utopia_manifesto_cities_local_charters
			utopia_manifesto_link_garden_district_network = yes
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.medium }
			utopia_manifesto_apply_prepared_ledger_delta = yes

				NOT = { has_country_flag = utopia_manifesto_evolution_cities_choice_setup }

			utopia_manifesto_link_garden_district_network = yes

			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_narrative_delta.small }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}
	if = {
		limit = {

			OR = {
				has_country_flag = utopia_manifesto_cities_foreign_circle
				NOT = { has_country_flag = utopia_manifesto_evolution_cities_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_cities_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_cities_choice_setup
			set_country_flag = utopia_manifesto_cities_foreign_circle
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.small }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}

	if = {
		limit = {
			check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.nowhere_plural_law }
			OR = {
				has_country_flag = utopia_manifesto_nowhere_law_plural
				NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_nowhere_choice_setup
			set_country_flag = utopia_manifesto_nowhere_law_plural
			set_temp_variable = { utopia_manifesto_league_cohesion_delta = constant:utopia_manifesto_narrative_delta.medium }
			utopia_manifesto_apply_prepared_league_cohesion_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}
	if = {
		limit = {
			check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.nowhere_sponsored_law }
			OR = {
				has_country_flag = utopia_manifesto_nowhere_law_sponsored
				NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_nowhere_choice_setup
			set_country_flag = utopia_manifesto_nowhere_law_sponsored
			set_temp_variable = { utopia_manifesto_plenty_delta = constant:utopia_manifesto_narrative_delta.small }
			set_temp_variable = { utopia_manifesto_league_cohesion_delta = constant:utopia_manifesto_narrative_delta.negative_small }
			utopia_manifesto_apply_prepared_league_cohesion_delta = yes
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}

		limit = {
			check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.nowhere_route_propaganda }
			OR = {
				has_country_flag = utopia_manifesto_nowhere_law_route_propaganda
				NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_nowhere_choice_setup
			set_country_flag = utopia_manifesto_nowhere_law_route_propaganda
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.negative_small }

	if = {

			NOT = { utopia_manifesto_is_closed_island_route = yes }
			OR = {
				has_country_flag = utopia_manifesto_perfect_island_refuge_network
				NOT = { has_country_flag = utopia_manifesto_evolution_perfect_choice_setup }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_perfect_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_perfect_choice_setup
			set_country_flag = utopia_manifesto_perfect_island_refuge_network
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.large }

			limit = { NOT = { has_country_flag = utopia_manifesto_evolution_perfect_choice_setup } }
			set_country_flag = utopia_manifesto_evolution_perfect_choice_setup
			set_country_flag = utopia_manifesto_perfect_island_bounded_self_sufficiency

			set_country_flag = utopia_manifesto_evolution_perfect_choice_setup
			set_country_flag = utopia_manifesto_perfect_island_closed_order
			utopia_manifesto_apply_route_institution_final = yes
			set_temp_variable = { utopia_manifesto_plenty_delta = constant:utopia_manifesto_narrative_delta.medium }
			set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_narrative_delta.negative_small }
			set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_narrative_delta.large }
			utopia_manifesto_apply_prepared_ledger_delta = yes
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one }
	}

	set_temp_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.none }
}

# Consumes only an explicit country-scoped prepared token. Merely enabling an
# evolution in the prefire snapshot does not set this token and therefore does
# not skip staged delivery. The explicit path shares the setup dispatcher and
# records the corresponding stage only after setup succeeds.
utopia_manifesto_consume_one_prepared_prefire_evolution_choice = {
	if = {
		limit = {
			has_variable = utopia_manifesto_prefire_evolution_choice
			NOT = {
				AND = {
					utopia_manifesto_has_unresolved_route = yes
					OR = {
						check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.perfect_refuge_network }
						check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.perfect_bounded_self_sufficiency }
						check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.perfect_closed_order }
					}
				}
			}
		}
		set_temp_variable = { utopia_manifesto_evolution_choice_input = utopia_manifesto_prefire_evolution_choice }
		if = {
			limit = {
				OR = {
					AND = {
						utopia_manifesto_glosses_delivery_is_enabled = yes
						OR = {
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.glosses_open_schools }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.glosses_chartered_schools }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.glosses_censored_margins }
						}
					}
					AND = {
						utopia_manifesto_necessary_shores_delivery_is_enabled = yes
						OR = {
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.shores_domestic_alternative }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.shores_peaceful_priority }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.shores_emergency_powers }
						}
					}
					AND = {
						utopia_manifesto_cities_delivery_is_enabled = yes
						OR = {
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.cities_local_charters }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.cities_linked_standard }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.cities_foreign_circle }
						}
					}
					AND = {
						utopia_manifesto_nowhere_law_delivery_is_enabled = yes
						OR = {
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.nowhere_plural_law }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.nowhere_sponsored_law }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.nowhere_route_propaganda }
						}
					}
					AND = {
						utopia_manifesto_perfect_island_delivery_is_enabled = yes
						OR = {
							AND = { check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.perfect_refuge_network } NOT = { utopia_manifesto_is_closed_island_route = yes } }
							check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.perfect_bounded_self_sufficiency }
							AND = { check_variable = { utopia_manifesto_evolution_choice_input = constant:utopia_manifesto_evolution_choice.perfect_closed_order } utopia_manifesto_is_closed_island_route = yes }
						}
					}
				}
			}
			utopia_manifesto_apply_prepared_evolution_choice = yes
			if = {
				limit = { check_variable = { utopia_manifesto_evolution_choice_applied = constant:utopia_manifesto_math.one } }
				if = {
					limit = {
						OR = {
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.glosses_open_schools }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.glosses_chartered_schools }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.glosses_censored_margins }
						}
						NOT = { has_country_flag = utopia_manifesto_evolution_glosses_recorded }
					}
					utopia_manifesto_apply_glosses_in_the_margin_evolution = yes
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.shores_domestic_alternative }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.shores_peaceful_priority }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.shores_emergency_powers }
						}
						NOT = { has_country_flag = utopia_manifesto_evolution_necessary_shores_recorded }
					}
					utopia_manifesto_apply_necessary_shores_evolution = yes
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.cities_local_charters }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.cities_linked_standard }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.cities_foreign_circle }
						}
						NOT = { has_country_flag = utopia_manifesto_evolution_cities_recorded }
					}
					utopia_manifesto_apply_cities_of_one_measure_evolution = yes
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.nowhere_plural_law }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.nowhere_sponsored_law }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.nowhere_route_propaganda }
						}
						NOT = { has_country_flag = utopia_manifesto_evolution_nowhere_law_recorded }
					}
					utopia_manifesto_apply_nowhere_made_law_evolution = yes
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.perfect_refuge_network }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.perfect_bounded_self_sufficiency }
							check_variable = { utopia_manifesto_prefire_evolution_choice = constant:utopia_manifesto_evolution_choice.perfect_closed_order }
						}
						NOT = { has_country_flag = utopia_manifesto_evolution_perfect_island_recorded }
					}
					utopia_manifesto_apply_perfect_island_evolution = yes
				}
			}
		}
		else = { set_country_flag = utopia_manifesto_evolution_disabled_before_delivery }
		clear_variable = utopia_manifesto_prefire_evolution_choice
	}
}

# A scenario may prepare one explicit choice per accepted stage. Each token is
# consumed independently through the one-choice validator above, so multiple
# genuinely pre-delivered evolutions remain possible without treating enabled
# settings as choices. The generic token remains available for a one-off call.
utopia_manifesto_consume_prepared_prefire_evolution_choice = {
	if = {
		limit = { has_variable = utopia_manifesto_prefire_evolution_choice }
		utopia_manifesto_consume_one_prepared_prefire_evolution_choice = yes
	}
	if = {
		limit = { has_variable = utopia_manifesto_prefire_glosses_choice }
