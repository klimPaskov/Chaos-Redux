# ============================================================================
# CHAOS REDUX - EVENT 039 CXT TEST REGISTRATION
# ============================================================================
# Overview:
# - Registers the operations-kit equipment, five Assassin Forces formations,
#   and the support-only Saboteur Cell with the shared CXT extension bus.
# - The carrier is modifier-free and the setup effect is idempotent, so a
#   bounded startup scope and the tag-scoped daily repair can converge safely.
# - Uses grouped, combat-ready assassin and mechanized formations.
# ============================================================================

murder_mystery_register_cxt_test_content = {
	set_temp_variable = {
		var = chaosx_test_country_registration_extension_effect
		value = token:chaosx_cxt_extension_event039_assassin_forces
	}
	chaosx_test_country_register_extension_effect = yes
}

chaosx_cxt_extension_event039_assassin_forces_apply = {
	if = {
		limit = { NOT = { has_global_flag = chaosx_cxt_extension_event039_assassin_forces_registered } }
		set_temp_variable = { var = chaosx_test_country_current_unit_family value = token:chaosx_cxt_extension_event039_assassin_forces }
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:assassin_cadre }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:shadow_company }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:silent_guard }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:master_assassin }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_frontline_subunit value = token:mechanized_assassin }
		chaosx_test_country_register_frontline_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_support_subunit value = token:saboteur_cell }
		set_temp_variable = { var = chaosx_test_country_registration_support_anchor value = token:assassin_cadre }
		chaosx_test_country_register_support_subunit = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:assassin_operations_kit_1 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:infantry_equipment_1 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:support_equipment_1 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:motorized_equipment_1 }
		chaosx_test_country_register_equipment = yes
		set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:mechanized_equipment_1 }
		chaosx_test_country_register_equipment = yes
		set_global_flag = chaosx_cxt_extension_event039_assassin_forces_registered
	}
	chaosx_cxt_event039_define_formations = yes
	chaosx_cxt_event039_spawn_assassin_cadres_initial = yes
	if = {
		limit = { NOT = { has_country_flag = chaosx_cxt_assassin_formations_ready } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:assassin_cadre } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:assassin_cadre } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:shadow_company } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:shadow_company } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:silent_guard } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:silent_guard } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:master_assassin } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:master_assassin } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_frontline_subunits value = token:mechanized_assassin } } } add_to_array = { array = chaosx_test_country_processed_frontline_subunits value = token:mechanized_assassin } }
		if = { limit = { NOT = { is_in_array = { array = chaosx_test_country_processed_support_subunits value = token:saboteur_cell } } } add_to_array = { array = chaosx_test_country_processed_support_subunits value = token:saboteur_cell } }
		chaosx_cxt_event039_spawn_formations_initial = yes
		if = {
			limit = {
				AND = {
					is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:saboteur_cell }
					is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:assassin_cadre }
					is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:mechanized_assassin }
				}
			}
			set_country_flag = chaosx_cxt_assassin_formations_ready
		}
	}
}

				mechanized_assassin = { x = 0 y = 1 }

				mechanized_assassin = { x = 1 y = 0 }
				mechanized_assassin = { x = 1 y = 1 }
				mechanized_assassin = { x = 1 y = 2 }
			}
		}
	}
}


chaosx_cxt_event039_spawn_assassin_cadres_initial = {
	if = {
		limit = { has_template = "CXT - Assassin Cadres" NOT = { is_in_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:saboteur_cell } } }
		set_temp_variable = { var = chaosx_test_country_current_support_subunit value = 0 }
		set_temp_variable = { var = chaosx_test_country_current_family value = token:chaosx_cxt_extension_event039_assassin_forces }
		set_temp_variable = { var = chaosx_test_country_family_spawn_succeeded value = 0 }
		set_variable = { global.chaosx_test_country_spawn_template = token:cxt_template_assassin_cadres }
		chaosx_test_country_spawn_family_template = yes
		if = {
			limit = { check_variable = { chaosx_test_country_family_spawn_succeeded > 0 } }
			add_to_array = { array = global.chaosx_test_country_spawned_owner_formations value = token:saboteur_cell }
		}
	}
}
