# Global scope. Event 027 stores the last accepted evolution stage separately
# from the Chaos Meter. The first firing may use the highest stage already
# eligible in the campaign; later promotions are one enabled stage at a time
# after the centralized pacing interval has elapsed.
doctrine_research_get_highest_enabled_stage = {
	set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.baseline_calm_world }
	if = { limit = { doctrine_research_evolution_i_enabled = yes } set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.gathering_storm } }
	if = { limit = { doctrine_research_evolution_ii_enabled = yes } set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.rising_chaos } }
	if = { limit = { doctrine_research_evolution_iii_enabled = yes } set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.chaos_tier } }
	if = { limit = { doctrine_research_evolution_iv_enabled = yes } set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.totalen_chaos } }
}

doctrine_research_get_next_enabled_stage = {
	set_temp_variable = { doctrine_research_next_stage = global.doctrine_research_current_stage }
	if = {
		limit = {
			check_variable = { var = doctrine_research_next_stage value = global.doctrine_research_current_stage compare = equals }
			check_variable = { var = global.doctrine_research_current_stage value = constant:doctrine_research_stage.gathering_storm compare = less_than }
			doctrine_research_evolution_i_enabled = yes
		}
		set_temp_variable = { doctrine_research_next_stage = constant:doctrine_research_stage.gathering_storm }
	}
	if = {
		limit = {
			check_variable = { var = doctrine_research_next_stage value = global.doctrine_research_current_stage compare = equals }
			check_variable = { var = global.doctrine_research_current_stage value = constant:doctrine_research_stage.rising_chaos compare = less_than }
			doctrine_research_evolution_ii_enabled = yes
		}
		set_temp_variable = { doctrine_research_next_stage = constant:doctrine_research_stage.rising_chaos }
	}
	if = {
		limit = {
			check_variable = { var = doctrine_research_next_stage value = global.doctrine_research_current_stage compare = equals }
			check_variable = { var = global.doctrine_research_current_stage value = constant:doctrine_research_stage.chaos_tier compare = less_than }
			doctrine_research_evolution_iii_enabled = yes
		}
		set_temp_variable = { doctrine_research_next_stage = constant:doctrine_research_stage.chaos_tier }
	}
	if = {
		limit = {
			check_variable = { var = doctrine_research_next_stage value = global.doctrine_research_current_stage compare = equals }
			check_variable = { var = global.doctrine_research_current_stage value = constant:doctrine_research_stage.totalen_chaos compare = less_than }
			doctrine_research_evolution_iv_enabled = yes
		}
		set_temp_variable = { doctrine_research_next_stage = constant:doctrine_research_stage.totalen_chaos }
	}
}

# Global scope. Stores the next evolution date one pacing interval after today.
# The pacing interval is tuned in days and converted to date units, where one
# game hour is 0.00001, before it is added to the stored copy of global.date.
doctrine_research_schedule_next_evolution = {
	set_temp_variable = { chaosx_date_days = constant:doctrine_research_event.base_pacing_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_variable = { global.doctrine_research_next_evolution_date = global.date }
	add_to_variable = { global.doctrine_research_next_evolution_date = chaosx_date_offset }
}

doctrine_research_process_evolution = {
	if = { limit = { NOT = { has_variable = global.doctrine_research_current_stage } } set_variable = { global.doctrine_research_current_stage = constant:doctrine_research_stage.baseline_calm_world } }
	set_global_flag = doctrine_research_evolution_clock_started
	set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.baseline_calm_world }
	doctrine_research_get_highest_enabled_stage = yes
	if = {
		limit = {
			check_variable = {
				var = doctrine_research_target_stage
				value = global.doctrine_research_current_stage
				compare = less_than_or_equals
			}
		}
		clear_variable = global.doctrine_research_next_evolution_date
	}
	else_if = {
		limit = { NOT = { has_variable = global.doctrine_research_next_evolution_date } }
		doctrine_research_schedule_next_evolution = yes
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.date
				value = global.doctrine_research_next_evolution_date
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { doctrine_research_next_stage = global.doctrine_research_current_stage }
		doctrine_research_get_next_enabled_stage = yes
		if = {
			limit = {
				check_variable = {
					var = doctrine_research_next_stage
					value = global.doctrine_research_current_stage
					compare = greater_than
				}
			}
			set_variable = { global.doctrine_research_current_stage = doctrine_research_next_stage }
			clear_variable = global.doctrine_research_next_evolution_date
			doctrine_research_record_new_evolutions = yes
			if = {
				limit = {
					check_variable = {
						var = doctrine_research_target_stage
						value = global.doctrine_research_current_stage
						compare = greater_than
					}
				}
				doctrine_research_schedule_next_evolution = yes
			}
		}
		else = { clear_variable = global.doctrine_research_next_evolution_date }
	}
}

# Daily global-host pulse. Reads and writes only global state and never scans
# countries. Before the first firing the stage stays unset, so that firing can
# still open at the highest enabled stage; afterwards each eligible stage is
# promoted one pacing interval after it became eligible, one stage at a time.
# While Event 027 is disabled in the event settings no stage is promoted or
# recorded; the pulse resumes from the stored clock once it is enabled again.
doctrine_research_evolution_pulse = {
	if = {
		limit = {
			has_global_flag = doctrine_research_fired_once
			NOT = { is_in_array = { global.disabled_events = constant:doctrine_research_event.id } }
		}
		doctrine_research_process_evolution = yes
	}
}

doctrine_research_prepare_evolution_stage = {
	if = { limit = { NOT = { has_variable = global.doctrine_research_current_stage } } set_variable = { global.doctrine_research_current_stage = constant:doctrine_research_stage.baseline_calm_world } }
	set_temp_variable = { doctrine_research_target_stage = constant:doctrine_research_stage.baseline_calm_world }
	doctrine_research_get_highest_enabled_stage = yes
	if = {
		limit = {
			NOT = { has_global_flag = doctrine_research_evolution_clock_started }
			NOT = { has_global_flag = doctrine_research_fired_once }
		}
		set_variable = { global.doctrine_research_current_stage = doctrine_research_target_stage }
		set_global_flag = doctrine_research_evolution_clock_started
		clear_variable = global.doctrine_research_next_evolution_date
	}
	else = { doctrine_research_process_evolution = yes }
}

# Country scope. A ledger whose rows no longer line up, or whose receipts point
# at a missing batch, cannot prove which choices remain. Its unused choices and
# receipts are discarded without compensation, so the country takes part in
# later firings again. Doctrine progress already granted is never touched.
doctrine_research_repair_country_state = {
	doctrine_research_discard_annexed_country_state = yes
	set_variable = { doctrine_research_last_transaction_result = constant:doctrine_research_transaction_result.none }

# Country scope. Choice receipts are intentionally a separate append-only
# ledger: one batch owns one-to-five receipts, while a receipt belongs to one
# batch. Validate that foreign keys still point at a retained batch row without
# forcing the two ledgers to have the same cardinality.
doctrine_research_validate_receipt_parent_links = {
	set_temp_variable = { doctrine_research_receipt_parent_validation_failed = 0 }
	if = {
		limit = { check_variable = { doctrine_research_receipt_ids^num > 0 } }
		for_loop_effect = {
			start = constant:doctrine_research_event.first_index
			end = doctrine_research_receipt_ids^num
			value = doctrine_research_receipt_parent_receipt_index
			set_temp_variable = { doctrine_research_receipt_parent_found = 0 }
			for_loop_effect = {
				start = constant:doctrine_research_event.first_index
				end = doctrine_research_batch_ids^num
				value = doctrine_research_receipt_parent_batch_index
				break = doctrine_research_receipt_parent_found
				if = {
					limit = {
						check_variable = { doctrine_research_receipt_parent_found < 1 }
						check_variable = {
							var = doctrine_research_receipt_batch_ids^doctrine_research_receipt_parent_receipt_index
							value = doctrine_research_batch_ids^doctrine_research_receipt_parent_batch_index
							compare = equals
						}
					}
					set_temp_variable = { doctrine_research_receipt_parent_found = 1 }
				}
			}
			if = {
				limit = { check_variable = { doctrine_research_receipt_parent_found < 1 } }
				set_temp_variable = { doctrine_research_receipt_parent_validation_failed = 1 }
			}
		}
	}
	if = {
		limit = { check_variable = { doctrine_research_receipt_parent_validation_failed > 0 } }
		set_country_flag = doctrine_research_state_corrupt
	}
}

	# A damaged ledger is repaired in place; a country whose ledger cannot be
	# repaired is the only country left out of this firing.
	every_country = {
		limit = { doctrine_research_country_has_valid_action = yes }
		doctrine_research_initialize_country_state = yes
		if = {
			limit = { NOT = { has_country_flag = doctrine_research_state_corrupt } }
			doctrine_research_validate_receipt_parent_links = yes
		}
		if = {
			limit = { has_country_flag = doctrine_research_state_corrupt }
			doctrine_research_repair_country_state = yes
		}
		if = {
			limit = { NOT = { has_country_flag = doctrine_research_state_corrupt } }
			add_to_temp_array = { doctrine_research_firing_recipients = THIS }
		}
	}
	# Every append is isolated. A country that cannot take its row is simply
	# absent from the published list; no other country's row is touched.
	for_each_scope_loop = {
		array = doctrine_research_firing_recipients
		doctrine_research_append_country_batch = yes
	}
	clear_temp_array = doctrine_research_firing_recipients
	set_global_flag = doctrine_research_fired_once
	clear_variable = global.doctrine_research_firing_stage
	clear_variable = global.doctrine_research_firing_size
	if = {
		limit = { check_variable = { doctrine_research_firing_published_count > 0 } }

# Country scope, called only while no batch is active. Achievement checks and
# the batch summary read the active or just-closed batch, so every older
# completed or quarantined row is removed together with its receipts. The most
# recent closed row and all queued rows remain. Rows are removed back to front
# so lower indices stay valid, which keeps every scan bounded by the queue.
doctrine_research_compact_closed_batches = {
	set_temp_variable = { doctrine_research_compact_keep_index = constant:doctrine_research_event.invalid_index }
	set_temp_variable = { doctrine_research_compact_scan_break = 0 }
	for_loop_effect = {
		start = constant:doctrine_research_event.first_index
		end = doctrine_research_batch_ids^num
		value = doctrine_research_compact_scan_index
		break = doctrine_research_compact_scan_break
		if = {
			limit = {
				OR = {
					check_variable = { var = doctrine_research_batch_statuses^doctrine_research_compact_scan_index value = constant:doctrine_research_batch_status.completed compare = equals }
					check_variable = { var = doctrine_research_batch_statuses^doctrine_research_compact_scan_index value = constant:doctrine_research_batch_status.quarantined compare = equals }
				}
			}
			set_temp_variable = { doctrine_research_compact_keep_index = doctrine_research_compact_scan_index }
		}
		# FIFO closure forms a closed prefix followed by queued rows. Keep the
		# unvisited suffix untouched rather than repeatedly scanning its rows.
		else_if = {
			limit = { check_variable = { var = doctrine_research_batch_statuses^doctrine_research_compact_scan_index value = constant:doctrine_research_batch_status.queued compare = equals } }
			set_temp_variable = { doctrine_research_compact_scan_break = 1 }
		}
	}
	set_temp_variable = { doctrine_research_compact_batch_start = doctrine_research_compact_keep_index }
	subtract_from_temp_variable = { doctrine_research_compact_batch_start = 1 }
	for_loop_effect = {
		start = doctrine_research_compact_batch_start
		end = constant:doctrine_research_event.invalid_index
		compare = greater_than
		add = -1
		value = doctrine_research_compact_batch_index
		if = {
			limit = {
				OR = {
					check_variable = { var = doctrine_research_batch_statuses^doctrine_research_compact_batch_index value = constant:doctrine_research_batch_status.completed compare = equals }
					check_variable = { var = doctrine_research_batch_statuses^doctrine_research_compact_batch_index value = constant:doctrine_research_batch_status.quarantined compare = equals }
				}
			}
			set_temp_variable = { doctrine_research_compact_batch_id = doctrine_research_batch_ids^doctrine_research_compact_batch_index }
			set_temp_variable = { doctrine_research_compact_receipt_start = doctrine_research_receipt_ids^num }
			subtract_from_temp_variable = { doctrine_research_compact_receipt_start = 1 }
			for_loop_effect = {
				start = doctrine_research_compact_receipt_start
				end = constant:doctrine_research_event.invalid_index
				compare = greater_than
				add = -1
				value = doctrine_research_compact_receipt_index
				if = {
					limit = { check_variable = { var = doctrine_research_receipt_batch_ids^doctrine_research_compact_receipt_index value = doctrine_research_compact_batch_id compare = equals } }
					remove_from_array = { array = doctrine_research_receipt_ids index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_batch_ids index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_choice_numbers index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_states index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_actions index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_domains index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_grands index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_tracks index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_track_indices index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_subdoctrines index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_pre_levels index = doctrine_research_compact_receipt_index }
					remove_from_array = { array = doctrine_research_receipt_post_levels index = doctrine_research_compact_receipt_index }
				}
			}
			remove_from_array = { array = doctrine_research_batch_ids index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_firing_ids index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_stages index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_sizes index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_remaining_choices index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_statuses index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_created_dates index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_human_started index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_start_empty_army index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_start_empty_navy index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_start_empty_air index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_start_empty_special_forces index = doctrine_research_compact_batch_index }
			remove_from_array = { array = doctrine_research_batch_start_empty_chaos_warfare index = doctrine_research_compact_batch_index }
		}
	}
}

doctrine_research_start_next_batch = {
	doctrine_research_initialize_country_state = yes
	if = {
		limit = {
			NOT = { has_country_flag = doctrine_research_active_batch }
			NOT = { has_country_flag = doctrine_research_state_corrupt }
			check_variable = { doctrine_research_queued_batches > 0 }
			doctrine_research_country_is_live = yes
		}
		# Closed rows are compacted before the next batch opens, so the queue
		# scan below only walks the kept closed row and the queued rows.
		doctrine_research_compact_closed_batches = yes
		set_variable = { doctrine_research_active_batch_index = constant:doctrine_research_event.invalid_index }
		set_temp_variable = { doctrine_research_batch_scan_break = 0 }
		for_loop_effect = {
			start = constant:doctrine_research_event.first_index
			end = doctrine_research_batch_ids^num
			value = doctrine_research_batch_scan_index
			break = doctrine_research_batch_scan_break
			if = { limit = { check_variable = { var = doctrine_research_batch_statuses^doctrine_research_batch_scan_index value = constant:doctrine_research_batch_status.queued compare = equals } }
				set_variable = { doctrine_research_active_batch_index = doctrine_research_batch_scan_index }
				set_variable = { doctrine_research_batch_statuses^doctrine_research_batch_scan_index = constant:doctrine_research_batch_status.active }
				subtract_from_variable = { doctrine_research_queued_batches = 1 }
				set_temp_variable = { doctrine_research_batch_scan_break = 1 }
			}
		}
		if = { limit = { check_variable = { doctrine_research_active_batch_index > constant:doctrine_research_event.invalid_index } }
			set_variable = { doctrine_research_active_batch_id = doctrine_research_batch_ids^doctrine_research_active_batch_index }
			set_variable = { doctrine_research_active_batch_stage = doctrine_research_batch_stages^doctrine_research_active_batch_index }
			set_variable = { doctrine_research_active_batch_size = doctrine_research_batch_sizes^doctrine_research_active_batch_index }
			set_variable = { doctrine_research_current_choice_number = 1 }
			set_variable = { doctrine_research_active_batch_successful_adoptions = 0 }
			set_variable = { doctrine_research_active_batch_successful_mastery = 0 }
			set_variable = { doctrine_research_summary_milestone_reached = 0 }
			set_country_flag = doctrine_research_active_batch
			doctrine_research_clear_selection = yes
			if = {
				limit = { is_ai = yes }
				clr_country_flag = doctrine_research_active_batch_prompted_human
				doctrine_research_ai_resolve_active_batch = yes
			}
			else = {
				set_country_flag = doctrine_research_active_batch_prompted_human
				if = { limit = { doctrine_research_country_has_valid_action = yes } country_event = { id = chaosx.nr27.3 } }
				else = { country_event = { id = chaosx.nr27.11 } }
			}
		}
		else = {
			# A bounded scan can stop before a long retained closed prefix ends.
			# Compaction made progress, but that does not prove the queue empty.
			# Preserve the saved count and every untouched row for the next pass.
			doctrine_research_schedule_country_batch_wake = yes
		}
	}
}

# Country scope. Lifecycle callbacks call this only for the exact country
# affected by the engine callback. The ledger remains country-owned across
# controller, subject, government, and faction changes; only the unresolved
# presentation or AI resolver is resumed when a queued or active batch exists.
doctrine_research_reconcile_country_lifecycle = {
	# Countries that never received a batch, or have nothing active or queued,
	# return here without touching any ledger array.
	if = {
		limit = {
			has_variable = doctrine_research_schema_version
			OR = {
				has_country_flag = doctrine_research_active_batch
				has_country_flag = doctrine_research_state_corrupt
				check_variable = { doctrine_research_queued_batches > 0 }
				NOT = { has_variable = doctrine_research_queued_batches }
			}
		}
		doctrine_research_initialize_country_state = yes
		if = {
			limit = {
				doctrine_research_country_is_live = yes
				NOT = { has_country_flag = doctrine_research_state_corrupt }
			}
			if = {
				limit = { has_country_flag = doctrine_research_active_batch }
				if = {
					limit = { is_ai = yes }
					clr_country_flag = doctrine_research_active_batch_prompted_human
					doctrine_research_ai_resolve_active_batch = yes
				}
				# A human page is already open whenever the prompted flag is set.
				# Only a batch without an open page is shown again, once.
				else_if = {
					limit = { NOT = { has_country_flag = doctrine_research_active_batch_prompted_human } }
					set_country_flag = doctrine_research_active_batch_prompted_human
					if = {
						limit = { check_variable = { var = doctrine_research_last_transaction_result value = constant:doctrine_research_transaction_result.adoption_only compare = equals } }
						country_event = { id = chaosx.nr27.13 }
					}
					else_if = { limit = { doctrine_research_country_has_valid_action = yes } doctrine_research_clear_selection = yes country_event = { id = chaosx.nr27.3 } }
					else = { country_event = { id = chaosx.nr27.11 } }
				}
			}
			else = { doctrine_research_start_next_batch = yes }
		}
	}
}

# Country scope. Scores each domain that has a valid action. Every term reads
# capability that belongs to that domain, so war, production, geography, and
# alliance pressure separate Army, Navy, Air, Special Forces, and Chaos Warfare
# instead of raising them all equally:
# - the industry focus term goes to Navy when dockyards are a large share of
#   war industry, to Air when deployed planes are large for the army's size,
#   and to Army otherwise;
# - Army takes the war and geography terms for a land front, while Navy takes
#   the war term only for a war without one and the geography term only for a
#   coastal capital; an air-led power's Air also takes the land-front term;
# - Special Forces readiness, war, production, and alliance terms require
#   templates that field Special Forces battalions.
# Navy without a coast, dockyards, ships, convoys, or naval doctrine scores
# zero, unless a landlocked country has a maritime plan (a core, claim, or war
# goal on the coast), which gives it the base and strategy weights. When no
# capability-weighted domain scores at all, every valid domain falls back to
# the base weight so the choice is never wasted.

		if = { limit = { has_war = yes num_divisions > 0 } add_to_temp_variable = { doctrine_research_ai_score_army = constant:doctrine_research_ai_weight.war } }
		if = { limit = { doctrine_research_ai_has_land_front = yes } add_to_temp_variable = { doctrine_research_ai_score_army = constant:doctrine_research_ai_weight.geography } }
		if = { limit = { is_in_faction = yes num_divisions > 0 } add_to_temp_variable = { doctrine_research_ai_score_army = constant:doctrine_research_ai_weight.theater } }

doctrine_research_ai_resolve_active_batch = {
	set_temp_variable = { doctrine_research_transaction_success = 0 }
	set_temp_variable = { doctrine_research_ai_choice_guard = 0 }
	set_temp_variable = { doctrine_research_ai_batch_id = doctrine_research_active_batch_id }
	while_loop_effect = {
		limit = { has_country_flag = doctrine_research_active_batch check_variable = { var = doctrine_research_active_batch_id value = doctrine_research_ai_batch_id compare = equals } check_variable = { var = doctrine_research_batch_remaining_choices^doctrine_research_active_batch_index value = 0 compare = greater_than } check_variable = { doctrine_research_ai_choice_guard < constant:doctrine_research_event.max_batch_choices } doctrine_research_country_has_valid_action = yes }
	doctrine_research_ai_pick_choice = yes
		if = {
			limit = { doctrine_research_selected_choice_is_valid = yes }
			doctrine_research_confirm_selection = yes
			if = { limit = { check_variable = { doctrine_research_transaction_success > 0 } } add_to_temp_variable = { doctrine_research_ai_choice_guard = 1 } }
			else_if = {
				limit = { check_variable = { var = doctrine_research_last_transaction_result value = constant:doctrine_research_transaction_result.adoption_only compare = equals } }
				doctrine_research_clear_selection = yes
				set_variable = { doctrine_research_last_transaction_result = constant:doctrine_research_transaction_result.none }
			}
			else = { doctrine_research_quarantine_active_batch_and_start_next = yes }
		}
		else = { doctrine_research_quarantine_active_batch_and_start_next = yes }
	}
	if = {
		limit = { has_country_flag = doctrine_research_active_batch check_variable = { var = doctrine_research_active_batch_id value = doctrine_research_ai_batch_id compare = equals } }
		if = {
			limit = {
				OR = {
					check_variable = { var = doctrine_research_batch_remaining_choices^doctrine_research_active_batch_index value = 0 compare = equals }
					NOT = { doctrine_research_country_has_valid_action = yes }
				}
			}
			if = {
				limit = { check_variable = { doctrine_research_ai_summary_page > 0 } }
				# The summary page needs a finished curriculum; unspendable
				# choices go to the no-action page, which closes it.
				if = {
					limit = { check_variable = { var = doctrine_research_batch_remaining_choices^doctrine_research_active_batch_index value = 0 compare = equals } }
					country_event = { id = chaosx.nr27.10 }
				}
				else = { country_event = { id = chaosx.nr27.11 } }
			}
			else = { doctrine_research_complete_active_batch = yes }
		}
		else = { doctrine_research_quarantine_active_batch_and_start_next = yes }
	}
}
