# Expects any positive temporary inputs named below. Every supplied value is
# negated before payment so callers never use unary minus on a variable token.
# All inputs are consumed and reset before this effect returns.
# Inputs: utopia_manifesto_cost_{manpower,infantry,support,motorized,trains,
# convoys,army_xp,command_power,stability,war_support}.
utopia_manifesto_pay_prepared_decision_cost = {
	if = {
		limit = { has_variable = utopia_manifesto_cost_manpower }
		multiply_temp_variable = { utopia_manifesto_cost_manpower = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_manpower = utopia_manifesto_cost_manpower
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_infantry }
		multiply_temp_variable = { utopia_manifesto_cost_infantry = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = utopia_manifesto_cost_infantry
		}
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_support }
		multiply_temp_variable = { utopia_manifesto_cost_support = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = utopia_manifesto_cost_support
		}
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_motorized }
		multiply_temp_variable = { utopia_manifesto_cost_motorized = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_equipment_to_stockpile = {
			type = motorized_equipment
			amount = utopia_manifesto_cost_motorized
		}
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_trains }
		multiply_temp_variable = { utopia_manifesto_cost_trains = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_equipment_to_stockpile = {
			type = train_equipment
			amount = utopia_manifesto_cost_trains
		}
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_convoys }
		multiply_temp_variable = { utopia_manifesto_cost_convoys = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_equipment_to_stockpile = {
			type = convoy_1
			amount = utopia_manifesto_cost_convoys
		}
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_army_xp }
		multiply_temp_variable = { utopia_manifesto_cost_army_xp = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_army_experience = utopia_manifesto_cost_army_xp
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_command_power }
		multiply_temp_variable = { utopia_manifesto_cost_command_power = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_command_power = utopia_manifesto_cost_command_power
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_stability }
		multiply_temp_variable = { utopia_manifesto_cost_stability = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_stability = utopia_manifesto_cost_stability
	}
	if = {
		limit = { has_variable = utopia_manifesto_cost_war_support }
		multiply_temp_variable = { utopia_manifesto_cost_war_support = constant:utopia_manifesto_decision_value.cost_negation_multiplier }
		add_war_support = utopia_manifesto_cost_war_support
	}
	set_temp_variable = { utopia_manifesto_cost_manpower = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_infantry = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_support = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_motorized = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_trains = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_convoys = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_army_xp = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_command_power = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_stability = constant:utopia_manifesto_decision_value.none }
	set_temp_variable = { utopia_manifesto_cost_war_support = constant:utopia_manifesto_decision_value.none }
}

# Inputs: utopia_manifesto_duration_base_input and
# utopia_manifesto_duration_maximum_input. Output: temporary
# utopia_manifesto_duration_result. Duration grows with war and country size,
# and falls when the Ledger can credibly support administration.
utopia_manifesto_prepare_dynamic_decision_duration = {
	set_temp_variable = {
		utopia_manifesto_duration_result = utopia_manifesto_duration_base_input
	}
	if = {
		limit = { has_war = yes }
		add_to_temp_variable = {
			utopia_manifesto_duration_result = constant:utopia_manifesto_decision_duration.war_addition
		}
	}
	if = {
		limit = { num_of_owned_states > constant:utopia_manifesto_decision_threshold.large_country_states }
		add_to_temp_variable = {
			utopia_manifesto_duration_result = constant:utopia_manifesto_decision_duration.large_country_addition
		}
	}
	if = {
		limit = { num_of_owned_states > constant:utopia_manifesto_decision_threshold.very_large_country_states }
		add_to_temp_variable = {
			utopia_manifesto_duration_result = constant:utopia_manifesto_decision_duration.very_large_country_addition
		}
	}
	if = {
		limit = { utopia_manifesto_plenty_is_stable = yes }
		subtract_from_temp_variable = {
			utopia_manifesto_duration_result = constant:utopia_manifesto_decision_duration.stable_plenty_reduction
		}
	}
	if = {
		limit = { utopia_manifesto_concord_is_trusted = yes }
		subtract_from_temp_variable = {
			utopia_manifesto_duration_result = constant:utopia_manifesto_decision_duration.trusted_concord_reduction
		}
	}
	clamp_temp_variable = {
		var = utopia_manifesto_duration_result
		min = constant:utopia_manifesto_decision_duration.minimum
		max = utopia_manifesto_duration_maximum_input
	}
}

utopia_manifesto_resolve_unpopular_calling = {
	clr_country_flag = utopia_manifesto_calling_mission_active
	if = {
		limit = {
			OR = {
				AND = {
					check_variable = { utopia_manifesto_selected_calling_method = constant:utopia_manifesto_calling_method.open_call }
					utopia_manifesto_concord_is_cooperative = yes
					utopia_manifesto_plenty_is_stable = yes
				}
				AND = {
					check_variable = { utopia_manifesto_selected_calling_method = constant:utopia_manifesto_calling_method.guaranteed_placement }
					utopia_manifesto_plenty_is_stable = yes
				}
				AND = {
					check_variable = { utopia_manifesto_selected_calling_method = constant:utopia_manifesto_calling_method.assignment_quota }
					utopia_manifesto_assignment_is_planned = yes
				}
				check_variable = { utopia_manifesto_selected_calling_method = constant:utopia_manifesto_calling_method.emergency_levy }
			}
		}
		utopia_manifesto_clear_selected_calling_shortage = yes
		set_country_flag = utopia_manifesto_calling_mission_succeeded
		set_temp_variable = { utopia_manifesto_need_delta = constant:utopia_manifesto_decision_change.ledger_step_loss }
		set_temp_variable = { utopia_manifesto_plenty_delta = constant:utopia_manifesto_decision_change.ledger_nudge }
		set_temp_variable = { utopia_manifesto_concord_delta = constant:utopia_manifesto_decision_change.ledger_nudge }
		set_temp_variable = { utopia_manifesto_assignment_delta = constant:utopia_manifesto_decision_value.none }
		if = {
			limit = {
				OR = {
					check_variable = { utopia_manifesto_selected_calling_method = constant:utopia_manifesto_calling_method.assignment_quota }
					check_variable = { utopia_manifesto_selected_calling_method = constant:utopia_manifesto_calling_method.emergency_levy }
