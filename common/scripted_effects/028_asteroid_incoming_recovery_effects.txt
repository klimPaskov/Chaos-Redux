# ============================================================================
# CHAOS REDUX - EVENT 028 ASTEROID INCOMING RECOVERY EFFECTS
# ============================================================================
# State-targeted recovery projects, mission starts and mission outcomes for
# Event 028. Every repair reads the state's pre-impact snapshot, restores only
# a bounded fraction of the observed gap, and refuses the permanent main crater.
# One paid project can be pending per country; each mission keeps its own state
# pointers and hold date, so projects continue while missions run. These
# helpers do not restore population and do not create nuclear, fallout, or
# generic world-scan side effects.
# ============================================================================

# STATE SCOPE. Inputs are the pre-impact receipt variables captured by the
# physical transaction and asteroid_incoming_restore_fraction as a temporary
# fixed-point value. Each explicit building call keeps the construction type
# static so a malformed token can never create an unintended building.
asteroid_incoming_restore_infrastructure = {
	if = {
		limit = {
			NOT = { has_state_flag = asteroid_incoming_main_crater }
			has_variable = asteroid_incoming_preimpact_infrastructure
		}
		set_temp_variable = { asteroid_incoming_restore_levels = asteroid_incoming_preimpact_infrastructure }
		subtract_from_temp_variable = { asteroid_incoming_restore_levels = building_level@infrastructure }
		clamp_temp_variable = {
			var = asteroid_incoming_restore_levels
			min = constant:asteroid_incoming_dust.minimum
			max = asteroid_incoming_preimpact_infrastructure
		}
		multiply_temp_variable = { asteroid_incoming_restore_levels = asteroid_incoming_restore_fraction }
		round_temp_variable = asteroid_incoming_restore_levels
		if = {
			limit = { check_variable = { asteroid_incoming_restore_levels > constant:asteroid_incoming_dust.minimum } }
			add_building_construction = {
				type = infrastructure
				level = asteroid_incoming_restore_levels
				instant_build = yes
			}
		}
	}
}

# STATE SCOPE. Finish one local emergency project and shorten every Event 028
# timed burden on the selected state. Persistent crater identities remain.
asteroid_incoming_close_state_emergency = {
	remove_dynamic_modifier = { modifier = asteroid_incoming_catastrophic_impact_zone }
	remove_dynamic_modifier = { modifier = asteroid_incoming_major_impact_zone }
	remove_dynamic_modifier = { modifier = asteroid_incoming_outer_shock_zone }
	remove_dynamic_modifier = { modifier = asteroid_incoming_heavy_fragment_damage }
	remove_dynamic_modifier = { modifier = asteroid_incoming_light_fragment_damage }
	set_state_flag = asteroid_incoming_local_emergency_resolved
}

# STATE SCOPE. At impact, a non-crater state with a live local emergency joins
# the bounded monthly closure list; a state with only outer-shock or
# light-fragment damage starts with its emergency already resolved.
asteroid_incoming_register_local_emergency = {
	if = {
		limit = { NOT = { has_state_flag = asteroid_incoming_main_crater } }
		if = {
			limit = { asteroid_incoming_state_has_active_local_emergency_burden = yes }
			if = {
				limit = { NOT = { is_in_array = { array = global.asteroid_incoming_emergency_states value = THIS } } }
				add_to_array = { array = global.asteroid_incoming_emergency_states value = THIS }
			}
		}
		else = { set_state_flag = asteroid_incoming_local_emergency_resolved }
	}
}

# GLOBAL-HOST SCOPE. Closes each listed local emergency once its rescue burden
# and every emergency-class impact modifier have ended, which opens
# reconstruction for that state. Closed states leave the list.
asteroid_incoming_process_emergency_closures = {
	clear_array = global.asteroid_incoming_emergency_states_next
	for_each_scope_loop = {
		array = global.asteroid_incoming_emergency_states
		if = {
			limit = {
				NOT = { has_state_flag = asteroid_incoming_local_emergency_resolved }
				asteroid_incoming_state_has_active_local_emergency_burden = yes
			}
			add_to_array = { array = global.asteroid_incoming_emergency_states_next value = THIS }
		}
		else = { set_state_flag = asteroid_incoming_local_emergency_resolved }
	}
	clear_array = global.asteroid_incoming_emergency_states
	for_each_scope_loop = {
		array = global.asteroid_incoming_emergency_states_next
		add_to_array = { array = global.asteroid_incoming_emergency_states value = THIS }
	}
	clear_array = global.asteroid_incoming_emergency_states_next
}

# COUNTRY SCOPE. Bounded global mitigation from an accepted action.
asteroid_incoming_add_mitigation = {
	if = { limit = { NOT = { has_variable = global.asteroid_incoming_mitigation_points } } set_variable = { global.asteroid_incoming_mitigation_points = 0 } }
	add_to_variable = { global.asteroid_incoming_mitigation_points = 1 }
	clamp_variable = {
		var = global.asteroid_incoming_mitigation_points
		min = 0
		max = constant:asteroid_incoming_dust.maximum_mitigation_decay
	}
}

# STATE SCOPE. Fixes the public repair targets for recovery missions from the
# pre-impact snapshot. Each target is a constant share of the level the state
# had before the impact, rounded to whole levels, so a building the state never
# had is never required. Called once, when the snapshot is recorded.
asteroid_incoming_set_mission_repair_targets = {
	set_variable = { asteroid_incoming_mission_infrastructure_target = asteroid_incoming_preimpact_infrastructure }
	multiply_variable = { asteroid_incoming_mission_infrastructure_target = constant:asteroid_incoming_recovery.mission_repair_share }
	round_variable = asteroid_incoming_mission_infrastructure_target
	set_variable = { asteroid_incoming_mission_railway_target = asteroid_incoming_preimpact_railway }
	multiply_variable = { asteroid_incoming_mission_railway_target = constant:asteroid_incoming_recovery.mission_repair_share }
	round_variable = asteroid_incoming_mission_railway_target
	set_variable = { asteroid_incoming_mission_supply_node_target = asteroid_incoming_preimpact_supply_node }
	multiply_variable = { asteroid_incoming_mission_supply_node_target = constant:asteroid_incoming_recovery.mission_repair_share }
	round_variable = asteroid_incoming_mission_supply_node_target
	set_variable = { asteroid_incoming_mission_civilian_factory_target = asteroid_incoming_preimpact_industrial_complex }
	multiply_variable = { asteroid_incoming_mission_civilian_factory_target = constant:asteroid_incoming_recovery.mission_repair_share }
	round_variable = asteroid_incoming_mission_civilian_factory_target
	set_variable = { asteroid_incoming_mission_military_factory_target = asteroid_incoming_preimpact_arms_factory }
	multiply_variable = { asteroid_incoming_mission_military_factory_target = constant:asteroid_incoming_recovery.mission_repair_share }
	round_variable = asteroid_incoming_mission_military_factory_target
}

# STATE SCOPE. Before any damage or capital move, records whether this state is
# on its controller's rail network. Recovery missions later ask only for rail
# links that existed before the impact, so an island or a state that never had
# a railway is never asked to reach the capital by rail.
asteroid_incoming_record_preimpact_rail_link = {
	if = {
		limit = { NOT = { has_state_flag = asteroid_incoming_preimpact_snapshot_recorded } }
		clr_state_flag = asteroid_incoming_preimpact_rail_linked
		if = {
			limit = {
				CONTROLLER = { exists = yes }
				OR = {
					AND = {
						is_capital = yes
						is_controlled_by = OWNER
					}
					CONTROLLER = {
						has_railway_connection = {
							start_state = THIS.capital
							target_state = PREV
						}
					}
				}
			}
			set_state_flag = asteroid_incoming_preimpact_rail_linked
		}
	}
}

# STATE SCOPE, decision actor in ROOT. Called at payment time after the
# exact action and affordability gates pass. No defaults or global targets.
# Output: ROOT's persistent project pointer and pending-project reservation.
asteroid_incoming_save_recovery_target = {
	if = {
		limit = {
			asteroid_incoming_decision_target_state_is_valid = yes
			ROOT = { exists = yes asteroid_incoming_recovery_project_slot_is_free = yes }
		}
		ROOT = {
			set_variable = { asteroid_incoming_recovery_target_state = PREV }
			set_country_flag = asteroid_incoming_recovery_project_pending
		}
	}
}

# COUNTRY SCOPE. Lifecycle entry for annexation. Removes the five native
# missions without outcome callbacks, then releases every mission contract and
# the pending project. Pending decision rows cancel from the lost reservation.
asteroid_incoming_cancel_recovery_runtime = {
	remove_mission = asteroid_incoming_rescue_route_mission
	remove_mission = asteroid_incoming_stabilize_damaged_zone_mission
	remove_mission = asteroid_incoming_reconnect_national_network_mission
	remove_mission = asteroid_incoming_restore_outer_ring_mission
	remove_mission = asteroid_incoming_rebuild_regional_economy_mission
	asteroid_incoming_clear_rescue_mission = yes
	asteroid_incoming_clear_stabilize_mission = yes
	asteroid_incoming_clear_network_mission = yes
	asteroid_incoming_clear_outer_mission = yes
	asteroid_incoming_clear_economy_mission = yes
	asteroid_incoming_clear_recovery_target = yes
}

# COUNTRY SCOPE, original decision state in FROM. Revalidates the reservation
# before dispatching one paid project, then always releases the project. A
# started mission keeps its own pointer. Repeated terminal callbacks cannot
# grant work without a pending receipt.
asteroid_incoming_finish_recovery_project = {
	if = {
		limit = { asteroid_incoming_recovery_project_target_is_valid = yes }
		clr_country_flag = asteroid_incoming_recovery_project_pending
		if = {
			limit = { has_country_flag = asteroid_incoming_mobile_hospitals_project_active }
			FROM = { asteroid_incoming_deploy_mobile_hospitals = yes }
		}
		else_if = {
			limit = { has_country_flag = asteroid_incoming_rail_corridor_project_active }
			FROM = { asteroid_incoming_open_emergency_rail_corridor = yes }
		}
		else_if = {
			limit = { has_country_flag = asteroid_incoming_debris_project_active }
			FROM = { asteroid_incoming_clear_unstable_debris = yes }
		}
		else_if = {
			limit = { has_country_flag = asteroid_incoming_filters_project_active }
			FROM = { asteroid_incoming_distribute_emergency_filters = yes }
		}
		else_if = {
			limit = { has_country_flag = asteroid_incoming_supply_spine_project_active }
			FROM = { asteroid_incoming_rebuild_supply_spine = yes }
		}
		else_if = {
			limit = { has_country_flag = asteroid_incoming_outer_industry_project_active }
			FROM = { asteroid_incoming_restore_outer_ring_industry = yes }
		}
		else_if = {
			limit = { has_country_flag = asteroid_incoming_rehousing_project_active }
			FROM = { asteroid_incoming_rehouse_displaced_workers = yes }
		}
	}
	asteroid_incoming_clear_recovery_target = yes
}

# COUNTRY SCOPE. Input: temporary asteroid_incoming_mission_hold_days, set by
# the caller to the mission's native timeout. Outputs, which the caller sets to
# zero before the call: temporary asteroid_incoming_mission_complete_day (the
# first day the objective can be met) and asteroid_incoming_mission_ready_date
# (the same moment as a date for the mission text). The hold period ends a
# constant grace window before the native timeout.
asteroid_incoming_quote_mission_hold = {
	subtract_from_temp_variable = { asteroid_incoming_mission_hold_days = constant:asteroid_incoming_recovery.mission_grace_days }
	set_temp_variable = { asteroid_incoming_mission_complete_day = global.num_days }
	add_to_temp_variable = { asteroid_incoming_mission_complete_day = asteroid_incoming_mission_hold_days }
	set_temp_variable = { chaosx_date_days = asteroid_incoming_mission_hold_days }
	set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
	chaosx_convert_days_to_date_offset = yes
	set_temp_variable = { asteroid_incoming_mission_ready_date = global.date }
	add_to_temp_variable = { asteroid_incoming_mission_ready_date = chaosx_date_offset }
}

# STATE SCOPE, acting country in ROOT. Starts Keep the Rescue Route Open on
# this state when that mission is not running and a mission slot is free.
asteroid_incoming_try_start_rescue_route_mission = {
	if = {
		limit = {
			asteroid_incoming_decision_target_state_is_valid = yes
			ROOT = {
				NOT = { has_active_mission = asteroid_incoming_rescue_route_mission }
				asteroid_incoming_country_can_start_recovery_mission = yes
			}
		}
		ROOT = {
			asteroid_incoming_clear_rescue_mission = yes
			set_variable = { asteroid_incoming_rescue_mission_state = PREV }
			set_temp_variable = { asteroid_incoming_mission_hold_days = constant:asteroid_incoming_recovery.mission_rescue_route_days }
			set_temp_variable = { asteroid_incoming_mission_complete_day = constant:asteroid_incoming_runtime.zero }
			set_temp_variable = { asteroid_incoming_mission_ready_date = constant:asteroid_incoming_runtime.zero }
			asteroid_incoming_quote_mission_hold = yes
			set_variable = { asteroid_incoming_rescue_mission_complete_day = asteroid_incoming_mission_complete_day }
			set_variable = { asteroid_incoming_rescue_mission_ready_date = asteroid_incoming_mission_ready_date }
			set_country_flag = asteroid_incoming_rescue_route_mission_ready
			activate_mission = asteroid_incoming_rescue_route_mission
		}
	}
}

# STATE SCOPE, acting country in ROOT. Starts Stabilize the Damaged Zone on the
# two worst-hit states the country holds that still fall short of their repair
# targets. Severity is the strongest impact profile first, then the larger
# infrastructure and railway gap. One qualifying state names only that state;
# none starts no mission.
asteroid_incoming_try_start_stabilize_mission = {
	if = {
		limit = {
			ROOT = {
				NOT = { has_active_mission = asteroid_incoming_stabilize_damaged_zone_mission }
				asteroid_incoming_country_can_start_recovery_mission = yes
			}
		}
		ROOT = {
			set_temp_variable = { asteroid_incoming_stabilize_best_score = constant:asteroid_incoming_runtime.zero }
			set_temp_variable = { asteroid_incoming_stabilize_second_score = constant:asteroid_incoming_runtime.zero }
			set_temp_variable = { asteroid_incoming_stabilize_best_state = constant:asteroid_incoming_runtime.zero }
			set_temp_variable = { asteroid_incoming_stabilize_second_state = constant:asteroid_incoming_runtime.zero }
			set_temp_variable = { asteroid_incoming_state_severity = constant:asteroid_incoming_runtime.zero }
			every_controlled_state = {
				limit = {
					asteroid_incoming_state_is_prev_recovery_site = yes
					NOT = { asteroid_incoming_state_meets_mission_logistics_targets = yes }
				}
				set_temp_variable = { asteroid_incoming_state_severity = asteroid_incoming_state_profile }
				multiply_temp_variable = { asteroid_incoming_state_severity = constant:asteroid_incoming_recovery.stabilize_profile_rank_weight }
				add_to_temp_variable = { asteroid_incoming_state_severity = asteroid_incoming_preimpact_infrastructure }
				subtract_from_temp_variable = { asteroid_incoming_state_severity = building_level@infrastructure }
				add_to_temp_variable = { asteroid_incoming_state_severity = asteroid_incoming_preimpact_railway }
				subtract_from_temp_variable = { asteroid_incoming_state_severity = building_level@rail_way }
				if = {
					limit = { check_variable = { asteroid_incoming_state_severity > asteroid_incoming_stabilize_best_score } }
					set_temp_variable = { asteroid_incoming_stabilize_second_score = asteroid_incoming_stabilize_best_score }
					set_temp_variable = { asteroid_incoming_stabilize_second_state = asteroid_incoming_stabilize_best_state }
					set_temp_variable = { asteroid_incoming_stabilize_best_score = asteroid_incoming_state_severity }
					set_temp_variable = { asteroid_incoming_stabilize_best_state = THIS }
				}
				else_if = {
					limit = { check_variable = { asteroid_incoming_state_severity > asteroid_incoming_stabilize_second_score } }
					set_temp_variable = { asteroid_incoming_stabilize_second_score = asteroid_incoming_state_severity }
					set_temp_variable = { asteroid_incoming_stabilize_second_state = THIS }
				}
			}
			if = {
				limit = { check_variable = { asteroid_incoming_stabilize_best_score > constant:asteroid_incoming_runtime.zero } }
				asteroid_incoming_clear_stabilize_mission = yes
				set_variable = { asteroid_incoming_stabilize_mission_state_one = asteroid_incoming_stabilize_best_state }
				if = {
					limit = { check_variable = { asteroid_incoming_stabilize_second_score > constant:asteroid_incoming_runtime.zero } }
					set_variable = { asteroid_incoming_stabilize_mission_state_two = asteroid_incoming_stabilize_second_state }
				}

# COUNTRY SCOPE. Compares current factories, infrastructure and railway across
# the surviving damaged states this country holds with the same totals before
# the impact. Outputs the country ratios read by Rebuild the Regional Economy.
# A total that had nothing before the impact counts as fully restored. The
# monthly recovery pass calls this without relying on ROOT.
asteroid_incoming_measure_regional_capacity = {
	set_temp_variable = { asteroid_incoming_regional_factories_now = constant:asteroid_incoming_runtime.zero }
	set_temp_variable = { asteroid_incoming_regional_factories_before = constant:asteroid_incoming_runtime.zero }
	set_temp_variable = { asteroid_incoming_regional_infrastructure_now = constant:asteroid_incoming_runtime.zero }
	set_temp_variable = { asteroid_incoming_regional_infrastructure_before = constant:asteroid_incoming_runtime.zero }
	set_temp_variable = { asteroid_incoming_regional_railway_now = constant:asteroid_incoming_runtime.zero }
	set_temp_variable = { asteroid_incoming_regional_railway_before = constant:asteroid_incoming_runtime.zero }
	every_controlled_state = {
		limit = { asteroid_incoming_state_is_prev_recovery_site = yes }
		add_to_temp_variable = { asteroid_incoming_regional_factories_now = building_level@industrial_complex }
		add_to_temp_variable = { asteroid_incoming_regional_factories_now = building_level@arms_factory }
		add_to_temp_variable = { asteroid_incoming_regional_factories_before = asteroid_incoming_preimpact_industrial_complex }
		add_to_temp_variable = { asteroid_incoming_regional_factories_before = asteroid_incoming_preimpact_arms_factory }
		add_to_temp_variable = { asteroid_incoming_regional_infrastructure_now = building_level@infrastructure }
		add_to_temp_variable = { asteroid_incoming_regional_infrastructure_before = asteroid_incoming_preimpact_infrastructure }
		add_to_temp_variable = { asteroid_incoming_regional_railway_now = building_level@rail_way }
		add_to_temp_variable = { asteroid_incoming_regional_railway_before = asteroid_incoming_preimpact_railway }
	}
	set_variable = { asteroid_incoming_regional_factory_ratio = constant:asteroid_incoming_runtime.one }
	if = {
		limit = { check_variable = { asteroid_incoming_regional_factories_before > constant:asteroid_incoming_runtime.zero } }
		set_variable = { asteroid_incoming_regional_factory_ratio = asteroid_incoming_regional_factories_now }
		divide_variable = { asteroid_incoming_regional_factory_ratio = asteroid_incoming_regional_factories_before }
	}
	set_variable = { asteroid_incoming_regional_infrastructure_ratio = constant:asteroid_incoming_runtime.one }
	if = {
		limit = { check_variable = { asteroid_incoming_regional_infrastructure_before > constant:asteroid_incoming_runtime.zero } }
		set_variable = { asteroid_incoming_regional_infrastructure_ratio = asteroid_incoming_regional_infrastructure_now }
		divide_variable = { asteroid_incoming_regional_infrastructure_ratio = asteroid_incoming_regional_infrastructure_before }
	}
	set_variable = { asteroid_incoming_regional_railway_ratio = constant:asteroid_incoming_runtime.one }
	if = {
		limit = { check_variable = { asteroid_incoming_regional_railway_before > constant:asteroid_incoming_runtime.zero } }
		set_variable = { asteroid_incoming_regional_railway_ratio = asteroid_incoming_regional_railway_now }
		divide_variable = { asteroid_incoming_regional_railway_ratio = asteroid_incoming_regional_railway_before }
	}
}

		min = constant:asteroid_incoming_cost.minimum_multiplier
		max = constant:asteroid_incoming_cost.maximum_multiplier

	set_variable = { asteroid_incoming_quote_manpower = constant:asteroid_incoming_recovery.decision_manpower }

# STATE SCOPE, failing country in PREV. Applies the local consequences of an

	if = {
		limit = {
			OR = {
				NOT = { has_variable = asteroid_incoming_mission_failure_day }
				NOT = { check_variable = { asteroid_incoming_mission_failure_day = global.num_days } }
			}
		}
		set_variable = { asteroid_incoming_mission_failure_day = global.num_days }
		asteroid_incoming_apply_mission_failure_deaths = yes
		asteroid_incoming_reapply_state_impact_damage = yes

		add_dynamic_modifier = {
			modifier = asteroid_incoming_mission_failure_pressure
			days = asteroid_incoming_failure_state_pressure_days
		}
	}
}

# STATE SCOPE. Records the people lost when relief fails as one Deaths entry
# through the shared protected civilian-loss contract. The loss is a constant
# share of the current population, doubled for the catastrophic ring and
# fragment centres, and capped in people.
asteroid_incoming_apply_mission_failure_deaths = {
	set_temp_variable = { asteroid_incoming_failure_death_fraction = constant:asteroid_incoming_recovery_fraction.rescue_death_fraction }
	if = {
		limit = {
			OR = {
				check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_one compare = equals }
				check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.fragment_center compare = equals }
			}
		}
		set_temp_variable = { asteroid_incoming_failure_death_fraction = constant:asteroid_incoming_recovery_fraction.rescue_death_fraction_heavy }
	}
	set_temp_variable = { asteroid_incoming_failure_deaths_requested = state_population_k }
	multiply_temp_variable = { asteroid_incoming_failure_deaths_requested = constant:chaos_meter_deaths.people_per_k }
	multiply_temp_variable = { asteroid_incoming_failure_deaths_requested = asteroid_incoming_failure_death_fraction }
	clamp_temp_variable = {
		var = asteroid_incoming_failure_deaths_requested
		min = constant:asteroid_incoming_runtime.zero
		max = constant:asteroid_incoming_recovery_fraction.rescue_death_cap_people
	}
	round_temp_variable = asteroid_incoming_failure_deaths_requested
	if = {
		limit = { check_variable = { asteroid_incoming_failure_deaths_requested > constant:asteroid_incoming_runtime.zero } }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:asteroid_incoming_runtime.one }
		set_temp_variable = { state_civilian_population_loss_requested = asteroid_incoming_failure_deaths_requested }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:asteroid_incoming_population_loss.rescue_minimum_remaining_people }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.asteroid_rescue_failure }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:asteroid_incoming_runtime.one }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:asteroid_incoming_runtime.one }
		apply_exact_state_civilian_population_loss = yes
	}
}

# STATE SCOPE. Brings back the state's impact-zone damage for a constant period
# when it has already ended; damage that is still running keeps its timer.
# Emergency-class damage reopens the local emergency, which rejoins the
# monthly closure list.
asteroid_incoming_reapply_state_impact_damage = {
	set_temp_variable = { asteroid_incoming_failure_damage_days = constant:asteroid_incoming_recovery.mission_failure_recovery_days }

		limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_one compare = equals } }

	else_if = {

			limit = { NOT = { has_dynamic_modifier = { modifier = asteroid_incoming_major_impact_zone } } }
			add_dynamic_modifier = { modifier = asteroid_incoming_major_impact_zone days = asteroid_incoming_failure_damage_days }

	}
	else_if = {
		limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.main_ring_three compare = equals } }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = asteroid_incoming_outer_shock_zone } } }
			add_dynamic_modifier = { modifier = asteroid_incoming_outer_shock_zone days = asteroid_incoming_failure_damage_days }
		}
	}
	else_if = {
		limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.fragment_center compare = equals } }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = asteroid_incoming_fragment_center_recovery } } }
			add_dynamic_modifier = { modifier = asteroid_incoming_fragment_center_recovery days = asteroid_incoming_failure_damage_days }
		}

	else_if = {
		limit = { check_variable = { var = asteroid_incoming_state_profile value = constant:asteroid_incoming_profile.fragment_ring_one compare = equals } }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = asteroid_incoming_heavy_fragment_damage } } }

# COUNTRY SCOPE. The national consequences of an expired mission: the full
# recovery burden replaces any recovery legacy for the stated window, and
# repair prices rise for a limited period. The timed flag holds the monthly
# recovery pass and the closeout report off the national idea until the
# window ends; a later failure restarts both timers together.
asteroid_incoming_apply_mission_failure_to_country = {
	set_temp_variable = { asteroid_incoming_failure_pressure_days = constant:asteroid_incoming_recovery.mission_failure_pressure_days }
	remove_ideas = asteroid_incoming_recovery_legacy
	remove_ideas = asteroid_incoming_recovery_burden
	add_timed_idea = {
		idea = asteroid_incoming_recovery_burden
		days = asteroid_incoming_failure_pressure_days
	}
	# A timed flag's days take a literal only, so the same duration is
	# injected as an integer.
	meta_effect = {
		text = {
			set_country_flag = { flag = asteroid_incoming_recovery_failure_pressure days = [FAILURE_PRESSURE_DAYS] }
		}
		FAILURE_PRESSURE_DAYS = "[?asteroid_incoming_failure_pressure_days|.0]"
	}
	asteroid_incoming_apply_repair_cost_surcharge = yes
}

# COUNTRY SCOPE. National recovery follows the country's real exact-state
# burdens rather than the unrelated global dust clock. The closeout report is
# queued once after every timed local burden and active mission has resolved.
# While a failed mission's burden window runs, the national idea and the
# closeout stay untouched; the normal legacy, burden or closeout logic resumes
# on the first pass after it ends.
# A country that rebuilt its regional economy keeps the recovery legacy after
# its last local burden ends; annexation clears that completion flag.
# The monthly pass reaches this through every_country, so ROOT is not used.
asteroid_incoming_refresh_national_recovery = {
	if = {
		limit = {
			has_variable = asteroid_incoming_recovery_target_state
			NOT = { has_country_flag = asteroid_incoming_recovery_project_pending }
		}
		asteroid_incoming_clear_recovery_target = yes
	}
	if = {
		limit = { has_country_flag = asteroid_incoming_regional_economy_mission_ready }
		asteroid_incoming_measure_regional_capacity = yes
	}
	if = {
		limit = { NOT = { has_country_flag = asteroid_incoming_recovery_failure_pressure } }
		remove_ideas = asteroid_incoming_recovery_burden
		remove_ideas = asteroid_incoming_recovery_legacy
		if = {
			limit = {
				has_country_flag = asteroid_incoming_affected_country
				asteroid_incoming_country_has_unresolved_local_burden = yes
			}
			if = {
				limit = { has_country_flag = asteroid_incoming_recovery_legacy_active }
				add_ideas = asteroid_incoming_recovery_legacy
			}
			else = { add_ideas = asteroid_incoming_recovery_burden }
		}
		else = {
			# Rebuild the Regional Economy leaves a lasting legacy that
			# outlives the local burdens and the closeout report. Other
			# mission legacies end with the burden they replaced.
			if = {
				limit = { has_country_flag = asteroid_incoming_regional_economy_complete }
				add_ideas = asteroid_incoming_recovery_legacy
			}
			if = {
				limit = {
					has_country_flag = asteroid_incoming_affected_country
					has_global_flag = asteroid_incoming_impact_recorded
					NOT = { has_country_flag = asteroid_incoming_recovery_closeout_sent }
					NOT = { has_country_flag = asteroid_incoming_recovery_closeout_pending }
				}
				set_country_flag = asteroid_incoming_recovery_closeout_pending
				country_event = { id = chaosx.nr28.6 }
			}
		}
	}
}
