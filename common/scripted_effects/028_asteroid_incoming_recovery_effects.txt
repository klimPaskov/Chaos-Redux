# ============================================================================
# CHAOS REDUX - EVENT 028 ASTEROID INCOMING RECOVERY EFFECTS
# ============================================================================
# State-targeted recovery actions and mission outcomes for Event 028. Every
# repair reads the state's Event 028 pre-impact receipt, restores only a
# bounded fraction of the observed gap, and refuses the permanent main crater.
# These helpers do not restore population and do not create nuclear, fallout,
# or generic world-scan side effects.
# ============================================================================

# STATE SCOPE. Inputs are the pre-impact receipt variables captured by the
# physical transaction and asteroid_incoming_restore_fraction as a temporary
# fixed-point value. Each explicit building call keeps the construction type
# static so a malformed token can never create an unintended building.
asteroid_incoming_restore_infrastructure = {
	if = {
		limit = {
			NOT = { has_state_flag = asteroid_incoming_main_crater }
			has_variable = asteroid_incoming_preimpact_infrastructure
		}
		set_temp_variable = { asteroid_incoming_restore_levels = asteroid_incoming_preimpact_infrastructure }
		subtract_from_temp_variable = { asteroid_incoming_restore_levels = building_level@infrastructure }
		clamp_temp_variable = {
			var = asteroid_incoming_restore_levels
			min = constant:asteroid_incoming_dust.minimum
			max = asteroid_incoming_preimpact_infrastructure
		}
		multiply_temp_variable = { asteroid_incoming_restore_levels = asteroid_incoming_restore_fraction }
		round_temp_variable = asteroid_incoming_restore_levels
		if = {
			limit = { check_variable = { asteroid_incoming_restore_levels > constant:asteroid_incoming_dust.minimum } }
			add_building_construction = {
				type = infrastructure
				level = asteroid_incoming_restore_levels
				instant_build = yes
			}
		}
	}
}

# STATE SCOPE. Finish one local emergency project and shorten every Event 028
# timed burden on the selected state. Persistent crater identities remain.
asteroid_incoming_close_state_emergency = {
	remove_dynamic_modifier = { modifier = asteroid_incoming_catastrophic_impact_zone }
	remove_dynamic_modifier = { modifier = asteroid_incoming_major_impact_zone }
	remove_dynamic_modifier = { modifier = asteroid_incoming_outer_shock_zone }
	remove_dynamic_modifier = { modifier = asteroid_incoming_heavy_fragment_damage }
	remove_dynamic_modifier = { modifier = asteroid_incoming_light_fragment_damage }
	set_state_flag = asteroid_incoming_local_emergency_resolved
	add_to_variable = { asteroid_incoming_recovery_actions_completed = 1 }
}

# COUNTRY SCOPE. Bounded global mitigation from an accepted action.
asteroid_incoming_add_mitigation = {
	if = { limit = { NOT = { has_variable = global.asteroid_incoming_mitigation_points } } set_variable = { global.asteroid_incoming_mitigation_points = 0 } }
	add_to_variable = { global.asteroid_incoming_mitigation_points = 1 }
	clamp_variable = {
		var = global.asteroid_incoming_mitigation_points
		min = 0
		max = constant:asteroid_incoming_dust.maximum_mitigation_decay
	}
}

# STATE SCOPE. Records the exact selected state for a mission and prevents a
# stale target from being used after ownership or controller changes.
asteroid_incoming_save_recovery_target = {
	if = {
		limit = { asteroid_incoming_decision_target_state_is_valid = yes }
		save_global_event_target_as = asteroid_incoming_decision_target_state
	}
}

# COUNTRY SCOPE. National recovery follows the country's real exact-state
# burdens rather than the unrelated global dust clock. The closeout report is
# queued once after every timed local burden and active mission has resolved.
asteroid_incoming_refresh_national_recovery = {
	remove_ideas = asteroid_incoming_recovery_burden
	remove_ideas = asteroid_incoming_recovery_legacy
	if = {
		limit = {
			has_country_flag = asteroid_incoming_affected_country
			asteroid_incoming_country_has_unresolved_local_burden = yes
		}
		if = {
			limit = { has_country_flag = asteroid_incoming_recovery_legacy_active }
			add_ideas = asteroid_incoming_recovery_legacy
		}
		else = { add_ideas = asteroid_incoming_recovery_burden }
	}
	else_if = {
		limit = {
			has_country_flag = asteroid_incoming_affected_country
			has_global_flag = asteroid_incoming_impact_recorded
			NOT = { has_country_flag = asteroid_incoming_recovery_closeout_sent }
			NOT = { has_country_flag = asteroid_incoming_recovery_closeout_pending }
		}
		set_country_flag = asteroid_incoming_recovery_closeout_pending
		country_event = { id = chaosx.nr28.6 }
	}
}
