infantry_spawn_recognize_selected_claimant = {
	if = {
		limit = { infantry_spawn_can_recognize_selected_claimant = yes }
		set_temp_variable = { infantry_spawn_claimant_response_cost = constant:infantry_spawn_claimant_response.recognition_political_power }
		multiply_temp_variable = { infantry_spawn_claimant_response_cost = -1 }
		add_political_power = infantry_spawn_claimant_response_cost
		set_variable = { infantry_spawn_claimant_status_entries^infantry_spawn_selected_claimant_index = constant:infantry_spawn_claimant_status.recognized }
		add_to_variable = { infantry_spawn_claimant_influence_entries^infantry_spawn_selected_claimant_index = constant:infantry_spawn_claimant_response.recognition_influence_gain }
		set_country_flag = infantry_spawn_claimant_formal_appointment
		add_ideas = infantry_spawn_recognized_claimant_command
	}
}

	if = {
		limit = { infantry_spawn_can_counter_command_selected_claimant = yes }
		set_temp_variable = { infantry_spawn_claimant_response_cost = constant:infantry_spawn_claimant_cost.counter_command_army_experience }
		multiply_temp_variable = { infantry_spawn_claimant_response_cost = -1 }
		add_army_experience = infantry_spawn_claimant_response_cost
		set_temp_variable = { infantry_spawn_claimant_response_cost = constant:infantry_spawn_claimant_cost.counter_command_command_power }
		multiply_temp_variable = { infantry_spawn_claimant_response_cost = -1 }
		add_command_power = infantry_spawn_claimant_response_cost
		subtract_from_variable = { infantry_spawn_claimant_influence_entries^infantry_spawn_selected_claimant_index = constant:infantry_spawn_claimant_response.counter_command_influence_loss }
		clamp_variable = { var = infantry_spawn_claimant_influence_entries^infantry_spawn_selected_claimant_index min = constant:infantry_spawn_claimant.influence_minimum max = constant:infantry_spawn_claimant.influence_maximum }
		add_to_variable = { infantry_spawn_muster_control = constant:infantry_spawn_claimant_response.counter_command_control_gain }
		clamp_variable = { var = infantry_spawn_muster_control min = constant:infantry_spawn_control.minimum max = constant:infantry_spawn_control.maximum }
		set_variable = { infantry_spawn_claimant_status_entries^infantry_spawn_selected_claimant_index = constant:infantry_spawn_claimant_status.countermanded }
		if = {
			limit = { check_variable = { var = infantry_spawn_claimant_influence_entries^infantry_spawn_selected_claimant_index value = constant:infantry_spawn_claimant.revolt_threshold compare = less_than } }
			infantry_spawn_clear_claimant_crisis_state = yes
		}
		infantry_spawn_recalculate_country_pressure = yes
	}
}
