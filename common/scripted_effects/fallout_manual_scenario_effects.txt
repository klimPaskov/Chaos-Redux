fallout_manual_fail_sweep = {
	if = {
		limit = { NOT = { has_global_flag = fallout_manual_sweep_error } }
		set_global_flag = fallout_manual_scenario_active
		set_global_flag = fallout_manual_sweep_error
		set_variable = { global.fallout_manual_error_code = fallout_manual_failure_code }
		set_variable = { global.fallout_manual_status = constant:fallout_manual_status.error }
		clr_global_flag = fallout_manual_sweep_active

			limit = { NOT = { has_global_flag = world_end_fallout } }
			clr_global_flag = fallout_air_cleanliness_disabled
			clr_global_flag = fallout_air_winter_shutdown_complete
			air_contamination_apply_state_modifier = yes
		}
		clear_variable = global.fallout_manual_scheduled_batch
		clear_variable = global.fallout_manual_scheduled_batch_generation
		clear_variable = global.fallout_manual_scheduled_verifier_attempt
		clear_variable = global.fallout_manual_scheduled_verifier_generation
		clear_variable = global.fallout_manual_scheduled_countdown_generation
	}
}

# Supported scope: COUNTRY
# Input: fallout_manual_selected_intensity_input
fallout_manual_initialize_sweep = {
	fallout_manual_reset_runtime_ledger = yes
	fallout_manual_capture_population_baselines = yes
	clear_global_event_target = fallout_manual_sweep_coordinator
	set_temp_variable = { fallout_manual_intensity_candidate = fallout_manual_selected_intensity_input }
	if = {
		limit = {
			check_variable = {
				var = fallout_manual_intensity_candidate
				value = constant:fallout_manual_sweep.minimum_intensity
				compare = greater_than_or_equals
			}
			check_variable = {
				var = fallout_manual_intensity_candidate
				value = constant:fallout_manual_sweep.maximum_intensity
				compare = less_than_or_equals
			}
		}
		set_variable = { global.fallout_manual_selected_intensity = fallout_manual_intensity_candidate }
		save_global_event_target_as = fallout_manual_sweep_coordinator
		set_country_flag = fallout_manual_synthetic_launcher
		set_global_flag = fallout_manual_scenario_active
		set_global_flag = fallout_air_cleanliness_disabled
		# The manual sweep is a Fallout-owned destructive start. Clear current
		# Air Cleanliness pressure before native strikes begin.
		air_contamination_apply_state_modifier = yes
		clr_global_flag = fallout_air_winter_shutdown_complete
		air_winter_suspend_all_states_for_fallout = yes
		set_global_flag = fallout_manual_sweep_active
		set_global_flag = fallout_synthetic_strike_batch
		set_variable = { global.fallout_manual_status = constant:fallout_manual_status.sweeping }
		fallout_manual_schedule_next_batch = yes
	}
	else = {
		set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.invalid_intensity }
		fallout_manual_fail_sweep = yes
	}
}

# Supported scope: COUNTRY
# Captures the state population before any native strike callback can mutate the
# map. The generation binds the per-state baseline to this sweep instance.
fallout_manual_capture_population_baselines = {
	set_variable = { global.fallout_manual_prestrike_population_baseline_count = constant:fallout_runtime.zero }
	clr_global_flag = fallout_manual_prestrike_population_baseline_ledger_complete
	clear_variable = global.fallout_manual_prestrike_population_baseline_ledger_generation
	every_state = {
		set_variable = { fallout_manual_prestrike_population_k = state_population_k }
		set_variable = { fallout_manual_prestrike_population_people = state_population_k }
		multiply_variable = { fallout_manual_prestrike_population_people = constant:fallout_runtime.population_scale }
		round_variable = fallout_manual_prestrike_population_people
		set_variable = { fallout_manual_prestrike_generation = global.fallout_manual_generation }
		set_state_flag = fallout_manual_prestrike_population_recorded
		add_to_variable = { global.fallout_manual_prestrike_population_baseline_count = constant:fallout_runtime.one }
	}
	if = {
		limit = {
			check_variable = {
				var = global.fallout_manual_prestrike_population_baseline_count
				value = constant:fallout_manual_sweep.expected_struck_states
				compare = equals
			}
		}
		set_variable = { global.fallout_manual_prestrike_population_baseline_ledger_generation = global.fallout_manual_generation }
		set_global_flag = fallout_manual_prestrike_population_baseline_ledger_complete
	}
}

fallout_manual_scenario_begin = {
	set_temp_variable = { fallout_manual_requested_intensity = fallout_manual_selected_intensity_input }
	if = {
		limit = { any_country = { has_country_flag = is_global_host } }
		every_country = {
			limit = {
				has_country_flag = is_global_host
				fallout_manual_sweep_can_begin = yes
			}
			set_temp_variable = { fallout_manual_selected_intensity_input = fallout_manual_requested_intensity }
			fallout_manual_initialize_sweep = yes
		}
	}
	else_if = {
		limit = { fallout_manual_sweep_can_begin = yes }
		set_temp_variable = { fallout_manual_selected_intensity_input = fallout_manual_requested_intensity }
		fallout_manual_initialize_sweep = yes
	}
}

# Supported scope: COUNTRY
fallout_manual_execute_next_batch = {
	if = {
		limit = {
			has_variable = fallout_manual_event_batch_index
			has_variable = global.fallout_manual_generation
			has_variable = global.fallout_manual_next_batch
			has_variable = global.fallout_manual_scheduled_batch
			has_variable = global.fallout_manual_scheduled_batch_generation
			has_variable = global.fallout_manual_last_completed_batch
			has_variable = global.fallout_manual_issued_strike_count
			has_variable = global.fallout_manual_observed_strike_count
			has_variable = global.fallout_manual_unique_struck_state_count
			fallout_manual_batch_cursor_is_valid = yes
			check_variable = {
				var = fallout_manual_event_batch_index
				value = global.fallout_manual_next_batch
				compare = equals
			}
			check_variable = {
				var = fallout_manual_event_batch_index
				value = global.fallout_manual_scheduled_batch
				compare = equals
			}
			check_variable = {
				var = global.fallout_manual_scheduled_batch_generation
				value = global.fallout_manual_generation
				compare = equals
			}
		}
		set_temp_variable = { fallout_manual_dispatch_batch = global.fallout_manual_next_batch }
		set_temp_variable = { fallout_manual_issued_before_batch = global.fallout_manual_issued_strike_count }
		set_temp_variable = { fallout_manual_expected_batch_size = constant:fallout_manual_sweep.batch_size }
		set_temp_variable = { fallout_manual_generated_batch_index = constant:fallout_runtime.negative_one }
		set_temp_variable = { fallout_manual_generated_batch_size = constant:fallout_runtime.zero }
		if = {
			limit = {
				check_variable = {
					var = fallout_manual_dispatch_batch
					value = constant:fallout_manual_sweep.final_batch_index
					compare = equals
				}
			}
			set_temp_variable = { fallout_manual_expected_batch_size = constant:fallout_manual_sweep.last_batch_size }
		}

		set_temp_variable = { fallout_manual_expected_issued_before_batch = fallout_manual_dispatch_batch }
		multiply_temp_variable = { fallout_manual_expected_issued_before_batch = constant:fallout_manual_sweep.batch_size }
		set_temp_variable = { fallout_manual_expected_last_completed_before_batch = fallout_manual_dispatch_batch }
		subtract_from_temp_variable = { fallout_manual_expected_last_completed_before_batch = constant:fallout_runtime.one }
		if = {
			limit = {
				OR = {
					NOT = {
						check_variable = {
							var = global.fallout_manual_issued_strike_count
							value = fallout_manual_expected_issued_before_batch
							compare = equals
						}
					}
					NOT = {
						check_variable = {
							var = global.fallout_manual_last_completed_batch
							value = fallout_manual_expected_last_completed_before_batch
							compare = equals
						}
					}
					check_variable = { global.fallout_manual_observed_strike_count < constant:fallout_runtime.zero }
					check_variable = {
						var = global.fallout_manual_observed_strike_count
						value = global.fallout_manual_issued_strike_count
						compare = greater_than
					}
					check_variable = { global.fallout_manual_unique_struck_state_count < constant:fallout_runtime.zero }
					check_variable = { global.fallout_manual_unique_struck_state_count > constant:fallout_manual_sweep.expected_struck_states }
				}
			}
			set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.runtime_ledger_inconsistent }
			fallout_manual_fail_sweep = yes
		}
		else = {
			set_global_flag = fallout_manual_batch_launch_window
			meta_effect = {
				text = { fallout_manual_execute_batch_[BATCH_ID] = yes }
				BATCH_ID = "[?fallout_manual_dispatch_batch|.0]"
			}
			clr_global_flag = fallout_manual_batch_launch_window

			set_temp_variable = { fallout_manual_issued_batch_delta = global.fallout_manual_issued_strike_count }
			subtract_from_temp_variable = { fallout_manual_issued_batch_delta = fallout_manual_issued_before_batch }
			if = {
				limit = {
					NOT = {
						check_variable = {
							var = fallout_manual_generated_batch_index
							value = fallout_manual_dispatch_batch
							compare = equals
						}
					}
				}
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.generated_batch_identity_mismatch }
				fallout_manual_fail_sweep = yes
			}
			else_if = {
				limit = {
					NOT = {
						check_variable = {
							var = fallout_manual_generated_batch_size
							value = fallout_manual_expected_batch_size
							compare = equals
						}
					}
				}
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.generated_batch_size_mismatch }
				fallout_manual_fail_sweep = yes
			}
			else_if = {
				limit = {
					NOT = {
						check_variable = {
							var = fallout_manual_issued_batch_delta
							value = fallout_manual_expected_batch_size
							compare = equals
						}
					}
				}
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.issued_count_mismatch }
				fallout_manual_fail_sweep = yes
			}
			else_if = {
				limit = {
					check_variable = {
						var = global.fallout_manual_observed_strike_count
						value = global.fallout_manual_issued_strike_count
						compare = greater_than
					}
				}
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.observed_count_mismatch }
				fallout_manual_fail_sweep = yes
			}
			else = {
				set_variable = { global.fallout_manual_last_completed_batch = fallout_manual_dispatch_batch }
				add_to_variable = { global.fallout_manual_next_batch = constant:fallout_runtime.one }
				clear_variable = global.fallout_manual_scheduled_batch
				clear_variable = global.fallout_manual_scheduled_batch_generation
				if = {
					limit = { fallout_manual_batch_cursor_is_valid = yes }
					fallout_manual_schedule_next_batch = yes
				}
				else = { fallout_manual_schedule_verifier = yes }
			}
		}
	}
	else = {
		set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.invalid_batch_token }
		fallout_manual_fail_sweep = yes
	}
}

# Supported scope: STATE
# The verified native-strike consequence removes a direct 90 to 95 percent
# band from the prestrike state population. The later standard rewrite is
# idempotent because it requests only the remaining grade-specific delta.
fallout_manual_apply_state_aggregate_consequence = {
	set_temp_variable = { fallout_manual_additional_state_strikes = fallout_manual_native_strike_count }
	subtract_from_temp_variable = { fallout_manual_additional_state_strikes = constant:fallout_runtime.one }
	clamp_temp_variable = {
		var = fallout_manual_additional_state_strikes
		min = constant:fallout_runtime.zero
		max = constant:fallout_manual_sweep.expected_province_strikes
	}

	set_temp_variable = { chaos_state_deaths_percent = fallout_manual_additional_state_strikes }
	multiply_temp_variable = { chaos_state_deaths_percent = constant:fallout_manual_aggregate.death_percent_per_additional_strike }
	add_to_temp_variable = { chaos_state_deaths_percent = constant:fallout_manual_aggregate.death_percent_base }
	clamp_temp_variable = {
		var = chaos_state_deaths_percent
		min = constant:fallout_manual_aggregate.death_percent_base
		max = constant:fallout_manual_aggregate.death_percent_max
	}
	set_temp_variable = { chaos_state_deaths_mult = constant:fallout_runtime.one }
	chaos_meter_apply_nuclear_state_death_environment_multiplier = yes
	multiply_temp_variable = { chaos_state_deaths_percent = chaos_state_deaths_mult }
	clamp_temp_variable = {
		var = chaos_state_deaths_percent
		min = constant:fallout_manual_aggregate.death_percent_base
		max = constant:fallout_manual_aggregate.death_percent_max
	}
	# Resolve the direct loss against the captured prestrike baseline. The live
	# population may already be lower because native nuclear hooks ran first.
	# Request only the delta needed to reach the direct survivor target so the
	# aggregate cannot overshoot the approved 90 to 95 percent band.
	set_temp_variable = { state_civilian_population_loss_requested = fallout_manual_prestrike_population_people }
	multiply_temp_variable = { state_civilian_population_loss_requested = constant:fallout_runtime.percentage_scale }
	multiply_temp_variable = { state_civilian_population_loss_requested = chaos_state_deaths_percent }
	round_temp_variable = state_civilian_population_loss_requested
	set_temp_variable = { fallout_manual_direct_target_survivor_population = fallout_manual_prestrike_population_people }
	subtract_from_temp_variable = { fallout_manual_direct_target_survivor_population = state_civilian_population_loss_requested }
	set_temp_variable = { fallout_manual_current_population_people = state_population_k }
	multiply_temp_variable = { fallout_manual_current_population_people = constant:fallout_runtime.population_scale }
	round_temp_variable = fallout_manual_current_population_people
	set_temp_variable = { state_civilian_population_loss_requested = fallout_manual_current_population_people }
	subtract_from_temp_variable = { state_civilian_population_loss_requested = fallout_manual_direct_target_survivor_population }
	clamp_temp_variable = {
		var = state_civilian_population_loss_requested
		min = constant:fallout_runtime.zero
	}

			fallout_manual_fail_sweep = yes
		}
		else_if = {
			limit = { NOT = { fallout_manual_selected_intensity_is_valid = yes } }
			set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.invalid_intensity }
			fallout_manual_fail_sweep = yes
		}
		else_if = {
			limit = { NOT = { fallout_manual_prestrike_population_baselines_are_current = yes } }
			set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.runtime_ledger_inconsistent }
			fallout_manual_fail_sweep = yes
		}
		else = {
			set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.zero }
			if = {
				limit = {
					OR = {
						NOT = { has_variable = global.fallout_manual_generation }
						NOT = { has_variable = global.fallout_manual_status }
						NOT = { has_variable = global.fallout_manual_next_batch }
						NOT = { has_variable = global.fallout_manual_last_completed_batch }
						NOT = { has_variable = global.fallout_manual_issued_strike_count }
						NOT = { has_variable = global.fallout_manual_observed_strike_count }
						NOT = { has_variable = global.fallout_manual_unique_struck_state_count }
						NOT = { has_variable = global.fallout_manual_state_strike_sum }
						NOT = { has_variable = global.fallout_manual_verifier_retry_count }
					}
				}
				set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
			}

			set_temp_variable = { fallout_manual_expected_last_completed_batch = global.fallout_manual_next_batch }
			subtract_from_temp_variable = { fallout_manual_expected_last_completed_batch = constant:fallout_runtime.one }
			if = {
				limit = {
					NOT = {
						check_variable = {
							var = global.fallout_manual_last_completed_batch
							value = fallout_manual_expected_last_completed_batch
							compare = equals
						}
					}
				}
				set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
			}

			set_temp_variable = { fallout_manual_expected_issued_at_cursor = global.fallout_manual_next_batch }
			multiply_temp_variable = { fallout_manual_expected_issued_at_cursor = constant:fallout_manual_sweep.batch_size }
			if = {
				limit = {
					check_variable = {
						var = global.fallout_manual_next_batch
						value = constant:fallout_manual_sweep.total_batches
						compare = equals
					}
				}
				set_temp_variable = { fallout_manual_expected_issued_at_cursor = constant:fallout_manual_sweep.expected_province_strikes }
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							var = global.fallout_manual_issued_strike_count
							value = fallout_manual_expected_issued_at_cursor
							compare = equals
						}
					}
				}
				set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
			}
			if = {
				limit = {
					OR = {
						check_variable = { global.fallout_manual_issued_strike_count < constant:fallout_runtime.zero }
						check_variable = { global.fallout_manual_observed_strike_count < constant:fallout_runtime.zero }
						check_variable = { global.fallout_manual_unique_struck_state_count < constant:fallout_runtime.zero }
						check_variable = { global.fallout_manual_issued_strike_count > constant:fallout_manual_sweep.expected_province_strikes }
						check_variable = {
							var = global.fallout_manual_observed_strike_count
							value = global.fallout_manual_issued_strike_count
							compare = greater_than
						}
						check_variable = { global.fallout_manual_unique_struck_state_count > constant:fallout_manual_sweep.expected_struck_states }
					}
				}
				set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
			}

			if = {
				limit = {
					OR = {
						AND = {
							has_global_flag = fallout_manual_batch_event_scheduled
							OR = {
								NOT = { has_variable = global.fallout_manual_scheduled_batch_generation }
								AND = {
									has_variable = global.fallout_manual_scheduled_batch_generation
									NOT = {
										check_variable = {
											var = global.fallout_manual_scheduled_batch_generation
											value = global.fallout_manual_generation
											compare = equals
										}
									}
								}
							}
						}
						AND = {
							NOT = { has_global_flag = fallout_manual_batch_event_scheduled }
							has_variable = global.fallout_manual_scheduled_batch_generation
						}
						AND = {
							has_global_flag = fallout_manual_verifier_event_scheduled
							OR = {
								NOT = { has_variable = global.fallout_manual_scheduled_verifier_generation }
								AND = {
									has_variable = global.fallout_manual_scheduled_verifier_generation
									NOT = {
										check_variable = {
											var = global.fallout_manual_scheduled_verifier_generation
											value = global.fallout_manual_generation
											compare = equals
										}
									}
								}
							}
						}
						AND = {
							NOT = { has_global_flag = fallout_manual_verifier_event_scheduled }
							has_variable = global.fallout_manual_scheduled_verifier_generation
						}
						AND = {
							has_global_flag = fallout_manual_countdown_event_scheduled
							OR = {
								NOT = { has_variable = global.fallout_manual_scheduled_countdown_generation }
								AND = {
									has_variable = global.fallout_manual_scheduled_countdown_generation
									NOT = {
										check_variable = {
											var = global.fallout_manual_scheduled_countdown_generation
											value = global.fallout_manual_generation
											compare = equals
										}
									}
								}
							}
						}
						AND = {
							NOT = { has_global_flag = fallout_manual_countdown_event_scheduled }
							has_variable = global.fallout_manual_scheduled_countdown_generation
						}
					}
				}
				set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
			}

			if = {
				limit = { check_variable = { global.fallout_manual_status = constant:fallout_manual_status.sweeping } }
				if = {
					limit = {
						OR = {
							NOT = { has_global_flag = fallout_manual_sweep_active }
							NOT = { has_global_flag = fallout_synthetic_strike_batch }
							has_global_flag = fallout_manual_native_sweep_complete
							has_global_flag = fallout_manual_countdown_active
							has_global_flag = fallout_manual_countdown_event_scheduled
							NOT = { fallout_manual_batch_cursor_is_valid = yes }
							has_global_flag = fallout_manual_verifier_event_scheduled
							has_variable = global.fallout_manual_scheduled_verifier_attempt
							has_variable = global.fallout_manual_scheduled_verifier_generation
							has_variable = global.fallout_manual_scheduled_countdown_generation
							has_global_flag = fallout_manual_batch_launch_window
							AND = {
									has_global_flag = fallout_manual_batch_event_scheduled
									OR = {
										NOT = { has_variable = global.fallout_manual_scheduled_batch }
										NOT = {
											check_variable = {
												var = global.fallout_manual_scheduled_batch
												value = global.fallout_manual_next_batch
												compare = equals
											}
										}
									}
								}
								AND = {
									NOT = { has_global_flag = fallout_manual_batch_event_scheduled }
									has_variable = global.fallout_manual_scheduled_batch
								}
							}
						}
						set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
				}
			}
			else_if = {
				limit = { check_variable = { global.fallout_manual_status = constant:fallout_manual_status.verifying } }
				if = {
					limit = {
						OR = {
							NOT = { has_global_flag = fallout_manual_sweep_active }
							NOT = { has_global_flag = fallout_synthetic_strike_batch }
							has_global_flag = fallout_manual_native_sweep_complete
							has_global_flag = fallout_manual_countdown_active
							has_global_flag = fallout_manual_countdown_event_scheduled
							NOT = { check_variable = { global.fallout_manual_next_batch = constant:fallout_manual_sweep.total_batches } }
							NOT = { check_variable = { global.fallout_manual_last_completed_batch = constant:fallout_manual_sweep.final_batch_index } }
							check_variable = { global.fallout_manual_verifier_retry_count < constant:fallout_runtime.zero }
							check_variable = { global.fallout_manual_verifier_retry_count > constant:fallout_manual_sweep.verifier_retry_limit }
							has_global_flag = fallout_manual_batch_event_scheduled
							has_variable = global.fallout_manual_scheduled_batch
							has_variable = global.fallout_manual_scheduled_batch_generation
							has_variable = global.fallout_manual_scheduled_countdown_generation
							has_global_flag = fallout_manual_batch_launch_window
							AND = {
									has_global_flag = fallout_manual_verifier_event_scheduled
									OR = {
										NOT = { has_variable = global.fallout_manual_scheduled_verifier_attempt }
										NOT = {
											check_variable = {
												var = global.fallout_manual_scheduled_verifier_attempt
												value = global.fallout_manual_verifier_retry_count
												compare = equals
											}
										}
									}
								}
								AND = {
									NOT = { has_global_flag = fallout_manual_verifier_event_scheduled }
									has_variable = global.fallout_manual_scheduled_verifier_attempt
								}
							}
						}
						set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
				}
			}
			else_if = {
				limit = { check_variable = { global.fallout_manual_status = constant:fallout_manual_status.countdown } }
				if = {
					limit = {
						OR = {
							has_global_flag = fallout_manual_sweep_active
							has_global_flag = fallout_synthetic_strike_batch
							NOT = { has_global_flag = fallout_manual_native_sweep_complete }
							NOT = { has_global_flag = fallout_manual_countdown_active }
							NOT = { has_global_flag = fallout_manual_countdown_event_scheduled }
							NOT = { has_variable = global.fallout_manual_countdown_start_day }
							NOT = { has_variable = global.fallout_manual_countdown_end_day }
							NOT = { has_variable = global.fallout_manual_scheduled_countdown_generation }
							NOT = { check_variable = { global.fallout_manual_next_batch = constant:fallout_manual_sweep.total_batches } }
							NOT = { check_variable = { global.fallout_manual_last_completed_batch = constant:fallout_manual_sweep.final_batch_index } }
							has_global_flag = fallout_manual_batch_event_scheduled
							has_global_flag = fallout_manual_verifier_event_scheduled
							has_global_flag = fallout_manual_batch_launch_window
							has_variable = global.fallout_manual_scheduled_batch
							has_variable = global.fallout_manual_scheduled_batch_generation
							has_variable = global.fallout_manual_scheduled_verifier_attempt
							has_variable = global.fallout_manual_scheduled_verifier_generation
							}
						}
					set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
				}
				set_temp_variable = { fallout_manual_expected_countdown_end_day = global.fallout_manual_countdown_start_day }
				add_to_temp_variable = { fallout_manual_expected_countdown_end_day = constant:fallout_manual_sweep.countdown_days }
				if = {
					limit = {
						NOT = {
							check_variable = {
								var = global.fallout_manual_countdown_end_day
								value = fallout_manual_expected_countdown_end_day
								compare = equals
							}
						}
					}
					set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
				}
			}
			else = {
				set_temp_variable = { fallout_manual_runtime_ledger_invalid = constant:fallout_runtime.one }
			}

			if = {
				limit = { check_variable = { fallout_manual_runtime_ledger_invalid > constant:fallout_runtime.zero } }
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.runtime_ledger_inconsistent }
				fallout_manual_fail_sweep = yes
			}
		}
	}
}

# Supported scope: COUNTRY. Called only from the existing host daily pulse.
fallout_manual_reconcile = {
	if = {
		limit = { has_global_flag = fallout_manual_scenario_active }
		fallout_manual_validate_active_runtime = yes
		if = {
			limit = {
				NOT = { has_global_flag = fallout_manual_sweep_error }
				NOT = { has_event_target = fallout_manual_sweep_coordinator }
			}
			if = {
				limit = { check_variable = { global.fallout_manual_status = constant:fallout_manual_status.countdown } }
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.countdown_coordinator_lost }
				fallout_manual_fail_sweep = yes
			}
			else = { fallout_manual_rehome_hourly_coordinator = yes }
		}
		else_if = {
			limit = {
				NOT = { has_global_flag = fallout_manual_sweep_error }
				NOT = {
					event_target:fallout_manual_sweep_coordinator = {
						exists = yes
						has_country_flag = is_global_host
						has_country_flag = fallout_manual_synthetic_launcher
					}
				}
			}
			if = {
				limit = { check_variable = { global.fallout_manual_status = constant:fallout_manual_status.countdown } }
				set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.countdown_coordinator_lost }
				fallout_manual_fail_sweep = yes
			}
			else = { fallout_manual_rehome_hourly_coordinator = yes }
		}

		if = {
			limit = {
				NOT = { has_global_flag = fallout_manual_sweep_error }
				has_global_flag = fallout_manual_sweep_active
				fallout_manual_batch_cursor_is_valid = yes
				NOT = { has_global_flag = fallout_manual_batch_event_scheduled }
			}
			fallout_manual_schedule_next_batch = yes
		}
		else_if = {
			limit = {
				NOT = { has_global_flag = fallout_manual_sweep_error }
				has_global_flag = fallout_manual_sweep_active
				check_variable = { global.fallout_manual_next_batch = constant:fallout_manual_sweep.total_batches }
				NOT = { has_global_flag = fallout_manual_verifier_event_scheduled }
			}
			fallout_manual_schedule_verifier = yes
		}

		if = {
			limit = {
				NOT = { has_global_flag = fallout_manual_sweep_error }
				has_global_flag = fallout_manual_countdown_active
				check_variable = {
					var = global.num_days
					value = global.fallout_manual_countdown_end_day
					compare = greater_than
				}
			}
			set_temp_variable = { fallout_manual_failure_code = constant:fallout_manual_error_code.countdown_overdue }
			fallout_manual_fail_sweep = yes
		}
	}
}
