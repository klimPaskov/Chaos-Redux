# ============================================================================
# CHAOS REDUX - CONDEMNATION SANCTIONS SCRIPTED EFFECTS
# ============================================================================
# Shared system for source-aware condemnation, tier transitions, sanction-pair
# scoring, native embargo ownership, practical penalties, counterplay, decay,
# and cleanup. All public condemnation gains must call condemnation_add_source.
#
# condemnation_add_source inputs (temporary variables on the condemned actor):
# - condemnation_source_type: constant:condemnation_source.*
# - condemnation_base_gain: positive pre-scaled source gain
# - condemnation_visibility_type: constant:condemnation_visibility.*
# - condemnation_severity_type: constant:condemnation_severity.*
# - condemnation_civilian_deaths: optional raw death count
# - condemnation_contamination_value: optional contamination pressure
# - condemnation_victim_country: optional country scope pointer
# - condemnation_context_type: optional constant:condemnation_context.*
# ============================================================================

	if = { limit = { NOT = { has_variable = condemnation_biological } } set_variable = { condemnation_biological = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_nuclear } } set_variable = { condemnation_nuclear = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_atrocity } } set_variable = { condemnation_atrocity = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_coverup } } set_variable = { condemnation_coverup = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_repeat_use } } set_variable = { condemnation_repeat_use = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_hidden_chemical } } set_variable = { condemnation_hidden_chemical = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_hidden_biological } } set_variable = { condemnation_hidden_biological = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_hidden_nuclear } } set_variable = { condemnation_hidden_nuclear = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_hidden_atrocity } } set_variable = { condemnation_hidden_atrocity = 0 } }

condemnation_recalculate_total = {
	set_variable = { condemnation_total = condemnation_chemical }
	add_to_variable = { condemnation_total = condemnation_biological }
	add_to_variable = { condemnation_total = condemnation_nuclear }
	add_to_variable = { condemnation_total = condemnation_atrocity }
	add_to_variable = { condemnation_total = condemnation_coverup }
	add_to_variable = { condemnation_total = condemnation_repeat_use }
	clamp_variable = { var = condemnation_total min = constant:condemnation_decay.minimum_total max = constant:condemnation_threshold.total_cap }
}

condemnation_recalculate_hidden_total = {
	set_variable = { condemnation_hidden_total = condemnation_hidden_chemical }
	add_to_variable = { condemnation_hidden_total = condemnation_hidden_biological }
	add_to_variable = { condemnation_hidden_total = condemnation_hidden_nuclear }
	add_to_variable = { condemnation_hidden_total = condemnation_hidden_atrocity }
	add_to_variable = { condemnation_hidden_total = condemnation_hidden_coverup }
	clamp_variable = { var = condemnation_hidden_total min = 0 max = constant:condemnation_threshold.total_cap }
}

condemnation_register_target = {
	if = {
		limit = { NOT = { is_in_array = { array = global.condemnation_targets value = THIS } } }
		add_to_array = { array = global.condemnation_targets value = THIS }
	}
}

condemnation_unregister_target_if_inactive = {
	if = {
		limit = {
			check_variable = { condemnation_total < constant:condemnation_gain.minimum_visible_gain }
			check_variable = { condemnation_hidden_total < constant:condemnation_gain.minimum_visible_gain }
			check_variable = { condemnation_active_sanctions < 1 }
			NOT = { condemnation_has_compliance_path = yes }
			NOT = { condemnation_has_defiance_path = yes }
			NOT = { condemnation_has_evasion_path = yes }
		}
		remove_from_array = { array = global.condemnation_targets value = THIS }
	}
}

condemnation_append_public_source_log = {
	add_to_array = { array = condemnation_source_log_types value = condemnation_source_type index = 0 }
	add_to_array = { array = condemnation_source_log_contexts value = condemnation_context_type index = 0 }
	add_to_array = { array = condemnation_source_log_gains value = condemnation_gain_final index = 0 }
	add_to_array = { array = condemnation_source_log_dates value = global.date index = 0 }
	set_temp_variable = { condemnation_source_log_limit = constant:condemnation_runtime.source_log_limit }
	if = {
		limit = { check_variable = { var = condemnation_source_log_types^num value = condemnation_source_log_limit compare = greater_than } }
		resize_array = { array = condemnation_source_log_types size = condemnation_source_log_limit }
		resize_array = { array = condemnation_source_log_contexts size = condemnation_source_log_limit }
		resize_array = { array = condemnation_source_log_gains size = condemnation_source_log_limit }
		resize_array = { array = condemnation_source_log_dates size = condemnation_source_log_limit }
	}
}

condemnation_expose_hidden_bucket = {
	# Inputs: condemnation_source_type, condemnation_hidden_exposure_fraction,
	# condemnation_context_type.
	set_temp_variable = { condemnation_hidden_exposed_gain = 0 }
	if = {
		limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.chemical compare = equals } }
		set_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_chemical }
		multiply_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_exposure_fraction }
		subtract_from_variable = { condemnation_hidden_chemical = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_chemical = condemnation_hidden_exposed_gain }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.biological compare = equals } }
		set_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_biological }
		multiply_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_exposure_fraction }
		subtract_from_variable = { condemnation_hidden_biological = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_biological = condemnation_hidden_exposed_gain }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.nuclear compare = equals } }
		set_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_nuclear }
		multiply_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_exposure_fraction }
		subtract_from_variable = { condemnation_hidden_nuclear = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_nuclear = condemnation_hidden_exposed_gain }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.atrocity compare = equals } }
		set_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_atrocity }
		multiply_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_exposure_fraction }
		subtract_from_variable = { condemnation_hidden_atrocity = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_atrocity = condemnation_hidden_exposed_gain }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.coverup compare = equals } }
		set_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_coverup }
		multiply_temp_variable = { condemnation_hidden_exposed_gain = condemnation_hidden_exposure_fraction }
		subtract_from_variable = { condemnation_hidden_coverup = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_coverup = condemnation_hidden_exposed_gain }
	}
	if = {
		limit = { check_variable = { condemnation_hidden_exposed_gain > constant:condemnation_gain.minimum_visible_gain } }
		set_temp_variable = { condemnation_gain_final = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_recent_gain = condemnation_hidden_exposed_gain }
		add_to_variable = { condemnation_participant_refresh_credit = condemnation_hidden_exposed_gain }
		set_variable = { condemnation_last_source_type = condemnation_source_type }
		set_variable = { condemnation_last_source_context = condemnation_context_type }
		set_variable = { condemnation_last_source_gain = condemnation_hidden_exposed_gain }
		set_variable = { condemnation_last_source_severity = constant:condemnation_severity.major }
		set_variable = { condemnation_last_gain_date = global.date }
		condemnation_append_public_source_log = yes
	}
}

condemnation_expose_hidden_evidence = {
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.chemical }
	condemnation_expose_hidden_bucket = yes
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.biological }
	condemnation_expose_hidden_bucket = yes
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.nuclear }
	condemnation_expose_hidden_bucket = yes
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.atrocity }
	condemnation_expose_hidden_bucket = yes
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.coverup }
	condemnation_expose_hidden_bucket = yes

	condemnation_recalculate_total = yes
	clamp_variable = { var = condemnation_recent_gain min = 0 max = constant:condemnation_threshold.recent_gain_cap }

		limit = { check_variable = { condemnation_recent_gain > 0 } }
		set_temp_variable = { condemnation_recent_gain_days = constant:condemnation_duration_days.recent_gain }
		set_country_flag = { flag = condemnation_recent_gain_visible days = condemnation_recent_gain_days }
	}
	condemnation_refresh_tier = yes
}

condemnation_start_targeted_pulse = {
	if = {
		limit = { NOT = { has_country_flag = condemnation_targeted_pulse_active } }
		set_country_flag = condemnation_targeted_pulse_active
		set_temp_variable = { condemnation_pulse_days = constant:condemnation_duration_days.pulse }
		country_event = {
			id = condemnation_sanctions.1
			days = condemnation_pulse_days
		}
	}
}

# Input: condemnation_fuel_fraction_cost (temporary fixed-point value).
condemnation_pay_fuel_fraction = {
	set_temp_variable = { condemnation_fuel_ratio_after_cost = fuel_ratio }
	set_temp_variable = { condemnation_fuel_ratio_cost = condemnation_fuel_ratio_after_cost }
	multiply_temp_variable = { condemnation_fuel_ratio_cost = condemnation_fuel_fraction_cost }
	subtract_from_temp_variable = { condemnation_fuel_ratio_after_cost = condemnation_fuel_ratio_cost }
	clamp_temp_variable = { var = condemnation_fuel_ratio_after_cost min = 0 max = 1 }
	set_fuel_ratio = condemnation_fuel_ratio_after_cost
}

condemnation_add_source = {
	condemnation_initialize_country = yes
	if = { limit = { NOT = { has_variable = condemnation_source_type } } set_temp_variable = { condemnation_source_type = constant:condemnation_source.mixed } }
	if = { limit = { NOT = { has_variable = condemnation_base_gain } } set_temp_variable = { condemnation_base_gain = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_visibility_type } } set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.public } }
	if = { limit = { NOT = { has_variable = condemnation_severity_type } } set_temp_variable = { condemnation_severity_type = constant:condemnation_severity.combat } }
	if = { limit = { NOT = { has_variable = condemnation_civilian_deaths } } set_temp_variable = { condemnation_civilian_deaths = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_contamination_value } } set_temp_variable = { condemnation_contamination_value = 0 } }
	if = { limit = { NOT = { has_variable = condemnation_context_type } } set_temp_variable = { condemnation_context_type = constant:condemnation_context.unspecified } }
	set_temp_variable = { condemnation_source_is_weapon_use = constant:condemnation_tier.normal }
	if = {
		limit = {
			OR = {
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.chemical_combat compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.chemical_air_strike compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.chemical_doomsday compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.biological_strike compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.biological_outbreak compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.biological_doomsday compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.weaponized_zombies compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.nuclear_strike compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.thermonuclear_strike compare = equals }
				check_variable = { var = condemnation_context_type value = constant:condemnation_context.final_silence compare = equals }
			}
		}
		set_temp_variable = { condemnation_source_is_weapon_use = constant:condemnation_tier.concern }
	}

	set_temp_variable = { condemnation_scaled_base_gain = condemnation_base_gain }
	set_temp_variable = { condemnation_death_gain = condemnation_civilian_deaths }
	divide_temp_variable = { condemnation_death_gain = constant:condemnation_gain.civilian_deaths_divisor }
	multiply_temp_variable = { condemnation_death_gain = constant:condemnation_gain.civilian_deaths_factor }
	clamp_temp_variable = { var = condemnation_death_gain min = 0 max = constant:condemnation_gain.civilian_deaths_gain_cap }
	add_to_temp_variable = { condemnation_scaled_base_gain = condemnation_death_gain }
	set_temp_variable = { condemnation_contamination_gain = condemnation_contamination_value }
	multiply_temp_variable = { condemnation_contamination_gain = constant:condemnation_gain.contamination_factor }
	clamp_temp_variable = { var = condemnation_contamination_gain min = 0 max = constant:condemnation_gain.contamination_gain_cap }
	add_to_temp_variable = { condemnation_scaled_base_gain = condemnation_contamination_gain }
	set_temp_variable = { condemnation_gain_final = condemnation_scaled_base_gain }
	if = {
		limit = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.hidden compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.visibility_hidden }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.suspected compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.visibility_suspected }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.discovered compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.visibility_discovered }
	}
	else = {
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.visibility_public }
	}

	if = {
		limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.minor compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.severity_minor }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.major compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.severity_major }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.mass_death compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.severity_mass_death }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.doomsday compare = equals } }
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.severity_doomsday }
	}
	else = {
		multiply_temp_variable = { condemnation_gain_final = constant:condemnation_gain.severity_combat }
	}

	set_temp_variable = { condemnation_gain_hidden = 0 }
	if = {
		limit = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.hidden compare = equals } }
		set_temp_variable = { condemnation_gain_hidden = condemnation_scaled_base_gain }
		if = { limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.minor compare = equals } } multiply_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.severity_minor } }
		else_if = { limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.major compare = equals } } multiply_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.severity_major } }
		else_if = { limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.mass_death compare = equals } } multiply_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.severity_mass_death } }
		else_if = { limit = { check_variable = { var = condemnation_severity_type value = constant:condemnation_severity.doomsday compare = equals } } multiply_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.severity_doomsday } }
		else = { multiply_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.severity_combat } }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.suspected compare = equals } }
		set_temp_variable = { condemnation_gain_hidden = condemnation_gain_final }
		divide_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.visibility_suspected }
		subtract_from_temp_variable = { condemnation_gain_hidden = condemnation_gain_final }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.discovered compare = equals } }
		set_temp_variable = { condemnation_gain_hidden = condemnation_gain_final }
		divide_temp_variable = { condemnation_gain_hidden = constant:condemnation_gain.visibility_discovered }
		subtract_from_temp_variable = { condemnation_gain_hidden = condemnation_gain_final }
	}

	if = {
		limit = { check_variable = { condemnation_gain_hidden > constant:condemnation_gain.minimum_visible_gain } }
		if = { limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.chemical compare = equals } } add_to_variable = { condemnation_hidden_chemical = condemnation_gain_hidden } }
		else_if = { limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.biological compare = equals } } add_to_variable = { condemnation_hidden_biological = condemnation_gain_hidden } }
		else_if = { limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.nuclear compare = equals } } add_to_variable = { condemnation_hidden_nuclear = condemnation_gain_hidden } }
		else_if = { limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.atrocity compare = equals } } add_to_variable = { condemnation_hidden_atrocity = condemnation_gain_hidden } }
		else_if = { limit = { check_variable = { var = condemnation_source_type value = constant:condemnation_source.coverup compare = equals } } add_to_variable = { condemnation_hidden_coverup = condemnation_gain_hidden } }
		else = {
			set_temp_variable = { condemnation_hidden_mixed_half = condemnation_gain_hidden }
			divide_temp_variable = { condemnation_hidden_mixed_half = constant:condemnation_math.two_way_average_divisor }
			add_to_variable = { condemnation_hidden_atrocity = condemnation_hidden_mixed_half }
			add_to_variable = { condemnation_hidden_coverup = condemnation_hidden_mixed_half }
		}
		clamp_variable = { var = condemnation_hidden_chemical min = 0 max = constant:condemnation_threshold.bucket_cap }
		clamp_variable = { var = condemnation_hidden_biological min = 0 max = constant:condemnation_threshold.bucket_cap }
		clamp_variable = { var = condemnation_hidden_nuclear min = 0 max = constant:condemnation_threshold.bucket_cap }
		clamp_variable = { var = condemnation_hidden_atrocity min = 0 max = constant:condemnation_threshold.bucket_cap }
		clamp_variable = { var = condemnation_hidden_coverup min = 0 max = constant:condemnation_threshold.bucket_cap }
		condemnation_recalculate_hidden_total = yes
		condemnation_register_target = yes
	}

	if = {
		limit = { check_variable = { condemnation_gain_final > constant:condemnation_gain.minimum_visible_gain } }
		set_temp_variable = { condemnation_repeat_gain = 0 }
		if = {
			limit = {
				has_country_flag = condemnation_recent_use
				check_variable = { condemnation_source_is_weapon_use > constant:condemnation_tier.normal }
			}
			set_temp_variable = { condemnation_repeat_gain = condemnation_gain_final }
			multiply_temp_variable = { condemnation_repeat_gain = constant:condemnation_gain.repeat_fraction }
			add_to_variable = { condemnation_repeat_use = condemnation_repeat_gain }
		}

		set_temp_variable = { condemnation_compliance_breach_applied = constant:condemnation_tier.normal }
		if = {
			limit = {
				has_country_flag = condemnation_non_use_pledge
				check_variable = { condemnation_source_is_weapon_use > constant:condemnation_tier.normal }
			}
			clr_country_flag = condemnation_non_use_pledge
			set_temp_variable = { condemnation_pledge_break_days = constant:condemnation_duration_days.pledge_break_memory }
			set_country_flag = { flag = condemnation_pledge_broken days = condemnation_pledge_break_days }
			add_to_variable = { condemnation_coverup = constant:condemnation_gain.pledge_break_coverup }
			add_to_variable = { condemnation_repeat_use = constant:condemnation_gain.pledge_break_repeat }

condemnation_refresh_tier = {
	condemnation_initialize_country = yes
	set_temp_variable = { condemnation_previous_tier = condemnation_tier }
	if = {
		limit = { check_variable = { var = condemnation_total value = constant:condemnation_threshold.pariah_state compare = greater_than_or_equals } }
		set_variable = { condemnation_tier = constant:condemnation_tier.pariah_state }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.total_cap }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_total value = constant:condemnation_threshold.total_embargo compare = greater_than_or_equals } }
		set_variable = { condemnation_tier = constant:condemnation_tier.total_embargo }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.pariah_state }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_total value = constant:condemnation_threshold.strategic_embargo compare = greater_than_or_equals } }
		set_variable = { condemnation_tier = constant:condemnation_tier.strategic_embargo }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.total_embargo }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_total value = constant:condemnation_threshold.arms_embargo compare = greater_than_or_equals } }
		set_variable = { condemnation_tier = constant:condemnation_tier.arms_embargo }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.strategic_embargo }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_total value = constant:condemnation_threshold.formal_censure compare = greater_than_or_equals } }
		set_variable = { condemnation_tier = constant:condemnation_tier.formal_censure }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.arms_embargo }
	}
	else_if = {
		limit = { check_variable = { var = condemnation_total value = constant:condemnation_threshold.concern compare = greater_than_or_equals } }
		set_variable = { condemnation_tier = constant:condemnation_tier.concern }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.formal_censure }
	}
	else = {
		set_variable = { condemnation_tier = constant:condemnation_tier.normal }
		set_variable = { condemnation_next_threshold = constant:condemnation_threshold.concern }
	}
	set_variable = { condemnation_compensation_convoy_requirement = condemnation_tier }
	multiply_variable = { condemnation_compensation_convoy_requirement = constant:condemnation_cost.compensation_convoys_per_tier }
	add_to_variable = { condemnation_compensation_convoy_requirement = constant:condemnation_cost.compensation_convoys_base }

	if = {
		limit = {
			check_variable = {
				var = condemnation_tier
				value = condemnation_peak_tier
				compare = greater_than
			}
		}
		set_variable = { condemnation_peak_tier = condemnation_tier }
	}

	clr_country_flag = condemnation_tier_changed
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = condemnation_tier
					value = condemnation_previous_tier
					compare = equals
				}
			}
		}
		set_country_flag = condemnation_tier_changed
	}
	condemnation_refresh_target_modifiers = yes
}

condemnation_recalculate_participants = {
	condemnation_initialize_country = yes
	set_temp_variable = { condemnation_recalculation_target = THIS }
	for_each_scope_loop = {
		array = condemnation_likely_participants
		limit = { exists = yes }
		var:condemnation_recalculation_target = { set_temp_variable = { condemnation_pair_target_id = id } }
		set_temp_variable = { condemnation_watch_tier = constant:condemnation_tier.formal_censure }
		meta_effect = {
			text = {
				if = {
					limit = { has_variable = condemnation_pair_[TARGET_ID]_watch_tier }
					set_temp_variable = { condemnation_watch_tier = condemnation_pair_[TARGET_ID]_watch_tier }
				}
				clear_variable = condemnation_pair_[TARGET_ID]_watch_tier
			}
			TARGET_ID = "[?condemnation_pair_target_id|.0]"
		}
		if = {
			limit = {
				is_ai = yes
				check_variable = { var = condemnation_watch_tier value = constant:condemnation_tier.concern compare = equals }
			}
			add_ai_strategy = { type = send_lend_lease_desire id = var:condemnation_recalculation_target value = constant:condemnation_ai_strategy.concern_lend_lease_restore }
			add_ai_strategy = { type = send_volunteers_desire id = var:condemnation_recalculation_target value = constant:condemnation_ai_strategy.concern_volunteers_restore }
			add_ai_strategy = { type = diplo_action_acceptance id = var:condemnation_recalculation_target target = military_access value = constant:condemnation_ai_strategy.concern_military_access_restore }
		}
		else_if = {
			limit = { is_ai = yes }
			add_ai_strategy = { type = send_lend_lease_desire id = var:condemnation_recalculation_target value = constant:condemnation_ai_strategy.censure_lend_lease_restore }
			add_ai_strategy = { type = send_volunteers_desire id = var:condemnation_recalculation_target value = constant:condemnation_ai_strategy.censure_volunteers_restore }
			add_ai_strategy = { type = diplo_action_acceptance id = var:condemnation_recalculation_target target = military_access value = constant:condemnation_ai_strategy.censure_military_access_restore }
		}
		remove_opinion_modifier = { target = var:condemnation_recalculation_target modifier = condemnation_monitoring_target }
	}
	set_variable = { condemnation_candidate_industry_weight = 0 }
	set_variable = { condemnation_candidate_trade_weight = 0 }
	clear_array = condemnation_likely_participants
	every_other_country = {
		limit = { exists = yes }
		set_temp_variable = { condemnation_pair_participant = THIS }
		set_temp_variable = { condemnation_pair_target = var:condemnation_recalculation_target }
		condemnation_cleanup_pair_statuses = yes
		var:condemnation_recalculation_target = { condemnation_calculate_pair_score = yes }
		condemnation_read_pair_record = yes
		set_temp_variable = { condemnation_pair_is_active = condemnation_pair_previous_tier }
		if = {
			limit = {
				has_dlc = "By Blood Alone"
				check_variable = { condemnation_pair_is_active > constant:condemnation_tier.normal }
				NOT = { is_embargoing = var:condemnation_recalculation_target }
			}
			set_temp_variable = { condemnation_pair_previous_expires = 0 }
		}
		set_temp_variable = { condemnation_pair_desired_tier = constant:condemnation_tier.normal }

		if = {
			limit = {
				var:condemnation_recalculation_target = {
					condemnation_is_arms_embargo_tier = yes
				}
				check_variable = { var = condemnation_pair_score value = constant:condemnation_participant_band.arms compare = greater_than_or_equals }
				OR = {
					check_variable = { condemnation_pair_is_active > constant:condemnation_tier.normal }
					condemnation_can_bear_enforcement_cost = yes
				}
			}
			set_temp_variable = { condemnation_pair_desired_tier = constant:condemnation_tier.arms_embargo }
			if = {
				limit = {
