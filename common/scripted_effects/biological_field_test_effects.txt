# ============================================================================
# CHAOS REDUX - ORDINARY BIOLOGICAL FIELD-TEST EFFECTS
# ============================================================================
# Overview:
# - Resolves the four special-project prototype field tests only in the exact
#   native facility state supplied by iteration_output.facility_state_effects.
# - Uses one shared accident chance for every ordinary agent.
# - A contained test grants project progress and records no outbreak.
# - A containment accident enters the ordinary biological lifecycle through
#   field_test_release. It never searches for an enemy or replacement state.
# - Weaponized-zombie field testing remains separate.
# ============================================================================

# State scope, FROM = active special project.
# Inputs:
# - bio_field_test_agent
# - bio_field_test_progress_gain
# - bio_field_test_success_event_id
# - bio_field_test_accident_event_id
bio_field_test_resolve_exact_facility_test = {
	save_event_target_as = field_test_state
	FROM.owner = { save_event_target_as = bio_field_test_actor }
	CONTROLLER = { save_event_target_as = bio_field_test_victim }
	set_temp_variable = { bio_field_test_accident = constant:bio_lifecycle_proof.missing }

	if = {
		limit = {
			has_event_target = bio_field_test_actor
			has_event_target = bio_field_test_victim
			event_target:bio_field_test_actor = { exists = yes }
			event_target:bio_field_test_victim = { exists = yes }
			CONTROLLER = { tag = event_target:bio_field_test_actor }
		}
		FROM = { add_project_progress_ratio = bio_field_test_progress_gain }
		random = {
			chance = constant:bio_field_test.accident_chance
			set_temp_variable = { bio_field_test_accident = constant:bio_lifecycle_proof.supplied }
		}

		if = {
			limit = { check_variable = { var = bio_field_test_accident value = constant:bio_lifecycle_proof.supplied compare = equals } }
			bio_lifecycle_reset_seed_record = yes
			set_temp_variable = { bio_seed_agent = bio_field_test_agent }
			set_temp_variable = { bio_seed_route = constant:bio_lifecycle_route.field_test_release }
			set_temp_variable = { bio_seed_source = constant:bio_lifecycle_source.field_test }
			set_temp_variable = { bio_seed_result = constant:bio_lifecycle_result.accident }
			set_temp_variable = { bio_seed_actor_proof = constant:bio_lifecycle_proof.supplied }
			set_temp_variable = { bio_seed_victim_proof = constant:bio_lifecycle_proof.supplied }
			set_temp_variable = { bio_seed_force_detection_on_activation = constant:bio_lifecycle_proof.supplied }
			event_target:bio_field_test_actor = { save_event_target_as = bio_seed_actor }
			event_target:bio_field_test_victim = { save_event_target_as = bio_seed_victim }
			bio_lifecycle_dispatch_seed = yes

			if = {
				limit = { check_variable = { var = bio_seed_dispatch_status value = constant:bio_lifecycle_proof.supplied compare = equals } }
				event_target:bio_field_test_actor = {
					set_country_flag = bio_field_test_accident_history
					meta_effect = {
						text = {
							set_country_flag = bio_agent_[BIO_AGENT]_field_test_accident_history
							country_event = { id = chaosx_bioweapon.[BIO_EVENT_ID] }
						}
						BIO_AGENT = "[?bio_field_test_agent|.0]"
						BIO_EVENT_ID = "[?bio_field_test_accident_event_id|.0]"
					}
				}
			}
		}
		else = {
			event_target:bio_field_test_actor = {
				set_country_flag = bio_field_test_completed_history
				meta_effect = {
					text = {
						set_country_flag = bio_agent_[BIO_AGENT]_field_test_completed_history
						country_event = { id = chaosx_bioweapon.[BIO_EVENT_ID] }
					}
					BIO_AGENT = "[?bio_field_test_agent|.0]"
					BIO_EVENT_ID = "[?bio_field_test_success_event_id|.0]"
				}
			}
		}
	}
}

# State scope, FROM = active special project.
conduct_anthrax_field_test = {
	set_temp_variable = { bio_field_test_agent = constant:bio_lifecycle_agent.anthrax }
	set_temp_variable = { bio_field_test_progress_gain = constant:sp_progress.gain.high }

	set_temp_variable = { bio_field_test_accident_event_id = 3 }
	bio_field_test_resolve_exact_facility_test = yes
