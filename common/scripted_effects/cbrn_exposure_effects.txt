# ============================================================================
# CHAOS REDUX - SHARED CBRN EXPOSURE EFFECTS
# ============================================================================
# Overview:
# - Defines the temporary action-record contract for deliberate chemical use.
# - Validates exact target, payload-debit, protection, and release proofs.
# - Applies environmental and command-condition modifiers only when an adapter
#   supplies a verified current-version receipt. Missing condition surfaces are
#   omitted rather than estimated or replaced by neutral receipts.
# - Resolves one normalized set of disruption, military-death fraction,
#   civilian exposure/death fraction, contamination, duration, medical burden,
#   evidence, attribution, Condemnation base, and friendly-exposure risk.
# - The Stage 6 consequence dispatcher consumes an accepted record once, after
#   real payload debit, and forwards these outputs to units, states, Deaths,
#   Air Cleanliness, evidence, Condemnation, treaty memory, and history.
# - Continuous ordinary-air missions are recognized only so they can be
#   rejected explicitly. Exact-state chemical air raids are supported routes.
# ============================================================================

cbrn_prepare_chemical_action_record = {
	cbrn_reset_action_outputs = yes

	if = {
		limit = { NOT = { cbrn_action_actor_is_valid = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_actor }
	}
	else_if = {
		limit = { NOT = { cbrn_action_target_state_is_valid = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.missing_target_state }
	}
	else_if = {
		limit = { NOT = { cbrn_action_weapon_class_is_chemical = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_weapon_class }
	}
	else_if = {
		limit = { NOT = { cbrn_action_agent_class_is_chemical = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_agent_class }
	}
	else_if = {
		limit = { NOT = { cbrn_action_agent_is_valid_chemical = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_agent }
	}
	else_if = {
		limit = { NOT = { cbrn_action_agent_matches_class = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.agent_class_mismatch }
	}
	else_if = {
		limit = { NOT = { cbrn_action_delivery_route_is_recognized = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_delivery_route }
	}
	else_if = {
		limit = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.continuous_air_mission compare = equals } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.unsupported_continuous_air_route }
	}
	else_if = {
		limit = { NOT = { cbrn_action_agent_is_eligible_for_route = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.agent_route_mismatch }
	}
	else_if = {
		limit = { NOT = { cbrn_action_severity_is_valid = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_severity }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = cbrn_action_payload_consumed_proof }
				NOT = { check_variable = { var = cbrn_action_payload_consumed_proof value = constant:cbrn_proof.supplied compare = equals } }
			}
		}
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.payload_not_consumed }
	}
	else_if = {
		limit = { NOT = { cbrn_action_payload_debit_is_valid = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.invalid_payload_values }
	}
	else_if = {
		limit = { NOT = { cbrn_action_protection_is_resolved = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.protection_not_resolved }
	}
	else_if = {
		limit = { NOT = { cbrn_action_release_is_resolved = yes } }
		set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.conditions_not_resolved }
	}
	else = {
		cbrn_bind_action_victim_country = yes
		if = {
			limit = { NOT = { has_event_target = cbrn_action_victim_country } }
			set_temp_variable = { cbrn_action_reject_reason = constant:cbrn_action_reject_reason.missing_victim_country }
		}
		else = {
			set_temp_variable = { cbrn_action_victim_country_supplied = constant:cbrn_proof.supplied }
			event_target:cbrn_action_victim_country = {
				save_event_target_as = cbrn_retaliation_target_country
			}
			set_temp_variable = { cbrn_retaliation_action_proportionate_proof = constant:cbrn_proof.missing }
			if = {
				limit = {
					OR = {
						check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.cylinder_release compare = equals }
						check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.projector_barrage compare = equals }
						check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals }
						check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.armored_delivery compare = equals }
					}
					check_variable = {
						var = cbrn_action_severity
						value = constant:cbrn_operation_severity.strategic
						compare = less_than
					}
					event_target:cbrn_action_target_state = {
						CONTROLLER = {
							divisions_in_state = {
								state = PREV
								size > constant:cbrn_math.zero
							}
						}
					}
				}
				set_temp_variable = { cbrn_retaliation_action_proportionate_proof = constant:cbrn_proof.supplied }
			}
			cbrn_diplomacy_classify_retaliation_action = yes
			# Doctrine mitigation is owned here so every accepted delivery route uses
			# the same Condemnation-only multiplier. Adapters cannot select a
			# different doctrine discount.
			cbrn_set_doctrine_condemnation_mult_from_country = yes
			cbrn_calculate_chemical_action_outputs = yes
			set_temp_variable = { cbrn_action_result = constant:cbrn_action_result.accepted }
		}
	}
}

cbrn_refresh_state_contamination_class = {
	set_temp_variable = { cbrn_state_contamination_class_output = constant:cbrn_contamination_class.none }
	if = {
		limit = { has_variable = cbrn_chemical_contamination }
		clamp_variable = { var = cbrn_chemical_contamination min = constant:cbrn_meter.minimum max = constant:cbrn_contamination_threshold.maximum }
		if = {
			limit = { check_variable = { var = cbrn_chemical_contamination value = constant:cbrn_contamination_threshold.trace compare = less_than } }
			clear_variable = cbrn_chemical_contamination
			clear_variable = cbrn_chemical_contamination_class
			clear_variable = cbrn_chemical_contamination_duration_days
		}
		else = {
			set_temp_variable = { cbrn_state_contamination_class_output = constant:cbrn_contamination_class.trace }
			if = {
				limit = { check_variable = { var = cbrn_chemical_contamination value = constant:cbrn_contamination_threshold.catastrophic compare = greater_than_or_equals } }
				set_temp_variable = { cbrn_state_contamination_class_output = constant:cbrn_contamination_class.catastrophic }
			}
			else_if = {
				limit = { check_variable = { var = cbrn_chemical_contamination value = constant:cbrn_contamination_threshold.severe compare = greater_than_or_equals } }
				set_temp_variable = { cbrn_state_contamination_class_output = constant:cbrn_contamination_class.severe }
			}
			else_if = {
				limit = { check_variable = { var = cbrn_chemical_contamination value = constant:cbrn_contamination_threshold.serious compare = greater_than_or_equals } }
				set_temp_variable = { cbrn_state_contamination_class_output = constant:cbrn_contamination_class.serious }
			}
			else_if = {
				limit = { check_variable = { var = cbrn_chemical_contamination value = constant:cbrn_contamination_threshold.local compare = greater_than_or_equals } }
				set_temp_variable = { cbrn_state_contamination_class_output = constant:cbrn_contamination_class.local }
			}
			set_variable = { cbrn_chemical_contamination_class = cbrn_state_contamination_class_output }
		}
	}
	else = {
		clear_variable = cbrn_chemical_contamination_class
		clear_variable = cbrn_chemical_contamination_duration_days
	}
}

cbrn_apply_state_contamination_delta_internal = {
	cbrn_refresh_state_contamination_class = yes
	set_temp_variable = { cbrn_state_previous_contamination_class = cbrn_state_contamination_class_output }
	set_temp_variable = { cbrn_state_previous_contamination_value = constant:cbrn_math.zero }
	if = {
		limit = { has_variable = cbrn_chemical_contamination }
		set_temp_variable = { cbrn_state_previous_contamination_value = cbrn_chemical_contamination }
	}

	if = {
		limit = { has_variable = cbrn_state_contamination_delta }
		if = {
			limit = { NOT = { has_variable = cbrn_chemical_contamination } }
			set_variable = { cbrn_chemical_contamination = constant:cbrn_math.zero }
		}
		add_to_variable = { cbrn_chemical_contamination = cbrn_state_contamination_delta }
		clamp_variable = { var = cbrn_chemical_contamination min = constant:cbrn_meter.minimum max = constant:cbrn_contamination_threshold.maximum }
	}

	if = {
		limit = {
			has_variable = cbrn_state_contamination_duration_input
			check_variable = { cbrn_state_contamination_duration_input > constant:cbrn_math.zero }
		}
		set_temp_variable = { cbrn_state_contamination_duration_local = cbrn_state_contamination_duration_input }
		clamp_temp_variable = { var = cbrn_state_contamination_duration_local min = constant:cbrn_scheduler.contamination_minimum_days max = constant:cbrn_scheduler.contamination_maximum_days }
		if = {
			limit = { NOT = { has_variable = cbrn_chemical_contamination_duration_days } }
			set_variable = { cbrn_chemical_contamination_duration_days = cbrn_state_contamination_duration_local }
		}
		else_if = {
			limit = { check_variable = { var = cbrn_chemical_contamination_duration_days value = cbrn_state_contamination_duration_local compare = less_than } }
			set_variable = { cbrn_chemical_contamination_duration_days = cbrn_state_contamination_duration_local }
		}
	}

	cbrn_refresh_state_contamination_class = yes
	set_temp_variable = { cbrn_state_new_contamination_class = cbrn_state_contamination_class_output }
	set_temp_variable = { cbrn_state_new_contamination_value = constant:cbrn_math.zero }
	if = {
		limit = { has_variable = cbrn_chemical_contamination }
		set_temp_variable = { cbrn_state_new_contamination_value = cbrn_chemical_contamination }
	}
	# Air Cleanliness is a state-class transition, not a monthly presence scan.
	# The state helper compares the stored contribution with the new class and
	# applies only the difference to the existing global system.
	set_temp_variable = { cbrn_air_contribution_class = cbrn_state_new_contamination_class }
	cbrn_chemical_set_air_contribution_from_class = yes
	cbrn_air_cleanliness_update_chemical_state_contribution = yes
	cbrn_chemical_refresh_state_modifier = yes
	if = {
		limit = { has_variable = cbrn_last_chemical_agent_class }
		cbrn_chemical_refresh_continuing_death_profile = yes
	}
}

cbrn_refresh_state_attribution = {
	set_temp_variable = { cbrn_state_attribution_output = constant:cbrn_attribution_state.unknown }
	if = {
		limit = { has_variable = cbrn_evidence_quality }
		clamp_variable = { var = cbrn_evidence_quality min = constant:cbrn_meter.minimum max = constant:cbrn_evidence_threshold.maximum }
		if = {
			limit = { check_variable = { var = cbrn_evidence_quality value = constant:cbrn_evidence_threshold.confirmed compare = greater_than_or_equals } }
			set_temp_variable = { cbrn_state_attribution_output = constant:cbrn_attribution_state.confirmed }
		}
		else_if = {
			limit = { check_variable = { var = cbrn_evidence_quality value = constant:cbrn_evidence_threshold.probable compare = greater_than_or_equals } }
			set_temp_variable = { cbrn_state_attribution_output = constant:cbrn_attribution_state.probable }
		}
		else_if = {
			limit = { check_variable = { var = cbrn_evidence_quality value = constant:cbrn_evidence_threshold.suspected compare = greater_than_or_equals } }
			set_temp_variable = { cbrn_state_attribution_output = constant:cbrn_attribution_state.suspected }
		}
		if = {
			limit = { check_variable = { var = cbrn_state_attribution_output value = constant:cbrn_attribution_state.unknown compare = equals } }
			clear_variable = cbrn_attribution_state
		}
		else = {
			set_variable = { cbrn_attribution_state = cbrn_state_attribution_output }
		}
	}
	else = {
		clear_variable = cbrn_attribution_state
	}
}
