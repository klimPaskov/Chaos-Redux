# COUNTRY scope. Inputs: temporary candidate route and score.
brilliant_scientist_evaluate_portrait_candidate = {
	if = {
		limit = {
			check_variable = {
				var = brilliant_scientist_portrait_candidate_score
				value = brilliant_scientist_portrait_maximum_score
				compare = greater_than
			}
		}
		set_temp_variable = { brilliant_scientist_portrait_maximum_score = brilliant_scientist_portrait_candidate_score }
		set_temp_variable = { brilliant_scientist_portrait_selected_route = brilliant_scientist_portrait_candidate_route }
		set_temp_variable = { brilliant_scientist_portrait_route_tied = constant:brilliant_scientist_value.zero }
	}
	else_if = {
		limit = {
			check_variable = {
				var = brilliant_scientist_portrait_candidate_score
				value = brilliant_scientist_portrait_maximum_score
				compare = equals
			}
			check_variable = { brilliant_scientist_portrait_candidate_score > constant:brilliant_scientist_value.zero }
		}
		set_temp_variable = { brilliant_scientist_portrait_route_tied = constant:brilliant_scientist_value.one }
	}
}
