# ============================================================================
# CHAOS REDUX - EVENT 006 REGIONAL FRAMEWORK PACKAGE EFFECTS
# ============================================================================
# Overview:
# - Shared Level 1 package engine for regional rows that have no bespoke
#   package of their own (agent decision D25). A framework country receives the
#   full common survival system, its region's identity overlay, its archetype
#   and ambition family, the shared Event 006 focus tree with the government
#   routes its research supports, its mapped dynamic starting force, and its
#   leadership: the vanilla roster of a registered tag, or the leader the game
#   appoints for an institutional government.
# - `independence_wave_regional_framework_apply_identity` holds the one
#   per-package block of every admitted row. Level 2 rows call their regional
#   layer from that block; Level 1 rows mark the layer as applied directly.
# - Setup reads the released country and its stored anchor and former host.
#   Cleanup runs only for the generation that set the package up.
# - The central dispatchers call the three `independence_wave_dispatch_
#   regional_framework_package_*` effects.
# No effect iterates countries or states.
# ============================================================================

# COUNTRY scope. Records the ruling party before a package changes it, so
# cleanup can restore the carrier's own government.
independence_wave_regional_framework_record_ruling_party = {
	if = {
		limit = { NOT = { has_variable = independence_wave_framework_restore_party } }
		if = { limit = { has_government = democratic } set_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_democratic } }
		else_if = { limit = { has_government = communism } set_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_communism } }
		else_if = { limit = { has_government = fascism } set_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_fascism } }
		else = { set_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_neutrality } }
	}
}

# COUNTRY scope. Restores the party recorded above.
independence_wave_regional_framework_restore_ruling_party = {
	if = {
		limit = { has_variable = independence_wave_framework_restore_party }
		if = { limit = { check_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_democratic } } set_politics = { ruling_party = democratic } }
		else_if = { limit = { check_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_communism } } set_politics = { ruling_party = communism } }
		else_if = { limit = { check_variable = { independence_wave_framework_restore_party = constant:independence_wave_regional_framework.party_fascism } } set_politics = { ruling_party = fascism } }
		else = { set_politics = { ruling_party = neutrality } }
		clear_variable = independence_wave_framework_restore_party
	}
}
