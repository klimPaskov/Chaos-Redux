# Appends one material history entry, not a notification. Population reports are separate.
# Inputs: temp camp_admin_record_kind and optional camp_admin_record_state_id (zero = national).
camp_admin_record_history = {
	if = {
		limit = { check_variable = { camp_admin_record_kind > constant:camp_admin_value.zero } }
		camp_admin_prepare_month_key = yes
		add_to_variable = { camp_admin_history_next_id = constant:camp_admin_value.one }
		add_to_array = { array = camp_admin_history_ids value = camp_admin_history_next_id }
		add_to_array = { array = camp_admin_history_kinds value = camp_admin_record_kind }
		add_to_array = { array = camp_admin_history_states value = camp_admin_record_state_id }
		add_to_array = { array = camp_admin_history_years value = global.year }
		add_to_array = { array = camp_admin_history_months value = camp_admin_month_number }
		if = {
			limit = { check_variable = { camp_admin_history_ids^num > constant:camp_admin_command.history_limit } }
			remove_from_array = { array = camp_admin_history_ids index = 0 }
			remove_from_array = { array = camp_admin_history_kinds index = 0 }
			remove_from_array = { array = camp_admin_history_states index = 0 }
			remove_from_array = { array = camp_admin_history_years index = 0 }
			remove_from_array = { array = camp_admin_history_months index = 0 }
			add_to_variable = { camp_admin_history_archived = constant:camp_admin_value.one }
		}
	}
	set_temp_variable = { camp_admin_record_kind = constant:camp_admin_value.zero }
	set_temp_variable = { camp_admin_record_state_id = constant:camp_admin_value.zero }
}

# Rebuild only the display list from the package's bounded active and civilian registries.
# Country-owned projection over the existing package registries only.
# A co-located occupation/custody host contributes one warning location.
camp_admin_project_origin_uncertainty = {
	set_variable = { camp_admin_gui_uncertain_locations = constant:camp_admin_value.zero }
	for_each_scope_loop = {
		array = camp_admin_known_site_states
		if = {
			limit = { camp_admin_site_is_original_record = yes camp_admin_gui_host_records_uncertain = yes }
			PREV = { add_to_variable = { camp_admin_gui_uncertain_locations = constant:camp_admin_value.one } }
		}
	}
	for_each_scope_loop = {
		array = camp_active_site_states
		set_temp_variable = { camp_admin_origin_projection_host = THIS }
		if = {
			limit = {
				NOT = { PREV = { is_in_array = { array = camp_admin_known_site_states value = var:camp_admin_origin_projection_host } } }
				camp_admin_gui_host_records_uncertain = yes
			}
			PREV = { add_to_variable = { camp_admin_gui_uncertain_locations = constant:camp_admin_value.one } }
		}
	}
	for_each_scope_loop = {
		array = camp_occ_states
		set_temp_variable = { camp_admin_origin_projection_host = THIS }
		if = {
			limit = {
				NOT = { PREV = { is_in_array = { array = camp_active_site_states value = var:camp_admin_origin_projection_host } } }
				NOT = { PREV = { is_in_array = { array = camp_admin_known_site_states value = var:camp_admin_origin_projection_host } } }
				camp_admin_gui_host_records_uncertain = yes
			}
			PREV = { add_to_variable = { camp_admin_gui_uncertain_locations = constant:camp_admin_value.one } }
		}
	}
	# Every iteration initializes this scratch target before reading it.
}

# Country scope. Read paid workflow counters into display variables only.
camp_admin_project_institution_progress = {
	set_variable = { camp_admin_gui_institution_progress = constant:camp_admin_value.zero }
	set_variable = { camp_admin_gui_institution_target = constant:camp_admin_value.zero }
	if = {
		limit = { has_country_flag = camp_admin_civil_claims_active }
		set_variable = { camp_admin_gui_institution_progress = camp_admin_civil_claims_progress }
		set_variable = { camp_admin_gui_institution_target = constant:camp_admin_civil.claims_months }
	}
	else_if = {
		limit = { has_country_flag = camp_admin_civil_review_active }
		set_variable = { camp_admin_gui_institution_progress = camp_admin_civil_review_progress }
		set_variable = { camp_admin_gui_institution_target = constant:camp_admin_civil.review_months }
	}
	else_if = {
		limit = { has_country_flag = camp_admin_country_archive_work_active }
		set_variable = { camp_admin_gui_institution_progress = camp_admin_country_archive_progress }
		set_variable = { camp_admin_gui_institution_target = constant:camp_admin_country.archive_months }
	}
	else_if = {
		limit = { has_country_flag = camp_admin_country_oversight_active }
		set_variable = { camp_admin_gui_institution_progress = camp_admin_country_review_progress }
		set_variable = { camp_admin_gui_institution_target = constant:camp_admin_country.oversight_months }
	}
	else_if = {
		limit = {
			camp_admin_country_is_major_kit = yes
			check_variable = { camp_admin_country_program > constant:camp_admin_value.zero }
			OR = {
				has_country_flag = camp_admin_institution_slot_active
				AND = {
					has_country_flag = camp_admin_country_institution_paused
					check_variable = { camp_admin_institution_progress > constant:camp_admin_value.zero }
				}
			}
		}
		set_variable = { camp_admin_gui_institution_progress = camp_admin_institution_progress }
		set_variable = { camp_admin_gui_institution_target = camp_admin_institution_target }
	}
	if = {
		limit = { check_variable = { camp_admin_gui_institution_target > constant:camp_admin_value.zero } }
		divide_variable = { camp_admin_gui_institution_progress = camp_admin_gui_institution_target }
	}
	else = { set_variable = { camp_admin_gui_institution_progress = constant:camp_admin_value.zero } }
	clamp_variable = { var = camp_admin_gui_institution_progress min = constant:camp_admin_value.zero max = constant:camp_admin_value.one }
}

camp_admin_refresh_interface = {
	camp_admin_project_origin_uncertainty = yes
	camp_admin_refresh_command_quotes = yes
	camp_admin_generic_reform_refresh_quote = yes
	if = { limit = { NOT = { has_variable = camp_admin_view } } set_variable = { camp_admin_view = constant:camp_admin_view.administration } }
	if = { limit = { NOT = { has_variable = camp_admin_location_filter } } set_variable = { camp_admin_location_filter = constant:camp_admin_filter.all } }
	clear_array = camp_admin_gui_states
	set_variable = { camp_admin_gui_works_sites = constant:camp_admin_value.zero }
	set_variable = { camp_admin_gui_extraction_sites = constant:camp_admin_value.zero }
	set_variable = { camp_admin_gui_works_factor = constant:camp_admin_value.zero }
	set_variable = { camp_admin_gui_extraction_factor = constant:camp_admin_value.zero }
	save_event_target_as = camp_admin_ui_actor
	clear_array = camp_admin_gui_source_states
	for_each_scope_loop = { array = camp_active_site_states camp_admin_append_display_source = yes }
	for_each_scope_loop = { array = camp_admin_civilian_states camp_admin_append_display_source = yes }
	for_each_scope_loop = { array = camp_admin_known_site_states camp_admin_append_display_source = yes }
	for_each_scope_loop = {
		array = camp_admin_gui_source_states
		if = {
			limit = { camp_admin_gui_site_has_current_output = yes check_variable = { camp_admin_construction_factor > constant:camp_admin_value.zero } }
			set_temp_variable = { camp_admin_display_factor = camp_admin_construction_factor }
			event_target:camp_admin_ui_actor = {
				add_to_variable = { camp_admin_gui_works_sites = constant:camp_admin_value.one }
				add_to_variable = { camp_admin_gui_works_factor = camp_admin_display_factor }
			}
		}
		if = {
			limit = { camp_admin_gui_site_has_current_output = yes check_variable = { camp_admin_extraction_factor > constant:camp_admin_value.zero } }
			set_temp_variable = { camp_admin_display_factor = camp_admin_extraction_factor }
			event_target:camp_admin_ui_actor = {
				add_to_variable = { camp_admin_gui_extraction_sites = constant:camp_admin_value.one }
				add_to_variable = { camp_admin_gui_extraction_factor = camp_admin_display_factor }
			}
		}
		if = {
			limit = {
				camp_admin_gui_site_record_visible = yes
				OR = {
					event_target:camp_admin_ui_actor = { check_variable = { camp_admin_location_filter = constant:camp_admin_filter.all } }
					AND = {
						event_target:camp_admin_ui_actor = { check_variable = { camp_admin_location_filter = constant:camp_admin_filter.labor } }
						OR = { check_variable = { camp_admin_role = constant:camp_admin_role.labor } check_variable = { camp_admin_role = constant:camp_admin_role.penal } }
					}
					AND = {
						event_target:camp_admin_ui_actor = { check_variable = { camp_admin_location_filter = constant:camp_admin_filter.custody } }
						check_variable = { camp_admin_role = constant:camp_admin_role.custody }
					}
					AND = {
						event_target:camp_admin_ui_actor = { check_variable = { camp_admin_location_filter = constant:camp_admin_filter.attention } }
						OR = { camp_admin_gui_host_records_uncertain = yes has_state_flag = camp_admin_site_suspended check_variable = { camp_admin_support_ratio < constant:camp_admin_value.one } }
					}
				}
			}
			set_temp_variable = { camp_admin_display_state = THIS.id }
			event_target:camp_admin_ui_actor = { add_to_array = { array = camp_admin_gui_states value = camp_admin_display_state } }
		}
	}
	clear_array = camp_admin_gui_source_states
	if = {
		limit = { check_variable = { camp_admin_gui_works_sites > constant:camp_admin_value.zero } }
		divide_variable = { camp_admin_gui_works_factor = camp_admin_gui_works_sites }
	}
	if = {
		limit = { check_variable = { camp_admin_gui_extraction_sites > constant:camp_admin_value.zero } }
		divide_variable = { camp_admin_gui_extraction_factor = camp_admin_gui_extraction_sites }
	}
	if = {
		limit = {
			has_variable = camp_selected_state_id
			NOT = { is_in_array = { array = camp_admin_gui_states value = camp_selected_state_id } }
		}
		clear_variable = camp_selected_state_id
	}
	if = {
		limit = { NOT = { has_variable = camp_selected_state_id } check_variable = { camp_admin_gui_states^num > constant:camp_admin_value.zero } }
		set_variable = { camp_selected_state_id = camp_admin_gui_states^0 }
	}
	set_variable = { camp_admin_gui_economy_progress = camp_admin_economy_progress }
	if = { limit = { check_variable = { camp_admin_economy_target > 0 } } divide_variable = { camp_admin_gui_economy_progress = camp_admin_economy_target } }
	else = { set_variable = { camp_admin_gui_economy_progress = 0 } }
	clamp_variable = { var = camp_admin_gui_economy_progress min = 0 max = 1 }
	camp_admin_project_institution_progress = yes
	set_variable = { camp_admin_gui_month_deaths_k = camp_admin_month_deaths_k }
	add_to_variable = { camp_admin_gui_month_deaths_k = camp_occ_month_deaths_k }
	set_variable = { camp_admin_gui_country_support_pct = camp_admin_country_support_ratio }
	multiply_variable = { camp_admin_gui_country_support_pct = constant:camp_admin_value.display_percent_multiplier }
	set_variable = { camp_admin_gui_total_deaths_k = camp_admin_total_deaths_k }
	add_to_variable = { camp_admin_gui_total_deaths_k = camp_occ_total_deaths_k }
	camp_admin_prepare_project_quote = yes
	set_variable = { camp_admin_gui_project_support = camp_admin_quote_support }
	set_variable = { camp_admin_gui_project_pp = camp_admin_quote_pp }
	set_temp_variable = { camp_admin_quote_support = 0 }
	set_temp_variable = { camp_admin_quote_pp = 0 }
	if = {
		limit = { has_variable = camp_selected_state_id }
		var:camp_selected_state_id = { camp_admin_cache_selected_site = yes }
	}
}
