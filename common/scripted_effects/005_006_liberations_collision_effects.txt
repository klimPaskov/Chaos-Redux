# ============================================================================
# CHAOS REDUX - EVENT 005 / EVENT 006 LIBERATIONS COLLISION EFFECTS
# ============================================================================
# Overview:
# - Freezes Event 005's opening republics before Event 006 draws packages.
# - Uses the shared coordinator for absent-tag, unique-anchor, exact-state, and
#   protected-host reservations.
# - Executes both frozen contributions behind one lock and one ownership-
#   mutation barrier when both Liberations cluster members were selected.
# - Cancels the complete joint incident before mutation when an exact Event 006
#   count or any shared invariant cannot be proven. It never falls back to an
#   order-dependent pair of releases.
# - Leaves standalone Event 005 and Event 006 dispatch to their owning events.
# ============================================================================

# Reads event target `soviet_collapse_joint_candidate_country` and builds a
# unique temporary pool of release hosts accepted by Event 005's existing
# SOV/original-union/breakaway/subject owner-controller precedence.
soviet_collapse_joint_add_current_candidate_host = {
	if = {
		limit = {
			exists = yes
			NOT = { is_in_array = { array = soviet_collapse_joint_candidate_host_pool value = THIS } }
		}
		add_to_temp_array = { soviet_collapse_joint_candidate_host_pool = THIS }
	}
}

soviet_collapse_joint_build_candidate_host_pool = {
	clear_temp_array = soviet_collapse_joint_candidate_host_pool
	event_target:soviet_collapse_joint_candidate_country = {
		every_core_state = {
			limit = {
				is_soviet_collapse_state_free_of_independence_wave_origin = yes
				is_liberation_release_state_available_for_reservation = yes
			}
			if = {
				limit = {
					OR = {
						is_owned_by = SOV
						is_controlled_by = SOV
					}
				}
				SOV = { soviet_collapse_joint_add_current_candidate_host = yes }
			}
			if = {
				limit = { has_state_flag = soviet_collapse_original_union_state }
				owner = { soviet_collapse_joint_add_current_candidate_host = yes }
			}
			if = {
				limit = {
					owner = {
						OR = {
							has_country_flag = soviet_collapse_breakaway
							has_country_flag = soviet_collapse_event_created_republic
							has_country_flag = soviet_collapse_high_chaos_successor
							is_subject_of = SOV
						}
					}
				}
				owner = { soviet_collapse_joint_add_current_candidate_host = yes }
			}
			if = {
				limit = {
					controller = {
						OR = {
							has_country_flag = soviet_collapse_breakaway
							has_country_flag = soviet_collapse_event_created_republic
							has_country_flag = soviet_collapse_high_chaos_successor
							is_subject_of = SOV
						}
					}
				}
				controller = { soviet_collapse_joint_add_current_candidate_host = yes }
			}
		}
	}
}

# Attempts one Event 005 candidate against each viable host. The Soviet Union
# is preferred when it owns a usable core; otherwise hosts are drawn without
# replacement. One unique anchor is mandatory. All additional same-host cores
# are optional and are trimmed before the candidate can fail.
soviet_collapse_joint_reserve_current_candidate = {
	set_temp_variable = { soviet_collapse_joint_candidate_reserved = constant:liberations_joint_planner.no }
	set_temp_variable = { soviet_collapse_joint_candidate_country_reserved = constant:liberations_joint_planner.no }
	set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.missing_anchor }
	event_target:soviet_collapse_joint_candidate_country = {
		soviet_collapse_joint_get_current_candidate_package_id = yes
	}
	if = {
		limit = {
			check_variable = {
				var = soviet_collapse_joint_candidate_package_id
				value = constant:liberations_joint_planner.first_array_index
				compare = greater_than
			}
		}
		soviet_collapse_joint_build_candidate_host_pool = yes
		while_loop_effect = {
			limit = {
				check_variable = { soviet_collapse_joint_candidate_reserved = constant:liberations_joint_planner.no }
				check_variable = {
					var = soviet_collapse_joint_candidate_host_pool^num
					value = constant:liberations_joint_planner.no
					compare = greater_than
				}
			}
			if = {
				limit = {
					SOV = {
						is_in_array = {
							array = soviet_collapse_joint_candidate_host_pool
							value = THIS
						}
					}
				}
				SOV = { save_event_target_as = soviet_collapse_joint_candidate_host }
			}
			else = {
				random_scope_in_array = {
					array = soviet_collapse_joint_candidate_host_pool
					save_event_target_as = soviet_collapse_joint_candidate_host
				}
			}
			remove_from_temp_array = {
				array = soviet_collapse_joint_candidate_host_pool
				value = event_target:soviet_collapse_joint_candidate_host
			}
			set_temp_variable = { soviet_collapse_joint_candidate_anchor_found = constant:liberations_joint_planner.no }
			event_target:soviet_collapse_joint_candidate_country = {
				random_core_state = {
					limit = {
						is_owned_by = event_target:soviet_collapse_joint_candidate_host
						is_soviet_collapse_state_free_of_independence_wave_origin = yes
						is_liberation_release_state_available_for_reservation = yes
					}
					save_event_target_as = soviet_collapse_joint_candidate_anchor
					set_temp_variable = { soviet_collapse_joint_candidate_anchor_found = constant:liberations_joint_planner.yes }
				}
			}
			if = {
				limit = { check_variable = { soviet_collapse_joint_candidate_anchor_found = constant:liberations_joint_planner.yes } }
				event_target:soviet_collapse_joint_candidate_anchor = { save_event_target_as = liberation_candidate_anchor }
				event_target:soviet_collapse_joint_candidate_host = {
					liberation_release_select_and_reserve_host_state = yes
				}
			}
			if = {
				limit = {
					check_variable = { soviet_collapse_joint_candidate_anchor_found = constant:liberations_joint_planner.yes }
					check_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none }
				}
				if = {
					limit = { check_variable = { soviet_collapse_joint_candidate_anchor_found = constant:liberations_joint_planner.yes } }
					event_target:soviet_collapse_joint_candidate_country = { save_event_target_as = liberation_candidate_country }
					event_target:soviet_collapse_joint_candidate_host = { save_event_target_as = liberation_candidate_primary_host }
					set_temp_variable = { liberation_candidate_state_row_start = global.liberation_plan_states^num }
					set_temp_variable = { liberation_candidate_plan_owner = constant:liberation_plan_owner.soviet_collapse }
					set_temp_variable = { liberation_candidate_package_id = soviet_collapse_joint_candidate_package_id }
					set_temp_variable = { liberation_candidate_reservation_group = constant:liberation_release_limit.minimum }
					set_temp_variable = { liberation_candidate_territory_level = constant:liberation_territory_level.anchor }
					set_temp_variable = { liberation_candidate_force_level = constant:liberation_force_level.fragile }
					liberation_release_add_country_reservation = yes
					if = {
						limit = { check_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none } }
						set_temp_variable = { soviet_collapse_joint_candidate_country_reserved = constant:liberations_joint_planner.yes }
						event_target:soviet_collapse_joint_candidate_host = { save_event_target_as = liberation_candidate_state_host }
						liberation_release_measure_candidate_host_loss_capacity = yes
						if = {
							limit = { check_variable = { liberation_candidate_host_has_loss_capacity = constant:liberation_release_limit.one } }
							set_temp_variable = { liberation_candidate_state_role = constant:liberation_plan_state_role.anchor }
							event_target:soviet_collapse_joint_candidate_anchor = {
								liberation_release_add_state_reservation = yes
							}
						}
						else = {
							set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.host_survival }
						}
					}
					if = {
						limit = { check_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none } }
						set_temp_variable = { soviet_collapse_joint_candidate_reserved = constant:liberations_joint_planner.yes }
						set_temp_variable = { soviet_collapse_joint_candidate_highest_territory = constant:liberation_territory_level.anchor }
						set_temp_variable = { soviet_collapse_joint_candidate_country_row = global.liberation_plan_countries^num }
						subtract_from_temp_variable = { soviet_collapse_joint_candidate_country_row = constant:liberations_joint_planner.loop_increment }
						set_variable = {
							global.liberation_plan_territory_levels^soviet_collapse_joint_candidate_country_row = soviet_collapse_joint_candidate_highest_territory
						}
						event_target:soviet_collapse_joint_candidate_country = {
							set_country_flag = soviet_collapse_joint_pending_origin
							set_variable = { soviet_collapse_joint_pending_plan_id = global.liberation_plan_id }
							set_variable = { soviet_collapse_joint_pending_package_id = soviet_collapse_joint_candidate_package_id }
							set_variable = { soviet_collapse_joint_pending_family = soviet_collapse_joint_current_family }
							clear_variable = soviet_collapse_joint_last_rejected_plan_id
						}
						add_to_array = {
							array = global.soviet_collapse_joint_plan_selected_countries
							value = event_target:soviet_collapse_joint_candidate_country
						}
						add_to_array = {
							array = global.soviet_collapse_joint_plan_selected_package_ids
							value = soviet_collapse_joint_candidate_package_id
						}
						add_to_array = {
							array = global.soviet_collapse_joint_plan_selected_families
							value = soviet_collapse_joint_current_family
						}
						add_to_array = {
							array = global.soviet_collapse_joint_plan_country_row_indices
							value = soviet_collapse_joint_candidate_country_row
						}
						add_to_array = {
							array = global.soviet_collapse_first_wave_republics
							value = event_target:soviet_collapse_joint_candidate_country
						}
						add_to_variable = {
							global.soviet_collapse_joint_plan_selected_count = constant:liberations_joint_planner.loop_increment
						}
						set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none }
					}
					else_if = {
						limit = { check_variable = { soviet_collapse_joint_candidate_country_reserved = constant:liberations_joint_planner.yes } }
						liberation_release_rollback_candidate_reservation = yes
						set_temp_variable = { soviet_collapse_joint_candidate_country_reserved = constant:liberations_joint_planner.no }
					}
					else = { liberation_release_cleanup_unused_host_reservations = yes }
				}
				else = {
					set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.missing_anchor }
					liberation_release_cleanup_unused_host_reservations = yes
				}
			}
		}
	}
	if = {
		limit = { check_variable = { soviet_collapse_joint_candidate_reserved = constant:liberations_joint_planner.no } }
		if = {
			limit = { check_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.none } }
			set_temp_variable = { liberation_candidate_reject_reason = constant:liberation_plan_reject_reason.missing_anchor }
		}
		event_target:soviet_collapse_joint_candidate_country = {
			set_variable = { soviet_collapse_joint_last_rejected_plan_id = global.liberation_plan_id }
		}
		set_temp_variable = { liberation_rejected_package_id = soviet_collapse_joint_candidate_package_id }
		set_temp_variable = { liberation_reject_reason = liberation_candidate_reject_reason }
		liberation_release_record_rejection = yes
	}
	clear_temp_array = soviet_collapse_joint_candidate_host_pool
}

# Fills one category slot and rerolls rejected tags without replacement. A
# category with no valid tag contributes zero, matching Event 005's existing
# first-wave semantics rather than reducing Event 006's exact wave count.
soviet_collapse_joint_fill_current_family_slot = {
	set_temp_variable = { soviet_collapse_joint_slot_filled = constant:liberations_joint_planner.no }
	set_temp_variable = { soviet_collapse_joint_slot_attempts = constant:liberations_joint_planner.no }
	while_loop_effect = {
		limit = {
			check_variable = { soviet_collapse_joint_slot_filled = constant:liberations_joint_planner.no }
			check_variable = {
				var = soviet_collapse_joint_slot_attempts
				value = constant:liberations_joint_planner.maximum_event5_slot_attempts
				compare = less_than
			}
		}
		add_to_temp_variable = { soviet_collapse_joint_slot_attempts = constant:liberations_joint_planner.loop_increment }
		add_to_variable = { global.soviet_collapse_joint_plan_attempt_count = constant:liberations_joint_planner.loop_increment }
		soviet_collapse_joint_build_current_family_pool = yes
		if = {
			limit = {
				check_variable = {
					var = soviet_collapse_joint_family_pool^num
					value = constant:liberations_joint_planner.no
					compare = greater_than
				}
			}
			random_scope_in_array = {
				array = soviet_collapse_joint_family_pool
				save_event_target_as = soviet_collapse_joint_candidate_country
			}
			soviet_collapse_joint_reserve_current_candidate = yes
			if = {
				limit = { check_variable = { soviet_collapse_joint_candidate_reserved = constant:liberations_joint_planner.yes } }
				set_temp_variable = { soviet_collapse_joint_slot_filled = constant:liberations_joint_planner.yes }
			}
		}
		else = {
			set_temp_variable = { soviet_collapse_joint_slot_attempts = constant:liberations_joint_planner.maximum_event5_slot_attempts }
		}
	}
	clear_temp_array = soviet_collapse_joint_family_pool
}

# Counts central-Asian republics that are already Event 005 breakaways or are
# frozen in this plan. This reproduces the release-then-Kazakhstan gate without
# mutating ownership before Event 006 has drawn.
soviet_collapse_joint_measure_central_asian_plan = {
	set_temp_variable = { soviet_collapse_joint_smaller_central_asian_count = constant:liberations_joint_planner.no }
	if = {
		limit = {
			OR = {
				UZB = { exists = yes has_country_flag = soviet_collapse_breakaway }
				UZB = { is_in_array = { array = global.soviet_collapse_joint_plan_selected_countries value = THIS } }
			}
		}
		add_to_temp_variable = { soviet_collapse_joint_smaller_central_asian_count = constant:liberations_joint_planner.loop_increment }
	}
	if = {
		limit = {
			OR = {
				KYR = { exists = yes has_country_flag = soviet_collapse_breakaway }
				KYR = { is_in_array = { array = global.soviet_collapse_joint_plan_selected_countries value = THIS } }

					set_variable = { global.liberation_plan_finalization_failure = constant:liberation_finalization_failure.event5_package_setup }
					liberation_release_mark_finalization_failed = yes
				}
			}
		}
	}
	if = {
		limit = {
			NOT = { check_variable = { liberations_joint_dispatch_result = constant:liberations_joint_result.committed } }
		}
		if = {
			limit = {
				check_variable = { liberations_joint_plan_begun = constant:liberations_joint_planner.yes }
				has_liberation_release_plan_finalization_failed = yes
			}
			set_global_flag = liberations_joint_incident_failed_after_mutation
			set_global_flag = liberations_joint_incident_failed_during_finalization
			set_temp_variable = { liberations_joint_dispatch_result = constant:liberations_joint_result.cancelled }
		}
		else_if = {
			limit = {
				check_variable = { liberations_joint_plan_begun = constant:liberations_joint_planner.yes }
				has_global_flag = liberation_release_execution_started
			}
			liberations_joint_record_failure_after_ownership_mutation = yes
		}
		else_if = {
			limit = { check_variable = { liberations_joint_plan_begun = constant:liberations_joint_planner.yes } }
			liberations_joint_cancel_before_ownership_mutation = yes
		}
		else = {
			set_global_flag = liberations_joint_incident_cancelled_before_mutation
			set_temp_variable = { liberations_joint_dispatch_result = constant:liberations_joint_result.cancelled }
		}
	}
}
