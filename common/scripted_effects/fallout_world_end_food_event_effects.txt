# CHAOS REDUX - FALLOUT THE LAST INVENTORY EFFECTS
# ============================================================================
# Dormant receipt-authenticated food-security chain. The opening request is
# consumed by one coordinator, result and callback rows use the Fallout-owned
# delayed queue, and all population loss remains in the shared Deaths API.
# The file never sets scheduler activation flags.
# ============================================================================

fallout_event_100_record_history = {
	set_temp_variable = { fallout_event_100_history_should_record = constant:fallout_event_runtime.zero }
	if = {
		limit = {
			check_variable = { fallout_event_100_history_stage = constant:fallout_event_runtime.one }
			NOT = { has_country_flag = fallout_event_100_result_history_recorded }
		}
		set_temp_variable = { fallout_event_100_history_should_record = constant:fallout_event_runtime.one }
	}
	else_if = {
		limit = {
			check_variable = { fallout_event_100_history_stage = constant:fallout_event_runtime.two }
			NOT = { has_country_flag = fallout_event_100_callback_history_recorded }
		}
		set_temp_variable = { fallout_event_100_history_should_record = constant:fallout_event_runtime.one }
	}
	if = {
		limit = { check_variable = { fallout_event_100_history_should_record = constant:fallout_event_runtime.one } }
		set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.callback_failure }
		if = {
			limit = { check_variable = { fallout_event_100_history_stage = constant:fallout_event_runtime.one } }
			if = {
				limit = { check_variable = { fallout_event_100_selected_branch = constant:fallout_event_100_branch.publish_strict_rations } }
				if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.public_success } }
				else_if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.public_partial } }
				else = { set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.public_failure } }
			}
			else_if = {
				limit = { check_variable = { fallout_event_100_selected_branch = constant:fallout_event_100_branch.hide_shortage_protect_caches } }
				if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.household_success } }
				else_if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.household_partial } }
				else = { set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.household_failure } }
			}
			else_if = {
				limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.success } }
				set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.requisition_success }
			}
			else_if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.requisition_partial } }
			else = { set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.requisition_failure } }
		}
		else_if = {
			if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.success } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.callback_success } }
			else_if = { limit = { check_variable = { fallout_event_100_result_outcome = constant:fallout_event_outcome_type.partial } } set_temp_variable = { fallout_event_100_history_payload = constant:fallout_event_100_log.callback_partial } }
		}
		set_temp_variable = { events_log_system_event_id = constant:fallout_event_100_log.history_id }
		set_temp_variable = { events_log_system_event_type = constant:event_system_event_type.fallout_country_memory }
		set_temp_variable = { events_log_system_payload = fallout_event_100_history_payload }
		set_temp_variable = { events_log_system_actor = ROOT.id }
		set_temp_variable = { events_log_system_has_actor = constant:fallout_event_runtime.one }
		set_temp_variable = { events_log_system_secondary_actor = constant:fallout_event_runtime.zero }
		set_temp_variable = { events_log_system_has_secondary_actor = constant:fallout_event_runtime.zero }
		record_events_log_system_history_entry = yes
		refresh_events_log_system_history_views = yes
		if = { limit = { check_variable = { fallout_event_100_history_stage = constant:fallout_event_runtime.one } } set_country_flag = fallout_event_100_result_history_recorded }
		else = { set_country_flag = fallout_event_100_callback_history_recorded }
	}
	clear_temp_variable = fallout_event_100_history_should_record
	clear_temp_variable = fallout_event_100_history_payload
	clear_temp_variable = events_log_system_event_id
	clear_temp_variable = events_log_system_event_type
	clear_temp_variable = events_log_system_payload
	clear_temp_variable = events_log_system_actor
	clear_temp_variable = events_log_system_has_actor
	clear_temp_variable = events_log_system_secondary_actor
	clear_temp_variable = events_log_system_has_secondary_actor
	clear_temp_variable = fallout_event_100_history_stage
}

fallout_event_100_apply_requisition_failure_deaths = {
	set_variable = { fallout_event_100_deaths_applied = constant:fallout_event_runtime.zero }
	every_owned_state = {
		if = {
			limit = { check_variable = { state_population_k > constant:fallout_event_runtime.zero } }
			set_temp_variable = { fallout_event_100_state_death_request = state_population_k }
			multiply_temp_variable = { fallout_event_100_state_death_request = constant:chaos_meter_deaths.people_per_k }
			multiply_temp_variable = { fallout_event_100_state_death_request = constant:fallout_event_100_threshold.requisition_failure_death_percent }
			round_temp_variable = fallout_event_100_state_death_request
			set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_requested = fallout_event_100_state_death_request }
			set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:fallout_event_100_threshold.minimum_remaining_people }
			set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.fallout_aftermath }
			set_temp_variable = { state_civilian_population_loss_log_deaths = constant:fallout_event_runtime.one }
			set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
			set_temp_variable = { state_civilian_population_loss_has_target_country = constant:fallout_event_runtime.one }
			apply_exact_state_civilian_population_loss = yes
			add_to_variable = { ROOT.fallout_event_100_deaths_applied = state_civilian_population_loss_applied }
		}
	}
	clear_temp_variable = fallout_event_100_state_death_request
}

fallout_event_100_calculate_outcome = {
	set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.failure }
	if = {
		limit = { check_variable = { fallout_event_100_requested_branch = constant:fallout_event_100_branch.publish_strict_rations } }
		if = {
			limit = {
				check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_success compare = greater_than_or_equals }
				check_variable = { fallout_survival_cohesion_current value = constant:fallout_event_100_threshold.cohesion_success compare = greater_than_or_equals }
			}
			set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.success }
		}
		else_if = { limit = { check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_partial compare = greater_than_or_equals } } set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	else_if = {
		limit = { check_variable = { fallout_event_100_requested_branch = constant:fallout_event_100_branch.hide_shortage_protect_caches } }
		if = {
			limit = {
				check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_success compare = greater_than_or_equals }
				check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_100_threshold.recognition_household_success compare = greater_than_or_equals }
			}
			set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.success }
		}
		else_if = { limit = { check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_partial compare = greater_than_or_equals } } set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
	else_if = {
		if = {
			limit = {
				check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_success compare = greater_than_or_equals }
				OR = {
					has_government = fascism
					has_government = neutrality
				}
			}
			set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.success }
		}
		else_if = { limit = { check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_partial compare = greater_than_or_equals } } set_temp_variable = { fallout_event_100_calculated_outcome = constant:fallout_event_outcome_type.partial } }
	}
}

			check_variable = { fallout_event_100_requested_branch < constant:fallout_event_100_branch.upper_bound }
		}

		set_temp_variable = { fallout_event_requested_cleanup_token = constant:fallout_event_id.last_inventory_cleanup }

			limit = { check_variable = { fallout_event_transaction_accepted = constant:fallout_event_runtime.one } }
			set_temp_variable = { fallout_event_requested_event_token = fallout_event_100_opening_event_token }

fallout_event_100_choose_ai_branch_and_schedule_result = {
	set_temp_variable = { fallout_event_100_requested_branch = constant:fallout_event_100_branch.hide_shortage_protect_caches }
	if = {
		limit = { check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_100_threshold.food_partial compare = less_than } }
		set_temp_variable = { fallout_event_100_requested_branch = constant:fallout_event_100_branch.requisition_private_stores }
	}
	else_if = {
		limit = {
			OR = {
				has_government = democratic

		set_temp_variable = { fallout_event_100_requested_branch = constant:fallout_event_100_branch.publish_strict_rations }
