# CHAOS REDUX - EVENT 006 OYO, ACEH AND MON SIGNATURE EFFECTS
# Overview:
# - Registers the three accepted Level 3 country modules and owns their
#   package-specific route, project, formable-entry and cleanup effects.
# - State construction uses the current owned Event 006 anchor and checks native
#   construction legality again when each timed project expires.
# - Aceh's naval base and Mon's bunker use their exact vanilla anchor provinces.
# - Paid PUSA education and Mon river services publish durable generation awards
#   before rewards; cleanup preserves award history while removing live content.
# - Regional federation entry delegates to the existing ready-formable
#   selection and readiness contract; it does not transfer territory.
# - The CXT adapter iterates only the existing aligned active-country registry;
#   no world-country or state scan is introduced.
# ============================================================================

# COUNTRY scope. The startup/daily registration adapter places this inert
# carrier in the CXT extension registry. Applying it only revisits the three
# exact, live Event 006 package rows in the aligned active-country registry.
# It never grants country content or creates a released package.
independence_wave_signature_modules_register_cxt_test_content = {
	set_temp_variable = { chaosx_test_country_extension_registration_added = constant:independence_wave_signature_cxt.registration_none }
	set_temp_variable = {
		var = chaosx_test_country_registration_extension_effect
		value = token:independence_wave_signature_modules_cxt_extension
	}
	chaosx_test_country_register_extension_effect = yes
}

independence_wave_signature_modules_cxt_extension_apply = {
	if = {
		limit = {
			tag = CXT
			has_country_flag = chaosx_test_country_initialized
			independence_wave_active_country_arrays_are_aligned = yes
		}
		for_each_scope_loop = {
			array = global.independence_wave_active_countries
			if = {
				limit = { is_independence_wave_signature_modules_exact_active_package = yes }
				if = {
					limit = {
						original_tag = DSX
						check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_097 }
					}
					independence_wave_dsx_signature_setup = yes
				}
				if = {
					limit = {
						original_tag = ATJ
						check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_150 }
					}
					independence_wave_atj_signature_setup = yes
					if = {
						limit = {
							has_country_flag = independence_wave_formable_family_selected
							has_variable = independence_wave_formable_family
							check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.maritime_federation }
							NOT = { has_country_flag = independence_wave_form36_readiness_attested }
							NOT = { has_country_flag = independence_wave_formable_active }
							NOT = { has_country_flag = independence_wave_formable_committed }
							NOT = { has_country_flag = independence_wave_formable_commit_pending }
							NOT = { has_pending_independence_wave_formable_founding_invitation = yes }
							OR = {
								NOT = { has_variable = independence_wave_formable_consent_count }
								check_variable = { var = independence_wave_formable_consent_count value = constant:independence_wave_formable_tuning.no_entries compare = equals }
							}
						}
						# The parent-owned relay refreshes readiness with the released
						# country as ROOT, preserving this bounded CXT repair path.
						independence_wave_ready_formable_request_cxt_readiness_refresh = yes
					}
				}
				if = {
					limit = {
						original_tag = IAX
						check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_161 }
					}
					independence_wave_iax_signature_setup = yes
					if = {
						limit = {
							has_country_flag = independence_wave_formable_family_selected
							has_variable = independence_wave_formable_family
							check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.mainland_southeast_asian_federation }
							NOT = { has_country_flag = independence_wave_form37_readiness_attested }
							NOT = { has_country_flag = independence_wave_formable_active }
							NOT = { has_country_flag = independence_wave_formable_committed }
							NOT = { has_country_flag = independence_wave_formable_commit_pending }
							NOT = { has_pending_independence_wave_formable_founding_invitation = yes }
							OR = {
								NOT = { has_variable = independence_wave_formable_consent_count }
								check_variable = { var = independence_wave_formable_consent_count value = constant:independence_wave_formable_tuning.no_entries compare = equals }
							}
						}
						independence_wave_ready_formable_request_cxt_readiness_refresh = yes
					}
				}
			}
		}
	}
}
