sanitize_events_log_history_actor_entries = {
    if = {
        limit = {
            check_variable = {
                var = global.events_log_history_event_id_entries^num
                value = global.events_log_history_actor_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_actor_entries = global.events_log_history_event_id_entries^num }
        subtract_from_temp_variable = { missing_actor_entries = global.events_log_history_actor_entries^num }

        while_loop_effect = {
            limit = { check_variable = { missing_actor_entries > 0 } }
            add_to_array = {
                array = global.events_log_history_actor_entries
                value = 0
            }
            add_to_array = {
                array = global.events_log_history_has_actor_entries
                value = 0
            }
            subtract_from_temp_variable = { missing_actor_entries = 1 }
        }
    }

    if = {
        limit = {
            check_variable = {
                var = global.events_log_history_event_id_entries^num
                value = global.events_log_history_has_actor_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_has_actor_entries = global.events_log_history_event_id_entries^num }
        subtract_from_temp_variable = { missing_has_actor_entries = global.events_log_history_has_actor_entries^num }

        while_loop_effect = {
            limit = { check_variable = { missing_has_actor_entries > 0 } }
            add_to_array = {
                array = global.events_log_history_has_actor_entries
                value = 0
            }
            subtract_from_temp_variable = { missing_has_actor_entries = 1 }
        }
    }

    set_temp_variable = { events_log_sanitize_index = 0 }
    while_loop_effect = {
        limit = {
            check_variable = {
                var = events_log_sanitize_index
                value = global.events_log_history_event_id_entries^num
                compare = less_than
            }
        }

        if = {
            limit = {
                check_variable = {
                    var = global.events_log_history_has_actor_entries^events_log_sanitize_index
                    value = 1
                    compare = less_than
                }
            }
            set_variable = { global.events_log_history_actor_entries^events_log_sanitize_index = 0 }
            set_variable = { global.events_log_history_has_actor_entries^events_log_sanitize_index = 0 }
        }

        if = {
            limit = {
                check_variable = {
                    var = global.events_log_history_has_actor_entries^events_log_sanitize_index
                    value = 0
                    compare = greater_than
                }
                check_variable = {
                    var = global.events_log_history_actor_entries^events_log_sanitize_index
                    value = 1
                    compare = less_than
                }
            }
            set_variable = { global.events_log_history_actor_entries^events_log_sanitize_index = 0 }
            set_variable = { global.events_log_history_has_actor_entries^events_log_sanitize_index = 0 }
        }

        add_to_temp_variable = { events_log_sanitize_index = 1 }
    }
	sanitize_events_log_history_secondary_actor_entries = yes
	sanitize_events_log_history_payload_entries = yes
}

		subtract_from_temp_variable = { events_log_missing_payload_entries = global.events_log_history_payload_entries^num }

			limit = { check_variable = { events_log_missing_payload_entries > 0 } }

		}

}

sanitize_events_log_history_secondary_actor_entries = {

		}

		subtract_from_temp_variable = { events_log_missing_secondary_actor_entries = global.events_log_history_secondary_actor_entries^num }
		while_loop_effect = {
			limit = { check_variable = { events_log_missing_secondary_actor_entries > 0 } }
			add_to_array = { array = global.events_log_history_secondary_actor_entries value = 0 }
			add_to_array = { array = global.events_log_history_has_secondary_actor_entries value = 0 }
			subtract_from_temp_variable = { events_log_missing_secondary_actor_entries = 1 }
		}
	}
	if = {
		limit = {
			check_variable = {
				var = global.events_log_history_event_id_entries^num
				value = global.events_log_history_has_secondary_actor_entries^num
				compare = greater_than
			}
		}
		set_temp_variable = { events_log_missing_has_secondary_actor_entries = global.events_log_history_event_id_entries^num }
		subtract_from_temp_variable = { events_log_missing_has_secondary_actor_entries = global.events_log_history_has_secondary_actor_entries^num }
		while_loop_effect = {
			limit = { check_variable = { events_log_missing_has_secondary_actor_entries > 0 } }
			add_to_array = { array = global.events_log_history_has_secondary_actor_entries value = 0 }
			subtract_from_temp_variable = { events_log_missing_has_secondary_actor_entries = 1 }
		}
	}
	set_temp_variable = { events_log_secondary_sanitize_index = 0 }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = events_log_secondary_sanitize_index
				value = global.events_log_history_event_id_entries^num
				compare = less_than
			}
		}
		if = {
			limit = {
				check_variable = {
					var = global.events_log_history_has_secondary_actor_entries^events_log_secondary_sanitize_index
					value = 1
					compare = less_than
				}
			}
			set_variable = { global.events_log_history_secondary_actor_entries^events_log_secondary_sanitize_index = 0 }
			set_variable = { global.events_log_history_has_secondary_actor_entries^events_log_secondary_sanitize_index = 0 }
		}
		if = {
			limit = {
				check_variable = {
					var = global.events_log_history_has_secondary_actor_entries^events_log_secondary_sanitize_index
					value = 0
					compare = greater_than
				}
				check_variable = {
					var = global.events_log_history_secondary_actor_entries^events_log_secondary_sanitize_index
					value = 1
					compare = less_than
				}
			}
			set_variable = { global.events_log_history_secondary_actor_entries^events_log_secondary_sanitize_index = 0 }
			set_variable = { global.events_log_history_has_secondary_actor_entries^events_log_secondary_sanitize_index = 0 }
		}
		add_to_temp_variable = { events_log_secondary_sanitize_index = 1 }
	}
}

# Appends a system-owned history row without borrowing global.last_fired_event_id.
# Inputs are temporary variables set by the calling system:
# events_log_system_event_id, events_log_system_event_type,
# events_log_system_payload, events_log_system_actor, and
# events_log_system_has_actor. The caller must finish its private payload row
# and then call refresh_events_log_system_history_views.
record_events_log_system_history_entry = {
	if = {
		limit = { NOT = { has_variable = global.events_log_history_sequence } }
		set_variable = { global.events_log_history_sequence = 0 }
	}

	sanitize_events_log_history_actor_entries = yes

	set_temp_variable = { event_id = events_log_system_event_id }
	set_temp_variable = { event_type = events_log_system_event_type }
	set_temp_variable = { events_log_entry_payload = events_log_system_payload }
	set_temp_variable = { events_log_default_actor = events_log_system_actor }
	set_temp_variable = { events_log_default_has_actor = events_log_system_has_actor }
	set_temp_variable = { events_log_entry_secondary_actor = events_log_system_secondary_actor }
	set_temp_variable = { events_log_entry_has_secondary_actor = events_log_system_has_secondary_actor }

	set_temp_variable = { total_unfired_events = global.unfired_major_events_count }
	add_to_temp_variable = { total_unfired_events = global.unfired_fire_once_events_count }
	add_to_temp_variable = { total_unfired_events = global.unfired_repeatable_events_count }

	add_to_variable = { global.events_log_history_sequence = 1 }

	add_to_array = {
		array = global.events_log_history_sequence_entries
		value = global.events_log_history_sequence
		index = 0

sanitize_events_log_evolution_actor_entries = {
    if = {
        limit = {
            check_variable = {
                var = global.events_log_evolution_sequence_entries^num
                value = global.events_log_evolution_event_id_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_evolution_event_id_entries = global.events_log_evolution_sequence_entries^num }
        subtract_from_temp_variable = { missing_evolution_event_id_entries = global.events_log_evolution_event_id_entries^num }
        while_loop_effect = {
            limit = { check_variable = { missing_evolution_event_id_entries > 0 } }
            add_to_array = {
                array = global.events_log_evolution_event_id_entries
                value = 0
            }
            subtract_from_temp_variable = { missing_evolution_event_id_entries = 1 }
        }
    }

    if = {
        limit = {
            check_variable = {
                var = global.events_log_evolution_sequence_entries^num
                value = global.events_log_evolution_event_type_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_evolution_event_type_entries = global.events_log_evolution_sequence_entries^num }
        subtract_from_temp_variable = { missing_evolution_event_type_entries = global.events_log_evolution_event_type_entries^num }
        while_loop_effect = {
            limit = { check_variable = { missing_evolution_event_type_entries > 0 } }
            add_to_array = {
                array = global.events_log_evolution_event_type_entries
                value = constant:event_system_event_type.unknown
            }
            subtract_from_temp_variable = { missing_evolution_event_type_entries = 1 }
        }
    }

    if = {
        limit = {
            check_variable = {
                var = global.events_log_evolution_sequence_entries^num
                value = global.events_log_evolution_tier_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_evolution_tier_entries = global.events_log_evolution_sequence_entries^num }
        subtract_from_temp_variable = { missing_evolution_tier_entries = global.events_log_evolution_tier_entries^num }
        while_loop_effect = {
            limit = { check_variable = { missing_evolution_tier_entries > 0 } }
            add_to_array = {
                array = global.events_log_evolution_tier_entries
                value = 0
            }
            subtract_from_temp_variable = { missing_evolution_tier_entries = 1 }
        }
    }

    if = {
        limit = {
            check_variable = {
                var = global.events_log_evolution_sequence_entries^num
                value = global.events_log_evolution_actor_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_evolution_actor_entries = global.events_log_evolution_sequence_entries^num }
        subtract_from_temp_variable = { missing_evolution_actor_entries = global.events_log_evolution_actor_entries^num }

        while_loop_effect = {
            limit = { check_variable = { missing_evolution_actor_entries > 0 } }
            add_to_array = {
                array = global.events_log_evolution_actor_entries
                value = 0
            }
            add_to_array = {
                array = global.events_log_evolution_has_actor_entries
                value = 0
            }
            subtract_from_temp_variable = { missing_evolution_actor_entries = 1 }
        }
    }

    if = {
        limit = {
            check_variable = {
                var = global.events_log_evolution_sequence_entries^num
                value = global.events_log_evolution_has_actor_entries^num
                compare = greater_than
            }
        }
        set_temp_variable = { missing_evolution_has_actor_entries = global.events_log_evolution_sequence_entries^num }
        subtract_from_temp_variable = { missing_evolution_has_actor_entries = global.events_log_evolution_has_actor_entries^num }

        while_loop_effect = {
            limit = { check_variable = { missing_evolution_has_actor_entries > 0 } }
            add_to_array = {
                array = global.events_log_evolution_has_actor_entries
                value = 0
            }
            subtract_from_temp_variable = { missing_evolution_has_actor_entries = 1 }
        }
    }

    set_temp_variable = { events_log_evolution_sanitize_index = 0 }
    while_loop_effect = {
        limit = {
            check_variable = {
                var = events_log_evolution_sanitize_index
                value = global.events_log_evolution_sequence_entries^num
                compare = less_than
            }
        }

        if = {
            limit = {
                check_variable = {
                    var = global.events_log_evolution_has_actor_entries^events_log_evolution_sanitize_index
                    value = 1
                    compare = less_than
                }
            }
            set_variable = { global.events_log_evolution_actor_entries^events_log_evolution_sanitize_index = 0 }
            set_variable = { global.events_log_evolution_has_actor_entries^events_log_evolution_sanitize_index = 0 }
        }

        if = {
            limit = {
                check_variable = {
                    var = global.events_log_evolution_has_actor_entries^events_log_evolution_sanitize_index
                    value = 0
                    compare = greater_than
                }
                check_variable = {
                    var = global.events_log_evolution_actor_entries^events_log_evolution_sanitize_index
                    value = 1
                    compare = less_than
                }
            }
            set_variable = { global.events_log_evolution_actor_entries^events_log_evolution_sanitize_index = 0 }
            set_variable = { global.events_log_evolution_has_actor_entries^events_log_evolution_sanitize_index = 0 }
        }

        add_to_temp_variable = { events_log_evolution_sanitize_index = 1 }
    }
}

        array = global.events_log_evolution_event_id_entries

        index = 0

        index = 0
    }
    add_to_array = {
        array = global.events_log_evolution_tier_entries
        value = events_log_evolution_tier
        index = 0
    }
    add_to_array = {
        array = global.events_log_evolution_stage_entries
        value = events_log_evolution_stage

events_log_rebuild_event_detail_world_end_scenarios = {
	clear_array = global.events_log_event_detail_world_end_scenario_id_entries
	clear_array = global.events_log_event_detail_world_end_owner_event_id_entries
	clear_array = global.events_log_event_detail_world_end_sort_order_entries
	clear_array = global.events_log_event_detail_world_end_enabled_entries
	clear_array = global.events_log_event_detail_world_end_active_entries
	clear_array = global.events_log_event_detail_world_end_available_entries
	clear_array = global.events_log_event_detail_world_end_scenario_flag_entries
	clear_array = global.events_log_event_detail_world_end_super_event_id_entries
	clear_array = global.events_log_event_detail_world_end_title_key_entries
	clear_array = global.events_log_event_detail_world_end_details_key_entries
	clear_array = global.events_log_event_detail_world_end_availability_helper_entries

	set_temp_variable = { events_log_world_end_registry_index = 0 }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = events_log_world_end_registry_index
				value = global.world_end_scenario_registry_id_entries^num
				compare = less_than
			}
		}

		if = {
			limit = {
				check_variable = { var = global.world_end_scenario_registry_owner_event_id_entries^events_log_world_end_registry_index value = event_id compare = equals }
				check_variable = { var = global.world_end_scenario_registry_visibility_entries^events_log_world_end_registry_index value = constant:world_end_scenario_visibility.public compare = equals }
				check_variable = { var = global.world_end_scenario_registry_public_details_ready_entries^events_log_world_end_registry_index value = constant:world_end_scenario_public_details.ready compare = equals }
				OR = {
					NOT = {
						OR = {
							check_variable = { var = global.world_end_scenario_registry_id_entries^events_log_world_end_registry_index value = constant:world_end_scenario_id.world_is_the_larder compare = equals }
							check_variable = { var = global.world_end_scenario_registry_id_entries^events_log_world_end_registry_index value = constant:world_end_scenario_id.no_thaw_will_come compare = equals }
						}
					}
					has_global_flag = cannibalism_reveal_complete
				}
			}

			set_temp_variable = { events_log_world_end_scenario_id = global.world_end_scenario_registry_id_entries^events_log_world_end_registry_index }
			set_temp_variable = { events_log_world_end_owner_event_id = global.world_end_scenario_registry_owner_event_id_entries^events_log_world_end_registry_index }
			set_temp_variable = { events_log_world_end_sort_order = global.world_end_scenario_registry_sort_order_entries^events_log_world_end_registry_index }

			set_temp_variable = { events_log_world_end_super_event_id = global.world_end_scenario_registry_super_event_id_entries^events_log_world_end_registry_index }
			set_temp_variable = { events_log_world_end_title_key = global.world_end_scenario_registry_title_key_entries^events_log_world_end_registry_index }
			set_temp_variable = { events_log_world_end_details_key = global.world_end_scenario_registry_details_key_entries^events_log_world_end_registry_index }
			set_temp_variable = { events_log_world_end_scenario_availability_helper = global.world_end_scenario_registry_availability_helper_entries^events_log_world_end_registry_index }

				}
				add_to_temp_variable = { events_log_world_end_sort_scan_index = 1 }
			}

			# add_to_array only documents a literal index. Inject the computed index
			# so every aligned field is inserted at the same stable sort position.

				text = {
					add_to_array = { array = global.events_log_event_detail_world_end_scenario_id_entries value = events_log_world_end_scenario_id index = [INDEX] }

					add_to_array = { array = global.events_log_event_detail_world_end_active_entries value = events_log_world_end_scenario_active index = [INDEX] }
					add_to_array = { array = global.events_log_event_detail_world_end_available_entries value = events_log_world_end_scenario_available index = [INDEX] }
					add_to_array = { array = global.events_log_event_detail_world_end_scenario_flag_entries value = events_log_world_end_scenario_flag index = [INDEX] }
					add_to_array = { array = global.events_log_event_detail_world_end_super_event_id_entries value = events_log_world_end_super_event_id index = [INDEX] }

events_log_history_append_rows_by_actor_sort = {
    clear_array = temp_events_log_history_unique_actors

    # Build unique actor list from rows with actors.
    set_temp_variable = { events_log_actor_collect_index = 0 }
    while_loop_effect = {
        limit = { check_variable = { events_log_actor_collect_index < events_log_source_count } }
        set_temp_variable = { events_log_source_index = events_log_actor_collect_index }
        if = {
            limit = {
                check_variable = {
                    var = global.events_log_history_has_actor_entries^events_log_source_index
                    value = 0
                    compare = greater_than
                }
                NOT = {
                    is_in_array = {
                        temp_events_log_history_unique_actors = global.events_log_history_actor_entries^events_log_source_index
                    }
                }
            }
            add_to_array = {
                array = temp_events_log_history_unique_actors
                value = global.events_log_history_actor_entries^events_log_source_index
            }
        }
        add_to_temp_variable = { events_log_actor_collect_index = 1 }
    }

    # Deterministic actor-id sort passes.
    while_loop_effect = {
        limit = { check_variable = { temp_events_log_history_unique_actors^num > 0 } }

        if = {
            limit = { has_country_flag = events_log_sort_descending }
            find_highest_in_array = {
                array = temp_events_log_history_unique_actors
                value = events_log_actor_sort_target
            }
        }
        else = {
            find_lowest_in_array = {
                array = temp_events_log_history_unique_actors
                value = events_log_actor_sort_target
            }
        }

        if = {
            limit = { has_country_flag = events_log_sort_descending }
            set_temp_variable = { events_log_actor_member_index = 0 }
            while_loop_effect = {
                limit = { check_variable = { events_log_actor_member_index < events_log_source_count } }
                set_temp_variable = { events_log_source_index = events_log_actor_member_index }
                if = {
                    limit = {
                        check_variable = {
                            var = global.events_log_history_has_actor_entries^events_log_source_index
                            value = 0
                            compare = greater_than
                        }
                        check_variable = {
                            var = global.events_log_history_actor_entries^events_log_source_index
                            value = events_log_actor_sort_target
                            compare = equals
                        }
                    }
                    events_log_row_matches_active_filter = yes
                    if = {
                        limit = { check_variable = { events_log_row_matches_filter > 0 } }
                        events_log_append_view_row_from_source = yes
                    }
                }
                add_to_temp_variable = { events_log_actor_member_index = 1 }
            }
        }
        else = {
            set_temp_variable = { events_log_actor_member_index = events_log_source_count }
            while_loop_effect = {
                limit = { check_variable = { events_log_actor_member_index > 0 } }
                subtract_from_temp_variable = { events_log_actor_member_index = 1 }
                set_temp_variable = { events_log_source_index = events_log_actor_member_index }
                if = {
                    limit = {
                        check_variable = {
                            var = global.events_log_history_has_actor_entries^events_log_source_index
                            value = 0
                            compare = greater_than
                        }
                        check_variable = {
                            var = global.events_log_history_actor_entries^events_log_source_index
                            value = events_log_actor_sort_target
                            compare = equals
                        }
                    }
                    events_log_row_matches_active_filter = yes
                    if = {
                        limit = { check_variable = { events_log_row_matches_filter > 0 } }
                        events_log_append_view_row_from_source = yes
                    }
                }
            }
        }

        remove_from_array = {
            array = temp_events_log_history_unique_actors
            value = events_log_actor_sort_target
        }
    }

    # Non-actor rows are always appended after actor rows.
    if = {
        limit = { has_country_flag = events_log_sort_descending }
        set_temp_variable = { events_log_source_index = 0 }
        while_loop_effect = {
            limit = { check_variable = { events_log_source_index < events_log_source_count } }
            if = {
                limit = {
                    check_variable = {
                        var = global.events_log_history_has_actor_entries^events_log_source_index
                        value = 1
                        compare = less_than
                    }
                }
                events_log_row_matches_active_filter = yes
                if = {
                    limit = { check_variable = { events_log_row_matches_filter > 0 } }
                    events_log_append_view_row_from_source = yes
                }
            }
            add_to_temp_variable = { events_log_source_index = 1 }
        }
    }
    else = {
        set_temp_variable = { events_log_source_index = events_log_source_count }
        while_loop_effect = {
            limit = { check_variable = { events_log_source_index > 0 } }
            subtract_from_temp_variable = { events_log_source_index = 1 }
            if = {
                limit = {
                    check_variable = {
                        var = global.events_log_history_has_actor_entries^events_log_source_index
                        value = 1
                        compare = less_than
                    }
                }
                events_log_row_matches_active_filter = yes
                if = {
                    limit = { check_variable = { events_log_row_matches_filter > 0 } }
                    events_log_append_view_row_from_source = yes
                }
            }
        }
    }
}

rebuild_events_log_history_view = {
    sanitize_events_log_history_actor_entries = yes

    if = {
        limit = { NOT = { has_variable = settings_events_log_sort_mode } }
        set_variable = { settings_events_log_sort_mode = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_sort_mode > 2 } }
        set_variable = { settings_events_log_sort_mode = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_sort_mode < 0 } }
        set_variable = { settings_events_log_sort_mode = 0 }
    }
    if = {
        limit = { NOT = { has_variable = settings_events_log_filter_type } }
        set_variable = { settings_events_log_filter_type = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_filter_type > 4 } }
        set_variable = { settings_events_log_filter_type = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_filter_type < 0 } }
        set_variable = { settings_events_log_filter_type = 0 }
    }

    clear_array = global.events_log_view_sequence_entries
    clear_array = global.events_log_view_date_entries
    clear_array = global.events_log_view_event_id_entries
    clear_array = global.events_log_view_event_type_entries
	clear_array = global.events_log_view_payload_entries
    clear_array = global.events_log_view_actor_entries
    clear_array = global.events_log_view_has_actor_entries
	clear_array = global.events_log_view_secondary_actor_entries
	clear_array = global.events_log_view_has_secondary_actor_entries
    clear_array = global.events_log_view_cluster_id_entries
    clear_array = global.events_log_view_cluster_sequence_entries
    clear_array = global.events_log_view_cluster_type_entries

    set_temp_variable = { events_log_source_count = global.events_log_history_event_id_entries^num }

    if = {
        limit = { check_variable = { events_log_source_count > 0 } }
        if = {
            limit = { check_variable = { settings_events_log_sort_mode = 0 } }
            if = {
                limit = { has_country_flag = events_log_sort_descending }
                for_loop_effect = {
                    start = 0
                    end = events_log_source_count
                    value = i
                    set_temp_variable = { events_log_source_index = i }
                    events_log_row_matches_active_filter = yes
                    if = {
                        limit = { check_variable = { events_log_row_matches_filter > 0 } }
                        events_log_append_view_row_from_source = yes
                    }
                }
            }
            else = {
                set_temp_variable = { events_log_asc_index = events_log_source_count }
                while_loop_effect = {
                    limit = { check_variable = { events_log_asc_index > 0 } }
                    subtract_from_temp_variable = { events_log_asc_index = 1 }
                    set_temp_variable = { events_log_source_index = events_log_asc_index }
                    events_log_row_matches_active_filter = yes
                    if = {
                        limit = { check_variable = { events_log_row_matches_filter > 0 } }
                        events_log_append_view_row_from_source = yes
                    }
                }
            }
        }
        else_if = {
            limit = { check_variable = { settings_events_log_sort_mode = 1 } }
            if = {
                limit = { has_country_flag = events_log_sort_descending }
                set_temp_variable = { events_log_sort_id = 2000 }
                while_loop_effect = {
                    limit = { check_variable = { events_log_sort_id > 0 } }
                    subtract_from_temp_variable = { events_log_sort_id = 1 }
                    for_loop_effect = {
                        start = 0
                        end = events_log_source_count
                        value = i
                        set_temp_variable = { events_log_source_index = i }

events_log_evolution_append_rows_by_actor_sort = {
    clear_array = temp_events_log_evolution_unique_actors

    # Build unique actor list from rows with actors.
    set_temp_variable = { events_log_evolution_actor_collect_index = 0 }
    while_loop_effect = {
        limit = { check_variable = { events_log_evolution_actor_collect_index < events_log_evolution_source_count } }
        set_temp_variable = { events_log_evolution_source_index = events_log_evolution_actor_collect_index }
        if = {
            limit = {
                check_variable = {
                    var = global.events_log_evolution_has_actor_entries^events_log_evolution_source_index
                    value = 0
                    compare = greater_than
                }
                NOT = {
                    is_in_array = {
                        temp_events_log_evolution_unique_actors = global.events_log_evolution_actor_entries^events_log_evolution_source_index
                    }
                }
            }
            add_to_array = {
                array = temp_events_log_evolution_unique_actors
                value = global.events_log_evolution_actor_entries^events_log_evolution_source_index
            }
        }
        add_to_temp_variable = { events_log_evolution_actor_collect_index = 1 }
    }

    # Deterministic actor-id sort passes.
    while_loop_effect = {
        limit = { check_variable = { temp_events_log_evolution_unique_actors^num > 0 } }

        if = {
            limit = { has_country_flag = events_log_evolution_sort_descending }
            find_highest_in_array = {
                array = temp_events_log_evolution_unique_actors
                value = events_log_evolution_actor_sort_target
            }
        }
        else = {
            find_lowest_in_array = {
                array = temp_events_log_evolution_unique_actors
                value = events_log_evolution_actor_sort_target
            }
        }

        if = {
            limit = { has_country_flag = events_log_evolution_sort_descending }
            set_temp_variable = { events_log_evolution_actor_member_index = 0 }
            while_loop_effect = {
                limit = { check_variable = { events_log_evolution_actor_member_index < events_log_evolution_source_count } }
                set_temp_variable = { events_log_evolution_source_index = events_log_evolution_actor_member_index }
                if = {
                    limit = {
                        check_variable = {
                            var = global.events_log_evolution_has_actor_entries^events_log_evolution_source_index
                            value = 0
                            compare = greater_than
                        }
                        check_variable = {
                            var = global.events_log_evolution_actor_entries^events_log_evolution_source_index
                            value = events_log_evolution_actor_sort_target
                            compare = equals
                        }
                    }
                    events_log_evolution_row_matches_active_filter = yes
                    if = {
                        limit = { check_variable = { events_log_evolution_row_matches_filter > 0 } }
                        events_log_evolution_append_view_row_from_source = yes
                    }
                }
                add_to_temp_variable = { events_log_evolution_actor_member_index = 1 }
            }
        }
        else = {
            set_temp_variable = { events_log_evolution_actor_member_index = events_log_evolution_source_count }
            while_loop_effect = {
                limit = { check_variable = { events_log_evolution_actor_member_index > 0 } }
                subtract_from_temp_variable = { events_log_evolution_actor_member_index = 1 }
                set_temp_variable = { events_log_evolution_source_index = events_log_evolution_actor_member_index }
                if = {
                    limit = {
                        check_variable = {
                            var = global.events_log_evolution_has_actor_entries^events_log_evolution_source_index
                            value = 0
                            compare = greater_than
                        }
                        check_variable = {
                            var = global.events_log_evolution_actor_entries^events_log_evolution_source_index
                            value = events_log_evolution_actor_sort_target
                            compare = equals
                        }
                    }
                    events_log_evolution_row_matches_active_filter = yes
                    if = {
                        limit = { check_variable = { events_log_evolution_row_matches_filter > 0 } }
                        events_log_evolution_append_view_row_from_source = yes
                    }
                }
            }
        }

        remove_from_array = {
            array = temp_events_log_evolution_unique_actors
            value = events_log_evolution_actor_sort_target
        }
    }

    # Non-actor rows are always appended after actor rows.
    if = {
        limit = { has_country_flag = events_log_evolution_sort_descending }
        set_temp_variable = { events_log_evolution_source_index = 0 }
        while_loop_effect = {
            limit = { check_variable = { events_log_evolution_source_index < events_log_evolution_source_count } }
            if = {
                limit = {
                    check_variable = {
                        var = global.events_log_evolution_has_actor_entries^events_log_evolution_source_index
                        value = 1
                        compare = less_than
                    }
                }
                events_log_evolution_row_matches_active_filter = yes
                if = {
                    limit = { check_variable = { events_log_evolution_row_matches_filter > 0 } }
                    events_log_evolution_append_view_row_from_source = yes
                }
            }
            add_to_temp_variable = { events_log_evolution_source_index = 1 }
        }
    }
    else = {
        set_temp_variable = { events_log_evolution_source_index = events_log_evolution_source_count }
        while_loop_effect = {
            limit = { check_variable = { events_log_evolution_source_index > 0 } }
            subtract_from_temp_variable = { events_log_evolution_source_index = 1 }
            if = {
                limit = {
                    check_variable = {
                        var = global.events_log_evolution_has_actor_entries^events_log_evolution_source_index
                        value = 1
                        compare = less_than
                    }
                }
                events_log_evolution_row_matches_active_filter = yes
                if = {
                    limit = { check_variable = { events_log_evolution_row_matches_filter > 0 } }
                    events_log_evolution_append_view_row_from_source = yes
                }
            }
        }
    }
}

rebuild_events_log_events_view = {
    if = {
        limit = { NOT = { has_variable = settings_events_log_events_sort_mode } }
        set_variable = { settings_events_log_events_sort_mode = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_events_sort_mode > 2 } }
        set_variable = { settings_events_log_events_sort_mode = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_events_sort_mode < 0 } }
        set_variable = { settings_events_log_events_sort_mode = 0 }
    }
    if = {
        limit = { NOT = { has_variable = settings_events_log_events_filter_type } }
        set_variable = { settings_events_log_events_filter_type = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_events_filter_type > 5 } }
        set_variable = { settings_events_log_events_filter_type = 0 }
    }
    if = {
        limit = { check_variable = { settings_events_log_events_filter_type < 0 } }
        set_variable = { settings_events_log_events_filter_type = 0 }
    }

    clear_array = global.events_log_view_sequence_entries
    clear_array = global.events_log_view_date_entries
    clear_array = global.events_log_view_event_id_entries
    clear_array = global.events_log_view_event_type_entries
    clear_array = global.events_log_view_actor_entries
    clear_array = global.events_log_view_has_actor_entries
	clear_array = global.events_log_view_secondary_actor_entries
	clear_array = global.events_log_view_has_secondary_actor_entries
    clear_array = global.events_log_view_cluster_type_entries
    clear_array = global.events_log_events_view_fired_entries
    clear_array = global.events_log_events_view_enabled_entries
    clear_array = global.events_log_events_view_unique_entries
    clear_array = global.events_log_events_view_weight_entries

    clear_array = temp_events_log_events_source_event_id_entries
    clear_array = temp_events_log_events_source_event_type_entries
    clear_array = temp_events_log_events_source_fired_entries
    clear_array = temp_events_log_events_source_enabled_entries
    clear_array = temp_events_log_events_source_unique_entries
    clear_array = temp_events_log_events_source_weight_entries

    set_temp_variable = { events_log_events_all_count = global.all_events^num }
    for_loop_effect = {
        start = 0
        end = events_log_events_all_count
        value = i

        set_temp_variable = { event_id = global.all_events^i }
        set_temp_variable = { events_log_events_is_enabled = 1 }
        if = {
            limit = { is_in_array = { global.disabled_events = event_id } }
            set_temp_variable = { events_log_events_is_enabled = 0 }
        }
        get_event_type = yes

        set_temp_variable = { events_log_events_include = 1 }
        if = {
            limit = {
                check_variable = { settings_events_log_events_filter_type = 1 }
                check_variable = { events_log_events_is_enabled = 0 }
            }
            set_temp_variable = { events_log_events_include = 0 }
        }
        else_if = {
            limit = {
                check_variable = { settings_events_log_events_filter_type = 2 }
                check_variable = { events_log_events_is_enabled > 0 }
            }
            set_temp_variable = { events_log_events_include = 0 }
        }
        else_if = {
            limit = {
                check_variable = { settings_events_log_events_filter_type = 3 }
                NOT = { check_variable = { event_type = constant:event_system_event_type.repeatable } }
            }
            set_temp_variable = { events_log_events_include = 0 }
        }
        else_if = {
            limit = {
                check_variable = { settings_events_log_events_filter_type = 4 }
                NOT = { check_variable = { event_type = constant:event_system_event_type.fire_once } }
            }
            set_temp_variable = { events_log_events_include = 0 }
        }
        else_if = {
            limit = {
                check_variable = { settings_events_log_events_filter_type = 5 }
                NOT = { check_variable = { event_type = constant:event_system_event_type.major } }
            }
            set_temp_variable = { events_log_events_include = 0 }
        }

        if = {
            limit = { check_variable = { events_log_events_include > 0 } }

            set_temp_variable = { events_log_events_unique_flag = 0 }
            if = {
                limit = {
                    OR = {
                        check_variable = { event_type = constant:event_system_event_type.major }
                        check_variable = { event_type = constant:event_system_event_type.fire_once }
                    }
                }
                set_temp_variable = { events_log_events_unique_flag = 1 }
            }

            set_temp_variable = { events_log_events_fired_count = 0 }
            set_temp_variable = { events_log_events_history_scan_index = 0 }
            while_loop_effect = {
                limit = {
                    check_variable = {
                        events_log_events_history_scan_index < global.events_log_history_event_id_entries^num
                    }
                }
                if = {
                    limit = {
                        check_variable = {
                            var = global.events_log_history_event_id_entries^events_log_events_history_scan_index
                            value = event_id
                            compare = equals
                        }
                    }
                    add_to_temp_variable = { events_log_events_fired_count = 1 }
                }
                add_to_temp_variable = { events_log_events_history_scan_index = 1 }
            }
            if = {
                limit = {
                    check_variable = { settings_events_log_events_sort_mode = 1 }
                    check_variable = { events_log_events_fired_count < 1 }
                }
                set_temp_variable = { events_log_events_include = 0 }
            }
        }

        if = {
            limit = { check_variable = { events_log_events_include > 0 } }
            set_temp_variable = { events_log_events_live_weight = 0 }
            if = {
                limit = {
                    check_variable = { events_log_events_is_enabled > 0 }
                    NOT = {
                        AND = {
                            check_variable = { events_log_events_unique_flag > 0 }
                            check_variable = { events_log_events_fired_count > 0 }
                        }
                    }
                }
                get_event_weight = yes
                set_temp_variable = { events_log_events_live_weight = event_weight }
            }

            if = {
                limit = {
                    check_variable = { event_id = 91 }
                    NOT = { has_global_flag = communism_spread_world_revolution_unlocked }
                }
                set_temp_variable = { events_log_events_live_weight = -1 }
            }
			if = {
				limit = {
					check_variable = { event_id = 3 }
					NOT = { holy_realm_has_valid_refuge_host = yes }
				}
				set_temp_variable = { events_log_events_live_weight = -1 }
			}
			if = {
				limit = {
					check_variable = { event_id = constant:utopia_manifesto_event.id }
					NOT = { utopia_manifesto_automatic_event_is_available = yes }
				}
				set_temp_variable = { events_log_events_live_weight = -1 }
			}
			if = {
				limit = {
					check_variable = { event_id = constant:brilliant_scientist_event.id }
					NOT = { brilliant_scientist_automatic_event_is_available = yes }
				}
				set_temp_variable = { events_log_events_live_weight = -1 }
			}
			if = {
				limit = {
					check_variable = { event_id = constant:secret_alliance_event.id }
					NOT = { secret_alliance_automatic_event_is_available = yes }
				}
				set_temp_variable = { events_log_events_live_weight = -1 }
			}
			if = {
				limit = {
					check_variable = { event_id = constant:cannibalism_event.id }
					NOT = { cannibalism_automatic_event_is_available = yes }
				}
				set_temp_variable = { events_log_events_live_weight = -1 }
			}
            if = {
                limit = {
                    check_variable = { event_id = constant:tensions_rising_event_log.event_id }
                    NOT = { is_tensions_rising_automatic_event_available = yes }
                }
                set_temp_variable = { events_log_events_live_weight = -1 }
            }

            add_to_array = {
                array = temp_events_log_events_source_event_id_entries
                value = event_id
            }
            add_to_array = {
                array = temp_events_log_events_source_event_type_entries
                value = event_type
            }
            add_to_array = {
                array = temp_events_log_events_source_fired_entries
                value = events_log_events_fired_count
            }
            add_to_array = {
                array = temp_events_log_events_source_enabled_entries
                value = events_log_events_is_enabled
            }
            add_to_array = {
                array = temp_events_log_events_source_unique_entries
                value = events_log_events_unique_flag
            }
            add_to_array = {
                array = temp_events_log_events_source_weight_entries
                value = events_log_events_live_weight
            }
        }
    }

    set_temp_variable = { events_log_events_source_count = temp_events_log_events_source_event_id_entries^num }
    set_temp_variable = { events_log_events_sorted_count = 0 }
    while_loop_effect = {
        limit = { check_variable = { events_log_events_sorted_count < events_log_events_source_count } }

        set_temp_variable = { events_log_events_best_index = -1 }
        set_temp_variable = { events_log_events_scan_index = 0 }
        while_loop_effect = {
            limit = { check_variable = { events_log_events_scan_index < events_log_events_source_count } }
            if = {
                limit = {
                    check_variable = {
                        var = temp_events_log_events_source_event_id_entries^events_log_events_scan_index
                        value = 0
                        compare = greater_than
                    }
                }

                set_temp_variable = { events_log_events_candidate_metric = temp_events_log_events_source_event_id_entries^events_log_events_scan_index }
                if = {
                    limit = { check_variable = { settings_events_log_events_sort_mode = 1 } }
                    set_temp_variable = { events_log_events_candidate_metric = temp_events_log_events_source_fired_entries^events_log_events_scan_index }
                }
                else_if = {
                    limit = { check_variable = { settings_events_log_events_sort_mode = 2 } }
                    set_temp_variable = { events_log_events_candidate_metric = temp_events_log_events_source_weight_entries^events_log_events_scan_index }
                }

                if = {
                    limit = { check_variable = { events_log_events_best_index < 0 } }
                    set_temp_variable = { events_log_events_best_index = events_log_events_scan_index }
                    set_temp_variable = { events_log_events_best_metric = events_log_events_candidate_metric }
                    set_temp_variable = { events_log_events_best_event_id = temp_events_log_events_source_event_id_entries^events_log_events_scan_index }
                }
                else_if = {
                    limit = {
                        has_country_flag = events_log_events_sort_descending
                        check_variable = { events_log_events_candidate_metric > events_log_events_best_metric }
                    }
                    set_temp_variable = { events_log_events_best_index = events_log_events_scan_index }
                    set_temp_variable = { events_log_events_best_metric = events_log_events_candidate_metric }
                    set_temp_variable = { events_log_events_best_event_id = temp_events_log_events_source_event_id_entries^events_log_events_scan_index }
                }
                else_if = {
                    limit = {
                        NOT = { has_country_flag = events_log_events_sort_descending }
                        check_variable = { events_log_events_candidate_metric < events_log_events_best_metric }
                    }
                    set_temp_variable = { events_log_events_best_index = events_log_events_scan_index }
                    set_temp_variable = { events_log_events_best_metric = events_log_events_candidate_metric }
                    set_temp_variable = { events_log_events_best_event_id = temp_events_log_events_source_event_id_entries^events_log_events_scan_index }
                }
                else_if = {
                    limit = { check_variable = { events_log_events_candidate_metric = events_log_events_best_metric } }
                    if = {
                        limit = {
                            has_country_flag = events_log_events_sort_descending
                            check_variable = {
                                var = temp_events_log_events_source_event_id_entries^events_log_events_scan_index
                                value = events_log_events_best_event_id
                                compare = greater_than
                            }
                        }
                        set_temp_variable = { events_log_events_best_index = events_log_events_scan_index }
                        set_temp_variable = { events_log_events_best_metric = events_log_events_candidate_metric }
                        set_temp_variable = { events_log_events_best_event_id = temp_events_log_events_source_event_id_entries^events_log_events_scan_index }
                    }
                    else_if = {
                        limit = {
                            NOT = { has_country_flag = events_log_events_sort_descending }
                            check_variable = {
                                var = temp_events_log_events_source_event_id_entries^events_log_events_scan_index
                                value = events_log_events_best_event_id
                                compare = less_than
                            }
                        }
                        set_temp_variable = { events_log_events_best_index = events_log_events_scan_index }
                        set_temp_variable = { events_log_events_best_metric = events_log_events_candidate_metric }
                        set_temp_variable = { events_log_events_best_event_id = temp_events_log_events_source_event_id_entries^events_log_events_scan_index }
                    }
                }
            }
            add_to_temp_variable = { events_log_events_scan_index = 1 }
        }

        if = {
            limit = { check_variable = { events_log_events_best_index > -1 } }
            add_to_array = {
                array = global.events_log_view_event_id_entries
                value = temp_events_log_events_source_event_id_entries^events_log_events_best_index
            }
            add_to_array = {
                array = global.events_log_view_event_type_entries
                value = temp_events_log_events_source_event_type_entries^events_log_events_best_index
            }
            add_to_array = {
                array = global.events_log_events_view_fired_entries
                value = temp_events_log_events_source_fired_entries^events_log_events_best_index
            }
            add_to_array = {
                array = global.events_log_events_view_enabled_entries
                value = temp_events_log_events_source_enabled_entries^events_log_events_best_index
            }
            add_to_array = {
                array = global.events_log_events_view_unique_entries
                value = temp_events_log_events_source_unique_entries^events_log_events_best_index
            }
            add_to_array = {
                array = global.events_log_events_view_weight_entries
                value = temp_events_log_events_source_weight_entries^events_log_events_best_index
            }

            set_variable = { temp_events_log_events_source_event_id_entries^events_log_events_best_index = -1 }
            add_to_temp_variable = { events_log_events_sorted_count = 1 }
        }
        else = {
            set_temp_variable = { events_log_events_sorted_count = events_log_events_source_count }
        }
    }
}
