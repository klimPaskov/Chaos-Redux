# Event 012 host decision navigation. COUNTRY scope on the host.
# These effects change only one private presentation enum. They never alter
# gameplay, target arrays, prices, receipts, deadlines, doctrine or priority.
# Public API: 012_africa_gods_response_view_effects.md.

gods_of_africa_clear_response_view = {
	clear_variable = gods_of_africa_response_view
}

gods_of_africa_open_priorities = {
	if = {
		limit = { gods_of_africa_host_can_open_priorities = yes }
		set_variable = { gods_of_africa_response_view = constant:gods_of_africa_response_view.priorities }
	}
}

gods_of_africa_open_partner_relations = {
	if = {
		limit = { gods_of_africa_host_can_open_partners = yes }
		set_variable = { gods_of_africa_response_view = constant:gods_of_africa_response_view.partners }
	}
}

gods_of_africa_open_judgments = {
	if = {
		limit = { gods_of_africa_host_can_open_judgments = yes }
		set_variable = { gods_of_africa_response_view = constant:gods_of_africa_response_view.judgment }
	}
}

gods_of_africa_return_to_overview = {
	if = {
		limit = { gods_of_africa_host_can_return_to_overview = yes }
		gods_of_africa_clear_response_view = yes
	}
}
