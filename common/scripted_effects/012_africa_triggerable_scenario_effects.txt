# ============================================================================
# CHAOS REDUX - EVENT 012 TRIGGERABLE SCENARIO EFFECTS
# ============================================================================
# The shared scenario window dispatches here for the Africa Is One row. A fresh
# launch reuses Event 012's weighted African-capital host and commit barrier;
# an existing host receives the same scenario opening in place. The effect never
# manufactures a tag, skips the host target, or writes the terminal world-end
# identity. World Is One prepares the existing terminal-campaign route while its
# normal package, proof, and identity gates remain authoritative.
# ============================================================================

# Any scope. Copies the UI selectors into global launch state, then either
# applies the opening to the committed host or queues the canonical Event 012
# entry event for the weighted prefire host.
africa_triggerable_scenario_launch_selected = {
	initialize_triggerable_scenarios_settings = yes
	if = {
		limit = {
			check_variable = {
				var = triggerable_scenarios_selected
				value = constant:triggerable_scenario_id.africa_is_one
				compare = equals
			}
			africa_triggerable_scenario_can_launch = yes
		}
		set_global_flag = africa_triggerable_scenario_launch_active
		clr_global_flag = africa_triggerable_scenario_setup_failed
		set_variable = { global.africa_triggerable_scenario_type = triggerable_scenarios_africa_type }
		set_variable = { global.africa_triggerable_scenario_intensity = triggerable_scenarios_intensity }
		if = {
			limit = {
				has_global_flag = africa_event_active
				NOT = { has_global_flag = africa_event_resolved }
				africa_has_current_host = yes
			}
			event_target:africa_host = {
				africa_triggerable_scenario_apply_opening = yes
			}
		}
		else = {
			africa_prepare_manual_event_fire = yes
			if = {
				limit = { has_event_target = africa_prefire_host }
				event_target:africa_prefire_host = {
					set_country_flag = africa_triggerable_scenario_launch_queued
					set_temp_variable = { africa_triggerable_scenario_callback_days = constant:africa_triggerable_scenario_timing.opening_callback_days }
					country_event = { id = chaosx.nr12.1 days = africa_triggerable_scenario_callback_days }
				}
			}
			else = {
				clr_global_flag = africa_triggerable_scenario_launch_active
				set_global_flag = africa_triggerable_scenario_setup_failed
			}
		}
	}
	else = {
		clr_global_flag = africa_triggerable_scenario_launch_active
		set_global_flag = africa_triggerable_scenario_setup_failed
	}
}

# COUNTRY scope on the Event 012 host. This is the one shared opening writer;
# branch-specific setup below only changes the route that follows it.
africa_triggerable_scenario_apply_opening = {
	if = {
		limit = {
			africa_triggerable_scenario_host_scope_is_valid = yes
			NOT = { has_country_flag = africa_triggerable_scenario_applied }
		}
		set_country_flag = africa_triggerable_scenario_applied
		set_variable = { africa_triggerable_scenario_type = global.africa_triggerable_scenario_type }
		set_variable = { africa_triggerable_scenario_intensity = global.africa_triggerable_scenario_intensity }
		africa_triggerable_scenario_apply_starting_forces = yes
		africa_triggerable_scenario_apply_value_deltas = yes
		africa_clamp_host_values = yes
		if = {
			limit = {
				check_variable = {
					var = africa_triggerable_scenario_type
					value = constant:triggerable_scenario_africa_type.world_is_one
					compare = equals
				}
			}
			africa_triggerable_scenario_apply_world_is_one_opening = yes
		}
		else = {
			africa_triggerable_scenario_apply_africa_is_one_opening = yes
		}
		set_global_flag = africa_triggerable_scenario_launched
		clr_global_flag = africa_triggerable_scenario_launch_active
		if = {
			limit = { NOT = { has_variable = global.africa_triggerable_scenario_launch_count } }
			set_variable = { global.africa_triggerable_scenario_launch_count = constant:africa_value.zero }
		}
		add_to_variable = { global.africa_triggerable_scenario_launch_count = constant:africa_value.one }
		set_variable = { global.africa_triggerable_scenario_last_type = africa_triggerable_scenario_type }
		set_variable = { global.africa_triggerable_scenario_last_intensity = africa_triggerable_scenario_intensity }
		set_country_flag = africa_triggerable_scenario_notification_pending
		set_temp_variable = { africa_triggerable_scenario_notification_days = constant:africa_triggerable_scenario_timing.notification_days }
		country_event = { id = chaosx.triggerable_scenarios.11 days = africa_triggerable_scenario_notification_days }
	}
	else_if = {
		limit = { has_global_flag = africa_triggerable_scenario_launch_active }
		clr_global_flag = africa_triggerable_scenario_launch_active
		set_global_flag = africa_triggerable_scenario_setup_failed
	}
}

# COUNTRY scope on the host. The equipment package is deliberately intensity
# driven and uses vanilla level-one equipment so it works for every mapped host.
africa_triggerable_scenario_apply_starting_forces = {
	if = {
		limit = { check_variable = { africa_triggerable_scenario_intensity = constant:triggerable_scenario_intensity.low } }
		set_temp_variable = { africa_triggerable_scenario_manpower = constant:africa_triggerable_scenario_force.low_manpower }
		set_temp_variable = { africa_triggerable_scenario_infantry_equipment = constant:africa_triggerable_scenario_force.low_infantry_equipment }
		set_temp_variable = { africa_triggerable_scenario_support_equipment = constant:africa_triggerable_scenario_force.low_support_equipment }
		set_temp_variable = { africa_triggerable_scenario_convoys = constant:africa_triggerable_scenario_force.low_convoys }
	}
	else_if = {
		limit = { check_variable = { africa_triggerable_scenario_intensity = constant:triggerable_scenario_intensity.high } }
		set_temp_variable = { africa_triggerable_scenario_manpower = constant:africa_triggerable_scenario_force.high_manpower }
		set_temp_variable = { africa_triggerable_scenario_infantry_equipment = constant:africa_triggerable_scenario_force.high_infantry_equipment }
		set_temp_variable = { africa_triggerable_scenario_support_equipment = constant:africa_triggerable_scenario_force.high_support_equipment }
		set_temp_variable = { africa_triggerable_scenario_convoys = constant:africa_triggerable_scenario_force.high_convoys }
	}
	else_if = {
		limit = { check_variable = { africa_triggerable_scenario_intensity = constant:triggerable_scenario_intensity.maximum } }
		set_temp_variable = { africa_triggerable_scenario_manpower = constant:africa_triggerable_scenario_force.maximum_manpower }
		set_temp_variable = { africa_triggerable_scenario_infantry_equipment = constant:africa_triggerable_scenario_force.maximum_infantry_equipment }
		set_temp_variable = { africa_triggerable_scenario_support_equipment = constant:africa_triggerable_scenario_force.maximum_support_equipment }
		set_temp_variable = { africa_triggerable_scenario_convoys = constant:africa_triggerable_scenario_force.maximum_convoys }
	}
	else = {
		set_temp_variable = { africa_triggerable_scenario_manpower = constant:africa_triggerable_scenario_force.medium_manpower }
		set_temp_variable = { africa_triggerable_scenario_infantry_equipment = constant:africa_triggerable_scenario_force.medium_infantry_equipment }
		set_temp_variable = { africa_triggerable_scenario_support_equipment = constant:africa_triggerable_scenario_force.medium_support_equipment }
		set_temp_variable = { africa_triggerable_scenario_convoys = constant:africa_triggerable_scenario_force.medium_convoys }
	}
	add_manpower = var:africa_triggerable_scenario_manpower
	add_equipment_to_stockpile = { type = infantry_equipment_1 amount = var:africa_triggerable_scenario_infantry_equipment }
	add_equipment_to_stockpile = { type = support_equipment_1 amount = var:africa_triggerable_scenario_support_equipment }
	add_equipment_to_stockpile = { type = convoy_1 amount = var:africa_triggerable_scenario_convoys }
}
