# ============================================================================
# CHAOS REDUX - MIGRATION RECEPTION CAPACITY EFFECTS
# ============================================================================
# Migration-owned people-unit reception service adapters and sparse-capacity
# recalculation.  Candidates are registered explicitly by state owners and
# recalculated only for a dirty country during the existing host pulse.  The
# transport component below is availability evidence; it never proves a route.
# ============================================================================

migration_register_reception_candidate_state = {
	migration_initialize_runtime = yes
	set_temp_variable = { migration_reception_candidate_result = constant:migration_capacity.zero }
	if = {
		limit = { humanitarian_state_is_valid = yes }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.migration_reception_candidate_states
						value = THIS
					}
				}
			}
			add_to_array = { array = global.migration_reception_candidate_states value = THIS }
		}
		set_variable = { global.migration_reception_candidate_states_count = global.migration_reception_candidate_states^num }
		set_state_flag = migration_reception_candidate_active
		if = { limit = { NOT = { has_variable = migration_reception_candidate_revision } } set_variable = { migration_reception_candidate_revision = constant:migration_capacity.revision_increment } }
		OWNER = {
			migration_register_active_country = yes
			migration_mark_reception_capacity_dirty = yes
		}
		set_temp_variable = { migration_reception_candidate_result = constant:migration_capacity.one }
	}
}

migration_unregister_reception_candidate_state = {
	migration_initialize_runtime = yes
	if = {
		limit = {
			is_in_array = {
				array = global.migration_reception_candidate_states
				value = THIS
			}
		}
		remove_from_array = { array = global.migration_reception_candidate_states value = THIS }
		set_variable = { global.migration_reception_candidate_states_count = global.migration_reception_candidate_states^num }
	}
	clr_state_flag = migration_reception_candidate_active
	OWNER = { migration_mark_reception_capacity_dirty = yes }
	if = { limit = { has_event_target = migration_previous_controller } event_target:migration_previous_controller = { migration_mark_reception_capacity_dirty = yes } }
}

# COUNTRY scope.  Dirty is a write-only callback consumed by the sparse host
# pulse; it is never a recurring country-wide scan.
migration_mark_reception_capacity_dirty = {
	if = {
		limit = { humanitarian_country_is_valid = yes }
		if = { limit = { NOT = { has_variable = migration_reception_capacity_revision } } set_variable = { migration_reception_capacity_revision = constant:migration_capacity.zero } }
		if = { limit = { NOT = { has_variable = migration_reception_capacity_schema_version } } set_variable = { migration_reception_capacity_schema_version = constant:migration_capacity.schema_version } }
		set_variable = { migration_reception_capacity_dirty = constant:migration_capacity.one }
		migration_register_active_country = yes
	}
}

# STATE scope. Other mechanics call this seam after a state projection changes.
# Migration alone decides whether the state is a registered reception candidate
# and therefore whether its owner must be dirtied.
migration_mark_reception_capacity_dirty_if_candidate = {
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			has_state_flag = migration_reception_candidate_active
		}
		OWNER = { migration_mark_reception_capacity_dirty = yes }
	}
}

# STATE scope.  The decision owner supplies a positive people-unit service
# amount and explicit proof.  The adapter never writes civilian factories or
# population and can be adjusted idempotently.
migration_set_reception_shelter_service = {
	set_temp_variable = { migration_reception_shelter_service_result = constant:migration_capacity.zero }
	if = {
		limit = {
			humanitarian_state_is_valid = yes
			check_variable = { var = migration_reception_shelter_service_request value = constant:humanitarian_population.zero compare = greater_than }
			check_variable = { var = migration_reception_shelter_service_request_proven value = constant:migration_capacity.one compare = greater_than }
		}
		set_variable = { migration_reception_shelter_capacity = migration_reception_shelter_service_request }
		set_variable = { migration_reception_shelter_service_proven = constant:migration_capacity.one }
		migration_register_reception_candidate_state = yes
		set_temp_variable = { migration_reception_shelter_service_result = constant:migration_capacity.one }
	}
	set_temp_variable = { migration_reception_shelter_service_request = constant:humanitarian_population.zero }
	set_temp_variable = { migration_reception_shelter_service_request_proven = constant:migration_capacity.zero }
}

					multiply_temp_variable = { migration_capacity_quality_weighted = migration_capacity_state_population }
