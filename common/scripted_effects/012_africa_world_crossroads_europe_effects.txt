# ============================================================================
# CHAOS REDUX - EVENT 012 WORLD PACKAGE W2 CROSSROADS AND EUROPE EFFECTS
# ============================================================================
# Country-scope helpers for the non-model Middle East Crossroads and Europe
# Continental Settlement loops. Each helper owns one visible negotiation lane,
# its route-sensitive package variables, and the cleanup needed by the event
# chain. The shared world-order effects, constants, and ratification trigger
# remain parent-owned.
#
# These helpers deliberately do not activate high-chaos routes or set any
# implementation-readiness flag. The high-chaos route may receive a bounded
# balance adjustment when it is already selected, but it never receives a new
# route or readiness flag from this file.
# ============================================================================

# Middle East Crossroads opening boards and lane settlement

# COUNTRY scope. Desert and mountain command lane. The selected constitutional
# route changes which public Crossroads ledger receives the command settlement.
africa_crossroads_apply_command_settlement = {
	set_country_flag = africa_crossroads_command_settlement_ratified
	set_country_flag = africa_crossroads_command_lane_pending
	if = {
		limit = { has_country_flag = africa_middle_east_arab_federal_route }
		add_to_variable = { africa_crossroads_arab_compact = constant:africa_world_order.package_major_gain }
		add_to_variable = { africa_crossroads_anatolian_power = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_crossroads_command_joint_staff
	}
	else_if = {
		limit = { has_country_flag = africa_middle_east_plural_crossroads_route }
		add_to_variable = { africa_crossroads_minority_guarantees = constant:africa_world_order.package_major_gain }
		add_to_variable = { africa_crossroads_holy_site_legitimacy = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_crossroads_command_civic_guard
	}
	else_if = {
		limit = { has_country_flag = africa_middle_east_royal_concert_route }
		add_to_variable = { africa_crossroads_anatolian_power = constant:africa_world_order.package_major_gain }
		add_to_variable = { africa_crossroads_persian_influence = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_crossroads_command_balancing_council
	}
	else_if = {
		limit = { has_country_flag = africa_middle_east_socialist_republics_route }
		add_to_variable = { africa_crossroads_oil_sovereignty = constant:africa_world_order.package_minor_gain }
		add_to_variable = { africa_crossroads_arab_compact = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_crossroads_command_workers_board
	}
	else_if = {
		limit = { has_country_flag = africa_middle_east_desert_covenant_route }
		add_to_variable = { africa_crossroads_water_security = constant:africa_world_order.package_minor_gain }
		add_to_variable = { africa_crossroads_holy_site_legitimacy = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_crossroads_command_desert_accord
	}
	africa_world_clamp_package_mechanic = yes
	if = {
		limit = { NOT = { has_country_flag = africa_crossroads_shared_lane_event_open } }
		set_country_flag = africa_crossroads_shared_lane_event_open
		country_event = { id = africa_world_package.105 }
	}
}

# COUNTRY scope. Common army and air defence lane. The route branch determines
# whether defence is federal, socialist, royal, command-led, or neutral.
africa_europe_apply_common_defence_law = {
	set_country_flag = africa_europe_common_defence_ratified
	set_country_flag = africa_europe_defence_lane_pending
	if = {
		limit = { has_country_flag = africa_europe_democratic_federation_route }
		add_to_variable = { africa_europe_industrial_integration = constant:africa_world_order.package_minor_gain }
		add_to_variable = { africa_europe_national_sovereignty = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_europe_federal_defence_board
	}
	else_if = {
		limit = { has_country_flag = africa_europe_socialist_union_route }
		add_to_variable = { africa_europe_industrial_integration = constant:africa_world_order.package_major_gain }
		add_to_variable = { africa_europe_war_memory = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_europe_peoples_defence_command
	}
	else_if = {
		limit = { has_country_flag = africa_europe_royal_concert_route }
		add_to_variable = { africa_europe_national_sovereignty = constant:africa_world_order.package_major_gain }
		add_to_variable = { africa_europe_industrial_integration = constant:africa_world_order.package_minor_gain }
		set_country_flag = africa_europe_crown_defence_council
	}
	else_if = {
		limit = { has_country_flag = africa_europe_continental_command_route }
