# ============================================================================
# CHAOS REDUX - EVENT 012 GODS OF AFRICA SCRIPTED EFFECTS
# ============================================================================
# Event 012 owns this ledger. The helpers below are called by the Event 012
# evolution/owner-local callbacks and by its decision surface. They persist
# only normal country variables, use the authoritative global event target
# `africa_host`, and guard every participant mutation by host generation.
#
# Contracts are immutable snapshots: demand family, amount, deadline, status,
# and generation are written once at creation. Payment effects debit the
# participant's real stockpile/manpower/political power and credit the host.
# No effect here installs a recurring world-wide on_action.
# ============================================================================

# COUNTRY scope on the Event 012 host. This is the one-time post-Evolution-I
# entry point. The owner callback should call this once; the effect is
# idempotent and only starts the 180-day consolidation clock.
gods_of_africa_initialize_system = {
	if = {
		limit = {
			africa_is_current_host = yes
			gods_of_africa_epoch_permission_is_current = yes
			NOT = { has_capitulated = yes }
			has_global_flag = africa_is_one
			has_global_flag = africa_evolution_i_logged
			NOT = { has_country_flag = gods_of_africa_active }
			NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
			# A delayed activation event may still arrive after the host-loss
			# callback has cancelled this generation. Do not restart consolidation
			# until a successor generation explicitly clears the loss receipt.
			NOT = { has_country_flag = gods_of_africa_cancelled_on_unifier_loss }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = gods_of_africa_restoration_generation }
					NOT = { check_variable = { var = gods_of_africa_restoration_generation value = africa_host_generation compare = equals } }
				}
			}
			set_variable = { gods_of_africa_restoration_generation = africa_host_generation }
			# A restored host starts a fresh owner lifecycle. Do not let a loss
			# receipt from the previous generation suppress its future cleanup.
			clr_country_flag = gods_of_africa_cancelled_on_unifier_loss
			clr_country_flag = gods_of_africa_unifier_loss_recorded
			clear_array = gods_of_africa_participants
			clear_array = gods_of_africa_selected_participants
			set_variable = { gods_of_africa_participant_count = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_demand_sequence = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_active_demand_count = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_priority_offender_count = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_strength = constant:gods_of_africa_value.starting_strength }
			set_variable = { global.gods_of_africa_strength = constant:gods_of_africa_value.starting_strength }
			set_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.limited }
			# Zero is the unchosen doctrine state expected by the host decisions.
			set_variable = { gods_of_africa_doctrine = constant:gods_of_africa_value.zero }
			set_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.land_equipment }
			set_variable = { gods_of_africa_next_dispatch_day = global.num_days }
			set_variable = { gods_of_africa_next_registry_review_day = global.num_days }
			set_variable = { gods_of_africa_next_owner_tick_day = global.num_days }
			set_country_flag = gods_of_africa_initialized
		}
		gods_of_africa_start_consolidation = yes
	}
}

# COUNTRY scope on the host. The date variable is deliberately used instead
# of a timed flag so the 180-day boundary remains inspectable and generation
# safe across saves and event chains. Evolution I starts this interval; the
# host-establishment date may predate Evolution I by many months.
gods_of_africa_start_consolidation = {
	if = {
		limit = {
			africa_is_current_host = yes
			gods_of_africa_epoch_permission_is_current = yes
			NOT = { has_capitulated = yes }
			NOT = { has_country_flag = gods_of_africa_active }
			NOT = { has_country_flag = gods_of_africa_consolidation_pending }
			NOT = { has_country_flag = gods_of_africa_cancelled_on_unifier_loss }
			has_variable = africa_host_generation
		}
		set_country_flag = gods_of_africa_consolidation_pending
		set_country_flag = gods_of_africa_activation_clock_scheduled
		set_variable = { gods_of_africa_consolidation_start_day = global.num_days }
		set_variable = { gods_of_africa_consolidation_due_day = gods_of_africa_consolidation_start_day }
		add_to_variable = { gods_of_africa_consolidation_due_day = constant:gods_of_africa_timing.consolidation_days }
		set_variable = { gods_of_africa_restoration_generation = africa_host_generation }
		set_temp_variable = { gods_of_africa_clock_days = gods_of_africa_consolidation_due_day }
		subtract_from_temp_variable = { gods_of_africa_clock_days = global.num_days }
		if = { limit = { check_variable = { gods_of_africa_clock_days < constant:gods_of_africa_value.one } } set_temp_variable = { gods_of_africa_clock_days = constant:gods_of_africa_value.one } }
		country_event = { id = chaosx.nr12.600 days = var:gods_of_africa_clock_days }
	}
}

# COUNTRY scope on the host. Called by the owner-local 30-day callback.
gods_of_africa_run_owner_tick = {
	# The ordinary tribute loop ends as soon as the continent reaches its
	# secured settlement phase. This owner-local check keeps a pending demand
	# from being dispatched after the political settlement has already closed.
	if = {
		limit = {
			gods_of_africa_active_host_is_valid = yes
			gods_of_africa_continent_is_secured = yes
		}
		gods_of_africa_commit_final_settlement = yes
	}
	if = {
		limit = { gods_of_africa_active_host_is_valid = yes }
		# Keep the small need snapshot current for both demand selection and the
		# participant-side voluntary-aid action. This remains host-local and does
		# not introduce a second world scan.
		gods_of_africa_build_need_profile = yes
		# The native mission owns exact contract expiry. This bounded pass is a
		# recovery check for an interrupted or missing mission timer.
		for_each_scope_loop = {
			array = gods_of_africa_participants
			if = {
				limit = {
					gods_of_africa_participant_has_active_demand = yes
					has_war_with = event_target:africa_host
				}
				gods_of_africa_cancel_demand_for_war = yes
			}
			if = {
				limit = {
					gods_of_africa_participant_has_active_demand = yes
					check_variable = {
						var = global.num_days
						value = gods_of_africa_demand_deadline
						compare = greater_than_or_equals
					}
				}
				gods_of_africa_fail_demand = yes
			}
		}
		if = {
			limit = { gods_of_africa_owner_tick_due = yes }
			gods_of_africa_refresh_strength = yes
			for_each_scope_loop = {
				array = gods_of_africa_participants
				gods_of_africa_refresh_participant_floors = yes
				if = {
					limit = {
						has_country_flag = gods_of_africa_defiant
						NOT = { has_country_flag = gods_of_africa_recent_extreme_punishment }
						OR = {
							NOT = { has_variable = gods_of_africa_punishment_review_due_day }
							check_variable = { var = global.num_days value = gods_of_africa_punishment_review_due_day compare = greater_than_or_equals }
						}
					}
					clr_country_flag = gods_of_africa_punishment_applied
					gods_of_africa_evaluate_punishment = yes
				}
				# A window whose mission was cancelled by war closes as missed once
				# its due day passes, so the participant is offered a fresh one.
				if = {
					limit = {
						has_country_flag = gods_of_africa_reconciliation_pending
						NOT = { has_active_mission = gods_of_africa_reconciliation_mission }
						has_variable = gods_of_africa_reconciliation_due_day
						check_variable = { var = global.num_days value = gods_of_africa_reconciliation_due_day compare = greater_than_or_equals }
					}
					gods_of_africa_fail_reconciliation = yes
				}
				if = {
					limit = {
						gods_of_africa_participant_needs_reconciliation = yes
						NOT = { has_country_flag = gods_of_africa_reconciliation_pending }
						NOT = { has_country_flag = gods_of_africa_demand_active }
					}
					gods_of_africa_start_reconciliation = yes
				}
			}
			set_variable = { gods_of_africa_next_owner_tick_day = global.num_days }
			add_to_variable = { gods_of_africa_next_owner_tick_day = constant:gods_of_africa_timing.owner_tick_days }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = gods_of_africa_active_demand_count }
					check_variable = { gods_of_africa_active_demand_count < constant:gods_of_africa_timing.max_active_demands }
				}
				OR = {
					NOT = { has_variable = gods_of_africa_next_dispatch_day }
					check_variable = { var = global.num_days value = gods_of_africa_next_dispatch_day compare = greater_than_or_equals }
				}
			}
			gods_of_africa_dispatch_next_demand = yes
			set_variable = { gods_of_africa_next_dispatch_day = global.num_days }
			add_to_variable = { gods_of_africa_next_dispatch_day = constant:gods_of_africa_timing.introduction_stagger_days }
		}
	}
}


# COUNTRY scope on the host. This is a bounded review of only valid majors or
# player countries. It is intentionally invoked by the owner and not a
# generic daily/weekly world scan. The owner-local review also onboards a
# country that became major or player-controlled after activation.
gods_of_africa_review_registry = {
	if = {
		limit = { gods_of_africa_active_host_is_valid = yes }
		# Archive from a separate snapshot because archival removes entries
		# from the live participant array.
		clear_array = gods_of_africa_registry_review_participants
		for_each_scope_loop = {
			array = gods_of_africa_participants
			event_target:africa_host = {
				add_to_array = { array = gods_of_africa_registry_review_participants value = PREV }
			}
		}
		for_each_scope_loop = {
			array = gods_of_africa_registry_review_participants
			if = {
				limit = { NOT = { gods_of_africa_participant_is_live = yes } }
				gods_of_africa_archive_invalid_participant = yes
			}
		}
		clear_array = gods_of_africa_registry_review_participants
		for_each_scope_loop = {
			array = gods_of_africa_candidate_registry
			if = {
				limit = { gods_of_africa_participant_can_register = yes }
				gods_of_africa_register_participant = yes
			}
		}
	}
}

# COUNTRY scope on the host. This is the single bounded initial discovery pass
# after consolidation. The candidate array was frozen by Event 012's one-time
# census before Scramble cleanup. Later reviews only inspect the maintained
# array; transaction-boundary callbacks may register named countries directly,
# while the owner-local review provides the bounded fallback for later major or
# player-control changes.
gods_of_africa_initialize_registry = {
	if = {
		limit = { gods_of_africa_active_host_is_valid = yes }
		for_each_scope_loop = {
			array = gods_of_africa_candidate_registry
			if = {
				limit = { gods_of_africa_participant_can_register = yes }
				gods_of_africa_register_participant = yes
			}
		}
	}
}

# COUNTRY scope on the Event 012 host. World Is One owns its own terminal
# campaign, so it must not start the ordinary Gods tribute loop. It does,
# however, inherit the bounded Scramble relationship evidence that the World
# invitation and opposition routes can consume. This one-time pass is called
# after the World package initializer has frozen `africa_scramble_participants`.
gods_of_africa_initialize_world_order_history = {
	if = {
		limit = {
			africa_is_current_host = yes
			has_global_flag = africa_world_order_open
			has_variable = africa_host_generation
			NOT = { has_country_flag = gods_of_africa_world_order_history_initialized }
		}
		set_country_flag = gods_of_africa_world_order_history_initialized
		set_variable = { gods_of_africa_restoration_generation = africa_host_generation }
		for_each_scope_loop = {
			array = africa_scramble_participants
			if = {
				limit = { exists = yes NOT = { has_capitulated = yes } }
				set_country_flag = gods_of_africa_world_history_seeded
				set_variable = { gods_of_africa_world_history_generation = event_target:africa_host.africa_host_generation }
				clr_country_flag = gods_of_africa_world_history_favored
				clr_country_flag = gods_of_africa_world_history_respected
				clr_country_flag = gods_of_africa_world_history_distrusted
				clr_country_flag = gods_of_africa_world_history_enemy
				if = {
					limit = {
						OR = {
							has_country_flag = africa_scramble_recognition_rejected
							has_country_flag = africa_scramble_ultimatum_issuer
							has_country_flag = africa_scramble_ideological_rival
							has_country_flag = africa_scramble_continent_unifier_fear
						}
					}
					set_country_flag = gods_of_africa_world_history_enemy
					set_variable = { gods_of_africa_world_history_status = constant:gods_of_africa_settlement.enemy }
				}
				else_if = {
					limit = {
						OR = {
							has_country_flag = africa_scramble_former_colonial_relationship
							has_country_flag = africa_scramble_foreign_base_holder
							has_country_flag = africa_scramble_resource_nationalisation_exposure
						}
					}
					set_country_flag = gods_of_africa_world_history_distrusted
					set_variable = { gods_of_africa_world_history_status = constant:gods_of_africa_settlement.distrusted }
				}
				else_if = {
					limit = {
						OR = {
							has_country_flag = africa_scramble_recognised_africa
							has_country_flag = africa_scramble_coalition_member
							has_country_flag = africa_scramble_south_africa_allied_relationship
						}
					}
					set_country_flag = gods_of_africa_world_history_favored
					set_variable = { gods_of_africa_world_history_status = constant:gods_of_africa_settlement.favored }
				}
				else = {
					set_country_flag = gods_of_africa_world_history_respected
					set_variable = { gods_of_africa_world_history_status = constant:gods_of_africa_settlement.respected }
				}
			}
		}
	}
}

# COUNTRY scope on the participant. `ROOT` is the host when called from the
# registry loop; the participant is stored in the host array as a scope, not as
# a tag or numeric id.
gods_of_africa_register_participant = {
	# Direct callers establish the same durable census coverage as callbacks.
	gods_of_africa_remember_candidate = yes
	if = {
		limit = { gods_of_africa_participant_can_register = yes }
		gods_of_africa_withdraw_protection_package = yes
		gods_of_africa_clear_report_receipts = yes
		gods_of_africa_clear_mobile_quote = yes
		gods_of_africa_remember_permanent_opt_out = yes
		gods_of_africa_clear_reconciliation_review = yes
		set_country_flag = gods_of_africa_participant
		clr_country_flag = gods_of_africa_leniency_used
		clr_country_flag = gods_of_africa_escalation_used
		clr_country_flag = gods_of_africa_enemy_marked
		clr_country_flag = gods_of_africa_punishment_memory_applied
		clr_country_flag = gods_of_africa_defiant_memory_applied
		clr_country_flag = gods_of_africa_reliable_memory_applied
		clr_country_flag = gods_of_africa_territory_returned
		clr_country_flag = gods_of_africa_recognition_granted
		clr_country_flag = gods_of_africa_enemy_support_ended
		# Resource-rights concessions are recorded for the current host
		# generation. A fresh enrollment must be eligible for that lane again;
		# ordinary cleanup deliberately leaves the historical receipt intact.
		clr_country_flag = gods_of_africa_resource_rights_granted
		set_variable = { gods_of_africa_participant_generation = event_target:africa_host.africa_host_generation }
		gods_of_africa_initialize_participant_history = yes
		if = {
			limit = { NOT = { has_variable = gods_of_africa_history_owner_id } }
			set_variable = { gods_of_africa_history_owner_id = THIS.id }
		}
		if = {
			limit = { NOT = { has_variable = gods_of_africa_history_origin_host_id } }
			set_variable = { gods_of_africa_history_origin_host_id = event_target:africa_host.africa_origin_host_id }
		}
		if = {
			limit = { has_country_flag = gods_of_africa_permanent_opt_out }
			set_country_flag = gods_of_africa_defiant
		}
		clear_variable = gods_of_africa_last_aid_family
		clear_variable = gods_of_africa_last_aid_amount
		clear_variable = gods_of_africa_last_aid_day
		clr_country_flag = gods_of_africa_aid_report_pending
		set_variable = { gods_of_africa_next_demand_date = global.num_days }
		add_to_variable = { gods_of_africa_next_demand_date = constant:gods_of_africa_timing.first_demand_delay_days }
		gods_of_africa_clear_substitute_slots = yes
		event_target:africa_host = {
			add_to_array = { array = gods_of_africa_participants value = PREV }
			add_to_variable = { gods_of_africa_participant_count = constant:gods_of_africa_value.one }
		}
		gods_of_africa_bind_participant_epoch = yes
		gods_of_africa_refresh_participant_floors = yes
		set_country_flag = gods_of_africa_intro_pending
		country_event = { id = chaosx.nr12.602 }
	}
}

# COUNTRY scope on a known country object. The one-time census and bounded
# transaction callbacks share this idempotent candidate insertion.
gods_of_africa_remember_candidate = {
	if = {
		limit = { gods_of_africa_candidate_can_be_remembered = yes }
		event_target:africa_host = {
			add_to_array = { array = gods_of_africa_candidate_registry value = PREV }
			add_to_variable = { gods_of_africa_candidate_count = constant:gods_of_africa_value.one }
		}
	}
}

# COUNTRY scope on a country named by an engine transaction. Remember minors
# before activation so a later major/player change can use the bounded review.
gods_of_africa_try_register_participant = {
	if = {
		limit = {
			exists = yes
			gods_of_africa_has_current_host = yes
			event_target:africa_host = {
				has_country_flag = gods_of_africa_candidate_census_complete
				NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
			}
		}
		gods_of_africa_remember_candidate = yes
		gods_of_africa_register_participant = yes
	}
}

# COUNTRY scope on a participant. Caller supplies positive
# `gods_of_africa_wrath_request` and direction enum. This central transaction
# prevents callers from bypassing war, territory, broken-treaty, defiance, or
# recent-catastrophe floors.
gods_of_africa_apply_wrath = {
	if = {
		limit = {
			gods_of_africa_participant_generation_is_current = yes
			has_variable = gods_of_africa_wrath_request
			check_variable = { gods_of_africa_wrath_request > constant:gods_of_africa_value.zero }
		}
		if = {
			limit = { NOT = { has_variable = gods_of_africa_wrath } }
			set_variable = { gods_of_africa_wrath = constant:gods_of_africa_value.starting_wrath }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = gods_of_africa_wrath_direction }
					check_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_increase }
				}
			}
			add_to_variable = { gods_of_africa_wrath = gods_of_africa_wrath_request }
		}
		else = { subtract_from_variable = { gods_of_africa_wrath = gods_of_africa_wrath_request } }
		clamp_variable = { var = gods_of_africa_wrath min = constant:gods_of_africa_value.minimum_wrath max = constant:gods_of_africa_value.maximum_wrath }
		gods_of_africa_enforce_wrath_floor = yes
		if = { limit = { check_variable = { gods_of_africa_wrath < constant:gods_of_africa_wrath.quiet } } set_variable = { gods_of_africa_wrath_band = constant:gods_of_africa_band.limited } }
		else_if = { limit = { check_variable = { gods_of_africa_wrath < constant:gods_of_africa_wrath.watchful } } set_variable = { gods_of_africa_wrath_band = constant:gods_of_africa_band.regional } }
		else_if = { limit = { check_variable = { gods_of_africa_wrath < constant:gods_of_africa_wrath.angered } } set_variable = { gods_of_africa_wrath_band = constant:gods_of_africa_band.continental } }
		else_if = { limit = { check_variable = { gods_of_africa_wrath < constant:gods_of_africa_wrath.hostile } } set_variable = { gods_of_africa_wrath_band = constant:gods_of_africa_band.global } }
		else_if = { limit = { check_variable = { gods_of_africa_wrath < constant:gods_of_africa_wrath.vengeful } } set_variable = { gods_of_africa_wrath_band = constant:gods_of_africa_band.global } }
		else = { set_variable = { gods_of_africa_wrath_band = constant:gods_of_africa_band.extreme } }
		clear_variable = gods_of_africa_wrath_request
		clear_variable = gods_of_africa_wrath_direction
	}
}

# The applicable floor is the highest floor among the current causes; a later,
# lower cause never lowers it.
gods_of_africa_enforce_wrath_floor = {
	if = {
		limit = { gods_of_africa_participant_generation_is_current = yes }
		set_temp_variable = { gods_of_africa_wrath_floor = constant:gods_of_africa_value.minimum_wrath }
		if = { limit = { has_country_flag = gods_of_africa_war_floor } clamp_temp_variable = { var = gods_of_africa_wrath_floor min = constant:gods_of_africa_wrath.floor_war } }
		if = { limit = { has_country_flag = gods_of_africa_territory_floor } clamp_temp_variable = { var = gods_of_africa_wrath_floor min = constant:gods_of_africa_wrath.floor_territory } }
		if = { limit = { has_country_flag = gods_of_africa_broken_treaty } clamp_temp_variable = { var = gods_of_africa_wrath_floor min = constant:gods_of_africa_wrath.floor_broken_treaty } }
		if = { limit = { has_country_flag = gods_of_africa_defiant } clamp_temp_variable = { var = gods_of_africa_wrath_floor min = constant:gods_of_africa_wrath.floor_defiant } }
		if = { limit = { has_country_flag = gods_of_africa_recent_extreme_punishment } clamp_temp_variable = { var = gods_of_africa_wrath_floor min = constant:gods_of_africa_wrath.floor_recent_extreme } }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = gods_of_africa_wrath }
					check_variable = { gods_of_africa_wrath < gods_of_africa_wrath_floor }
				}
			}
			set_variable = { gods_of_africa_wrath = gods_of_africa_wrath_floor }
		}
		clamp_variable = { var = gods_of_africa_wrath min = constant:gods_of_africa_value.minimum_wrath max = constant:gods_of_africa_value.maximum_wrath }
	}
}

# COUNTRY scope on a participant, called only from the host registry loop.
gods_of_africa_refresh_participant_floors = {
	if = {
		limit = { gods_of_africa_participant_generation_is_current = yes }
		if = { limit = { has_war_with = event_target:africa_host } set_country_flag = gods_of_africa_war_floor }
		else = { clr_country_flag = gods_of_africa_war_floor }
		gods_of_africa_observe_agreement_war = yes
		if = {
			limit = { gods_of_africa_participant_holds_host_core_state = yes }
			set_country_flag = gods_of_africa_territory_floor
		}
		else = { clr_country_flag = gods_of_africa_territory_floor }
		if = {
			limit = {
				has_country_flag = gods_of_africa_recent_extreme_punishment
				has_variable = gods_of_africa_recent_extreme_until
				check_variable = { var = global.num_days value = gods_of_africa_recent_extreme_until compare = greater_than_or_equals }
			}
			clr_country_flag = gods_of_africa_recent_extreme_punishment
		}
		# Bounded conduct observation for this participant only: war episodes,
		# occupied and owned host-core states, and prior hostility. It refreshes
		# the war and territory floor causes from the same observation.
		gods_of_africa_refresh_grading_conduct = yes
		gods_of_africa_enforce_wrath_floor = yes
	}
}

# Capacity requirements include the protected operating reserve. The public
# payment amount remains the amount transferred; these helpers keep the extra
# floor in a separate hidden variable used only by the affordability triggers.
gods_of_africa_compute_capacity_requirement = {
	if = {
		limit = {
			has_variable = gods_of_africa_capacity_work_family
			has_variable = gods_of_africa_capacity_work_amount
		}
		set_variable = { gods_of_africa_capacity_requirement = gods_of_africa_capacity_work_amount }
		if = {
			limit = {
				OR = {
					check_variable = { gods_of_africa_capacity_work_family = constant:gods_of_africa_demand_family.armor }
					check_variable = { gods_of_africa_capacity_work_family = constant:gods_of_africa_demand_family.mobile_artillery }
				}
			}
			# Typed capacity already removes operating and reinforcement reserves.
			set_variable = { gods_of_africa_capacity_requirement = gods_of_africa_capacity_work_amount }
		}
		else_if = {
			limit = {
				OR = {

# COUNTRY scope on the host. The profile is a small, inspectable snapshot of
# maintained Event 012 pressure rather than a second world-wide scan.
gods_of_africa_build_need_profile = {
	if = {
		limit = { africa_is_current_host = yes }
		gods_of_africa_build_mobile_need_profile = yes
		set_variable = { gods_of_africa_need_land_equipment = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_fuel = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_manpower = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_industry = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_emergency = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_transport = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_aircraft = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_support_equipment = constant:gods_of_africa_value.zero }
		set_variable = { gods_of_africa_need_resources = constant:gods_of_africa_value.zero }
		clr_country_flag = gods_of_africa_armor_needed
		set_temp_variable = { gods_of_africa_armor_stock = constant:gods_of_africa_value.zero }
		set_temp_variable = { gods_of_africa_armor_deficit = constant:gods_of_africa_value.zero }
		set_temp_variable = { gods_of_africa_armor_spare = constant:gods_of_africa_value.zero }
		if = {
			limit = { set_temp_variable = { gods_of_africa_armor_required_amount = constant:gods_of_africa_value.zero } gods_of_africa_armor_capacity_is_sufficient = yes }
			set_temp_variable = { gods_of_africa_armor_host_requirement = gods_of_africa_armor_deficit }
			if = {
				limit = { has_global_flag = africa_evolution_ii_logged }
				add_to_temp_variable = { gods_of_africa_armor_host_requirement = constant:gods_of_africa_armor.host_stockpile_floor }
			}
			if = {
				limit = { check_variable = { gods_of_africa_armor_stock < gods_of_africa_armor_host_requirement } }
				set_country_flag = gods_of_africa_armor_needed
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = africa_common_defence_stockpile }
					check_variable = { africa_common_defence_stockpile < constant:gods_of_africa_transaction.land_equipment_amount }
				}
			}
			set_variable = { gods_of_africa_need_land_equipment = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
				NOT = { has_fuel > constant:gods_of_africa_transaction.fuel_amount }
			}
			set_variable = { gods_of_africa_need_fuel = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
			OR = {
					NOT = { has_variable = africa_continental_war_readiness }
					check_variable = { africa_continental_war_readiness < constant:africa_measure.high }
				}
			}
			set_variable = { gods_of_africa_need_manpower = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
			OR = {
					NOT = { has_variable = africa_central_capacity }
					check_variable = { africa_central_capacity < constant:africa_measure.medium }
				}
			}
			set_variable = { gods_of_africa_need_industry = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
				# An absent debt variable is treated as a healthy zero, not as an
				# emergency. Africa Is One initializes the variable, but this explicit
				# guard also keeps a stale or partially restored host from manufacturing
				# an emergency demand. The needs-profile focus deliberately opens the
				# relief lane earlier, at the medium debt band.
				OR = {
					AND = {
						has_variable = africa_emergency_rule_debt
						check_variable = { africa_emergency_rule_debt > constant:africa_measure.high }
					}
					AND = {
						has_country_flag = gods_of_africa_focus_needs_profile_unlocked
						has_variable = africa_emergency_rule_debt
						check_variable = { africa_emergency_rule_debt > constant:africa_measure.medium }
					}
				}
			}
			set_variable = { gods_of_africa_need_emergency = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
				OR = {
					AND = {
						has_equipment = { convoy_1 < constant:gods_of_africa_transaction.transport_amount }
						has_equipment = { train_equipment < constant:gods_of_africa_transaction.transport_amount }
					}
					AND = {
						has_country_flag = gods_of_africa_focus_logistics_support_unlocked
						OR = {
							has_equipment = { convoy_1 < constant:gods_of_africa_transaction.transport_amount }
							has_equipment = { train_equipment < constant:gods_of_africa_transaction.transport_amount }
						}
					}
				}
			}
			set_variable = { gods_of_africa_need_transport = constant:gods_of_africa_value.one }
		}
		if = {
			limit = { has_equipment = { fighter_equipment_1 < constant:gods_of_africa_transaction.aircraft_amount } }
			set_variable = { gods_of_africa_need_aircraft = constant:gods_of_africa_value.one }
		}
		if = {
			limit = { has_equipment = { support_equipment < constant:gods_of_africa_transaction.support_equipment_amount } }
			set_variable = { gods_of_africa_need_support_equipment = constant:gods_of_africa_value.one }
		}
		if = {
			limit = {
				NOT = {
					set_temp_variable = { gods_of_africa_resource_threshold = constant:gods_of_africa_transaction.resources_amount }
					meta_trigger = {
						text = { has_resources_in_country = { resource = steel amount > [THRESHOLD] extracted = yes } }
						THRESHOLD = "[?gods_of_africa_resource_threshold|.0]"
					}
				}
			}
			set_variable = { gods_of_africa_need_resources = constant:gods_of_africa_value.one }
		}
	}
}

# COUNTRY scope on the host. A priority family is tried first, then a fixed
# deterministic fallback ladder. This avoids adding an implicit random-list
# probability surface to the Gods contract system.

	clear_variable = gods_of_africa_selected_family
	clear_variable = gods_of_africa_selected_mobile_type
	clear_variable = gods_of_africa_selected_amount

	if = {
		limit = { has_event_target = gods_of_africa_demand_participant }
		event_target:gods_of_africa_demand_participant = { gods_of_africa_quote_mobile_tribute = yes }
	}
	if = {
		# A priority is meaningful only when its maintained need is active. This
		# keeps host policy as a real tie-breaker without manufacturing a demand
		# for a healthy lane.
		limit = {
			has_variable = gods_of_africa_priority_family
			OR = {
				AND = {
					check_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.land_equipment }
					check_variable = { gods_of_africa_need_land_equipment > constant:gods_of_africa_value.zero }

					check_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.armor }
					has_country_flag = gods_of_africa_armor_needed
				}
				AND = {
					check_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.mobile_artillery }
					has_event_target = gods_of_africa_demand_participant

						has_variable = gods_of_africa_mobile_quote_type
						check_variable = { gods_of_africa_mobile_quote_amount > constant:gods_of_africa_value.zero }
					}
				}
				AND = {
					check_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.fuel }
					check_variable = { gods_of_africa_need_fuel > constant:gods_of_africa_value.zero }

					check_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.manpower }
					check_variable = { gods_of_africa_need_manpower > constant:gods_of_africa_value.zero }

					check_variable = { gods_of_africa_priority_family = constant:gods_of_africa_demand_family.emergency_relief }

			limit = { event_target:gods_of_africa_demand_participant = { check_variable = { gods_of_africa_capacity_result > constant:gods_of_africa_value.zero } } }
			set_variable = { gods_of_africa_selected_family = event_target:gods_of_africa_demand_participant.gods_of_africa_payment_family }
			set_variable = { gods_of_africa_selected_amount = event_target:gods_of_africa_demand_participant.gods_of_africa_payment_amount }
			if = {
				limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.mobile_artillery } }
				set_variable = { gods_of_africa_selected_mobile_type = event_target:gods_of_africa_demand_participant.gods_of_africa_payment_mobile_type }
			}
			set_temp_variable = { gods_of_africa_family_selected = constant:gods_of_africa_value.one }
		}
	}
	if = {
		limit = {
			check_variable = { gods_of_africa_family_selected = constant:gods_of_africa_value.zero }
			has_variable = gods_of_africa_family_candidate
			has_event_target = gods_of_africa_demand_participant
		}
		event_target:gods_of_africa_demand_participant = {
			set_variable = { gods_of_africa_payment_family = constant:gods_of_africa_demand_family.emergency_relief }
			clear_variable = gods_of_africa_payment_mobile_type
			set_variable = { gods_of_africa_payment_amount = constant:gods_of_africa_transaction.emergency_amount }
			gods_of_africa_build_capacity_profile = yes
		}
		if = {
			limit = { event_target:gods_of_africa_demand_participant = { check_variable = { gods_of_africa_capacity_result > constant:gods_of_africa_value.zero } } }
			set_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.emergency_relief }
			set_variable = { gods_of_africa_selected_amount = constant:gods_of_africa_transaction.emergency_amount }
		}
	}
}

# COUNTRY scope on the host. The family identity is immutable once the
# contract is created, while the amount is scaled from current capacity,
# Strength, Wrath, Chaos, needs, conduct, doctrine, and repetition. The final
# quote is rechecked against the participant's live stockpile before creation.
gods_of_africa_calculate_demand_amount = {
	clear_variable = gods_of_africa_contract_amount_base
	if = {
		limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.land_equipment } }
		set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.land_equipment_amount }
	}
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.support_equipment } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.support_equipment_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.mobile_artillery } } set_variable = { gods_of_africa_contract_amount_base = gods_of_africa_selected_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.armor } } set_variable = { gods_of_africa_contract_amount_base = gods_of_africa_selected_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.aircraft } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.aircraft_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.fuel } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.fuel_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.transport } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.transport_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.manpower } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.manpower_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.industry_repair } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.industry_pp_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.resources } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.resources_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.recognition } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.recognition_pp_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.territory } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.territory_pp_amount } }
	else_if = { limit = { check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.alliance_access } } set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.recognition_pp_amount } }
	else = { set_variable = { gods_of_africa_contract_amount_base = constant:gods_of_africa_transaction.emergency_amount } }
	set_variable = { gods_of_africa_contract_amount = gods_of_africa_contract_amount_base }
	set_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.base_percent }
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				OR = {
					check_variable = { var = num_of_civilian_factories value = constant:gods_of_africa_demand_scale.very_high_factory_threshold compare = greater_than }
					check_variable = { var = num_of_military_factories value = constant:gods_of_africa_demand_scale.very_high_factory_threshold compare = greater_than }
				}
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.capacity_very_high_delta }
	}
	else_if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				OR = {
					check_variable = { var = num_of_civilian_factories value = constant:gods_of_africa_demand_scale.major_factory_threshold compare = greater_than }
					check_variable = { var = num_of_military_factories value = constant:gods_of_africa_demand_scale.major_factory_threshold compare = greater_than }
				}
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.capacity_major_delta }
	}
	if = { limit = { has_variable = gods_of_africa_strength check_variable = { gods_of_africa_strength > constant:gods_of_africa_value.global_strength } } add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.strength_delta } }
	if = { limit = { check_variable = { var = global.chaos_meter_value value = constant:gods_of_africa_value.catastrophe_chaos_gate compare = greater_than_or_equals } } add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.chaos_delta } }
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				has_variable = gods_of_africa_wrath
				check_variable = { gods_of_africa_wrath > constant:gods_of_africa_wrath.hostile }
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.wrath_delta }
	}
	if = {
		limit = {
			OR = {
					check_variable = { gods_of_africa_need_land_equipment > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_fuel > constant:gods_of_africa_value.zero }
					has_country_flag = gods_of_africa_armor_needed
					check_variable = { gods_of_africa_truck_need_amount > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_artillery_need_amount > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_manpower > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_industry > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_emergency > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_transport > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_aircraft > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_support_equipment > constant:gods_of_africa_value.zero }
					AND = {
						check_variable = { gods_of_africa_need_resources > constant:gods_of_africa_value.zero }
						event_target:gods_of_africa_demand_participant = { NOT = { has_country_flag = gods_of_africa_resource_rights_granted } }
					}
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.need_delta }
	}
	if = {
		limit = {
			OR = {
				check_variable = { gods_of_africa_doctrine = constant:gods_of_africa_doctrine.reciprocal_covenant }
				has_country_flag = gods_of_africa_focus_reciprocal_route
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.reciprocal_delta }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { gods_of_africa_doctrine = constant:gods_of_africa_doctrine.sovereign_exaction }
				has_country_flag = gods_of_africa_focus_exaction_route
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.sovereign_delta }
	}
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				OR = {
					check_variable = { gods_of_africa_demand_failures > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_refusal_count > constant:gods_of_africa_value.zero }
				}
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.conduct_delta }
	}
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				set_temp_variable = { gods_of_africa_repetition_family = event_target:africa_host.gods_of_africa_selected_family }
				set_temp_variable = { gods_of_africa_repetition_mobile_type = constant:gods_of_africa_value.zero }
				if = {
					limit = { check_variable = { gods_of_africa_repetition_family = constant:gods_of_africa_demand_family.mobile_artillery } }
					set_temp_variable = { gods_of_africa_repetition_mobile_type = event_target:africa_host.gods_of_africa_selected_mobile_type }
				}
				gods_of_africa_repetition_scale_applies = yes
			}
		}
		add_to_temp_variable = { gods_of_africa_demand_scale = constant:gods_of_africa_demand_scale.repetition_delta }
	}
	multiply_variable = { gods_of_africa_contract_amount = gods_of_africa_demand_scale }
	divide_variable = { gods_of_africa_contract_amount = constant:gods_of_africa_demand_scale.percent_divisor }
	clamp_variable = { var = gods_of_africa_contract_amount min = constant:gods_of_africa_value.minimum_demand_amount max = constant:gods_of_africa_value.maximum_demand_amount }
	if = {
		limit = { set_temp_variable = { gods_of_africa_family_to_check = gods_of_africa_selected_family } gods_of_africa_family_is_equipment = yes }
		set_temp_variable = { gods_of_africa_equipment_amount = gods_of_africa_contract_amount }
		gods_of_africa_floor_equipment_amount = yes
		set_variable = { gods_of_africa_contract_amount = gods_of_africa_equipment_amount }
	}
	if = {
		limit = {
			check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.armor }
			has_event_target = gods_of_africa_demand_participant
		}
		clamp_variable = { var = gods_of_africa_contract_amount min = constant:gods_of_africa_armor.minimum_shipment max = event_target:gods_of_africa_demand_participant.gods_of_africa_armor_quote_ceiling }
		set_temp_variable = { gods_of_africa_equipment_amount = gods_of_africa_contract_amount }
		gods_of_africa_floor_equipment_amount = yes
		set_variable = { gods_of_africa_contract_amount = gods_of_africa_equipment_amount }
	}
	if = {
		limit = {
			check_variable = { gods_of_africa_selected_family = constant:gods_of_africa_demand_family.mobile_artillery }
			has_event_target = gods_of_africa_demand_participant
		}
		clamp_variable = { var = gods_of_africa_contract_amount min = constant:gods_of_africa_mobile.minimum_shipment max = event_target:gods_of_africa_demand_participant.gods_of_africa_mobile_quote_ceiling }
		set_temp_variable = { gods_of_africa_equipment_amount = gods_of_africa_contract_amount }
		gods_of_africa_floor_equipment_amount = yes
		set_variable = { gods_of_africa_contract_amount = gods_of_africa_equipment_amount }
	}
	set_variable = { gods_of_africa_contract_band = constant:gods_of_africa_demand.standard_band }
	# A first contact is a small ceremonial test. Later contracts move up the
	# severity ladder only when the participant's conduct or the host's strength
	# makes that escalation credible. Emergency need is applied last so it can
	# override an ordinary band without making a routine demand punitive.
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				OR = {
					NOT = { has_variable = gods_of_africa_history_count }
					check_variable = { gods_of_africa_history_count = constant:gods_of_africa_value.zero }
				}
			}
		}
		set_variable = { gods_of_africa_contract_band = constant:gods_of_africa_demand.ceremonial_band }
	}
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = {
				has_variable = gods_of_africa_history_count
				check_variable = { gods_of_africa_history_count > constant:gods_of_africa_value.zero }
				OR = {
					has_country_flag = gods_of_africa_offender
					AND = {
						has_variable = gods_of_africa_demand_failures
						check_variable = { var = gods_of_africa_demand_failures value = constant:gods_of_africa_value.repeated_conduct_threshold compare = greater_than_or_equals }
					}
					AND = {
						has_variable = gods_of_africa_refusal_count
						check_variable = { var = gods_of_africa_refusal_count value = constant:gods_of_africa_value.repeated_conduct_threshold compare = greater_than_or_equals }
					}
					AND = {
						has_variable = gods_of_africa_offense_count
						check_variable = { gods_of_africa_offense_count > constant:gods_of_africa_value.zero }
					}
				}
				OR = {
					AND = {
						has_variable = gods_of_africa_wrath
						check_variable = { var = gods_of_africa_wrath value = constant:gods_of_africa_wrath.vengeful compare = greater_than_or_equals }
					}
					event_target:africa_host = {
						has_variable = gods_of_africa_strength
						check_variable = { var = gods_of_africa_strength value = constant:gods_of_africa_value.global_strength compare = greater_than_or_equals }
					}
				}
			}
		}
		set_variable = { gods_of_africa_contract_band = constant:gods_of_africa_demand.punitive_band }
	}
	if = {
		limit = {
			has_variable = gods_of_africa_strength
			check_variable = { var = gods_of_africa_strength value = constant:gods_of_africa_value.global_strength compare = greater_than_or_equals }
			has_event_target = gods_of_africa_demand_participant
			# A conduct-driven punitive quote is already the highest band; later
			# capability checks may not silently downgrade it.
			check_variable = {
				var = gods_of_africa_contract_band
				value = constant:gods_of_africa_demand.severe_band
				compare = less_than
			}
			event_target:gods_of_africa_demand_participant = {
				has_variable = gods_of_africa_history_count
				check_variable = { gods_of_africa_history_count > constant:gods_of_africa_value.zero }
			}
		}
		set_variable = { gods_of_africa_contract_band = constant:gods_of_africa_demand.severe_band }
	}
	if = {
		limit = {
			has_variable = gods_of_africa_need_emergency
			check_variable = { gods_of_africa_need_emergency > constant:gods_of_africa_value.zero }
			# Emergency need can lift a normal quote, but never soften the
			# punitive band already earned by repeated offenses.
			check_variable = {
				var = gods_of_africa_contract_band
				value = constant:gods_of_africa_demand.emergency_band
				compare = less_than
			}
		}
		set_variable = { gods_of_africa_contract_band = constant:gods_of_africa_demand.emergency_band }
	}
	if = {
		limit = { has_event_target = gods_of_africa_demand_participant }
		event_target:gods_of_africa_demand_participant = {
			set_variable = { gods_of_africa_payment_family = event_target:africa_host.gods_of_africa_selected_family }
			set_variable = { gods_of_africa_payment_amount = event_target:africa_host.gods_of_africa_contract_amount }
			clear_variable = gods_of_africa_payment_mobile_type
			if = {
				limit = { check_variable = { gods_of_africa_payment_family = constant:gods_of_africa_demand_family.mobile_artillery } }
				set_variable = { gods_of_africa_payment_mobile_type = event_target:africa_host.gods_of_africa_selected_mobile_type }
			}
			gods_of_africa_build_capacity_profile = yes
		}
		if = {
			limit = { event_target:gods_of_africa_demand_participant = { NOT = { check_variable = { gods_of_africa_capacity_result > constant:gods_of_africa_value.zero } } } }
			set_variable = { gods_of_africa_contract_amount = gods_of_africa_contract_amount_base }
			event_target:gods_of_africa_demand_participant = { set_variable = { gods_of_africa_payment_amount = event_target:africa_host.gods_of_africa_contract_amount } gods_of_africa_build_capacity_profile = yes }
		}
	}
}

# COUNTRY scope on a participant. The candidate families are frozen with the
# contract and deliberately stay compact: three useful alternatives rather
# than a store-like list of every resource. The participant's live capacity is
# checked later when negotiation opens, so an offer never becomes an implicit
# grant merely because a family was named here.
gods_of_africa_build_substitute_set = {
	gods_of_africa_clear_substitute_slots = yes
	if = {
		limit = {
			has_variable = gods_of_africa_demand_family
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.land_equipment }

# COUNTRY scope on the host. The participant target is selected and the
# contract is written in one transaction so a stale target cannot be rerolled.
gods_of_africa_prepare_demand_for_participant = {
	clear_variable = gods_of_africa_selected_family
	clear_variable = gods_of_africa_selected_amount
	clear_variable = gods_of_africa_selected_mobile_type
	if = {
		limit = { has_event_target = gods_of_africa_demand_participant }
		event_target:gods_of_africa_demand_participant = { gods_of_africa_clear_mobile_quote = yes }
	}
	if = {
		limit = {
			has_event_target = gods_of_africa_demand_participant
			event_target:gods_of_africa_demand_participant = { gods_of_africa_participant_can_receive_demand = yes }
		}
		gods_of_africa_build_need_profile = yes
		# A priority is only a tie-breaker among real African needs. Do not let
		# the default land priority turn an empty need profile into an arbitrary
		# recurring demand.
		if = {
			limit = {
				OR = {
					check_variable = { gods_of_africa_need_land_equipment > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_fuel > constant:gods_of_africa_value.zero }
					has_country_flag = gods_of_africa_armor_needed
					check_variable = { gods_of_africa_truck_need_amount > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_artillery_need_amount > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_manpower > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_industry > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_emergency > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_transport > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_aircraft > constant:gods_of_africa_value.zero }
					check_variable = { gods_of_africa_need_support_equipment > constant:gods_of_africa_value.zero }
					AND = {
						has_country_flag = gods_of_africa_focus_territory_unlocked
						event_target:gods_of_africa_demand_participant = { gods_of_africa_participant_holds_host_core_state = yes }
					}
					AND = {
						has_country_flag = gods_of_africa_focus_recognition_unlocked
						event_target:gods_of_africa_demand_participant = { NOT = { has_country_flag = gods_of_africa_recognition_granted } }
					}
					AND = {
						has_country_flag = gods_of_africa_focus_sponsorship_unlocked
						event_target:gods_of_africa_demand_participant = { NOT = { has_country_flag = gods_of_africa_enemy_support_ended } }
					}
				}
			}
			gods_of_africa_select_demand_family = yes
		}
		if = {
			limit = { has_variable = gods_of_africa_selected_family }
			gods_of_africa_calculate_demand_amount = yes
			# The scaled quote is checked again after the capacity fallback. If
			# neither the scaled nor base quote is affordable, fail closed instead
			# of creating an impossible contract with stale payment variables.
			if = {
				limit = {
					has_variable = gods_of_africa_selected_family
					event_target:gods_of_africa_demand_participant = {
						gods_of_africa_participant_has_capacity_for_family = yes
					}
				}
				gods_of_africa_calculate_demand_deadline = yes
				gods_of_africa_create_demand_contract = yes
			}
			else = {
				event_target:gods_of_africa_demand_participant = {
					clear_variable = gods_of_africa_payment_family
					clear_variable = gods_of_africa_payment_mobile_type
					clear_variable = gods_of_africa_payment_amount
					clear_variable = gods_of_africa_capacity_result
				}
				clear_variable = gods_of_africa_selected_family
				clear_variable = gods_of_africa_selected_mobile_type
				clear_variable = gods_of_africa_selected_amount
			}
		}
		event_target:gods_of_africa_demand_participant = {
			clear_variable = gods_of_africa_armor_quote_amount
			clear_variable = gods_of_africa_armor_quote_ceiling
			gods_of_africa_clear_mobile_quote = yes
		}
		clear_variable = gods_of_africa_selected_mobile_type
	}
}

gods_of_africa_dispatch_next_demand = {
	# This guard is a normal host variable. Temporary variables have no stable
	# scope inside `for_each_scope_loop`, so a participant must never read or
	# clear the dispatch lock through a scoped temp reference.
	set_variable = { gods_of_africa_dispatch_locked = constant:gods_of_africa_value.zero }
	for_each_scope_loop = {
		array = gods_of_africa_participants
		if = {
			limit = {
				check_variable = {
					var = ROOT.gods_of_africa_dispatch_locked
					value = constant:gods_of_africa_value.zero
					compare = equals
				}
				gods_of_africa_participant_can_receive_demand = yes
			}
			save_event_target_as = gods_of_africa_demand_participant
			ROOT = { set_variable = { gods_of_africa_dispatch_locked = constant:gods_of_africa_value.one } }
		}
	}
	if = { limit = { has_event_target = gods_of_africa_demand_participant } gods_of_africa_prepare_demand_for_participant = yes }
	clear_variable = gods_of_africa_dispatch_locked
}

# COUNTRY scope on a participant. This is the only payment debit path. Each
# material branch opens a measured receipt, emits its real native transfer and
# closes the receipt inside this effect, so gods_of_africa_payment_delivered is
# the amount that actually left the participant and reached Africa. Political
# and territorial families have no material quantity; their delivered amount is
# the political capital paid.
gods_of_africa_transfer_payment = {
	set_variable = { gods_of_africa_payment_success = constant:gods_of_africa_value.zero }
	set_variable = { gods_of_africa_payment_delivered = constant:gods_of_africa_value.zero }
	gods_of_africa_set_payment_capacity_requirement = yes
	if = {
		limit = {
			gods_of_africa_demand_contract_is_current = yes
			gods_of_africa_contract_ledger_can_close = yes
			has_variable = gods_of_africa_payment_family
			has_variable = gods_of_africa_payment_amount
			check_variable = { gods_of_africa_payment_amount > constant:gods_of_africa_value.zero }
			gods_of_africa_payment_family_is_valid = yes
			gods_of_africa_mobile_payment_matches_contract = yes
			gods_of_africa_participant_has_capacity_for_family = yes
			has_event_target = africa_host
		}
		save_event_target_as = gods_of_africa_payment_participant
		set_variable = { gods_of_africa_payment_credit = gods_of_africa_payment_amount }
		if = {
			limit = { check_variable = { gods_of_africa_payment_family = constant:gods_of_africa_demand_family.land_equipment } }
			set_temp_variable = { gods_of_africa_payment_receipt_kind = constant:gods_of_africa_contribution_kind.infantry }
			gods_of_africa_begin_payment_receipt = yes
			if = {
				limit = { has_country_flag = gods_of_africa_material_transfer_begin_ready }
				send_equipment = { equipment = infantry_equipment amount = gods_of_africa_payment_amount target = event_target:africa_host }
				gods_of_africa_complete_payment_receipt = yes
				event_target:africa_host = {
					# The maintained reserve is the same Event 012 ledger used by the
					# need profile and settlement delivery. It follows the measured
					# infantry delivery, never the requested amount.
					add_to_variable = { africa_common_defence_stockpile = event_target:gods_of_africa_payment_participant.gods_of_africa_payment_delivered }
				}
			}
		}
		else_if = {

				set_temp_variable = { gods_of_africa_payment_receipt_kind = constant:gods_of_africa_contribution_kind.trucks }

				gods_of_africa_complete_payment_receipt = yes

					set_temp_variable = { gods_of_africa_resource_threshold = constant:gods_of_africa_transaction.resources_amount }

					check_variable = { gods_of_africa_payment_receipt_kind = constant:gods_of_africa_contribution_kind.emergency_pp }
					has_country_flag = gods_of_africa_material_transfer_begin_ready
				}
				gods_of_africa_complete_material_transfer = yes
			}
			set_variable = { gods_of_africa_payment_delivered = gods_of_africa_payment_amount }
			if = {
				limit = {
					check_variable = { gods_of_africa_payment_family = constant:gods_of_africa_demand_family.territory }
					any_controlled_state = {
						is_on_continent = africa
						is_core_of = event_target:africa_host
					}
				}
				gods_of_africa_return_host_core_state = yes
			}
		}
		if = {
			limit = { check_variable = { gods_of_africa_payment_delivered > constant:gods_of_africa_value.zero } }
			if = {
				limit = { check_variable = { gods_of_africa_payment_family = constant:gods_of_africa_demand_family.recognition } }

			}

			}
			set_variable = { gods_of_africa_payment_success = constant:gods_of_africa_value.one }
		}
		clear_variable = gods_of_africa_payment_amount
		clear_variable = gods_of_africa_payment_family
		clear_variable = gods_of_africa_payment_mobile_type
		clear_variable = gods_of_africa_payment_credit
	}

# COUNTRY scope on a participant inside the territory payment. One controlled
# African core state of the host is returned. The return counts once per state,
# and only when Africa then both owns and controls it.
gods_of_africa_return_host_core_state = {
	random_controlled_state = {
		limit = {
			is_on_continent = africa
			is_core_of = event_target:africa_host
		}
		save_event_target_as = gods_of_africa_returned_core_state
		PREV = { set_variable = { gods_of_africa_return_state = PREV } }
	}

		limit = { has_event_target = gods_of_africa_returned_core_state }
		gods_of_africa_begin_core_return = yes
		event_target:gods_of_africa_returned_core_state = { transfer_state_to = event_target:africa_host }

			limit = { has_country_flag = gods_of_africa_core_return_begin_ready }
			gods_of_africa_complete_core_return = yes
		}
		set_country_flag = gods_of_africa_territory_returned
	}
	clear_variable = gods_of_africa_return_state
}

# COUNTRY scope on a participant. The contract closes only when the measured
# delivery covers the whole remaining balance. A short delivery stays a partial
# payment, so the remainder is still owed; full compliance and accepted
# substitutes close through the same idempotent cleanup.
gods_of_africa_resolve_full_compliance = {
	if = {
		limit = { gods_of_africa_demand_contract_is_current = yes gods_of_africa_contract_ledger_can_close = yes }
		set_variable = { gods_of_africa_payment_family = gods_of_africa_demand_family }
		gods_of_africa_bind_payment_mobile_type = yes
		set_variable = { gods_of_africa_payment_amount = gods_of_africa_demand_remaining }
		gods_of_africa_transfer_payment = yes
		if = {
			limit = {
				check_variable = { gods_of_africa_payment_success > constant:gods_of_africa_value.zero }
				check_variable = { var = gods_of_africa_payment_delivered value = gods_of_africa_demand_remaining compare = greater_than_or_equals }
			}
			set_temp_variable = { gods_of_africa_response_paid_amount = gods_of_africa_payment_delivered }
			set_country_flag = gods_of_africa_contract_fulfillment_authorized
			set_temp_variable = { gods_of_africa_contract_close_kind = constant:gods_of_africa_contract_close.full_original }
			if = { limit = { gods_of_africa_contract_has_substitute_terms = yes } set_temp_variable = { gods_of_africa_contract_close_kind = constant:gods_of_africa_contract_close.honored_substitute } }
			gods_of_africa_close_contract_ledger = yes
			if = {
				limit = { gods_of_africa_contract_has_substitute_terms = yes }
				set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.substitute_magnitude }
				set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.substitute }
				set_temp_variable = { gods_of_africa_report_kind = constant:gods_of_africa_history_kind.substitute }
				set_temp_variable = { gods_of_africa_report_amount = gods_of_africa_response_paid_amount }
				set_temp_variable = { gods_of_africa_report_balance = constant:gods_of_africa_value.zero }
				gods_of_africa_capture_response_report = yes
			}
			else = {
				set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.compliance_magnitude }
				set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.compliance }
				set_temp_variable = { gods_of_africa_report_kind = constant:gods_of_africa_history_kind.compliance }
				set_temp_variable = { gods_of_africa_report_amount = gods_of_africa_response_paid_amount }
				set_temp_variable = { gods_of_africa_report_balance = constant:gods_of_africa_value.zero }
				gods_of_africa_capture_response_report = yes
			}
			set_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_decrease }
			gods_of_africa_apply_wrath = yes
			gods_of_africa_record_history = yes
			set_variable = { gods_of_africa_last_demand_status = constant:gods_of_africa_demand_status.resolved }
			set_country_flag = gods_of_africa_outcome_report_pending
			if = {
				limit = { NOT = { has_country_flag = gods_of_africa_reliable_memory_applied } }
				set_country_flag = gods_of_africa_reliable_memory_applied
				save_event_target_as = gods_of_africa_memory_participant
				add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_reliable_partner }
				event_target:africa_host = { add_opinion_modifier = { target = event_target:gods_of_africa_memory_participant modifier = gods_of_africa_reliable_partner } }
			}
			gods_of_africa_clear_demand = yes
			country_event = { id = chaosx.nr12.604 }
		}
		else_if = {
			limit = { check_variable = { gods_of_africa_payment_success > constant:gods_of_africa_value.zero } }
			gods_of_africa_apply_partial_delivery = yes
		}
	}
}

# COUNTRY scope on a participant after a successful measured transfer that did
# not cover the whole balance, or after a partial shipment. Only the measured
# delivery reduces the balance; the remainder stays owed under the same running
# deadline.
gods_of_africa_apply_partial_delivery = {
	# A begun partial payment binds the participant to the remaining terms.
	gods_of_africa_mark_contract_agreed = yes
	subtract_from_variable = { gods_of_africa_demand_remaining = gods_of_africa_payment_delivered }
	if = {
		limit = { check_variable = { var = gods_of_africa_demand_remaining value = constant:gods_of_africa_value.zero compare = less_than_or_equals } }
		set_country_flag = gods_of_africa_contract_fulfillment_authorized
		set_temp_variable = { gods_of_africa_contract_close_kind = constant:gods_of_africa_contract_close.full_original }
		if = { limit = { gods_of_africa_contract_has_substitute_terms = yes } set_temp_variable = { gods_of_africa_contract_close_kind = constant:gods_of_africa_contract_close.honored_substitute } }
		gods_of_africa_close_contract_ledger = yes
	}
	set_temp_variable = { gods_of_africa_report_kind = constant:gods_of_africa_history_kind.partial }
	set_temp_variable = { gods_of_africa_report_amount = gods_of_africa_payment_delivered }
	set_temp_variable = { gods_of_africa_report_balance = gods_of_africa_demand_remaining }
	gods_of_africa_capture_response_report = yes
	set_variable = { gods_of_africa_demand_status = constant:gods_of_africa_demand_status.partial }
	set_variable = { gods_of_africa_next_partial_payment_day = global.num_days }
	add_to_variable = { gods_of_africa_next_partial_payment_day = constant:gods_of_africa_timing.partial_payment_cooldown_days }
	# Keep the ordinary comply cost pointed at the reduced balance. The partial
	# decision uses its own frozen requirement, so this does not change the
	# cooldown gate.
	set_variable = { gods_of_africa_payment_family = gods_of_africa_demand_family }
	gods_of_africa_bind_payment_mobile_type = yes
	set_variable = { gods_of_africa_payment_amount = gods_of_africa_demand_remaining }
	gods_of_africa_set_payment_capacity_requirement = yes
	set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.partial }
	gods_of_africa_record_history = yes
	set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.negotiation_magnitude }
	set_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_decrease }
	gods_of_africa_apply_wrath = yes
	if = {
		limit = { check_variable = { var = gods_of_africa_demand_remaining value = constant:gods_of_africa_value.zero compare = less_than_or_equals } }
		set_variable = { gods_of_africa_last_demand_status = constant:gods_of_africa_demand_status.resolved }
		gods_of_africa_clear_demand = yes
	}
	set_country_flag = gods_of_africa_outcome_report_pending
	country_event = { id = chaosx.nr12.604 }
}

gods_of_africa_resolve_substitute = {
	gods_of_africa_set_substitute_capacity_requirement = yes
	if = {
		limit = {
			gods_of_africa_demand_contract_is_current = yes
			has_country_flag = gods_of_africa_substitute_offered
			has_variable = gods_of_africa_substitute_family
			has_variable = gods_of_africa_substitute_amount
			gods_of_africa_substitute_offer_is_affordable = yes
		}
		# Preserve the immutable contract until the transfer has actually
		# succeeded. A participant can lose stockpile capacity between submitting
		# an offer and the host's acceptance decision, so a failed debit must
		# restore the original family and remaining amount instead of consuming
		# the negotiation state.
		set_variable = { gods_of_africa_substitute_original_family = gods_of_africa_demand_family }
		clear_variable = gods_of_africa_substitute_original_mobile_type
		if = {
			limit = { has_variable = gods_of_africa_demand_mobile_type check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.mobile_artillery } }
			set_variable = { gods_of_africa_substitute_original_mobile_type = gods_of_africa_demand_mobile_type }
		}
		set_variable = { gods_of_africa_substitute_original_remaining = gods_of_africa_demand_remaining }
		set_variable = { gods_of_africa_demand_family = gods_of_africa_substitute_family }
		clear_variable = gods_of_africa_demand_mobile_type
		if = {
			limit = { has_variable = gods_of_africa_substitute_mobile_type check_variable = { gods_of_africa_substitute_family = constant:gods_of_africa_demand_family.mobile_artillery } }
			set_variable = { gods_of_africa_demand_mobile_type = gods_of_africa_substitute_mobile_type }
		}
		set_variable = { gods_of_africa_demand_remaining = gods_of_africa_substitute_amount }
		set_country_flag = gods_of_africa_substitute_accepted
		gods_of_africa_resolve_full_compliance = yes
		if = {
			limit = {
				has_country_flag = gods_of_africa_demand_active
				check_variable = { gods_of_africa_payment_success > constant:gods_of_africa_value.zero }
			}
			# Part of the accepted substitute arrived. The substitute becomes the
			# terms of the contract and its remaining balance stays owed.
			set_country_flag = gods_of_africa_contract_substitute_terms
			clr_country_flag = gods_of_africa_substitute_offered
			clr_country_flag = gods_of_africa_negotiation_open
			clear_variable = gods_of_africa_substitute_family
			clear_variable = gods_of_africa_substitute_mobile_type
			clear_variable = gods_of_africa_substitute_amount
			clear_variable = gods_of_africa_substitute_capacity_requirement
			gods_of_africa_clear_substitute_slots = yes
			clr_country_flag = gods_of_africa_substitute_accepted
		}
		else_if = {
			limit = { has_country_flag = gods_of_africa_demand_active }
			# Nothing arrived. Restore the original contract and let the host
			# decide again.
			set_variable = { gods_of_africa_demand_family = gods_of_africa_substitute_original_family }
			clear_variable = gods_of_africa_demand_mobile_type
			if = {
				limit = { has_variable = gods_of_africa_substitute_original_mobile_type check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.mobile_artillery } }
				set_variable = { gods_of_africa_demand_mobile_type = gods_of_africa_substitute_original_mobile_type }
			}
			set_variable = { gods_of_africa_demand_remaining = gods_of_africa_substitute_original_remaining }
			set_variable = { gods_of_africa_payment_family = gods_of_africa_demand_family }
			gods_of_africa_bind_payment_mobile_type = yes
			set_variable = { gods_of_africa_payment_amount = gods_of_africa_demand_remaining }
			gods_of_africa_set_payment_capacity_requirement = yes
			clr_country_flag = gods_of_africa_substitute_accepted
		}
		else = {
			# Successful settlement already ran the idempotent demand cleanup.
			clr_country_flag = gods_of_africa_substitute_accepted
		}
		clear_variable = gods_of_africa_substitute_original_family
		clear_variable = gods_of_africa_substitute_original_mobile_type
		clear_variable = gods_of_africa_substitute_original_remaining
	}
}

# COUNTRY scope on a participant. Wrath is converted into a bounded tier,
# first refusal is capped before any side effect, and the host Strength band
# provides a second ceiling against disproportionate punishment.
gods_of_africa_evaluate_punishment = {
	if = {
		limit = {
			gods_of_africa_participant_host_is_active = yes
			gods_of_africa_participant_generation_is_current = yes
		}
		# Intent is computed from conduct and escalation evidence first. The
		# pending tier below is a separate capability result and is the only
		# value consumed by the punishment transaction.
		set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.pressure }
		# Conduct history raises intent even when Wrath has been partially
		# reduced. This keeps repeated failure and proven offenses meaningful while
		# the first-failure and host-Strength caps below still limit capability.
		if = {
			limit = {
				OR = {
					AND = {
						has_variable = gods_of_africa_demand_failures
						check_variable = { gods_of_africa_demand_failures > constant:gods_of_africa_value.zero }
					}
					AND = {
						has_variable = gods_of_africa_refusal_count
						check_variable = { gods_of_africa_refusal_count > constant:gods_of_africa_value.zero }
					}
				}
			}
			set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.disruption }
		}
		if = {
			limit = {
				OR = {
					AND = {
						has_variable = gods_of_africa_demand_failures
						check_variable = { var = gods_of_africa_demand_failures value = constant:gods_of_africa_value.repeated_conduct_threshold compare = greater_than_or_equals }
					}
					AND = {
						has_variable = gods_of_africa_refusal_count
						check_variable = { var = gods_of_africa_refusal_count value = constant:gods_of_africa_value.repeated_conduct_threshold compare = greater_than_or_equals }
					}
					AND = {
						has_variable = gods_of_africa_offense_count
						check_variable = { gods_of_africa_offense_count > constant:gods_of_africa_value.zero }
					}
				}
			}
			set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.crisis }
		}
		if = {
			limit = {
				has_variable = gods_of_africa_offense_count
				check_variable = { var = gods_of_africa_offense_count value = constant:gods_of_africa_value.repeated_conduct_threshold compare = greater_than_or_equals }
			}
			set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.ultimatum }
		}
		if = {
			limit = {
				has_variable = gods_of_africa_wrath
				check_variable = { gods_of_africa_wrath > constant:gods_of_africa_wrath.floor_war }
				check_variable = { var = gods_of_africa_punishment_intent_tier value = constant:gods_of_africa_punishment.disruption compare = less_than }
			}
			set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.disruption }
		}
		if = {
			limit = {
				has_variable = gods_of_africa_wrath
				check_variable = { gods_of_africa_wrath > constant:gods_of_africa_value.global_strength }
				check_variable = { var = gods_of_africa_punishment_intent_tier value = constant:gods_of_africa_punishment.crisis compare = less_than }
			}
			set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.crisis }
		}
		if = {
			limit = {
				has_variable = gods_of_africa_wrath
				check_variable = { gods_of_africa_wrath > constant:gods_of_africa_value.catastrophe_wrath_gate }
				check_variable = { var = gods_of_africa_punishment_intent_tier value = constant:gods_of_africa_punishment.ultimatum compare = less_than }
			}
			set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.ultimatum }
		}
		if = {
			limit = {
				has_event_target = africa_host
				event_target:africa_host = { check_variable = { gods_of_africa_doctrine = constant:gods_of_africa_doctrine.reciprocal_covenant } }
				check_variable = { var = gods_of_africa_punishment_intent_tier value = constant:gods_of_africa_punishment.pressure compare = greater_than }
			}
			subtract_from_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_doctrine_effect.reciprocal_punishment_tier_reduction }
		}
		else_if = {
			limit = {
				has_event_target = africa_host
				event_target:africa_host = { check_variable = { gods_of_africa_doctrine = constant:gods_of_africa_doctrine.sovereign_exaction } }
				has_country_flag = gods_of_africa_offender
				check_variable = { var = gods_of_africa_punishment_intent_tier value = constant:gods_of_africa_punishment.ultimatum compare = less_than }
			}
			add_to_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_doctrine_effect.sovereign_exaction_punishment_tier_increase }
		}
		if = { limit = { gods_of_africa_extreme_gate_is_open = yes } set_variable = { gods_of_africa_punishment_intent_tier = constant:gods_of_africa_punishment.catastrophe } }
		set_variable = { gods_of_africa_pending_punishment_tier = gods_of_africa_punishment_intent_tier }
		if = {
			limit = { gods_of_africa_first_failure_is_safe = yes check_variable = { gods_of_africa_pending_punishment_tier > constant:gods_of_africa_value.first_refusal_max_punishment_tier } }
			set_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_value.first_refusal_max_punishment_tier }
		}
		if = {
			limit = { event_target:africa_host = { check_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.limited } } check_variable = { gods_of_africa_pending_punishment_tier > constant:gods_of_africa_punishment.pressure } }
			set_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_punishment.pressure }
		}
		else_if = {
			limit = { event_target:africa_host = { check_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.regional } } check_variable = { gods_of_africa_pending_punishment_tier > constant:gods_of_africa_punishment.disruption } }
			set_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_punishment.disruption }
		}
		else_if = {
			limit = { event_target:africa_host = { check_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.continental } } check_variable = { gods_of_africa_pending_punishment_tier > constant:gods_of_africa_punishment.crisis } }
			set_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_punishment.crisis }
		}
		else_if = {
			limit = { event_target:africa_host = { check_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.global } } check_variable = { gods_of_africa_pending_punishment_tier > constant:gods_of_africa_punishment.ultimatum } }
			set_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_punishment.ultimatum }
		}
		else_if = {
			limit = {
				event_target:africa_host = { check_variable = { gods_of_africa_strength_band = constant:gods_of_africa_band.extreme } }
				check_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_punishment.catastrophe }
				NOT = { gods_of_africa_extreme_gate_is_open = yes }
			}
			set_variable = { gods_of_africa_pending_punishment_tier = constant:gods_of_africa_punishment.ultimatum }
		}
		gods_of_africa_apply_punishment = yes
	}
}

# agreement only when the contract carried agreed terms; otherwise it is

# caller, so this never scans the world or cancels the same contract twice.
gods_of_africa_cancel_demand_for_war = {
	if = {
		limit = {
			gods_of_africa_demand_contract_can_resolve = yes
			gods_of_africa_contract_ledger_can_close = yes
			has_war_with = event_target:africa_host
		}
		set_temp_variable = { gods_of_africa_wartime_contract_agreed = constant:gods_of_africa_value.zero }
		if = {
			limit = { gods_of_africa_contract_is_agreed = yes }
			set_temp_variable = { gods_of_africa_wartime_contract_agreed = constant:gods_of_africa_agreement.tribute_terms }
		}
		set_temp_variable = { gods_of_africa_contract_close_kind = constant:gods_of_africa_contract_close.war_breach }
		gods_of_africa_close_contract_ledger = yes
		set_variable = { gods_of_africa_last_demand_status = constant:gods_of_africa_demand_status.cancelled }
		clr_country_flag = gods_of_africa_wartime_breach_reported
		if = {
			limit = { check_variable = { gods_of_africa_wartime_contract_agreed = constant:gods_of_africa_agreement.tribute_terms } }
			set_temp_variable = { gods_of_africa_treaty_breach_kind = constant:gods_of_africa_agreement.tribute_terms }
			set_temp_variable = { gods_of_africa_treaty_breach_war = constant:gods_of_africa_value.one }
			gods_of_africa_record_treaty_breach = yes
			set_country_flag = gods_of_africa_wartime_breach_reported
		}
		else = {
			set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.war_cancellation }
			gods_of_africa_record_history = yes
			set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.war_magnitude }
			set_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_increase }
			gods_of_africa_apply_wrath = yes
		}
		set_country_flag = gods_of_africa_wartime_demand_report_pending
		gods_of_africa_clear_demand = yes
		country_event = { id = chaosx.nr12.613 }
	}
}

			has_variable = gods_of_africa_demand_deadline

	}

gods_of_africa_declare_defiance = {
	if = {
		limit = {
			gods_of_africa_participant_can_defy = yes
			gods_of_africa_contract_history_identity_is_current = yes
			OR = { NOT = { has_country_flag = gods_of_africa_demand_active } gods_of_africa_contract_ledger_can_close = yes }
		}
		set_country_flag = gods_of_africa_prior_defiance
		set_temp_variable = { gods_of_africa_defiance_breaks_terms = constant:gods_of_africa_value.zero }
		if = {
			limit = { gods_of_africa_contract_is_agreed = yes }
			set_temp_variable = { gods_of_africa_defiance_breaks_terms = constant:gods_of_africa_value.one }
		}
		set_temp_variable = { gods_of_africa_contract_close_kind = constant:gods_of_africa_contract_close.refusal }
		gods_of_africa_close_contract_ledger = yes
		# Defying agreed terms also breaks them; defiance keeps its own Wrath.
		if = {
			limit = { check_variable = { gods_of_africa_defiance_breaks_terms = constant:gods_of_africa_value.one } }
			set_temp_variable = { gods_of_africa_treaty_breach_kind = constant:gods_of_africa_agreement.tribute_terms }
			set_temp_variable = { gods_of_africa_treaty_breach_war = constant:gods_of_africa_value.zero }
			gods_of_africa_record_treaty_breach = yes
		}
		# Defiance replaces an accepted reconciliation; it is its own Enemy cause.
		clr_country_flag = gods_of_africa_reconciliation_agreement_live
		set_country_flag = gods_of_africa_defiant
		set_country_flag = gods_of_africa_permanent_opt_out
		gods_of_africa_withdraw_protection_package = yes
		clr_country_flag = gods_of_africa_punishment_applied
		set_variable = { gods_of_africa_defiance_day = global.num_days }
		set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.defiance_magnitude }
		set_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_increase }
		gods_of_africa_apply_wrath = yes
		gods_of_africa_evaluate_punishment = yes
		set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.defiance }
		gods_of_africa_record_history = yes
		if = {
			limit = { NOT = { has_country_flag = gods_of_africa_defiant_memory_applied } }
			set_country_flag = gods_of_africa_defiant_memory_applied
			save_event_target_as = gods_of_africa_memory_participant
			add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_defiant_offender }
			event_target:africa_host = { add_opinion_modifier = { target = event_target:gods_of_africa_memory_participant modifier = gods_of_africa_defiant_offender } }
		}
		if = {
			limit = { event_target:africa_host = { NOT = { has_country_flag = gods_of_africa_defiance_logged } } }
			event_target:africa_host = { set_country_flag = gods_of_africa_defiance_logged }
			set_variable = { gods_of_africa_milestone_payload = constant:africa_event_log.gods_defiance }
			set_variable = { gods_of_africa_milestone_actor = THIS.id }
			set_variable = { gods_of_africa_milestone_secondary_actor = event_target:africa_host.id }
			gods_of_africa_record_milestone = yes
		}
		if = { limit = { has_country_flag = gods_of_africa_demand_active } set_variable = { gods_of_africa_last_demand_status = constant:gods_of_africa_demand_status.failed } gods_of_africa_clear_demand = yes }
	}
}

gods_of_africa_start_reconciliation = {
	if = {
		limit = {
			gods_of_africa_participant_generation_is_current = yes
			gods_of_africa_participant_needs_reconciliation = yes
			# The window cannot be honoured during a war, so it opens after peace.
			NOT = { has_war = yes }
			NOT = { has_country_flag = gods_of_africa_reconciliation_pending }
			NOT = { has_country_flag = gods_of_africa_demand_active }
			OR = {
				NOT = { has_variable = gods_of_africa_reconciliation_next_attempt_day }
				check_variable = {
					var = global.num_days
					value = gods_of_africa_reconciliation_next_attempt_day
					compare = greater_than_or_equals
				}
			}
		}
		set_country_flag = gods_of_africa_reconciliation_pending
		clr_country_flag = gods_of_africa_reconciliation_failure_report_pending
		set_variable = { gods_of_africa_reconciliation_start_day = global.num_days }
		set_variable = { gods_of_africa_reconciliation_due_day = global.num_days }
		add_to_variable = { gods_of_africa_reconciliation_due_day = constant:gods_of_africa_timing.defiance_review_days }
		gods_of_africa_bind_reconciliation_epoch = yes
		activate_mission = gods_of_africa_reconciliation_mission
	}
}

gods_of_africa_complete_reconciliation = {
	if = {
		limit = {
			ROOT = { tag = event_target:africa_host gods_of_africa_active_host_is_valid = yes }
			gods_of_africa_participant_can_reconcile = yes
			gods_of_africa_contract_history_identity_is_current = yes
			has_country_flag = gods_of_africa_reparations_paid
			has_country_flag = gods_of_africa_reconciliation_offer_pending
			has_variable = gods_of_africa_reconciliation_offer_sequence
			check_variable = { gods_of_africa_reconciliation_offer_sequence > constant:gods_of_africa_value.zero }
			OR = {
				NOT = { has_variable = gods_of_africa_last_accepted_reconciliation_sequence }
				check_variable = { var = gods_of_africa_reconciliation_offer_sequence value = gods_of_africa_last_accepted_reconciliation_sequence compare = greater_than_or_equals }
			}
		}
		gods_of_africa_record_accepted_reconciliation = yes
		clr_country_flag = gods_of_africa_defiant
		clr_country_flag = gods_of_africa_permanent_opt_out
		# Resolve the outstanding hostility of the conduct record while the paid
		# offer is still on file. Prior war, occupation, defiance and breach
		# records stay permanent.
		gods_of_africa_resolve_accepted_grading_conduct = yes
		# This acceptance already proved peace and no held host-core territory.
		clr_country_flag = gods_of_africa_unresolved_major_offense
		gods_of_africa_clear_reconciliation_epoch = yes
		remove_mission = gods_of_africa_reconciliation_mission
		# Acceptance resolves an outstanding broken agreement and its Wrath floor.
		# The breach count and the prior-breach record remain permanent.
		clr_country_flag = gods_of_africa_unresolved_treaty_break
		clr_country_flag = gods_of_africa_broken_treaty
		# The accepted reconciliation is itself a live agreement from now on.
		set_country_flag = gods_of_africa_reconciliation_agreement_live
		clr_country_flag = gods_of_africa_reconciliation_pending
		clr_country_flag = gods_of_africa_reparations_paid
		clr_country_flag = gods_of_africa_reconciliation_offer_pending
		clr_country_flag = gods_of_africa_reconciliation_failure_report_pending
		clear_variable = gods_of_africa_reconciliation_next_attempt_day
		set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.pardon_magnitude }
		set_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_decrease }
		gods_of_africa_apply_wrath = yes
		set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.reconciliation }
		gods_of_africa_record_history = yes
		save_event_target_as = gods_of_africa_reconciled_participant
		add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_reconciliation }
		event_target:africa_host = { add_opinion_modifier = { target = event_target:gods_of_africa_reconciled_participant modifier = gods_of_africa_reconciliation } }
	}
}

gods_of_africa_resolve_participant_final_outcome = {
	if = {
		limit = { gods_of_africa_settlement_can_commit = yes }
		gods_of_africa_retire_open_contract = yes
		set_variable = { gods_of_africa_settlement_generation = event_target:africa_host.africa_host_generation }
		set_variable = { gods_of_africa_settlement_origin_host_id = event_target:africa_host.africa_origin_host_id }
		set_variable = { gods_of_africa_settlement_history_generation = gods_of_africa_participant_generation }
		set_variable = { gods_of_africa_settlement_strength = event_target:africa_host.gods_of_africa_strength }
		set_variable = { gods_of_africa_settlement_doctrine = event_target:africa_host.gods_of_africa_doctrine }
		if = {
			limit = { gods_of_africa_final_enemy_is_valid = yes }
			set_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.enemy }
		}
		else_if = {
			limit = { gods_of_africa_final_favored_is_valid = yes }
			set_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.favored }
		}
		else_if = {
			limit = { gods_of_africa_final_respected_is_valid = yes }
			set_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.respected }
		}
		else = { set_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.distrusted } }
		set_country_flag = gods_of_africa_final_outcome_committed
	}
}

# COUNTRY scope on a participant. Final settlement is a real, one-time
# transfer from the host when the reserve can support it. If the physical
# package is unavailable, the participant receives the smaller political-power
# equivalent instead of a fictional equipment grant.
gods_of_africa_grant_settlement_reward = {
	if = {
		limit = {
			gods_of_africa_participant_has_settlement_history = yes
			has_country_flag = gods_of_africa_final_outcome_committed
			event_target:africa_host = { has_country_flag = gods_of_africa_final_settlement_committed }
			gods_of_africa_final_settlement_value_is_valid = yes
			NOT = { has_country_flag = gods_of_africa_settlement_reward_granted }
			has_event_target = africa_host
		}
		set_country_flag = gods_of_africa_settlement_reward_granted
		save_event_target_as = gods_of_africa_settlement_participant
		if = {
			limit = {
				check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.favored }
				event_target:africa_host = {
					NOT = { has_equipment = { infantry_equipment < constant:gods_of_africa_transaction.settlement_favored_infantry_equipment } }
					NOT = { has_fuel < constant:gods_of_africa_transaction.settlement_favored_fuel }
					has_variable = africa_common_defence_stockpile
					check_variable = { var = africa_common_defence_stockpile value = constant:gods_of_africa_transaction.settlement_favored_infantry_equipment compare = greater_than_or_equals }
				}
			}
			event_target:africa_host = {
				send_equipment = { equipment = infantry_equipment amount = constant:gods_of_africa_transaction.settlement_favored_infantry_equipment target = event_target:gods_of_africa_settlement_participant }
				set_temp_variable = { fuel_stockpile_removal_amount = constant:gods_of_africa_transaction.settlement_favored_fuel }
				remove_fuel_from_stockpile = yes
				subtract_from_variable = { africa_common_defence_stockpile = constant:gods_of_africa_transaction.settlement_favored_infantry_equipment }
			}
			add_fuel = constant:gods_of_africa_transaction.settlement_favored_fuel
			add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_settlement_favor }
		}
		else_if = {
			limit = { check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.favored } }
			add_political_power = constant:gods_of_africa_transaction.settlement_favored_pp
			add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_settlement_favor }
		}
		else_if = {
			limit = {
				check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.respected }
				event_target:africa_host = {
					NOT = { has_equipment = { infantry_equipment < constant:gods_of_africa_transaction.settlement_respected_infantry_equipment } }
					NOT = { has_fuel < constant:gods_of_africa_transaction.settlement_respected_fuel }
					has_variable = africa_common_defence_stockpile
					check_variable = { var = africa_common_defence_stockpile value = constant:gods_of_africa_transaction.settlement_respected_infantry_equipment compare = greater_than_or_equals }
				}
			}
			event_target:africa_host = {
				send_equipment = { equipment = infantry_equipment amount = constant:gods_of_africa_transaction.settlement_respected_infantry_equipment target = event_target:gods_of_africa_settlement_participant }
				set_temp_variable = { fuel_stockpile_removal_amount = constant:gods_of_africa_transaction.settlement_respected_fuel }
				remove_fuel_from_stockpile = yes
				subtract_from_variable = { africa_common_defence_stockpile = constant:gods_of_africa_transaction.settlement_respected_infantry_equipment }
			}
			add_fuel = constant:gods_of_africa_transaction.settlement_respected_fuel
			add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_settlement_favor }
		}
		else_if = { limit = { check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.respected } } add_political_power = constant:gods_of_africa_transaction.settlement_respected_pp add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_settlement_favor } }
		else_if = { limit = { check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.distrusted } } add_political_power = constant:gods_of_africa_transaction.settlement_distrusted_pp }
		if = {
			limit = {
				OR = {
					check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.favored }
					check_variable = { gods_of_africa_final_settlement = constant:gods_of_africa_settlement.respected }
				}
			}
			event_target:africa_host = { add_opinion_modifier = { target = event_target:gods_of_africa_settlement_participant modifier = gods_of_africa_settlement_favor } }
		}
	}
}

	if = {
		limit = {
			OR = {
				has_capitulated = yes

			array = gods_of_africa_participants
			gods_of_africa_cancel_recovery_participant_preserving_receipts = yes
		}
		clear_array = gods_of_africa_participants
		clear_array = gods_of_africa_selected_participants

# Shared Event Log adapter. Callers provide temporary `gods_of_africa_milestone_*`
# values; routine demand receipts stay in the private relationship ledger and
# only the accepted milestone payloads enter the public Event 012 history.
gods_of_africa_record_milestone = {
	if = {
		limit = { has_variable = gods_of_africa_milestone_payload }
		set_temp_variable = { africa_event_log_payload = gods_of_africa_milestone_payload }
		set_temp_variable = { africa_event_log_actor = THIS.id }
		set_temp_variable = { africa_event_log_has_actor = constant:africa_value.one }
		set_temp_variable = { africa_event_log_secondary_actor = constant:africa_value.zero }
		set_temp_variable = { africa_event_log_has_secondary_actor = constant:africa_value.zero }
		if = {
			limit = { has_variable = gods_of_africa_milestone_actor }
			set_temp_variable = { africa_event_log_actor = gods_of_africa_milestone_actor }
		}
		if = {
			limit = { has_variable = gods_of_africa_milestone_secondary_actor }
			set_temp_variable = { africa_event_log_secondary_actor = gods_of_africa_milestone_secondary_actor }
			set_temp_variable = { africa_event_log_has_secondary_actor = constant:africa_value.one }
		}
		africa_record_event_log_world_order_entry = yes
		clear_variable = gods_of_africa_milestone_payload
		clear_variable = gods_of_africa_milestone_actor
		clear_variable = gods_of_africa_milestone_secondary_actor
	}
}

# COUNTRY scope on a selected participant. The shipment is an actual equipment
# transfer, limited to the maintained protection target and the host's reserve.
gods_of_africa_deliver_protection_transport = {
	if = {
		limit = {
			event_target:africa_host = {
				has_country_flag = gods_of_africa_focus_logistics_support_unlocked
			}
		}
		event_target:africa_host = {
			set_temp_variable = { gods_of_africa_protection_transport_minimum = constant:gods_of_africa_protection.transport_host_stockpile_minimum }
			add_to_temp_variable = { gods_of_africa_protection_transport_minimum = constant:gods_of_africa_transaction.transport_amount }
		}
		if = {
			limit = {
				has_equipment = { convoy_1 < constant:gods_of_africa_transaction.transport_amount }
				event_target:africa_host = {
					has_equipment = { convoy_1 > gods_of_africa_protection_transport_minimum }
				}
			}
			event_target:africa_host = {
				set_temp_variable = { equipment_stockpile_removal_amount = constant:gods_of_africa_transaction.transport_amount }
				remove_convoys_from_stockpile = yes
			}
			event_target:gods_of_africa_protected_participant = {
				add_equipment_to_stockpile = {
					type = convoy_1
					amount = constant:gods_of_africa_transaction.transport_amount
				}
				set_country_flag = gods_of_africa_protection_convoys_delivered
			}
		}
		else_if = {
			limit = {
				has_equipment = { train_equipment < constant:gods_of_africa_transaction.transport_amount }
				event_target:africa_host = {
					has_equipment = { train_equipment > gods_of_africa_protection_transport_minimum }
				}
			}
			event_target:africa_host = {
				set_temp_variable = { equipment_stockpile_removal_amount = constant:gods_of_africa_transaction.transport_amount }
				remove_trains_from_stockpile = yes
			}
			event_target:gods_of_africa_protected_participant = {
				add_equipment_to_stockpile = {
					type = train_equipment
					amount = constant:gods_of_africa_transaction.transport_amount
				}
				set_country_flag = gods_of_africa_protection_trains_delivered
			}
		}
	}
}

# COUNTRY scope on a selected participant. One capital infrastructure level is
# built only while the relief route is open and the participant controls its
# capital; the effect never scans or modifies unrelated states.
gods_of_africa_rebuild_protected_partner_capital = {
	if = {
		limit = {
			event_target:africa_host = {
				has_country_flag = gods_of_africa_focus_relief_unlocked
			}
			capital_scope = {
				is_owned_and_controlled_by = event_target:gods_of_africa_protected_participant
				infrastructure < constant:gods_of_africa_protection.reconstruction_infrastructure_cap
			}
		}
		capital_scope = {
			set_temp_variable = { gods_of_africa_reconstruction_levels = constant:gods_of_africa_protection.reconstruction_infrastructure_levels }
			add_building_construction = {
				type = infrastructure
				level = gods_of_africa_reconstruction_levels
				instant_build = yes
			}
		}
		set_country_flag = gods_of_africa_protection_capital_rebuilt
	}
}

# COUNTRY scope on a participant. This is the one voluntary aid route outside
# an active demand. The host need and the sender reserve are checked again at
# submission, then one real material shipment is debited and credited. The
# first offer has the larger relationship effect; later offers are still
# meaningful but use a smaller Wrath reduction.
gods_of_africa_offer_voluntary_aid = {
	if = {
		limit = {
			gods_of_africa_participant_can_offer_aid = yes
			has_event_target = africa_host
		}
		save_event_target_as = gods_of_africa_aid_participant
		set_variable = { gods_of_africa_aid_success = constant:gods_of_africa_value.zero }
		if = {
			limit = {
				event_target:africa_host = {
					check_variable = { gods_of_africa_need_land_equipment > constant:gods_of_africa_value.zero }
				}
				NOT = { has_equipment = { infantry_equipment < constant:gods_of_africa_transaction.voluntary_aid_equipment_capacity } }
			}
			set_variable = { gods_of_africa_aid_family = constant:gods_of_africa_demand_family.land_equipment }
			set_variable = { gods_of_africa_aid_amount = constant:gods_of_africa_transaction.voluntary_aid_equipment_amount }
			# A real shipment of the actual rifle variants, measured on both sides.
			set_temp_variable = { gods_of_africa_payment_receipt_kind = constant:gods_of_africa_contribution_kind.infantry }
			gods_of_africa_begin_voluntary_receipt = yes
			if = {
				limit = { has_country_flag = gods_of_africa_material_transfer_begin_ready }
				send_equipment = { equipment = infantry_equipment amount = gods_of_africa_aid_amount target = event_target:africa_host }
				gods_of_africa_complete_payment_receipt = yes
				if = {
					limit = { check_variable = { gods_of_africa_payment_delivered > constant:gods_of_africa_value.zero } }
					set_variable = { gods_of_africa_aid_amount = gods_of_africa_payment_delivered }
					event_target:africa_host = {
						add_to_variable = { africa_common_defence_stockpile = event_target:gods_of_africa_aid_participant.gods_of_africa_aid_amount }
					}
					set_variable = { gods_of_africa_aid_success = constant:gods_of_africa_value.one }
				}
			}
		}
		else_if = {
			limit = {
				event_target:africa_host = {
					check_variable = { gods_of_africa_need_fuel > constant:gods_of_africa_value.zero }
				}
				NOT = { has_fuel < constant:gods_of_africa_transaction.voluntary_aid_fuel_capacity }
			}
			set_variable = { gods_of_africa_aid_family = constant:gods_of_africa_demand_family.fuel }
			set_variable = { gods_of_africa_aid_amount = constant:gods_of_africa_transaction.voluntary_aid_fuel_amount }
			set_temp_variable = { gods_of_africa_payment_receipt_kind = constant:gods_of_africa_contribution_kind.fuel }
			gods_of_africa_begin_voluntary_receipt = yes
			if = {
				limit = { has_country_flag = gods_of_africa_material_transfer_begin_ready }
				set_temp_variable = { fuel_stockpile_removal_amount = gods_of_africa_aid_amount }
				remove_fuel_from_stockpile = yes
				event_target:africa_host = {
					add_fuel = event_target:gods_of_africa_aid_participant.gods_of_africa_aid_amount
				}
				gods_of_africa_complete_payment_receipt = yes
				if = {
					limit = { check_variable = { gods_of_africa_payment_delivered > constant:gods_of_africa_value.zero } }
					set_variable = { gods_of_africa_aid_amount = gods_of_africa_payment_delivered }
					set_variable = { gods_of_africa_aid_success = constant:gods_of_africa_value.one }
				}
			}
		}
		if = {
			limit = { check_variable = { gods_of_africa_aid_success > constant:gods_of_africa_value.zero } }
			if = {
				limit = { NOT = { has_variable = gods_of_africa_voluntary_aid_count } }
				set_variable = { gods_of_africa_voluntary_aid_count = constant:gods_of_africa_value.zero }
			}
			if = {
				limit = { check_variable = { gods_of_africa_voluntary_aid_count = constant:gods_of_africa_value.zero } }
				set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.voluntary_aid_magnitude }
			}
			else = {
				set_variable = { gods_of_africa_wrath_request = constant:gods_of_africa_wrath.voluntary_aid_repeat_magnitude }
			}
			add_to_variable = { gods_of_africa_voluntary_aid_count = constant:gods_of_africa_value.one }
			set_variable = { gods_of_africa_last_aid_family = gods_of_africa_aid_family }
			set_variable = { gods_of_africa_last_aid_amount = gods_of_africa_aid_amount }
			set_variable = { gods_of_africa_last_aid_day = global.num_days }
			set_variable = { gods_of_africa_next_aid_day = global.num_days }
			add_to_variable = { gods_of_africa_next_aid_day = constant:gods_of_africa_timing.voluntary_aid_cooldown_days }
			set_variable = { gods_of_africa_wrath_direction = constant:gods_of_africa_wrath.direction_decrease }
			gods_of_africa_apply_wrath = yes
			set_variable = { gods_of_africa_history_kind_request = constant:gods_of_africa_history_kind.aid }
			gods_of_africa_record_history = yes
			if = {
				limit = { NOT = { has_country_flag = gods_of_africa_reliable_memory_applied } }
				set_country_flag = gods_of_africa_reliable_memory_applied
				add_opinion_modifier = { target = event_target:africa_host modifier = gods_of_africa_reliable_partner }
				event_target:africa_host = {
					add_opinion_modifier = { target = event_target:gods_of_africa_aid_participant modifier = gods_of_africa_reliable_partner }
				}
			}
			event_target:africa_host = { gods_of_africa_build_need_profile = yes }
			set_country_flag = gods_of_africa_aid_report_pending
			country_event = { id = chaosx.nr12.614 }
		}
		clear_variable = gods_of_africa_aid_success
		clear_variable = gods_of_africa_aid_family
		clear_variable = gods_of_africa_aid_amount
	}
}
