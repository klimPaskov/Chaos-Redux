bio_sabotage_prepare_program_values_internal = {
	set_temp_variable = { bio_sabotage_weaponization_mult = constant:bio_sabotage_resolution.weaponization_operational }
	if = {
		limit = { cbrn_country_has_full_readiness = yes }
		set_temp_variable = { bio_sabotage_weaponization_mult = constant:bio_sabotage_resolution.weaponization_full }
	}
	else_if = {
		limit = { cbrn_country_has_integrated_readiness = yes }
		set_temp_variable = { bio_sabotage_weaponization_mult = constant:bio_sabotage_resolution.weaponization_integrated }
	}
	set_temp_variable = { bio_sabotage_concealment = cbrn_attribution_control }
	multiply_temp_variable = { bio_sabotage_concealment = constant:bio_sabotage_resolution.attribution_concealment_factor }
	clamp_temp_variable = {
		var = bio_sabotage_concealment
		min = constant:bio_lifecycle_math.zero
		max = constant:bio_sabotage_resolution.attribution_concealment_maximum
	}
}

bio_sabotage_record_failed_attempt_evidence_internal = {
	set_temp_variable = { bio_sabotage_attempt_lead_matches_actor = constant:bio_lifecycle_proof.missing }
	set_temp_variable = { bio_sabotage_attempt_evidence_points = constant:bio_sabotage_resolution.failed_attempt_evidence }
	event_target:bio_sabotage_actor_target = {
		if = {
			limit = { cbrn_designer_has_stable_weaponization = yes }
			multiply_temp_variable = { bio_sabotage_attempt_evidence_points = constant:cbrn_biosecurity_designer_effect.stable_weaponization_evidence_mult }
		}
		if = {
			limit = { cbrn_designer_has_strategic_dissemination_program = yes }
			multiply_temp_variable = { bio_sabotage_attempt_evidence_points = constant:cbrn_biosecurity_designer_effect.strategic_dissemination_evidence_mult }
		}
	}
	if = {
		limit = {
			has_variable = bio_sabotage_attempt_actor
			var:bio_sabotage_attempt_actor = { tag = event_target:bio_sabotage_actor_target }
		}
		if = {
			limit = { NOT = { has_variable = bio_sabotage_attempt_evidence } }
			set_variable = { bio_sabotage_attempt_evidence = constant:bio_lifecycle_math.zero }
		}
		add_to_variable = { bio_sabotage_attempt_evidence = bio_sabotage_attempt_evidence_points }
		set_temp_variable = { bio_sabotage_attempt_lead_matches_actor = constant:bio_lifecycle_proof.supplied }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = bio_sabotage_attempt_actor }
				NOT = { has_variable = bio_sabotage_attempt_evidence }
				check_variable = { var = bio_sabotage_attempt_evidence value = bio_sabotage_attempt_evidence_points compare = less_than }
			}
		}
		set_variable = { bio_sabotage_attempt_actor = event_target:bio_sabotage_actor_target }
		set_variable = { bio_sabotage_attempt_evidence = bio_sabotage_attempt_evidence_points }
		set_temp_variable = { bio_sabotage_attempt_lead_matches_actor = constant:bio_lifecycle_proof.supplied }
	}
	clamp_variable = {
		var = bio_sabotage_attempt_evidence
		min = constant:bio_lifecycle_math.zero
		max = constant:bio_lifecycle_math.hundred
	}
	set_state_flag = bio_sabotage_attempt_history
	if = {
		limit = { check_variable = { var = bio_sabotage_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals } }
		set_variable = { bio_sabotage_attempt_agent = bio_sabotage_agent }
		set_variable = { bio_sabotage_attempt_date = global.num_days }
		set_variable = { bio_sabotage_attempt_victim = event_target:bio_sabotage_victim_target }
	}
	set_temp_variable = { bio_active_agent = bio_sabotage_agent }
	meta_effect = {
		text = { set_state_flag = bio_agent_[BIO_AGENT]_sabotage_attempt_history }
		BIO_AGENT = "[?bio_active_agent|.0]"
	}
}

bio_sabotage_apply_failed_attempt_condemnation_internal = {
	set_temp_variable = { bio_sabotage_attempt_condemnation_mult = constant:bio_lifecycle_math.one }
	set_temp_variable = { condemnation_source_type = constant:condemnation_source.biological }
	set_temp_variable = { condemnation_base_gain = constant:bio_sabotage_resolution.failed_attempt_condemnation }
	set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.hidden }
	if = {
		limit = {
			check_variable = { var = bio_sabotage_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals }
			event_target:bio_sabotage_state_target = {
				check_variable = { var = bio_sabotage_attempt_evidence value = constant:bio_lifecycle_evidence.confirmed_threshold compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.public }
	}
	else_if = {
		limit = {
			check_variable = { var = bio_sabotage_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals }
			event_target:bio_sabotage_state_target = {
				check_variable = { var = bio_sabotage_attempt_evidence value = constant:bio_lifecycle_evidence.probable_threshold compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.discovered }
	}
	else_if = {
		limit = {
			check_variable = { var = bio_sabotage_attempt_lead_matches_actor value = constant:bio_lifecycle_proof.supplied compare = equals }
			event_target:bio_sabotage_state_target = {
				check_variable = { var = bio_sabotage_attempt_evidence value = constant:bio_lifecycle_evidence.suspicious_threshold compare = greater_than_or_equals }
			}
		}
		set_temp_variable = { condemnation_visibility_type = constant:condemnation_visibility.suspected }
	}
	if = {
		limit = {
			NOT = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.hidden compare = equals } }
			has_idea = cbrn_terminal_hazard_doctrine_spirit
		}
		set_temp_variable = { bio_sabotage_attempt_condemnation_mult = constant:bio_lifecycle_doctrine.terminal_hazard.condemnation_mult }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { var = condemnation_visibility_type value = constant:condemnation_visibility.hidden compare = equals } }
			has_idea = cbrn_theater_contamination_doctrine_spirit
		}
		set_temp_variable = { bio_sabotage_attempt_condemnation_mult = constant:bio_lifecycle_doctrine.theater_contamination.condemnation_mult }
	}
	if = {
		limit = { cbrn_designer_has_strategic_dissemination_program = yes }
		multiply_temp_variable = { bio_sabotage_attempt_condemnation_mult = constant:cbrn_biosecurity_designer_effect.strategic_dissemination_condemnation_mult }
	}
	multiply_temp_variable = { condemnation_base_gain = bio_sabotage_attempt_condemnation_mult }
	set_temp_variable = { condemnation_severity_type = constant:condemnation_severity.major }
	set_temp_variable = { condemnation_civilian_deaths = constant:bio_lifecycle_math.zero }
	set_temp_variable = { condemnation_contamination_value = constant:bio_lifecycle_math.zero }
	set_temp_variable = { condemnation_context_type = constant:condemnation_context.biological_strike }
	set_temp_variable = { condemnation_source_is_no_release_attempt = constant:condemnation_tier.concern }
	set_temp_variable = { condemnation_victim_country = event_target:bio_sabotage_victim_target }
	condemnation_add_source = yes
}
