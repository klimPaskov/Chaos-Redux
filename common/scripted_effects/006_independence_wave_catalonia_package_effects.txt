independence_wave_refresh_cat_compact_lifecycle = {
	if = {
		limit = { is_independence_wave_cat_package = yes }
		remove_ideas = { cat_contested_assembly cat_industrial_compact }
		if = {
			limit = { has_stable_independence_wave_cat_compact = yes }
			add_ideas = cat_industrial_compact
			set_country_flag = independence_wave_cat_compact_stabilized
		}
		else = {
			add_ideas = cat_contested_assembly
			clr_country_flag = independence_wave_cat_compact_stabilized
		}
	}
}

# Inputs: temporary `independence_wave_cat_industrial_delta` and
# `independence_wave_cat_assembly_delta`.
independence_wave_change_cat_compact_values = {
	if = {
		limit = { is_independence_wave_cat_package = yes }
		add_to_variable = { independence_wave_cat_industrial_cohesion = independence_wave_cat_industrial_delta }
		add_to_variable = { independence_wave_cat_assembly_legitimacy = independence_wave_cat_assembly_delta }
		clamp_variable = {
			var = independence_wave_cat_industrial_cohesion
			min = constant:independence_wave_catalonia_pressure.minimum
			max = constant:independence_wave_catalonia_pressure.maximum
		}
		clamp_variable = {
			var = independence_wave_cat_assembly_legitimacy
			min = constant:independence_wave_catalonia_pressure.minimum
			max = constant:independence_wave_catalonia_pressure.maximum
		}
		independence_wave_refresh_cat_compact_lifecycle = yes
	}
}
