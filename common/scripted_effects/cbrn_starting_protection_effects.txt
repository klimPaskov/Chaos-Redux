# ============================================================================
# CHAOS REDUX - 1936 CBRN PROTECTIVE-EQUIPMENT STARTING PROFILES
# ============================================================================
# Overview:
# - Applies the accepted gas-mask matrix as tuned 1936 crate totals.
# - Converts each country's military target into issue against its actual
#   deployed manpower and each civilian target into population-scaled state
#   distribution against actual controlled core states.
# - Keeps technology, warehouse stock, military issue, civilian distribution,
#   registration, filter condition, and annual maintenance separate.
# ============================================================================

cbrn_apply_starting_mask_profile = {
	cbrn_initialize_country_data = yes
	set_technology = {
		basic_gas_masks = 1
		popup = no
	}
	if = {
		limit = { check_variable = { cbrn_starting_mask_improved > constant:cbrn_math.zero } }
		set_technology = {
			improved_gas_masks = 1
			popup = no
		}
	}
	chaosx_startup_sync_chemical_technology_unlocks = yes

	if = {
		limit = { check_variable = { cbrn_starting_mask_basic > constant:cbrn_math.zero } }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_1
			amount = cbrn_starting_mask_basic
		}
	}
	if = {
		limit = { check_variable = { cbrn_starting_mask_improved > constant:cbrn_math.zero } }
		add_equipment_to_stockpile = {
			type = gas_mask_equipment_2
			amount = cbrn_starting_mask_improved
		}
	}

	set_variable = { cbrn_protection_program_profile = cbrn_starting_program_profile }
	set_country_flag = cbrn_program_established
	set_country_flag = cbrn_national_respirator_reserve
	if = {
		limit = {
			check_variable = {
				var = cbrn_starting_registration_proof
				value = constant:cbrn_proof.supplied
				compare = equals
			}
		}
		set_country_flag = cbrn_population_registered_and_fitted
		set_country_flag = cbrn_civil_defence_established
	}

	set_temp_variable = { cbrn_readiness_delta = constant:cbrn_protection_program_effect.reserve_readiness_gain }
	if = {
		limit = {
			check_variable = {
				var = cbrn_starting_registration_proof
				value = constant:cbrn_proof.supplied
				compare = equals
			}
		}
		add_to_temp_variable = { cbrn_readiness_delta = constant:cbrn_protection_program_effect.registration_readiness_gain }
	}
	cbrn_modify_chemical_readiness = yes

	cbrn_refresh_country_mask_snapshot = yes
	set_temp_variable = { cbrn_mask_issue_requested_crates = cbrn_military_mask_requirement }
	multiply_temp_variable = { cbrn_mask_issue_requested_crates = cbrn_starting_military_issue_target }
	subtract_from_temp_variable = { cbrn_mask_issue_requested_crates = cbrn_military_mask_crates }
	clamp_temp_variable = { var = cbrn_mask_issue_requested_crates min = constant:cbrn_math.zero }
	cbrn_issue_requested_masks_to_military = yes

	if = {
		limit = { check_variable = { cbrn_starting_civilian_distribution_target > constant:cbrn_math.zero } }
		set_temp_variable = { cbrn_distribution_target_fraction = cbrn_starting_civilian_distribution_target }
		set_temp_variable = { cbrn_distribution_base_cost_mult = constant:cbrn_math.one }
		set_temp_variable = { cbrn_distribution_effectiveness_mult = constant:cbrn_math.one }
		set_temp_variable = { cbrn_distribution_new_filter_condition = constant:cbrn_mask_filters.condition_full }
		every_owned_state = {
			limit = {
				is_controlled_by = PREV
				is_core_of = PREV
				state_population_k > constant:cbrn_math.zero
			}
			cbrn_distribute_requested_masks_to_state = yes
		}
	}
	cbrn_refresh_country_mask_snapshot = yes
	cbrn_start_protection_maintenance_job = yes
}
