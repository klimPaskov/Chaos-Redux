# ============================================================================
# CHAOS REDUX - EVENT 016 NEUTRAL CUSTOM TECHNOLOGY API
# Public API ownership is country scoped.
# ============================================================================
# Public country-scope effects expose only the eighteen existing Event 016
# custom technology IDs through selector variables. They never grant vanilla
# technology, Event 016 project history, project facilities, opening units, or
# stockpiles. External grant flags are a durable knowledge ledger: the Event
# 016 runtime rebuild preserves them, recreates the matching capped locked
# template, reopens matching custom-equipment production, and registers the
# neutral Event 019 provider row.
#
# Public inputs:
#   chaosx_grant_custom_operational_technology:
#       chaosx_custom_technology_family = constant:chaosx_custom_technology_family.*
#   chaosx_grant_custom_technology_upgrade:
#       chaosx_custom_technology_upgrade = constant:chaosx_custom_technology_upgrade.*
#   all public grants (optional):
#       chaosx_custom_technology_source = numeric source ID in [1, stride)
#   chaosx_grant_random_custom_operational_technology:
#       no selector; chooses one unresearched operational family.
#   chaosx_reconcile_custom_technology_runtime:
#       no selector; reconciles public control receipts and rebuilds runtime
#       consumers after external state migration or repair.
# Public outputs:
#   chaosx_custom_technology_grant_applied (temporary 0/1) for an operational
#   grant; chaosx_custom_technology_upgrade_applied (temporary 0/1) for an
#   upgrade grant; chaosx_custom_technology_random_grant_applied (temporary
#   0/1) for the random effect; chaosx_custom_technology_runtime_reconciled
#   (temporary 0/1) for explicit runtime reconciliation. In-range sources also
#   append idempotent encoded receipts to the country-scoped operational and
#   upgrade provenance arrays. Invalid selectors are safe no-ops.
# ============================================================================

# COUNTRY scope. Internal core used by the public operational and upgrade
# effects and by the runtime reapply helper. It deliberately does not rebuild
# templates or providers so a compound upgrade grant rebuilds exactly once.
chaosx_grant_custom_operational_technology_core = {
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.portal } }
		if = { limit = { NOT = { has_tech = brilliant_scientist_portal_warfare_tech } } set_technology = { brilliant_scientist_portal_warfare_tech = 1 popup = no } }
		set_country_flag = chaosx_custom_technology_portal_operational
		set_country_flag = chaosx_custom_technology_portal_warfare_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.clone } }
		if = {
			limit = { is_mengele_clone_directorate_country = yes }
			clone_select_mengele_refinement = yes
		}
		else = { clone_select_kruger_refinement = yes }
		set_country_flag = chaosx_custom_technology_clone_operational
		set_country_flag = chaosx_custom_technology_clone_formations_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.robot } }
		if = { limit = { NOT = { has_tech = brilliant_scientist_robot_formations_tech } } set_technology = { brilliant_scientist_robot_formations_tech = 1 popup = no } }
		set_country_flag = chaosx_custom_technology_robot_operational
		set_country_flag = chaosx_custom_technology_robot_formations_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.paleogenetic } }
		if = { limit = { NOT = { has_tech = brilliant_scientist_paleogenetic_formations_tech } } set_technology = { brilliant_scientist_paleogenetic_formations_tech = 1 popup = no } }
		set_country_flag = chaosx_custom_technology_paleogenetic_operational
		set_country_flag = chaosx_custom_technology_paleogenetic_formations_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.xenobiological } }
		if = { limit = { NOT = { has_tech = brilliant_scientist_xenobiological_formations_tech } } set_technology = { brilliant_scientist_xenobiological_formations_tech = 1 popup = no } }
		set_country_flag = chaosx_custom_technology_xenobiological_operational
		set_country_flag = chaosx_custom_technology_xenobiological_formations_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.alien_infantry } }
		if = { limit = { NOT = { has_tech = brilliant_scientist_alien_infantry_tech } } set_technology = { brilliant_scientist_alien_infantry_tech = 1 popup = no } }
		set_country_flag = chaosx_custom_technology_alien_infantry_operational
		set_country_flag = chaosx_custom_technology_alien_infantry_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
	if = {
		limit = { check_variable = { chaosx_custom_technology_family = constant:chaosx_custom_technology_family.temporal } }
		if = { limit = { NOT = { has_tech = brilliant_scientist_temporal_guard_tech } } set_technology = { brilliant_scientist_temporal_guard_tech = 1 popup = no } }
		set_country_flag = chaosx_custom_technology_temporal_operational
		set_country_flag = chaosx_custom_technology_temporal_guard_granted
		set_temp_variable = { chaosx_custom_technology_grant_applied = 1 }
	}
}

# COUNTRY scope. Private provenance writers used only by the public grant
# boundaries. A caller may provide a numeric chaosx_custom_technology_source
# from provenance_minimum_source (inclusive) to provenance_stride (exclusive)
# as a temporary or country variable. The
# receipt is selector * provenance_stride + source and is appended once to the
# corresponding country array; these arrays are never cleared by reconciliation
# or runtime rebuilds.
chaosx_record_custom_operational_technology_provenance = {
	if = {
		limit = {
			chaosx_custom_technology_source_is_valid = yes
			chaosx_custom_technology_family_is_valid = yes
		}
		set_temp_variable = { chaosx_custom_technology_provenance_receipt = chaosx_custom_technology_family }
		multiply_temp_variable = { chaosx_custom_technology_provenance_receipt = constant:chaosx_custom_technology_tuning.provenance_stride }
		add_to_temp_variable = { chaosx_custom_technology_provenance_receipt = chaosx_custom_technology_source }
		if = {
			limit = { NOT = { is_in_array = { array = chaosx_custom_technology_operational_provenance value = chaosx_custom_technology_provenance_receipt } } }
			add_to_array = { array = chaosx_custom_technology_operational_provenance value = chaosx_custom_technology_provenance_receipt }
		}
	}
}

chaosx_record_custom_technology_upgrade_provenance = {
	if = {
		limit = {
			chaosx_custom_technology_source_is_valid = yes
			chaosx_custom_technology_upgrade_is_valid = yes
		}
		set_temp_variable = { chaosx_custom_technology_provenance_receipt = chaosx_custom_technology_upgrade }
		multiply_temp_variable = { chaosx_custom_technology_provenance_receipt = constant:chaosx_custom_technology_tuning.provenance_stride }
		add_to_temp_variable = { chaosx_custom_technology_provenance_receipt = chaosx_custom_technology_source }
		if = {
			limit = { NOT = { is_in_array = { array = chaosx_custom_technology_upgrade_provenance value = chaosx_custom_technology_provenance_receipt } } }
			add_to_array = { array = chaosx_custom_technology_upgrade_provenance value = chaosx_custom_technology_provenance_receipt }
		}
	}
}

# COUNTRY scope. Reconciles the four xenobiological control channels before a
# public grant or runtime rebuild. Existing Event 016 control flags are the
# authoritative source when present. Otherwise the neutral API receipt or a
# learned control technology is used. Chemical, neural, machine, researched is
# the deterministic repair order when more than one source is stale. Only the
# canonical control survives. The chosen learned path is retained by its
# neutral external receipt when it is already visible to this API. This effect
# never creates project history, facilities, Directorate values, or achievements.
chaosx_reconcile_custom_xeno_control_grants = {
	set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.none }
	set_temp_variable = { chaosx_custom_technology_xeno_control_reconciled = 0 }

	# Existing project control modes take precedence over external receipts.
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			has_country_flag = brilliant_scientist_xeno_control_chemical
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.chemical }
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			has_country_flag = brilliant_scientist_xeno_control_neural
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.neural }
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			has_country_flag = brilliant_scientist_xeno_control_machine
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.machine }
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			has_country_flag = brilliant_scientist_xeno_control_researched
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.researched }
	}

	# If no project mode exists, repair the external/API ledger or a learned
	# hidden technology in the same deterministic order.
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_chemical
				has_country_flag = chaosx_custom_technology_xeno_chemical_control_granted
				has_tech = brilliant_scientist_xeno_chemical_control_tech
			}
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.chemical }
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_neural
				has_country_flag = chaosx_custom_technology_xeno_neural_control_granted
				has_tech = brilliant_scientist_xeno_neural_control_tech
			}
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.neural }
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_machine
				has_country_flag = chaosx_custom_technology_xeno_machine_control_granted
				has_tech = brilliant_scientist_xeno_machine_control_tech
			}
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.machine }
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.none compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_researched
				has_country_flag = chaosx_custom_technology_xeno_researched_control_granted
				has_tech = brilliant_scientist_xeno_researched_control_tech
			}
		}
		set_temp_variable = { chaosx_custom_technology_xeno_control_choice = constant:chaosx_custom_technology_control.researched }
	}

	# A selected learned/API path receives both neutral receipts so the normal
	# runtime clear cannot revoke it when a source event or project is gone.
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.chemical compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_chemical
				has_country_flag = chaosx_custom_technology_xeno_chemical_control_granted
				has_tech = brilliant_scientist_xeno_chemical_control_tech
			}
		}
		set_country_flag = chaosx_custom_technology_xeno_control_chemical
		set_country_flag = chaosx_custom_technology_xeno_chemical_control_granted
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.neural compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_neural
				has_country_flag = chaosx_custom_technology_xeno_neural_control_granted
				has_tech = brilliant_scientist_xeno_neural_control_tech
			}
		}
		set_country_flag = chaosx_custom_technology_xeno_control_neural
		set_country_flag = chaosx_custom_technology_xeno_neural_control_granted
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.machine compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_machine
				has_country_flag = chaosx_custom_technology_xeno_machine_control_granted
				has_tech = brilliant_scientist_xeno_machine_control_tech
			}
		}
		set_country_flag = chaosx_custom_technology_xeno_control_machine
		set_country_flag = chaosx_custom_technology_xeno_machine_control_granted
	}
	if = {
		limit = {
			check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.researched compare = equals }
			OR = {
				has_country_flag = chaosx_custom_technology_xeno_control_researched
				has_country_flag = chaosx_custom_technology_xeno_researched_control_granted
				has_tech = brilliant_scientist_xeno_researched_control_tech
			}
		}
		set_country_flag = chaosx_custom_technology_xeno_control_researched
		set_country_flag = chaosx_custom_technology_xeno_researched_control_granted
	}

	# Remove every alternate control channel while preserving the canonical one.
	if = {
		limit = { check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.chemical compare = equals } }
		chaosx_clear_custom_xeno_control_neural = yes
		chaosx_clear_custom_xeno_control_machine = yes
		chaosx_clear_custom_xeno_control_researched = yes
		set_temp_variable = { chaosx_custom_technology_xeno_control_reconciled = 1 }
	}
	if = {
		limit = { check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.neural compare = equals } }
		chaosx_clear_custom_xeno_control_chemical = yes
		chaosx_clear_custom_xeno_control_machine = yes
		chaosx_clear_custom_xeno_control_researched = yes
		set_temp_variable = { chaosx_custom_technology_xeno_control_reconciled = 1 }
	}
	if = {
		limit = { check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.machine compare = equals } }
		chaosx_clear_custom_xeno_control_chemical = yes
		chaosx_clear_custom_xeno_control_neural = yes
		chaosx_clear_custom_xeno_control_researched = yes
		set_temp_variable = { chaosx_custom_technology_xeno_control_reconciled = 1 }
	}
	if = {
		limit = { check_variable = { var = chaosx_custom_technology_xeno_control_choice value = constant:chaosx_custom_technology_control.researched compare = equals } }
		chaosx_clear_custom_xeno_control_chemical = yes
		chaosx_clear_custom_xeno_control_neural = yes
		chaosx_clear_custom_xeno_control_machine = yes
		set_temp_variable = { chaosx_custom_technology_xeno_control_reconciled = 1 }
	}
}

# COUNTRY scope. Public reconstruction boundary for external callers. It
# canonicalizes control state, then invokes the existing full Event 016 runtime
# rebuild so templates, production gates, provider rows, caps, and learned
# custom technologies are synchronized. The rebuild's normal clear keeps the
# chosen external ledger entry and never invents project history.
chaosx_reconcile_custom_technology_runtime = {
	chaosx_reconcile_custom_xeno_control_grants = yes
	brilliant_scientist_rebuild_project_force_runtime_package = yes
	set_temp_variable = { chaosx_custom_technology_runtime_reconciled = 1 }
}

# COUNTRY scope. Grants one operational family and immediately reconciles the
# existing Event 016 runtime consumers. No project-history flags are created.
chaosx_grant_custom_operational_technology = {
	set_temp_variable = { chaosx_custom_technology_grant_applied = 0 }
	if = {
		limit = { chaosx_can_grant_custom_operational_technology = yes }
		chaosx_grant_custom_operational_technology_core = yes
	}
	if = {
		limit = {
			check_variable = {
				var = chaosx_custom_technology_grant_applied
				value = 1
				compare = greater_than_or_equals
			}
		}
		chaosx_record_custom_operational_technology_provenance = yes
		chaosx_reconcile_custom_technology_runtime = yes
	}
}
