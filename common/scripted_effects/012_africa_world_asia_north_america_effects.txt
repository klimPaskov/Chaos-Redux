# ============================================================================
# CHAOS REDUX - EVENT 012 WORLD PACKAGE W2 ASIA AND NORTH AMERICA EFFECTS
# ============================================================================
# Country-scope helpers for the second-wave Asia and North America package
# loops. These effects keep route-specific gains and lifecycle proofs in one
# place while leaving the shared world-order effects, constants, triggers, and
# decision category owned by the parent implementation.
#
# The parent implementation provides the global crisis-result constants
# (africa_world_package_crisis_result.none/success/compromise/failure) and the
# africa_world_package_ratification_is_proven trigger. No high-chaos route or
# implementation-readiness flag is set here.
# ============================================================================

# COUNTRY scope. Named decision: seat the regional centres by lifting the
# weakest represented centre, rather than silently replacing any centre.
africa_asia_seat_regional_centres = {
	set_country_flag = africa_asia_centres_seated
	set_country_flag = africa_asia_regional_centres_decision_used
	set_variable = { africa_world_package_phase = constant:africa_world_package_phase.constituent_bargaining }
	if = {
		limit = {
			check_variable = { var = africa_asia_eastern_centre value = africa_asia_southern_centre compare = less_than }
			check_variable = { var = africa_asia_eastern_centre value = africa_asia_inland_centre compare = less_than }
			check_variable = { var = africa_asia_eastern_centre value = africa_asia_archipelago_centre compare = less_than }
		}
		add_to_variable = { africa_asia_eastern_centre = constant:africa_world_order.package_major_gain }
	}
	else_if = {
		limit = {
			check_variable = { var = africa_asia_southern_centre value = africa_asia_eastern_centre compare = less_than }
			check_variable = { var = africa_asia_southern_centre value = africa_asia_inland_centre compare = less_than }
			check_variable = { var = africa_asia_southern_centre value = africa_asia_archipelago_centre compare = less_than }
		}
		add_to_variable = { africa_asia_southern_centre = constant:africa_world_order.package_major_gain }
	}
	else_if = {
		limit = {
			check_variable = { var = africa_asia_inland_centre value = africa_asia_eastern_centre compare = less_than }
			check_variable = { var = africa_asia_inland_centre value = africa_asia_southern_centre compare = less_than }
			check_variable = { var = africa_asia_inland_centre value = africa_asia_archipelago_centre compare = less_than }
		}
		add_to_variable = { africa_asia_inland_centre = constant:africa_world_order.package_major_gain }
	}
	else = {
		add_to_variable = { africa_asia_archipelago_centre = constant:africa_world_order.package_major_gain }
	}
	add_to_variable = { africa_world_package_authority = constant:africa_world_order.package_minor_gain }
	add_to_variable = { africa_world_package_legitimacy = constant:africa_world_order.package_minor_gain }
	africa_world_clamp_package_mechanic = yes
}

# COUNTRY scope. Open the Asia shared-lane ratification event once all three
# grounded lane proofs are present. This also covers named decisions completed
# after a focus lane.
africa_asia_maybe_open_shared_lane_ratification = {
	if = {
		limit = {
			has_country_flag = africa_asia_food_lane_proof
			has_country_flag = africa_asia_rail_lane_proof
			has_country_flag = africa_asia_defence_autonomy_law_ready
			OR = {
				has_country_flag = africa_asia_founding_crisis_success
				has_country_flag = africa_asia_founding_crisis_compromise
			}
			NOT = { has_country_flag = africa_asia_shared_lane_event_fired }
		}
		set_country_flag = africa_asia_shared_lane_event_fired
		set_country_flag = africa_asia_shared_lane_ratification_pending
		country_event = { id = africa_world_package.305 }
	}
}
