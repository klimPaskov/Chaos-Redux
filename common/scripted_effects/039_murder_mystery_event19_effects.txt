# ============================================================================
# CHAOS REDUX - EVENT 039 EVENT 019 ASSASSIN FORCES PROVIDER
# ============================================================================
# Overview:
# - Provider 524 is the complete Event 019 owner adapter for Assassin Forces.
# - The adapter publishes one custom operations-kit obligation profile and a
#   locked baseline template of three Assassin Cadres plus one Saboteur Cell.
# - Shadow, Guard, Master, and Mechanized profiles remain Event 039 progression
#   formations and never leak into Event 019's standalone provider package.
# - Every callback is country-scoped and fails closed outside a validated
#   Assassin State or derivative country.
# ============================================================================

chaos_unit_family_provider_524_event19_evaluate_eligibility = {
	set_temp_variable = { chaos_unit_family_candidate_eligible = 0 }
	set_temp_variable = { chaos_unit_family_candidate_native = 0 }
	if = {
		limit = {
			OR = {
				has_country_flag = murder_mystery_assassin_state
				has_country_flag = murder_mystery_derivative_country
			}
			NOT = { is_special_chaos_country = yes }
			NOT = { is_actual_nonhuman_country = yes }
			NOT = { has_global_flag = murder_mystery_cleanup_in_progress }
		}
		set_temp_variable = { chaos_unit_family_candidate_eligible = 1 }
		set_temp_variable = { chaos_unit_family_candidate_native = 1 }
	}
}

chaos_unit_family_provider_524_event19_get_equipment_token = {
	set_temp_variable = { infantry_spawn_family_provider_equipment_is_specialist = 0 }
	set_temp_variable = { infantry_spawn_family_provider_equipment_token = 0 }
	if = {
		limit = {
			check_variable = {
				var = infantry_spawn_exact_current_profile
				value = constant:chaos_unit_family_event39_assassin_forces.operations_kit_resource_profile
				compare = equals
			}
		}
		set_temp_variable = { var = infantry_spawn_family_provider_equipment_token value = token:assassin_operations_kit_1 }
	}
	else = {
		infantry_spawn_provider_get_standard_equipment_token = yes
	}
}

chaos_unit_family_provider_524_event19_evaluate_management = {
	set_temp_variable = { infantry_spawn_family_provider_can_train = 0 }
	set_temp_variable = { infantry_spawn_family_provider_can_spawn = 0 }
	set_temp_variable = { infantry_spawn_family_provider_can_sustain = 0 }
	set_temp_variable = { infantry_spawn_family_provider_uses_training = 0 }
	if = {
		limit = {
			OR = {
				has_country_flag = murder_mystery_assassin_state
				has_country_flag = murder_mystery_derivative_country
			}
			infantry_spawn_country_has_eligible_state = yes
			OR = {
				has_country_flag = murder_mystery_force_reinforcement_open
				has_tech = murder_mystery_clandestine_combat_training
			}
			check_variable = { var = num_equipment@assassin_operations_kit_1 value = constant:chaos_unit_family_event39_assassin_forces.minimum_operations_kit_stockpile compare = greater_than_or_equals }
		}
		set_temp_variable = { infantry_spawn_family_provider_can_train = 1 }
		set_temp_variable = { infantry_spawn_family_provider_can_spawn = 1 }
		set_temp_variable = { infantry_spawn_family_provider_uses_training = 1 }
	}
	if = {
		limit = {
			check_variable = { infantry_spawn_family_live_division_count > constant:chaos_unit_family_registry.first_index }
			OR = {
				has_country_flag = murder_mystery_assassin_state
				has_country_flag = murder_mystery_derivative_country
			}
		}
		set_temp_variable = { infantry_spawn_family_provider_can_sustain = 1 }
	}
}

chaos_unit_family_provider_524_event19_pay_management_action = {
	set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.train_or_spawn }
			check_variable = { infantry_spawn_family_provider_can_spawn > constant:chaos_unit_family_registry.first_index }
		}
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
	else_if = {
		limit = {
			check_variable = { infantry_spawn_family_management_action = constant:infantry_spawn_family_management_action.sustainment }
			check_variable = { infantry_spawn_family_provider_can_sustain > constant:chaos_unit_family_registry.first_index }
		}
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 1 }
	}
}

chaos_unit_family_provider_524_event19_refund_management_action = {
	if = {
		limit = { check_variable = { infantry_spawn_family_provider_payment_succeeded > constant:chaos_unit_family_registry.first_index } }
		set_temp_variable = { infantry_spawn_family_provider_payment_succeeded = 0 }
	}
}
