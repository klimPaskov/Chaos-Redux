famine_release_reserves_action_begin_work = {
	if = {
			limit = { famine_release_reserves_can_start_in_context = yes }
			set_temp_variable = { migration_transport_payment_result = constant:migration_transport_cost.zero }
			set_temp_variable = { migration_transport_fuel_land = constant:migration_transport_cost.standard_fuel_land } set_temp_variable = { migration_transport_fuel_sea = constant:migration_transport_cost.standard_fuel_sea } set_temp_variable = { migration_transport_convoys_sea = constant:migration_transport_cost.standard_convoys_sea } set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:famine_release_reserves } migration_transport_pay_state_cost = yes
			if = {
				limit = { check_variable = { migration_transport_payment_result = constant:migration_transport_cost.one } }
				add_equipment_to_stockpile = { type = support_equipment amount = constant:humanitarian_legacy_cost.support_light_debit }
				var:humanitarian_action_target = {
					set_state_flag = famine_release_reserves_cost_paid
					set_variable = { famine_release_reserves_cost_payer = ROOT }
				}
				set_country_flag = famine_decision_active
			}
		}
}

famine_release_reserves_action_resolve_work = {
	if = {
			limit = { famine_release_reserves_cost_receipt_is_valid_in_context = yes }
			var:humanitarian_action_target = {
				clr_state_flag = famine_release_reserves_cost_paid
				clear_variable = famine_release_reserves_cost_payer
			}
			var:humanitarian_action_target = {
				famine_initialize_state = yes
				set_temp_variable = { famine_food_reserve_release_amount = famine_food_reserve_daily_need }
				multiply_temp_variable = { famine_food_reserve_release_amount = constant:famine_mission_contract.reserve_release_days }
				set_temp_variable = { famine_food_reserve_release_request_proven = constant:humanitarian_runtime.one }
				set_temp_variable = { famine_food_reserve_release_actor_proven = constant:humanitarian_runtime.one }
				set_temp_variable = { famine_food_reserve_consume_result = constant:civilian_transfer_route_result.invalid }
				set_temp_variable = { famine_food_reserve_consumed_output = constant:famine_food_reserve.zero }
				set_temp_variable = { famine_food_reserve_relief_granted_output = constant:famine_food_reserve.zero }
				famine_release_reserves = yes
				if = {
					limit = { check_variable = { var = famine_food_reserve_consumed_output value = constant:famine_food_reserve.zero compare = equals } }
					ROOT = {
						set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:famine_release_reserves } migration_transport_refund_state_cost = yes
	set_temp_variable = { humanitarian_action_refunded = constant:humanitarian_runtime.one }
						add_equipment_to_stockpile = { type = support_equipment amount = constant:humanitarian_legacy_cost.support_light }
					}
				}
				if = {
					limit = {
						check_variable = { var = famine_food_reserve_consume_result value = constant:civilian_transfer_route_result.valid compare = equals }
						check_variable = { var = famine_food_reserve_relief_granted_output value = constant:famine_mission_contract.minimum_relief_grant compare = greater_than }
					}
					set_temp_variable = { famine_reserve_requisition_protection_days = constant:famine_decision_threshold.reserve_requisition_protection_days }
					set_state_flag = {
						flag = famine_food_reserve_requisition_protected
						value = 1
						days = famine_reserve_requisition_protection_days
					}
					if = {
						limit = {
							has_state_flag = famine_extraction_active
							NOT = { has_state_flag = famine_extraction_suspended }
							NOT = { has_state_flag = famine_food_stage_catastrophic }
							has_state_flag = humanitarian_achievement_extraction_shock_recorded
							OR = {
								has_state_flag = famine_food_stage_acute_shortage
								has_state_flag = famine_food_stage_famine
							}
							check_variable = { var = famine_component_extraction value = constant:famine_food_component.zero compare = greater_than }
							OR = {
								check_variable = { var = famine_component_production value = constant:famine_food_component.zero compare = greater_than }
								check_variable = { var = famine_component_transport value = constant:famine_food_component.zero compare = greater_than }
							}
							check_variable = { var = famine_extraction_generation value = constant:humanitarian_achievement_value.zero compare = greater_than }
						}
						set_state_flag = famine_extraction_suspended
						clr_state_flag = famine_extraction_active
						set_variable = { famine_input_extraction = constant:famine_food_component.zero }
						set_variable = { famine_component_extraction = constant:famine_food_component.zero }
						ROOT = {
							humanitarian_achievement_record_extraction_suspended = yes
							humanitarian_achievement_record_alternative_logistics = yes
						}
					}
					if = {
						limit = { has_state_flag = famine_mission_route_subject }
						set_state_flag = famine_mission_route_success_proven
					}
					if = {
						limit = { has_state_flag = famine_mission_relief_subject }
						set_state_flag = famine_mission_relief_success_proven
					}
					else_if = {
						limit = {
							NOT = { ROOT = { has_country_flag = famine_mission_relief_active } }
							ROOT = { famine_country_has_free_mission_slot = yes }
						}
						set_state_flag = famine_mission_relief_subject
						famine_set_mission_relief_deadline = yes
						ROOT = {
							famine_claim_mission_slot = yes
							set_country_flag = famine_mission_relief_active
						}
					}
					ROOT = {
						famine_show_report_relief_arrival = yes
						add_stability = constant:famine_decision_outcome.decision_minor_stability_negative
					}
				}
			}
		}
		else_if = {
			limit = {
				var:humanitarian_action_target = {
					has_state_flag = famine_release_reserves_cost_paid
					has_variable = famine_release_reserves_cost_payer
					var:famine_release_reserves_cost_payer = { tag = ROOT }
				}
			}
			var:humanitarian_action_target = {
				clr_state_flag = famine_release_reserves_cost_paid
				clear_variable = famine_release_reserves_cost_payer
			}
		}
		set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:famine_release_reserves } migration_transport_forget_state_cost = yes
}

famine_emergency_imports_action_begin_work = {
	if = {
			limit = { famine_emergency_imports_can_start_in_context = yes }
			var:humanitarian_action_target = {
				set_temp_variable = { famine_emergency_imports_contract_preexisting = constant:famine_relief_runtime.zero }
				if = {
					limit = {
						famine_relief_contract_is_valid = yes
						check_variable = { var = famine_relief_route_mode value = constant:famine_relief_route_mode.land compare = equals }
						var:famine_relief_actor = { tag = ROOT }
					}
					set_temp_variable = { famine_emergency_imports_contract_preexisting = constant:famine_relief_runtime.one }
				}
				set_temp_variable = { famine_emergency_imports_attempt_generation = constant:famine_relief_runtime.zero }
				if = {
					limit = { NOT = { famine_relief_contract_is_valid = yes } }
					famine_refresh_reserve_capacity = yes
					set_variable = { famine_relief_route_mode = constant:famine_relief_route_mode.land }
					set_variable = { famine_relief_requested_amount = famine_food_reserve_daily_need }
					multiply_variable = { famine_relief_requested_amount = constant:famine_mission_contract.reserve_import_days }
					round_variable = famine_relief_requested_amount
					famine_relief_select_and_create_contract = yes
				}
				if = {
					limit = {
						check_variable = { var = famine_emergency_imports_contract_preexisting value = constant:famine_relief_runtime.zero compare = equals }
						famine_relief_contract_is_valid = yes
						check_variable = { var = famine_relief_route_mode value = constant:famine_relief_route_mode.land compare = equals }
						var:famine_relief_actor = { tag = ROOT }
					}
					set_temp_variable = { famine_emergency_imports_attempt_generation = famine_relief_contract_generation }
				}
			}
			if = {
				limit = {
					famine_emergency_imports_cost_is_affordable_in_context = yes
					var:humanitarian_action_target = {
						famine_relief_contract_is_valid = yes
						check_variable = { var = famine_relief_route_mode value = constant:famine_relief_route_mode.land compare = equals }
						var:famine_relief_actor = { tag = ROOT }
					}
				}
				set_temp_variable = { migration_transport_payment_result = constant:migration_transport_cost.zero }
				set_temp_variable = { migration_transport_fuel_land = constant:migration_transport_cost.standard_fuel_land } set_temp_variable = { migration_transport_fuel_sea = constant:migration_transport_cost.standard_fuel_sea } set_temp_variable = { migration_transport_convoys_sea = constant:migration_transport_cost.standard_convoys_sea } set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:famine_emergency_imports } migration_transport_pay_state_cost = yes
				if = {
					limit = { check_variable = { migration_transport_payment_result = constant:migration_transport_cost.one } }
					var:humanitarian_action_target = {
						set_state_flag = famine_emergency_imports_cost_paid
						set_variable = { famine_emergency_imports_cost_payer = ROOT }
						set_variable = { famine_emergency_imports_cost_contract_generation = famine_relief_contract_generation }
						set_variable = { famine_emergency_imports_cost_donor_state = famine_relief_donor_state }
					}
				}
			}
			else = {
				var:humanitarian_action_target = {
					if = {
						limit = {
							check_variable = { var = famine_emergency_imports_contract_preexisting value = constant:famine_relief_runtime.zero compare = equals }
							famine_relief_contract_is_valid = yes
							check_variable = { var = famine_relief_route_mode value = constant:famine_relief_route_mode.land compare = equals }
							var:famine_relief_actor = { tag = ROOT }
							check_variable = { var = famine_relief_contract_generation value = famine_emergency_imports_attempt_generation compare = equals }
						}
						famine_relief_cleanup_contract = yes
					}
				}
			}
		}
}

famine_emergency_imports_action_resolve_work = {
	if = {
			limit = { famine_emergency_imports_cost_receipt_is_valid_in_context = yes }
			set_temp_variable = { famine_emergency_imports_cost_work_proven = constant:famine_relief_runtime.zero }
			if = {
				limit = { famine_emergency_imports_cost_contract_is_current_in_context = yes }
				set_temp_variable = { famine_emergency_imports_cost_work_proven = constant:famine_relief_runtime.one }
			}
			var:humanitarian_action_target = {
				clr_state_flag = famine_emergency_imports_cost_paid
				clear_variable = famine_emergency_imports_cost_payer
				clear_variable = famine_emergency_imports_cost_contract_generation
				clear_variable = famine_emergency_imports_cost_donor_state
			}
			if = {
				limit = { check_variable = { var = famine_emergency_imports_cost_work_proven value = constant:famine_relief_runtime.one compare = equals } }
				var:humanitarian_action_target = {
					set_temp_variable = { famine_relief_delivery_result = constant:famine_relief_result.invalid }
					famine_relief_deliver_contract = yes
					if = {
						limit = {
							OR = {
								check_variable = { var = famine_relief_delivery_result value = constant:famine_relief_result.delivered compare = equals }
								check_variable = { var = famine_relief_delivery_result value = constant:famine_relief_result.partial compare = equals }
							}
						}
						if = {
							limit = { has_state_flag = famine_mission_route_subject }
							set_state_flag = famine_mission_route_success_proven
						}
						if = {
							limit = { has_state_flag = famine_mission_relief_subject }
							set_state_flag = famine_mission_relief_success_proven
						}
						else_if = {
							limit = {
								NOT = { ROOT = { has_country_flag = famine_mission_relief_active } }
								ROOT = { famine_country_has_free_mission_slot = yes }
							}
							set_state_flag = famine_mission_relief_subject
							famine_set_mission_relief_deadline = yes
							ROOT = {
								famine_claim_mission_slot = yes
								set_country_flag = famine_mission_relief_active
							}
						}
						ROOT = {
							famine_show_report_relief_arrival = yes
							humanitarian_achievement_record_alternative_logistics = yes
						}
					}
				}
			}
			else = {
				# A contract broken by an enemy capture of its donor is relief obstruction.
				set_temp_variable = { famine_condemnation_result = constant:humanitarian_runtime.zero }
				var:humanitarian_action_target = { famine_condemn_relief_contract_interruption = yes }
			}
		}
		else_if = {
			limit = {
				var:humanitarian_action_target = {
					has_state_flag = famine_emergency_imports_cost_paid
					has_variable = famine_emergency_imports_cost_payer
					var:famine_emergency_imports_cost_payer = { tag = ROOT }
				}
			}
			var:humanitarian_action_target = {
				clr_state_flag = famine_emergency_imports_cost_paid
				clear_variable = famine_emergency_imports_cost_payer
				clear_variable = famine_emergency_imports_cost_contract_generation
				clear_variable = famine_emergency_imports_cost_donor_state
			}
		}
		set_temp_variable = { migration_transport_target = var:humanitarian_action_target } set_temp_variable = { migration_transport_receipt = token:famine_emergency_imports } migration_transport_forget_state_cost = yes
}
