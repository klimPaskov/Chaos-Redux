# CHAOS REDUX - EVENT 019 FORMATION GENERATION
# ============================================================================
# Builds weighted ordinary lots for the baseline, Organized Muster, and
# Arsenal Lottery stages. Every generated template receives an exact component

infantry_spawn_prepare_current_state_context = {
	set_temp_variable = { infantry_spawn_context_core = 0 }
	set_temp_variable = { infantry_spawn_context_capital = 0 }
	set_temp_variable = { infantry_spawn_context_industry = 0 }
	set_temp_variable = { infantry_spawn_context_high_population = 0 }
	set_temp_variable = { infantry_spawn_context_supply = 0 }
	set_temp_variable = { infantry_spawn_context_remote = 0 }
	set_temp_variable = { infantry_spawn_context_mobile_terrain = 0 }
	set_temp_variable = { infantry_spawn_context_front = 0 }
	event_target:infantry_spawn_current_origin_state = {
		if = {
			limit = { is_core_of = event_target:infantry_spawn_generation_country }
			set_temp_variable = { infantry_spawn_context_core = 1 }
		}
		if = {
			limit = { is_capital = yes }
			set_temp_variable = { infantry_spawn_context_capital = 1 }
		}
		if = {
			limit = { industrial_complex > constant:infantry_spawn_state_threshold.industrial }
			set_temp_variable = { infantry_spawn_context_industry = 1 }
		}
		if = {
			limit = { state_population_k > constant:infantry_spawn_state_threshold.high_population_k }
			set_temp_variable = { infantry_spawn_context_high_population = 1 }
		}
		if = {
			limit = { OR = { supply_node > 0 rail_way > 0 naval_base > 0 } }
			set_temp_variable = { infantry_spawn_context_supply = 1 }
		}
		if = {
			limit = {
				OR = {
					infrastructure < constant:infantry_spawn_state_threshold.low_infrastructure
					NOT = { is_core_of = event_target:infantry_spawn_generation_country }
				}
			}
			set_temp_variable = { infantry_spawn_context_remote = 1 }
		}
		if = {
			limit = {
				OR = {
					has_state_category = pastoral
					has_state_category = rural
					has_state_category = town
					has_state_category = large_town
				}
			}
			set_temp_variable = { infantry_spawn_context_mobile_terrain = 1 }
		}
		if = {
			limit = {
				event_target:infantry_spawn_generation_country = { has_war = yes }
				any_neighbor_state = {
					CONTROLLER = { has_war_with = event_target:infantry_spawn_generation_country }
				}
			}
			set_temp_variable = { infantry_spawn_context_front = 1 }
		}
	}
}

infantry_spawn_add_current_family_candidate = {
	if = {
		limit = {
			has_country_flag = infantry_spawn_incident_preserved_depot_affinity
			check_variable = { infantry_spawn_current_candidate_profile > constant:infantry_spawn_lot_family.composite_territorial_division }
			check_variable = { infantry_spawn_current_candidate_profile < constant:infantry_spawn_lot_family.fully_random }
		}
		add_to_temp_variable = { infantry_spawn_current_candidate_weight = constant:infantry_spawn_incident.advanced_affinity_weight }
	}
	if = {
		limit = {
			has_country_flag = infantry_spawn_incident_mounted_tradition_affinity
			check_variable = { infantry_spawn_current_candidate_profile = constant:infantry_spawn_lot_family.mounted_patrol }
		}
		add_to_temp_variable = { infantry_spawn_current_candidate_weight = constant:infantry_spawn_incident.mounted_affinity_weight }
	}
	if = {
		limit = {
			has_country_flag = infantry_spawn_incident_motor_standardization_affinity
			OR = {
				check_variable = { infantry_spawn_current_candidate_profile = constant:infantry_spawn_lot_family.mobile_reserve }
				check_variable = { infantry_spawn_current_candidate_profile = constant:infantry_spawn_lot_family.motor_rifle_formation }
			}
		}
		add_to_temp_variable = { infantry_spawn_current_candidate_weight = constant:infantry_spawn_incident.motor_affinity_weight }
	}
	multiply_temp_variable = { infantry_spawn_current_candidate_weight = infantry_spawn_current_pool_scale }
	round_temp_variable = infantry_spawn_current_candidate_weight
	if = {
		limit = { check_variable = { infantry_spawn_current_candidate_weight > 0 } }
		add_to_temp_array = { infantry_spawn_family_candidate_profile_entries = infantry_spawn_current_candidate_profile }
		add_to_temp_array = { infantry_spawn_family_candidate_weight_entries = infantry_spawn_current_candidate_weight }
	}
}

infantry_spawn_select_current_family_candidate = {
	set_temp_variable = { infantry_spawn_candidate_total_weight = 0 }
	set_temp_variable = { infantry_spawn_candidate_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_candidate_index value = infantry_spawn_family_candidate_profile_entries^num compare = less_than } }
		add_to_temp_variable = { infantry_spawn_candidate_total_weight = infantry_spawn_family_candidate_weight_entries^infantry_spawn_candidate_index }
		add_to_temp_variable = { infantry_spawn_candidate_index = 1 }
	}
	set_temp_variable = { infantry_spawn_current_lot_profile = constant:infantry_spawn_event.invalid_index }
	if = {
		limit = { check_variable = { infantry_spawn_candidate_total_weight > 0 } }
		set_temp_variable = { infantry_spawn_candidate_roll_max = infantry_spawn_candidate_total_weight }
		add_to_temp_variable = { infantry_spawn_candidate_roll_max = 1 }
		set_temp_variable_to_random = {
			var = infantry_spawn_candidate_roll
			min = 1
			max = infantry_spawn_candidate_roll_max
			integer = yes
		}
		set_temp_variable = { infantry_spawn_candidate_index = constant:infantry_spawn_event.first_index }
		while_loop_effect = {
			limit = {
				check_variable = { infantry_spawn_current_lot_profile < constant:infantry_spawn_event.first_index }
				check_variable = { var = infantry_spawn_candidate_index value = infantry_spawn_family_candidate_profile_entries^num compare = less_than }
			}
			set_temp_variable = { infantry_spawn_candidate_current_weight = infantry_spawn_family_candidate_weight_entries^infantry_spawn_candidate_index }
			if = {
				limit = { check_variable = { var = infantry_spawn_candidate_roll value = infantry_spawn_candidate_current_weight compare = greater_than } }
				subtract_from_temp_variable = { infantry_spawn_candidate_roll = infantry_spawn_candidate_current_weight }
			}
			else = { set_temp_variable = { infantry_spawn_current_lot_profile = infantry_spawn_family_candidate_profile_entries^infantry_spawn_candidate_index } }
			add_to_temp_variable = { infantry_spawn_candidate_index = 1 }
		}
	}
	else = { infantry_spawn_mark_ledger_invariant_failure = yes }
	clear_temp_array = infantry_spawn_family_candidate_profile_entries
	clear_temp_array = infantry_spawn_family_candidate_weight_entries
}

infantry_spawn_choose_current_ordinary_family = {
	clear_temp_array = infantry_spawn_family_candidate_profile_entries
	clear_temp_array = infantry_spawn_family_candidate_weight_entries
	infantry_spawn_prepare_current_state_context = yes
	if = {
		limit = { check_variable = { infantry_spawn_current_generation_profile = constant:infantry_spawn_generation_profile.arsenal } }
		set_temp_variable = { infantry_spawn_current_pool_scale = constant:infantry_spawn_factor.tenth }
		infantry_spawn_add_baseline_family_candidates = yes
		set_temp_variable = { infantry_spawn_current_pool_scale = constant:infantry_spawn_factor.one_third }
		infantry_spawn_add_organized_family_candidates = yes
		set_temp_variable = { infantry_spawn_current_pool_scale = constant:infantry_spawn_factor.standard }
		infantry_spawn_add_arsenal_family_candidates = yes
	}
	else_if = {
		limit = { check_variable = { infantry_spawn_current_generation_profile = constant:infantry_spawn_generation_profile.organized } }
		set_temp_variable = { infantry_spawn_current_pool_scale = constant:infantry_spawn_factor.half }
		infantry_spawn_add_baseline_family_candidates = yes
		set_temp_variable = { infantry_spawn_current_pool_scale = constant:infantry_spawn_factor.standard }
		infantry_spawn_add_organized_family_candidates = yes
	}
	else = {
		set_temp_variable = { infantry_spawn_current_pool_scale = constant:infantry_spawn_factor.standard }
		infantry_spawn_add_baseline_family_candidates = yes
	}
	infantry_spawn_select_current_family_candidate = yes
}

infantry_spawn_find_compatible_current_lot = {
	set_temp_variable = { infantry_spawn_current_lot_row_index = constant:infantry_spawn_event.invalid_index }
	if = {
		limit = { infantry_spawn_lot_ledger_is_aligned = yes }
		set_temp_variable = { infantry_spawn_lot_scan_index = constant:infantry_spawn_event.first_index }
		while_loop_effect = {
			limit = {
				check_variable = { infantry_spawn_current_lot_row_index < constant:infantry_spawn_event.first_index }
				check_variable = { var = infantry_spawn_lot_scan_index value = infantry_spawn_lot_uid_entries^num compare = less_than }
			}
			if = {
				limit = {
					check_variable = { var = infantry_spawn_lot_generation_uid_entries^infantry_spawn_lot_scan_index value = infantry_spawn_current_generation_uid compare = equals }
					check_variable = { var = infantry_spawn_lot_family_id_entries^infantry_spawn_lot_scan_index value = infantry_spawn_current_family_id compare = equals }
					check_variable = { var = infantry_spawn_lot_profile_entries^infantry_spawn_lot_scan_index value = infantry_spawn_current_lot_profile compare = equals }
					check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_lot_scan_index value = constant:infantry_spawn_lot_state.demobilizing compare = less_than }
				}
				set_temp_variable = { infantry_spawn_current_lot_row_index = infantry_spawn_lot_scan_index }
			}
			add_to_temp_variable = { infantry_spawn_lot_scan_index = 1 }
		}
	}
	else = { infantry_spawn_mark_ledger_invariant_failure = yes }
}

	set_temp_variable = { infantry_spawn_current_support_count = 0 }
	set_temp_variable = { infantry_spawn_component_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = { check_variable = { var = infantry_spawn_component_scan_index value = infantry_spawn_template_component_template_uid_entries^num compare = less_than } }
		if = {
			limit = { check_variable = { var = infantry_spawn_template_component_template_uid_entries^infantry_spawn_component_scan_index value = infantry_spawn_current_template_uid compare = equals } }
			if = {
				limit = { check_variable = { infantry_spawn_template_component_slot_kind_entries^infantry_spawn_component_scan_index = constant:infantry_spawn_component_kind.combat } }
				add_to_temp_variable = { infantry_spawn_current_combat_count = 1 }

	}

infantry_spawn_accumulate_current_template_needs = {
	infantry_spawn_reset_component_need_accumulators = yes
	if = {
		limit = { infantry_spawn_template_component_ledger_is_aligned = yes }
		set_temp_variable = { infantry_spawn_component_scan_index = constant:infantry_spawn_event.first_index }
		while_loop_effect = {
			limit = { check_variable = { var = infantry_spawn_component_scan_index value = infantry_spawn_template_component_template_uid_entries^num compare = less_than } }
			if = {
				limit = { check_variable = { var = infantry_spawn_template_component_template_uid_entries^infantry_spawn_component_scan_index value = infantry_spawn_current_template_uid compare = equals } }
				set_temp_variable = { infantry_spawn_current_component_profile = infantry_spawn_template_component_unit_profile_entries^infantry_spawn_component_scan_index }
				meta_effect = {
					text = { infantry_spawn_accumulate_component_needs_[COMPONENT] = yes }
					COMPONENT = "[?infantry_spawn_current_component_profile|.0]"
				}
			}
			add_to_temp_variable = { infantry_spawn_component_scan_index = 1 }
		}
	}
	else = { infantry_spawn_mark_ledger_invariant_failure = yes }
}
