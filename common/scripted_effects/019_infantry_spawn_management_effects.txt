# ============================================================================
# CHAOS REDUX - EVENT 019 ORDINARY-COUNTRY MANAGEMENT
# ============================================================================
# Owns the baseline, Organized Muster, and Arsenal Lottery management surface.
# Every lot operation captures a stable lot UID, resolves the current row from
# that UID, and mutates the aligned ledgers only after the engine-side action
# succeeds. The paid-country request dispatcher hands composition to the
# unrestricted registry only at Evolution III; claimant logic and Evolution IV
# family logic live in their own files and are deliberately not implemented here.
# ============================================================================

}

infantry_spawn_record_management_success = {
	infantry_spawn_initialize_management_telemetry = yes
	add_to_variable = { infantry_spawn_successful_management_actions = 1 }
	add_to_variable = { global.infantry_spawn_total_management_successes = 1 }

	clr_country_flag = infantry_spawn_last_management_failed

# Called by the shared staged-pressure helper after its structural congestion
# pass and before tier ideas are selected. Persistent management relief and
# request pressure are applied here so recalculation remains idempotent.
infantry_spawn_apply_management_pressure_adjustments = {
	infantry_spawn_initialize_management_telemetry = yes
	set_temp_variable = { infantry_spawn_request_congestion_pressure = infantry_spawn_request_count_generation }
	multiply_temp_variable = { infantry_spawn_request_congestion_pressure = constant:infantry_spawn_congestion.request_pressure }
	add_to_variable = { infantry_spawn_army_congestion = infantry_spawn_request_congestion_pressure }
	subtract_from_variable = { infantry_spawn_army_congestion = infantry_spawn_management_congestion_relief }
	clamp_variable = { var = infantry_spawn_army_congestion min = constant:infantry_spawn_congestion.minimum max = constant:infantry_spawn_congestion.maximum }
	clamp_variable = { var = infantry_spawn_muster_control min = constant:infantry_spawn_control.minimum max = constant:infantry_spawn_control.maximum }
	if = { limit = { check_variable = { infantry_spawn_equipment_debt < 0 } } set_variable = { infantry_spawn_equipment_debt = 0 } }
	if = { limit = { check_variable = { infantry_spawn_manpower_liability < 0 } } set_variable = { infantry_spawn_manpower_liability = 0 } }
}

infantry_spawn_capture_selected_lot_uid = {
	set_temp_variable = { infantry_spawn_management_target_is_valid = 0 }
	if = {
		limit = { infantry_spawn_selected_lot_index_is_valid = yes }
		set_temp_variable = { infantry_spawn_management_target_lot_uid = infantry_spawn_lot_uid_entries^infantry_spawn_selected_lot_index }
		set_temp_variable = { infantry_spawn_lookup_lot_uid = infantry_spawn_management_target_lot_uid }
		infantry_spawn_find_lot_row = yes
		if = {
			limit = { check_variable = { infantry_spawn_lot_row_index > constant:infantry_spawn_event.invalid_index } }
			set_temp_variable = { infantry_spawn_management_target_is_valid = 1 }
		}
	}
}

infantry_spawn_select_first_unresolved_ordinary_lot = {
	set_variable = { infantry_spawn_selected_lot_index = constant:infantry_spawn_event.invalid_index }
	set_temp_variable = { infantry_spawn_management_scan_index = constant:infantry_spawn_event.first_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_selected_lot_index < constant:infantry_spawn_event.first_index }
			check_variable = { var = infantry_spawn_management_scan_index value = infantry_spawn_lot_uid_entries^num compare = less_than }
		}
		if = {
			limit = {
				check_variable = { infantry_spawn_lot_family_id_entries^infantry_spawn_management_scan_index = constant:infantry_spawn_lot_family.none }
				check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_management_scan_index value = constant:infantry_spawn_lot_state.training compare = less_than }
			}
			set_variable = { infantry_spawn_selected_lot_index = infantry_spawn_management_scan_index }
		}
		add_to_temp_variable = { infantry_spawn_management_scan_index = 1 }
	}
}

infantry_spawn_select_next_unresolved_ordinary_lot = {
	set_temp_variable = { infantry_spawn_management_lot_count = infantry_spawn_lot_uid_entries^num }
	set_temp_variable = { infantry_spawn_management_scan_index = constant:infantry_spawn_event.first_index }
	if = {
		limit = { infantry_spawn_selected_lot_index_is_valid = yes }
		set_temp_variable = { infantry_spawn_management_scan_index = infantry_spawn_selected_lot_index }
		add_to_temp_variable = { infantry_spawn_management_scan_index = 1 }
	}
	set_temp_variable = { infantry_spawn_management_rows_checked = 0 }
	set_variable = { infantry_spawn_selected_lot_index = constant:infantry_spawn_event.invalid_index }
	while_loop_effect = {
		limit = {
			check_variable = { infantry_spawn_selected_lot_index < constant:infantry_spawn_event.first_index }
			check_variable = { var = infantry_spawn_management_rows_checked value = infantry_spawn_management_lot_count compare = less_than }
		}
		if = {
			limit = { check_variable = { var = infantry_spawn_management_scan_index value = infantry_spawn_management_lot_count compare = greater_than_or_equals } }
			set_temp_variable = { infantry_spawn_management_scan_index = constant:infantry_spawn_event.first_index }
		}
		if = {
			limit = {
				check_variable = { infantry_spawn_lot_family_id_entries^infantry_spawn_management_scan_index = constant:infantry_spawn_lot_family.none }
				check_variable = { var = infantry_spawn_lot_status_entries^infantry_spawn_management_scan_index value = constant:infantry_spawn_lot_state.training compare = less_than }
			}
			set_variable = { infantry_spawn_selected_lot_index = infantry_spawn_management_scan_index }
		}
		add_to_temp_variable = { infantry_spawn_management_scan_index = 1 }
		add_to_temp_variable = { infantry_spawn_management_rows_checked = 1 }
	}
}

infantry_spawn_start_selected_lot_audit = {
	if = { limit = { is_ai = yes } infantry_spawn_select_first_auditing_ordinary_lot = yes }
	infantry_spawn_capture_selected_lot_uid = yes
	if = {
		limit = {
			check_variable = { infantry_spawn_management_target_is_valid > 0 }
			check_variable = { infantry_spawn_lot_family_id_entries^infantry_spawn_lot_row_index = constant:infantry_spawn_lot_family.none }
			check_variable = { infantry_spawn_lot_status_entries^infantry_spawn_lot_row_index = constant:infantry_spawn_lot_state.auditing }
			NOT = { has_country_flag = infantry_spawn_audit_decision_running }
		}
		infantry_spawn_pay_base_army_experience_cost = yes
		infantry_spawn_pay_base_support_equipment_cost = yes
		set_variable = { infantry_spawn_audit_target_lot_uid = infantry_spawn_management_target_lot_uid }
		set_variable = { infantry_spawn_audit_mission_days = constant:infantry_spawn_timer.audit_base_days }
		if = {
			limit = { is_major = yes }
			add_to_variable = { infantry_spawn_audit_mission_days = constant:infantry_spawn_timer.audit_large_country_bonus_days }
		}
		if = {
			limit = { has_war = yes }
			subtract_from_variable = { infantry_spawn_audit_mission_days = constant:infantry_spawn_timer.audit_war_penalty_days }
		}
		if = {
			limit = { has_country_flag = infantry_spawn_generation_stacked }
			add_to_variable = { infantry_spawn_audit_mission_days = constant:infantry_spawn_timer.audit_stacked_penalty_days }
		}
		if = {
			limit = { infantry_spawn_has_evolution_i = yes }
			multiply_variable = { infantry_spawn_audit_mission_days = constant:infantry_spawn_factor.three_quarters }
			round_variable = infantry_spawn_audit_mission_days
		}
		if = {
			limit = { check_variable = { infantry_spawn_audit_mission_days < constant:infantry_spawn_timer.audit_minimum_days } }
			set_variable = { infantry_spawn_audit_mission_days = constant:infantry_spawn_timer.audit_minimum_days }
		}
		set_country_flag = infantry_spawn_audit_decision_running
		activate_mission = infantry_spawn_formation_roll_call_mission
	}
	else = { infantry_spawn_record_management_failure = yes }
}

infantry_spawn_complete_selected_lot_audit = {
	set_temp_variable = { infantry_spawn_lookup_lot_uid = infantry_spawn_audit_target_lot_uid }
	infantry_spawn_find_lot_row = yes
	if = {
		limit = {
			check_variable = { infantry_spawn_lot_row_index > constant:infantry_spawn_event.invalid_index }
			check_variable = { infantry_spawn_lot_family_id_entries^infantry_spawn_lot_row_index = constant:infantry_spawn_lot_family.none }
			check_variable = { infantry_spawn_lot_status_entries^infantry_spawn_lot_row_index = constant:infantry_spawn_lot_state.auditing }
		}
		set_variable = { infantry_spawn_lot_status_entries^infantry_spawn_lot_row_index = constant:infantry_spawn_lot_state.revealed }
		set_temp_variable = { infantry_spawn_lookup_generation_uid = infantry_spawn_lot_generation_uid_entries^infantry_spawn_lot_row_index }
		infantry_spawn_find_generation_row = yes
		if = {
			limit = {
				check_variable = { infantry_spawn_generation_row_index > constant:infantry_spawn_event.invalid_index }
				check_variable = { infantry_spawn_generation_status_entries^infantry_spawn_generation_row_index = constant:infantry_spawn_generation_status.open }
			}
			set_variable = { infantry_spawn_generation_status_entries^infantry_spawn_generation_row_index = constant:infantry_spawn_generation_status.audited }
		}
		if = {
			limit = {
				OR = {
					NOT = { infantry_spawn_control_is_low = yes }
					has_country_flag = infantry_spawn_integration_staff_appointed
				}
			}
			add_to_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.audit_success }
			add_to_variable = { infantry_spawn_management_congestion_relief = constant:infantry_spawn_congestion.audit_relief }
			if = {
				limit = {
					OR = {
						has_country_flag = infantry_spawn_audit_organized_success_package
						infantry_spawn_has_evolution_i = yes
					}
				}
				add_to_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.territorial_assignment }
				add_to_variable = { infantry_spawn_management_congestion_relief = constant:infantry_spawn_congestion.audit_relief }
			}
			infantry_spawn_record_management_success = yes
		}
		else = {
			add_to_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.territorial_assignment }
			infantry_spawn_record_management_success = yes
			set_country_flag = infantry_spawn_last_management_partial
		}
	}
	else = {
		subtract_from_variable = { infantry_spawn_muster_control = constant:infantry_spawn_control.failed_mission_penalty }
		subtract_from_variable = { infantry_spawn_management_congestion_relief = constant:infantry_spawn_congestion.audit_relief }
		infantry_spawn_record_management_failure = yes
	}
	clr_country_flag = infantry_spawn_audit_decision_running
	clr_country_flag = infantry_spawn_audit_organized_success_package
	clear_variable = infantry_spawn_audit_target_lot_uid
	infantry_spawn_recalculate_country_pressure = yes
}

	if = { limit = { is_ai = yes } infantry_spawn_select_weakest_unresolved_ordinary_lot = yes }
	infantry_spawn_capture_selected_lot_uid = yes

			NOT = { has_country_flag = infantry_spawn_audit_decision_running }
		}

		set_variable = { infantry_spawn_lot_status_entries^infantry_spawn_lot_row_index = constant:infantry_spawn_lot_state.territorial }

		infantry_spawn_record_management_success = yes
		infantry_spawn_close_resolved_generations = yes
		infantry_spawn_select_first_unresolved_ordinary_lot = yes
	}
	else = { infantry_spawn_record_management_failure = yes }
	infantry_spawn_recalculate_country_pressure = yes

				set_state_flag = infantry_spawn_reserved_rail_corridor

			}
		}
	}

	clear_variable = infantry_spawn_rail_corridor_target_state
	infantry_spawn_recalculate_country_pressure = yes
}

# Public paid-request dispatch. Evolution II keeps its weighted serious,
# advanced, and strange family profiles. Evolution III alone crosses into the
# unrestricted component registry, where quality and coherence roll separately.
infantry_spawn_dispatch_management_request_lot = {
	set_temp_variable = { infantry_spawn_request_generation_succeeded = 0 }
	infantry_spawn_prepare_request_generation_context = yes
	if = {
		limit = {
			check_variable = { infantry_spawn_request_context_ready > 0 }
		}
		random_controlled_state = {
			limit = { infantry_spawn_state_is_eligible = yes }
			save_event_target_as = infantry_spawn_current_origin_state
		}
		if = {
			limit = { infantry_spawn_has_evolution_iii = yes }
			set_temp_variable = { var = infantry_spawn_unit_registry_request_mode value = infantry_spawn_request_mode_selection }
			set_temp_variable = { var = infantry_spawn_unit_registry_request_context value = constant:infantry_spawn_unit_registry_request_context.country_request }
			set_temp_variable = { var = infantry_spawn_unit_registry_request_command_owner value = constant:infantry_spawn_command_owner.state }
			infantry_spawn_unit_registry_draw_requested_lot = yes
			if = {
				limit = {
					is_in_array = {
						array = infantry_spawn_unit_registry_outcome_tag_ids
						value = constant:infantry_spawn_unit_registry_outcome_tag.draw_complete
					}
				}
				infantry_spawn_unit_registry_materialize_drawn_lot = yes
			}
			if = {
				limit = { has_country_flag = infantry_spawn_unit_registry_materialization_succeeded }
				set_temp_variable = { infantry_spawn_request_generation_succeeded = 1 }
			}
		}
		else = {
			set_temp_variable = { infantry_spawn_current_generation_profile = constant:infantry_spawn_generation_profile.arsenal }
			set_temp_variable = { infantry_spawn_current_family_id = constant:infantry_spawn_lot_family.none }
			infantry_spawn_build_request_candidate_pool = yes
			if = {
				limit = { check_variable = { infantry_spawn_current_lot_profile > constant:infantry_spawn_event.invalid_index } }
				infantry_spawn_create_current_ordinary_lot = yes
				if = {
					limit = { NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure } }
					infantry_spawn_append_selected_state_row = yes
					add_to_temp_variable = { infantry_spawn_current_selected_state_count = 1 }
					infantry_spawn_roll_units_for_current_state = yes
					set_temp_variable = { infantry_spawn_current_state_unit_index = 0 }
					while_loop_effect = {
						limit = {
							check_variable = { var = infantry_spawn_current_state_unit_index value = infantry_spawn_current_state_unit_target compare = less_than }
							NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
						}
						infantry_spawn_spawn_current_template_unit = yes
						add_to_temp_variable = { infantry_spawn_current_state_unit_index = 1 }
					}
					if = {
						limit = { NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure } }
						infantry_spawn_update_current_generation_totals = yes
						set_temp_variable = { infantry_spawn_request_generation_succeeded = 1 }
					}
				}
			}
		}
	}
}

		else = {
			set_country_flag = infantry_spawn_last_request_failed
			clr_country_flag = infantry_spawn_last_request_succeeded
			infantry_spawn_record_management_failure = yes
		}
		infantry_spawn_recalculate_country_pressure = yes
		infantry_spawn_refresh_request_costs = yes
	}
	else = { infantry_spawn_record_management_failure = yes }
}

infantry_spawn_finish_request_cooldown = {
	clr_country_flag = infantry_spawn_request_cooldown
	clear_variable = infantry_spawn_request_cooldown_days
	infantry_spawn_refresh_request_costs = yes
}

# Public country pulse / AI-parity hook. It does no global iteration and is
# safe for the Event 19 country pulse to call repeatedly.
infantry_spawn_run_country_management_pulse = {
	if = {
		limit = { infantry_spawn_country_is_active = yes }
		infantry_spawn_initialize_management_telemetry = yes
		infantry_spawn_close_resolved_generations = yes
		if = { limit = { infantry_spawn_has_evolution_i = yes } infantry_spawn_evolution_one_reevaluate_management = yes }
		if = { limit = { infantry_spawn_has_evolution_ii = yes } infantry_spawn_evolution_two_expand_management = yes }
		if = {
			limit = { NOT = { infantry_spawn_selected_lot_is_unresolved = yes } }
			infantry_spawn_select_first_unresolved_ordinary_lot = yes
		}
		if = {
			limit = { is_ai = yes }
			if = {
				limit = { infantry_spawn_congestion_is_severe = yes }
				infantry_spawn_select_weakest_unresolved_ordinary_lot = yes
			}
			else_if = {
				limit = { has_war = yes }
				infantry_spawn_select_strongest_unresolved_ordinary_lot = yes
			}
			else = { infantry_spawn_select_first_unresolved_ordinary_lot = yes }
		}
		infantry_spawn_recalculate_country_pressure = yes
	}
}
