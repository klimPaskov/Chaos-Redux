# Country scope. Preserve a legal installed campus or provision one exact
# building type. Rejected temporary transfers restore owner and controller.
# Input: temporary building token chaosx_test_country_current_facility.
# Resolve native construction checks in this template; do not nest a
# meta-trigger expansion in its search. Naval requires a coastal state.
chaosx_test_country_provision_facility_type = {
	if = {
		limit = { has_country_flag = chaosx_test_country_setup_trace_active }
		log = "CXT_SETUP_TRACE facility_begin type=[?chaosx_test_country_current_facility.GetTokenKey] date=[?global.date|.3]"
	}
	meta_effect = {
		text = {
			if = {
				limit = { has_country_flag = chaosx_test_country_setup_trace_active }
				log = "CXT_SETUP_TRACE facility_body_begin type=[FACILITY] date=[?global.date|.3]"
			}
			if = {
				limit = {
					NOT = {
						any_owned_state = {
							is_controlled_by = ROOT
							[FACILITY] > 0
						}
					}
				}
				clear_array = chaosx_test_facility_rejected_states
				set_temp_variable = { chaosx_test_facility_type_candidate_failed = 0 }
				if = {
					limit = { has_country_flag = chaosx_test_country_setup_trace_active }
					log = "CXT_SETUP_TRACE facility_search_begin type=[FACILITY] date=[?global.date|.3]"
				}
				while_loop_effect = {
					break = chaosx_test_facility_type_candidate_failed
					limit = {
						NOT = {
							any_owned_state = {
								chaosx_test_country_state_meets_facility_requirements = yes
								can_construct_building = [FACILITY]
							}
						}
					}
					set_temp_variable = { chaosx_test_facility_type_candidate_selected = 0 }
					random_state = {
						limit = {
							NOT = { is_owned_by = ROOT }
							NOT = { is_capital = yes }
							impassable = no
							is_demilitarized_zone = no
							NOT = { chaosx_test_country_state_has_special_facility = yes }
							OWNER = { exists = yes }
							CONTROLLER = { exists = yes }
							OR = {
								NOT = { check_variable = { chaosx_test_country_current_facility = token:naval_facility } }
								is_coastal = yes
							}
							ROOT = {
								NOT = {
									is_in_array = {
										array = chaosx_test_facility_rejected_states
										value = PREV
									}
								}
							}
						}
						set_temp_variable = { chaosx_test_facility_type_candidate_selected = 1 }
						OWNER = { save_event_target_as = chaosx_test_facility_original_owner }
						CONTROLLER = { save_event_target_as = chaosx_test_facility_original_controller }
						if = {
							limit = { ROOT = { has_country_flag = chaosx_test_country_setup_trace_active } }
							log = "CXT_SETUP_TRACE facility_transfer_begin state=[This.GetName] type=[FACILITY] date=[?global.date|.3]"
						}
						transfer_state_to = ROOT
						set_state_controller_to = ROOT
						if = {
							limit = { ROOT = { has_country_flag = chaosx_test_country_setup_trace_active } }
							log = "CXT_SETUP_TRACE facility_transfer_end state=[This.GetName] type=[FACILITY] date=[?global.date|.3]"
						}
						if = {
							limit = {
								NOT = {
									AND = {
										chaosx_test_country_state_meets_facility_requirements = yes
										can_construct_building = [FACILITY]
									}
								}
							}
							ROOT = {
								add_to_array = {
									array = chaosx_test_facility_rejected_states
									value = PREV
								}
							}
							if = {
								limit = { ROOT = { has_country_flag = chaosx_test_country_setup_trace_active } }
								log = "CXT_SETUP_TRACE facility_restore_begin state=[This.GetName] type=[FACILITY] date=[?global.date|.3]"
							}
							transfer_state_to = event_target:chaosx_test_facility_original_owner
							set_state_controller_to = event_target:chaosx_test_facility_original_controller
							if = {
								limit = { ROOT = { has_country_flag = chaosx_test_country_setup_trace_active } }
								log = "CXT_SETUP_TRACE facility_restore_end state=[This.GetName] type=[FACILITY] date=[?global.date|.3]"
							}
						}
					}
					if = {
						limit = { check_variable = { chaosx_test_facility_type_candidate_selected < 1 } }
						set_temp_variable = { chaosx_test_facility_type_candidate_failed = 1 }
					}
				}
				if = {
					limit = { has_country_flag = chaosx_test_country_setup_trace_active }
					log = "CXT_SETUP_TRACE facility_search_end type=[FACILITY] date=[?global.date|.3]"
				}
				random_owned_controlled_state = {
					limit = {
						chaosx_test_country_state_meets_facility_requirements = yes
						can_construct_building = [FACILITY]
					}
					if = {
						limit = { ROOT = { has_country_flag = chaosx_test_country_setup_trace_active } }
						log = "CXT_SETUP_TRACE facility_construction_begin state=[This.GetName] type=[FACILITY] date=[?global.date|.3]"
					}
					construct_building_in_random_province = { [FACILITY] = 1 }
					if = {
						limit = { ROOT = { has_country_flag = chaosx_test_country_setup_trace_active } }
						log = "CXT_SETUP_TRACE facility_construction_end state=[This.GetName] type=[FACILITY] date=[?global.date|.3]"
					}
				}
				clear_array = chaosx_test_facility_rejected_states
			}
			if = {
				limit = {
					any_owned_state = {
						is_controlled_by = ROOT
						[FACILITY] > 0
					}
				}
				clr_country_flag = chaosx_test_facility_missing_[FACILITY]
			}
			else = {
				set_country_flag = chaosx_test_country_facility_provisioning_incomplete
				set_country_flag = chaosx_test_facility_missing_[FACILITY]
			}
		}
		FACILITY = "[?chaosx_test_country_current_facility.GetTokenKey]"
	}
	if = {
		limit = { has_country_flag = chaosx_test_country_setup_trace_active }
		log = "CXT_SETUP_TRACE facility_end type=[?chaosx_test_country_current_facility.GetTokenKey] date=[?global.date|.3]"
	}
}

# Input: temporary variable chaosx_test_country_registration_extension_effect
# containing token:<hidden_idea_carrier>. The matching scripted effect must use
# the carrier id plus an _apply suffix, be safe in country scope, and be
# idempotent because initial setup and the manual package decision invoke it.
chaosx_test_country_register_extension_effect = {
	set_temp_variable = { chaosx_test_country_extension_registration_added = 0 }
	if = {
		limit = { has_variable = chaosx_test_country_registration_extension_effect }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.chaosx_test_country_registered_extension_effects
						value = chaosx_test_country_registration_extension_effect
					}
				}
			}
			add_to_array = {
				array = global.chaosx_test_country_registered_extension_effects
				value = chaosx_test_country_registration_extension_effect
			}
			set_temp_variable = { chaosx_test_country_extension_registration_added = 1 }
		}
	}
}

chaosx_test_country_apply_registered_extension_effects = {
	for_each_loop = {
		array = global.chaosx_test_country_registered_extension_effects
		value = chaosx_test_country_current_extension_effect
		if = {
			limit = { has_country_flag = chaosx_test_country_setup_trace_active }
			log = "CXT_SETUP_TRACE extension_begin carrier=[?chaosx_test_country_current_extension_effect.GetTokenKey] date=[?global.date|.3]"
		}
		meta_effect = {
			text = { [EXTENSION_EFFECT]_apply = yes }
			EXTENSION_EFFECT = "[?chaosx_test_country_current_extension_effect.GetTokenKey]"
		}
		if = {
			limit = { has_country_flag = chaosx_test_country_setup_trace_active }
			log = "CXT_SETUP_TRACE extension_end carrier=[?chaosx_test_country_current_extension_effect.GetTokenKey] date=[?global.date|.3]"
		}
	}
}

# Initial setup and the manual package decision consume registered content.
# Every package effect must remain idempotent. Trace output is temporary until
# the reported first-day freeze has an identified blocking stage.

	clr_country_flag = chaosx_test_country_registered_sync_pending
	clr_country_flag = chaosx_test_country_registered_sync_queued

		limit = { has_country_flag = chaosx_test_country_setup_trace_active }

		limit = { has_country_flag = chaosx_test_country_setup_trace_active }

		limit = { has_country_flag = chaosx_test_country_setup_trace_active }
		log = "CXT_SETUP_TRACE sync_complete country=[This.GetTag] date=[?global.date|.3]"
	}
}

chaosx_test = {
	# Temporary console-to-receiver trace for the unresolved activation freeze.
	log = "CXT_SETUP_TRACE console_entry date=[?global.date|.3]"
	if = {
		limit = { tag = CXT }
		country_event = { id = chaosx_test_country.1 }
	}
	else = {
		if = {
			limit = { any_owned_state = { is_capital = yes } }
			save_event_target_as = chaosx_test_origin_country
			capital_scope = {
				save_event_target_as = chaosx_test_origin_capital_state
				transfer_state_to = CXT
				set_state_controller_to = CXT
			}
			# Resolve the dormant tag only after its capital has activated it.
			# The receiver event gives every setup helper ROOT = CXT.
			meta_effect = {
				text = {
					CXT = {
						set_capital = {
							state = event_target:chaosx_test_origin_capital_state
							remember_old_capital = no
						}
						# Queue from a valid CXT scope; player transfer must run last.
						log = "CXT_SETUP_TRACE scheduling_activation_receiver"
						country_event = { id = chaosx_test_country.1 hours = [SETUP_DELAY] }
						change_tag_from = event_target:chaosx_test_origin_country
					}
				}
				SETUP_DELAY = @CXT_SETUP_DELAY_HOURS
			}
		}
	}
}
