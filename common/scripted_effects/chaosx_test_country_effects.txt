# CHAOS REDUX - TEST COUNTRY HARNESS
# ============================================================================
# Overview:
# - The public console entry point is "e chaosx_test".
# - Switches the current player to CXT, annexes the former player country
#   without importing vanilla divisions, and preserves the former capital.
# - Completes the merged technology and special-project inventories, activates
#   Chaos Warfare, provisions every Chaos Redux land unit, and turns on the
#   shared CBRN and camp systems.
# - Maintains tag-specific daily resources and weekly stockpile replenishment
#   without a world iteration.
# ============================================================================

chaosx_test_country_ensure_facility_states = {
	if = {
		limit = {
			NOT = {
				any_owned_state = {
					is_controlled_by = ROOT
					is_coastal = yes
					NOT = { chaosx_test_country_state_has_special_facility = yes }
				}
			}
		}
		random_state = {
			limit = {
				is_coastal = yes
				NOT = { is_owned_by = ROOT }
				NOT = { chaosx_test_country_state_has_special_facility = yes }
			}
			transfer_state_to = ROOT
		}
	}

	set_temp_variable = { chaosx_test_empty_facility_states = 0 }
	every_owned_state = {
		limit = { chaosx_test_country_state_can_host_special_facility = yes }
		add_to_temp_variable = { chaosx_test_empty_facility_states = 1 }
	}
	set_temp_variable = { chaosx_test_facility_state_transfer_failed = 0 }
	while_loop_effect = {
		break = chaosx_test_facility_state_transfer_failed
		limit = {
			check_variable = {
				var = chaosx_test_empty_facility_states
				value = constant:chaosx_test_country_count.minimum_facility_states
				compare = less_than
			}
		}
		set_temp_variable = { chaosx_test_facility_state_transfer_succeeded = 0 }
		random_state = {
			limit = {
				NOT = { is_owned_by = ROOT }
				NOT = { chaosx_test_country_state_has_special_facility = yes }
			}
			transfer_state_to = ROOT
			set_temp_variable = { chaosx_test_facility_state_transfer_succeeded = 1 }
		}
		if = {
			limit = { check_variable = { chaosx_test_facility_state_transfer_succeeded > 0 } }
			add_to_temp_variable = { chaosx_test_empty_facility_states = 1 }
		}
		else = {
			set_temp_variable = { chaosx_test_facility_state_transfer_failed = 1 }
		}
	}
}

chaosx_test_country_ensure_occupation_fixture = {
	if = {
		limit = {
			NOT = {
				any_controlled_state = {
					NOT = { is_owned_by = ROOT }
					NOT = { is_core_of = ROOT }
				}
			}
		}
		random_state = {
			limit = {
				NOT = { is_owned_by = ROOT }
				NOT = { is_core_of = ROOT }
				NOT = { is_capital = yes }
				state_population_k > 0
			}
			set_state_controller_to = ROOT
		}
	}
}

chaosx_test = {
	if = {
		limit = { tag = CXT }
		if = {
			limit = { NOT = { has_country_flag = chaosx_test_country_initialized } }
			chaosx_test_country_initial_setup = yes
		}
		else = {
			chaosx_test_country_refresh = yes
		}
	}
	else = {
		save_event_target_as = chaosx_test_origin_country
		capital_scope = { save_event_target_as = chaosx_test_origin_capital_state }
		CXT = {
			change_tag_from = event_target:chaosx_test_origin_country
			annex_country = {
				target = event_target:chaosx_test_origin_country
				transfer_troops = no
			}
			set_capital = {
				state = event_target:chaosx_test_origin_capital_state
				remember_old_capital = no
			}
			if = {
				limit = { NOT = { has_country_flag = chaosx_test_country_initialized } }
				chaosx_test_country_initial_setup = yes
			}
			else = {
				chaosx_test_country_refresh = yes
			}
		}
	}
}
