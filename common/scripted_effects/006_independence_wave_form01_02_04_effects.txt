# Member-country scope. The caller has already verified consent in the frozen
# array row. This establishes concrete bilateral protection without forcing a
# faction, subject relation, tree replacement, or territorial transfer.
independence_wave_form0124_install_autonomous_member = {
	save_event_target_as = independence_wave_form0124_autonomous_member
	set_country_flag = independence_wave_form0124_autonomous_member
	set_variable = { independence_wave_form0124_autonomous_family = ROOT.independence_wave_formable_family }
	set_variable = { independence_wave_form0124_autonomous_carrier = ROOT }
	set_variable = { independence_wave_form0124_autonomous_carrier_generation = ROOT.independence_wave_generation_id }
	if = {
		limit = { NOT = { gives_military_access_to = ROOT } }
		give_military_access = ROOT
		set_country_flag = independence_wave_form0124_member_granted_access_to_carrier
	}
	if = {
		limit = { NOT = { has_guaranteed = ROOT } }
		give_guarantee = ROOT
		set_country_flag = independence_wave_form0124_member_guaranteed_carrier
	}
	ROOT = {
		if = {
			limit = { NOT = { gives_military_access_to = event_target:independence_wave_form0124_autonomous_member } }
			give_military_access = event_target:independence_wave_form0124_autonomous_member
			event_target:independence_wave_form0124_autonomous_member = { set_country_flag = independence_wave_form0124_carrier_granted_access_to_member }
		}
		if = {
			limit = { NOT = { has_guaranteed = event_target:independence_wave_form0124_autonomous_member } }
			give_guarantee = event_target:independence_wave_form0124_autonomous_member
			event_target:independence_wave_form0124_autonomous_member = { set_country_flag = independence_wave_form0124_carrier_guaranteed_member }
		}
		set_country_flag = independence_wave_form0124_has_autonomous_member
	}
	if = {
		limit = { check_variable = { ROOT.independence_wave_formable_family = constant:independence_wave_formable_family.celtic_cooperation_state } }
		add_ideas = independence_wave_form01_autonomous_congress_member
	}
	else_if = {
		limit = { check_variable = { ROOT.independence_wave_formable_family = constant:independence_wave_formable_family.north_atlantic_compact } }
		add_ideas = independence_wave_form02_autonomous_union_member
	}
	else_if = {
		limit = { check_variable = { ROOT.independence_wave_formable_family = constant:independence_wave_formable_family.rhine_federation } }
		add_ideas = independence_wave_form04_autonomous_league_member
	}
}

# Autonomous-member country scope. Only relations whose directional ownership
# flags were set when Event 006 created them are removed.
independence_wave_form0124_remove_event6_autonomous_relations = {
	save_event_target_as = independence_wave_form0124_cleanup_member
	if = {
		limit = {
			has_variable = independence_wave_form0124_autonomous_carrier
			var:independence_wave_form0124_autonomous_carrier = { exists = yes }
		}
		var:independence_wave_form0124_autonomous_carrier = { save_event_target_as = independence_wave_form0124_cleanup_carrier }
		if = {
			limit = { has_country_flag = independence_wave_form0124_member_granted_access_to_carrier }
			diplomatic_relation = { country = event_target:independence_wave_form0124_cleanup_carrier relation = military_access active = no }
		}
		if = {
			limit = { has_country_flag = independence_wave_form0124_member_guaranteed_carrier }
			diplomatic_relation = { country = event_target:independence_wave_form0124_cleanup_carrier relation = guarantee active = no }
		}
		if = {
			limit = { has_country_flag = independence_wave_form0124_carrier_granted_access_to_member }
			event_target:independence_wave_form0124_cleanup_carrier = {
				diplomatic_relation = { country = event_target:independence_wave_form0124_cleanup_member relation = military_access active = no }
			}
		}
		if = {
			limit = { has_country_flag = independence_wave_form0124_carrier_guaranteed_member }
			event_target:independence_wave_form0124_cleanup_carrier = {
				diplomatic_relation = { country = event_target:independence_wave_form0124_cleanup_member relation = guarantee active = no }
			}
		}
	}
	clr_country_flag = independence_wave_form0124_member_granted_access_to_carrier
	clr_country_flag = independence_wave_form0124_member_guaranteed_carrier
	clr_country_flag = independence_wave_form0124_carrier_granted_access_to_member
	clr_country_flag = independence_wave_form0124_carrier_guaranteed_member
	clear_variable = independence_wave_form0124_autonomous_carrier
	clear_variable = independence_wave_form0124_autonomous_carrier_generation
}

# Carrier-country scope. Uses the frozen founding ledger; it never searches all
# countries and never touches FORM-03 sovereign associates.
independence_wave_form0124_cleanup_frozen_autonomous_members = {
	if = {
		limit = { independence_wave_formable_member_arrays_are_aligned = yes }
		set_temp_variable = { independence_wave_form0124_cleanup_member_index = constant:independence_wave_formable_tuning.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_form0124_cleanup_member_index value = independence_wave_formable_member_country_entries^num compare = less_than } }
			var:independence_wave_formable_member_country_entries^independence_wave_form0124_cleanup_member_index = {
				if = {
					limit = {
						has_country_flag = independence_wave_form0124_autonomous_member
						has_variable = independence_wave_form0124_autonomous_carrier
						has_variable = independence_wave_form0124_autonomous_carrier_generation
						var:independence_wave_form0124_autonomous_carrier = { tag = ROOT }
						check_variable = { var = independence_wave_form0124_autonomous_carrier_generation value = ROOT.independence_wave_generation_id compare = equals }
