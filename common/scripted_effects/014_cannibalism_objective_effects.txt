# Bounded observer registry

cannibalism_register_current_country_objective_observer = {
	if = {
		limit = {
			exists = yes
			NOT = { is_in_array = { array = global.cannibalism_actor_countries value = THIS } }
			NOT = { is_in_array = { array = global.cannibalism_objective_countries value = THIS } }
		}
		add_to_array = { array = global.cannibalism_objective_countries value = THIS }
	}
}

cannibalism_unregister_current_country_objective_observer_if_idle = {
	if = {
		limit = {
			NOT = { cannibalism_current_country_has_active_maintained_objective = yes }
			is_in_array = { array = global.cannibalism_objective_countries value = THIS }
		}
		remove_from_array = { array = global.cannibalism_objective_countries value = THIS }
	}
}

cannibalism_start_investigation_mission = {
	if = {
		limit = {
			cannibalism_containment_decisions_visible = yes
			cannibalism_has_valid_primary_theater = yes
			NOT = { has_country_flag = cannibalism_investigation_mission_active }
			NOT = { has_country_flag = cannibalism_investigation_objective_resolved }
		}
		set_country_flag = cannibalism_investigation_mission_active
		set_variable = { cannibalism_investigation_target_state = cannibalism_primary_theater_state }
		clr_country_flag = cannibalism_investigation_target_has_node_identity
		clear_variable = cannibalism_investigation_target_node_id
		clear_variable = cannibalism_investigation_target_node_generation
		var:cannibalism_investigation_target_state = {
			if = {
				limit = {
					has_state_flag = cannibalism_active_node
					has_variable = cannibalism_node_id
					has_variable = cannibalism_node_generation
				}
				ROOT = {
					set_country_flag = cannibalism_investigation_target_has_node_identity
					set_variable = { cannibalism_investigation_target_node_id = PREV.cannibalism_node_id }
					set_variable = { cannibalism_investigation_target_node_generation = PREV.cannibalism_node_generation }
				}
			}
		}
		set_variable = { cannibalism_investigation_progress = constant:cannibalism_objective_value.zero }
		set_variable = { cannibalism_investigation_goal = constant:cannibalism_objective_progress.investigation_goal }
		set_variable = { cannibalism_investigation_duration = constant:cannibalism_objective_timing.investigation_base_days }
		if = {
			limit = { has_global_flag = cannibalism_evolution_i_active }
			add_to_variable = { cannibalism_investigation_duration = constant:cannibalism_objective_timing.investigation_evolution_add_days }
		}
		if = {
			limit = { check_variable = { cannibalism_cell_strength > constant:cannibalism_meter.severe } }
			add_to_variable = { cannibalism_investigation_duration = constant:cannibalism_objective_timing.investigation_severe_cell_add_days }
		}
		if = {
			limit = { has_country_flag = cannibalism_policy_open_emergency }
			subtract_from_variable = { cannibalism_investigation_duration = constant:cannibalism_objective_timing.investigation_open_response_reduction_days }
		}
		clamp_variable = { var = cannibalism_investigation_duration min = constant:cannibalism_objective_timing.investigation_minimum_days max = constant:cannibalism_objective_timing.investigation_maximum_days }
		activate_mission = cannibalism_investigation_mission
	}
}

cannibalism_record_investigation_operation = {
	if = {
		limit = { NOT = { has_country_flag = cannibalism_investigation_mission_active } }
		cannibalism_start_investigation_mission = yes
	}
	if = {
		limit = {
			has_country_flag = cannibalism_investigation_mission_active
			cannibalism_investigation_target_is_valid = yes
		}
		add_to_variable = { cannibalism_investigation_progress = constant:cannibalism_objective_value.one }
		clamp_variable = { var = cannibalism_investigation_progress min = constant:cannibalism_objective_value.zero max = cannibalism_investigation_goal }
	}
}

cannibalism_start_hold_prison_mission = {
	if = {
		limit = {
			cannibalism_containment_decisions_visible = yes
			NOT = { has_country_flag = cannibalism_hold_prison_mission_active }
			NOT = { has_country_flag = cannibalism_hold_prison_objective_resolved }
		}
		random_controlled_state = {
			limit = {
				impassable = no
				cannibalism_state_has_prison_or_camp = yes
				has_state_flag = cannibalism_active_node
				has_variable = cannibalism_node_id
				has_variable = cannibalism_node_generation
			}
			save_event_target_as = cannibalism_prison_objective_state
			ROOT = {
				set_country_flag = cannibalism_hold_prison_mission_active
				set_variable = { cannibalism_prison_target_state = event_target:cannibalism_prison_objective_state }
				set_variable = { cannibalism_prison_target_node_id = event_target:cannibalism_prison_objective_state.cannibalism_node_id }
				set_variable = { cannibalism_prison_target_node_generation = event_target:cannibalism_prison_objective_state.cannibalism_node_generation }
				set_variable = { cannibalism_prison_hold_days = constant:cannibalism_objective_value.zero }
				set_variable = { cannibalism_prison_required_hold_days = constant:cannibalism_objective_progress.prison_goal_days }
				set_variable = { cannibalism_prison_division_floor = constant:cannibalism_objective_value.zero }
				set_variable = { cannibalism_prison_duration = constant:cannibalism_objective_timing.prison_base_days }
				if = {
					limit = { check_variable = { cannibalism_cell_strength > constant:cannibalism_meter.severe } }
					add_to_variable = { cannibalism_prison_duration = constant:cannibalism_objective_timing.prison_severe_cell_add_days }
				}
				if = {
					limit = { event_target:cannibalism_prison_objective_state = { OR = { is_island_state = yes NOT = { is_in_home_area = yes } } } }
					add_to_variable = { cannibalism_prison_duration = constant:cannibalism_objective_timing.prison_isolated_add_days }
				}
				clamp_variable = { var = cannibalism_prison_duration min = constant:cannibalism_objective_timing.prison_minimum_days max = constant:cannibalism_objective_timing.prison_maximum_days }
				activate_mission = cannibalism_hold_prison_mission
			}
		}
	}
}

			event_target:cannibalism_objective_country_target = { cannibalism_is_valid_silent_island_recon_target = yes }

			subtract_from_variable = { cannibalism_reach_island_duration = constant:cannibalism_objective_timing.island_blockade_reduction_days }

			add_to_variable = { cannibalism_reach_island_duration = constant:cannibalism_objective_timing.island_large_host_add_days }

# Reconstruction compact progress and bounded category retirement

cannibalism_refresh_compact_vigilance_progress = {
	if = {
		limit = { has_variable = cannibalism_compact_vigilance_start_date }
		set_variable = { cannibalism_compact_vigilance_progress_days = global.date }
		subtract_from_variable = { cannibalism_compact_vigilance_progress_days = cannibalism_compact_vigilance_start_date }
		clamp_variable = { var = cannibalism_compact_vigilance_progress_days min = constant:cannibalism_objective_value.zero max = constant:cannibalism_objective_timing.compact_vigilance_days }
	}
}

cannibalism_refresh_global_aftermath_category_lifecycle = {
	set_temp_variable = { cannibalism_reconstruction_participant_found = constant:cannibalism_objective_value.zero }
	for_each_scope_loop = {
		array = global.cannibalism_reconstruction_participants
		if = {
			limit = {
				exists = yes
				has_country_flag = cannibalism_reconstruction_participant
			}
			set_temp_variable = { cannibalism_reconstruction_participant_found = constant:cannibalism_objective_value.one }
		}
	}
	if = {
		limit = { check_variable = { cannibalism_reconstruction_participant_found < constant:cannibalism_objective_value.one } }
		clr_global_flag = cannibalism_reconstruction_system_active
	}
}

cannibalism_clear_current_country_compact_vigilance_runtime = {
	clr_country_flag = cannibalism_compact_vigilance_mission_active
	clr_country_flag = cannibalism_reconstruction_participant
	if = {
		limit = { is_in_array = { array = global.cannibalism_reconstruction_participants value = THIS } }
		remove_from_array = { array = global.cannibalism_reconstruction_participants value = THIS }
	}
	clear_variable = cannibalism_compact_vigilance_start_date
	clear_variable = cannibalism_compact_vigilance_progress_days
	clear_variable = cannibalism_compact_vigilance_duration
	cannibalism_refresh_current_country_reconstruction_idea = yes
	cannibalism_refresh_global_aftermath_category_lifecycle = yes
}

cannibalism_retire_current_country_reconstruction_participation = {
	cannibalism_clear_current_country_compact_vigilance_runtime = yes
	set_country_flag = cannibalism_reconstruction_participant_complete
}

cannibalism_partial_compact_vigilance = {
	set_country_flag = cannibalism_compact_vigilance_partial
	cannibalism_retire_current_country_reconstruction_participation = yes
}

cannibalism_resolve_compact_vigilance = {
	cannibalism_refresh_compact_vigilance_progress = yes
	if = {
		limit = {
			cannibalism_compact_vigilance_is_maintained = yes
			cannibalism_compact_vigilance_has_full_progress = yes
		}
		cannibalism_complete_compact_vigilance = yes
	}
	else_if = {
		limit = { cannibalism_compact_vigilance_has_partial_progress = yes }
		cannibalism_partial_compact_vigilance = yes
	}
	else = { cannibalism_fail_compact_vigilance = yes }
}
