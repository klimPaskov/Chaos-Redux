# CHAOS REDUX - EVENT 014 WARLORD DECISION EFFECTS
# ============================================================================
# Performs exact Deaths-backed state consumption and recruitment. Scripted
# units begin with no generated manpower or equipment; the exact population
# transaction fills the manpower pool and existing stockpiles reinforce them.

cannibalism_pay_current_warlord_larder_cost = {
	if = {
		limit = { has_country_flag = cannibalism_inherited_warlord_recruitment_open }
		set_temp_variable = { cannibalism_unified_larder_cost = cannibalism_warlord_larder_cost }
		cannibalism_unified_pay_current_larder_cost = yes
	}
	else = {
		set_temp_variable = { cannibalism_warlord_larder_cost_negative = cannibalism_warlord_larder_cost }
		multiply_temp_variable = { cannibalism_warlord_larder_cost_negative = constant:cannibalism_unified_profile.minus_one }
		add_to_variable = { cannibalism_larder = cannibalism_warlord_larder_cost_negative }
		clamp_variable = {
			var = cannibalism_larder
			min = constant:cannibalism_consumption.larder_min
			max = constant:cannibalism_consumption.larder_max
		}
	}
}

cannibalism_register_selected_state_to_current_warlord = {
	if = {
		limit = {
			has_event_target = cannibalism_warlord_recruitment_state
			has_event_target = cannibalism_warlord_recruitment_actor
			event_target:cannibalism_warlord_recruitment_state = { is_controlled_by = event_target:cannibalism_warlord_recruitment_actor }
		}
		event_target:cannibalism_warlord_recruitment_actor = { save_event_target_as = cannibalism_node_source_actor }
		event_target:cannibalism_warlord_recruitment_state = {
			if = {
				limit = { NOT = { has_state_flag = cannibalism_active_node } }
				set_temp_variable = { cannibalism_new_node_type = constant:cannibalism_node_type.warlord }
				set_temp_variable = { cannibalism_new_node_strength = constant:cannibalism_opening.network_cell_strength }
				set_temp_variable = { cannibalism_new_node_stage = constant:cannibalism_state_stage.feeding_state }
				cannibalism_register_current_state_node = yes
			}
			else_if = {
				limit = { NOT = { is_in_array = { array = global.cannibalism_node_states value = THIS } } }
				add_to_array = { array = global.cannibalism_node_states value = THIS }
			}
			if = {
				limit = { NOT = { has_state_flag = cannibalism_warlord_state } }
				set_state_flag = cannibalism_warlord_state
				event_target:cannibalism_warlord_recruitment_actor = { add_to_variable = { cannibalism_warlord_slot_reference_count = 1 } }
			}
			set_variable = { cannibalism_warlord_slot_index = event_target:cannibalism_warlord_recruitment_actor.cannibalism_warlord_slot_index }
			set_variable = { cannibalism_warlord_slot_generation = event_target:cannibalism_warlord_recruitment_actor.cannibalism_warlord_slot_generation }
			set_variable = { cannibalism_node_source_country = event_target:cannibalism_warlord_recruitment_actor }
			set_variable = { cannibalism_node_source_country_generation = event_target:cannibalism_warlord_recruitment_actor.cannibalism_actor_generation }
		}
	}
}

cannibalism_prepare_warlord_recruitment_contract = {
	set_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_decision.unit_start_experience }
	set_temp_variable = { cannibalism_warlord_unit_start_equipment = constant:cannibalism_warlord_decision.unit_start_equipment }
	set_temp_variable = { cannibalism_warlord_unit_start_manpower = constant:cannibalism_warlord_decision.unit_start_manpower }
	set_temp_variable = { cannibalism_warlord_recruitment_cooldown_days = constant:cannibalism_warlord_decision.recruitment_state_cooldown_days }
	if = {
		limit = { has_country_flag = cannibalism_warlord_survival_order_open }
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_small }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_origin_defense_open }
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_small }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_capital_repairs_open }
		subtract_from_temp_variable = { cannibalism_warlord_recruitment_cooldown_days = constant:cannibalism_warlord_contract.recruitment_cooldown_reduction_small }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_lieutenant_assignments_open }
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_medium }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_hierarchy_routes_open }
		subtract_from_temp_variable = { cannibalism_warlord_recruitment_cooldown_days = constant:cannibalism_warlord_contract.recruitment_cooldown_reduction_small }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_officer_training_open }
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_large }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_distributed_recruitment_open }
		subtract_from_temp_variable = { cannibalism_warlord_recruitment_cooldown_days = constant:cannibalism_warlord_contract.recruitment_cooldown_reduction_large }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_workshop_conversion_open }
		subtract_from_temp_variable = { cannibalism_warlord_recruitment_cooldown_days = constant:cannibalism_warlord_contract.recruitment_cooldown_reduction_medium }
	}
	if = {
		limit = {
			has_country_flag = cannibalism_warlord_personal_guard_doctrine_open

		}
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_elite }
	}
	if = {
		limit = {
			has_country_flag = cannibalism_warlord_leader_protection_open
			check_variable = { cannibalism_warlord_recruitment_template = constant:cannibalism_recruitment_template.bone_guard }
		}
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_large }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_tyrant_doctrine_open }
		add_to_temp_variable = { cannibalism_warlord_unit_start_experience = constant:cannibalism_warlord_contract.recruit_experience_large }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_council_doctrine_open }

	}

cannibalism_execute_warlord_recruitment_transaction = {
	set_temp_variable = { cannibalism_warlord_recruitment_result = 0 }
	cannibalism_prepare_warlord_recruitment_contract = yes
	if = {
		limit = {
			has_event_target = cannibalism_warlord_recruitment_state
			check_variable = { cannibalism_warlord_recruitment_population_k > 0 }
			check_variable = { cannibalism_warlord_larder_cost > 0 }
		}
		save_event_target_as = cannibalism_warlord_recruitment_actor
		event_target:cannibalism_warlord_recruitment_state = { save_event_target_as = cannibalism_consumption_target_state }
		set_temp_variable = { cannibalism_consumption_exact_request_people = cannibalism_warlord_recruitment_population_k }
		multiply_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_consumption.population_people_per_k }
		set_temp_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.recruitment }
		cannibalism_prepare_consumption_context = yes
		if = {
			limit = {
				check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied }
				check_variable = {
					var = cannibalism_population_loss_applied
					value = cannibalism_consumption_exact_request_people
					compare = equals
				}
			}
			if = {
				limit = { is_cannibalism_warlord_country = yes }
				cannibalism_register_selected_state_to_current_warlord = yes
			}
			cannibalism_pay_current_warlord_larder_cost = yes
			set_temp_variable = { cannibalism_warlord_recruitment_manpower = cannibalism_population_loss_applied }
			multiply_temp_variable = { cannibalism_warlord_recruitment_manpower = constant:cannibalism_warlord_decision.recruitment_manpower_factor }
			round_temp_variable = cannibalism_warlord_recruitment_manpower
			add_manpower = cannibalism_warlord_recruitment_manpower
			if = {
				limit = { check_variable = { cannibalism_warlord_recruitment_template < constant:cannibalism_recruitment_template.emergency_reinforcement } }
				event_target:cannibalism_warlord_recruitment_state = { cannibalism_spawn_selected_warlord_recruitment_unit = yes }
				cannibalism_record_current_warlord_recruitment = yes
			}
			if = {
				limit = { has_country_flag = cannibalism_warlord_workshop_conversion_open }
				set_temp_variable = { cannibalism_warlord_recovered_infantry = constant:cannibalism_warlord_contract.workshop_recovery_infantry }
				add_equipment_to_stockpile = { type = infantry_equipment amount = cannibalism_warlord_recovered_infantry }
			}
			if = {
				limit = {
					has_country_flag = cannibalism_warlord_survival_order_open
					check_variable = { cannibalism_warlord_recruitment_template = constant:cannibalism_recruitment_template.emergency_reinforcement }
				}
				add_to_variable = { cannibalism_frenzy = constant:cannibalism_warlord_decision.frenzy_relief }
				add_to_variable = { cannibalism_command_integrity = constant:cannibalism_warlord_decision.command_integrity_gain }
			}
			if = {
				limit = {
					has_country_flag = cannibalism_warlord_purge_decisions_open
					check_variable = { cannibalism_warlord_recruitment_template = constant:cannibalism_recruitment_template.bone_guard }
				}
				add_to_variable = { cannibalism_frenzy = constant:cannibalism_warlord_contract.rapid_frenzy_gain }
				add_to_variable = { cannibalism_command_integrity = constant:cannibalism_warlord_contract.provincial_integrity_gain }
			}
			clamp_variable = { var = cannibalism_frenzy min = constant:cannibalism_consumption.frenzy_min max = constant:cannibalism_consumption.frenzy_max }
			clamp_variable = { var = cannibalism_command_integrity min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
			event_target:cannibalism_warlord_recruitment_state = {
				set_state_flag = { flag = cannibalism_recruitment_cooldown days = cannibalism_warlord_recruitment_cooldown_days }
			}
			set_temp_variable = { cannibalism_warlord_recruitment_result = 1 }
		}
		set_temp_variable = { cannibalism_consumption_exact_request_people = 0 }
	}
}

}

cannibalism_consume_selected_state_for_larder = {
	set_temp_variable = { cannibalism_warlord_consumption_result = 0 }
	if = {
		limit = { has_event_target = cannibalism_warlord_recruitment_state }
		save_event_target_as = cannibalism_warlord_recruitment_actor
		event_target:cannibalism_warlord_recruitment_state = { save_event_target_as = cannibalism_consumption_target_state }
		set_temp_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.feeding }
		cannibalism_prepare_consumption_context = yes
		if = {
			limit = { check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied } }
