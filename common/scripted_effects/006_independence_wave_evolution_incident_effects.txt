# ============================================================================
# CHAOS REDUX - EVENT 006 EVOLUTION INCIDENT EFFECTS
# ============================================================================
# Visible, paid resolutions for the five accepted evolution incident families.
# Each effect is called by one option of a triggered country event after the
# corresponding stage decision has consumed its material commitment. Country,
# former-host, network, and league ledgers are changed through their public
# transaction helpers; no option grants political power or free formations.
# ============================================================================

# Global input: temporary `independence_wave_incident_revisionist_delta`.
# This is separate from the focus helper because incident outcomes can move
# pressure in either direction and must use their own centralized tuning.
independence_wave_change_revisionist_pressure = {
	add_to_variable = { global.independence_wave_league_revisionist_pressure = independence_wave_incident_revisionist_delta }
	clamp_variable = {
		var = global.independence_wave_league_revisionist_pressure
		min = constant:independence_wave_league.minimum
		max = constant:independence_wave_league.maximum
	}
}
