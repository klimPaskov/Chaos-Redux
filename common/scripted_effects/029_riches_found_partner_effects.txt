# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND CONCESSION PARTNERS
# ============================================================================
# STATE scope unless noted; THIS is the mine. A signed contract gives the
# partner a bounded concession share (one country dynamic modifier no matter
# how many Event 029 contracts it holds) and gives both governments a positive
# opinion modifier. A buyout or compensated nationalization pays the partner
# its compensation; an uncompensated nationalization leaves the sponsor
# aggrieved, hostile and more likely to interfere with this mine for a while.
# The partner pointer is the mine's existing contract record; nothing here
# scans the world.
# ============================================================================

# STATE scope. Called by riches_found_begin_contract after the partner pointer
# is stored, and by riches_found_repoint_contract_partner_share when a resumed
# contract passes to a new controller. Side effects: partner share count and
# modifier, the stored share controller riches_found_contract_share_controller,
# state flag riches_found_contract_share_recorded, and one mutual opinion pair
# per partner and controller however many contracts the two share.
riches_found_record_contract_partner_share = {
	if = {
		limit = {
			NOT = { has_state_flag = riches_found_contract_share_recorded }
			has_variable = riches_found_contract_partner
			var:riches_found_contract_partner = { exists = yes }
			CONTROLLER = { exists = yes }
		}
		var:riches_found_contract_partner = { save_event_target_as = riches_found_share_partner }
		CONTROLLER = { save_event_target_as = riches_found_share_controller }
		# Read before this mine records its own share.
		if = {
			limit = { NOT = { riches_found_partner_pair_has_other_share = yes } }
			event_target:riches_found_share_partner = {
				add_opinion_modifier = { target = event_target:riches_found_share_controller modifier = riches_found_concession_partner }
			}
			event_target:riches_found_share_controller = {
				add_opinion_modifier = { target = event_target:riches_found_share_partner modifier = riches_found_concession_partner }
			}
		}
		set_state_flag = riches_found_contract_share_recorded
		set_variable = { riches_found_contract_share_controller = CONTROLLER }
		event_target:riches_found_share_partner = {
			add_to_variable = { riches_found_concession_share_count = constant:riches_found_value.one }
			# A partner holding concessions has a network that makes it a
			# stronger bidder for the next mine.
			set_country_flag = riches_found_foreign_contract_network
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = riches_found_country_concession_share } } }
				add_dynamic_modifier = { modifier = riches_found_country_concession_share }
			}
		}
	}
}

# STATE scope. Called by riches_found_clear_contract_state while the partner
# pointer still exists. The share modifier ends with the partner's last
# contract; the opinion pair ends with the last contract between the partner
# and the controller that holds the share, even after the mine changed hands.
riches_found_release_contract_partner_share = {
	if = {
		limit = { has_state_flag = riches_found_contract_share_recorded }
		clr_state_flag = riches_found_contract_share_recorded
		if = {
			limit = {
				has_variable = riches_found_contract_partner
				var:riches_found_contract_partner = { exists = yes }
			}
			var:riches_found_contract_partner = {
				add_to_variable = { riches_found_concession_share_count = constant:riches_found_value.negative_one }
				if = {
					limit = { NOT = { check_variable = { riches_found_concession_share_count > constant:riches_found_value.zero } } }
					clear_variable = riches_found_concession_share_count
					clr_country_flag = riches_found_foreign_contract_network
					if = {
						limit = { has_dynamic_modifier = { modifier = riches_found_country_concession_share } }
						remove_dynamic_modifier = { modifier = riches_found_country_concession_share }
					}
				}
			}
			if = {
				limit = {
					has_variable = riches_found_contract_share_controller
					var:riches_found_contract_share_controller = { exists = yes }
				}
				var:riches_found_contract_partner = { save_event_target_as = riches_found_share_partner }
				var:riches_found_contract_share_controller = { save_event_target_as = riches_found_share_controller }
				if = {
					limit = { NOT = { riches_found_partner_pair_has_other_share = yes } }
					event_target:riches_found_share_partner = { remove_opinion_modifier = { target = event_target:riches_found_share_controller modifier = riches_found_concession_partner } }
					event_target:riches_found_share_controller = { remove_opinion_modifier = { target = event_target:riches_found_share_partner modifier = riches_found_concession_partner } }
				}
			}
		}
		clear_variable = riches_found_contract_share_controller
	}
}

# STATE scope. Called when a suspended contract resumes after a control or
# ownership change. A share recorded for the previous controller moves to the
# current one, so the old pair's opinion ends and the new pair's begins.
riches_found_repoint_contract_partner_share = {
	if = {
		limit = {
			has_state_flag = riches_found_contract_share_recorded
			has_variable = riches_found_contract_share_controller
			check_variable = { var = riches_found_contract_share_controller value = CONTROLLER compare = not_equals }
		}
		riches_found_release_contract_partner_share = yes
	}
	riches_found_record_contract_partner_share = yes
}

# STATE scope. Buyout and compensated nationalization: the partner receives
# the compensation the controller paid and remembers a fair settlement. Call
# before riches_found_end_contract clears the partner pointer.
riches_found_compensate_contract_partner = {
	if = {
		limit = {
			has_variable = riches_found_contract_partner
			var:riches_found_contract_partner = { exists = yes }
			CONTROLLER = { exists = yes }
		}
		var:riches_found_contract_partner = { save_event_target_as = riches_found_share_partner }
		CONTROLLER = { save_event_target_as = riches_found_share_controller }
		event_target:riches_found_share_partner = {
			add_political_power = constant:riches_found_concession.compensation_political_power
			add_opinion_modifier = { target = event_target:riches_found_share_controller modifier = riches_found_concession_settled }
		}
	}
}

# STATE scope. Uncompensated nationalization: the sponsor turns hostile and
# is remembered on this mine as an aggrieved party whose interference is more
# likely until the grievance fades. Call before riches_found_end_contract.
riches_found_expropriate_contract_partner = {
	if = {
		limit = {
			has_variable = riches_found_contract_partner
			var:riches_found_contract_partner = { exists = yes }
			CONTROLLER = { exists = yes }
		}
		var:riches_found_contract_partner = { save_event_target_as = riches_found_share_partner }
		CONTROLLER = { save_event_target_as = riches_found_share_controller }
		event_target:riches_found_share_partner = {
			add_opinion_modifier = { target = event_target:riches_found_share_controller modifier = riches_found_concession_expropriated }
		}
		set_variable = { riches_found_aggrieved_sponsor = riches_found_contract_partner }
		set_variable = { riches_found_aggrieved_sponsor_expiry_date = global.date }
		set_temp_variable = { chaosx_date_days = constant:riches_found_concession.grievance_days }
		set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
		chaosx_convert_days_to_date_offset = yes
		add_to_variable = { riches_found_aggrieved_sponsor_expiry_date = chaosx_date_offset }
		set_state_flag = riches_found_sponsor_aggrieved
	}
}

# STATE scope. Clears a faded or vanished grievance. Called by the controller
# pulse through the contract lifecycle.
riches_found_refresh_aggrieved_sponsor = {
	if = {
		limit = {
			has_state_flag = riches_found_sponsor_aggrieved
			OR = {
				NOT = { has_variable = riches_found_aggrieved_sponsor }
				NOT = { var:riches_found_aggrieved_sponsor = { exists = yes } }
				check_variable = { var = global.date value = riches_found_aggrieved_sponsor_expiry_date compare = greater_than_or_equals }
			}
		}
		clr_state_flag = riches_found_sponsor_aggrieved
		clear_variable = riches_found_aggrieved_sponsor
		clear_variable = riches_found_aggrieved_sponsor_expiry_date
	}
}

# COUNTRY scope. Stores the selected mine's current or offered partner on the
# controller so decision tooltips can name it. No scoring runs here.
riches_found_refresh_partner_display = {
	clear_variable = riches_found_selected_foreign_partner
	if = {
		limit = { riches_found_country_has_selected_mine = yes }
		var:riches_found_selected_mine = {
			if = {
				limit = {
					has_variable = riches_found_contract_partner
					var:riches_found_contract_partner = { exists = yes }
				}
				PREV = { set_variable = { riches_found_selected_foreign_partner = PREV.riches_found_contract_partner } }
			}
			else_if = {
				limit = {
					has_state_flag = riches_found_foreign_target_available
					has_variable = riches_found_foreign_target
					var:riches_found_foreign_target = { exists = yes }
				}
				PREV = { set_variable = { riches_found_selected_foreign_partner = PREV.riches_found_foreign_target } }
			}
		}
	}
}
