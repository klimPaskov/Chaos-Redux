independence_wave_refresh_rhi_corridor_lifecycle = {
	if = {
		limit = { is_independence_wave_rhi_package = yes }
		remove_ideas = { rhi_divided_river_authority rhi_rhine_civic_industrial_compact }
		if = {
			limit = { has_stable_independence_wave_rhi_corridor = yes }
