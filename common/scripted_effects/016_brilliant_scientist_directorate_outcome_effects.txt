# COUNTRY scope. Selects one actual Prototype-or-higher family that has not
# already been damaged. The scan starts with the most consequential family.
brilliant_scientist_identify_damageable_relocation_project = {
	clear_variable = brilliant_scientist_directorate_relocation_selected_damage_family
	set_temp_variable = { brilliant_scientist_project_family = constant:brilliant_scientist_project_family.singularity }
	while = {
		limit = {
			check_variable = { brilliant_scientist_project_family > constant:brilliant_scientist_project_family.none }
			NOT = { has_variable = brilliant_scientist_directorate_relocation_selected_damage_family }
		}
		brilliant_scientist_load_project_index = yes
		if = {
			limit = {
				check_variable = { brilliant_scientist_project_stage_entries^brilliant_scientist_project_index > constant:brilliant_scientist_project_stage.theory }
				NOT = { brilliant_scientist_project_is_damaged = yes }
			}
			set_variable = { brilliant_scientist_directorate_relocation_selected_damage_family = brilliant_scientist_project_family }
		}
		subtract_from_temp_variable = { brilliant_scientist_project_family = constant:brilliant_scientist_directorate_value.one }
	}
}
