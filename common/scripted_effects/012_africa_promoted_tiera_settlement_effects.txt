africa_promoted_tiera_accept_league = {
	if = {
		limit = { africa_promoted_tiera_package_is_pan = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.protected }
		africa_apply_relationship_transition = yes
		set_country_flag = africa_promoted_pan_league_protected
	}
	if = {
		limit = { africa_promoted_tiera_package_is_gorilla_kingdom = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered }
		africa_apply_relationship_transition = yes
		set_country_flag = africa_promoted_gorilla_league_guardian
	}
	if = {
		limit = { africa_promoted_tiera_package_is_the_green = yes }
		if = { limit = { africa_member_can_enter_autonomous_federal = yes } set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.autonomous_federal } }
		else = { set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered } }
		africa_apply_relationship_transition = yes
		set_country_flag = africa_promoted_green_league_covenant
	}
	if = {
		limit = { africa_promoted_tiera_package_is_living_rivers = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.protected }
		africa_apply_relationship_transition = yes
		set_country_flag = africa_promoted_rivers_league_waterway
	}
	if = {
		limit = { africa_promoted_tiera_package_is_stoneborn = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.associate }
		africa_apply_relationship_transition = yes
		set_country_flag = africa_promoted_stoneborn_league_heritage
	}
	if = {
		limit = { africa_promoted_tiera_package_is_ancient_hosts = yes }
		set_temp_variable = { africa_requested_relationship_state = constant:africa_relationship.chartered }
		africa_apply_relationship_transition = yes
		set_country_flag = africa_promoted_ancient_league_relic
	}
	set_country_flag = africa_promoted_tiera_league_settled
	set_country_flag = africa_promoted_tiera_league_accepted
	set_country_flag = africa_member_allied_settlement_accepted
	if = {
		limit = {
			has_global_flag = africa_achievement_normal_launch
			has_country_flag = africa_registered_high_chaos_actor
		}
		africa_achievement_record_nonhuman_constitutional_member = yes
	}
	if = { limit = { has_event_target = africa_host } event_target:africa_host = { add_to_variable = { africa_charter_authority = constant:africa_value.one africa_consent_commitment = constant:africa_value.one } } }
}

africa_promoted_tiera_post_settlement = {
	if = {
		limit = { africa_promoted_tiera_package_is_pan = yes }
		set_country_flag = africa_promoted_pan_toolmaker_expansion
		add_army_experience = 8
		add_political_power = 10
	}
	if = {
		limit = { africa_promoted_tiera_package_is_gorilla_kingdom = yes }
		set_country_flag = africa_promoted_gorilla_sanctuary_guard
		add_war_support = 0.10
		add_stability = 0.05
	}
	if = {
		limit = { africa_promoted_tiera_package_is_the_green = yes }
		set_country_flag = africa_promoted_green_biome_restoration
		if = {
			limit = { has_global_flag = africa_achievement_normal_launch }
			set_global_flag = africa_achievement_ecological_covenant_owner_ready
		}
		add_stability = 0.10
		add_war_support = 0.03
	}
	if = {
		limit = { africa_promoted_tiera_package_is_living_rivers = yes }
		set_country_flag = africa_promoted_rivers_basin_works
		add_command_power = 10
		add_political_power = 5
	}
	if = {
		limit = { africa_promoted_tiera_package_is_stoneborn = yes }
		set_country_flag = africa_promoted_stoneborn_monument_restoration
		if = {
			limit = { has_global_flag = africa_achievement_normal_launch }
			set_temp_variable = { africa_achievement_milestone_id = constant:africa_achievement_milestone.stoneborn_constitutional_status }
			africa_achievement_record_milestone = yes
		}
		add_army_experience = 10
		add_stability = 0.05
	}
	if = {
		limit = { africa_promoted_tiera_package_is_ancient_hosts = yes }
		set_country_flag = africa_promoted_ancient_relic_network_complete
		add_war_support = 0.10
		add_political_power = 10
	}
	set_country_flag = africa_promoted_tiera_post_settlement_complete
}
