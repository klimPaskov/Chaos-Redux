# ============================================================================
# CHAOS REDUX - CAMP REPRESSION PLAYER ACTION DISPATCHER
# ============================================================================
# Overview:
# - Routes every player-started repression decision through one country bus.
# - Uses camp_rework_action_state_id for state actions; mission outcomes remain
#   in their owning decision files and never enter this dispatcher.
# - Adapts the existing colonial payloads without allowing AI selection to
#   overwrite a human player's persistent Repression Ledger selection.
# ============================================================================

camp_rework_prepare_selected_action_state = {
	if = {
		limit = {
			is_ai = no
			has_variable = camp_selected_state_id
		}
		set_variable = { camp_rework_action_state_id = camp_selected_state_id }
	}
}

camp_rework_dispatch_prepare_colonial_selection = {
	set_temp_variable = { camp_rework_had_ledger_selection = 0 }
	if = {
		limit = { has_variable = camp_selected_state_id }
		set_temp_variable = { camp_rework_had_ledger_selection = 1 }
		set_variable = { camp_rework_saved_ledger_state_id = camp_selected_state_id }
	}
	if = {
		limit = { has_variable = camp_rework_action_state_id }
		set_variable = { camp_selected_state_id = camp_rework_action_state_id }
	}
}

camp_rework_dispatch_restore_colonial_selection = {
	if = {
		limit = {
			check_variable = { camp_rework_had_ledger_selection > 0 }
			has_variable = camp_rework_saved_ledger_state_id
		}
		set_variable = { camp_selected_state_id = camp_rework_saved_ledger_state_id }
	}
	else = { clear_variable = camp_selected_state_id }
	clear_variable = camp_rework_saved_ledger_state_id
}

camp_rework_dispatch_generic_action = {
	clear_variable = camp_rework_exact_custody_action_request
	clear_variable = camp_rework_exact_custody_actor_token
	if = {
		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_activate_detention_network } }
		set_variable = { camp_rework_exact_custody_action_request = constant:migration_custody_action.internment }
		set_variable = { camp_rework_exact_custody_actor_token = camp_rework_action_id }
	}
	if = {
		limit = {
			has_variable = camp_rework_action_state_id
			has_variable = camp_rework_exact_custody_action_request
		}
		var:camp_rework_action_state_id = { camp_rework_prepare_exact_cohort_custody = yes }
	}
	if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_activate_detention_network }
			has_variable = camp_rework_action_state_id
			OR = {
				camp_rework_country_can_activate_generic = yes
				AND = {
					camp_rework_country_is_eligible = yes
					NOT = { has_country_flag = camp_rework_expansion_frozen }
					has_variable = camp_rework_exact_custody_action_request
					var:camp_rework_action_state_id = {
						check_variable = { camp_rework_exact_cohort_staged = constant:humanitarian_runtime.one }
					}
				}
			}
			var:camp_rework_action_state_id = {
				camp_rework_state_can_accept_responsibility_from_root = yes
				is_generic_player_selectable_camp_pool_state = yes
				NOT = { is_valid_camp_active_site_state = yes }
				NOT = { concentration_camp > 0 }
			}
		}
		camp_rework_activate_detention_in_action_state = yes
	}
	else_if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_inspect_active_site }
			camp_rework_country_uses_generic_kit = yes
			set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id }
			camp_rework_can_pay_site_inspect = yes
			camp_rework_country_can_open_site_inspection = yes
			has_variable = camp_rework_action_state_id
			var:camp_rework_action_state_id = { is_valid_camp_inspection_state = yes }
		}
		set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id }
		camp_rework_pay_site_inspect = yes
		camp_rework_apply_site_inspection_in_action_state = yes
	}
	else_if = {
		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_restricted_contaminated_site_escalation } has_variable = camp_rework_action_state_id }
		if = {
			limit = {
				camp_rework_country_can_apply_chemical_camp_method = yes
				var:camp_rework_action_state_id = { camp_rework_state_is_available_chemical_method_target = yes }
			}
			if = {
				limit = { set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id } camp_rework_can_pay_site_restricted = yes }
				set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id }
				camp_rework_pay_site_restricted = yes
				camp_rework_apply_chemical_escalation_in_action_state = yes
			}
		}
		else_if = {
			limit = {
				camp_rework_country_has_biological_escalation_capacity = yes
				var:camp_rework_action_state_id = { camp_rework_state_is_biological_escalation_target = yes }
			}
			if = {
				limit = { set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id } camp_rework_can_pay_site_restricted = yes }
				set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id }
				camp_rework_pay_site_restricted = yes
				camp_rework_apply_biological_escalation_in_action_state = yes
			}
		}
	}
	else_if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_destroy_evidence_before_retreat }
			has_variable = camp_rework_action_state_id
			set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id }
			camp_rework_can_pay_site_evidence = yes
			camp_rework_country_can_destroy_retreat_evidence = yes
			var:camp_rework_action_state_id = {
				camp_rework_state_is_actor_responsible_controlled = yes
				is_valid_camp_enemy_proximity_state = yes
			}
		}
		set_temp_variable = { camp_site_cost_state_id = camp_rework_action_state_id }
		camp_rework_pay_site_evidence = yes
		genocide_destroy_camp_evidence_in_action_state = yes
	}
	clear_variable = camp_rework_exact_custody_action_request
	clear_variable = camp_rework_exact_custody_actor_token
}

		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.japan_open_epidemic_containment_office } }
		camp_rework_consume_labor_project_resources = yes
		camp_rework_japan_open_containment_office = yes
	}
	else_if = {
		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.japan_destroy_pingfang_records } }
		camp_rework_consume_evidence_resources = yes
		camp_rework_japan_destroy_pingfang_records = yes
	}
	else_if = {
		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.japan_evacuate_pingfang_research_staff } }
		camp_rework_consume_labor_project_resources = yes
		camp_rework_japan_evacuate_pingfang_records = yes
	}
	else_if = { limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.japan_submit_to_army_review } } camp_rework_japan_submit_to_army_review = yes }
	else_if = { limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.japan_remove_ishii_from_program_control } } camp_rework_japan_remove_ishii_from_program_control = yes }
	else_if = {
		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.japan_shut_down_prisoner_experiments } camp_rework_japan_dismantlement_sites_ready = yes }
		camp_rework_consume_dismantlement_resources = yes
		camp_rework_japan_shutdown_prisoner_experiments = yes
	}
}

camp_rework_dispatch_soviet_action = {
	if = { limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.sov_release_prisoners_for_military_service } } camp_rework_soviet_release_prisoners_for_military_service = yes }
	else_if = { limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.sov_dismantle_overextended_gulags } } camp_rework_soviet_start_gulag_dismantlement = yes }
	else_if = {
		limit = { check_variable = { camp_rework_action_id = constant:camp_rework_action.sov_emergency_famine_relief } }
		camp_rework_soviet_emergency_famine_relief = yes
		if = {
			limit = {
				NOT = { has_country_flag = camp_rework_soviet_relief_order_accepted }
				check_variable = { camp_rework_native_pp_paid > 0 }
			}
			add_political_power = camp_rework_native_pp_paid
			set_temp_variable = { camp_rework_native_pp_paid = 0 }
		}
	}
	else_if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.sov_admit_local_administrative_collapse }
			has_variable = camp_rework_action_state_id
			var:camp_rework_action_state_id = {
				camp_rework_state_is_actor_responsible_controlled = yes
				has_state_flag = genocide_soviet_famine_pressure_site
				NOT = { has_state_flag = genocide_site_discovered }
			}
		}
		camp_rework_soviet_admit_administrative_collapse = yes
	}
}
