# ============================================================================
# CHAOS REDUX - CAMP REPRESSION PLAYER ACTION DISPATCHER
# ============================================================================
# Overview:
# - Routes every player-started repression decision through one country bus.
# - Uses camp_rework_action_state_id for state actions; mission outcomes remain
#   in their owning decision files and never enter this dispatcher.
# - Adapts the existing colonial payloads without allowing AI selection to
#   overwrite a human player's persistent Repression Ledger selection.
# ============================================================================

camp_rework_prepare_selected_action_state = {
	if = {
		limit = {
			is_ai = no
			has_variable = camp_selected_state_id
		}
		set_variable = { camp_rework_action_state_id = camp_selected_state_id }
	}
}

camp_rework_dispatch_prepare_colonial_selection = {
	set_temp_variable = { camp_rework_had_ledger_selection = 0 }
	if = {
		limit = { has_variable = camp_selected_state_id }
		set_temp_variable = { camp_rework_had_ledger_selection = 1 }
		set_variable = { camp_rework_saved_ledger_state_id = camp_selected_state_id }
	}
	if = {
		limit = { has_variable = camp_rework_action_state_id }
		set_variable = { camp_selected_state_id = camp_rework_action_state_id }
	}
}

camp_rework_dispatch_restore_colonial_selection = {
	if = {
		limit = {
			check_variable = { camp_rework_had_ledger_selection > 0 }
			has_variable = camp_rework_saved_ledger_state_id
		}
		set_variable = { camp_selected_state_id = camp_rework_saved_ledger_state_id }
	}
	else = { clear_variable = camp_selected_state_id }
	clear_variable = camp_rework_saved_ledger_state_id
}

camp_rework_dispatch_generic_action = {
	if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_activate_detention_network }
			has_variable = camp_rework_action_state_id
			camp_rework_country_can_activate_generic = yes
			var:camp_rework_action_state_id = {
				camp_rework_state_can_accept_responsibility_from_root = yes
				is_generic_player_selectable_camp_pool_state = yes
				NOT = { is_valid_camp_active_site_state = yes }
				NOT = { concentration_camp > 0 }
			}
		}
		camp_rework_activate_detention_in_action_state = yes
	}
	else_if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_expand_labor_quotas }
			has_variable = camp_rework_action_state_id
			var:camp_rework_action_state_id = {
				is_valid_camp_active_site_state = yes
				camp_rework_state_is_actor_responsible_controlled = yes
				NOT = { has_state_flag = camp_rework_expanded_labor_site }
				NOT = { extermination_camp > 0 }
				check_variable = { building_level@concentration_camp < constant:genocide_camp_level.maximum }
			}
		}
		camp_rework_expand_labor_in_action_state = yes
	}
	else_if = {
		limit = {
			check_variable = { camp_rework_action_id = constant:camp_rework_action.generic_redirect_labor_to_construction }

				camp_rework_state_is_actor_responsible_controlled = yes

			}
		}
		set_temp_variable = { camp_requested_project_type = constant:camp_rework_project_type.construction }
		camp_rework_start_generic_labor_project_in_action_state = yes

	}
	else_if = {

			camp_rework_country_can_open_site_inspection = yes

		if = {

			}
