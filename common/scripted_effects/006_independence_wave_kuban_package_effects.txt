independence_wave_refresh_kub_compact_lifecycle = {

		limit = { is_independence_wave_kub_package = yes }
		remove_ideas = { kub_fragmented_mounted_mandate kub_kuban_host_compact }

			limit = { has_stable_independence_wave_kub_compact = yes }
