# ============================================================================
# CHAOS REDUX - EVENT 040 LAWRENCE OF ARABIA ACTION EFFECTS
# ============================================================================
# Effects behind the target and British decisions, the missions and the
# incident and reaction events. Every paid action debits through the charge
# helpers below, so the cost row, the affordability trigger and the payment
# read the same quote.
#
# Scope: COUNTRY. Target actions run in the live target (ROOT). British
# actions run in ENG and reach the target through var:global.lawrence_target.
# Temporary variables carry inputs across scope changes inside one chain.
# ============================================================================

# ---------------------------------------------------------------------------
# Payment helpers. Input: temporary lawrence_charge (a positive amount).
# ---------------------------------------------------------------------------

lawrence_charge_political_power = {
	multiply_temp_variable = { lawrence_charge = -1 }
	add_political_power = lawrence_charge
}

lawrence_charge_command_power = {
	multiply_temp_variable = { lawrence_charge = -1 }
	add_command_power = lawrence_charge
}

lawrence_charge_army_experience = {
	multiply_temp_variable = { lawrence_charge = -1 }
	add_army_experience = lawrence_charge
}

lawrence_charge_support_equipment = {
	multiply_temp_variable = { lawrence_charge = -1 }
	add_equipment_to_stockpile = { type = support_equipment amount = lawrence_charge }
}

lawrence_charge_infantry_equipment = {
	multiply_temp_variable = { lawrence_charge = -1 }
	add_equipment_to_stockpile = { type = infantry_equipment amount = lawrence_charge }
}

	multiply_temp_variable = { lawrence_charge = -1 }
	add_equipment_to_stockpile = { type = convoy_1 amount = lawrence_charge }
}

	multiply_temp_variable = { lawrence_charge = -1 }
	add_equipment_to_stockpile = { type = train_equipment amount = lawrence_charge }
}

lawrence_charge_fuel = {
	multiply_temp_variable = { lawrence_charge = -1 }
	add_fuel = lawrence_charge
}

# Bounded bookkeeping

# Input: temporary lawrence_evidence_change (signed). Evidence stays 0 to 4.
lawrence_add_evidence = {
	add_to_variable = { lawrence_evidence = lawrence_evidence_change }
	clamp_variable = { var = lawrence_evidence min = constant:lawrence_event.zero max = constant:lawrence_evidence.maximum }
}

# Input: temporary lawrence_exposure_change (signed). Exposure stays 0 to 100.
lawrence_add_exposure = {
	add_to_variable = { lawrence_exposure = lawrence_exposure_change }
	clamp_variable = { var = lawrence_exposure min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
}

# Input: temporary lawrence_officers_change (signed). Officers stay 0 to 100.
lawrence_add_officers = {
	add_to_variable = { lawrence_officers = lawrence_officers_change }
	clamp_variable = { var = lawrence_officers min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
}

# Global hidden values. Input: temporary lawrence_trust_change or
# lawrence_credibility_change (signed).
lawrence_add_trust = {
	add_to_variable = { global.lawrence_trust = lawrence_trust_change }
	clamp_variable = { var = global.lawrence_trust min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
}

lawrence_add_credibility = {
	add_to_variable = { global.lawrence_credibility = lawrence_credibility_change }
	clamp_variable = { var = global.lawrence_credibility min = constant:lawrence_event.zero max = constant:lawrence_influence.maximum }
}

# A mission started or ended. Missions stay at most two per target.
lawrence_mission_started = {
	add_to_variable = { lawrence_missions_active = constant:lawrence_event.one }
}

lawrence_mission_closed = {
	subtract_from_variable = { lawrence_missions_active = constant:lawrence_event.one }
	clamp_variable = { var = lawrence_missions_active min = constant:lawrence_event.zero max = constant:lawrence_mission.max_active }
}

# A mission was completed in the target's favour.
lawrence_mission_succeeded = {
	lawrence_mission_closed = yes
	add_to_variable = { lawrence_missions_done = constant:lawrence_event.one }
}

# Terms of access (chaosx.nr40.4 options)
# ---------------------------------------------------------------------------

# COUNTRY, the target. Input: temporary lawrence_terms_choice.
lawrence_apply_terms = {
	set_variable = { lawrence_terms = lawrence_terms_choice }
	set_variable = { lawrence_phase = constant:lawrence_phase.contest }
	if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.broad compare = equals } }
		set_temp_variable = { lawrence_change = constant:lawrence_terms_effect.broad_influence }
		lawrence_add_influence = yes
		set_temp_variable = { lawrence_trust_change = constant:lawrence_terms_effect.broad_trust }
		lawrence_add_trust = yes
		set_temp_variable = { lawrence_exposure_change = constant:lawrence_terms_effect.broad_exposure }
		lawrence_add_exposure = yes
		set_variable = { lawrence_support_power = constant:lawrence_support.opening_power }
		set_variable = { lawrence_support_until = global.num_days }
		add_to_variable = { lawrence_support_until = constant:lawrence_support.opening_days }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.liaison compare = equals } }
		set_variable = { lawrence_support_power = constant:lawrence_support.opening_power }
		set_variable = { lawrence_support_until = global.num_days }
		add_to_variable = { lawrence_support_until = constant:lawrence_support.opening_days_liaison }
	}
	else_if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.escorted compare = equals } }
		set_country_flag = lawrence_escorted
		set_temp_variable = { lawrence_change = constant:lawrence_terms_effect.escort_influence }
		lawrence_add_influence = yes
		set_temp_variable = { lawrence_trust_change = constant:lawrence_terms_effect.escort_trust }
		lawrence_add_trust = yes
		set_temp_variable = { lawrence_exposure_change = constant:lawrence_terms_effect.escort_exposure }
		lawrence_add_exposure = yes
	}
	else_if = {
		limit = { check_variable = { var = lawrence_terms value = constant:lawrence_terms.refused compare = equals } }
		set_temp_variable = { lawrence_settle_outcome = constant:lawrence_outcome.expelled }
		set_temp_variable = { lawrence_settle_clean = constant:lawrence_event.one }
		lawrence_settle = yes
	}
}

# Weekly: the ledger and dismantle missions count the days the named nodes
# stay garrisoned, and the securing mission moves officers only while the
# capital is held.
lawrence_track_mission_holds = {
	if = {
		limit = { has_active_mission = lawrence_trace_ledger_mission }
		if = {
			limit = {
				lawrence_capital_garrisoned = yes
				lawrence_port_garrisoned = yes
			}
			add_to_variable = { lawrence_ledger_hold_days = constant:lawrence_event.pulse_days }
		}
		else = {
			set_variable = { lawrence_ledger_hold_days = constant:lawrence_event.zero }
		}
	}
	if = {
		limit = { has_active_mission = lawrence_dismantle_mission }
		if = {
			limit = { lawrence_all_nodes_garrisoned = yes }
			add_to_variable = { lawrence_dismantle_hold_days = constant:lawrence_event.pulse_days }
		}
		else = {
			set_variable = { lawrence_dismantle_hold_days = constant:lawrence_event.zero }
		}
	}
	lawrence_track_officer_securing = yes
}

# COUNTRY, the target. Inputs: temporary lawrence_ship_infantry and
# lawrence_ship_support, already paid by the sender.
lawrence_queue_shipment = {
	add_to_variable = { lawrence_delivery_serial = constant:lawrence_event.one }
	set_variable = { lawrence_delivery_infantry = lawrence_ship_infantry }
	set_variable = { lawrence_delivery_support = lawrence_ship_support }
	set_country_flag = lawrence_delivery_pending
	clr_country_flag = lawrence_delivery_custody
	clr_country_flag = lawrence_delivery_intercepted
	set_temp_variable = { lawrence_travel = constant:lawrence_delivery.base_days }
	if = {
		limit = { has_country_flag = lawrence_route_sea }
		subtract_from_temp_variable = { lawrence_travel = constant:lawrence_delivery.sea_route_days }
	}
	if = {
		limit = { has_country_flag = lawrence_rail_repaired }
		subtract_from_temp_variable = { lawrence_travel = constant:lawrence_delivery.rail_days }
	}
	if = {
		limit = { has_country_flag = lawrence_bases_agreed }
		subtract_from_temp_variable = { lawrence_travel = constant:lawrence_delivery.bases_days }
	}
	if = {
		limit = {
			NOT = { any_owned_state = { is_coastal = yes } }
			NOT = { any_neighbor_country = { OR = { tag = ENG is_subject_of = ENG } } }
		}
		add_to_temp_variable = { lawrence_travel = constant:lawrence_delivery.remote_days }
	}
	if = {
		limit = { NOT = { has_country_flag = lawrence_route_open } }
		add_to_temp_variable = { lawrence_travel = constant:lawrence_delivery.closed_route_days }
	}
	clamp_temp_variable = { var = lawrence_travel min = constant:lawrence_delivery.minimum_days max = constant:lawrence_delivery.maximum_days }
	set_variable = { lawrence_delivery_arrival_day = global.num_days }
	add_to_variable = { lawrence_delivery_arrival_day = lawrence_travel }
	# A shipment crossing a country with a strong security service and no
	# secured route can be seized.
	if = {
		limit = {
			NOT = { has_country_flag = lawrence_route_sea }
			NOT = { has_country_flag = lawrence_rail_repaired }
			check_variable = { var = lawrence_security value = constant:lawrence_delivery.seizure_security compare = greater_than_or_equals }
		}
		random_list = {
			35 = {
				modifier = { factor = 0.4 has_country_flag = lawrence_bases_agreed }
				modifier = { factor = 1.4 has_country_flag = lawrence_nodes_secured }
				set_country_flag = lawrence_delivery_intercepted
			}
			65 = { }
		}
	}
	set_temp_variable = { lawrence_report_delay = constant:lawrence_event.report_delay_days }
	country_event = { id = chaosx.nr40.12 days = lawrence_report_delay }
}
