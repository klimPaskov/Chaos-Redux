# ============================================================================
# CHAOS REDUX - SHARED OUTBREAK-MEASURE EFFECTS
# ============================================================================
# Overview:
# - Preserves the named national hygiene, migration, quarantine, border, and
#   emergency-law controls shared with the weaponized-zombie system.
# - Recalculates zombie outbreak chance immediately.
# - Ordinary biological lifecycle ticks read the resulting flags and ideas
#   directly where accepted; no all-state modifier refresh or death pass is
#   performed here.
# ============================================================================

apply_shared_hygiene_measures = {
	if = {
		limit = { NOT = { has_country_flag = hygiene_measures_enforced } }
		set_country_flag = hygiene_measures_enforced
	}
	if = {
		limit = { NOT = { has_idea = outbreak_prevention_hygiene_measures } }
		add_ideas = outbreak_prevention_hygiene_measures
	}
	calculate_zombie_outbreak_chance = yes
}

revoke_shared_hygiene_measures = {
	if = {
		limit = { has_country_flag = hygiene_measures_enforced }
		clr_country_flag = hygiene_measures_enforced
	}
	if = {
		limit = { has_idea = outbreak_prevention_hygiene_measures }
		remove_ideas = outbreak_prevention_hygiene_measures
	}
	calculate_zombie_outbreak_chance = yes
}
