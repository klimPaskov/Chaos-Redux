# ============================================================================
# CHAOS REDUX - EVENT 006 DYNAMIC STARTING FORCE EFFECTS
# ============================================================================
# Overview:
# - Snapshots every accepted starting-force input after synchronized release.
# - Builds a bounded budget and tier-specific division, equipment, experience,
#   logistics, officer, and military-influence outputs.
# - Defines one researched profile template and creates the opening force once
#   per Event 006 generation at the frozen anchor, then stamps each newly
#   created division with package/generation provenance for exact operations.
# - Transfers only explicitly approved small naval or air fractions and never
#   transfers host land units or general stockpiles.
# - Records all inputs and outputs for Event Details, balancing, and audits.
# ============================================================================

independence_wave_collect_force_inputs = {
	set_temp_variable = { independence_wave_force_population_k = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_owned_states = num_owned_states }
	set_temp_variable = { independence_wave_force_factory_count = num_of_factories }
	set_temp_variable = { independence_wave_force_military_factory_count = num_of_military_factories }
	set_temp_variable = { independence_wave_force_naval_factory_count = num_of_naval_factories }
	set_temp_variable = { independence_wave_force_infrastructure_score = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_railway_states = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_port_levels = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_supply_nodes = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_minimum_railway_level = constant:independence_wave_force_budget.minimum_railway_level }
	every_owned_state = {
		add_to_temp_variable = { independence_wave_force_population_k = state_population_k }
		add_to_temp_variable = { independence_wave_force_infrastructure_score = infrastructure_level }
		add_to_temp_variable = { independence_wave_force_port_levels = building_level@naval_base }
		add_to_temp_variable = { independence_wave_force_supply_nodes = num_of_supply_nodes }
		if = {
			limit = {
				meta_trigger = {
					text = {
						has_railway_level = {
							level = [IW_RAIL_LEVEL]
							state = THIS
						}
					}
					IW_RAIL_LEVEL = "[?independence_wave_force_minimum_railway_level|.0]"
				}
			}
			add_to_temp_variable = { independence_wave_force_railway_states = constant:independence_wave_plan.loop_increment }
		}
	}

	set_temp_variable = { independence_wave_force_host_divisions = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_host_surrender = constant:independence_wave_force_budget.minimum }
	set_temp_variable = { independence_wave_force_host_at_war = constant:independence_wave_planner.no_candidates }
	event_target:independence_wave_setup_former_host = {
		set_temp_variable = { independence_wave_force_host_divisions = num_divisions }
		set_temp_variable = { independence_wave_force_host_surrender = surrender_progress }
		if = {
			limit = { has_war = yes }
			set_temp_variable = { independence_wave_force_host_at_war = constant:independence_wave_planner.selected }
		}
	}

	set_temp_variable = { independence_wave_force_chaos_band = independence_wave_origin_chaos_band }
	set_temp_variable = { independence_wave_force_legitimacy_input = independence_wave_legitimacy }
	set_temp_variable = { independence_wave_force_tradition_input = independence_wave_military_tradition }
	set_temp_variable = { independence_wave_force_patron_input = independence_wave_strongest_patron_influence }
	set_temp_variable = { independence_wave_force_network_input = independence_wave_network_standing }

	set_variable = { independence_wave_opening_population_k = independence_wave_force_population_k }
	set_variable = { independence_wave_opening_factory_count = independence_wave_force_factory_count }
	set_variable = { independence_wave_opening_infrastructure_score = independence_wave_force_infrastructure_score }
	set_variable = { independence_wave_opening_railway_state_count = independence_wave_force_railway_states }
	set_variable = { independence_wave_opening_port_level_count = independence_wave_force_port_levels }
	set_variable = { independence_wave_opening_supply_node_count = independence_wave_force_supply_nodes }
	set_variable = { independence_wave_opening_host_division_count = independence_wave_force_host_divisions }
	set_variable = { independence_wave_opening_host_surrender_progress = independence_wave_force_host_surrender }
	clr_country_flag = independence_wave_opening_host_was_at_war
	if = {
		limit = { check_variable = { independence_wave_force_host_at_war = constant:independence_wave_planner.selected } }
		set_country_flag = independence_wave_opening_host_was_at_war
	}
	set_variable = { independence_wave_opening_patron_support = independence_wave_force_patron_input }
	set_variable = { independence_wave_opening_network_support = independence_wave_force_network_input }
}

independence_wave_calculate_force_budget = {
	set_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_tier.fragile_budget }
	if = {
		limit = { check_variable = { independence_wave_force_level = constant:liberation_force_level.viable } }
		set_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_tier.viable_budget }
	}
	else_if = {
		limit = { check_variable = { independence_wave_force_level = constant:liberation_force_level.armed } }
		set_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_tier.armed_budget }
	}
	else_if = {
		limit = { check_variable = { independence_wave_force_level = constant:liberation_force_level.high_chaos } }
		set_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_tier.high_chaos_budget }
	}

	set_temp_variable = { independence_wave_force_population_component = independence_wave_force_population_k }
	divide_temp_variable = { independence_wave_force_population_component = constant:independence_wave_force_budget.population_divisor_k }
	clamp_temp_variable = {
		var = independence_wave_force_population_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.population_component_maximum
	}

	set_temp_variable = { independence_wave_force_factory_component = independence_wave_force_factory_count }
	multiply_temp_variable = { independence_wave_force_factory_component = constant:independence_wave_force_budget.factory_multiplier }
	clamp_temp_variable = {
		var = independence_wave_force_factory_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.factory_component_maximum
	}

	set_temp_variable = { independence_wave_force_infrastructure_component = independence_wave_force_infrastructure_score }
	divide_temp_variable = { independence_wave_force_infrastructure_component = constant:independence_wave_force_budget.infrastructure_divisor }
	clamp_temp_variable = {
		var = independence_wave_force_infrastructure_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.infrastructure_component_maximum
	}

	set_temp_variable = { independence_wave_force_transport_component = independence_wave_force_railway_states }
	multiply_temp_variable = { independence_wave_force_transport_component = constant:independence_wave_force_budget.railway_state_multiplier }
	set_temp_variable = { independence_wave_force_port_component = independence_wave_force_port_levels }
	multiply_temp_variable = { independence_wave_force_port_component = constant:independence_wave_force_budget.port_level_multiplier }
	add_to_temp_variable = { independence_wave_force_transport_component = independence_wave_force_port_component }
	set_temp_variable = { independence_wave_force_supply_component = independence_wave_force_supply_nodes }
	multiply_temp_variable = { independence_wave_force_supply_component = constant:independence_wave_force_budget.supply_node_multiplier }
	add_to_temp_variable = { independence_wave_force_transport_component = independence_wave_force_supply_component }
	clamp_temp_variable = {
		var = independence_wave_force_transport_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.transport_component_maximum
	}

	set_temp_variable = { independence_wave_force_host_force_component = independence_wave_force_host_divisions }
	divide_temp_variable = { independence_wave_force_host_force_component = constant:independence_wave_force_budget.host_division_divisor }
	clamp_temp_variable = {
		var = independence_wave_force_host_force_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.host_force_component_maximum
	}

	set_temp_variable = { independence_wave_force_host_weakness_component = independence_wave_force_host_surrender }
	multiply_temp_variable = { independence_wave_force_host_weakness_component = constant:independence_wave_force_budget.host_surrender_multiplier }
	if = {
		limit = { check_variable = { independence_wave_force_host_at_war = constant:independence_wave_planner.selected } }
		add_to_temp_variable = { independence_wave_force_host_weakness_component = constant:independence_wave_force_budget.host_war_bonus }
	}
	if = {
		limit = { independence_wave_force_host_divisions < constant:independence_wave_force_budget.host_weak_army_division_ceiling }
		add_to_temp_variable = { independence_wave_force_host_weakness_component = constant:independence_wave_force_budget.host_weak_army_bonus }
	}
	clamp_temp_variable = {
		var = independence_wave_force_host_weakness_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.host_weakness_component_maximum
	}

	set_temp_variable = { independence_wave_force_chaos_component = independence_wave_force_chaos_band }
	multiply_temp_variable = { independence_wave_force_chaos_component = constant:independence_wave_force_budget.chaos_band_multiplier }
	set_temp_variable = { independence_wave_force_legitimacy_component = independence_wave_force_legitimacy_input }
	divide_temp_variable = { independence_wave_force_legitimacy_component = constant:independence_wave_force_budget.legitimacy_divisor }
	set_temp_variable = { independence_wave_force_tradition_component = independence_wave_force_tradition_input }
	divide_temp_variable = { independence_wave_force_tradition_component = constant:independence_wave_force_budget.military_tradition_divisor }
	set_temp_variable = { independence_wave_force_patron_component = independence_wave_force_patron_input }
	divide_temp_variable = { independence_wave_force_patron_component = constant:independence_wave_force_budget.patron_support_divisor }
	set_temp_variable = { independence_wave_force_network_component = independence_wave_force_network_input }
	divide_temp_variable = { independence_wave_force_network_component = constant:independence_wave_force_budget.network_support_divisor }
	clamp_temp_variable = {
		var = independence_wave_force_network_component
		min = constant:independence_wave_force_budget.network_component_minimum
		max = constant:independence_wave_force_budget.network_component_maximum
	}
	set_temp_variable = { independence_wave_force_territory_component = independence_wave_force_owned_states }
	multiply_temp_variable = { independence_wave_force_territory_component = constant:independence_wave_force_budget.territory_state_multiplier }
	clamp_temp_variable = {
		var = independence_wave_force_territory_component
		min = constant:independence_wave_force_budget.minimum
		max = constant:independence_wave_force_budget.territory_component_maximum
	}

	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_population_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_factory_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_infrastructure_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_transport_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_host_force_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_host_weakness_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_chaos_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_legitimacy_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_tradition_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_patron_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_network_component }
	add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_force_territory_component }

	if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.industrial_security } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.industrial_security } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.regular_defectors } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.regular_defectors } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.mountain_frontier } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.mountain_frontier } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.coastal_maritime } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.coastal_maritime } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.mounted_mobile } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.mounted_mobile } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.desert_nomadic } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.desert_nomadic } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.river_jungle } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.river_jungle } }
	else_if = { limit = { check_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.foreign_volunteers } } add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_force_profile_budget.foreign_volunteers } }
	if = {
		limit = { has_country_flag = independence_wave_armed_birth_force_authorized }
		add_to_temp_variable = { independence_wave_force_budget_score = constant:independence_wave_evolution_force.armed_birth_budget_bonus }
	}
	if = {
		limit = {
			has_country_flag = independence_wave_sponsored_release
			has_variable = independence_wave_sponsorship_opening_strength
			check_variable = {
				var = independence_wave_sponsorship_opening_strength
				value = constant:independence_wave_value.minimum
				compare = greater_than
			}
		}
		add_to_temp_variable = { independence_wave_force_budget_score = independence_wave_sponsorship_opening_strength }
	}

	clamp_temp_variable = { var = independence_wave_force_budget_score min = constant:independence_wave_force_budget.minimum }
	set_variable = { independence_wave_opening_force_budget = independence_wave_force_budget_score }
}

independence_wave_create_opening_divisions = {
	# Snapshot division scopes before materialisation. `create_unit` exposes no
	# division-scoped payload, so the post-create scope delta is the only safe
	# way to stamp the supplied formations without accepting a matching
	# player-built division later.
	clear_temp_array = independence_wave_precreate_division_scopes
	every_country_division = {
		add_to_temp_array = { array = independence_wave_precreate_division_scopes value = THIS }
	}
	meta_effect = {
		text = {
			event_target:independence_wave_setup_anchor_state = {
				create_unit = {
					division = "division_template = \"[IW_TEMPLATE]\" start_experience_factor = [IW_EXPERIENCE] start_equipment_factor = [IW_EQUIPMENT] start_manpower_factor = [IW_MANPOWER]"
					owner = ROOT
					allow_spawning_on_enemy_provs = no
					count = [IW_COUNT]
				}
			}
		}
		IW_TEMPLATE = "[This.GetIndependenceWaveForceTemplateName]"
		IW_EXPERIENCE = "[?independence_wave_force_experience_factor|.2]"
		IW_EQUIPMENT = "[?independence_wave_force_equipment_factor|.2]"
		IW_MANPOWER = "[?independence_wave_force_manpower_factor|.2]"
		IW_COUNT = "[?independence_wave_force_division_count|.0]"
	}
	# Every division outside the pre-create scope array was materialised by the
	# opening-force allocator in this call. Persist both immutable package and
	# generation receipts on the division itself; country-level receipts remain
	# only an admission summary and are not treated as unit identity.
	every_country_division = {
		limit = { NOT = { is_in_array = { array = independence_wave_precreate_division_scopes value = THIS } } }
		set_variable = { independence_wave_force_division_package_id = independence_wave_setup_package_id }
		set_variable = { independence_wave_force_division_generation_id = ROOT.independence_wave_generation_id }
	}
	clear_temp_array = independence_wave_precreate_division_scopes
}
