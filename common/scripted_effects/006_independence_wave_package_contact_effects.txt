# ============================================================================
# CHAOS REDUX - EVENT 006 PACKAGE CONTACT REGISTRY
# ============================================================================
# Overview:
# - Every admitted Event 006 package registers two or three researched foreign
#   contacts when its origin commits. Each contact carries a short recorded
#   basis in the comment beside it and in
#   docs/events/006_independence_wave/package_contacts.md.
# - The contacts are a bounded actor source for the foreign reaction queue and
#   count as the recorded-relationship family of the foreign score. A contact
#   that is the country's own former host is excluded by the actor check.
# - `independence_wave_clear_package_contacts` runs in the generation reset and
#   at origin end.
# No effect scans the world.
# ============================================================================

# COUNTRY scope: the released country. PREV: the contact country.
independence_wave_add_prev_as_package_contact = {
	if = {
		limit = { NOT = { is_in_array = { array = independence_wave_package_contact_entries value = PREV } } }
		add_to_array = { array = independence_wave_package_contact_entries value = PREV }
	}
}

# COUNTRY scope: the released country.
independence_wave_clear_package_contacts = {
	clear_array = independence_wave_package_contact_entries
}
