# Overview:
# - Count distinct committed Event 006 state transfers per former host.
# - Use the exact host owned-state snapshot as the loss denominator.

						set_variable = { independence_wave_host_response_original_capital = independence_wave_host_response_validated_original_capitals^independence_wave_host_response_stage_index }

# COUNTRY scope: host .63 calls this before selecting an action effect.

		while_loop_effect = {
			limit = { check_variable = { var = independence_wave_host_response_country_index value = global.independence_wave_plan_selected_count compare = less_than } }
			set_temp_variable = { independence_wave_host_response_country_row = global.independence_wave_plan_country_row_indices^independence_wave_host_response_country_index }
			if = {
				limit = {
					check_variable = { var = independence_wave_host_response_country_row value = constant:independence_wave_plan.first_array_index compare = greater_than_or_equals }
					check_variable = { var = independence_wave_host_response_country_row value = global.liberation_plan_countries^num compare = less_than }
					check_variable = { var = global.liberation_plan_country_owners^independence_wave_host_response_country_row value = constant:liberation_plan_owner.independence_wave compare = equals }
				}
				add_to_temp_variable = { independence_wave_host_response_valid_country_rows = constant:independence_wave_host_response_value.one }
			}
			if = {
				limit = {
					check_variable = { var = independence_wave_host_response_country_row value = constant:independence_wave_plan.first_array_index compare = greater_than_or_equals }
					check_variable = { var = independence_wave_host_response_country_row value = global.liberation_plan_countries^num compare = less_than }
					check_variable = { var = global.liberation_plan_country_owners^independence_wave_host_response_country_row value = constant:liberation_plan_owner.independence_wave compare = equals }
					var:global.liberation_plan_primary_hosts^independence_wave_host_response_country_row = { tag = event_target:independence_wave_host_response_ai_host }
				}
				var:global.liberation_plan_countries^independence_wave_host_response_country_row = {
					if = {
						limit = {
							is_independence_wave_active_country = yes
							has_country_flag = independence_wave_origin_committed
							has_variable = independence_wave_generation_id
							check_variable = { independence_wave_generation_id > constant:independence_wave_host_response_value.zero }
