@NATURAL_DISASTER_WARNING_DAYS = 45
@NATURAL_DISASTER_RECENT_TARGET_DAYS = 90
@NATURAL_DISASTER_RECENT_FAMILY_DAYS = 180
@NATURAL_DISASTER_AFTERMATH_LOCAL_DAYS = 90

@NATURAL_DISASTER_AFTERMATH_REGIONAL_DAYS = 150

@NATURAL_DISASTER_AFTERMATH_ABNORMAL_DAYS = 210
@NATURAL_DISASTER_AFTERMATH_SCENARIO_MAXIMUM_DAYS = 240

@NATURAL_DISASTER_WEIGHT_STORM = 12

@NATURAL_DISASTER_WEIGHT_WILDFIRE = 7
@NATURAL_DISASTER_WEIGHT_LANDSLIDE = 7

@NATURAL_DISASTER_WEIGHT_METEOR = 1

natural_disaster_prepare_launch_context = {
	natural_disaster_select_stage = yes
	if = {
		limit = { NOT = { has_variable = natural_disaster_severity } }
		set_temp_variable = { natural_disaster_severity = constant:natural_disaster_severity.local }
	}
	if = {
		limit = {
			check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_varied_local }
			check_variable = { natural_disaster_severity < constant:natural_disaster_severity.serious }
		}
		set_temp_variable = { natural_disaster_severity = constant:natural_disaster_severity.serious }
	}
	if = {
		limit = {
			check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_regional_systems }
			check_variable = { natural_disaster_severity < constant:natural_disaster_severity.regional }
		}
		set_temp_variable = { natural_disaster_severity = constant:natural_disaster_severity.regional }
	}
	if = {
		limit = {
			check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_chained_aftermaths }
			check_variable = { natural_disaster_severity < constant:natural_disaster_severity.cascading }
		}
		set_temp_variable = { natural_disaster_severity = constant:natural_disaster_severity.cascading }
	}
	if = {
		limit = {
			check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_abnormal_age }
			check_variable = { natural_disaster_severity < constant:natural_disaster_severity.abnormal }
		}
		set_temp_variable = { natural_disaster_severity = constant:natural_disaster_severity.abnormal }
	}
	if = {
		limit = { NOT = { has_variable = natural_disaster_family } }
		natural_disaster_select_family_for_stage = yes
	}
	set_temp_variable = { natural_disaster_phase = constant:natural_disaster_phase.impact }
}

natural_disaster_prepare_random_event_context = {
	natural_disaster_prepare_launch_context = yes
	natural_disaster_select_target_state = yes
	if = {
		limit = { has_event_target = natural_disaster_current_target_state }
		natural_disaster_roll_warning_chance = yes
	}
}

natural_disaster_select_stage = {
	set_temp_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_baseline }
	if = {
		limit = {
			OR = {
				has_global_flag = { flag = chaos_tier value = 4 }
				has_global_flag = { flag = chaos_tier value = 5 }
			}
		}
		set_temp_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_abnormal_age }
	}
	else_if = {
		limit = { has_global_flag = { flag = chaos_tier value = 3 } }
		set_temp_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_chained_aftermaths }
	}
	else_if = {
		limit = { has_global_flag = { flag = chaos_tier value = 2 } }
		set_temp_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_regional_systems }
	}
	else_if = {
		limit = { has_global_flag = { flag = chaos_tier value = 1 } }
		set_temp_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_varied_local }
	}
}

natural_disaster_select_family_for_stage = {
	random_list = {
		@NATURAL_DISASTER_WEIGHT_EARTHQUAKE = {
			modifier = {
				factor = 0
				NOT = { any_controlled_state = { natural_disaster_is_valid_base_target_state = yes } }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.earthquake }
		}
		@NATURAL_DISASTER_WEIGHT_FLOOD = {
			modifier = {
				factor = 0
				NOT = {
					any_controlled_state = {
						OR = {
							is_coastal = yes
							infrastructure > 0
							rail_way > 0
							supply_node > 0
						}
					}
				}
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.flood }
		}
		@NATURAL_DISASTER_WEIGHT_STORM = {
			modifier = {
				factor = 0
				NOT = {
					any_controlled_state = {
						OR = {
							is_coastal = yes
							naval_base > 0
							air_base > 0
							infrastructure > 0
						}
					}
				}
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.storm }
		}
		@NATURAL_DISASTER_WEIGHT_DROUGHT = {
			modifier = {
				factor = 0
				NOT = { any_controlled_state = { natural_disaster_is_dry_candidate_state = yes } }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.drought }
		}
		@NATURAL_DISASTER_WEIGHT_WILDFIRE = {
			modifier = {
				factor = 0
				check_variable = { natural_disaster_stage < constant:natural_disaster_event.stage_varied_local }
			}
			modifier = {
				factor = 0
				NOT = { any_controlled_state = { natural_disaster_is_wildfire_candidate_state = yes } }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.wildfire }
		}
		@NATURAL_DISASTER_WEIGHT_LANDSLIDE = {
			modifier = {
				factor = 0
				NOT = { any_controlled_state = { natural_disaster_is_mountain_candidate_state = yes } }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.landslide }
		}
		@NATURAL_DISASTER_WEIGHT_VOLCANO = {
			modifier = {
				factor = 0
				NOT = { any_controlled_state = { natural_disaster_is_volcanic_candidate_state = yes } }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.volcano }
		}
		@NATURAL_DISASTER_WEIGHT_TSUNAMI = {
			modifier = {
				factor = 0
				check_variable = { natural_disaster_stage < constant:natural_disaster_event.stage_chained_aftermaths }
			}
			modifier = {
				factor = 0
				NOT = { any_controlled_state = { natural_disaster_is_coastal_candidate_state = yes } }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.tsunami }
		}
		@NATURAL_DISASTER_WEIGHT_METEOR = {
			modifier = {
				factor = 0
				check_variable = { natural_disaster_stage < constant:natural_disaster_event.stage_abnormal_age }
			}
			set_temp_variable = { natural_disaster_family = constant:natural_disaster_family.meteor }
		}
	}
}

natural_disaster_select_target_state = {
	clr_country_flag = natural_disaster_target_selection_failed
	random_owned_controlled_state = {
		limit = { natural_disaster_is_valid_target_state = yes }
		save_event_target_as = natural_disaster_current_target_state
		CONTROLLER = { save_event_target_as = natural_disaster_current_controller }
		natural_disaster_score_current_state = yes
	}
	if = {
		limit = { NOT = { has_event_target = natural_disaster_current_target_state } }
		set_country_flag = natural_disaster_target_selection_failed
	}
}

natural_disaster_score_current_state = {
	set_temp_variable = { natural_disaster_target_score = 0 }
	if = {
		limit = { state_population_k > constant:natural_disaster_target.large_population_k }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.capital_score }
	}
	if = {
		limit = { is_capital = yes }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.capital_score }
	}
	if = {
		limit = {
			OR = {
				industrial_complex > 0
				arms_factory > 0
				dockyard > 0
			}
		}
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.industrial_building_score }
	}
	if = {
		limit = { infrastructure > 0 }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.infrastructure_score }
	}
	if = {
		limit = { rail_way > 0 }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.rail_score }
	}
	if = {
		limit = { supply_node > 0 }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.supply_score }
	}
	if = {
		limit = { naval_base > 0 }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.port_score }
	}
	if = {
		limit = { is_coastal = yes }
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.coastal_score }
	}
	if = {
		limit = {
			OR = {
				has_terrain = mountain
				has_terrain = hills
			}
		}
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.mountain_score }
	}
	if = {
		limit = {
			OR = {
				is_desert_state = yes
				has_state_category = rural
				has_state_category = pastoral
			}
		}
		add_to_temp_variable = { natural_disaster_target_score = constant:natural_disaster_target.dry_state_score }
	}
	set_variable = { natural_disaster_last_target_score = natural_disaster_target_score }
}

natural_disaster_setup_warning = {
	set_temp_variable = { natural_disaster_phase = constant:natural_disaster_phase.warning }
	if = {
		limit = { has_event_target = natural_disaster_current_target_state }
		set_country_flag = { flag = natural_disaster_warning_active days = @NATURAL_DISASTER_WARNING_DAYS }
		event_target:natural_disaster_current_target_state = {
			set_state_flag = { flag = natural_disaster_warning_target days = @NATURAL_DISASTER_WARNING_DAYS }
			set_variable = { natural_disaster_warning_family = natural_disaster_family }
			set_variable = { natural_disaster_warning_severity = natural_disaster_severity }
			CONTROLLER = { save_event_target_as = natural_disaster_current_controller }
		}
		set_variable = { natural_disaster_current_family = natural_disaster_family }
		set_variable = { natural_disaster_current_severity = natural_disaster_severity }
		set_variable = { natural_disaster_current_phase = natural_disaster_phase }
	}
}

natural_disaster_roll_warning_chance = {
	set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.baseline_chance }
	if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.earthquake } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.earthquake_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.flood } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.flood_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.storm } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.storm_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.drought } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.drought_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.wildfire } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.wildfire_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.landslide } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.landslide_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.volcano } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.volcano_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.tsunami } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.tsunami_chance }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.meteor } }
		set_temp_variable = { natural_disaster_warning_chance = constant:natural_disaster_warning.meteor_chance }
	}
	set_temp_variable_to_random = {
		var = natural_disaster_warning_roll
		min = constant:natural_disaster_warning.roll_min
		max = constant:natural_disaster_warning.roll_max
	}
	set_temp_variable = { natural_disaster_should_warn = 0 }
	if = {
		limit = {
			check_variable = {
				var = natural_disaster_warning_roll
				value = natural_disaster_warning_chance
				compare = less_than_or_equals
			}
		}
		set_temp_variable = { natural_disaster_should_warn = 1 }
	}
}

natural_disaster_apply_impact_to_current_state = {
	if = {
		limit = { NOT = { has_variable = natural_disaster_severity } }
		set_temp_variable = { natural_disaster_severity = constant:natural_disaster_severity.local }
	}
	natural_disaster_set_damage_profile = yes
	if = {
		limit = { has_state_flag = natural_disaster_prepared_warning }
		multiply_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_warning.prepared_damage_multiplier }
		multiply_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.prepared_multiplier }
	}
	natural_disaster_apply_family_building_damage = yes
	natural_disaster_apply_population_loss = yes
	natural_disaster_setup_recovery_context = yes
}

natural_disaster_apply_current_impact = {
	if = {
		limit = { has_event_target = natural_disaster_current_target_state }
		event_target:natural_disaster_current_target_state = {
			natural_disaster_apply_impact_to_current_state = yes
		}
	}
	else = {
		set_country_flag = natural_disaster_target_selection_failed
	}
}

natural_disaster_set_damage_profile = {
	set_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_impact_damage.local }
	set_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.local }
	if = {
		limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.serious } }
		set_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_impact_damage.serious }
		set_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.serious }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.regional } }
		set_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_impact_damage.regional }
		set_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.regional }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.cascading } }
		set_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_impact_damage.cascading }
		set_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.cascading }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.abnormal } }
		set_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_impact_damage.abnormal }
		set_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.abnormal }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_severity = constant:natural_disaster_severity.scenario_maximum } }
		set_temp_variable = { natural_disaster_damage_amount = constant:natural_disaster_impact_damage.scenario_maximum }
		set_temp_variable = { natural_disaster_population_loss_per_k = constant:natural_disaster_population_loss_per_k.scenario_maximum }
	}
}

natural_disaster_apply_population_loss = {
	set_temp_variable = { natural_disaster_population_loss = state_population_k }
	multiply_temp_variable = { natural_disaster_population_loss = natural_disaster_population_loss_per_k }
	round_temp_variable = natural_disaster_population_loss
	add_manpower = natural_disaster_population_loss
	set_variable = { natural_disaster_last_population_delta = natural_disaster_population_loss }
}

natural_disaster_damage_selected_building_type = {
	meta_effect = {
		text = {
			damage_building = {
				type = [ND_BUILDING_TYPE]
				damage = [ND_DAMAGE]
			}
		}
		ND_BUILDING_TYPE = "[?natural_disaster_building_type.GetTokenKey]"
		ND_DAMAGE = "[?natural_disaster_damage_amount]"
	}
}

	if = {
		limit = {
			OR = {
				check_variable = { natural_disaster_family = constant:natural_disaster_family.earthquake }
				check_variable = { natural_disaster_family = constant:natural_disaster_family.meteor }
			}
		}
		natural_disaster_damage_earthquake_profile = yes
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.flood } }
		natural_disaster_damage_flood_profile = yes
	}

		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.storm } }
		natural_disaster_damage_storm_profile = yes

	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.drought } }

		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.wildfire } }
		natural_disaster_damage_wildfire_profile = yes
	}
	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.landslide } }
		natural_disaster_damage_landslide_profile = yes

	else_if = {
		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.volcano } }

natural_disaster_refresh_recovery_pressure = {
	set_temp_variable = { natural_disaster_pressure_active = 0 }
	set_temp_variable = { natural_disaster_pressure_worst = 0 }
	every_controlled_state = {
		limit = { natural_disaster_state_has_active_aftermath = yes }
		add_to_temp_variable = { natural_disaster_pressure_active = 1 }
		if = {
			limit = { check_variable = { natural_disaster_state_severity > natural_disaster_pressure_worst } }
			set_temp_variable = { natural_disaster_pressure_worst = natural_disaster_state_severity }
		}
	}
	set_variable = { natural_disaster_active_aftermath_count = natural_disaster_pressure_active }
	set_variable = { natural_disaster_worst_active_severity = natural_disaster_pressure_worst }
	if = {
		limit = { check_variable = { natural_disaster_pressure_active > 0 } }
		set_country_flag = natural_disaster_response_visible
	}
	else = {
		clr_country_flag = natural_disaster_response_visible
	}
}

natural_disaster_record_evolution_if_needed = {
	set_temp_variable = { events_log_evolution_event_id = constant:natural_disaster_event.id }
	set_temp_variable = { events_log_evolution_type = constant:natural_disaster_event.evolution_type }
	set_temp_variable = { events_log_evolution_stage = natural_disaster_stage }
	set_temp_variable = { events_log_evolution_tier = natural_disaster_stage }
	if = {
		limit = {
			is_current_evolution_enabled = yes
			check_variable = { natural_disaster_stage > constant:natural_disaster_event.stage_baseline }
			OR = {
				AND = {
					check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_varied_local }
					NOT = { has_global_flag = natural_disaster_evolution_1_recorded }
				}
				AND = {
					check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_regional_systems }
					NOT = { has_global_flag = natural_disaster_evolution_2_recorded }
				}
				AND = {
					check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_chained_aftermaths }
					NOT = { has_global_flag = natural_disaster_evolution_3_recorded }
				}
				AND = {
					check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_abnormal_age }
					NOT = { has_global_flag = natural_disaster_evolution_4_recorded }
				}
			}
		}
		if = {
			limit = { has_event_target = natural_disaster_current_controller }
			event_target:natural_disaster_current_controller = { save_event_target_as = events_log_evolution_actor }
			set_temp_variable = { events_log_evolution_has_actor = 1 }
		}
		record_events_log_evolution_entry = yes
		if = {
			limit = { check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_varied_local } }
			set_global_flag = natural_disaster_evolution_1_recorded
		}
		else_if = {
			limit = { check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_regional_systems } }
			set_global_flag = natural_disaster_evolution_2_recorded
		}
		else_if = {
			limit = { check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_chained_aftermaths } }
			set_global_flag = natural_disaster_evolution_3_recorded
		}
		else_if = {
			limit = { check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_abnormal_age } }
			set_global_flag = natural_disaster_evolution_4_recorded
		}
	}
}
