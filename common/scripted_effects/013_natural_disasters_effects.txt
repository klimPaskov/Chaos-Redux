natural_disasters_initialize_sequence_context = {
	if = {
		limit = { NOT = { has_variable = global.natural_disasters_sequence_counter } }
		set_variable = { global.natural_disasters_sequence_counter = 0 }
	}
	add_to_variable = { global.natural_disasters_sequence_counter = 1 }
	set_variable = { natural_disasters_sequence_id = global.natural_disasters_sequence_counter }
	set_country_flag = natural_disasters_sequence_active

	if = {
		limit = { NOT = { has_variable = natural_disasters_sequence_evolution_stage } }
		set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.baseline }
	}
	if = {
		limit = { NOT = { has_variable = natural_disasters_sequence_incidents_planned } }
		natural_disasters_set_sequence_incident_count_from_context = yes
	}
	set_variable = { natural_disasters_sequence_incidents_completed = 0 }
	set_variable = { natural_disasters_sequence_estimated_deaths = 0 }
	set_variable = { natural_disasters_display_incidents_completed = 0 }
	set_variable = { natural_disasters_display_incidents_planned = natural_disasters_sequence_incidents_planned }
	set_variable = { natural_disasters_display_estimated_deaths = 0 }
	natural_disasters_prepare_delay_band = yes
}

natural_disasters_set_sequence_incident_count_from_context = {
	set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.baseline_incidents }

	if = {
		limit = {
			check_variable = {
				var = natural_disasters_sequence_evolution_stage
				value = constant:natural_disasters_evolution.varied_local
				compare = equals
			}
		}
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.varied_local_incidents }
	}
	else_if = {
		limit = {
			check_variable = {
				var = natural_disasters_sequence_evolution_stage
				value = constant:natural_disasters_evolution.regional_systems
				compare = equals
			}
		}
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.regional_incidents }
	}
	else_if = {
		limit = {
			check_variable = {
				var = natural_disasters_sequence_evolution_stage
				value = constant:natural_disasters_evolution.abnormal_age
				compare = equals
			}
		}
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.abnormal_incidents }
	}

	if = {
		limit = { has_country_flag = natural_disasters_cluster_local_slot }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.cluster_local_incidents }
	}
	if = {
		limit = { has_country_flag = natural_disasters_cluster_regional_slot }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.cluster_regional_incidents }
	}
	if = {
		limit = { has_country_flag = natural_disasters_cluster_abnormal_slot }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.cluster_abnormal_incidents }
	}
	if = {
		limit = { has_country_flag = natural_disasters_scenario_low_intensity }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.scenario_low_incidents }
	}
	if = {
		limit = { has_country_flag = natural_disasters_scenario_medium_intensity }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.scenario_medium_incidents }
	}
	if = {
		limit = { has_country_flag = natural_disasters_scenario_high_intensity }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.scenario_high_incidents }
	}
	if = {
		limit = { has_country_flag = natural_disasters_scenario_maximum_intensity }
		set_variable = { natural_disasters_sequence_incidents_planned = constant:natural_disasters_sequence.scenario_maximum_incidents }
	}
}

natural_disasters_prepare_delay_band = {
	set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.two_min }
	set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.two_max }

	if = {
		limit = { check_variable = { natural_disasters_sequence_incidents_planned < 2 } }
		set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.single_min }
		set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.single_max }
	}
	else_if = {
		limit = { check_variable = { natural_disasters_sequence_incidents_planned < 3 } }
		set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.two_min }
		set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.two_max }
	}
	else_if = {
		limit = { check_variable = { natural_disasters_sequence_incidents_planned < 4 } }
		set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.three_min }
		set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.three_max }
	}
	else_if = {
		limit = { check_variable = { natural_disasters_sequence_incidents_planned < 7 } }
		set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.regional_min }
		set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.regional_max }
	}
	else = {
		set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.abnormal_min }
		set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.abnormal_max }
	}

	if = {
		limit = { has_country_flag = natural_disasters_scenario_maximum_intensity }
		set_variable = { natural_disasters_delay_min_days = constant:natural_disasters_delay.scenario_maximum_min }
		set_variable = { natural_disasters_delay_max_days = constant:natural_disasters_delay.scenario_maximum_max }
	}
}

	set_temp_variable = { events_log_evolution_stage = constant:natural_disasters_event_log.stage_varied_local }
	set_temp_variable = { events_log_evolution_tier = constant:natural_disasters_event_log.tier_varied_local }

		limit = {
			is_current_evolution_enabled = yes
			check_variable = {
				var = current_chaos_tier_value

		}

natural_disasters_apply_cluster_launch_context = {
	if = {
		limit = {
			has_country_flag = event_cluster_member_fire_context
			has_variable = event_cluster_current_member_sequence
		}
		if = {
			limit = { check_variable = { event_cluster_current_member_sequence < 3 } }
			set_country_flag = natural_disasters_cluster_local_slot
			set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.baseline }
		}
		else_if = {
			limit = { check_variable = { event_cluster_current_member_sequence < 5 } }
			set_country_flag = natural_disasters_cluster_regional_slot
			set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.varied_local }
			natural_disasters_record_varied_local_evolution = yes
		}
		else_if = {
			limit = { check_variable = { event_cluster_current_member_sequence < 6 } }
			set_country_flag = natural_disasters_cluster_regional_slot
			set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.regional_systems }
			natural_disasters_record_regional_systems_evolution = yes
		}
		else = {
			set_country_flag = natural_disasters_cluster_abnormal_slot
			set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.abnormal_age }
			natural_disasters_record_abnormal_age_evolution = yes
		}
	}
}

natural_disasters_apply_scenario_launch_context = {
	if = {
		limit = { has_country_flag = natural_disasters_disaster_barrage_launch }
		set_country_flag = natural_disasters_scenario_launch_active
		set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.abnormal_age }

		if = {
			limit = { check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.low } }
			set_country_flag = natural_disasters_scenario_low_intensity
			set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.varied_local }
			natural_disasters_record_varied_local_evolution = yes
		}
		else_if = {
			limit = { check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.medium } }
			set_country_flag = natural_disasters_scenario_medium_intensity
			set_variable = { natural_disasters_sequence_evolution_stage = constant:natural_disasters_evolution.regional_systems }
			natural_disasters_record_varied_local_evolution = yes
			natural_disasters_record_regional_systems_evolution = yes
		}
		else_if = {
			limit = { check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.high } }
			set_country_flag = natural_disasters_scenario_high_intensity
			natural_disasters_record_varied_local_evolution = yes
			natural_disasters_record_regional_systems_evolution = yes
			natural_disasters_record_abnormal_age_evolution = yes
		}
		else = {
			set_country_flag = natural_disasters_scenario_maximum_intensity
			natural_disasters_record_varied_local_evolution = yes
			natural_disasters_record_regional_systems_evolution = yes
			natural_disasters_record_abnormal_age_evolution = yes
		}
	}
}

natural_disasters_prepare_manual_controller_state = {
	natural_disasters_clear_current_manual_controller_state = yes
	clear_variable = natural_disasters_manual_controller_state
	set_variable = { natural_disasters_manual_controller_found = 0 }
	set_temp_variable = { natural_disasters_manual_controller_busy_days = constant:natural_disasters_manual_slot.controller_busy_days }
	random_state = {
		limit = {
			NOT = { has_state_flag = natural_disaster_manual_controller_busy }
		}
		set_state_flag = natural_disaster_manual_controller_state
		set_state_flag = { flag = natural_disaster_manual_controller_busy days = natural_disasters_manual_controller_busy_days }
		set_variable = { natural_disaster_manual_controller_country = ROOT.id }
		set_variable = { natural_disaster_manual_sequence_id = ROOT.natural_disasters_sequence_id }
		ROOT = {
			set_variable = { natural_disasters_manual_controller_state = PREV.id }
			set_variable = { natural_disasters_manual_controller_found = 1 }
		}
	}
	clear_variable = natural_disasters_manual_controller_found
}

natural_disasters_run_manual_scheduler_controller = {
	if = {
		limit = { has_country_flag = natural_disasters_chain_active }
		if = { limit = { NOT = { has_variable = natural_disasters_manual_controller_state } } natural_disasters_prepare_manual_controller_state = yes }
		set_variable = { natural_disasters_manual_controller_delay_days = constant:natural_disasters_chain.controller_delay_days }
		if = {
			limit = { has_variable = natural_disasters_manual_controller_state }
			var:natural_disasters_manual_controller_state = {
				set_variable = { natural_disaster_manual_controller_delay_days = ROOT.natural_disasters_manual_controller_delay_days }
				state_event = { id = chaosx.nr13.51 days = natural_disaster_manual_controller_delay_days }
			}
		}
		clear_variable = natural_disasters_manual_controller_delay_days
	}
	else = {
		natural_disasters_prepare_next_incident = yes
		if = {
			limit = {
				check_variable = { natural_disasters_current_incident_has_target > 0 }
				check_variable = { natural_disasters_current_incident_has_warning > 0 }
			}
			every_owned_state = {
				limit = { has_state_flag = natural_disaster_current_target }
				state_event = { id = chaosx.nr13.11 }
			}
		}
		else_if = {
			limit = { check_variable = { natural_disasters_current_incident_has_target > 0 } }
			every_owned_state = {
				limit = { has_state_flag = natural_disaster_current_target }
				state_event = { id = chaosx.nr13.21 days = constant:natural_disasters_warning.missed_warning_days }
			}
		}
		else_if = {
			limit = {
				check_variable = {
					var = natural_disasters_sequence_incidents_completed
					value = natural_disasters_sequence_incidents_planned
					compare = less_than
				}
			}
			natural_disasters_schedule_next_warning_report = yes
		}
		else = {
			natural_disasters_schedule_sequence_finish = yes
		}
	}
}

natural_disasters_run_sequence_finish = {
	if = {
		limit = {
			check_variable = { natural_disasters_sequence_incidents_planned > 1 }
			check_variable = { natural_disasters_sequence_estimated_deaths < 1 }
			has_country_flag = natural_disasters_country_hit
			NOT = { has_country_flag = achievement_nd_no_deaths_sequence_missed_warning }
			NOT = { has_country_flag = natural_disasters_objective_failed }
		}
		set_country_flag = natural_disasters_no_deaths_sequence_cleanup_watch
		set_country_flag = achievement_nd_no_deaths_sequence_multi_incident
	}
	if = {
		limit = {
			has_country_flag = natural_disasters_scenario_maximum_intensity
			check_variable = {
				var = natural_disasters_sequence_incidents_planned
				value = constant:natural_disasters_sequence.scenario_maximum_incidents
				compare = greater_than_or_equals
			}
			check_variable = {
				var = natural_disasters_sequence_incidents_completed
				value = constant:natural_disasters_achievement.maximum_barrage_incident_floor
				compare = greater_than_or_equals
			}
			NOT = { has_global_flag = world_end }
		}
		set_country_flag = natural_disasters_maximum_barrage_cleanup_watch
		set_country_flag = achievement_nd_barrage_maximum_completed
	}
	if = {
		limit = {
			check_variable = {
				var = natural_disasters_sequence_evolution_stage
				value = constant:natural_disasters_evolution.abnormal_age
				compare = equals
			}
			has_country_flag = natural_disasters_country_hit
			NOT = { has_global_flag = world_end }
		}
		set_country_flag = natural_disasters_abnormal_sequence_cleanup_watch
		set_country_flag = achievement_nd_abnormal_sequence_completed
	}
	natural_disasters_maybe_send_sequence_news = yes
	natural_disasters_cleanup_sequence_context = yes
}

				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.massive_eruption }
				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.dust_sandstorm }
				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.skyfall_meteor }
			}
		}
		add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.airbase_bonus }
	}
	if = {
		limit = {
			natural_disasters_state_is_coastal_or_port = yes
			OR = {
				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.cyclone_storm }
				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.tsunami }
				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.volcano }
				check_variable = { natural_disaster_family_id = constant:natural_disasters_family.massive_eruption }
			}
		}
		add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.coastal_station_bonus }
	}
	if = {
		limit = { natural_disasters_state_has_active_aftermath = yes }
		add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.active_aftermath_penalty }
	}
	OWNER = {
		if = {
			limit = { NOT = { has_war = yes } }
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.peace_bonus } }
		}
		else = {
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.war_penalty } }
		}
		if = {
			limit = { has_stability > constant:natural_disasters_warning_threshold.stable_country }
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.stability_bonus } }
		}
		else_if = {
			limit = { has_stability < constant:natural_disasters_warning_threshold.unstable_country }
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.low_stability_penalty } }
		}
		if = {
			limit = { has_variable = natural_disasters_preparedness_actions_taken }
			set_temp_variable = { natural_disasters_warning_preparedness_bonus = natural_disasters_preparedness_actions_taken }
			multiply_temp_variable = { natural_disasters_warning_preparedness_bonus = constant:natural_disasters_warning_score.preparedness_action_bonus }
			PREV = { add_to_variable = { natural_disaster_warning_score = natural_disasters_warning_preparedness_bonus } }
		}
		if = {
			limit = { has_tech = radio }
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.observatory_network_bonus } }
		}
		if = {
			limit = {
				check_variable = {
					var = natural_disasters_sequence_evolution_stage
					value = constant:natural_disasters_evolution.regional_systems
					compare = equals
				}
			}
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.regional_penalty } }
		}
		else_if = {
			limit = {
				check_variable = {
					var = natural_disasters_sequence_evolution_stage
					value = constant:natural_disasters_evolution.abnormal_age
					compare = equals
				}
			}
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.abnormal_penalty } }
		}
		if = {
			limit = { has_country_flag = natural_disasters_scenario_maximum_intensity }
			PREV = { add_to_variable = { natural_disaster_warning_score = constant:natural_disasters_warning_score.maximum_barrage_penalty } }
		}
	}
	clamp_variable = {
		var = natural_disaster_warning_score
		min = constant:natural_disasters_warning_score.min_roll
		max = constant:natural_disasters_warning_score.max_roll
	}
}

		var = natural_disaster_warning_roll
		min = constant:natural_disasters_warning_score.min_roll
		max = constant:natural_disasters_warning_score.max_roll

	}

natural_disasters_prepare_loss_rate = {
	if = {
		limit = { NOT = { has_variable = natural_disaster_severity_id } }
		set_temp_variable = { natural_disaster_severity_id = constant:natural_disasters_severity.moderate }
	}
	if = {
		limit = { NOT = { has_variable = natural_disaster_evolution_stage } }
		set_temp_variable = { natural_disaster_evolution_stage = constant:natural_disasters_evolution.baseline }
	}
	if = {
		limit = { NOT = { has_variable = natural_disaster_preparedness_mult } }
		set_temp_variable = { natural_disaster_preparedness_mult = constant:natural_disasters_preparedness_multiplier.none }
	}
	if = {
		limit = { NOT = { has_variable = natural_disaster_aftermath_mult } }
		set_temp_variable = { natural_disaster_aftermath_mult = 1 }
	}

	set_temp_variable = { natural_disaster_loss_rate_ceiling = constant:natural_disasters_death_rate_ceiling.moderate }
	set_temp_variable = { natural_disaster_severity_mult = constant:natural_disasters_severity_multiplier.moderate }

	if = {
		limit = {
			check_variable = {
				var = natural_disaster_severity_id
				value = constant:natural_disasters_severity.light
				compare = equals
			}
		}
		set_temp_variable = { natural_disaster_loss_rate_ceiling = constant:natural_disasters_death_rate_ceiling.light }
		set_temp_variable = { natural_disaster_severity_mult = constant:natural_disasters_severity_multiplier.light }
	}
	else_if = {
		limit = {
			check_variable = {
				var = natural_disaster_severity_id
				value = constant:natural_disasters_severity.severe
				compare = equals
			}
		}
		set_temp_variable = { natural_disaster_loss_rate_ceiling = constant:natural_disasters_death_rate_ceiling.severe }
		set_temp_variable = { natural_disaster_severity_mult = constant:natural_disasters_severity_multiplier.severe }
	}
	else_if = {
		limit = {
			check_variable = {
				var = natural_disaster_severity_id
				value = constant:natural_disasters_severity.extreme
				compare = equals
			}
		}
		set_temp_variable = { natural_disaster_loss_rate_ceiling = constant:natural_disasters_death_rate_ceiling.extreme }
		set_temp_variable = { natural_disaster_severity_mult = constant:natural_disasters_severity_multiplier.extreme }
	}

	set_temp_variable = { natural_disaster_evolution_mult = constant:natural_disasters_evolution_multiplier.baseline }
	if = {
		limit = {
			check_variable = {
				var = natural_disaster_evolution_stage
				value = constant:natural_disasters_evolution.varied_local
				compare = equals
			}
		}
		set_temp_variable = { natural_disaster_evolution_mult = constant:natural_disasters_evolution_multiplier.varied_local }
	}
	else_if = {
		limit = {
			check_variable = {
				var = natural_disaster_evolution_stage
				value = constant:natural_disasters_evolution.regional_systems
				compare = equals
			}
		}
		set_temp_variable = { natural_disaster_evolution_mult = constant:natural_disasters_evolution_multiplier.regional_systems }
	}
	else_if = {
		limit = {
			check_variable = {
				var = natural_disaster_evolution_stage
				value = constant:natural_disasters_evolution.abnormal_age
				compare = equals
			}
		}
		set_temp_variable = { natural_disaster_evolution_mult = constant:natural_disasters_evolution_multiplier.abnormal_age }
	}

	set_temp_variable = { natural_disaster_final_loss_rate = natural_disaster_base_loss_rate }
	multiply_temp_variable = { natural_disaster_final_loss_rate = natural_disaster_severity_mult }
	multiply_temp_variable = { natural_disaster_final_loss_rate = natural_disaster_evolution_mult }
	multiply_temp_variable = { natural_disaster_final_loss_rate = natural_disaster_preparedness_mult }
	multiply_temp_variable = { natural_disaster_final_loss_rate = natural_disaster_aftermath_mult }
	clamp_temp_variable = {
		var = natural_disaster_final_loss_rate
		min = 0
		max = natural_disaster_loss_rate_ceiling
	}
}

natural_disasters_register_state_population_loss = {
	set_variable = { natural_disaster_last_impact_loss_rate = natural_disaster_final_loss_rate }
	set_variable = { natural_disaster_last_impact_estimated_deaths = state_population_k }
	multiply_variable = { natural_disaster_last_impact_estimated_deaths = 1000 }
	multiply_variable = { natural_disaster_last_impact_estimated_deaths = natural_disaster_final_loss_rate }
	round_variable = natural_disaster_last_impact_estimated_deaths

	if = {
		limit = {
			has_variable = natural_disasters_death_reason
			check_variable = { natural_disasters_death_reason > 0 }
		}
		set_temp_variable = { chaos_deaths_change = state_population_k }
		multiply_temp_variable = { chaos_deaths_change = 1000 }
		multiply_temp_variable = { chaos_deaths_change = natural_disaster_final_loss_rate }
		round_temp_variable = chaos_deaths_change
		set_temp_variable = { chaos_deaths_is_civilian = 1 }
		set_temp_variable = { chaos_deaths_apply_state_pop = 1 }
		set_temp_variable = { chaos_deaths_target_country = OWNER }
		set_temp_variable = { chaos_deaths_has_target_country = 1 }
		set_temp_variable = { chaos_deaths_reason = natural_disasters_death_reason }
		chaos_meter_register_deaths = yes
		set_variable = { natural_disaster_last_registered_deaths = chaos_deaths_change }
		set_temp_variable = { chaos_deaths_is_civilian = 0 }
		set_temp_variable = { chaos_deaths_apply_state_pop = 0 }
		set_temp_variable = { chaos_deaths_reason = 0 }
	}
	else = {
		set_state_flag = natural_disaster_death_reason_pending
		set_variable = { natural_disaster_last_registered_deaths = 0 }
	}
}

		limit = {

				var = natural_disaster_evolution_stage

			}

			limit = {
				natural_disasters_state_can_receive_impact = yes
				NOT = { has_state_flag = natural_disaster_recent_impact }

					}
				}

				limit = { PREV = { check_variable = { natural_disaster_family_id = constant:natural_disasters_family.wildfire } } }
				ROOT = { set_country_flag = achievement_nd_firebreak_spread_failed }
