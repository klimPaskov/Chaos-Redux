# ============================================================================
# CHAOS REDUX - EVENT 013 NATURAL DISASTERS EFFECTS
# ============================================================================
# Public API and runtime for delayed Natural Disasters seasons.
# - A public Event 013 firing starts one season and one Event Log entry.
# - Internal disaster pulses are hidden delayed events and never call the Event
#   Log dispatcher.
# - All state population loss goes through the Chaos Meter deaths system.
# - Direct disaster calls, SCN-007, and event clusters share this controller.
# ============================================================================

natural_disasters_initialize_global_state = {
	if = {
		limit = { NOT = { has_variable = global.natural_disaster_sequences_started } }
		set_variable = { global.natural_disaster_sequences_started = 0 }
	}
	if = {
		limit = { NOT = { has_variable = global.natural_disaster_pulses_resolved } }
		set_variable = { global.natural_disaster_pulses_resolved = 0 }
	}
	if = {
		limit = { NOT = { has_variable = global.natural_disaster_states_hit } }
		set_variable = { global.natural_disaster_states_hit = 0 }
	}
	if = {
		limit = { NOT = { has_variable = global.natural_disaster_abnormal_pulses } }
		set_variable = { global.natural_disaster_abnormal_pulses = 0 }
	}
}

natural_disasters_select_profile_for_current_chaos = {
	if = {
		limit = { NOT = { has_variable = natural_disasters_start_profile } }
		set_temp_variable = { natural_disasters_start_profile = constant:natural_disaster_sequence_profile.baseline }
		if = {
			limit = { has_global_flag = { flag = chaos_tier value = 1 } }
			set_temp_variable = { natural_disasters_start_profile = constant:natural_disaster_sequence_profile.evolution_1 }
		}
		if = {
			limit = { has_global_flag = { flag = chaos_tier value = 2 } }
			set_temp_variable = { natural_disasters_start_profile = constant:natural_disaster_sequence_profile.evolution_2 }
		}
		if = {
			limit = {
				OR = {
					has_global_flag = { flag = chaos_tier value = 3 }
					has_global_flag = { flag = chaos_tier value = 4 }
					has_global_flag = { flag = chaos_tier value = 5 }
				}
			}
			set_temp_variable = { natural_disasters_start_profile = constant:natural_disaster_sequence_profile.evolution_3 }
		}
	}
}

natural_disasters_record_evolution_if_needed = {
	if = {
		limit = {
			check_variable = {
				var = natural_disasters_start_profile
				value = constant:natural_disaster_sequence_profile.evolution_1
				compare = equals
			}
			NOT = { has_global_flag = natural_disasters_evolution_1_recorded }
		}
		set_global_flag = natural_disasters_evolution_1_recorded
		save_event_target_as = events_log_evolution_actor
		set_temp_variable = { events_log_evolution_event_id = constant:natural_disaster_event.id }
		set_temp_variable = { events_log_evolution_type = constant:natural_disaster_event.evolution_type }
		set_temp_variable = { events_log_evolution_stage = constant:natural_disaster_evolution_stage.diversified_seasons }

				compare = equals

natural_disasters_start_achievement_window = {
	if = {
		limit = {
			has_country_flag = achievement_nd_skyfall_candidate
			NOT = { has_country_flag = achievement_nd_skyfall_survivor_ready }
		}
		clr_country_flag = achievement_nd_skyfall_candidate
	}
	clr_country_flag = achievement_nd_clean_season_disqualified
	set_country_flag = { flag = achievement_nd_clean_season_candidate days = constant:natural_disaster_aftermath.response_visibility_days }
}

natural_disasters_track_achievement_hit = {
	OWNER = {
		set_country_flag = { flag = achievement_nd_clean_season_candidate days = constant:natural_disaster_aftermath.response_visibility_days }
		if = {
			limit = { NOT = { has_variable = natural_disaster_open_recovery_states } }
			set_variable = { natural_disaster_open_recovery_states = 0 }
		}
	}
	if = {
		limit = { has_state_flag = natural_disaster_family_earthquake }
		OWNER = { set_country_flag = { flag = achievement_nd_aftershock_candidate days = constant:natural_disaster_aftermath.response_visibility_days } }
	}
	if = {
		limit = {
			has_state_flag = natural_disaster_family_dust_storm
			OWNER = { has_war = yes }
		}
		OWNER = { set_country_flag = { flag = achievement_nd_dust_war_candidate days = constant:natural_disaster_aftermath.response_visibility_days } }
	}
	if = {
		limit = { has_state_flag = natural_disaster_family_meteor_shower }
		OWNER = { set_country_flag = { flag = achievement_nd_skyfall_candidate days = constant:natural_disaster_aftermath.response_visibility_days } }
	}
	if = {
		limit = { has_state_flag = natural_disaster_family_massive_eruption }
		OWNER = { set_country_flag = { flag = achievement_nd_ash_winter_candidate days = constant:natural_disaster_aftermath.response_visibility_days } }
	}
	if = {
		limit = { has_state_flag = natural_disaster_family_storm_corridor }
		OWNER = {
			if = {
				limit = { NOT = { has_country_flag = achievement_nd_eye_of_road_candidate } }
				set_variable = { natural_disaster_achievement_corridor_prepared_states = 0 }
			}
			set_country_flag = { flag = achievement_nd_eye_of_road_candidate days = constant:natural_disaster_aftermath.response_visibility_days }
		}
	}
	if = {
		limit = { has_state_flag = natural_disaster_abnormal_aftermath }
		OWNER = {
			if = {

		limit = {

			8 = { set_temp_variable = { natural_disaster_current_family = constant:natural_disaster_family.drought } }
			8 = { set_temp_variable = { natural_disaster_current_family = constant:natural_disaster_family.dust_storm } }
			8 = { set_temp_variable = { natural_disaster_current_family = constant:natural_disaster_family.blizzard } }

natural_disasters_select_requested_target_state = {
	if = {
		limit = {
			has_event_target = natural_disasters_requested_target_state
			event_target:natural_disasters_requested_target_state = { natural_disaster_target_current_family = yes }
		}
		event_target:natural_disasters_requested_target_state = {
			save_event_target_as = natural_disaster_target_state
			set_temp_variable = { natural_disaster_target_found = 1 }
		}
	}
}

natural_disasters_select_regional_seed_target = {
	if = {
		limit = { has_event_target = natural_disasters_requested_target_state }
		event_target:natural_disasters_requested_target_state = {
			random_neighbor_state = {
				limit = { natural_disaster_target_current_family = yes }
				save_event_target_as = natural_disaster_target_state
				set_temp_variable = { natural_disaster_target_found = 1 }
			}
		}
	}
	if = {
		limit = { check_variable = { natural_disaster_target_found < 1 } }
		natural_disasters_select_requested_target_state = yes
	}
}

natural_disasters_select_random_world_target = {
	random_state = {
		limit = { natural_disaster_target_current_family = yes }
		save_event_target_as = natural_disaster_target_state
		set_temp_variable = { natural_disaster_target_found = 1 }
	}

natural_disasters_select_target = {
	set_temp_variable = { natural_disaster_target_found = 0 }
	natural_disasters_load_requested_targets_for_current_slot = yes
	if = {
		limit = { check_variable = { natural_disaster_current_target_mode = constant:natural_disaster_target_mode.exact_state } }
		natural_disasters_select_requested_target_state = yes
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_target_mode = constant:natural_disaster_target_mode.regional_seed } }
		natural_disasters_select_regional_seed_target = yes
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_target_mode = constant:natural_disaster_target_mode.target_country } }
		natural_disasters_select_target_country_state = yes
	}
	else = {
		natural_disasters_select_random_world_target = yes
	}
}

natural_disasters_should_schedule_report = {
	set_temp_variable = { natural_disaster_should_report = 0 }
	if = {
		limit = { check_variable = { natural_disaster_current_report_policy = constant:natural_disaster_report_policy.always } }
		set_temp_variable = { natural_disaster_should_report = 1 }
	}
	else_if = {
		limit = {
			check_variable = { natural_disaster_current_report_policy = constant:natural_disaster_report_policy.first_only }
			check_variable = { natural_disaster_current_pulse < 2 }
		}
		set_temp_variable = { natural_disaster_should_report = 1 }
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_report_policy = constant:natural_disaster_report_policy.important } }
		natural_disasters_mark_important_report_if_relevant = yes
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_report_policy = constant:natural_disaster_report_policy.auto } }
		if = {
			limit = {
				OR = {
					check_variable = { natural_disaster_current_profile = constant:natural_disaster_sequence_profile.direct }
					check_variable = { natural_disaster_current_profile = constant:natural_disaster_sequence_profile.baseline }
					check_variable = { natural_disaster_current_profile = constant:natural_disaster_sequence_profile.evolution_1 }
				}
			}
			if = { limit = { check_variable = { natural_disaster_current_pulse < 3 } } set_temp_variable = { natural_disaster_should_report = 1 } }
			natural_disasters_mark_important_report_if_relevant = yes
		}
		else = {
			natural_disasters_mark_important_report_if_relevant = yes
		}
	}
	if = {
		limit = {
			OWNER = { has_country_flag = natural_disaster_report_recent }
			NOT = { OWNER = { is_ai = no } }
			NOT = { is_capital = yes }
			NOT = { has_state_flag = natural_disaster_abnormal_aftermath }
		}
		set_temp_variable = { natural_disaster_should_report = 0 }
	}
	if = {
		limit = {
			OR = {
				check_variable = { natural_disaster_current_report_policy = constant:natural_disaster_report_policy.quiet }
				check_variable = { natural_disaster_current_report_policy = constant:natural_disaster_report_policy.news_only }
			}
		}
		set_temp_variable = { natural_disaster_should_report = 0 }
	}
}
