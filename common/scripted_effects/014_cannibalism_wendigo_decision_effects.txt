cannibalism_calculate_current_counterwar_anchor_score = {
	set_temp_variable = { cannibalism_wendigo_counterwar_anchor_score = constant:cannibalism_wendigo_decision.selection_base_score }
	if = {
		limit = { CONTROLLER = { has_war_with = event_target:cannibalism_wendigo_counterwar_actor } }
		add_to_temp_variable = { cannibalism_wendigo_counterwar_anchor_score = constant:cannibalism_wendigo_decision.selection_at_war_score }
	}
	if = {
		limit = { any_neighbor_state = { CONTROLLER = { tag = event_target:cannibalism_wendigo_counterwar_actor } } }
		add_to_temp_variable = { cannibalism_wendigo_counterwar_anchor_score = constant:cannibalism_wendigo_decision.selection_adjacent_score }
	}
	if = {
		limit = { has_state_flag = cannibalism_wendigo_anchor_identified }
		add_to_temp_variable = { cannibalism_wendigo_counterwar_anchor_score = constant:cannibalism_wendigo_decision.selection_identified_score }
	}
	set_temp_variable = { cannibalism_wendigo_counterwar_strength_component = constant:cannibalism_wendigo_decision.selection_maximum_strength }
	subtract_from_temp_variable = { cannibalism_wendigo_counterwar_strength_component = cannibalism_wendigo_anchor_strength }
	add_to_temp_variable = { cannibalism_wendigo_counterwar_anchor_score = cannibalism_wendigo_counterwar_strength_component }
}

cannibalism_select_counterwar_anchor = {
	set_temp_variable = { cannibalism_wendigo_counterwar_anchor_found = 0 }
	set_temp_variable = { cannibalism_wendigo_counterwar_best_score = 0 }
	save_event_target_as = cannibalism_wendigo_counterwar_actor
	for_each_scope_loop = {
		array = global.cannibalism_transformation_anchor_states
		if = {
			limit = {
				cannibalism_current_wendigo_anchor_is_reachable_by_root = yes
				OR = {
					check_variable = { cannibalism_wendigo_counterwar_require_unidentified < 1 }
					NOT = { has_state_flag = cannibalism_wendigo_anchor_identified }
				}
				OR = {
					check_variable = { cannibalism_wendigo_counterwar_require_active_site < 1 }
					NOT = { has_state_flag = cannibalism_wendigo_recruitment_site_broken }
				}
			}
			cannibalism_calculate_current_counterwar_anchor_score = yes
			if = {
				limit = {
					check_variable = {
						var = cannibalism_wendigo_counterwar_anchor_score
						value = cannibalism_wendigo_counterwar_best_score
						compare = greater_than
					}
				}
				set_temp_variable = { cannibalism_wendigo_counterwar_best_score = cannibalism_wendigo_counterwar_anchor_score }
				set_temp_variable = { cannibalism_wendigo_counterwar_anchor_found = 1 }
				save_event_target_as = cannibalism_wendigo_counterwar_anchor
			}
		}
	}
}

cannibalism_apply_counterwar_anchor_damage = {
	event_target:cannibalism_wendigo_counterwar_anchor = {
		subtract_from_variable = { cannibalism_wendigo_anchor_strength = cannibalism_wendigo_counterwar_anchor_damage }
		if = {
			limit = {
				check_variable = {
					var = cannibalism_wendigo_anchor_strength
					value = constant:cannibalism_wendigo_anchor.minimum_live_strength
					compare = less_than
				}
			}
			cannibalism_break_loaded_wendigo_anchor = yes
			set_temp_variable = { cannibalism_wendigo_counterwar_anchor_destroyed = 1 }
		}
		else = {
			clamp_variable = {
				var = cannibalism_wendigo_anchor_strength
				min = constant:cannibalism_wendigo_anchor.minimum_live_strength
				max = constant:cannibalism_wendigo_anchor.maximum_strength
			}
		}
	}
	if = {
		limit = {
			check_variable = { cannibalism_wendigo_counterwar_anchor_destroyed > 0 }
			is_ai = no
		}
		set_country_flag = achievement_cannibalism_wendigo_counterwar_contributor
		set_country_flag = achievement_cannibalism_decisive_aid
	}
}
