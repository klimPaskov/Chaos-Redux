# ============================================================================
# CHAOS REDUX - EVENT 021 FRACTURE EVIDENCE RECEIPTS
# ============================================================================
# Source-owned, expiring receipts for Event 021 route eligibility. The parent
# planner may derive a route only from a live receipt; government form, leader
# count, state count, and division count alone are not route evidence.
#
# Inputs to `event021_register_fracture_evidence` are temporary variables:
# `event021_evidence_source`, `event021_evidence_strength`, and
# `event021_evidence_duration`. Optional scope identifiers are carried in
# `event021_evidence_actor_id`, `event021_evidence_package_id`, and
# `event021_evidence_region_id`. A caller with a live country scope may also
# set the normal scope variable `event021_evidence_actor_scope`; registration
# copies it into the source-specific receipt and clears the API input. Every
# receipt stores its source, strength, start, expiry, and cleanup owner on the
# country that owns the evidence.
#
# Event 004 and Event 007 call only the convenience recorders below. They add
# pressure evidence and never fire Event 021 or Event 006. The parent owns
# route flags, actor creation, and all route cleanup.
# ============================================================================

# COUNTRY scope. Copies optional identifiers into the receipt family that owns
# them. Regional, command, and Event 006 receipts always carry a scope pointer:
# an explicit referenced actor when supplied, otherwise the current country as
# the institutional owner. Refresh can therefore invalidate a receipt when its
# referenced actor ceases to exist without consulting overwriteable last-* data.
event021_capture_fracture_evidence_identity = {
	if = {
		limit = { check_variable = { var = event021_evidence_source value = constant:random_civil_war_evidence_source.regional compare = equals } }
		set_variable = { random_civil_war_regional_evidence_actor_scope = THIS }
		if = {
			limit = { has_variable = event021_evidence_actor_scope var:event021_evidence_actor_scope = { exists = yes } }
			set_variable = { random_civil_war_regional_evidence_actor_scope = var:event021_evidence_actor_scope }
		}
		if = { limit = { has_variable = event021_evidence_actor_id } set_variable = { random_civil_war_regional_evidence_actor_id = event021_evidence_actor_id } }
		if = { limit = { has_variable = event021_evidence_region_id } set_variable = { random_civil_war_regional_evidence_region_id = event021_evidence_region_id } }
	}
	if = {
		limit = { check_variable = { var = event021_evidence_source value = constant:random_civil_war_evidence_source.command compare = equals } }
		set_variable = { random_civil_war_command_evidence_actor_scope = THIS }
		if = {
			limit = { has_variable = event021_evidence_actor_scope var:event021_evidence_actor_scope = { exists = yes } }
			set_variable = { random_civil_war_command_evidence_actor_scope = var:event021_evidence_actor_scope }
		}
		if = { limit = { has_variable = event021_evidence_actor_id } set_variable = { random_civil_war_command_evidence_actor_id = event021_evidence_actor_id } }
	}
	if = {
		limit = { check_variable = { var = event021_evidence_source value = constant:random_civil_war_evidence_source.event006 compare = equals } }
		set_variable = { random_civil_war_event6_evidence_actor_scope = THIS }
		if = {
			limit = { has_variable = event021_evidence_actor_scope var:event021_evidence_actor_scope = { exists = yes } }
			set_variable = { random_civil_war_event6_evidence_actor_scope = var:event021_evidence_actor_scope }
		}
		if = { limit = { has_variable = event021_evidence_actor_id } set_variable = { random_civil_war_event6_evidence_actor_id = event021_evidence_actor_id } }
		if = { limit = { has_variable = event021_evidence_package_id } set_variable = { random_civil_war_event6_evidence_package_id = event021_evidence_package_id } }
		if = { limit = { has_variable = event021_evidence_region_id } set_variable = { random_civil_war_event6_evidence_region_id = event021_evidence_region_id } }
	}
}
