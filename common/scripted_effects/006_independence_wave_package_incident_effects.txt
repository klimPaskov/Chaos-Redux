# ============================================================================
# CHAOS REDUX - EVENT 006 PACKAGE INCIDENT SCHEDULING
# ============================================================================
# Overview:
# - `independence_wave_schedule_package_incident_a` runs in the released
#   country when it resolves its founding posture and schedules that package's
#   first incident.
# - `independence_wave_schedule_package_incident_b` runs from either answer to
#   the first incident and schedules the second.
# - Both store the current generation and a due date. Public delivery also
#   consumes a phase-specific flag before the player chooses an answer.
# - `independence_wave_clear_package_incident_runtime` clears every receipt; the
#   generation reset and origin end call it.
# No effect iterates countries or runs on a periodic hook.
# ============================================================================

# COUNTRY scope: the released country after its founding posture resolves.
independence_wave_schedule_package_incident_a = {
	if = {
		limit = {
			is_independence_wave_active_country = yes
			has_variable = independence_wave_package_id
			NOT = { has_country_flag = independence_wave_package_incident_a_scheduled }
		}
		set_temp_variable = { independence_wave_package_incident_event_id = constant:chaosx_date_units.zero }
		independence_wave_package_incident_select_a_event = yes
		if = {
			limit = { check_variable = { var = independence_wave_package_incident_event_id value = constant:chaosx_date_units.zero compare = greater_than } }
			set_variable = { independence_wave_package_incident_generation = independence_wave_generation_id }
			set_temp_variable = { chaosx_date_days = constant:independence_wave_package_incident.first_delay_days }
			set_temp_variable = { chaosx_date_offset = constant:chaosx_date_units.zero }
			chaosx_convert_days_to_date_offset = yes
			set_variable = { independence_wave_package_incident_a_due_date = global.date }
			add_to_variable = { independence_wave_package_incident_a_due_date = chaosx_date_offset }
			set_country_flag = independence_wave_package_incident_a_scheduled
			independence_wave_package_incident_queue_a_appointment = yes
		}
	}
}

# COUNTRY scope: called by either answer to the first incident.

# COUNTRY scope. Clear the generation-local appointment and answer receipts.
# A shown slot and its two snapshot variables belong to the outstanding native
# popup and survive this cleanup; only that popup's AFTER releases them.
independence_wave_clear_package_incident_runtime = {
	clr_country_flag = independence_wave_package_incident_a_scheduled
	clr_country_flag = independence_wave_package_incident_a_resolved
	clr_country_flag = independence_wave_package_incident_b_scheduled
	clr_country_flag = independence_wave_package_incident_b_resolved
	clr_country_flag = independence_wave_package_incident_a_delivered
	clr_country_flag = independence_wave_package_incident_b_delivered
	clear_variable = independence_wave_package_incident_a_due_date
	clear_variable = independence_wave_package_incident_b_due_date
	clear_variable = independence_wave_package_incident_generation
}

# Existing appointment only. Preserve its due date when an old shown slot closes.
independence_wave_package_incident_queue_a_appointment = {
	if = {
		limit = {
			can_fire_independence_wave_package_incident = yes
			has_variable = independence_wave_package_id
			has_country_flag = independence_wave_package_incident_a_scheduled
			has_variable = independence_wave_package_incident_a_due_date
			NOT = { has_country_flag = independence_wave_package_incident_a_delivered }
			NOT = { has_country_flag = independence_wave_package_incident_a_resolved }
			NOT = { has_country_flag = independence_wave_package_incident_a_shown }
		}
		set_temp_variable = { independence_wave_package_incident_event_id = constant:chaosx_date_units.zero }
		independence_wave_package_incident_select_a_event = yes
		if = {
			limit = { check_variable = { var = independence_wave_package_incident_event_id value = constant:chaosx_date_units.zero compare = greater_than } }
			set_temp_variable = { independence_wave_package_incident_remaining_hours = independence_wave_package_incident_a_due_date }
			subtract_from_temp_variable = { independence_wave_package_incident_remaining_hours = global.date }
			multiply_temp_variable = { independence_wave_package_incident_remaining_hours = constant:chaosx_date_units.hours_per_date_unit }
			clamp_temp_variable = { var = independence_wave_package_incident_remaining_hours min = constant:chaosx_date_units.zero }
			meta_effect = {
				text = { country_event = { id = chaosx.nr6.[INCIDENT] hours = [HOURS] } }
				INCIDENT = "[?independence_wave_package_incident_event_id|.0]"
				HOURS = "[?independence_wave_package_incident_remaining_hours|.0]"
			}
		}
	}
}
