# ============================================================================
# CHAOS REDUX - EVENT 039 MOVEMENT SUCCESSION
# ============================================================================
# Overview:
# - Installs one existing route successor package for the Assassin State.
# - Verifies the ruling leader before the registered founder is captured.
# - Retires a registered national leader only after a different sitting ruler
#   and the exact unused registry row have been verified and consumed.
# - Restores the consumed row if native retirement leaves the victim recruited.
# - Does not create countries. Outcomes remain owned by integration callers.

# COUNTRY investigator or victim government. The selected national-leadership
# row and its owner callback are inputs. The owner must have installed a
# successor. A synchronous global guard protects the private chain targets from
# recursive retirement calls. No event is queued while this guard is held.
# Output: named_removal_committed only after the victim leaves the owner roster.
murder_mystery_retire_selected_leader = {
	clr_country_flag = murder_mystery_named_removal_committed
	set_country_flag = murder_mystery_named_removal_failed
	clear_variable = murder_mystery_named_removal_target_id
	clear_variable = murder_mystery_named_removal_generation
	if = {
		limit = { NOT = { has_global_flag = murder_mystery_named_removal_active } }
		set_global_flag = murder_mystery_named_removal_active
		clr_country_flag = murder_mystery_target_consumption_committed
		murder_mystery_prepare_safe_target_action = yes
		if = {
			limit = {
				murder_mystery_selected_target_allows_named_action = yes
				check_variable = { var = murder_mystery_selected_target_role value = constant:murder_mystery_target_role.national_leadership compare = equals }
				event_target:murder_mystery_selected_target_owner = {
					OR = {
						has_country_flag = murder_mystery_emergency_successor_installed
						has_country_flag = murder_mystery_capture_successor_verified
					}
				}
			}
			# Freeze caller inputs before entering another helper or country.
			set_variable = { murder_mystery_removal_target_id = murder_mystery_selected_target_id }
			set_variable = { murder_mystery_removal_generation = murder_mystery_selected_target_generation }
			set_variable = { murder_mystery_removal_disposition = murder_mystery_selected_target_disposition }
			set_variable = { murder_mystery_removal_protection = murder_mystery_selected_target_protection }
			event_target:murder_mystery_selected_target_character = { save_event_target_as = murder_mystery_removal_character }
			event_target:murder_mystery_selected_target_owner = {
				save_event_target_as = murder_mystery_removal_owner
				set_country_flag = murder_mystery_ruling_selection_authorized
				murder_mystery_capture_current_ruling_character = yes
			}
			if = {
				limit = { event_target:murder_mystery_removal_owner = { has_country_flag = murder_mystery_ruling_character_resolved } }
				event_target:murder_mystery_current_ruling_character = { save_event_target_as = murder_mystery_removal_successor }
			}
			if = {
				limit = {
					event_target:murder_mystery_removal_owner = { has_country_flag = murder_mystery_ruling_character_resolved }
					has_event_target = murder_mystery_removal_successor
					NOT = { event_target:murder_mystery_removal_successor = { is_character = event_target:murder_mystery_removal_character } }
				}
				# Restore the frozen selection and revalidate it after ruler
				# resolution. A changed row or overwritten public selector
				# cannot redirect this transaction to another character.
				set_variable = { murder_mystery_selected_target_id = murder_mystery_removal_target_id }
				set_variable = { murder_mystery_selected_target_expected_generation = murder_mystery_removal_generation }
				murder_mystery_prepare_safe_target_action = yes
				# Tombstone detection during validation is not proof that this
				# invocation consumed the row. Require its fresh receipt.
				clr_country_flag = murder_mystery_target_consumption_committed
				if = {
					limit = {
						murder_mystery_selected_target_allows_named_action = yes
						check_variable = { var = murder_mystery_selected_target_role value = constant:murder_mystery_target_role.national_leadership compare = equals }
						check_variable = { var = murder_mystery_selected_target_disposition value = murder_mystery_removal_disposition compare = equals }
						check_variable = { var = murder_mystery_selected_target_protection value = murder_mystery_removal_protection compare = equals }
						event_target:murder_mystery_selected_target_owner = { tag = event_target:murder_mystery_removal_owner }
						event_target:murder_mystery_selected_target_character = { is_character = event_target:murder_mystery_removal_character }
					}
					murder_mystery_mark_selected_target_removed = yes
				}
				if = {
					limit = { has_country_flag = murder_mystery_target_consumption_committed }
					event_target:murder_mystery_removal_character = { retire = yes }
					if = {
						limit = {
							event_target:murder_mystery_removal_owner = {
								exists = yes
								NOT = { any_character = { is_character = event_target:murder_mystery_removal_character } }
								any_character = { is_character = event_target:murder_mystery_removal_successor }
							}
						}
						set_country_flag = murder_mystery_named_removal_committed
						clr_country_flag = murder_mystery_named_removal_failed
						set_variable = { murder_mystery_named_removal_target_id = murder_mystery_removal_target_id }
						set_variable = { murder_mystery_named_removal_generation = murder_mystery_removal_generation }
					}
					else_if = {
						limit = {
							murder_mystery_target_registry_is_aligned = yes
							event_target:murder_mystery_removal_owner = { any_character = { is_character = event_target:murder_mystery_removal_character } }
						}
						# Restore only this transaction's still-present victim. A
						# missing victim keeps its tombstone even if another
						# postcondition failed, preventing a second removal.
						set_temp_variable = { murder_mystery_removal_row = constant:murder_mystery_identity.first_array_index }
						while_loop_effect = {
							limit = { check_variable = { var = murder_mystery_removal_row value = global.murder_mystery_target_id_entries^num compare = less_than } }
							if = {
								limit = {
									check_variable = { var = global.murder_mystery_target_id_entries^murder_mystery_removal_row value = murder_mystery_removal_target_id compare = equals }
									check_variable = { var = global.murder_mystery_target_generation_entries^murder_mystery_removal_row value = murder_mystery_removal_generation compare = equals }
									check_variable = { var = global.murder_mystery_target_disposition_entries^murder_mystery_removal_row value = constant:murder_mystery_target_disposition.already_removed compare = equals }
									check_variable = { var = global.murder_mystery_target_protection_entries^murder_mystery_removal_row value = constant:murder_mystery_target_protection.hardened compare = equals }
									check_variable = { var = global.murder_mystery_target_used_entries^murder_mystery_removal_row value = constant:murder_mystery_value.one compare = equals }
									var:global.murder_mystery_target_owner_entries^murder_mystery_removal_row = { tag = event_target:murder_mystery_removal_owner }
									var:global.murder_mystery_target_subject_entries^murder_mystery_removal_row = { is_character = event_target:murder_mystery_removal_character }
								}
								set_variable = { global.murder_mystery_target_disposition_entries^murder_mystery_removal_row = murder_mystery_removal_disposition }
								set_variable = { global.murder_mystery_target_protection_entries^murder_mystery_removal_row = murder_mystery_removal_protection }
								set_variable = { global.murder_mystery_target_used_entries^murder_mystery_removal_row = constant:murder_mystery_value.zero }
								clr_country_flag = murder_mystery_selected_target_retired
								clr_country_flag = murder_mystery_target_consumption_committed
							}
							add_to_temp_variable = { murder_mystery_removal_row = constant:murder_mystery_identity.loop_increment }
						}
					}
				}
			}
		}
		clear_variable = murder_mystery_removal_target_id
		clear_variable = murder_mystery_removal_generation
		clear_variable = murder_mystery_removal_disposition
		clear_variable = murder_mystery_removal_protection
		clr_global_flag = murder_mystery_named_removal_active
	}
}

# COUNTRY scope. Resolve the actual ruler without selecting another party's
# leader. The native COUNTRY trait operation addresses the sitting leader.
# Output: a chain-local character target and country-local success flag.
murder_mystery_capture_current_ruling_character = {
	clr_country_flag = murder_mystery_ruling_character_resolved
	set_temp_variable = { murder_mystery_ruling_character_matches = constant:murder_mystery_value.zero }
	if = {
		limit = {
			exists = yes
			has_country_flag = murder_mystery_ruling_selection_authorized
		}
		# An inherited anchor is not identity proof. Fail closed instead of
		# selecting one of several tagged characters or modifying their jobs.
		if = {
			limit = {
				NOT = {
					any_character = { has_trait = murder_mystery_ruling_selection_anchor }
				}
			}
			add_country_leader_trait = murder_mystery_ruling_selection_anchor
			every_character = {
				limit = {
					is_country_leader = yes
					has_trait = murder_mystery_ruling_selection_anchor
				}
				add_to_temp_variable = { murder_mystery_ruling_character_matches = constant:murder_mystery_value.one }
				save_event_target_as = murder_mystery_current_ruling_character
			}
			remove_country_leader_trait = murder_mystery_ruling_selection_anchor
			if = {
				limit = {
					check_variable = { var = murder_mystery_ruling_character_matches value = constant:murder_mystery_value.one compare = equals }
					has_event_target = murder_mystery_current_ruling_character
					any_character = { is_character = event_target:murder_mystery_current_ruling_character }
				}
				set_country_flag = murder_mystery_ruling_character_resolved
			}
		}
	}
	clr_country_flag = murder_mystery_ruling_selection_authorized
}

# COUNTRY scope. Input: the current movement route flags and neutrality party.
# COUNTRY candidate after transfer. No historical recruitment or roster theft.
murder_mystery_register_candidate_founder = {
	clr_country_flag = murder_mystery_first_knife_registered
	clr_country_flag = murder_mystery_target_registration_succeeded
	clr_country_flag = murder_mystery_ruling_character_resolved
	if = {
		limit = {
			exists = yes
			murder_mystery_assassin_candidate_is_reserved = yes
			tag = event_target:murder_mystery_assassin_state_candidate
			has_country_flag = murder_mystery_assassin_state
			has_country_leader = { name = MURDER_MYSTERY_FIRST_KNIFE_NAME ruling_only = yes }
		}
		clear_global_event_target = murder_mystery_first_knife_character
		set_country_flag = murder_mystery_ruling_selection_authorized
		murder_mystery_capture_current_ruling_character = yes
		if = {
			limit = { has_country_flag = murder_mystery_ruling_character_resolved }
			event_target:murder_mystery_current_ruling_character = {
				set_character_flag = murder_mystery_first_knife_character
				save_global_event_target_as = murder_mystery_first_knife_character
				save_global_event_target_as = murder_mystery_target_registration_character
			}
			set_variable = { murder_mystery_target_registration_id = constant:murder_mystery_value.first_knife_target_id }
			set_variable = { murder_mystery_target_registration_disposition = constant:murder_mystery_target_disposition.eligible_named_target }
			set_variable = { murder_mystery_target_registration_role = constant:murder_mystery_target_role.national_leadership }
			set_variable = { murder_mystery_target_registration_protection = constant:murder_mystery_target_protection.normal }
			set_variable = { murder_mystery_target_registration_generation = global.murder_mystery_transaction_sequence }
			murder_mystery_register_safe_target = yes
			clear_global_event_target = murder_mystery_target_registration_character
			if = {
				limit = { has_country_flag = murder_mystery_target_registration_succeeded }
				set_country_flag = murder_mystery_first_knife_registered
				set_variable = { murder_mystery_first_knife_generation = murder_mystery_target_registration_generation }
			}
			clear_variable = murder_mystery_target_registration_id
			clear_variable = murder_mystery_target_registration_disposition
			clear_variable = murder_mystery_target_registration_role
			clear_variable = murder_mystery_target_registration_protection
			clear_variable = murder_mystery_target_registration_generation
		}
	}
}

# COUNTRY scope. Input: the current movement route flags and neutrality party.
# Output: murder_mystery_capture_successor_verified after a ruling-leader check.
# Reusing a successor name follows native create_country_leader idempotence.
murder_mystery_install_movement_successor = {
	clr_country_flag = murder_mystery_capture_successor_verified
	if = {
		limit = {
			exists = yes
			has_country_flag = murder_mystery_assassin_state
			has_government = neutrality
		}
		if = {
			limit = { has_country_flag = murder_mystery_cells_without_masters_route }
			create_country_leader = {
				name = MURDER_MYSTERY_CELLS_WITHOUT_MASTERS_SUCCESSOR_NAME
				desc = MURDER_MYSTERY_CELLS_WITHOUT_MASTERS_SUCCESSOR_DESC
				picture = GFX_portrait_039_cells_without_masters
				ideology = despotism
			}
		}
		else_if = {
			limit = { has_country_flag = murder_mystery_necessary_mask_route }
			create_country_leader = {
				name = MURDER_MYSTERY_NECESSARY_MASK_SUCCESSOR_NAME
				desc = MURDER_MYSTERY_NECESSARY_MASK_SUCCESSOR_DESC
				picture = GFX_portrait_039_necessary_mask
				ideology = despotism
			}
		}
		else = {
			create_country_leader = {
				name = MURDER_MYSTERY_HIDDEN_HAND_SUCCESSOR_NAME
				desc = MURDER_MYSTERY_HIDDEN_HAND_SUCCESSOR_DESC
				picture = GFX_portrait_039_hidden_hand
				ideology = despotism
			}
		}
		if = {
			limit = {
				OR = {
					AND = {
						has_country_flag = murder_mystery_cells_without_masters_route
						has_country_leader = { ruling_only = yes name = MURDER_MYSTERY_CELLS_WITHOUT_MASTERS_SUCCESSOR_NAME }
					}
					AND = {
						NOT = { has_country_flag = murder_mystery_cells_without_masters_route }
						has_country_flag = murder_mystery_necessary_mask_route
						has_country_leader = { ruling_only = yes name = MURDER_MYSTERY_NECESSARY_MASK_SUCCESSOR_NAME }
					}
					AND = {
						NOT = { has_country_flag = murder_mystery_cells_without_masters_route }
						NOT = { has_country_flag = murder_mystery_necessary_mask_route }
						has_country_leader = { ruling_only = yes name = MURDER_MYSTERY_HIDDEN_HAND_SUCCESSOR_NAME }
					}
				}
			}
			set_country_flag = murder_mystery_capture_successor_verified
		}
	}
}

# COUNTRY investigator. A living central country keeps its government,
# territory, forces, foreign network, route and runtime after founder capture.
murder_mystery_apply_live_central_capture_shock = {
	clr_country_flag = murder_mystery_live_central_after_capture
	if = {
		limit = {
			has_global_flag = murder_mystery_core_capture_committed
			has_event_target = murder_mystery_assassin_state
			event_target:murder_mystery_assassin_state = {
				exists = yes
				has_country_flag = murder_mystery_assassin_state
				check_variable = { num_owned_states > constant:murder_mystery_value.zero }
			}
		}
		set_country_flag = murder_mystery_live_central_after_capture
		clr_country_flag = murder_mystery_permanent_resolution_ready
		if = {
			limit = { NOT = { has_global_flag = murder_mystery_capture_shock_applied } }
			set_global_flag = murder_mystery_capture_shock_applied
			event_target:murder_mystery_assassin_state = {
				set_temp_variable = { murder_mystery_capture_cohesion_loss = murder_mystery_brotherhood_cohesion }
				multiply_temp_variable = { murder_mystery_capture_cohesion_loss = constant:murder_mystery_capture_shock.cohesion_ratio }
				clamp_temp_variable = { var = murder_mystery_capture_cohesion_loss min = constant:murder_mystery_capture_shock.cohesion_minimum max = constant:murder_mystery_capture_shock.cohesion_maximum }
				subtract_from_variable = { murder_mystery_brotherhood_cohesion = murder_mystery_capture_cohesion_loss }
				set_temp_variable = { murder_mystery_capture_reach_loss = murder_mystery_network_reach }
				multiply_temp_variable = { murder_mystery_capture_reach_loss = constant:murder_mystery_capture_shock.reach_ratio }
				clamp_temp_variable = { var = murder_mystery_capture_reach_loss min = constant:murder_mystery_capture_shock.reach_minimum max = constant:murder_mystery_capture_shock.reach_maximum }
				subtract_from_variable = { murder_mystery_network_reach = murder_mystery_capture_reach_loss }
				murder_mystery_clamp_country_values = yes
			}
		}
	}
}

# COUNTRY scope: a government whose ruler is about to be murdered.
# Input: chain-local event target murder_mystery_victim_character (the sitting
# ruler resolved by murder_mystery_capture_current_ruling_character) and temp
# murder_mystery_successor_style (constant:murder_mystery_successor.style_*).
# Installs the replacement before the victim is removed: a curated native
# successor when a verified row applies, otherwise an institutional emergency
# body named for the country in the ruling ideology. No invented individual is
# created for a real government.
# Output: country flag murder_mystery_emergency_successor_installed only when
# the sitting ruler is now the replacement and not the victim.
murder_mystery_install_emergency_successor = {
	clr_country_flag = murder_mystery_emergency_successor_installed
	clr_country_flag = murder_mystery_curated_successor_used
	murder_mystery_try_curated_successor = yes
	if = {
		limit = { NOT = { has_country_flag = murder_mystery_curated_successor_used } }
		murder_mystery_create_emergency_body = yes
	}
	set_country_flag = murder_mystery_ruling_selection_authorized
	murder_mystery_capture_current_ruling_character = yes
	if = {
		limit = {
			has_country_flag = murder_mystery_ruling_character_resolved
			has_event_target = murder_mystery_victim_character
			NOT = { event_target:murder_mystery_current_ruling_character = { is_character = event_target:murder_mystery_victim_character } }
			OR = {
				has_country_flag = murder_mystery_curated_successor_used
				event_target:murder_mystery_current_ruling_character = {
					OR = {
						has_trait = murder_mystery_emergency_council
						has_trait = murder_mystery_acting_cabinet
					}
				}
			}
		}
		set_country_flag = murder_mystery_emergency_successor_installed
	}
}

# COUNTRY scope. Curated native successor rows: the exact victim token, the
# successor token from the same roster and ideology, and a ruling check after
# promotion. Add a row only for a verified historical succession.
murder_mystery_try_curated_successor = {
	if = {
		limit = {
			original_tag = ENG
			has_government = democratic
			has_event_target = murder_mystery_victim_character
			event_target:murder_mystery_victim_character = { is_character = ENG_stanley_baldwin }
		}
		# Recruitment is history-only; attach the defined character by nationality,
		# as the Assassin State setup does for its commanders.
		if = {
			limit = { NOT = { has_character = ENG_neville_chamberlain } }
			ENG_neville_chamberlain = { set_nationality = PREV }
		}
		promote_character = ENG_neville_chamberlain
		if = {
			limit = { has_country_leader = { character = ENG_neville_chamberlain ruling_only = yes } }
			set_country_flag = murder_mystery_curated_successor_used
		}
	}
}

# COUNTRY scope. Input: temp murder_mystery_successor_style. Creates an
# emergency council (opening murder) or acting cabinet (foreign murder) named
# from the country's adjective, in the ruling party's ideology, with an
# institutional portrait and a marker trait that verification reads.
murder_mystery_create_emergency_body = {

		limit = { check_variable = { var = murder_mystery_successor_style value = constant:murder_mystery_successor.style_acting_cabinet compare = equals } }

			limit = { has_government = democratic }
			meta_effect = {
				text = { create_country_leader = { name = "[BODY_NAME]" desc = MURDER_MYSTERY_ACTING_CABINET_DESC picture = GFX_portrait_039_emergency_council ideology = liberalism traits = { murder_mystery_acting_cabinet } } }

			}
		}
		else_if = {
			limit = { has_government = fascism }

			limit = { has_government = fascism }
			meta_effect = {
				text = { create_country_leader = { name = "[BODY_NAME]" desc = MURDER_MYSTERY_EMERGENCY_COUNCIL_DESC picture = GFX_portrait_039_emergency_council ideology = fascism_ideology traits = { murder_mystery_emergency_council } } }
				BODY_NAME = "[This.GetMurderMysteryEmergencyCouncilName]"

			meta_effect = {
				text = { create_country_leader = { name = "[BODY_NAME]" desc = MURDER_MYSTERY_EMERGENCY_COUNCIL_DESC picture = GFX_portrait_039_emergency_council ideology = marxism traits = { murder_mystery_emergency_council } } }
				BODY_NAME = "[This.GetMurderMysteryEmergencyCouncilName]"
			}
		}
