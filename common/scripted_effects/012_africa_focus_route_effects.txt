# Compatibility wrapper for older route call sites. The canonical loader owns
# the one-shot tree test, completion preservation, loaded flag, and layout
# refresh so those rules cannot drift between helpers.

	africa_load_continental_focus_tree = yes
}

# The seven constitutional axes are displayed on the Charter Council and are
# consumed by action pricing and risk. Initialising them from a route makes the
# constitutional choice a rules change before its first specialised focus.
africa_clamp_constitutional_payoff_axes = {
	clamp_variable = { var = africa_constitutional_representation min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
	clamp_variable = { var = africa_constitutional_executive_power min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
	clamp_variable = { var = africa_constitutional_resource_governance min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
	clamp_variable = { var = africa_constitutional_command_integration min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
	clamp_variable = { var = africa_constitutional_withdrawal_protection min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
	clamp_variable = { var = africa_constitutional_crisis_resilience min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
	clamp_variable = { var = africa_constitutional_post_unification_rule min = constant:africa_constitution_axis.minimum max = constant:africa_constitution_axis.maximum }
}

# Overlay completion is allowed before the constitutional principle is chosen.
# Keep those regional flags as the source of truth, then apply the route's
# overlay payoff when a constitution is committed. If an overlay is completed
# after commitment, the same helper applies its payoff immediately. Every
# route uses all seven live axes, but the six overlay milestones deliberately
# place the major and minor gains on different axes for each constitution.

# Replays completed overlay milestones at constitutional commitment. The
# normal overlay reward calls the same payoff helper immediately once a route
# is active; pre-commit milestones are therefore applied exactly once here.
africa_replay_constitutional_overlay_payoff = {
	if = { limit = { has_country_flag = africa_overlay_authority_settled } set_temp_variable = { africa_focus_overlay_step = constant:africa_focus_overlay_step.settle_authority } africa_apply_constitutional_overlay_payoff = yes }
	if = { limit = { has_country_flag = africa_overlay_corridor_charter_written } set_temp_variable = { africa_focus_overlay_step = constant:africa_focus_overlay_step.build_corridor } africa_apply_constitutional_overlay_payoff = yes }

# Event-driven AI hook. Crisis-opening focuses call it once. Action-resolution
# or capacity-recovery events should call it again while the pending flag is
# present; no daily/weekly/monthly all-country scan is authorised.
africa_focus_ai_retry_pending_route_action = {
	if = {
		limit = {
			is_ai = yes
			has_country_flag = africa_focus_route_action_ai_contract_pending
			africa_focus_pending_route_action_is_unresolved = yes
			africa_has_open_action_capacity = yes
		}
		set_temp_variable = { africa_requested_action_id = africa_focus_pending_route_action }
		if = {
			limit = {
				OR = {
					check_variable = { africa_focus_pending_route_action = constant:africa_action.conduct_first_continental_election }
					check_variable = { africa_focus_pending_route_action = constant:africa_action.hold_postwar_constitutional_review }
				}
			}
			africa_save_host_as_selected_country_target = yes
		}
		else = {
			random_scope_in_array = {
				array = africa_selected_country_targets
				limit = {
					africa_action_target_is_usable = yes
					OR = {
						AND = { ROOT = { check_variable = { africa_focus_pending_route_action = constant:africa_action.convene_federal_deadlock_conference } } }
						AND = { ROOT = { check_variable = { africa_focus_pending_route_action = constant:africa_action.arbitrate_continental_succession } } has_country_flag = africa_succession_dispute_active }
						AND = { ROOT = { check_variable = { africa_focus_pending_route_action = constant:africa_action.balance_food_and_industrial_plan } } has_country_flag = africa_plan_balance_crisis_active }
						AND = { ROOT = { check_variable = { africa_focus_pending_route_action = constant:africa_action.review_victorious_commander_loyalty } } has_country_flag = africa_commander_loyalty_review_target }
						AND = { ROOT = { check_variable = { africa_focus_pending_route_action = constant:africa_action.ratify_confederal_emergency_action } } has_country_flag = africa_confederal_emergency_ratification_target }
						AND = { ROOT = { check_variable = { africa_focus_pending_route_action = constant:africa_action.review_covenant_obligation } } has_country_flag = africa_covenant_obligation_target }
					}
				}
				save_event_target_as = africa_focus_route_action_target
			}
			if = {
				limit = { has_event_target = africa_focus_route_action_target }
				africa_refresh_selected_action_state_candidates = yes
			}
		}
		if = {
			limit = {
				check_variable = { africa_focus_pending_route_action = constant:africa_action.balance_food_and_industrial_plan }
				check_variable = { africa_selected_action_state_candidates^num > constant:africa_value.zero }
			}
			random_scope_in_array = { array = africa_selected_action_state_candidates save_event_target_as = africa_focus_route_action_state }
			if = {
				limit = { has_event_target = africa_focus_route_action_state }
				add_to_array = { array = africa_selected_action_state_targets value = event_target:africa_focus_route_action_state }
				event_target:africa_focus_route_action_state = { set_state_flag = africa_selected_action_state }
			}
		}
		africa_select_action_for_quote = yes
		africa_begin_quoted_action_against_target = yes
		if = {
			limit = { check_variable = { africa_action_begin_result > constant:africa_value.zero } }
			clr_country_flag = africa_focus_route_action_ai_contract_pending
		}
	}
}

# COUNTRY scope on an active action target. The same constitutional rules alter
# the action's risk tier once, after its immutable record is written. High
# fitness lowers risk by one band; a weak relevant institution raises it.
africa_apply_constitutional_action_risk_adjustment = {
	if = {
		limit = {
			has_event_target = africa_host
			event_target:africa_host = { has_country_flag = africa_constitutional_payoff_axes_active }
		}
		set_temp_variable = { africa_action_axis_fitness = constant:africa_value.zero }
		event_target:africa_host = {
			if = { limit = { ROOT = { check_variable = { africa_active_action_family = constant:africa_action_family.protection } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_command_integration } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_crisis_resilience } }
			else_if = { limit = { ROOT = { OR = { check_variable = { africa_active_action_family = constant:africa_action_family.accession } check_variable = { africa_active_action_family = constant:africa_action_family.regional_congress } check_variable = { africa_active_action_family = constant:africa_action_family.integration } } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_representation } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_withdrawal_protection } }
			else_if = { limit = { ROOT = { check_variable = { africa_active_action_family = constant:africa_action_family.economy } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_resource_governance } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_resource_governance } }
			else_if = { limit = { ROOT = { check_variable = { africa_active_action_family = constant:africa_action_family.diaspora } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_withdrawal_protection } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_representation } }
			else_if = { limit = { ROOT = { check_variable = { africa_active_action_family = constant:africa_action_family.rival_bloc } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_withdrawal_protection } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_crisis_resilience } }
			else_if = { limit = { ROOT = { check_variable = { africa_active_action_family = constant:africa_action_family.high_chaos } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_crisis_resilience } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_command_integration } }
			else_if = { limit = { ROOT = { check_variable = { africa_active_action_family = constant:africa_action_family.scramble } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_command_integration } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_crisis_resilience } }
			else_if = { limit = { ROOT = { OR = { check_variable = { africa_active_action_family = constant:africa_action_family.world_order } check_variable = { africa_active_action_family = constant:africa_action_family.post_unification } } } } set_temp_variable = { africa_action_axis_fitness = africa_constitutional_post_unification_rule } add_to_temp_variable = { africa_action_axis_fitness = africa_constitutional_crisis_resilience } }

					var = africa_action_axis_fitness

# Event-driven AI equivalent for the overlay's exact physical project. It uses
# only the bounded relationship and controlled-state arrays and is retried from
# action-resolution hooks, never from a recurring all-country on_action.
africa_focus_ai_retry_overlay_corridor_action = {
	if = {
		limit = {
			is_ai = yes
			has_country_flag = africa_overlay_corridor_ai_contract_pending
			africa_has_open_action_capacity = yes
		}
		africa_clear_selected_country_target = yes
		random_scope_in_array = {
			array = africa_relationship_countries
			limit = {
				africa_action_target_is_usable = yes
				africa_matches_current_host_overlay = yes
				OR = {
					africa_relationship_is_protected = yes
					africa_relationship_is_associate = yes
					africa_relationship_is_chartered = yes
					africa_relationship_is_autonomous_federal = yes
					africa_relationship_is_integrated_region = yes
				}
			}
			save_event_target_as = africa_overlay_corridor_ai_target
		}
		if = {
			limit = { has_event_target = africa_overlay_corridor_ai_target }
			set_variable = { africa_selected_country_id = event_target:africa_overlay_corridor_ai_target.id }
			add_to_array = { array = africa_selected_country_targets value = event_target:africa_overlay_corridor_ai_target }
			event_target:africa_overlay_corridor_ai_target = { set_country_flag = africa_selected_country_target }
			africa_refresh_selected_action_state_candidates = yes
			set_temp_variable = { africa_requested_action_id = africa_overlay_corridor_required_action }
			africa_select_action_for_quote = yes
			if = {
				limit = { check_variable = { africa_overlay_corridor_required_action = constant:africa_action.build_regional_rail_spine } }
				random_scope_in_array = {
					array = africa_selected_action_state_candidates
					limit = {
						NOT = { is_capital = yes }
						CONTROLLER = {
							ROOT = {
								can_build_railway = {
									level = constant:africa_action_building.rail_level
									start_state = THIS.capital
									target_state = PREV
								}
							}
						}
					}
					save_event_target_as = africa_overlay_corridor_ai_state
				}
			}
			else_if = {
				limit = { check_variable = { africa_overlay_corridor_required_action = constant:africa_action.expand_river_transport } }
				random_scope_in_array = {
					array = africa_selected_action_state_candidates
					limit = { africa_state_is_registered_river_or_lake_project_site = yes }
					save_event_target_as = africa_overlay_corridor_ai_state
				}
			}
			else_if = {
				limit = { check_variable = { africa_overlay_corridor_required_action = constant:africa_action.modernise_continental_port } }
				random_scope_in_array = {
					array = africa_selected_action_state_candidates
					limit = {
						is_coastal = yes
						naval_base > constant:africa_value.zero
					}
					save_event_target_as = africa_overlay_corridor_ai_state
				}
			}
			if = {
				limit = { has_event_target = africa_overlay_corridor_ai_state }
				add_to_array = { array = africa_selected_action_state_targets value = event_target:africa_overlay_corridor_ai_state }
				event_target:africa_overlay_corridor_ai_state = { set_state_flag = africa_selected_action_state }
				africa_refresh_selected_action_quote = yes
				africa_begin_quoted_action_against_target = yes
				if = {
					limit = { check_variable = { africa_action_begin_result > constant:africa_value.zero } }
					clr_country_flag = africa_overlay_corridor_ai_contract_pending
				}
			}
		}
	}
}
