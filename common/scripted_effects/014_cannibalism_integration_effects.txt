# ============================================================================
# CHAOS REDUX - EVENT 014 CROSS-EVENT INTEGRATION EFFECTS
# ============================================================================
# Applies shortage pressure, routes Event 14 relief into the existing disaster
# and famine systems, damages cells exposed to contamination, and maintains the
# explicit exploitation/protection lifecycle for detention sites.
# ============================================================================

cannibalism_apply_external_pressure_to_current_actor = {
	if = {
		limit = { cannibalism_country_has_external_hunger_pressure = yes }
		add_to_variable = { cannibalism_field_hunger = constant:cannibalism_integration.weekly_shortage_hunger }
		set_country_flag = cannibalism_external_shortage_pressure_active
	}
	else = { clr_country_flag = cannibalism_external_shortage_pressure_active }
	if = {
		limit = { cannibalism_country_has_severe_external_hunger_pressure = yes }
		add_to_variable = { cannibalism_field_hunger = constant:cannibalism_integration.weekly_severe_shortage_hunger }
		set_country_flag = cannibalism_external_shortage_pressure_severe
	}
	else = { clr_country_flag = cannibalism_external_shortage_pressure_severe }
}

cannibalism_process_current_state_contamination_against_cell = {
	if = {
		limit = {
			cannibalism_state_has_active_cell = yes
			cannibalism_state_has_any_larder_contamination = yes
			NOT = { has_state_flag = cannibalism_contamination_cell_damage_cooldown }
			CONTROLLER = {
				NOT = {
					AND = {
						is_cannibalism_wendigo_hannibal_country = yes
						has_country_flag = cannibalism_wendigo_transformation_locked
					}
				}
			}
		}
		set_temp_variable = { cannibalism_contamination_cell_damage = 0 }
		set_temp_variable = { cannibalism_contamination_evidence_gain = 0 }
		if = {
			limit = { has_dynamic_modifier = { modifier = nuclear_fallout_state } }
			set_temp_variable = { cannibalism_contamination_cell_damage = constant:cannibalism_integration.fallout_cell_damage }
			set_state_flag = cannibalism_cell_exposed_to_fallout
		}
		else_if = {
			limit = { cannibalism_state_has_bioweapon_contamination = yes }
			set_temp_variable = { cannibalism_contamination_cell_damage = constant:cannibalism_integration.biological_cell_damage }
			set_temp_variable = { cannibalism_contamination_evidence_gain = constant:cannibalism_integration.biological_evidence_gain }
			set_state_flag = cannibalism_cell_exposed_to_biological_attack
		}
		else_if = {
			limit = { cannibalism_state_has_chemical_contamination = yes }
			set_temp_variable = { cannibalism_contamination_cell_damage = constant:cannibalism_integration.chemical_cell_damage }
			set_temp_variable = { cannibalism_contamination_evidence_gain = constant:cannibalism_integration.chemical_evidence_gain }
			set_state_flag = cannibalism_cell_exposed_to_chemical_attack
		}
		if = {
			limit = { has_variable = cannibalism_node_strength }
			add_to_variable = { cannibalism_node_strength = cannibalism_contamination_cell_damage }
			clamp_variable = { var = cannibalism_node_strength min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		}
		CONTROLLER = {
			if = {
				limit = { is_cannibalism_active_country = yes }
				add_to_variable = { cannibalism_cell_strength = cannibalism_contamination_cell_damage }
				add_to_variable = { cannibalism_evidence = cannibalism_contamination_evidence_gain }
				cannibalism_clamp_current_country_evidence = yes
				cannibalism_clamp_current_country_meters = yes
			}
		}
		set_temp_variable = { cannibalism_contamination_cell_damage_days = constant:cannibalism_integration.contamination_cell_damage_cooldown_days }
		set_state_flag = { flag = cannibalism_contamination_cell_damage_cooldown days = cannibalism_contamination_cell_damage_days }
	}
}

cannibalism_apply_shared_relief_to_current_state = {
	if = {
		limit = { has_state_flag = natural_disaster_aftermath_active }
		if = {
			limit = { check_variable = { natural_disaster_phase = constant:natural_disaster_phase.early_rescue } }
			add_to_variable = { natural_disaster_rescue_score = constant:cannibalism_integration.shared_relief_score }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_phase = constant:natural_disaster_phase.middle_stabilization } }
			add_to_variable = { natural_disaster_stabilization_score = constant:cannibalism_integration.shared_relief_score }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_phase = constant:natural_disaster_phase.late_reconstruction } }
			add_to_variable = { natural_disaster_reconstruction_score = constant:cannibalism_integration.shared_relief_score }
		}
		clr_state_flag = natural_disaster_famine_pressure
		clr_state_flag = natural_disaster_supply_collapse
		set_temp_variable = { cannibalism_shared_relief_days = constant:cannibalism_integration.shared_relief_days }
		set_state_flag = { flag = natural_disaster_inbound_relief_arrived days = cannibalism_shared_relief_days }
		set_state_flag = { flag = cannibalism_shared_relief_delivered days = cannibalism_shared_relief_days }
	}
	if = {
		limit = {
			OR = {
			has_state_flag = genocide_soviet_famine_pressure_site
			has_state_flag = camp_rework_soviet_famine_pressure
			has_dynamic_modifier = { modifier = sov_famine_pressure_state }
			}
		}
		OWNER = {
			if = {
				limit = { check_variable = { famine_pressure > 0 } }
				add_to_variable = { famine_pressure = constant:cannibalism_integration.shared_relief_famine_reduction }
				clamp_variable = { var = famine_pressure min = 0 max = constant:camp_rework_threshold.max_value }
			}
		}
		set_temp_variable = { cannibalism_shared_relief_days = constant:cannibalism_integration.shared_relief_days }
		set_state_flag = { flag = cannibalism_shared_relief_delivered days = cannibalism_shared_relief_days }
	}
}

cannibalism_apply_shared_medical_protection_to_current_state = {
	if = {
		limit = {
			has_state_flag = natural_disaster_aftermath_active
			has_state_flag = natural_disaster_disease_pressure
		}
		clr_state_flag = natural_disaster_disease_pressure
		if = {
			limit = { check_variable = { natural_disaster_phase = constant:natural_disaster_phase.early_rescue } }
			add_to_variable = { natural_disaster_rescue_score = constant:cannibalism_integration.shared_relief_score }
		}
		else_if = {
			limit = { check_variable = { natural_disaster_phase = constant:natural_disaster_phase.middle_stabilization } }
			add_to_variable = { natural_disaster_stabilization_score = constant:cannibalism_integration.shared_relief_score }
		}
		set_temp_variable = { cannibalism_shared_relief_days = constant:cannibalism_integration.shared_relief_days }
		set_state_flag = { flag = natural_disaster_foreign_medical_mission days = cannibalism_shared_relief_days }
		set_state_flag = { flag = cannibalism_shared_medical_protection days = cannibalism_shared_relief_days }
	}
}

cannibalism_record_detention_exploitation_links = {
	every_controlled_state = {
		limit = {
			cannibalism_state_has_prison_or_camp = yes
			cannibalism_state_has_active_cell = yes
			NOT = { has_state_flag = cannibalism_feeding_program_link }
		}
		set_state_flag = cannibalism_feeding_program_link
		set_state_flag = cannibalism_prisoners_abandoned_to_cell
		if = {
			limit = { has_variable = cannibalism_node_strength }
			add_to_variable = { cannibalism_node_strength = constant:cannibalism_integration.camp_link_cell_gain }
			clamp_variable = { var = cannibalism_node_strength min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		}
	}
	if = {
		limit = { cannibalism_country_has_active_detention_exploitation_link = yes }
		add_to_variable = { cannibalism_cell_strength = constant:cannibalism_integration.camp_link_cell_gain }
		if = {
			limit = { has_variable = cannibalism_cult_cohesion }
			add_to_variable = { cannibalism_cult_cohesion = constant:cannibalism_integration.camp_link_cult_gain }
		}
		add_to_variable = { cannibalism_evidence = constant:cannibalism_integration.camp_link_evidence_gain }
		set_country_flag = cannibalism_detention_exploitation_active
		set_global_flag = cannibalism_detention_exploitation_occurred
	}
}

cannibalism_protect_detention_sites_from_current_policy = {
	every_controlled_state = {
		limit = {
			cannibalism_state_has_prison_or_camp = yes
			OR = {
				has_state_flag = cannibalism_feeding_program_link
				has_state_flag = cannibalism_prisoners_abandoned_to_cell
			}
		}
		clr_state_flag = cannibalism_feeding_program_link
		clr_state_flag = cannibalism_prisoners_abandoned_to_cell
		set_state_flag = cannibalism_prisoners_under_protected_custody
		if = {
			limit = { has_variable = cannibalism_node_strength }
			add_to_variable = { cannibalism_node_strength = constant:cannibalism_integration.camp_protection_cell_damage }
			clamp_variable = { var = cannibalism_node_strength min = constant:cannibalism_meter.min max = constant:cannibalism_meter.max }
		}
	}
	clr_country_flag = cannibalism_detention_exploitation_active
	add_to_variable = { cannibalism_command_integrity = constant:cannibalism_integration.camp_protection_integrity_gain }
}

cannibalism_expose_detention_exploitation_in_current_country = {
	if = {
		limit = { cannibalism_country_has_active_detention_exploitation_link = yes }
		add_to_variable = { cannibalism_evidence = constant:cannibalism_integration.camp_exposure_evidence_gain }
		set_country_flag = cannibalism_feeding_program_exposed
		set_country_flag = cannibalism_clean_route_disqualified
		set_global_flag = cannibalism_feeding_program_exposed_abroad
		cannibalism_clamp_current_country_evidence = yes
	}
}

cannibalism_log_mixed_prisoner_feeding_deaths = {
	set_temp_variable = { cannibalism_prisoner_military_deaths = cannibalism_population_loss_applied }
	multiply_temp_variable = { cannibalism_prisoner_military_deaths = constant:cannibalism_integration.prisoner_feeding_military_share }
	round_temp_variable = cannibalism_prisoner_military_deaths
	set_temp_variable = { cannibalism_prisoner_civilian_deaths = cannibalism_population_loss_applied }
	subtract_from_temp_variable = { cannibalism_prisoner_civilian_deaths = cannibalism_prisoner_military_deaths }
	if = {
		limit = { check_variable = { cannibalism_prisoner_civilian_deaths > 0 } }
		set_temp_variable = { chaos_deaths_change = cannibalism_prisoner_civilian_deaths }
		set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.cannibalism_consumption }
		set_temp_variable = { chaos_deaths_is_civilian = 1 }
		set_temp_variable = { chaos_deaths_apply_state_pop = 0 }
		set_temp_variable = { chaos_deaths_target_country = event_target:cannibalism_consumption_actor }
		set_temp_variable = { chaos_deaths_has_target_country = 1 }
		chaos_meter_register_deaths = yes
	}
	if = {
		limit = { check_variable = { cannibalism_prisoner_military_deaths > 0 } }
		set_temp_variable = { chaos_deaths_change = cannibalism_prisoner_military_deaths }
		set_temp_variable = { chaos_deaths_reason = constant:chaos_meter_deaths_reason.cannibalism_consumption }
		set_temp_variable = { chaos_deaths_is_civilian = 0 }
		set_temp_variable = { chaos_deaths_apply_state_pop = 0 }
		set_temp_variable = { chaos_deaths_target_country = event_target:cannibalism_consumption_actor }
		set_temp_variable = { chaos_deaths_has_target_country = 1 }
		chaos_meter_register_deaths = yes
	}
}

cannibalism_pay_selected_prisoner_feeding_cost = {

	cannibalism_pay_current_command_cost = yes

}

cannibalism_execute_selected_prisoner_feeding = {
	if = {
		limit = {
			cannibalism_can_pay_selected_prisoner_feeding_cost = yes
			has_event_target = cannibalism_selected_prisoner_feeding_state
			event_target:cannibalism_selected_prisoner_feeding_state = { cannibalism_state_can_feed_selected_prisoners = yes }
		}
		cannibalism_pay_selected_prisoner_feeding_cost = yes
		cannibalism_record_detention_exploitation_links = yes
		event_target:cannibalism_selected_prisoner_feeding_state = {
			save_event_target_as = cannibalism_consumption_target_state
			set_state_flag = cannibalism_feeding_program_link
			set_state_flag = cannibalism_prisoners_abandoned_to_cell
		}
		set_temp_variable = { cannibalism_consumption_exact_request_people = constant:cannibalism_integration.prisoner_feeding_people }
		set_temp_variable = { cannibalism_consumption_context = constant:cannibalism_consumption_mode.prisoner_feeding }
		cannibalism_prepare_consumption_context = yes
		if = {
			limit = { check_variable = { cannibalism_consumption_result = constant:cannibalism_consumption_result.applied } }
			add_to_variable = { cannibalism_evidence = constant:cannibalism_integration.prisoner_feeding_evidence_gain }
			add_to_variable = { cannibalism_cell_strength = constant:cannibalism_integration.prisoner_feeding_cell_gain }
			if = {
				limit = { has_variable = cannibalism_cult_cohesion }
				add_to_variable = { cannibalism_cult_cohesion = constant:cannibalism_integration.prisoner_feeding_cult_gain }
			}
			add_to_variable = { cannibalism_selected_prisoner_feedings = 1 }
			set_country_flag = cannibalism_selected_prisoner_feeding_used
			set_country_flag = cannibalism_clean_route_disqualified
			set_country_flag = cannibalism_policy_exploitation_unresolved
			cannibalism_achievement_record_exploitation = yes
			set_temp_variable = { cannibalism_chaos_change = constant:cannibalism_integration.prisoner_feeding_chaos }
			set_temp_variable = { cannibalism_chaos_reason = constant:chaos_meter_history_reason.special.cannibalism_spread }
			cannibalism_apply_chaos_change = yes
			cannibalism_clamp_current_country_evidence = yes
			cannibalism_clamp_current_country_meters = yes
			cannibalism_refresh_country_penalty_variables = yes
		}
	}
}
