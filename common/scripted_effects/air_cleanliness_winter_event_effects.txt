# ============================================================================
# CHAOS REDUX - AIR WINTER EVENT EFFECTS
# ============================================================================
# Public contract:
#   air_winter_event_prepare_candidate_cycle = yes (COUNTRY/host before pass)
#   air_winter_schedule_phase_event = yes  (STATE during the monthly pass)
#   air_winter_dispatch_phase_events = yes  (COUNTRY/host after the pass)
#   air_winter_event_refresh_state = yes   (STATE after an event choice)
#   air_winter_event_apply_deaths = yes    (STATE with temp death percent)
#
# The scheduler adds no country or state scan. It records candidates during the
# existing monthly state pass, compares them by score and state id, then uses a
# bounded owner array for post-pass dispatch.
# ============================================================================

# Supported scope: STATE. The regular country target is preferred because it
# preserves the country that made the original choice.
air_winter_event_bind_pending_chain = {
	set_state_flag = air_winter_pending_delayed_result
	if = {
		limit = { has_event_target = air_winter_event_country }
		set_variable = { air_winter_pending_event_owner = event_target:air_winter_event_country }
	}
	else = { set_variable = { air_winter_pending_event_owner = OWNER } }
}

# Supported scope: STATE
air_winter_event_reconcile_pending_chain = {
	if = {
		limit = {
			has_state_flag = air_winter_pending_delayed_result
			OR = {
				NOT = { air_winter_event_has_pending_chain = yes }
				NOT = { air_winter_event_pending_owner_is_valid = yes }
			}
		}
		air_winter_event_cancel_pending_chain = yes
	}
}

# Supported scope: COUNTRY. Used only when ownership changed after a visible
# event opened. It never touches a newer owner's delayed chain.
air_winter_event_reject_stale_choice = {
	if = {
		limit = {
			has_event_target = air_winter_event_state
			has_event_target = air_winter_event_country
			event_target:air_winter_event_state = {
				has_state_flag = air_winter_pending_delayed_result
				has_variable = air_winter_pending_event_owner
				check_variable = {
					var = air_winter_pending_event_owner
					value = event_target:air_winter_event_country
					compare = equals
				}
			}
		}
		event_target:air_winter_event_state = { air_winter_event_cancel_pending_chain = yes }
	}
	country_event = { id = chaosx.fallout.203 }
}

# Supported scope: STATE
air_winter_event_refresh_state = {
	air_winter_normalize_state_ledger = yes
	air_winter_calculate_survival_value = yes
	set_variable = { air_winter_last_event_date = global.date }
	if = {
		limit = {
			air_winter_event_has_pending_chain = yes
			NOT = { has_state_flag = air_winter_pending_delayed_result }
		}
		air_winter_event_bind_pending_chain = yes
	}
	air_winter_event_reconcile_pending_chain = yes
}

# Supported scope: STATE
# Input: temporary variable air_winter_event_death_percent
air_winter_event_apply_deaths = {
	if = {
		limit = { NOT = { has_global_flag = settings_chaos_deaths_disabled } }
		set_temp_variable = { air_winter_event_death_request = state_population_k }
		multiply_temp_variable = { air_winter_event_death_request = constant:air_winter_event_deaths.population_scale }
		multiply_temp_variable = { air_winter_event_death_request = air_winter_event_death_percent }
		round_temp_variable = air_winter_event_death_request
		set_temp_variable = { state_civilian_population_loss_requested = air_winter_event_death_request }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:air_winter_event_deaths.minimum_remaining }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:air_winter_event_runtime.one }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.air_winter_exposure }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:air_winter_event_runtime.one }
		set_temp_variable = { state_civilian_population_loss_applied = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_result = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.one }
		apply_exact_state_civilian_population_loss = yes
		if = {
			limit = { check_variable = { state_civilian_population_loss_applied > constant:air_winter_event_runtime.zero } }
			add_to_variable = { air_winter_population_loss_memory = state_civilian_population_loss_applied }
		}
	}
	set_temp_variable = { air_winter_event_death_request = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_event_death_percent = constant:air_winter_event_runtime.zero }
}

# Supported scope: STATE. Called once from the existing monthly state pass.
# It records one deterministic candidate per owner and never fires directly.
air_winter_schedule_phase_event = {
	set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.none }
	if = {
		limit = {
			OWNER = {
				exists = yes
				NOT = { has_country_flag = air_winter_event_cooldown }
			}
		}
		if = {
			limit = {
				air_winter_is_active = yes
				NOT = { air_winter_presentation_is_unclassified = yes }
			}
			if = {
				limit = {
					air_winter_is_phase_1 = yes
					OWNER = { NOT = { has_country_flag = air_winter_event_seen_phase_1 } }
				}
				if = {
					limit = {
						OR = {
							air_winter_presentation_is_boreal = yes
							air_winter_presentation_is_polar = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_1_boreal }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_maritime = yes
							air_winter_presentation_is_oceanic = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_1_maritime }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_arid = yes
							air_winter_presentation_is_mediterranean = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_1_arid }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_tropical = yes
							air_winter_presentation_is_equatorial = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_1_tropical }
				}
				else_if = {
					limit = { air_winter_presentation_is_highland = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_1_highland }
				}
			}
			else_if = {
				limit = {
					air_winter_is_phase_2 = yes
					OWNER = { NOT = { has_country_flag = air_winter_event_seen_phase_2 } }
				}
				if = {
					limit = { air_winter_response_is_city = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_2_city }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_maritime = yes
							air_winter_presentation_is_oceanic = yes
							air_winter_presentation_is_tropical = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_2_coast }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_arid = yes
							air_winter_presentation_is_mediterranean = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_2_arid }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_highland = yes
							air_winter_presentation_is_polar = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_2_highland }
				}
				else_if = {
					limit = {
						OR = {
							air_winter_presentation_is_boreal = yes
							air_winter_presentation_is_equatorial = yes
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_2_food }
				}
			}
			else_if = {
				limit = {
					air_winter_is_phase_3 = yes
					OWNER = { NOT = { has_country_flag = air_winter_event_seen_phase_3 } }
				}
				if = {
					limit = { air_winter_response_is_transport_state = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_3_transport }
				}
				else = { set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_3_clinics } }
			}
			else_if = {
				limit = {
					air_winter_is_phase_4 = yes
					OWNER = { NOT = { has_country_flag = air_winter_event_seen_phase_4 } }
				}
				if = {
					limit = { air_winter_response_is_city = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_4_city }
				}
				else_if = {
					limit = { air_winter_response_is_food_state = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_4_food }
				}
				else = { set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_4_shelter } }
			}
			else_if = {
				limit = {
					air_winter_is_phase_5 = yes
					OWNER = { NOT = { has_country_flag = air_winter_event_seen_phase_5 } }
				}
				if = {
					limit = { air_winter_response_is_city = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_5_city }
				}
				else_if = {
					limit = { air_winter_has_low_shelter = yes }
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_5_abandonment }
				}
				else = { set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_5_archive } }
			}
			else_if = {
				limit = {
					air_winter_is_phase_6 = yes
					OWNER = { NOT = { has_country_flag = air_winter_event_seen_phase_6 } }
				}
				set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.phase_6_terminal }
			}
		}
		if = {
			limit = {
				check_variable = { air_winter_selected_event_id = constant:air_winter_event_id.none }
				NOT = { air_winter_presentation_is_unclassified = yes }
				has_state_flag = air_winter_phase_changed_this_tick
				check_variable = {
					var = air_winter_phase
					value = air_winter_previous_phase
					compare = less_than
				}
			}
			OWNER = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable = air_winter_recovery_event_count }
							check_variable = { air_winter_recovery_event_count < constant:air_winter_event_runtime.recovery_arc_cap }
						}
					}
					set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.recovery }
				}
			}
		}

		if = {
			limit = { check_variable = { air_winter_selected_event_id > constant:air_winter_event_id.none } }
			set_temp_variable = { air_winter_event_candidate_state_id = THIS.id }
			set_temp_variable = { air_winter_event_candidate_score_value = air_winter_phase }
			multiply_temp_variable = { air_winter_event_candidate_score_value = constant:air_winter_event_runtime.candidate_phase_weight }
			set_temp_variable = { air_winter_event_candidate_pressure_score = air_winter_pressure }
			multiply_temp_variable = { air_winter_event_candidate_pressure_score = constant:air_winter_event_runtime.candidate_pressure_weight }
			add_to_temp_variable = { air_winter_event_candidate_score_value = air_winter_event_candidate_pressure_score }
			if = {
				limit = { check_variable = { air_winter_selected_event_id = constant:air_winter_event_id.recovery } }
				set_temp_variable = { air_winter_event_candidate_score_value = constant:air_winter_event_runtime.candidate_recovery_base }
				add_to_temp_variable = { air_winter_event_candidate_score_value = air_winter_event_candidate_pressure_score }
			}
			OWNER = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable = air_winter_event_candidate_cycle_id }
							NOT = {
								check_variable = {
									var = air_winter_event_candidate_cycle_id
									value = global.air_winter_cycle_id
									compare = equals
								}
							}
							check_variable = {
								var = air_winter_event_candidate_score_value
								value = air_winter_event_candidate_score
								compare = greater_than
							}
							AND = {
								check_variable = {
									var = air_winter_event_candidate_score_value
									value = air_winter_event_candidate_score
									compare = equals
								}
								check_variable = {
									var = air_winter_event_candidate_state_id
									value = air_winter_event_candidate_state
									compare = less_than
								}
							}
						}
					}
					set_variable = { air_winter_event_candidate_state = air_winter_event_candidate_state_id }
					set_variable = { air_winter_event_candidate_id = air_winter_selected_event_id }
					set_variable = { air_winter_event_candidate_score = air_winter_event_candidate_score_value }
					set_variable = { air_winter_event_candidate_cycle_id = global.air_winter_cycle_id }
				}
				if = {
					limit = {
						NOT = {
							is_in_array = {
								array = global.air_winter_event_candidate_countries
								value = THIS
							}
						}
					}
					add_to_array = { array = global.air_winter_event_candidate_countries value = THIS }
				}
			}
		}
	}
	set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.none }
	set_temp_variable = { air_winter_event_candidate_state_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_event_candidate_score_value = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_event_candidate_pressure_score = constant:air_winter_event_runtime.zero }
}

# Supported scope: COUNTRY (the existing host country). Iterates only the
# bounded country array collected during the existing state pass.
air_winter_dispatch_phase_events = {
	for_each_scope_loop = {
		array = global.air_winter_event_candidate_countries
		if = {
			limit = {
				has_global_flag = air_winter_cycle_open
				exists = yes
				NOT = { has_country_flag = air_winter_event_cooldown }
				has_variable = air_winter_event_candidate_state
				has_variable = air_winter_event_candidate_id
				has_variable = air_winter_event_candidate_cycle_id
				check_variable = {
					var = air_winter_event_candidate_cycle_id
					value = global.air_winter_cycle_id
					compare = equals
				}
				check_variable = { air_winter_event_candidate_id > constant:air_winter_event_id.none }
				var:air_winter_event_candidate_state = {
					air_winter_state_is_valid = yes
					is_owned_by = PREV
				}
			}
			save_event_target_as = air_winter_event_country
			var:air_winter_event_candidate_state = { save_event_target_as = air_winter_event_state }
			if = {
				limit = { check_variable = { air_winter_event_candidate_id = constant:air_winter_event_id.recovery } }
				add_to_variable = { air_winter_recovery_event_count = constant:air_winter_event_runtime.one }
			}
			else = {
				var:air_winter_event_candidate_state = {
					if = { limit = { air_winter_is_phase_1 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_1 } }
					else_if = { limit = { air_winter_is_phase_2 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_2 } }
					else_if = { limit = { air_winter_is_phase_3 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_3 } }
					else_if = { limit = { air_winter_is_phase_4 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_4 } }
					else_if = { limit = { air_winter_is_phase_5 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_5 } }
					else_if = { limit = { air_winter_is_phase_6 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_6 } }
				}
			}
			set_country_flag = {
				flag = air_winter_event_cooldown
				days = @AIR_WINTER_EVENT_COUNTRY_COOLDOWN_DAYS
			}
			meta_effect = {
				text = { country_event = { id = chaosx.fallout.[AIR_WINTER_EVENT_ID] } }
				AIR_WINTER_EVENT_ID = "[?air_winter_event_candidate_id|.0]"
			}
		}
		air_winter_event_clear_candidate = yes
	}
	clear_array = global.air_winter_event_candidate_countries
}
