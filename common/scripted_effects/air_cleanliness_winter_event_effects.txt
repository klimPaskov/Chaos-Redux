# ============================================================================
# CHAOS REDUX - AIR WINTER EVENT EFFECTS
# ============================================================================
# Public contract:
#   air_winter_event_prepare_candidate_cycle = yes (COUNTRY/host before pass)
#   air_winter_schedule_phase_event = yes  (STATE during the monthly pass)
#   air_winter_dispatch_phase_events = yes  (COUNTRY/host after the pass)
#   air_winter_event_refresh_state = yes   (STATE after an event choice)
#   air_winter_event_apply_deaths = yes    (STATE with temp death percent)
#
# The scheduler adds no country or state scan. It records candidates during the
# existing monthly state pass, compares typed priority, origin cycle, score,
# and state id, then uses a bounded owner array for post-pass dispatch.
# ============================================================================

# Supported scope: STATE. The regular country target is preferred because it
# preserves the country that made the original choice.
air_winter_event_bind_pending_chain = {
	set_state_flag = air_winter_pending_delayed_result
	if = {
		limit = { has_event_target = air_winter_event_country }
		set_variable = { air_winter_pending_event_owner = event_target:air_winter_event_country }
	}
	else = { set_variable = { air_winter_pending_event_owner = OWNER } }
}

# Supported scope: STATE
air_winter_event_reconcile_pending_chain = {
	if = {
		limit = {
			has_state_flag = air_winter_pending_delayed_result
			OR = {
				NOT = { air_winter_event_has_pending_chain = yes }
				NOT = { air_winter_event_pending_owner_is_valid = yes }
			}
		}
		air_winter_event_cancel_pending_chain = yes
	}
}

# Supported scope: COUNTRY. Used only when ownership changed after a visible
# event opened. It never touches a newer owner's delayed chain.
air_winter_event_reject_stale_choice = {
	if = {
		limit = {
			has_event_target = air_winter_event_state
			has_event_target = air_winter_event_country
			event_target:air_winter_event_state = {
				has_state_flag = air_winter_pending_delayed_result
				has_variable = air_winter_pending_event_owner
				check_variable = {
					var = air_winter_pending_event_owner
					value = event_target:air_winter_event_country
					compare = equals
				}
			}
		}
		event_target:air_winter_event_state = { air_winter_event_cancel_pending_chain = yes }
	}
	if = {
		limit = {
			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_active }
		}
		country_event = { id = chaosx.fallout.203 }
	}
}

# Supported scope: STATE
air_winter_event_refresh_state = {
	air_winter_normalize_state_ledger = yes
	air_winter_calculate_survival_value = yes
	set_variable = { air_winter_last_event_date = global.date }
	if = {
		limit = {
			air_winter_event_has_pending_chain = yes
			NOT = { has_state_flag = air_winter_pending_delayed_result }
		}
		air_winter_event_bind_pending_chain = yes
	}
	air_winter_event_reconcile_pending_chain = yes
}

# Input: temporary variable air_winter_event_death_percent
air_winter_event_apply_deaths = {
	if = {
		limit = { NOT = { has_global_flag = settings_chaos_deaths_disabled } }
		set_temp_variable = { air_winter_event_death_request = state_population_k }
		multiply_temp_variable = { air_winter_event_death_request = constant:air_winter_event_deaths.population_scale }
		multiply_temp_variable = { air_winter_event_death_request = air_winter_event_death_percent }
		round_temp_variable = air_winter_event_death_request
		set_temp_variable = { state_civilian_population_loss_requested = air_winter_event_death_request }
		set_temp_variable = { state_civilian_population_loss_minimum_remaining = constant:air_winter_event_deaths.minimum_remaining }
		set_temp_variable = { state_civilian_population_loss_log_deaths = constant:air_winter_event_runtime.one }
		set_temp_variable = { state_civilian_population_loss_reason = constant:chaos_meter_deaths_reason.air_winter_exposure }
		set_temp_variable = { state_civilian_population_loss_target_country = OWNER }
		set_temp_variable = { state_civilian_population_loss_has_target_country = constant:air_winter_event_runtime.one }
		set_temp_variable = { state_civilian_population_loss_applied = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_result = constant:chaos_meter_deaths.zero }
		set_temp_variable = { state_civilian_population_loss_contract_supplied = constant:chaos_meter_deaths.one }
		apply_exact_state_civilian_population_loss = yes
		if = {
			limit = { check_variable = { state_civilian_population_loss_applied > constant:air_winter_event_runtime.zero } }
			add_to_variable = { air_winter_population_loss_memory = state_civilian_population_loss_applied }
		}
	}
	set_temp_variable = { air_winter_event_death_request = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_event_death_percent = constant:air_winter_event_runtime.zero }
}

# Supported scope: STATE. The value is frozen when a seasonal observation is
# recorded, so later pressure changes cannot reorder old observations.
air_winter_event_calculate_candidate_score = {
	set_temp_variable = { air_winter_marker_score = air_winter_phase }
	multiply_temp_variable = { air_winter_marker_score = constant:air_winter_event_runtime.candidate_phase_weight }
	set_temp_variable = { air_winter_marker_pressure_score = air_winter_pressure }
	multiply_temp_variable = { air_winter_marker_pressure_score = constant:air_winter_event_runtime.candidate_pressure_weight }
	add_to_temp_variable = { air_winter_marker_score = air_winter_marker_pressure_score }
}

# Supported scope: STATE. All five observations are stored before the owner
# cooldown gate so a transient transition or threshold cannot be lost.
air_winter_event_capture_seasonal_markers = {
	air_winter_event_reconcile_seasonal_markers = yes
	if = {
		limit = {
			air_winter_event_cycle_year_is_valid = yes
			OWNER = { exists = yes }
			NOT = { air_winter_presentation_is_unclassified = yes }
		}
		if = {
			limit = {
				NOT = { has_state_flag = air_winter_first_frost_marker }
				has_state_flag = air_winter_phase_changed_this_tick
				check_variable = { air_winter_previous_phase < constant:air_winter_phase.crop_shock }
				check_variable = { var = air_winter_phase value = constant:air_winter_phase.crop_shock compare = greater_than_or_equals }
				OWNER = {
					OR = {
						NOT = { has_variable = air_winter_first_frost_event_year }
						check_variable = { var = air_winter_first_frost_event_year value = global.air_winter_event_cycle_year compare = less_than }
					}
				}
			}
			set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.none }
			air_winter_event_select_phase_2_route = yes
			if = {
				limit = { check_variable = { air_winter_selected_event_id > constant:air_winter_event_id.none } }
				air_winter_event_calculate_candidate_score = yes
				set_variable = { air_winter_first_frost_origin_year = global.air_winter_event_cycle_year }
				set_variable = { air_winter_first_frost_origin_cycle_id = global.air_winter_cycle_id }
				set_variable = { air_winter_first_frost_origin_owner = OWNER }
				set_variable = { air_winter_first_frost_origin_region = air_winter_presentation_class }
				set_variable = { air_winter_first_frost_origin_score = air_winter_marker_score }
				set_variable = { air_winter_first_frost_origin_event_id = air_winter_selected_event_id }
				set_state_flag = air_winter_first_frost_marker
			}
		}

		if = {

				limit = {
					NOT = { has_state_flag = air_winter_second_winter_marker }

				set_variable = { air_winter_second_winter_origin_event_id = constant:air_winter_event_id.second_winter }
				set_state_flag = air_winter_second_winter_marker
			}
		}

		if = {
			limit = {
				NOT = { has_state_flag = air_winter_terminal_season_marker }
				air_winter_is_terminal = yes
				has_variable = global.air_winter_cycle_contamination_bp_snapshot
				check_variable = { global.air_winter_cycle_contamination_bp_snapshot > constant:air_winter_monitoring.terminal_contamination_bp }
				OWNER = {
					OR = {
						NOT = { has_variable = air_winter_terminal_season_event_year }
						check_variable = { var = air_winter_terminal_season_event_year value = global.air_winter_event_cycle_year compare = less_than }

			set_variable = { air_winter_terminal_season_origin_cycle_id = global.air_winter_cycle_id }

			set_variable = { air_winter_terminal_season_origin_region = air_winter_presentation_class }
			set_variable = { air_winter_terminal_season_origin_score = air_winter_marker_score }
			set_variable = { air_winter_terminal_season_origin_event_id = constant:air_winter_event_id.phase_6_terminal }
			set_state_flag = air_winter_terminal_season_marker
		}
	}
	set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.none }
	set_temp_variable = { air_winter_marker_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_marker_pressure_score = constant:air_winter_event_runtime.zero }

				set_temp_variable = { air_winter_selected_origin_cycle_id = global.air_winter_cycle_id }
				set_temp_variable = { air_winter_selected_score = air_winter_marker_score }
			}

				set_temp_variable = { air_winter_selected_event_id = air_winter_second_winter_origin_event_id }
				set_temp_variable = { air_winter_selected_family = constant:air_winter_event_family.second_winter }

				set_temp_variable = { air_winter_selected_priority = constant:air_winter_event_priority.ash_thaw }

					limit = {
						OR = {
							NOT = { has_variable = air_winter_event_candidate_state }
							NOT = { has_variable = air_winter_event_candidate_id }
							NOT = { has_variable = air_winter_event_candidate_score }
							NOT = { has_variable = air_winter_event_candidate_cycle_id }
							NOT = { check_variable = { var = air_winter_event_candidate_cycle_id value = global.air_winter_cycle_id compare = equals } }

											NOT = { has_variable = air_winter_event_candidate_score }
											check_variable = { var = air_winter_selected_score value = air_winter_event_candidate_score compare = greater_than }

												check_variable = { var = air_winter_selected_score value = air_winter_event_candidate_score compare = equals }
												OR = {
													NOT = { has_variable = air_winter_event_candidate_state }
													check_variable = { var = air_winter_event_candidate_state_id value = air_winter_event_candidate_state compare = less_than }

											}
										}
									}
								}
							}
						}
					}
					set_variable = { air_winter_event_candidate_state = air_winter_event_candidate_state_id }
					set_variable = { air_winter_event_candidate_id = air_winter_selected_event_id }
					set_variable = { air_winter_event_candidate_score = air_winter_selected_score }
					set_variable = { air_winter_event_candidate_cycle_id = global.air_winter_cycle_id }
					set_variable = { air_winter_event_candidate_family = air_winter_selected_family }
					set_variable = { air_winter_event_candidate_priority = air_winter_selected_priority }
					set_variable = { air_winter_event_candidate_year = air_winter_selected_year }
					set_variable = { air_winter_event_candidate_region = air_winter_selected_region }
					set_variable = { air_winter_event_candidate_origin_cycle_id = air_winter_selected_origin_cycle_id }

				if = {
					limit = {
						NOT = {
							is_in_array = {
								array = global.air_winter_event_candidate_countries
								value = THIS
							}
						}
					}
					add_to_array = { array = global.air_winter_event_candidate_countries value = THIS }
				}
			}
		}
	}

	set_temp_variable = { air_winter_selected_event_id = constant:air_winter_event_id.none }

	set_temp_variable = { air_winter_selected_year = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_selected_region = constant:air_winter_presentation_class.unclassified }
	set_temp_variable = { air_winter_selected_origin_cycle_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_selected_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_event_candidate_state_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_marker_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_marker_pressure_score = constant:air_winter_event_runtime.zero }
}
# Supported scope: COUNTRY (the existing host country). Iterates only the
# bounded country array collected during the existing state pass.
air_winter_dispatch_phase_events = {
	for_each_scope_loop = {
		array = global.air_winter_event_candidate_countries
		if = {
			limit = {
				exists = yes
				NOT = { has_country_flag = air_winter_event_cooldown }
				air_winter_event_candidate_is_dispatchable = yes
			}
			save_event_target_as = air_winter_event_country
			var:air_winter_event_candidate_state = { save_event_target_as = air_winter_event_state }

			if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.first_frost } }
				set_variable = { air_winter_first_frost_event_year = air_winter_event_candidate_year }
				var:air_winter_event_candidate_state = { air_winter_event_clear_first_frost_marker = yes }
				set_country_flag = air_winter_event_seen_phase_2
			}
			else_if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.dark_harvest } }
				set_variable = { air_winter_dark_harvest_event_year = air_winter_event_candidate_year }
				var:air_winter_event_candidate_state = { air_winter_event_clear_dark_harvest_marker = yes }
				set_country_flag = air_winter_event_seen_phase_4
			}
			else_if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.ash_thaw } }
				set_variable = { air_winter_ash_thaw_event_year = air_winter_event_candidate_year }
				var:air_winter_event_candidate_state = { air_winter_event_clear_ash_thaw_marker = yes }
				if = {
					limit = {
						OR = {
							NOT = { has_variable = air_winter_recovery_event_count }
							check_variable = { air_winter_recovery_event_count < constant:air_winter_event_runtime.recovery_arc_cap }
						}
					}
					add_to_variable = { air_winter_recovery_event_count = constant:air_winter_event_runtime.one }
				}
			}
			else_if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.second_winter } }
				set_variable = { air_winter_second_winter_event_year = air_winter_event_candidate_year }
				air_winter_event_commit_second_winter_region = yes
				var:air_winter_event_candidate_state = { air_winter_event_clear_second_winter_marker = yes }
				set_country_flag = air_winter_event_seen_phase_4
			}
			else_if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.terminal_season } }
				set_variable = { air_winter_terminal_season_event_year = air_winter_event_candidate_year }
				var:air_winter_event_candidate_state = { air_winter_event_clear_terminal_season_marker = yes }
				set_country_flag = air_winter_event_seen_phase_6
			}
			else_if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.recovery } }
				add_to_variable = { air_winter_recovery_event_count = constant:air_winter_event_runtime.one }
			}
			else_if = {
				limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.ordinary_phase } }
				var:air_winter_event_candidate_state = {
