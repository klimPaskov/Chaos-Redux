# ============================================================================
# CHAOS REDUX - AIR WINTER EVENT EFFECTS
# ============================================================================
# Public contract:
#   air_winter_event_prepare_candidate_cycle = yes (COUNTRY/host before pass)
#   air_winter_schedule_phase_event = yes  (STATE during the monthly pass)
#   air_winter_dispatch_phase_events = yes  (COUNTRY/host after the pass)
#   air_winter_event_refresh_state = yes   (STATE after an event choice)
#   air_winter_event_apply_deaths = yes    (STATE with temp death percent)
#
# The scheduler adds no country or state scan. It records candidates during the
# existing monthly state pass, compares typed priority, origin cycle, score,
# and state id, then uses a bounded owner array for post-pass dispatch.
# ============================================================================

# Timed flag durations reject script-constant and variable tokens on some
# engine builds. Keep this literal mirrored with
# constant:air_winter_event_runtime.country_cooldown_days.
@AIR_WINTER_EVENT_COUNTRY_COOLDOWN_DAYS = 46

# Supported scope: COUNTRY. Reanchors the event cooldown when a delayed
# result is scheduled after the opening popup has already been displayed.
air_winter_event_refresh_country_cooldown = {
	set_country_flag = {
		flag = air_winter_event_cooldown
		days = @AIR_WINTER_EVENT_COUNTRY_COOLDOWN_DAYS
	}
}

# Supported scope: STATE. The regular country target is preferred because it
# preserves the country that made the original choice.
air_winter_event_bind_pending_chain = {
	set_state_flag = air_winter_pending_delayed_result
	if = {
		limit = { has_event_target = air_winter_event_country }
		set_variable = { air_winter_pending_event_owner = event_target:air_winter_event_country }
	}
	else = { set_variable = { air_winter_pending_event_owner = OWNER } }
}

# Supported scope: STATE
air_winter_event_reconcile_pending_chain = {
	if = {
		limit = {
			OR = {
				AND = {
					air_winter_event_has_any_phase_1_branch = yes
					NOT = { has_state_flag = air_winter_pending_delayed_result }
				}
				air_winter_event_has_malformed_phase_1_branches = yes
				AND = {
					air_winter_event_has_exact_desert_city_branch = yes
					NOT = { has_state_flag = air_winter_pending_delayed_result }
				}
				AND = {
					has_state_flag = air_winter_pending_delayed_result
					OR = {
						NOT = { air_winter_event_has_pending_chain = yes }
						NOT = { air_winter_event_pending_owner_is_valid = yes }
					}
				}
				AND = {
					has_state_flag = air_winter_furnace_chain_full_shifts
					has_state_flag = air_winter_furnace_chain_shutdown
				}
				air_winter_event_has_invalid_island_refugee_branches = yes
				air_winter_event_has_invalid_desert_city_branches = yes
				air_winter_event_has_invalid_dead_city_branches = yes
			}
		}
		air_winter_event_cancel_pending_chain = yes
	}
	if = {
		limit = {
			OR = {
				has_state_flag = air_winter_desert_city_opening_receipt
				has_variable = air_winter_desert_city_opening_owner
			}
			NOT = { air_winter_event_desert_city_opening_receipt_is_valid = yes }
		}
		air_winter_desert_city_clear_opening_receipt = yes
		remove_dynamic_modifier = { modifier = air_winter_desert_city_waterworks_state }
	}
}

# Supported scope: COUNTRY. Used only when ownership changed after a visible
# event opened. It never touches a newer owner's delayed chain.
air_winter_event_reject_stale_choice = {
	if = {
		limit = {
			has_event_target = air_winter_event_state
			has_event_target = air_winter_event_country
			event_target:air_winter_event_state = {
				has_state_flag = air_winter_desert_city_opening_receipt
				has_variable = air_winter_desert_city_opening_owner
				check_variable = {
					var = air_winter_desert_city_opening_owner
					value = event_target:air_winter_event_country
					compare = equals
				}
			}
		}
		event_target:air_winter_event_state = { air_winter_desert_city_clear_opening_receipt = yes }
	}
	if = {
		limit = {
			has_event_target = air_winter_event_state
			has_event_target = air_winter_event_country
			event_target:air_winter_event_state = {
				has_state_flag = air_winter_pending_delayed_result
				has_variable = air_winter_pending_event_owner
				check_variable = {
					var = air_winter_pending_event_owner
					value = event_target:air_winter_event_country
					compare = equals
				}
			}
		}
		event_target:air_winter_event_state = { air_winter_event_cancel_pending_chain = yes }
	}
	if = {
		limit = {
			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_active }
		}
		country_event = { id = chaosx.fallout.203 }
	}
}

# Supported scope: COUNTRY. An old Phase 1 opening popup may coexist with a

air_winter_event_reject_stale_opening_choice = {
	if = {

			NOT = { has_global_flag = fallout_transition_active }
			NOT = { has_global_flag = fallout_active }
		}
		country_event = { id = chaosx.fallout.203 }
	}
}

# Supported scope: STATE. Called during the existing monthly state pass. Each
# owner retains only its highest-pressure coastal source, with lower state id
# breaking an exact score tie.
air_winter_island_refugee_collect_source_candidate = {
	set_temp_variable = { air_winter_island_refugee_source_state_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_source_candidate_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_source_pressure_score = constant:air_winter_event_runtime.zero }
	if = {
		limit = { air_winter_island_refugee_source_state_is_live = yes }
		set_temp_variable = { air_winter_island_refugee_source_state_id = THIS.id }
		set_temp_variable = { air_winter_island_refugee_source_candidate_score = air_winter_phase }
		multiply_temp_variable = { air_winter_island_refugee_source_candidate_score = constant:air_winter_event_runtime.candidate_phase_weight }
		set_temp_variable = { air_winter_island_refugee_source_pressure_score = air_winter_pressure }
		multiply_temp_variable = { air_winter_island_refugee_source_pressure_score = constant:air_winter_event_runtime.candidate_pressure_weight }
		add_to_temp_variable = { air_winter_island_refugee_source_candidate_score = air_winter_island_refugee_source_pressure_score }
		OWNER = {
			if = {
				limit = {
					OR = {
						NOT = { has_variable = air_winter_island_refugee_source_state }
						NOT = { has_variable = air_winter_island_refugee_source_score }
						NOT = { has_variable = air_winter_island_refugee_source_cycle_id }
						NOT = { check_variable = { var = air_winter_island_refugee_source_cycle_id value = global.air_winter_cycle_id compare = equals } }
						check_variable = { var = air_winter_island_refugee_source_candidate_score value = air_winter_island_refugee_source_score compare = greater_than }
						AND = {
							check_variable = { var = air_winter_island_refugee_source_candidate_score value = air_winter_island_refugee_source_score compare = equals }
							check_variable = { var = air_winter_island_refugee_source_state_id value = air_winter_island_refugee_source_state compare = less_than }
						}
					}
				}
				set_variable = { air_winter_island_refugee_source_state = air_winter_island_refugee_source_state_id }
				set_variable = { air_winter_island_refugee_source_score = air_winter_island_refugee_source_candidate_score }
				set_variable = { air_winter_island_refugee_source_cycle_id = global.air_winter_cycle_id }
				set_variable = { air_winter_island_refugee_source_region = PREV.air_winter_presentation_class }
			}
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = global.air_winter_island_refugee_source_countries
							value = THIS
						}
					}
				}
				add_to_array = { array = global.air_winter_island_refugee_source_countries value = THIS }
			}
		}
	}
	set_temp_variable = { air_winter_island_refugee_source_state_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_source_candidate_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_source_pressure_score = constant:air_winter_event_runtime.zero }
}

# Supported scope: COUNTRY. The bounded source-owner array is compared by
# frozen score and state id. The dedicated found output prevents an older
# regular target from being mistaken for the current receiver's source.
air_winter_island_refugee_select_source = {
	set_temp_variable = { air_winter_island_refugee_source_found = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_selected_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_selected_state = constant:air_winter_event_runtime.zero }
	if = {
		limit = { NOT = { has_country_flag = air_winter_island_refugee_offer_pending } }
		for_each_scope_loop = {
			array = global.air_winter_island_refugee_source_countries
			if = {
				limit = {
					exists = yes
					NOT = { tag = event_target:air_winter_event_country }
					air_winter_island_refugee_source_receipt_is_current = yes
					OR = {
						check_variable = { air_winter_island_refugee_source_found = constant:air_winter_event_runtime.zero }
						check_variable = { var = air_winter_island_refugee_source_score value = air_winter_island_refugee_selected_score compare = greater_than }
						AND = {
							check_variable = { var = air_winter_island_refugee_source_score value = air_winter_island_refugee_selected_score compare = equals }
							check_variable = { var = air_winter_island_refugee_source_state value = air_winter_island_refugee_selected_state compare = less_than }
						}
					}
				}
				set_temp_variable = { air_winter_island_refugee_source_found = constant:air_winter_event_runtime.one }
				set_temp_variable = { air_winter_island_refugee_selected_score = air_winter_island_refugee_source_score }
				set_temp_variable = { air_winter_island_refugee_selected_state = air_winter_island_refugee_source_state }
				save_event_target_as = air_winter_refugee_source_country
				var:air_winter_island_refugee_source_state = { save_event_target_as = air_winter_refugee_source_state }
			}
		}
	}
	if = {
		limit = { check_variable = { air_winter_island_refugee_source_found = constant:air_winter_event_runtime.one } }
		set_country_flag = air_winter_island_refugee_offer_pending
		set_variable = { air_winter_island_refugee_offer_source_state = event_target:air_winter_refugee_source_state }
		set_variable = { air_winter_island_refugee_offer_source_country = event_target:air_winter_refugee_source_country }
		set_variable = { air_winter_island_refugee_offer_source_region = event_target:air_winter_refugee_source_country.air_winter_island_refugee_source_region }
		set_variable = { air_winter_island_refugee_offer_destination_state = air_winter_event_candidate_state }
		set_variable = { air_winter_island_refugee_offer_family = air_winter_event_candidate_family }
		set_variable = { air_winter_island_refugee_offer_year = air_winter_event_candidate_year }
		set_variable = { air_winter_island_refugee_offer_cycle_id = global.air_winter_cycle_id }
		air_winter_event_refresh_country_cooldown = yes
	}
}

					has_variable = air_winter_first_frost_origin_event_id
					has_variable = air_winter_first_frost_origin_route_subtype
					check_variable = { air_winter_first_frost_origin_event_id = constant:air_winter_event_id.phase_2_island_refugees }
					check_variable = { air_winter_first_frost_origin_route_subtype = constant:air_winter_event_route_subtype.none }

					check_variable = { var = air_winter_first_frost_origin_region value = air_winter_presentation_class compare = equals }
					check_variable = { air_winter_first_frost_origin_score > constant:air_winter_event_runtime.zero }
					is_owned_by = PREV
					is_owned_by = var:air_winter_first_frost_origin_owner
				}

		}
	}
	set_country_flag = air_winter_event_seen_phase_2
	air_winter_event_refresh_country_cooldown = yes
	air_winter_island_refugee_clear_offer = yes
	set_temp_variable = { air_winter_island_refugee_marker_year = constant:air_winter_event_runtime.zero }

# Supported scope: STATE. PREV is the dispatching country. An ordinary phase
# event may be the exact authored route stored by first frost or dark harvest.
# Consume that matching observation once so one physical incident cannot open
# the same event twice under two family labels.
air_winter_event_coalesce_matching_seasonal_marker = {
	if = {
		limit = {
			air_winter_first_frost_marker_is_valid = yes
			check_variable = {
				var = air_winter_first_frost_origin_event_id
				value = PREV.air_winter_event_candidate_id
				compare = equals
			}
			check_variable = {
				var = air_winter_first_frost_origin_route_subtype
				value = PREV.air_winter_event_candidate_route_subtype
				compare = equals
			}
		}
		OWNER = { set_variable = { air_winter_first_frost_event_year = PREV.air_winter_first_frost_origin_year } }
		air_winter_event_clear_first_frost_marker = yes
	}
	if = {
		limit = {
			air_winter_dark_harvest_marker_is_valid = yes
			check_variable = {
				var = air_winter_dark_harvest_origin_event_id
				value = PREV.air_winter_event_candidate_id
				compare = equals
			}
		}
		OWNER = { set_variable = { air_winter_dark_harvest_event_year = PREV.air_winter_dark_harvest_origin_year } }
		air_winter_event_clear_dark_harvest_marker = yes
	}
}

						OR = {
							NOT = { has_variable = air_winter_event_candidate_state }
							NOT = { has_variable = air_winter_event_candidate_id }
							NOT = { has_variable = air_winter_event_candidate_score }
							NOT = { has_variable = air_winter_event_candidate_cycle_id }
							NOT = { check_variable = { var = air_winter_event_candidate_cycle_id value = global.air_winter_cycle_id compare = equals } }
							NOT = { has_variable = air_winter_event_candidate_family }
							NOT = { has_variable = air_winter_event_candidate_priority }
							NOT = { has_variable = air_winter_event_candidate_year }
							NOT = { has_variable = air_winter_event_candidate_region }
							NOT = { has_variable = air_winter_event_candidate_origin_cycle_id }
							NOT = { has_variable = air_winter_event_candidate_route_subtype }
							check_variable = { var = air_winter_selected_priority value = air_winter_event_candidate_priority compare = greater_than }
							AND = {
								check_variable = { var = air_winter_selected_priority value = air_winter_event_candidate_priority compare = equals }
								OR = {
									NOT = { has_variable = air_winter_event_candidate_origin_cycle_id }
									check_variable = { var = air_winter_selected_origin_cycle_id value = air_winter_event_candidate_origin_cycle_id compare = less_than }
									AND = {
										check_variable = { var = air_winter_selected_origin_cycle_id value = air_winter_event_candidate_origin_cycle_id compare = equals }
										OR = {
											NOT = { has_variable = air_winter_event_candidate_score }
											check_variable = { var = air_winter_selected_score value = air_winter_event_candidate_score compare = greater_than }
											AND = {
												check_variable = { var = air_winter_selected_score value = air_winter_event_candidate_score compare = equals }
												OR = {
													NOT = { has_variable = air_winter_event_candidate_state }
													check_variable = { var = air_winter_event_candidate_state_id value = air_winter_event_candidate_state compare = less_than }
												}
											}
										}
									}
								}
							}
						}
					}
					set_variable = { air_winter_event_candidate_state = air_winter_event_candidate_state_id }
					set_variable = { air_winter_event_candidate_id = air_winter_selected_event_id }
					set_variable = { air_winter_event_candidate_score = air_winter_selected_score }
					set_variable = { air_winter_event_candidate_cycle_id = global.air_winter_cycle_id }
					set_variable = { air_winter_event_candidate_family = air_winter_selected_family }
					set_variable = { air_winter_event_candidate_priority = air_winter_selected_priority }
					set_variable = { air_winter_event_candidate_year = air_winter_selected_year }
					set_variable = { air_winter_event_candidate_region = air_winter_selected_region }
					set_variable = { air_winter_event_candidate_origin_cycle_id = air_winter_selected_origin_cycle_id }
					set_variable = { air_winter_event_candidate_route_subtype = air_winter_selected_route_subtype }
				}
				if = {
					limit = {
						NOT = {
							is_in_array = {
								array = global.air_winter_event_candidate_countries
								value = THIS
							}
						}
					}
					add_to_array = { array = global.air_winter_event_candidate_countries value = THIS }
				}
			}
		}
	}

	set_temp_variable = { air_winter_selected_priority = constant:air_winter_event_priority.none }
	set_temp_variable = { air_winter_selected_year = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_selected_region = constant:air_winter_presentation_class.unclassified }
	set_temp_variable = { air_winter_selected_origin_cycle_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_selected_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_event_candidate_state_id = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_marker_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_marker_pressure_score = constant:air_winter_event_runtime.zero }

# Supported scope: COUNTRY (the existing host country). Iterates only the
# bounded country array collected during the existing state pass.
air_winter_dispatch_phase_events = {
	for_each_scope_loop = {
		array = global.air_winter_event_candidate_countries
		if = {
			limit = {
				exists = yes
				NOT = { has_global_flag = fallout_transition_active }
				NOT = { has_global_flag = fallout_active }
				NOT = { has_country_flag = air_winter_event_cooldown }
				NOT = { has_country_flag = air_winter_island_refugee_offer_pending }
				air_winter_event_candidate_is_dispatchable = yes
			}
			save_event_target_as = air_winter_event_country
			var:air_winter_event_candidate_state = { save_event_target_as = air_winter_event_state }

			if = {
				limit = { check_variable = { air_winter_event_candidate_id = constant:air_winter_event_id.phase_2_island_refugees } }
				air_winter_island_refugee_select_source = yes
				if = {
					limit = { check_variable = { air_winter_island_refugee_source_found = constant:air_winter_event_runtime.one } }
					country_event = { id = chaosx.fallout.38 }
				}
			}
			else = {
				if = {
					limit = { check_variable = { air_winter_event_candidate_id = constant:air_winter_event_id.phase_2_arid } }
					var:air_winter_event_candidate_state = {
						air_winter_desert_city_clear_opening_receipt = yes
						if = {
							limit = {
								PREV = { check_variable = { air_winter_event_candidate_route_subtype = constant:air_winter_event_route_subtype.desert_city } }
							}
							set_state_flag = air_winter_desert_city_opening_receipt
							set_variable = { air_winter_desert_city_opening_owner = PREV }
						}
					}
				}
				if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.first_frost } }
					set_variable = { air_winter_first_frost_event_year = air_winter_event_candidate_year }
					var:air_winter_event_candidate_state = { air_winter_event_clear_first_frost_marker = yes }
					set_country_flag = air_winter_event_seen_phase_2
				}
				else_if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.dark_harvest } }
					set_variable = { air_winter_dark_harvest_event_year = air_winter_event_candidate_year }
					var:air_winter_event_candidate_state = { air_winter_event_clear_dark_harvest_marker = yes }
					set_country_flag = air_winter_event_seen_phase_4
				}
				else_if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.ash_thaw } }
					set_variable = { air_winter_ash_thaw_event_year = air_winter_event_candidate_year }
					var:air_winter_event_candidate_state = { air_winter_event_clear_ash_thaw_marker = yes }
					if = {
						limit = {
							OR = {
								NOT = { has_variable = air_winter_recovery_event_count }
								check_variable = { air_winter_recovery_event_count < constant:air_winter_event_runtime.recovery_arc_cap }
							}
						}
						add_to_variable = { air_winter_recovery_event_count = constant:air_winter_event_runtime.one }
					}
				}
				else_if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.second_winter } }
					set_variable = { air_winter_second_winter_event_year = air_winter_event_candidate_year }
					air_winter_event_commit_second_winter_region = yes
					var:air_winter_event_candidate_state = { air_winter_event_clear_second_winter_marker = yes }
					set_country_flag = air_winter_event_seen_phase_4
				}
				else_if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.terminal_season } }
					set_variable = { air_winter_terminal_season_event_year = air_winter_event_candidate_year }
					var:air_winter_event_candidate_state = { air_winter_event_clear_terminal_season_marker = yes }
					set_country_flag = air_winter_event_seen_phase_6
				}
				else_if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.recovery } }
					add_to_variable = { air_winter_recovery_event_count = constant:air_winter_event_runtime.one }
				}
				else_if = {
					limit = { check_variable = { air_winter_event_candidate_family = constant:air_winter_event_family.ordinary_phase } }
					var:air_winter_event_candidate_state = {
						air_winter_event_coalesce_matching_seasonal_marker = yes
						if = { limit = { air_winter_is_phase_1 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_1 } }
						else_if = { limit = { air_winter_is_phase_2 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_2 } }
						else_if = { limit = { air_winter_is_phase_3 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_3 } }
						else_if = { limit = { air_winter_is_phase_4 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_4 } }
						else_if = { limit = { air_winter_is_phase_5 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_5 } }
						else_if = { limit = { air_winter_is_phase_6 = yes } OWNER = { set_country_flag = air_winter_event_seen_phase_6 } }
					}
				}

				air_winter_event_refresh_country_cooldown = yes
				meta_effect = {
					text = { country_event = { id = chaosx.fallout.[AIR_WINTER_EVENT_ID] } }
					AIR_WINTER_EVENT_ID = "[?air_winter_event_candidate_id|.0]"
				}
			}
		}
		air_winter_event_clear_candidate = yes
	}

		array = global.air_winter_island_refugee_source_countries
		air_winter_island_refugee_clear_source_receipt = yes
	}
	clear_array = global.air_winter_island_refugee_source_countries
	set_temp_variable = { air_winter_island_refugee_source_found = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_selected_score = constant:air_winter_event_runtime.zero }
	set_temp_variable = { air_winter_island_refugee_selected_state = constant:air_winter_event_runtime.zero }
