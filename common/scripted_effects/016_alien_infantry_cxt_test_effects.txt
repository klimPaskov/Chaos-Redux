# ============================================================================
# CHAOS REDUX - EVENT 016 CXT TEST REGISTRATION
# ============================================================================
# Overview:
# - Registers every reusable Event 016 project-force unit and concrete
#   equipment type, plus the D'Rhondan envoy project, with the shared CXT bus.
# - Keeps ordinary Event 016 recruitment rules intact while giving CXT a
#   recruitable ten-battalion landing cohort for designer and stat validation.
# - The carrier and setup effect are idempotent; daily CXT work only registers
#   pending content for a player-triggered package application.
# ============================================================================

alien_infantry_register_cxt_test_content = {
	set_temp_variable = {
		var = chaosx_test_country_registration_extension_effect
		value = token:chaosx_cxt_extension_event016_alien_infantry
	}
	chaosx_test_country_register_extension_effect = yes
}

	# packages. Test grants do not create Kruger project or source history.
	if = {
		limit = { NOT = { has_country_flag = conventional_technology_computation_weaponized } }
		set_temp_variable = { chaosx_conventional_technology_family = constant:brilliant_scientist_project_family.computation }
		set_temp_variable = { chaosx_conventional_technology_tier = constant:brilliant_scientist_project_stage.weaponization }
		chaosx_grant_conventional_technology_package_core = yes
	}
	if = {
		limit = { NOT = { has_country_flag = conventional_technology_electronics_weaponized } }
		set_temp_variable = { chaosx_conventional_technology_family = constant:brilliant_scientist_project_family.electronics }
		set_temp_variable = { chaosx_conventional_technology_tier = constant:brilliant_scientist_project_stage.weaponization }
		chaosx_grant_conventional_technology_package_core = yes
	}
	if = {

	set_temp_variable = { chaosx_conventional_technology_family = constant:brilliant_scientist_value.zero }
	set_temp_variable = { chaosx_conventional_technology_tier = constant:brilliant_scientist_value.zero }
	chaosx_reconcile_conventional_technology_runtime = yes

	# inventing paid project history from the harness's completed technologies.
	brilliant_scientist_mengele_reconcile_project_availability = yes

		limit = { NOT = { brilliant_scientist_mengele_project_stage_provider_is_valid = yes } }
		brilliant_scientist_mengele_cleanup_provider_receipts = yes

	# invalid pending receipt. The harness never fabricates a site or component.
	if = {
		limit = {
			OR = {
				has_country_flag = brilliant_scientist_singularity_construction_receipt_active
				check_variable = { var = brilliant_scientist_singularity_site_state_entries^num value = constant:brilliant_scientist_value.zero compare = greater_than }
			}
		}
		brilliant_scientist_singularity_site_reconcile_receipts = yes

	# Registration helpers reject duplicates, so these calls also repair an
	# existing CXT save whose older Event 016 carrier contained only aliens.
	set_temp_variable = {

	}
	chaosx_test_country_register_special_project = yes

	set_temp_variable = { var = chaosx_test_country_registration_equipment value = token:teleportation_equipment_1 }

	# CXT's isolated test copy is recruitable so its unit, equipment, counter,
	# designer, and Event 019 provider surfaces can all be inspected.

		limit = {
			NOT = {
				is_in_array = {

chaosx_cxt_event016_spawn_alien_cohort = {
	if = {
		limit = { has_template = "CXT Test - Alien Landing Cohort" }
		for_loop_effect = {
			end = constant:chaosx_test_country_count.divisions_per_template
			value = chaosx_test_country_spawn_index
			random_owned_controlled_state = {
				create_unit = {
					division = "name = \"CXT Test - Alien Landing Cohort\" division_template = \"CXT Test - Alien Landing Cohort\" start_experience_factor = 1 start_equipment_factor = 1 start_manpower_factor = 1"
					owner = PREV
					count = constant:chaosx_test_country_count.divisions_per_state_pick
				}
			}
		}
	}
}
