# COUNTRY scope. Player-triggered CXT-only setup for the real Mengele provider
# decision surface. It creates no stage receipt, native completion, or package.
chaosx_cxt_event016_enable_mengele_provider_fixture = {
	if = {
		limit = {
			tag = CXT
			has_country_flag = chaosx_test_country_initialized
			NOT = { has_country_flag = chaosx_cxt_event016_mengele_provider_fixture_ready }
		}
		set_country_flag = mengele_clone_army_scenario_country
		if = {
			limit = { NOT = { has_idea = mengele_clone_army_scenario_state } }
			add_ideas = mengele_clone_army_scenario_state
		}
		set_country_flag = mengele_clone_focus_directorate_project_registry
		brilliant_scientist_mengele_initialize_project_stage_receipts = yes
		brilliant_scientist_mengele_initialize_native_prototype_provenance = yes
		brilliant_scientist_mengele_reconcile_project_availability = yes
		if = {
			limit = {
				brilliant_scientist_mengele_project_stage_provider_is_valid = yes
				has_country_flag = mengele_event016_provider_receipts_initialized
			}
			set_country_flag = chaosx_cxt_event016_mengele_provider_fixture_ready
		}
	}
}
