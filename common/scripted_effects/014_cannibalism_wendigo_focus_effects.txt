# ============================================================================
# CHAOS REDUX - EVENT 014 WENDIGO FOCUS OVERLAY EFFECTS
# ============================================================================
# Executes the 28 post-reveal original-ZZZ Wendigo Hannibal focus rewards.
# These effects preserve the live zombie country's OOB, technology, units,
# leader identity, division-template lock, and 16-battalion Wendigo Pack. The
# winter capstone uses the existing staged-idea helper. No effect grants consumed
# population or Larder or sets world-end state. The pulse remains the sole lock.
# ============================================================================

# ---------------------------------------------------------------------------
# Loader and shared reward helpers
# ---------------------------------------------------------------------------

cannibalism_load_wendigo_focus_overlay = {
	if = {
		limit = {
			original_tag = ZZZ
			is_cannibalism_wendigo_hannibal_country = yes
			has_global_flag = cannibalism_reveal_complete
			has_country_flag = cannibalism_wendigo_focus_overlay_available
			has_character = ZZZ_hannibal_wendigo
			NOT = { has_global_flag = world_end }
			NOT = { has_country_flag = cannibalism_wendigo_focus_overlay_loaded }
		}
		load_focus_tree = { tree = cannibalism_wendigo_focus_tree keep_completed = no }
		set_country_flag = cannibalism_wendigo_focus_overlay_loaded
	}
}

cannibalism_wendigo_focus_prepare_runtime = {
	if = {
		limit = { NOT = { has_variable = cannibalism_unified_authority } }
		set_variable = { cannibalism_unified_authority = constant:cannibalism_unification.starting_authority }
	}
	if = {
		limit = { NOT = { has_variable = cannibalism_frenzy } }
		set_variable = { cannibalism_frenzy = constant:cannibalism_meter.min }
	}
	if = {
		limit = { NOT = { has_variable = cannibalism_wendigo_pack_capacity } }
		set_variable = { cannibalism_wendigo_pack_capacity = constant:cannibalism_wendigo_decision.train_pack_base_cap }
	}
	if = {
		limit = { NOT = { has_variable = cannibalism_wendigo_focus_pack_capacity_bonus } }
		set_variable = { cannibalism_wendigo_focus_pack_capacity_bonus = constant:cannibalism_meter.min }
	}
	if = {
		limit = { NOT = { has_variable = cannibalism_wendigo_trained_pack_count } }
		set_variable = { cannibalism_wendigo_trained_pack_count = constant:cannibalism_meter.min }
	}
}

cannibalism_wendigo_focus_apply_authority_gain = {
	cannibalism_wendigo_focus_prepare_runtime = yes
	add_to_variable = { cannibalism_unified_authority = cannibalism_wendigo_focus_authority_gain }
	clamp_variable = {
		var = cannibalism_unified_authority
		min = constant:cannibalism_meter.min
		max = constant:cannibalism_meter.max
	}
}

cannibalism_wendigo_focus_apply_frenzy_gain = {
	cannibalism_wendigo_focus_prepare_runtime = yes
	add_to_variable = { cannibalism_frenzy = cannibalism_wendigo_focus_frenzy_gain }
	clamp_variable = {
		var = cannibalism_frenzy
		min = constant:cannibalism_consumption.frenzy_min
		max = constant:cannibalism_consumption.frenzy_max
	}
}

cannibalism_wendigo_focus_expand_pack_capacity = {
	cannibalism_wendigo_focus_prepare_runtime = yes
	add_to_variable = { cannibalism_wendigo_focus_pack_capacity_bonus = cannibalism_wendigo_focus_pack_capacity_gain }
	cannibalism_rebuild_wendigo_anchor_runtime = yes
}

cannibalism_wendigo_focus_preserve_pack_contract = {
	country_lock_all_division_template = yes
	if = {
		limit = { has_template = "Wendigo Pack" }
		set_division_force_allow_recruiting = {
			division_template = "Wendigo Pack"
			force_allow_recruiting = no
		}
		set_division_template_lock = {
			division_template = "Wendigo Pack"
			is_locked = yes
		}
	}
	if = {
		limit = { NOT = { has_country_flag = cannibalism_wendigo_pack_contract_ai_applied } }
		add_ai_strategy = { type = template_prio id = zombies value = constant:cannibalism_wendigo_ai.train_pack }
		add_ai_strategy = { type = role_ratio id = zombies value = constant:cannibalism_wendigo_ai.train_pack }
		set_country_flag = cannibalism_wendigo_pack_contract_ai_applied
	}
}

cannibalism_wendigo_focus_strengthen_live_anchors = {
	for_each_scope_loop = {
		array = global.cannibalism_transformation_anchor_states
		if = {
			limit = { cannibalism_current_wendigo_anchor_is_live = yes }
			add_to_variable = { cannibalism_wendigo_anchor_strength = cannibalism_wendigo_focus_anchor_strength_gain }
			clamp_variable = {
				var = cannibalism_wendigo_anchor_strength
				min = constant:cannibalism_wendigo_anchor.minimum_live_strength
				max = constant:cannibalism_wendigo_anchor.maximum_strength
			}
		}
	}
	cannibalism_rebuild_wendigo_anchor_runtime = yes
}
