cannibalism_calculate_current_unification_host_score = {
	set_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_unification.base_host_score }
	if = {
		limit = { is_ai = no }
		add_to_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_unification.human_host_score }
	}

	set_temp_variable = { cannibalism_unification_score_component = num_of_controlled_states }
	multiply_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.state_score }
	add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }

	set_temp_variable = { cannibalism_unification_score_component = num_divisions }
	multiply_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.division_score }
	add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }

	set_temp_variable = { cannibalism_unification_controlled_population_k = 0 }
	every_controlled_state = {
		add_to_temp_variable = { cannibalism_unification_controlled_population_k = state_population_k }
	}
	set_temp_variable = { cannibalism_unification_score_component = cannibalism_unification_controlled_population_k }
	divide_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.population_k_per_million }
	multiply_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.population_score_per_million }
	add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }

	if = {
		limit = { has_variable = cannibalism_larder }
		set_temp_variable = { cannibalism_unification_score_component = cannibalism_larder }
		divide_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.larder_score_divisor }
		add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }
	}
	if = {
		limit = { has_variable = cannibalism_network_alignment }
		set_temp_variable = { cannibalism_unification_score_component = cannibalism_network_alignment }
		divide_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.alignment_score_divisor }
		add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }
	}
	if = {
		limit = { has_variable = cannibalism_inbound_spread_count }
		set_temp_variable = { cannibalism_unification_score_component = cannibalism_inbound_spread_count }
		multiply_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.network_centrality_score }
		add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }
	}
	if = {
		limit = { has_variable = cannibalism_outbound_spread_count }
		set_temp_variable = { cannibalism_unification_score_component = cannibalism_outbound_spread_count }
		multiply_temp_variable = { cannibalism_unification_score_component = constant:cannibalism_unification.network_centrality_score }
		add_to_temp_variable = { cannibalism_current_unification_host_score = cannibalism_unification_score_component }
	}

	if = {
		limit = { any_controlled_state = { is_coastal = yes naval_base > 0 } }
		add_to_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_unification.port_score }
	}
	if = {
		limit = { capital_scope = { OR = { supply_node > 0 naval_base > 0 } } }
		add_to_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_unification.host_capital_supply_score }
	}
	if = {
		limit = { capital_scope = { rail_way > 0 } }
		add_to_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_unification.host_rail_score }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_convergence_leverage_open }
		add_to_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_warlord_contract.convergence_leverage_host_score }
	}
	if = {
		limit = { has_country_flag = cannibalism_warlord_manipulation_endgame_open }
		add_to_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_warlord_contract.manipulation_endgame_host_score }
	}
	if = {
		limit = {
			capital_scope = {
				NOT = { any_neighbor_state = { is_controlled_by = PREV } }
				is_island_state = no
			}
		}
		subtract_from_temp_variable = { cannibalism_current_unification_host_score = constant:cannibalism_unification.host_isolation_penalty }
	}
	clamp_temp_variable = {
		var = cannibalism_current_unification_host_score
		min = constant:cannibalism_unification.minimum_host_score
		max = constant:cannibalism_unification.maximum_host_score
	}
}

cannibalism_select_unification_host = {
	clear_global_event_target = cannibalism_unification_host
	clear_variable = global.cannibalism_unification_host_generation
	set_temp_variable = { cannibalism_best_unification_host_score = 0 }
	for_each_scope_loop = {
		array = global.cannibalism_actor_countries
		if = {
			limit = { cannibalism_is_viable_unification_host = yes }
			cannibalism_calculate_current_unification_host_score = yes
			if = {
				limit = {
					check_variable = {
						var = cannibalism_current_unification_host_score
						value = cannibalism_best_unification_host_score
						compare = greater_than
					}
				}
				set_temp_variable = { cannibalism_best_unification_host_score = cannibalism_current_unification_host_score }
				save_global_event_target_as = cannibalism_unification_host
				set_variable = { global.cannibalism_unification_host_generation = cannibalism_actor_generation }
			}
		}
	}
	set_variable = { global.cannibalism_unification_host_score = cannibalism_best_unification_host_score }
}

cannibalism_prepare_current_warlord_slot_for_unification = {
	set_temp_variable = { cannibalism_absorbed_warlord_slot_index = cannibalism_warlord_slot_index }
	set_temp_variable = { cannibalism_absorbed_warlord_slot_generation = cannibalism_warlord_slot_generation }
	save_event_target_as = cannibalism_warlord_integration_release_country
	if = {
		limit = { is_in_array = { array = global.cannibalism_convergence_actor_countries value = THIS } }
		remove_from_array = { array = global.cannibalism_convergence_actor_countries value = THIS }
	}
	clr_country_flag = cannibalism_convergence_participant
	set_country_flag = cannibalism_warlord_integration_release_pending
	every_state = {
		limit = {
			has_variable = cannibalism_warlord_slot_index
			check_variable = {
				var = cannibalism_warlord_slot_index
				value = cannibalism_absorbed_warlord_slot_index
				compare = equals
			}
			check_variable = {
				var = cannibalism_warlord_slot_generation
				value = cannibalism_absorbed_warlord_slot_generation
				compare = equals
			}
		}
		if = {
			limit = {
				has_event_target = cannibalism_runtime_migration_destination_country
				event_target:cannibalism_runtime_migration_destination_country = { exists = yes }
				NOT = { is_core_of = event_target:cannibalism_runtime_migration_destination_country }
			}
			add_state_core = event_target:cannibalism_runtime_migration_destination_country
		}
		if = {
			limit = { is_core_of = event_target:cannibalism_warlord_integration_release_country }
			remove_state_core = event_target:cannibalism_warlord_integration_release_country
		}
		clr_state_flag = cannibalism_warlord_state
		clr_state_flag = cannibalism_warlord_transfer_selected
		clr_state_flag = cannibalism_warlord_formation_pending
		clear_variable = cannibalism_warlord_slot_index
		clear_variable = cannibalism_warlord_slot_generation
	}
	# The in-use flags and country incarnation remain intact until on_annex calls
	# the canonical reset and the reference audit authorizes final release.
}

cannibalism_migrate_runtime_references_to_unified_country = {
	# Contract: cannibalism_runtime_migration_source_country and
	# cannibalism_runtime_migration_destination_country must both be valid
	# regular event targets. Never infer the destination from ROOT: absorption
	# runs from the responding warlord's event scope.
	set_temp_variable = { cannibalism_runtime_migration_result = 0 }
	# The spread owner validates all nine aligned arrays, updates both country
	# directions and exact state metadata, then rebuilds every derived cache.
	cannibalism_migrate_spread_entries_between_countries = yes
	if = {
		limit = { check_variable = { cannibalism_spread_migration_result > 0 } }
		for_each_scope_loop = {
			array = global.cannibalism_node_states
			if = {
				limit = {
					has_variable = cannibalism_node_source_country
					check_variable = {
						var = cannibalism_node_source_country
						value = event_target:cannibalism_runtime_migration_source_country
						compare = equals
					}
				}
				set_variable = { cannibalism_node_source_country = event_target:cannibalism_runtime_migration_destination_country }
				set_variable = { cannibalism_node_source_country_generation = event_target:cannibalism_runtime_migration_destination_country.cannibalism_actor_generation }
			}
		}
		for_each_scope_loop = {
			array = global.cannibalism_actor_countries
			if = {
				limit = {
					has_variable = cannibalism_external_infection_source

						var = cannibalism_external_infection_source
						value = event_target:cannibalism_runtime_migration_source_country

					}
				}
				set_variable = { cannibalism_external_infection_source = event_target:cannibalism_runtime_migration_destination_country }
			}
		}
		set_temp_variable = { cannibalism_runtime_migration_result = 1 }
	}

					limit = { cannibalism_selected_unification_host_is_valid = yes }

				}
				else = {
					cannibalism_break_convergence = yes
				}
			}
		}
	}

				add_building_construction = { type = bunker level = constant:cannibalism_warlord_contract.independent_capital_fort_levels instant_build = yes }
			}
		}
		if = {
			limit = { has_country_flag = cannibalism_warlord_defiance_endgame_open }
			capital_scope = {
				add_building_construction = { type = bunker level = constant:cannibalism_warlord_contract.defiance_resistance_fort_levels instant_build = yes }
			}
		}
		if = {
			limit = { NOT = { has_war_with = event_target:cannibalism_unification_host } }
			event_target:cannibalism_unification_host = { declare_war_on = { target = ROOT type = annex_everything } }
		}
	}
}

cannibalism_challenge_unified_host = {
	if = {
		limit = {
			is_cannibalism_warlord_country = yes
			has_country_flag = cannibalism_pending_absorption
			cannibalism_has_valid_unification_destination = yes
		}
		clr_country_flag = cannibalism_pending_absorption
		clr_country_flag = cannibalism_convergence_participant
		if = {
			limit = { is_in_array = { array = global.cannibalism_convergence_actor_countries value = THIS } }
			remove_from_array = { array = global.cannibalism_convergence_actor_countries value = THIS }
		}
		set_country_flag = cannibalism_challenger_warlord
		set_variable = { cannibalism_warlord_disposition = constant:cannibalism_warlord_disposition.challenger }
		add_dynamic_modifier = { modifier = cannibalism_warlord_challenge_to_unification days = @CANNIBALISM_WARLORD_CHALLENGE_DAYS }
		if = {
			limit = { NOT = { has_war_with = event_target:cannibalism_unification_host } }
			event_target:cannibalism_unification_host = { declare_war_on = { target = ROOT type = annex_everything } }
		}
	}
}
