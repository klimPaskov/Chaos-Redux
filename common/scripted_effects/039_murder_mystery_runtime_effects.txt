# ============================================================================
# CHAOS REDUX - EVENT 039 MURDER MYSTERY RUNTIME EFFECTS
# ============================================================================
# Overview:
# - This file is the bounded runtime layer for Event 039 Murder Mystery.
# - It owns country values, event-owned flags, aligned target/cell/derivative/
#   terminal registries, six-phase action transactions, evolution pacing, and
#   setup/rollback state machines.
# - Character removal, dynamic country creation, final war declaration, and
#   state/country owner callbacks are deliberately not performed here. The
#   parent event package must prove those operations and call the commit APIs.
# - Long-lived pointers use named global event targets and are cleared by the
#   cleanup effects below. No on_action or whole-world recurring scan belongs
#   in this file.
# ============================================================================

murder_mystery_register_original_host = {
	murder_mystery_initialize_runtime = yes
	clr_country_flag = murder_mystery_host_registration_failed
	clr_country_flag = murder_mystery_host_registration_succeeded
	if = {
		limit = { murder_mystery_host_registration_is_valid = yes }
		murder_mystery_initialize_country_runtime = yes
		save_global_event_target_as = murder_mystery_original_host
		capital_scope = {
			save_global_event_target_as = murder_mystery_original_host_capital_state
		}
		set_country_flag = murder_mystery_original_host
		set_global_flag = murder_mystery_host_registered
		set_global_flag = murder_mystery_runtime_active
		clr_global_flag = murder_mystery_runtime_dormant
		clr_global_flag = murder_mystery_runtime_resolved
		set_variable = { global.murder_mystery_lifecycle = constant:murder_mystery_lifecycle.opening_shock }
		add_to_variable = { global.murder_mystery_transaction_sequence = constant:murder_mystery_value.one }
		set_variable = { global.murder_mystery_host_generation = global.murder_mystery_transaction_sequence }
		set_variable = { murder_mystery_host_generation = global.murder_mystery_host_generation }
		set_country_flag = murder_mystery_host_registration_succeeded
	}
	else = {
		set_country_flag = murder_mystery_host_registration_failed
	}
}

murder_mystery_set_investigation_phase = {
	if = {
		limit = {
			has_variable = murder_mystery_requested_phase
			check_variable = { var = murder_mystery_requested_phase value = constant:murder_mystery_phase.secure_succession compare = greater_than_or_equals }
			check_variable = { var = murder_mystery_requested_phase value = constant:murder_mystery_phase.final_operation compare = less_than_or_equals }
		}
		set_variable = { murder_mystery_investigation_phase = murder_mystery_requested_phase }

# checked for ownership, control, connectivity, and remnant viability.

			has_event_target = murder_mystery_original_host

			is_owned_and_controlled_by = event_target:murder_mystery_original_host
			NOT = { is_in_array = { array = global.murder_mystery_split_state_entries value = THIS } }
			check_variable = { var = global.murder_mystery_split_state_entries^num value = constant:murder_mystery_registry.maximum_split_state_rows compare = less_than }

murder_mystery_evaluate_terminal_victory = {
	clr_global_flag = murder_mystery_terminal_registry_victory_ready
	if = {
		limit = {
			murder_mystery_terminal_registry_is_aligned = yes
			check_variable = { var = global.murder_mystery_terminal_country_entries^num value = constant:murder_mystery_world_end.minimum_terminal_registry_rows compare = greater_than_or_equals }
		}
		set_temp_variable = { murder_mystery_terminal_registry_index = constant:murder_mystery_identity.first_array_index }
		set_temp_variable = { murder_mystery_terminal_unresolved_rows = constant:murder_mystery_value.zero }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_terminal_registry_index value = global.murder_mystery_terminal_country_entries^num compare = less_than } }
			set_temp_variable = { murder_mystery_terminal_registry_row_status = global.murder_mystery_terminal_status_entries^murder_mystery_terminal_registry_index }
			set_temp_variable = { murder_mystery_terminal_registry_row_victory = global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_registry_index }
			if = {
				limit = {
					OR = {
						check_variable = { var = murder_mystery_terminal_registry_row_status value = constant:murder_mystery_terminal.status_unclassified compare = equals }
						check_variable = { var = murder_mystery_terminal_registry_row_status value = constant:murder_mystery_terminal.status_registered compare = equals }
						check_variable = { var = murder_mystery_terminal_registry_row_status value = constant:murder_mystery_terminal.status_engaged compare = equals }
						check_variable = { var = murder_mystery_terminal_registry_row_victory value = constant:murder_mystery_terminal.victory_unresolved compare = equals }
					}
				}
				add_to_temp_variable = { murder_mystery_terminal_unresolved_rows = constant:murder_mystery_value.one }
			}
			add_to_temp_variable = { murder_mystery_terminal_registry_index = constant:murder_mystery_identity.loop_increment }
		}
		if = {
			limit = { check_variable = { var = murder_mystery_terminal_unresolved_rows value = constant:murder_mystery_value.zero compare = equals } }
			set_global_flag = murder_mystery_terminal_registry_victory_ready
			set_global_flag = murder_mystery_terminal_registry_completion_proven
		}
	}
}
