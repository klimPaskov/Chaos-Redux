# ============================================================================
# CHAOS REDUX - EVENT 039 MURDER MYSTERY RUNTIME EFFECTS
# ============================================================================
# Overview:
# - This file is the bounded runtime layer for Event 039 Murder Mystery.
# - It owns country values, event-owned flags, aligned target/cell/derivative/
#   terminal registries, six-phase action transactions, evolution pacing, and
#   setup/rollback state machines.
# - Character removal, dynamic country creation, final war declaration, and
#   state/country owner callbacks are deliberately not performed here. The
#   parent event package must prove those operations and call the commit APIs.
# - Long-lived pointers use named global event targets and are cleared by the
#   cleanup effects below. No on_action or whole-world recurring scan belongs
#   in this file.
# ============================================================================

murder_mystery_register_original_host = {
	murder_mystery_initialize_runtime = yes
	clr_country_flag = murder_mystery_host_registration_failed
	clr_country_flag = murder_mystery_host_registration_succeeded
	if = {
		limit = { murder_mystery_host_registration_is_valid = yes }
		murder_mystery_initialize_country_runtime = yes
		save_global_event_target_as = murder_mystery_original_host
		capital_scope = {
			save_global_event_target_as = murder_mystery_original_host_capital_state
		}
		set_country_flag = murder_mystery_original_host
		set_global_flag = murder_mystery_host_registered
		set_global_flag = murder_mystery_runtime_active
		clr_global_flag = murder_mystery_runtime_dormant
		clr_global_flag = murder_mystery_runtime_resolved
		set_variable = { global.murder_mystery_lifecycle = constant:murder_mystery_lifecycle.opening_shock }
		add_to_variable = { global.murder_mystery_transaction_sequence = constant:murder_mystery_value.one }
		set_variable = { global.murder_mystery_host_generation = global.murder_mystery_transaction_sequence }
		set_variable = { murder_mystery_host_generation = global.murder_mystery_host_generation }
		set_country_flag = murder_mystery_host_registration_succeeded
	}
	else = {
		set_country_flag = murder_mystery_host_registration_failed
	}
}

	if = {
		limit = {
			has_variable = murder_mystery_requested_phase
			check_variable = { var = murder_mystery_requested_phase value = constant:murder_mystery_phase.secure_succession compare = greater_than_or_equals }

# State scope. The parent map selector adds only states that it has already
# checked for ownership, control, connectivity, and remnant viability.
murder_mystery_register_split_state = {
	if = {
		limit = {
			has_event_target = murder_mystery_original_host
			impassable = no
			is_owned_and_controlled_by = event_target:murder_mystery_original_host
			NOT = { is_in_array = { array = global.murder_mystery_split_state_entries value = THIS } }
			check_variable = { var = global.murder_mystery_split_state_entries^num value = constant:murder_mystery_registry.maximum_split_state_rows compare = less_than }
		}
		add_to_array = { array = global.murder_mystery_split_state_entries value = THIS }

		set_variable = { global.murder_mystery_split_state_count = global.murder_mystery_split_state_entries^num }
	}

# Country scope. This validates an already-selected connected cluster and
# records the caller's remnant proof. It does not choose a random state or
# infer an unsafe split from a single country-size number.
murder_mystery_validate_split_state_registry = {
	clr_global_flag = murder_mystery_split_preflight_passed
	clr_country_flag = murder_mystery_split_cluster_validated
	clr_country_flag = murder_mystery_split_anchor_validated
	clr_country_flag = murder_mystery_host_remnant_validated
	clr_country_flag = murder_mystery_split_share_validated
	if = {
		limit = {
			murder_mystery_host_is_registered = yes
			has_event_target = murder_mystery_split_anchor_state
			has_variable = global.murder_mystery_split_state_entries^num
			check_variable = { var = global.murder_mystery_split_state_entries^num value = constant:murder_mystery_split.minimum_cluster_states compare = greater_than_or_equals }
			check_variable = { var = global.murder_mystery_split_state_entries^num value = constant:murder_mystery_registry.maximum_split_state_rows compare = less_than_or_equals }
			has_variable = murder_mystery_host_remnant_state_count
			check_variable = { var = murder_mystery_host_remnant_state_count value = constant:murder_mystery_split.minimum_remnant_states compare = greater_than_or_equals }
			check_variable = { var = murder_mystery_host_remnant_state_count value = num_of_controlled_states compare = less_than_or_equals }
			has_variable = murder_mystery_actual_split_share
			check_variable = { var = murder_mystery_actual_split_share value = constant:murder_mystery_split.minimum_share compare = greater_than_or_equals }
			check_variable = { var = murder_mystery_actual_split_share value = constant:murder_mystery_split.maximum_share compare = less_than_or_equals }
			is_in_array = { array = global.murder_mystery_split_state_entries value = event_target:murder_mystery_split_anchor_state }
		}
		set_country_flag = murder_mystery_split_cluster_validated
		set_country_flag = murder_mystery_split_anchor_validated
		set_country_flag = murder_mystery_host_remnant_validated
		set_country_flag = murder_mystery_split_share_validated
		set_global_flag = murder_mystery_split_preflight_passed
		set_variable = { global.murder_mystery_split_state_count = global.murder_mystery_split_state_entries^num }
	}
}

murder_mystery_reconcile_derivative_registry = {
	if = {
		limit = { murder_mystery_derivative_registry_is_aligned = yes }
		set_temp_variable = { murder_mystery_derivative_registry_index = global.murder_mystery_derivative_country_entries^num }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_derivative_registry_index value = constant:murder_mystery_value.zero compare = greater_than } }
			subtract_from_temp_variable = { murder_mystery_derivative_registry_index = constant:murder_mystery_identity.loop_increment }
			set_temp_variable = { murder_mystery_derivative_registry_row_invalid = constant:murder_mystery_value.zero }
			set_temp_variable = { murder_mystery_derivative_registry_row_status = global.murder_mystery_derivative_status_entries^murder_mystery_derivative_registry_index }
			if = {
				limit = {
					OR = {
						check_variable = { var = murder_mystery_derivative_registry_row_status value = constant:murder_mystery_derivative.status_pending compare = less_than }
						check_variable = { var = murder_mystery_derivative_registry_row_status value = constant:murder_mystery_derivative.status_retired compare = greater_than }
					}
				}
				set_temp_variable = { murder_mystery_derivative_registry_row_invalid = constant:murder_mystery_value.one }
			}
			var:global.murder_mystery_derivative_country_entries^murder_mystery_derivative_registry_index = {
				if = { limit = { NOT = { exists = yes } } ROOT = { set_temp_variable = { murder_mystery_derivative_registry_row_invalid = constant:murder_mystery_value.one } } }
			}
			# A defeated parent does not remove the row: inheritance reads it.
			if = {
				limit = { check_variable = { var = murder_mystery_derivative_registry_row_invalid value = constant:murder_mystery_value.one compare = equals } }
				remove_from_array = { array = global.murder_mystery_derivative_country_entries index = murder_mystery_derivative_registry_index }
				remove_from_array = { array = global.murder_mystery_derivative_parent_entries index = murder_mystery_derivative_registry_index }
				remove_from_array = { array = global.murder_mystery_derivative_stage_entries index = murder_mystery_derivative_registry_index }
				remove_from_array = { array = global.murder_mystery_derivative_status_entries index = murder_mystery_derivative_registry_index }
				remove_from_array = { array = global.murder_mystery_derivative_generation_entries index = murder_mystery_derivative_registry_index }
			}
		}
		set_variable = { global.murder_mystery_derivative_count = global.murder_mystery_derivative_country_entries^num }
	}
}

# COUNTRY scope: an ordinary government. Adds one registry row, sorted into
# the home, near or far sector by the distance from its capital to the central
# capital, and marks the government as a campaign target. Requires
# event_target:murder_mystery_terminal_sector_origin.
# Output: global flag murder_mystery_terminal_target_registration_succeeded.
murder_mystery_register_terminal_country = {
	clr_global_flag = murder_mystery_terminal_target_registration_succeeded
	if = {
		limit = {
			has_global_flag = murder_mystery_terminal_registry_initialized
			murder_mystery_terminal_registry_has_capacity = yes
			murder_mystery_terminal_country_is_eligible = yes
			has_event_target = murder_mystery_terminal_sector_origin
			NOT = { is_in_array = { array = global.murder_mystery_terminal_country_entries value = THIS } }
		}
		set_temp_variable = { murder_mystery_terminal_row_sector = constant:murder_mystery_terminal.sector_three }
		if = {
			limit = { murder_mystery_terminal_capital_is_in_home_sector = yes }
			set_temp_variable = { murder_mystery_terminal_row_sector = constant:murder_mystery_terminal.sector_one }
		}
		else_if = {
			limit = { murder_mystery_terminal_capital_is_in_near_sector = yes }
			set_temp_variable = { murder_mystery_terminal_row_sector = constant:murder_mystery_terminal.sector_two }
		}
		set_temp_variable = { murder_mystery_terminal_row_cell = constant:murder_mystery_value.zero }
		if = {

# COUNTRY scope: a registered government. Inputs: temp
# murder_mystery_terminal_row_status, _war_state, _administration, _victory and
# _protection. Writes the government's registry row. Only the war hooks and
# the successor administration transaction write a settled victory value.
murder_mystery_record_terminal_row = {
	if = {
		limit = { murder_mystery_terminal_registry_is_aligned = yes }
		save_event_target_as = murder_mystery_terminal_row_target
		set_temp_variable = { murder_mystery_terminal_row_index = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_terminal_row_index value = global.murder_mystery_terminal_country_entries^num compare = less_than } }
			set_temp_variable = { murder_mystery_terminal_row_match = constant:murder_mystery_value.zero }
			var:global.murder_mystery_terminal_country_entries^murder_mystery_terminal_row_index = {
				if = {
					limit = { tag = event_target:murder_mystery_terminal_row_target }
					set_temp_variable = { murder_mystery_terminal_row_match = constant:murder_mystery_value.one }
				}
			}
			if = {
				limit = { check_variable = { var = murder_mystery_terminal_row_match value = constant:murder_mystery_value.one compare = equals } }
				set_temp_variable = { murder_mystery_terminal_previous_victory = global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_row_index }
				set_variable = { global.murder_mystery_terminal_status_entries^murder_mystery_terminal_row_index = murder_mystery_terminal_row_status }
				set_variable = { global.murder_mystery_terminal_war_state_entries^murder_mystery_terminal_row_index = murder_mystery_terminal_row_war_state }
				set_variable = { global.murder_mystery_terminal_administration_entries^murder_mystery_terminal_row_index = murder_mystery_terminal_row_administration }
				set_variable = { global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_row_index = murder_mystery_terminal_row_victory }
				set_variable = { global.murder_mystery_terminal_protection_entries^murder_mystery_terminal_row_index = murder_mystery_terminal_row_protection }
				if = {
					limit = {
						check_variable = { var = murder_mystery_terminal_previous_victory value = constant:murder_mystery_terminal.victory_unresolved compare = equals }
						check_variable = { var = murder_mystery_terminal_row_victory value = constant:murder_mystery_terminal.victory_contribution compare = equals }
					}
					add_to_variable = { global.murder_mystery_terminal_target_completed_count = constant:murder_mystery_value.one }
				}
				set_temp_variable = { murder_mystery_terminal_row_index = global.murder_mystery_terminal_country_entries^num }
			}
			add_to_temp_variable = { murder_mystery_terminal_row_index = constant:murder_mystery_identity.loop_increment }
		}
	}
}

# COUNTRY scope: an open registered government. Brings it into the war with
# the central state: a government already at war is only marked, an
# independent one receives a declaration (after leaving a faction the central
# state leads), and a subject follows its overlord into the war unless that
# overlord is outside the campaign. Input: temp
# murder_mystery_terminal_wave_declared, the declarations made in this wave.
# Output: country flag murder_mystery_terminal_target_at_war.
murder_mystery_engage_terminal_country = {
	if = {
		limit = {
			murder_mystery_terminal_target_is_open = yes
			murder_mystery_terminal_country_is_eligible = yes
			has_event_target = murder_mystery_assassin_state
			event_target:murder_mystery_assassin_state = { exists = yes }
		}
		if = {
			limit = {
				NOT = { has_war_with = event_target:murder_mystery_assassin_state }
				murder_mystery_terminal_country_takes_declaration = yes
				check_variable = { var = murder_mystery_terminal_wave_declared value = constant:murder_mystery_terminal.wave_declaration_limit compare = less_than }
			}
			if = {
				limit = {
					is_in_faction_with = event_target:murder_mystery_assassin_state
					event_target:murder_mystery_assassin_state = { is_faction_leader = yes }
				}
				event_target:murder_mystery_assassin_state = { remove_from_faction = PREV }
			}
			event_target:murder_mystery_assassin_state = {
				declare_war_on = { target = PREV type = annex_everything }
			}
			if = {
				limit = { has_war_with = event_target:murder_mystery_assassin_state }
				add_to_temp_variable = { murder_mystery_terminal_wave_declared = constant:murder_mystery_value.one }
			}
		}
		if = {
			limit = { has_war_with = event_target:murder_mystery_assassin_state }
			set_country_flag = murder_mystery_terminal_target_at_war
			murder_mystery_set_white_peace_protection = yes
		}
	}
}

# Any scope. Input: temp murder_mystery_terminal_wave_sector. One bounded war
# wave over the registered governments of that sector: every open government
# already at war is marked, and up to wave_declaration_limit independent ones
# receive declarations. The pulse runs the same wave again for active sectors.
murder_mystery_run_terminal_war_wave = {
	if = {
		limit = {
			has_global_flag = murder_mystery_terminal_war_active
			murder_mystery_terminal_registry_is_aligned = yes
			has_event_target = murder_mystery_assassin_state
			event_target:murder_mystery_assassin_state = { exists = yes }
		}
		set_temp_variable = { murder_mystery_terminal_wave_declared = constant:murder_mystery_value.zero }
		set_temp_variable = { murder_mystery_terminal_wave_index = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_terminal_wave_index value = global.murder_mystery_terminal_country_entries^num compare = less_than } }
			if = {
				limit = {
					check_variable = { var = global.murder_mystery_terminal_sector_entries^murder_mystery_terminal_wave_index value = murder_mystery_terminal_wave_sector compare = equals }
					check_variable = { var = global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_wave_index value = constant:murder_mystery_terminal.victory_unresolved compare = equals }
				}
				var:global.murder_mystery_terminal_country_entries^murder_mystery_terminal_wave_index = {
					murder_mystery_engage_terminal_country = yes
					if = {
						limit = { has_country_flag = murder_mystery_terminal_target_at_war }
						set_variable = { global.murder_mystery_terminal_status_entries^murder_mystery_terminal_wave_index = constant:murder_mystery_terminal.status_engaged }
						set_variable = { global.murder_mystery_terminal_war_state_entries^murder_mystery_terminal_wave_index = constant:murder_mystery_terminal.status_engaged }
					}
				}
			}
			add_to_temp_variable = { murder_mystery_terminal_wave_index = constant:murder_mystery_identity.loop_increment }
		}
		murder_mystery_select_terminal_campaign_target = yes
		murder_mystery_refresh_terminal_sector_counts = yes
	}
}

# Any scope. Victory needs every registered row settled by a conquest, a
# capitulation followed by a successor administration, or a reconciled
# annexation. A subject administration that broke free reopens its row. When
# every row is settled, one bounded reconciliation registers any ordinary
# government created since activation; victory waits for those as well.
murder_mystery_evaluate_terminal_victory = {
	clr_global_flag = murder_mystery_terminal_registry_victory_ready
	if = {
		limit = {
			has_global_flag = murder_mystery_terminal_war_active
			murder_mystery_terminal_registry_is_aligned = yes
		}
		murder_mystery_reconcile_terminal_registry = yes
		set_temp_variable = { murder_mystery_terminal_unresolved_rows = constant:murder_mystery_value.zero }
		set_temp_variable = { murder_mystery_terminal_registry_index = constant:murder_mystery_identity.first_array_index }
		while_loop_effect = {
			limit = { check_variable = { var = murder_mystery_terminal_registry_index value = global.murder_mystery_terminal_country_entries^num compare = less_than } }
			if = {
				limit = { check_variable = { var = global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_registry_index value = constant:murder_mystery_terminal.victory_contribution compare = equals } }
				var:global.murder_mystery_terminal_country_entries^murder_mystery_terminal_registry_index = {
					if = {
						limit = {
							has_country_flag = murder_mystery_terminal_dismantled
							murder_mystery_terminal_country_is_eligible = yes
						}
						clr_country_flag = murder_mystery_terminal_dismantled
						set_variable = { global.murder_mystery_terminal_status_entries^murder_mystery_terminal_registry_index = constant:murder_mystery_terminal.status_registered }
						set_variable = { global.murder_mystery_terminal_war_state_entries^murder_mystery_terminal_registry_index = constant:murder_mystery_terminal.status_registered }
						set_variable = { global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_registry_index = constant:murder_mystery_terminal.victory_unresolved }
						subtract_from_variable = { global.murder_mystery_terminal_target_completed_count = constant:murder_mystery_value.one }
					}
				}
			}
			if = {
				limit = { check_variable = { var = global.murder_mystery_terminal_victory_entries^murder_mystery_terminal_registry_index value = constant:murder_mystery_terminal.victory_unresolved compare = equals } }
				add_to_temp_variable = { murder_mystery_terminal_unresolved_rows = constant:murder_mystery_value.one }
			}
			add_to_temp_variable = { murder_mystery_terminal_registry_index = constant:murder_mystery_identity.loop_increment }
		}
		if = {
			limit = {
				check_variable = { var = murder_mystery_terminal_unresolved_rows value = constant:murder_mystery_value.zero compare = equals }
				check_variable = { var = global.murder_mystery_terminal_target_completed_count value = constant:murder_mystery_world_end.minimum_terminal_registry_rows compare = greater_than_or_equals }
			}
			murder_mystery_save_terminal_sector_origin = yes
			set_temp_variable = { murder_mystery_terminal_wave_declared = constant:murder_mystery_value.zero }
			every_country = {
				limit = {
					murder_mystery_terminal_country_is_eligible = yes
					NOT = { has_country_flag = murder_mystery_terminal_target_registered }
				}
				murder_mystery_register_terminal_country = yes
				if = {
					limit = { has_global_flag = murder_mystery_terminal_target_registration_succeeded }
					add_to_temp_variable = { murder_mystery_terminal_unresolved_rows = constant:murder_mystery_value.one }
					set_temp_variable = { murder_mystery_terminal_sector_query = murder_mystery_terminal_sector }
					if = {
						limit = { murder_mystery_terminal_sector_query_is_active = yes }
						murder_mystery_engage_terminal_country = yes
					}
				}
			}
			if = {
				limit = { check_variable = { var = murder_mystery_terminal_unresolved_rows value = constant:murder_mystery_value.zero compare = equals } }
				set_global_flag = murder_mystery_terminal_registry_victory_ready
				set_global_flag = murder_mystery_terminal_registry_completion_proven
			}
		}
		murder_mystery_refresh_terminal_sector_counts = yes
	}
}

# ---------------------------------------------------------------------------
# Event 009 White Peace protection
# ---------------------------------------------------------------------------
# Event 009 skips any war actor that carries white_peace_protected_war_actor.
# Event 039 sets it on its own war actors and records that it did, so clearing
# never removes a protection another owner gave the country. The central
# state, every active derivative, the original host while it fights the
# movement and every campaign target at war with it are protected, so an
# unrelated white peace cannot strand a split, an inheritance or a successor
# administration half done.

	if = {
		limit = { NOT = { has_country_flag = white_peace_protected_war_actor } }
		set_country_flag = white_peace_protected_war_actor
		set_country_flag = murder_mystery_white_peace_protection_set
	}
}

# COUNTRY scope.

	if = {

# Any scope, called by the movement pulse. Keeps the protection on exactly the
# countries that need it: the central state, active derivatives, the original
# host while at war with the central state, and campaign targets at war with
# it. Reads only the host, the center and the derivative and terminal
# registries.
murder_mystery_reconcile_white_peace_protection = {
	if = {
		limit = {
			has_global_flag = murder_mystery_assassin_state_exists
			has_event_target = murder_mystery_assassin_state
			event_target:murder_mystery_assassin_state = { exists = yes }
		}
		event_target:murder_mystery_assassin_state = { murder_mystery_set_white_peace_protection = yes }
	}
	for_each_scope_loop = {
		array = global.murder_mystery_derivative_country_entries
		if = {
			limit = {
				exists = yes
				has_country_flag = murder_mystery_derivative_active
			}
			murder_mystery_set_white_peace_protection = yes
		}
		else = { murder_mystery_clear_white_peace_protection = yes }
	}
	if = {
		limit = { has_event_target = murder_mystery_original_host }
		event_target:murder_mystery_original_host = {
			if = {
				limit = {
					exists = yes
					has_global_flag = murder_mystery_assassin_state_exists
					has_event_target = murder_mystery_assassin_state
					has_war_with = event_target:murder_mystery_assassin_state
				}
				murder_mystery_set_white_peace_protection = yes
			}
			else = { murder_mystery_clear_white_peace_protection = yes }
		}
	}
	for_each_scope_loop = {
		array = global.murder_mystery_terminal_country_entries
		if = {
			limit = {
				has_global_flag = murder_mystery_terminal_war_active
				murder_mystery_terminal_target_is_open = yes
				has_event_target = murder_mystery_assassin_state
				has_war_with = event_target:murder_mystery_assassin_state
			}
			murder_mystery_set_white_peace_protection = yes
		}
		else = { murder_mystery_clear_white_peace_protection = yes }
	}
}
