	set_variable = { black_plague_cleanup_attempts = constant:black_plague_response_value.zero }
	set_variable = { black_plague_city_rat_clearance_count = constant:black_plague_response_value.zero }

	clr_state_flag = black_plague_first_response_committed

# subsequent quotes immediately return to their normal population scale.

		black_plague_refresh_current_state_response_quotes = yes
	}
}

# STATE scope; ROOT is the country. Inputs: project and lane.
black_plague_cancel_current_state_response_project = {
	set_temp_variable = { black_plague_response_cancellation_applied = constant:black_plague_response_value.zero }
	if = {
		limit = {
			check_variable = { black_plague_response_lane = constant:black_plague_response_lane.civil }
			has_variable = black_plague_response_civil_lane_owner
			has_variable = black_plague_response_civil_lane_project
			check_variable = { var = black_plague_response_civil_lane_owner value = ROOT compare = equals }
			check_variable = { var = black_plague_response_civil_lane_project value = black_plague_response_project compare = equals }
		}
		black_plague_clear_current_state_civil_lane = yes
		set_temp_variable = { black_plague_response_cancellation_applied = constant:black_plague_response_value.one }
	}
	else_if = {
		limit = {
			check_variable = { black_plague_response_lane = constant:black_plague_response_lane.quarantine }
			has_variable = black_plague_response_quarantine_lane_owner
			has_variable = black_plague_response_quarantine_lane_project
			check_variable = { var = black_plague_response_quarantine_lane_owner value = ROOT compare = equals }
			check_variable = { var = black_plague_response_quarantine_lane_project value = black_plague_response_project compare = equals }
		}
		black_plague_clear_current_state_quarantine_lane = yes
		set_temp_variable = { black_plague_response_cancellation_applied = constant:black_plague_response_value.one }
	}
	else_if = {
		limit = {
			check_variable = { black_plague_response_lane = constant:black_plague_response_lane.cordon }

		}
		clr_country_flag = black_plague_countermeasure_setup_active
		clr_country_flag = black_plague_countermeasure_program_active
		clr_country_flag = black_plague_countermeasure_program_completed
		clr_country_flag = black_plague_countermeasure_findings_available
		clr_country_flag = black_plague_countermeasure_hospital_discount_unlocked
		clr_country_flag = black_plague_countermeasure_cleanup_unlocked
		clr_country_flag = black_plague_medical_reserve_batch_active
		clr_country_flag = black_plague_crisis_board_available
		clr_country_flag = black_plague_response_burden_medium
		clr_country_flag = black_plague_response_burden_high
		clr_country_flag = black_plague_response_burden_extreme
		clear_variable = black_plague_countermeasure_setup_state
		clear_variable = black_plague_medical_reserve
		clear_variable = black_plague_medical_reserve_capacity
		clear_variable = black_plague_medical_reserve_room
		clear_variable = black_plague_medical_reserve_batch
		clear_variable = black_plague_medical_reserve_active_batch
		clear_variable = black_plague_countermeasure_progress
		clear_variable = black_plague_response_capacity
		clear_variable = black_plague_response_capacity_remaining
		clear_variable = black_plague_response_burden
		set_variable = { black_plague_response_persistent_factory_use = constant:black_plague_response_value.zero }
		set_variable = { black_plague_response_consumer_goods_penalty = constant:black_plague_response_value.zero }
		set_variable = { black_plague_response_supply_penalty = constant:black_plague_response_value.zero }
		set_variable = { black_plague_response_factory_output_penalty = constant:black_plague_response_value.zero }
		set_variable = { black_plague_response_stability_penalty = constant:black_plague_response_value.zero }
		if = {
			limit = { has_dynamic_modifier = { modifier = black_plague_persistent_response_burden } }
			remove_dynamic_modifier = { modifier = black_plague_persistent_response_burden }
		}
	}
}

# COUNTRY scope. The active-state callback is invoked after disease growth and
# before mortality/transition, so reductions cannot be erased at a zero floor.
black_plague_process_current_state_response_callback = {
	if = {
		limit = {
			has_event_target = black_plague_response_pulse_state
			event_target:black_plague_response_pulse_state = {
				is_controlled_by = ROOT
				black_plague_state_is_tracked = yes
			}
		}
		event_target:black_plague_response_pulse_state = {
			black_plague_refresh_current_state_response_quotes = yes
			black_plague_register_current_response_state = yes
			black_plague_process_current_state_persistent_response = yes
		}
	}
}

# Any root scope. Rebuilds the two response registries from their live records.
black_plague_rebuild_response_registries = {
	clear_temp_array = black_plague_live_response_states
	clear_temp_array = black_plague_live_response_countries
	for_each_scope_loop = {
		array = global.black_plague_response_states
		black_plague_administratively_reconcile_current_state_response = yes
		if = {
			limit = {
				OR = {
					has_state_flag = black_plague_response_civil_lane_active
					has_state_flag = black_plague_response_quarantine_lane_active
					has_state_flag = black_plague_response_cordon_lane_active
					has_state_flag = black_plague_response_treatment_lane_active
					has_state_flag = black_plague_response_state_target
					has_state_flag = black_plague_countermeasure_site_active
					has_state_flag = black_plague_emergency_hospital_active
					has_state_flag = black_plague_quarantine_active
					has_state_flag = black_plague_quarantine_monitoring
					has_state_flag = black_plague_cordon_active
					has_state_flag = black_plague_sealed_food_stores_active
					has_state_flag = black_plague_hidden_nests_suppressed
					has_state_flag = black_plague_flea_control_active
					has_state_flag = black_plague_transport_purge_active
					has_state_flag = black_plague_treatment_relief_active
					has_state_flag = black_plague_liberated_warrens_suppressed
					AND = {
						NOT = { black_plague_state_is_cured = yes }
						check_variable = { black_plague_cleanup_attempts > constant:black_plague_response_value.zero }
					}
				}
			}
			add_to_temp_array = { black_plague_live_response_states = THIS }
			if = {
				limit = { has_variable = black_plague_response_civil_lane_owner }
				var:black_plague_response_civil_lane_owner = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
			if = {
				limit = { has_variable = black_plague_response_quarantine_lane_owner }
				var:black_plague_response_quarantine_lane_owner = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
			if = {
				limit = { has_variable = black_plague_response_cordon_lane_owner }
				var:black_plague_response_cordon_lane_owner = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
			if = {
				limit = { has_variable = black_plague_response_treatment_lane_owner }
				var:black_plague_response_treatment_lane_owner = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
			if = {
				limit = { has_variable = black_plague_response_persistent_owner }
				var:black_plague_response_persistent_owner = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
			if = {
				limit = { has_variable = black_plague_shared_action_owner }
				var:black_plague_shared_action_owner = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
			if = {
				limit = {
					check_variable = { black_plague_cleanup_attempts > constant:black_plague_response_value.zero }
					CONTROLLER = { black_plague_country_is_human_host = yes }
				}
				CONTROLLER = {
					if = {
						limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
						add_to_temp_array = { black_plague_live_response_countries = THIS }
					}
				}
			}
		}
	}
	clear_array = global.black_plague_response_states
	for_each_scope_loop = {
		array = black_plague_live_response_states
		add_to_array = { array = global.black_plague_response_states value = THIS }
	}
	clear_temp_array = black_plague_live_response_states

	for_each_scope_loop = {
		array = global.black_plague_response_countries
		if = {
			limit = {
				exists = yes
				OR = {
					has_country_flag = black_plague_countermeasure_setup_active
					has_country_flag = black_plague_countermeasure_program_active
					has_country_flag = black_plague_medical_reserve_batch_active
					has_country_flag = black_plague_shared_country_action_active
					has_variable = black_plague_selected_response_state
					check_variable = { black_plague_medical_reserve > constant:black_plague_response_value.zero }
					check_variable = { black_plague_countermeasure_progress > constant:black_plague_response_value.zero }
					black_plague_country_has_strong_prevention = yes
				}
			}
			if = {
				limit = { NOT = { is_in_array = { array = black_plague_live_response_countries value = THIS } } }
				add_to_temp_array = { black_plague_live_response_countries = THIS }
			}
		}
	}
	clear_array = global.black_plague_response_countries
	for_each_scope_loop = {
		array = black_plague_live_response_countries
		add_to_array = { array = global.black_plague_response_countries value = THIS }
	}
	clear_temp_array = black_plague_live_response_countries
}
