# Adds the Event 006 diplomacy bias against the frozen former-host target.
# The target is an execution event target rather than a country literal: a
# later ICE release can be hosted by any valid surviving country. Each channel
# is applied once per package generation and is reversed before cleanup.
independence_wave_ice_apply_host_ai = {
	if = {
		limit = {
			is_independence_wave_ice_package = yes
			has_country_flag = independence_wave_ice_ai_profile
			NOT = { has_country_flag = independence_wave_ice_host_ai_applied }
			has_event_target = independence_wave_setup_former_host
		}
		add_ai_strategy = { type = befriend id = event_target:independence_wave_setup_former_host value = @CR_SC_INDEPENDENCE_WAVE_ICE_AI_TRADE }
		set_country_flag = independence_wave_ice_host_ai_trade_applied
		if = {
			limit = { has_independence_wave_living_former_host = yes }
			add_ai_strategy = { type = befriend id = event_target:independence_wave_setup_former_host value = @CR_SC_INDEPENDENCE_WAVE_ICE_AI_DIPLOMACY }
			add_ai_strategy = { type = prepare_for_war id = event_target:independence_wave_setup_former_host value = @CR_SC_INDEPENDENCE_WAVE_ICE_AI_WAR_PREPARATION }
			set_country_flag = independence_wave_ice_host_ai_charter_applied
		}
		if = {
			limit = { has_country_flag = independence_wave_formable_family_registered }
			add_ai_strategy = { type = befriend id = event_target:independence_wave_setup_former_host value = @CR_SC_INDEPENDENCE_WAVE_ICE_AI_COMPACT_DIPLOMACY }
			add_ai_strategy = { type = befriend id = event_target:independence_wave_setup_former_host value = @CR_SC_INDEPENDENCE_WAVE_ICE_AI_COMPACT_RELATIONS }

	}
}
