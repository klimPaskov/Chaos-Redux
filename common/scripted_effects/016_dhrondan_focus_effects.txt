# Regime, army, laboratory, diplomacy, expansion, and crisis focus mechanics

# Each reward below changes a live D'Rhondan system: the paid landing (price
# through tithes and refunds, recovery, descent time, and per-landing results),
# landing-state buildings and state modifiers, cores and claims, the three
# sovereignty decisions, the Two-World Compact and partner accord, the army
# lifecycle and commanders, laser production, and intelligence. Values live in
# dhrondan_focus_mechanic and dhrondan_focus_mechanic_factor. Static fields
# that reject constants use the file-scoped values below, which mirror
# enclave_network_civilian_intel, enclave_network_army_intel,
# field_calculant_planning, materials_board_laser_factories,
# materials_board_laser_efficiency, and synod_industrial_sites.

# COUNTRY scope on DHR. Called by dhrondan_record_successful_landing after every
# successful landing while var:dhrondan_landing_state_id names the landed state.
# The underwritten landing returns the weapons it actually cost: temp
# alien_infantry_landing_settled_amount from a settled reservation, otherwise
# the full reserve.
dhrondan_apply_landing_dividends = {
	if = {
		limit = { tag = DHR }
		if = {
			limit = { has_country_flag = dhrondan_convention_landing_underwritten }
			clr_country_flag = dhrondan_convention_landing_underwritten
			set_temp_variable = { dhrondan_underwritten_lasers = constant:alien_infantry_landing.reserve_equipment }
			if = {
				limit = { check_variable = { alien_infantry_landing_settled_amount > constant:alien_infantry_landing.zero } }
				set_temp_variable = { dhrondan_underwritten_lasers = alien_infantry_landing_settled_amount }
			}
			add_equipment_to_stockpile = { type = alien_laser_weapon_equipment_1 amount = dhrondan_underwritten_lasers }
		}
		if = {
			limit = {
				has_country_flag = dhrondan_landing_lords_bound
				dhrondan_is_imperial_regime = yes
			}
			add_equipment_to_stockpile = { type = alien_laser_weapon_equipment_1 amount = constant:dhrondan_focus_mechanic.landing_tithe_lasers }
		}
		if = {
			limit = { has_country_flag = dhrondan_expeditionary_cadres_rebuilt }
			army_experience = constant:dhrondan_focus_mechanic.cadre_army_experience_per_landing
		}
		if = {
			limit = { has_variable = dhrondan_landing_state_id }
			var:dhrondan_landing_state_id = {
				if = {
					limit = {
						is_owned_by = PREV
						PREV = {
							has_country_flag = dhrondan_ninth_diadem_restored
							dhrondan_is_imperial_regime = yes
						}
					}
					dhrondan_fortify_landing_state = yes
				}
				if = {
					limit = {
						is_owned_by = PREV
						PREV = {
							has_country_flag = dhrondan_dual_citizenship_code_written
							dhrondan_is_covenant_regime = yes
						}
					}
					dhrondan_grant_dual_citizenship = yes
				}
			}
		}
	}
}

# COUNTRY scope. Called by alien_infantry_apply_landing_cooldown for DHR after
# the shared recovery ladder chose its tier in temp
# alien_infantry_landing_cooldown_days.
dhrondan_adjust_landing_cooldown = {
	if = {
		limit = {
			has_country_flag = dhrondan_command_nodes_audited
			dhrondan_is_synod_regime = yes
		}
		subtract_from_temp_variable = { alien_infantry_landing_cooldown_days = constant:dhrondan_focus_mechanic.audited_cooldown_reduction }
	}
	if = {
		limit = {
			has_country_flag = dhrondan_foreseen_counterstroke_prepared
			has_war = yes
		}
		subtract_from_temp_variable = { alien_infantry_landing_cooldown_days = constant:dhrondan_focus_mechanic.wartime_cooldown_reduction }
	}
	clamp_temp_variable = { var = alien_infantry_landing_cooldown_days min = constant:dhrondan_focus_mechanic.minimum_landing_cooldown_days }
}

# COUNTRY scope. Called by both landing reservation entry points for DHR after
# the reservation length is set and before the landing mission starts.
dhrondan_adjust_landing_reservation = {
	if = {
		limit = {
			has_country_flag = dhrondan_human_signal_teams_trained
			check_variable = { alien_infantry_landing_reservation_days > constant:dhrondan_focus_mechanic.signal_reservation_days }
		}
		set_variable = { alien_infantry_landing_reservation_days = constant:dhrondan_focus_mechanic.signal_reservation_days }
	}
}

# STATE scope on an owned landing state. Imperial strongholds.

		}
	}
}

# STATE scope on an owned landing state. Covenant dual citizenship.
dhrondan_grant_dual_citizenship = {
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = dhrondan_dual_citizenship_enclave } } }
		add_dynamic_modifier = { modifier = dhrondan_dual_citizenship_enclave }
	}
}

# COUNTRY scope on DHR. Input temp dhrondan_calculant_refund_cost: the Political
# Power the completed sovereignty order cost. A Synod with elevated calculants
# recovers the configured share.
dhrondan_apply_calculant_refund = {
	if = {
		limit = {
			dhrondan_is_synod_regime = yes
			has_country_flag = dhrondan_first_calculants_elevated
		}
		custom_effect_tooltip = dhrondan_calculant_refund_tt
		multiply_temp_variable = { dhrondan_calculant_refund_cost = constant:dhrondan_focus_mechanic_factor.calculant_refund_share }
		add_political_power = dhrondan_calculant_refund_cost
	}
}

		}
	}
