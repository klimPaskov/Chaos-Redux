# ============================================================================
# CHAOS REDUX - EMPIRE OF D'RHONDA FOCUS EFFECTS
# ============================================================================
# Narrow focus-owned transitions for DHR. These effects deliberately expose
# API and decision contracts through flags and variables; they never create a
# division, train alien infantry, or grant a free cohort. The parent-owned
# generic landing API consumes dhrondan_landing_equipment_cost when enabled.
# ============================================================================

@dhr_focus_building_level = 1

dhrondan_focus_clear_homeworld_lifecycle = {
	if = { limit = { has_idea = dhrondan_homeworld_fragmentation } remove_ideas = dhrondan_homeworld_fragmentation }
	if = { limit = { has_idea = dhrondan_homeworld_cohesion } remove_ideas = dhrondan_homeworld_cohesion }
	if = { limit = { has_idea = dhrondan_imperial_mandate } remove_ideas = dhrondan_imperial_mandate }
	if = { limit = { has_idea = dhrondan_synod_calculus } remove_ideas = dhrondan_synod_calculus }
	if = { limit = { has_idea = dhrondan_covenant_compact } remove_ideas = dhrondan_covenant_compact }
}

dhrondan_focus_expand_laser_production = {
	set_country_flag = dhrondan_laser_production_expanded
	random_owned_controlled_state = {
		limit = {
			free_building_slots = {
				building = arms_factory
				size > 0
				include_locked = yes
			}
		}
		add_extra_state_shared_building_slots = @dhr_focus_building_level
		add_building_construction = {
			type = arms_factory
			level = @dhr_focus_building_level
			instant_build = yes
		}
	}
}
