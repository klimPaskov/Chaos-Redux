		limit = { has_country_flag = utopia_manifesto_accepted }

	}
}

# Actor scope, with FROM as the verified active Necessary Ground target.
utopia_manifesto_record_enforcement_war_milestone = {
	if = {
		limit = {
			has_country_flag = utopia_manifesto_accepted
			has_country_flag = utopia_manifesto_case_enforcement_war_active
			has_country_flag = utopia_manifesto_need_case_active
			FROM = { has_country_flag = utopia_manifesto_active_case_target }
		}
		if = {
			limit = { NOT = { has_country_flag = utopia_manifesto_necessary_ground_war_news_emitted } }
			set_country_flag = utopia_manifesto_necessary_ground_war_news_emitted
			news_event = { id = chaosx.nr15.161 }
		}
		utopia_manifesto_try_dispatch_colonial_reaction = yes
	}
}
