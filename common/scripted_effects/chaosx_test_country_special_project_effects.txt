# ============================================================================
# Overview:
# - Completes every installed vanilla and Chaos Redux special project.

# - Projects are completed before the final technology grant so all project outputs resolve first.

# Country scope. PROJECT is a static installed project id; repeated calls are safe.
chaosx_test_country_complete_special_project = {
	if = {
		limit = { NOT = { is_special_project_completed = sp:$PROJECT$ } }
		complete_special_project = {
			project = sp:$PROJECT$
			show_modifiers = no
		}
	}
}

# Input: temporary variable chaosx_test_country_registration_special_project
# containing an sp:<project_id> special-project scope. Special projects use the
# dedicated sp: scope form rather than the token:<id> form used by equipment and
# sub-units. The owning package's registered CXT setup effect calls this helper;
# the registry persists globally for the save.
chaosx_test_country_register_special_project = {
	if = {
		limit = { has_variable = chaosx_test_country_registration_special_project }
		if = {
			limit = {
				NOT = {
					is_in_array = {
						array = global.chaosx_test_country_registered_special_projects
						value = chaosx_test_country_registration_special_project
					}
				}
			}
			add_to_array = {
				array = global.chaosx_test_country_registered_special_projects
				value = chaosx_test_country_registration_special_project
			}
		}
	}
}

chaosx_test_country_complete_registered_special_projects = {
	set_country_flag = chaosx_test_country_silent_unlocks
	for_each_loop = {
		array = global.chaosx_test_country_registered_special_projects
		value = chaosx_test_country_current_special_project
		if = {
			limit = { NOT = { is_special_project_completed = var:chaosx_test_country_current_special_project } }
			complete_special_project = { project = var:chaosx_test_country_current_special_project show_modifiers = no }
		}
	}
	clr_country_flag = chaosx_test_country_silent_unlocks
}
