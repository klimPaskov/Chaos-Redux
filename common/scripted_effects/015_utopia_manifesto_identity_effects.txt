# ============================================================================
# CHAOS REDUX - EVENT 015 UTOPIA MANIFESTO IDENTITY EFFECTS
# ============================================================================
# Country-scope API for staged institutions, parties, cosmetic identities,
# institutional leaders, succession, and durable achievement evidence.
#
# Public integration effects:
# - utopia_manifesto_initialize_identity_package
# - utopia_manifesto_commit_current_route_identity
# - utopia_manifesto_form_current_route_identity
# - utopia_manifesto_advance_current_route_succession
#
# Formation delegates proof and lifecycle state to the central Event 015
# formation helper. It changes only politics, leader presentation, and the
# cosmetic identity. It never annexes league members, transfers states, adds
# cores, restores reserves, creates units, or refunds formation costs.
# ============================================================================

utopia_manifesto_commit_current_route_identity = {
	utopia_manifesto_clear_route_institution_identity_flags = yes
	clr_country_flag = utopia_manifesto_route_institution_established
	utopia_manifesto_clear_found_manifesto_idea_stages = yes
	if = {
		limit = {
			OR = {
				has_idea = utopia_manifesto_unmeasured_country
				has_idea = utopia_manifesto_unmeasured_country_mitigated
				has_idea = utopia_manifesto_unmeasured_country_failure
			}
		}
		utopia_manifesto_finalize_public_administration = yes
	}
	utopia_manifesto_apply_route_property_settlement = yes
	if = {
		limit = { has_country_flag = utopia_manifesto_route_consent_of_households }
		set_country_flag = utopia_manifesto_institution_household_assembly
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_common_table }
		set_country_flag = utopia_manifesto_institution_council_of_callings
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_guardians_of_measure }
		set_country_flag = utopia_manifesto_institution_board_of_measure
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_closed_island }
		set_country_flag = utopia_manifesto_institution_first_steward_regime
	}
	else_if = {
		limit = { has_country_flag = utopia_manifesto_route_joke_understood }
		set_country_flag = utopia_manifesto_institution_humanist_cabinet
		utopia_manifesto_name_current_party_humanist = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = utopia_manifesto_institution_household_assembly
				has_country_flag = utopia_manifesto_institution_council_of_callings
				has_country_flag = utopia_manifesto_institution_board_of_measure
				has_country_flag = utopia_manifesto_institution_first_steward_regime
				has_country_flag = utopia_manifesto_institution_humanist_cabinet
			}
		}
		set_country_flag = utopia_manifesto_route_institution_established
		utopia_manifesto_apply_route_institution_founding = yes
	}
}

utopia_manifesto_form_current_route_identity = {
	utopia_manifesto_complete_formation = yes

		limit = {
			has_country_flag = utopia_manifesto_commonwealth_formed
			has_country_flag = utopia_manifesto_formation_proof_met
			NOT = { has_country_flag = utopia_manifesto_identity_formed }
		}
		utopia_manifesto_clear_formed_identity_flags = yes
		utopia_manifesto_clear_succession_identity_flags = yes
		if = {
			limit = { has_country_flag = utopia_manifesto_route_consent_of_households }
			utopia_manifesto_form_voluntary_commonwealth_identity = yes
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_common_table }
			utopia_manifesto_form_council_union_identity = yes
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_guardians_of_measure }
			utopia_manifesto_form_planned_utopia_identity = yes
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_closed_island }
			utopia_manifesto_form_closed_island_identity = yes
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_route_joke_understood }
			utopia_manifesto_form_practical_commonwealth_identity = yes
		}
		if = {
			limit = {
				OR = {
					has_country_flag = utopia_manifesto_identity_voluntary_commonwealth
					has_country_flag = utopia_manifesto_identity_council_union
					has_country_flag = utopia_manifesto_identity_planned_utopia
					has_country_flag = utopia_manifesto_identity_closed_island
					has_country_flag = utopia_manifesto_identity_practical_commonwealth
				}
			}
			set_country_flag = utopia_manifesto_identity_formed
		}
	}

			limit = { has_country_flag = utopia_manifesto_identity_closed_island }
			utopia_manifesto_install_directorate_of_service_successor = yes
		}
		else_if = {
			limit = { has_country_flag = utopia_manifesto_identity_practical_commonwealth }
			utopia_manifesto_install_constitutional_election_successor = yes
		}

		}
	}
}

# Call once for each completed Need case before the central case runtime is
# cleared. The country array prevents one target from satisfying the distinct-
# case requirement more than once.
utopia_manifesto_record_achievement_peaceful_case_resolution = {
	if = {
		limit = { has_variable = utopia_manifesto_case_target_id }
		set_temp_variable = { utopia_manifesto_achievement_case_target_input = utopia_manifesto_case_target_id }
	}
	if = {
		limit = { has_variable = utopia_manifesto_case_method }
		set_temp_variable = { utopia_manifesto_achievement_case_method_input = utopia_manifesto_case_method }
	}
	if = {
		limit = {
			has_variable = utopia_manifesto_achievement_case_target_input
			has_variable = utopia_manifesto_achievement_case_method_input
			OR = {
				check_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.purchase }
				check_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.long_supply_contract }
				check_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.lease }
				check_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.settlement_agreement }
				check_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.joint_administration }
				check_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.association }
				has_country_flag = utopia_manifesto_case_closed_for_domestic_alternative
			}
			NOT = {
				is_in_array = {
					array = utopia_manifesto_achievement_peaceful_case_targets
					value = utopia_manifesto_achievement_case_target_input
				}
			}
		}
		add_to_array = {
			array = utopia_manifesto_achievement_peaceful_case_targets
			value = utopia_manifesto_achievement_case_target_input
		}
		set_variable = { utopia_manifesto_achievement_peaceful_case_count = utopia_manifesto_achievement_peaceful_case_targets^num }
		if = {
			limit = {
				check_variable = {
					var = utopia_manifesto_achievement_peaceful_case_count
					value = constant:utopia_manifesto_achievement_requirement.peaceful_case_count
					compare = greater_than_or_equals
				}
			}
			set_country_flag = utopia_manifesto_achievement_peaceful_cases_proven
		}
	}
	set_temp_variable = { utopia_manifesto_achievement_case_target_input = constant:utopia_manifesto_narrative_count.zero }
	set_temp_variable = { utopia_manifesto_achievement_case_method_input = constant:utopia_manifesto_case_method.none }
}

		set_country_flag = utopia_manifesto_achievement_obsolete_case_renounced

# Called only by the successful end of the sustained Choice/Concord/Plenty

			}

		}
		set_country_flag = utopia_manifesto_achievement_reserve_challenge_survived
	}
}

# Refresh after league membership, reserve, or defense arrays change.

			check_variable = {
				var = utopia_manifesto_league_member_count
				value = constant:utopia_manifesto_achievement_requirement.league_member_count
				compare = greater_than_or_equals
			}
			check_variable = {
				var = utopia_manifesto_league_reserve_count
				value = constant:utopia_manifesto_narrative_count.zero
				compare = greater_than
			}
			check_variable = {
				var = utopia_manifesto_league_defense_count
				value = constant:utopia_manifesto_narrative_count.zero
				compare = greater_than

		}
		set_country_flag = utopia_manifesto_achievement_five_member_league_proven
	}
}

# Call before the current case target is cleared. Repeating a vote for a target
# already in the array permanently disqualifies the achievement.
utopia_manifesto_record_achievement_status_vote = {
	if = {
		limit = {
			has_variable = utopia_manifesto_case_target_id
			OR = {
				has_country_flag = utopia_manifesto_route_consent_of_households
				has_country_flag = utopia_manifesto_route_joke_understood
			}
			has_country_flag = utopia_manifesto_stewardship_status_vote_enabled
			has_country_flag = utopia_manifesto_partner_autonomy_retained
		}
		if = {
			limit = {
				is_in_array = {
					array = utopia_manifesto_achievement_status_vote_targets
					value = utopia_manifesto_case_target_id
				}
			}
			set_country_flag = utopia_manifesto_achievement_disqualifier_repeated_status_vote
		}
		else = {
			add_to_array = {
				array = utopia_manifesto_achievement_status_vote_targets
				value = utopia_manifesto_case_target_id
			}
			set_variable = { utopia_manifesto_achievement_status_vote_count = utopia_manifesto_achievement_status_vote_targets^num }
			if = {
				limit = {
					check_variable = {
						var = utopia_manifesto_achievement_status_vote_count
						value = constant:utopia_manifesto_achievement_requirement.status_vote_count
						compare = greater_than_or_equals
					}
				}
				set_country_flag = utopia_manifesto_achievement_status_votes_proven
			}
		}
	}
}

utopia_manifesto_record_achievement_crisis_recovery = {
	if = {
		limit = {
			has_country_flag = utopia_manifesto_crisis_branch_entered
			OR = {
				has_country_flag = utopia_manifesto_crisis_corrected_by_consent
				has_country_flag = utopia_manifesto_crisis_corrected_by_councils
				has_country_flag = utopia_manifesto_crisis_corrected_by_measure
				has_country_flag = utopia_manifesto_crisis_corrected_by_revision
				has_country_flag = utopia_manifesto_crisis_corrected_by_separation
			}
			check_variable = {
				var = utopia_plenty
				value = constant:utopia_manifesto_formation_limits.minimum_plenty
				compare = greater_than_or_equals
			}
			check_variable = {
				var = utopia_concord
				value = constant:utopia_manifesto_formation_limits.minimum_concord
				compare = greater_than_or_equals
			}
			OR = {
				has_idea = utopia_manifesto_common_store_network
				has_idea = utopia_manifesto_common_store_network_mitigated
				has_idea = utopia_manifesto_common_store_network_final
			}
			NOT = { has_country_flag = utopia_manifesto_institutions_abandoned }
			NOT = { has_country_flag = utopia_manifesto_achievement_disqualifier_total_repeal }
		}
		set_country_flag = utopia_manifesto_achievement_crisis_recovery_proven
	}
}
