# ============================================================================
# CHAOS REDUX - EVENT 014 SCRIPTED GUI EFFECTS
# ============================================================================
# Player-only view builders for the Evolution II network ledger. They copy
# already-registered actor and node scopes into country-scoped view arrays,
# preserving gameplay ownership in the core ledger and decision systems.
# No gameplay cost, population transaction, cooldown, or AI action lives here.
# ============================================================================

@cannibalism_gui_loop_zero = 0
@cannibalism_gui_loop_one = 1

cannibalism_gui_append_current_country_to_view_if_valid = {
	if = {
		limit = { cannibalism_gui_country_matches_selected_tab = yes }
		ROOT = { add_to_array = { cannibalism_network_gui_country_entries = PREV } }
	}
}

cannibalism_gui_append_current_state_to_view_if_valid = {
	if = {
		limit = { cannibalism_gui_state_matches_selected_tab = yes }
		ROOT = { add_to_array = { cannibalism_network_gui_state_entries = PREV } }
	}
}

cannibalism_gui_rebuild_network_view = {
	clear_array = cannibalism_network_gui_country_entries
	clear_array = cannibalism_network_gui_state_entries
	if = {
		limit = {
			check_variable = { cannibalism_network_gui_sort = constant:cannibalism_gui_sort.newest_first }
		}
		set_temp_variable = { cannibalism_network_gui_reverse_index = global.cannibalism_actor_countries^num }
		subtract_from_temp_variable = { cannibalism_network_gui_reverse_index = constant:cannibalism_gui_value.one }
		for_loop_effect = {
			start = @cannibalism_gui_loop_zero
			end = global.cannibalism_actor_countries^num
			add = @cannibalism_gui_loop_one
			value = cannibalism_network_gui_loop_index
			var:global.cannibalism_actor_countries^cannibalism_network_gui_reverse_index = {
				cannibalism_gui_append_current_country_to_view_if_valid = yes
			}
			subtract_from_temp_variable = { cannibalism_network_gui_reverse_index = constant:cannibalism_gui_value.one }
		}

		set_temp_variable = { cannibalism_network_gui_reverse_index = global.cannibalism_node_states^num }
		subtract_from_temp_variable = { cannibalism_network_gui_reverse_index = constant:cannibalism_gui_value.one }
		for_loop_effect = {
			start = @cannibalism_gui_loop_zero
			end = global.cannibalism_node_states^num
			add = @cannibalism_gui_loop_one
			value = cannibalism_network_gui_loop_index
			var:global.cannibalism_node_states^cannibalism_network_gui_reverse_index = {
				cannibalism_gui_append_current_state_to_view_if_valid = yes
			}
			subtract_from_temp_variable = { cannibalism_network_gui_reverse_index = constant:cannibalism_gui_value.one }
		}
	}
	else = {
		for_each_scope_loop = {
			array = global.cannibalism_actor_countries
			cannibalism_gui_append_current_country_to_view_if_valid = yes
		}
		for_each_scope_loop = {
			array = global.cannibalism_node_states
			cannibalism_gui_append_current_state_to_view_if_valid = yes
		}
	}
	add_to_variable = { cannibalism_network_gui_dirty = constant:cannibalism_gui_value.one }
}

cannibalism_gui_open_network_view = {
	if = {
		limit = { NOT = { has_variable = cannibalism_network_gui_tab } }
		set_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.domestic_cells }
	}
	if = {
		limit = { NOT = { has_variable = cannibalism_network_gui_sort } }
		set_variable = { cannibalism_network_gui_sort = constant:cannibalism_gui_sort.oldest_first }
	}
	set_country_flag = cannibalism_network_gui_open
	cannibalism_gui_rebuild_network_view = yes
}

cannibalism_gui_close_network_view = {
	clr_country_flag = cannibalism_network_gui_open
	clear_array = cannibalism_network_gui_country_entries
	clear_array = cannibalism_network_gui_state_entries
	clear_variable = cannibalism_network_gui_selected_country
	clear_variable = cannibalism_network_gui_selected_state
	add_to_variable = { cannibalism_network_gui_dirty = constant:cannibalism_gui_value.one }
}

cannibalism_gui_toggle_network_sort = {
	if = {
		limit = { check_variable = { cannibalism_network_gui_sort = constant:cannibalism_gui_sort.newest_first } }
		set_variable = { cannibalism_network_gui_sort = constant:cannibalism_gui_sort.oldest_first }
	}
	else = {
		set_variable = { cannibalism_network_gui_sort = constant:cannibalism_gui_sort.newest_first }
	}
	cannibalism_gui_rebuild_network_view = yes
}

cannibalism_gui_select_country_entry = {
	set_variable = {
		cannibalism_network_gui_selected_country = cannibalism_network_gui_country_entries^cannibalism_network_gui_country_index
	}
	clear_variable = cannibalism_network_gui_selected_state
	add_to_variable = { cannibalism_network_gui_dirty = constant:cannibalism_gui_value.one }
}

cannibalism_gui_select_state_entry = {
	set_variable = {
		cannibalism_network_gui_selected_state = cannibalism_network_gui_state_entries^cannibalism_network_gui_state_index
	}
	clear_variable = cannibalism_network_gui_selected_country
	add_to_variable = { cannibalism_network_gui_dirty = constant:cannibalism_gui_value.one }
}

cannibalism_gui_toggle_animations = {
	if = {
		limit = { has_country_flag = cannibalism_gui_animations_disabled }
		clr_country_flag = cannibalism_gui_animations_disabled
	}
	else = {
		set_country_flag = cannibalism_gui_animations_disabled
	}
	add_to_variable = { cannibalism_network_gui_dirty = constant:cannibalism_gui_value.one }
}
