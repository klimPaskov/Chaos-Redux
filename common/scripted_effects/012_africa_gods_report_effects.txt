# Event 012 Gods response receipts.
# Capture a response before demand cleanup, then render only that outcome.
# These private fields never add counters to the decision categories.

gods_of_africa_capture_response_report = {
	if = {
		limit = { has_variable = gods_of_africa_demand_family }
		set_variable = { gods_of_africa_last_response_kind = gods_of_africa_report_kind }
		set_variable = { gods_of_africa_last_response_family = gods_of_africa_demand_family }
		clear_variable = gods_of_africa_last_response_mobile_type
		if = {
			limit = {
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.mobile_artillery }
				has_variable = gods_of_africa_demand_mobile_type
			}
			set_variable = { gods_of_africa_last_response_mobile_type = gods_of_africa_demand_mobile_type }
		}
		set_variable = { gods_of_africa_last_response_amount = gods_of_africa_report_amount }
		set_variable = { gods_of_africa_last_response_balance = gods_of_africa_report_balance }
		gods_of_africa_capture_response_subject = yes
		# Owners that can break agreed terms set this temporary kind explicitly
		# before every capture; zero means the answer broke no agreement.
		clear_variable = gods_of_africa_last_response_breach_kind
		if = {
			limit = { check_variable = { gods_of_africa_report_breach_kind > constant:gods_of_africa_value.zero } }
			set_variable = { gods_of_africa_last_response_breach_kind = gods_of_africa_report_breach_kind }
		}
	}
	set_temp_variable = { gods_of_africa_report_breach_kind = constant:gods_of_africa_value.zero }
}

# COUNTRY scope on a participant. A payment or voluntary shipment that moved
# nothing is credited nothing. The participant sees the result, and the same
# answer stays closed for a short time so neither a player nor the AI resends
# it every day. Caller sets temporary gods_of_africa_report_family.
gods_of_africa_record_undelivered_payment = {
	set_variable = { gods_of_africa_last_response_kind = constant:gods_of_africa_history_kind.undelivered }
	set_variable = { gods_of_africa_last_response_family = gods_of_africa_report_family }
	clear_variable = gods_of_africa_last_response_mobile_type
	if = {
		limit = {
			check_variable = { gods_of_africa_report_family = constant:gods_of_africa_demand_family.mobile_artillery }
			has_variable = gods_of_africa_demand_mobile_type
		}
		set_variable = { gods_of_africa_last_response_mobile_type = gods_of_africa_demand_mobile_type }
	}
	set_variable = { gods_of_africa_last_response_amount = constant:gods_of_africa_value.zero }
	set_variable = { gods_of_africa_last_response_balance = constant:gods_of_africa_value.zero }
	clear_variable = gods_of_africa_last_response_breach_kind
	set_variable = { gods_of_africa_payment_retry_day = global.num_days }
	add_to_variable = { gods_of_africa_payment_retry_day = constant:gods_of_africa_timing.undelivered_retry_days }
	set_country_flag = gods_of_africa_outcome_report_pending
	country_event = { id = chaosx.nr12.604 }
}

# COUNTRY scope on a participant during gods_of_africa_capture_response_report.
# The states or enemy a diplomatic demand named survive demand cleanup so the
# report can name them (decision L25).
gods_of_africa_capture_response_subject = {
	clear_variable = gods_of_africa_last_state_count
	clear_variable = gods_of_africa_last_state_1
	clear_variable = gods_of_africa_last_state_2
	clear_variable = gods_of_africa_last_state_3
	clear_variable = gods_of_africa_last_support_enemy
	clr_country_flag = gods_of_africa_last_support_kind_guarantee
	clr_country_flag = gods_of_africa_last_support_kind_access
	clr_country_flag = gods_of_africa_last_support_kind_volunteers
	if = { limit = { has_variable = gods_of_africa_demand_state_count } set_variable = { gods_of_africa_last_state_count = gods_of_africa_demand_state_count } }
	if = { limit = { has_variable = gods_of_africa_demand_state_1 } set_variable = { gods_of_africa_last_state_1 = gods_of_africa_demand_state_1 } }
	if = { limit = { has_variable = gods_of_africa_demand_state_2 } set_variable = { gods_of_africa_last_state_2 = gods_of_africa_demand_state_2 } }
	if = { limit = { has_variable = gods_of_africa_demand_state_3 } set_variable = { gods_of_africa_last_state_3 = gods_of_africa_demand_state_3 } }
	if = { limit = { has_variable = gods_of_africa_support_enemy } set_variable = { gods_of_africa_last_support_enemy = gods_of_africa_support_enemy } }
	if = { limit = { has_country_flag = gods_of_africa_support_kind_guarantee } set_country_flag = gods_of_africa_last_support_kind_guarantee }
	if = { limit = { has_country_flag = gods_of_africa_support_kind_access } set_country_flag = gods_of_africa_last_support_kind_access }
	if = { limit = { has_country_flag = gods_of_africa_support_kind_volunteers } set_country_flag = gods_of_africa_last_support_kind_volunteers }
}
