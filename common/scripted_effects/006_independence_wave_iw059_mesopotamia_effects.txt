# CHAOS REDUX - EVENT 006 IW-059 MESOPOTAMIA OVERLAY EFFECTS
# ============================================================================
# Additive mechanics for a carrier that has completed the vanilla
# neo_mesopotamia formable. The effect never creates a country, transfers
# territory, rewrites autonomy, or replaces the carrier's existing tree.
# ============================================================================

independence_wave_iw059_mesopotamia_remove_overlay_ideas = {
	remove_ideas = {
		independence_wave_iw059_mesopotamia_contested_river_cabinet
		independence_wave_iw059_mesopotamia_integrated_river_authority
		independence_wave_iw059_mesopotamia_minority_guarantees
		independence_wave_iw059_mesopotamia_constitutional_compact
	}
}

independence_wave_iw059_mesopotamia_refresh_idea_lifecycle = {
	if = {
		limit = { is_independence_wave_iw059_mesopotamia_overlay_active = yes }
		independence_wave_iw059_mesopotamia_remove_overlay_ideas = yes
		if = {
			limit = { has_country_flag = independence_wave_iw059_mesopotamia_minority_guarantees }
			add_ideas = independence_wave_iw059_mesopotamia_minority_guarantees
		}
		else_if = {
			limit = { has_country_flag = independence_wave_iw059_mesopotamia_constitutional_compact }
			add_ideas = independence_wave_iw059_mesopotamia_constitutional_compact
		}
		else_if = {
			limit = {
				check_variable = {
					var = independence_wave_iw059_mesopotamia_river_logistics
					value = constant:independence_wave_iw059_mesopotamia_value.settlement_threshold
					compare = greater_than_or_equals
				}
				check_variable = {
					var = independence_wave_iw059_mesopotamia_civic_legitimacy
					value = constant:independence_wave_iw059_mesopotamia_value.settlement_threshold
					compare = greater_than_or_equals
				}
			}
			add_ideas = independence_wave_iw059_mesopotamia_integrated_river_authority
		}
		else = { add_ideas = independence_wave_iw059_mesopotamia_contested_river_cabinet }
	}
}

			var = independence_wave_iw059_mesopotamia_river_logistics
			min = constant:independence_wave_iw059_mesopotamia_value.minimum
			max = constant:independence_wave_iw059_mesopotamia_value.maximum
		}
		clamp_variable = {
			var = independence_wave_iw059_mesopotamia_civic_legitimacy
			min = constant:independence_wave_iw059_mesopotamia_value.minimum
			max = constant:independence_wave_iw059_mesopotamia_value.maximum

}

independence_wave_iw059_mesopotamia_update_guard_hold = {
	if = {
		limit = {
			is_independence_wave_iw059_mesopotamia_overlay_active = yes
			has_country_flag = independence_wave_iw059_mesopotamia_guard_running
			has_independence_wave_iw059_mesopotamia_guard_objective = yes
		}
		add_to_variable = { independence_wave_iw059_mesopotamia_guard_hold_days = constant:independence_wave_iw059_mesopotamia_value.guard_daily_gain }
		clamp_variable = {
			var = independence_wave_iw059_mesopotamia_guard_hold_days
			min = constant:independence_wave_iw059_mesopotamia_value.minimum
			max = constant:independence_wave_iw059_mesopotamia_value.guard_required
		}
	}
	else_if = {
		limit = {
			is_independence_wave_iw059_mesopotamia_overlay_active = yes
			has_country_flag = independence_wave_iw059_mesopotamia_guard_running
		}
		set_variable = { independence_wave_iw059_mesopotamia_guard_hold_days = constant:independence_wave_iw059_mesopotamia_value.minimum }
	}
}

independence_wave_iw059_mesopotamia_initialize_overlay = {
	if = {
		limit = {
			is_independence_wave_iw059_mesopotamia_route_active = yes
			NOT = { has_country_flag = independence_wave_iw059_mesopotamia_overlay_ever_activated }
		}
		set_country_flag = independence_wave_iw059_mesopotamia_overlay_ever_activated
		set_country_flag = independence_wave_iw059_mesopotamia_package_initialized
		set_country_flag = independence_wave_iw059_mesopotamia_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		set_variable = { independence_wave_iw059_mesopotamia_river_logistics = constant:independence_wave_iw059_mesopotamia_value.river_logistics_start }
		set_variable = { independence_wave_iw059_mesopotamia_civic_legitimacy = constant:independence_wave_iw059_mesopotamia_value.civic_legitimacy_start }
		set_variable = { independence_wave_iw059_mesopotamia_provincial_autonomy = constant:independence_wave_iw059_mesopotamia_value.provincial_autonomy_start }
		set_variable = { independence_wave_iw059_mesopotamia_guard_hold_days = constant:independence_wave_iw059_mesopotamia_value.minimum }
		set_variable = { independence_wave_anchor_state = 291 }
		set_variable = { independence_wave_package_archetype = constant:independence_wave_package_archetype.river_or_corridor }
		set_variable = { independence_wave_force_profile = constant:independence_wave_force_profile.river_jungle }
		independence_wave_iw059_mesopotamia_refresh_idea_lifecycle = yes
	}
}

independence_wave_iw059_mesopotamia_suspend_overlay = {
	if = {
		limit = { has_country_flag = independence_wave_iw059_mesopotamia_overlay_active }
		if = {
			limit = {
				has_country_flag = independence_wave_iw059_mesopotamia_guard_running
				NOT = { has_country_flag = independence_wave_iw059_mesopotamia_guard_completed }
			}
			set_country_flag = independence_wave_iw059_mesopotamia_guard_interrupted
		}
		independence_wave_iw059_mesopotamia_remove_overlay_ideas = yes
		clr_country_flag = independence_wave_iw059_mesopotamia_overlay_active
		clr_country_flag = independence_wave_route_overlay_active
		set_country_flag = independence_wave_iw059_mesopotamia_overlay_suspended
	}
}

independence_wave_iw059_mesopotamia_pause_guard_mission = {
	if = {
		limit = {
			has_country_flag = independence_wave_iw059_mesopotamia_overlay_suspended
			has_country_flag = independence_wave_iw059_mesopotamia_guard_interrupted
			has_country_flag = independence_wave_iw059_mesopotamia_guard_running
			has_active_mission = independence_wave_iw059_hold_baghdad_river_guard
		}
		set_temp_variable = { independence_wave_iw059_mesopotamia_pause_days = constant:independence_wave_iw059_mesopotamia_duration.guard_pause_daily_extension }
		add_days_mission_timeout = { mission = independence_wave_iw059_hold_baghdad_river_guard days = independence_wave_iw059_mesopotamia_pause_days }
	}
}

independence_wave_iw059_mesopotamia_resume_overlay = {
	if = {
		limit = {
			is_independence_wave_iw059_mesopotamia_route_active = yes
			has_country_flag = independence_wave_iw059_mesopotamia_overlay_ever_activated
			NOT = { has_country_flag = independence_wave_iw059_mesopotamia_overlay_active }
		}
		set_country_flag = independence_wave_iw059_mesopotamia_overlay_active
		set_country_flag = independence_wave_route_overlay_active
		clr_country_flag = independence_wave_iw059_mesopotamia_overlay_suspended
		independence_wave_iw059_mesopotamia_refresh_idea_lifecycle = yes
	}
}

# Called only from the exact original-tag daily hooks below. No global daily
# iteration is introduced; every state change is identity-gated.
independence_wave_iw059_mesopotamia_refresh_overlay = {
	if = {
		limit = { is_independence_wave_iw059_mesopotamia_route_active = yes }
		if = {
			limit = { NOT = { has_country_flag = independence_wave_iw059_mesopotamia_overlay_ever_activated } }
			independence_wave_iw059_mesopotamia_initialize_overlay = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = independence_wave_iw059_mesopotamia_overlay_active } }
			independence_wave_iw059_mesopotamia_resume_overlay = yes
		}
		else = {
			if = {
				limit = { has_country_flag = independence_wave_iw059_mesopotamia_guard_interrupted }
				if = {
					limit = { has_country_flag = independence_wave_iw059_mesopotamia_guard_running has_active_mission = independence_wave_iw059_hold_baghdad_river_guard }
					clr_country_flag = independence_wave_iw059_mesopotamia_guard_interrupted
				}
				else = { independence_wave_iw059_mesopotamia_fail_guard_mission = yes }
			}
			else_if = {
				limit = {
					has_country_flag = independence_wave_iw059_mesopotamia_guard_running
					NOT = { has_active_mission = independence_wave_iw059_hold_baghdad_river_guard }
				}
				independence_wave_iw059_mesopotamia_fail_guard_mission = yes
			}
			independence_wave_iw059_mesopotamia_update_guard_hold = yes
			if = {
				limit = {
					NOT = {
						OR = {
							has_idea = independence_wave_iw059_mesopotamia_contested_river_cabinet
							has_idea = independence_wave_iw059_mesopotamia_integrated_river_authority
							has_idea = independence_wave_iw059_mesopotamia_minority_guarantees
							has_idea = independence_wave_iw059_mesopotamia_constitutional_compact
						}
					}
				}
				independence_wave_iw059_mesopotamia_refresh_idea_lifecycle = yes
			}
		}
	}
	else = {
		if = {
			limit = { has_country_flag = independence_wave_iw059_mesopotamia_overlay_active }
			independence_wave_iw059_mesopotamia_suspend_overlay = yes
		}
		independence_wave_iw059_mesopotamia_pause_guard_mission = yes
	}
}
