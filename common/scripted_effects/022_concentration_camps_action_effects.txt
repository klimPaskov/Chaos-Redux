# ============================================================================
# CHAOS REDUX - EVENT 022 CONCENTRATION CAMPS ACTION EFFECTS
# ============================================================================
# Overview:
# - The opening policy options, the Event 22 decisions, the incident answers and
#   the observer answers. Every effect uses the shared public API only: the
#   Ledger mandate and role variables with a recorded history entry, the
#   custody release and death helpers, the camp_policy_* impulses (which the
#   shared monthly recalculation decays), and the shared hidden cover-up source.
# - Payments are support equipment, trains, fuel, manpower and political power,
#   quoted in the cached concentration_camps_cost_* country variables.
#
# Scope notes: COUNTRY unless noted. ROOT is the acting country in every
# caller. Decisions that target a state read it as FROM.
# ============================================================================

# ----------------------------------------------------------------------------
# Payments
# ----------------------------------------------------------------------------

# COUNTRY. Inputs (temp, all optional, zero when unused):
# concentration_camps_pay_support, concentration_camps_pay_trains,
# concentration_camps_pay_fuel and concentration_camps_pay_manpower_k.
# Clears the inputs after paying.
concentration_camps_pay = {
	if = {
		limit = { check_variable = { concentration_camps_pay_support > constant:concentration_camps_value.zero } }
		set_temp_variable = { equipment_stockpile_removal_amount = concentration_camps_pay_support }
		remove_support_equipment_from_stockpile = yes
	}
	if = {
		limit = { check_variable = { concentration_camps_pay_trains > constant:concentration_camps_value.zero } }
		set_temp_variable = { concentration_camps_payment = concentration_camps_pay_trains }
		multiply_temp_variable = { concentration_camps_payment = constant:concentration_camps_value.negative_one }
		add_equipment_to_stockpile = { type = train_equipment amount = concentration_camps_payment }
	}
	if = {
		limit = { check_variable = { concentration_camps_pay_fuel > constant:concentration_camps_value.zero } }
		set_temp_variable = { concentration_camps_payment = concentration_camps_pay_fuel }
		multiply_temp_variable = { concentration_camps_payment = constant:concentration_camps_value.negative_one }
		add_fuel = concentration_camps_payment
	}
	if = {
		limit = { check_variable = { concentration_camps_pay_manpower_k > constant:concentration_camps_value.zero } }
		set_temp_variable = { concentration_camps_payment = concentration_camps_pay_manpower_k }
		multiply_temp_variable = { concentration_camps_payment = constant:camp_admin_value.people_per_k }
		multiply_temp_variable = { concentration_camps_payment = constant:concentration_camps_value.negative_one }
		add_manpower = concentration_camps_payment
	}
	set_temp_variable = { concentration_camps_pay_support = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_pay_trains = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_pay_fuel = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_pay_manpower_k = constant:concentration_camps_value.zero }
}

# Shared national controls
# ----------------------------------------------------------------------------

# COUNTRY. Input: temp concentration_camps_mandate_target, one of
# constant:camp_admin_mandate.*. Sets the Ledger mandate without a payment,
# because the opening choice and the incident answers are the policy decision,
# and records it in the Ledger history like the Ledger command does.
concentration_camps_apply_mandate = {
	camp_admin_initialize_country = yes
	if = {
		limit = { NOT = { check_variable = { camp_admin_mandate = concentration_camps_mandate_target } } }
		set_variable = { camp_admin_mandate = concentration_camps_mandate_target }
		set_temp_variable = { camp_admin_record_kind = constant:camp_admin_history_kind.mandate }
		camp_admin_record_history = yes
		camp_admin_refresh_country_economy = yes
	}
	camp_admin_refresh_interface = yes
}

# COUNTRY. One impulse of shared pressure. Inputs (temp, optional, zero when
# unused): concentration_camps_push_resistance, _visibility, _evidence,
# _overstretch, _reform. The shared recalculation folds them into the public
# values and decays them monthly.
concentration_camps_push_pressure = {
	camp_rework_initialize_country_variables = yes
	add_to_variable = { camp_policy_resistance_pressure = concentration_camps_push_resistance }
	add_to_variable = { camp_policy_visibility = concentration_camps_push_visibility }
	add_to_variable = { camp_policy_evidence = concentration_camps_push_evidence }
	add_to_variable = { camp_policy_overstretch = concentration_camps_push_overstretch }
	add_to_variable = { camp_policy_reform = concentration_camps_push_reform }
	set_temp_variable = { concentration_camps_push_resistance = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_push_visibility = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_push_evidence = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_push_overstretch = constant:concentration_camps_value.zero }
	set_temp_variable = { concentration_camps_push_reform = constant:concentration_camps_value.zero }
	camp_rework_recalculate_country_values = yes
}

# COUNTRY. The case review: inspectors go through the files, the mandate returns
# to custody, and part of every living cohort is released in place.
concentration_camps_review_cases = {
	concentration_camps_pay_review = yes
	set_variable = { concentration_camps_policy = constant:concentration_camps_policy.review }
	set_temp_variable = { concentration_camps_mandate_target = constant:camp_admin_mandate.custody }
	concentration_camps_apply_mandate = yes

			set_temp_variable = { camp_admin_release_requested_k = camp_admin_detainees_k }
			multiply_temp_variable = { camp_admin_release_requested_k = constant:concentration_camps_cohort.review_release_share }
			camp_admin_release_detainees = yes
			PREV = { add_to_variable = { concentration_camps_released_k = camp_admin_release_actual_k } }
		}
	}
	set_temp_variable = { concentration_camps_push_resistance = constant:concentration_camps_incident.relief_resistance_relief }
	set_temp_variable = { concentration_camps_push_visibility = constant:camp_rework_country_value.visibility_minor }

# ----------------------------------------------------------------------------
# Deaths through the shared API
# ----------------------------------------------------------------------------

# STATE. Inputs (temp): concentration_camps_death_share (of the living cohort)
# and concentration_camps_death_cause (constant:camp_admin_cause.*). Requests the
# loss from the shared custody owner, which clamps it to the people actually
# held and records the Deaths entry. Output: temp camp_admin_death_actual_k.
concentration_camps_request_deaths = {
	if = {
		limit = {
			camp_admin_site_is_operational = yes
			check_variable = { camp_admin_detainees_k > constant:concentration_camps_value.zero }
		}
		set_temp_variable = { camp_admin_death_requested_k = camp_admin_detainees_k }
		multiply_temp_variable = { camp_admin_death_requested_k = concentration_camps_death_share }
		set_temp_variable = { camp_admin_death_cause = concentration_camps_death_cause }
		set_temp_variable = { camp_admin_death_contract = constant:concentration_camps_value.one }
		camp_admin_record_deaths = yes
		PREV = { add_to_variable = { concentration_camps_incident_deaths_k = camp_admin_death_actual_k } }
	}
}

			limit = {
				has_variable = famine_food_stage

			}
			set_temp_variable = { concentration_camps_death_share = constant:concentration_camps_incident.epidemic_neglect_share }

	set_temp_variable = { concentration_camps_push_evidence = constant:concentration_camps_incident.conceal_evidence_gain }
	concentration_camps_push_pressure = yes
}

# COUNTRY. The records are rewritten and a tour is staged. It works for a
# while when the files hold together, and fails visibly when they do not.

	concentration_camps_register_coverup = yes
	set_temp_variable_to_random = { var = concentration_camps_stage_roll min = 0 max = 1 }
	if = {
		limit = { check_variable = { concentration_camps_stage_roll < constant:concentration_camps_incident.staged_success_chance } }
		set_temp_variable = { concentration_camps_push_visibility = constant:concentration_camps_incident.conceal_visibility_relief }

		concentration_camps_push_pressure = yes
	}
	else = {
		set_temp_variable = { concentration_camps_push_visibility = constant:camp_rework_country_value.visibility_major }
		set_temp_variable = { concentration_camps_push_evidence = constant:camp_rework_country_value.evidence_major }
