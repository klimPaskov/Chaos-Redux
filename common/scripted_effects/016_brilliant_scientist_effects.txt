#   foreign targets, temporal debt, origin evidence, formation inputs, and
#   terminal preparation.

brilliant_scientist_clear_prefire_context = {
	set_temp_variable = { brilliant_scientist_prefire_ready = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_prefire_evolution_stage = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_prefire_evolution_tier = constant:brilliant_scientist_value.zero }
}

# Any scope, called by record_events_log_history_entry immediately after its
# Event 016 row is appended. The sequence, not array position zero, is the
# stable identity because the history view is newest-first and can be rebuilt.
brilliant_scientist_bind_opening_history_sequence = {
	if = {
		limit = {
			has_variable = global.events_log_history_sequence
			check_variable = { global.last_fired_event_id = constant:brilliant_scientist_event.id }
		}
		set_variable = { global.brilliant_scientist_pending_history_sequence = global.events_log_history_sequence }
		set_global_flag = brilliant_scientist_history_actor_rebind_pending
	}
}

# COUNTRY scope on the final accepting host. Locates the exact Event 016 row,
# replaces actor/has_actor, then clears pending state only after a successful
# rebind. This corrects both direct acceptance and the player send-away route.
brilliant_scientist_finalize_opening_history_actor = {
	set_temp_variable = { brilliant_scientist_history_actor_rebound = constant:brilliant_scientist_value.zero }
	if = {
		limit = {
			has_country_flag = brilliant_scientist_current_host
			has_global_flag = brilliant_scientist_history_actor_rebind_pending
			has_variable = global.brilliant_scientist_pending_history_sequence
		}
		sanitize_events_log_history_actor_entries = yes
		set_temp_variable = { brilliant_scientist_history_actor_index = constant:brilliant_scientist_value.zero }
		set_temp_variable = { brilliant_scientist_history_accepting_actor = THIS }
		while_loop_effect = {
			limit = {
				check_variable = {
					var = brilliant_scientist_history_actor_index
					value = global.events_log_history_sequence_entries^num
					compare = less_than
				}
				check_variable = { brilliant_scientist_history_actor_rebound < constant:brilliant_scientist_value.one }
			}
			if = {
				limit = {
					check_variable = {
						var = global.events_log_history_sequence_entries^brilliant_scientist_history_actor_index
						value = global.brilliant_scientist_pending_history_sequence
						compare = equals
					}
					check_variable = {
						var = global.events_log_history_event_id_entries^brilliant_scientist_history_actor_index
						value = constant:brilliant_scientist_event.id
						compare = equals
					}
				}
				set_variable = { global.events_log_history_actor_entries^brilliant_scientist_history_actor_index = brilliant_scientist_history_accepting_actor }
				set_variable = { global.events_log_history_has_actor_entries^brilliant_scientist_history_actor_index = constant:brilliant_scientist_value.one }
				set_temp_variable = { brilliant_scientist_history_actor_rebound = constant:brilliant_scientist_value.one }
			}
			add_to_temp_variable = { brilliant_scientist_history_actor_index = constant:brilliant_scientist_value.one }
		}
	}
	if = {
		limit = { check_variable = { brilliant_scientist_history_actor_rebound > constant:brilliant_scientist_value.zero } }
		clear_variable = global.brilliant_scientist_pending_history_sequence
		clr_global_flag = brilliant_scientist_history_actor_rebind_pending
		every_country = {
			limit = {
				is_ai = no
				has_country_flag = events_log_window_open
			}
			if = { limit = { has_country_flag = events_log_history_tab } rebuild_events_log_history_view = yes }
			if = { limit = { has_country_flag = events_log_event_details_open } events_log_rebuild_open_event_details_view = yes }
		}
	}
}

# COUNTRY scope. Produces a bounded whole-number fire-once selection weight.
# Research depth, industry/urban capacity, wartime pressure, special-project
# infrastructure, government structure, chaos pressure, and prior scientific
# chaos exposure all remain independently tunable.
brilliant_scientist_prepare_host_selection_weight = {
	set_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.base_weight }
	if = {
		limit = { amount_research_slots > constant:brilliant_scientist_value.two }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.research_slots_three_bonus }
	}
	if = {
		limit = { amount_research_slots > constant:brilliant_scientist_value.three }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.research_slots_four_bonus }
	}
	if = {
		limit = { amount_research_slots > constant:brilliant_scientist_value.four }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.research_slots_five_bonus }
	}
	if = {
		limit = { has_tech = electronic_mechanical_engineering }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.electronics_technology_bonus }
	}
	if = {
		limit = { has_tech = computing_machine }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.computing_technology_bonus }
	}
	if = {
		limit = {
			NOT = {
				OR = {
					has_tech = electronic_mechanical_engineering
					has_tech = computing_machine
				}
			}
			OR = {
				any_enemy_country = {
					OR = {
						has_tech = electronic_mechanical_engineering
						has_tech = computing_machine
					}
				}
				any_neighbor_country = {
					OR = {
						has_tech = electronic_mechanical_engineering
						has_tech = computing_machine
					}
				}
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.technological_gap_bonus }
	}
	if = {
		limit = { num_of_factories < constant:brilliant_scientist_host_selection.factory_gate_low }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.factory_low_bonus }
	}
	if = {
		limit = { num_of_factories > constant:brilliant_scientist_host_selection.factory_gate_high }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.factory_high_bonus }
	}
	if = {
		limit = {
			any_owned_state = {
				is_owned_and_controlled_by = PREV
				OR = {
					has_state_category = city
					has_state_category = large_city
					has_state_category = metropolis
					has_state_category = megalopolis
				}
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.urban_state_bonus }
	}
	if = {
		limit = { is_major = yes }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.major_bonus }
	}
	if = {
		limit = { has_war = yes }
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.war_pressure_bonus }
	}
	if = {
		limit = {
			OR = {
				has_facility_specialization = specialization_nuclear
				has_facility_specialization = specialization_air
				has_facility_specialization = specialization_land
				has_facility_specialization = specialization_naval
				has_facility_specialization = specialization_biowarfare
				has_facility_specialization = specialization_cw
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.special_project_facility_bonus }
	}
	if = {
		limit = {
			OR = {
				has_government = fascism
				has_government = communism
				has_government = neutrality
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.centralized_government_bonus }
	}
	else = {
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.democratic_government_bonus }
	}
	if = {
		limit = {
			has_variable = global.chaos_meter_value
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_3.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.chaos_high_bonus }
	}
	if = {
		limit = {
			has_variable = global.chaos_meter_value
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_4.min
				compare = greater_than_or_equals
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.chaos_extreme_bonus }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = bioweapon_available
				has_country_flag = used_bioweapon
				has_country_flag = germany_mengele_facility_built_from_demand
				has_country_flag = mengele_clone_army_facility_built
				has_country_flag = holy_realm_thermonuclear_doctrine
				has_country_flag = nlc_doctrine_scientific_refuge_organized
				has_country_flag = resources_found_first_battle_analyzed
			}
		}
		add_to_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_host_selection.science_chaos_history_bonus }
	}
	round_temp_variable = brilliant_scientist_host_selection_weight
	clamp_temp_variable = {
		var = brilliant_scientist_host_selection_weight
		min = constant:brilliant_scientist_host_selection.minimum_weight
		max = constant:brilliant_scientist_host_selection.maximum_weight
	}
}

# COUNTRY scope. Mirrors the verified Fury weighted-scope pool contract.
brilliant_scientist_add_current_country_to_host_selection_pool = {
	brilliant_scientist_prepare_host_selection_weight = yes
	set_temp_variable = { brilliant_scientist_host_pool_weight_remaining = brilliant_scientist_host_selection_weight }
	while_loop_effect = {
		limit = { check_variable = { brilliant_scientist_host_pool_weight_remaining > constant:brilliant_scientist_value.zero } }
		add_to_temp_array = { brilliant_scientist_host_selection_pool = THIS }
		add_to_temp_variable = { brilliant_scientist_host_selection_pool_count = constant:brilliant_scientist_value.one }
		subtract_from_temp_variable = { brilliant_scientist_host_pool_weight_remaining = constant:brilliant_scientist_value.one }
	}
	set_temp_variable = { brilliant_scientist_host_pool_weight_remaining = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_value.zero }
}

# Any scope. This is an explicit fire-once scan, never a periodic on_action.
brilliant_scientist_select_weighted_prefire_host = {
	clear_temp_array = brilliant_scientist_host_selection_pool
	set_temp_variable = { brilliant_scientist_host_selection_pool_count = constant:brilliant_scientist_value.zero }
	every_country = {
		limit = { brilliant_scientist_is_eligible_host = yes }
		brilliant_scientist_add_current_country_to_host_selection_pool = yes
	}
	if = {
		limit = { check_variable = { brilliant_scientist_host_selection_pool_count > constant:brilliant_scientist_value.zero } }
		random_scope_in_array = {
			array = brilliant_scientist_host_selection_pool
			save_event_target_as = brilliant_scientist_prefire_host
		}
	}
	clear_temp_array = brilliant_scientist_host_selection_pool
	set_temp_variable = { brilliant_scientist_host_selection_pool_count = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_host_pool_weight_remaining = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_host_selection_weight = constant:brilliant_scientist_value.zero }
}

# Any country scope. Output is one regular event target, which survives into
# events fired from this effect chain and then clears automatically.
brilliant_scientist_prepare_random_event_fire = {
	brilliant_scientist_clear_prefire_context = yes
	if = {
		limit = { brilliant_scientist_automatic_event_is_available = yes }
		brilliant_scientist_select_weighted_prefire_host = yes
		if = {
			limit = { has_event_target = brilliant_scientist_prefire_host }
			set_temp_variable = { brilliant_scientist_prefire_ready = constant:brilliant_scientist_value.one }
			brilliant_scientist_prepare_prefire_evolution_context = yes
		}
	}
}

					has_country_flag = brilliant_scientist_directorate_publication_completed

	}

# COUNTRY scope. Reconstructs available Project Capacity from the durable gross
# capacity receipt and the exact current ledger. Completed stages retain their
# full cumulative burden; suspended stages retain half. A live stage mission
# also occupies its incremental reservation until completion or cancellation.
brilliant_scientist_refresh_project_capacity_from_gross = {
	set_temp_variable = { brilliant_scientist_project_capacity_used = constant:brilliant_scientist_value.zero }
	set_temp_variable = { brilliant_scientist_project_capacity_index = constant:brilliant_scientist_value.zero }
	while = {
		limit = { check_variable = { brilliant_scientist_project_capacity_index < constant:brilliant_scientist_event.project_family_count } }
		set_temp_variable = { brilliant_scientist_project_family = brilliant_scientist_project_capacity_index }
		add_to_temp_variable = { brilliant_scientist_project_family = constant:brilliant_scientist_value.one }
		set_temp_variable = { brilliant_scientist_project_capacity_stage = brilliant_scientist_project_stage_entries^brilliant_scientist_project_capacity_index }
		set_temp_variable = { brilliant_scientist_project_capacity_burden = constant:brilliant_scientist_value.zero }
		if = {
			limit = { check_variable = { brilliant_scientist_project_capacity_stage = constant:brilliant_scientist_project_stage.theory } }
			set_temp_variable = { brilliant_scientist_project_capacity_burden = constant:brilliant_scientist_project_capacity.theory }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_project_capacity_stage = constant:brilliant_scientist_project_stage.prototype } }
			set_temp_variable = { brilliant_scientist_project_capacity_burden = constant:brilliant_scientist_project_capacity.prototype }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_project_capacity_stage = constant:brilliant_scientist_project_stage.deployment } }
			set_temp_variable = { brilliant_scientist_project_capacity_burden = constant:brilliant_scientist_project_capacity.deployment }
		}
		else_if = {
			limit = { check_variable = { brilliant_scientist_project_capacity_stage = constant:brilliant_scientist_project_stage.weaponization } }
			set_temp_variable = { brilliant_scientist_project_capacity_burden = constant:brilliant_scientist_project_capacity.weaponization }
		}
		if = {
			limit = { brilliant_scientist_project_is_suspended = yes }
			multiply_temp_variable = { brilliant_scientist_project_capacity_burden = constant:brilliant_scientist_project_capacity.suspended_factor }
		}
		add_to_temp_variable = { brilliant_scientist_project_capacity_used = brilliant_scientist_project_capacity_burden }
		add_to_temp_variable = { brilliant_scientist_project_capacity_index = constant:brilliant_scientist_value.one }
	}
	if = {
		limit = {
			has_country_flag = brilliant_scientist_project_stage_in_progress
			has_variable = brilliant_scientist_active_project_capacity_delta
		}
		add_to_temp_variable = { brilliant_scientist_project_capacity_used = brilliant_scientist_active_project_capacity_delta }
	}
	set_variable = { brilliant_scientist_project_capacity = brilliant_scientist_project_capacity_gross }
	subtract_from_variable = { brilliant_scientist_project_capacity = brilliant_scientist_project_capacity_used }
	clamp_variable = { var = brilliant_scientist_project_capacity min = constant:brilliant_scientist_measure.minimum max = constant:brilliant_scientist_measure.maximum }
}
