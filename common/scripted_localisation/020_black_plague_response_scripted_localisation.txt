# Decision localisation cannot interpolate script constants directly. Each
# quote sets a temporary value from the same constant used by the payment or
# capacity effect before the corresponding localisation key is displayed.

	name = GetBlackPlagueReserveSupportQuote
	text = {
		trigger = { set_temp_variable = { black_plague_reserve_support_quote = constant:black_plague_medical_reserve_cost.support_equipment } always = yes }
		localization_key = black_plague_reserve_support_quote_value
