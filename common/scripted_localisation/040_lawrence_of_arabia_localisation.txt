# The British header: what is committed and what blocks further support.

	name = GetLawrenceBritishLine
	text = { trigger = { lawrence_status_is_detained = yes } localization_key = lawrence_british_detained_line }

		trigger = {
			has_global_flag = lawrence_campaign_active
			has_variable = global.lawrence_target
			var:global.lawrence_target = { has_country_flag = lawrence_delivery_pending }
		}
		localization_key = lawrence_british_shipment_line
	}
	text = {
		trigger = {
			has_global_flag = lawrence_campaign_active
			has_variable = global.lawrence_target

		}
		localization_key = lawrence_british_route_closed_line
	}
	text = {
		trigger = { NOT = { lawrence_sponsor_has_regional_base = yes } }
		localization_key = lawrence_british_no_base_line
	}
	text = { localization_key = lawrence_british_open_line }
