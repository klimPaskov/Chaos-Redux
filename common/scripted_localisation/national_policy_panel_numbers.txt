# Presentation-only population formatting. Inputs stay in thousands to avoid overflowing
# the engine's fixed-point variables when a count reaches millions or billions of people.
# Callers set the unscoped temporary national_policy_panel_number_k, then select
# national_policy_panel_number_from_thousands. No persistent state is changed here.
defined_text = {
	name = GetNationalPolicyPanelNumberFromThousands
	text = {
		trigger = {
			check_variable = { var = national_policy_panel_number_k value = constant:national_policy_panel_number.billion_in_thousands compare = greater_than_or_equals }
			set_temp_variable = { national_policy_panel_number_display = national_policy_panel_number_k }
			divide_temp_variable = { national_policy_panel_number_display = constant:national_policy_panel_number.billion_in_thousands }
		}
		localization_key = national_policy_panel_number_billion
	}
	text = {
		trigger = {
			check_variable = { var = national_policy_panel_number_k value = constant:national_policy_panel_number.thousand compare = greater_than_or_equals }
			set_temp_variable = { national_policy_panel_number_display = national_policy_panel_number_k }
			divide_temp_variable = { national_policy_panel_number_display = constant:national_policy_panel_number.thousand }
		}
		localization_key = national_policy_panel_number_million
	}
	text = {
		trigger = {
			check_variable = { var = national_policy_panel_number_k value = constant:national_policy_panel_number.one compare = greater_than_or_equals }
			set_temp_variable = { national_policy_panel_number_display = national_policy_panel_number_k }
		}
		localization_key = national_policy_panel_number_thousand
	}
	text = {
		trigger = {
			set_temp_variable = { national_policy_panel_number_display = national_policy_panel_number_k }
			multiply_temp_variable = { national_policy_panel_number_display = constant:national_policy_panel_number.thousand }
		}
		localization_key = national_policy_panel_number_integer
	}
}

# Round scaled counts to one decimal, omitting a redundant decimal when it is zero.
defined_text = {
	name = GetNationalPolicyPanelNumberPrecision
	text = {
		trigger = {
			set_temp_variable = { national_policy_panel_number_decimal = national_policy_panel_number_display }
			multiply_temp_variable = { national_policy_panel_number_decimal = constant:national_policy_panel_number.decimal_scale }
			round_temp_variable = national_policy_panel_number_decimal
			divide_temp_variable = { national_policy_panel_number_decimal = constant:national_policy_panel_number.decimal_scale }
			set_temp_variable = { national_policy_panel_number_whole = national_policy_panel_number_decimal }
			round_temp_variable = national_policy_panel_number_whole
			check_variable = { national_policy_panel_number_decimal = national_policy_panel_number_whole }
			set_temp_variable = { national_policy_panel_number_display = national_policy_panel_number_decimal }
		}
		localization_key = national_policy_panel_number_integer
	}
	text = {
		trigger = { set_temp_variable = { national_policy_panel_number_display = national_policy_panel_number_decimal } }
		localization_key = national_policy_panel_number_decimal
	}
}

# Live source adapters. Selected-state values are read directly, never from copied GUI data.
defined_text = {
	name = GetCampAdminNumberDetainees
	text = {
		trigger = {
			set_temp_variable = { national_policy_panel_number_k = constant:camp_admin_value.zero }
			clear_temp_array = national_policy_panel_number_seen
			all_of_scopes = {
				array = camp_active_site_states
				if = {
					limit = {
						NOT = { has_state_flag = camp_admin_closed }
						NOT = { is_in_array = { array = national_policy_panel_number_seen value = THIS } }
					}
					add_to_temp_array = { array = national_policy_panel_number_seen value = THIS }
					add_to_temp_variable = { national_policy_panel_number_k = camp_admin_detainees_k }
				}
			}
			all_of_scopes = {
				array = camp_admin_civilian_states
				if = {
					limit = { NOT = { is_in_array = { array = national_policy_panel_number_seen value = THIS } } }
					add_to_temp_array = { array = national_policy_panel_number_seen value = THIS }
					add_to_temp_variable = { national_policy_panel_number_k = camp_admin_detainees_k }
				}
			}
		}
		localization_key = national_policy_panel_number_from_thousands
	}
	text = { localization_key = national_policy_panel_number_zero }
}
