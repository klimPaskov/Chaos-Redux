# ============================================================================
# CHAOS REDUX - EVENT 016 D'RHONDAN COUNTRY SCRIPTED LOCALISATION
# ============================================================================
# Adds the sovereignty aftermath to the existing Event 016 details entry only
# after the Empire of D'Rhonda has formed. The fallback is deliberately empty.
# ============================================================================

defined_text = {
	name = GetDhrondanEventDetailClause
	text = {
		trigger = { has_global_flag = dhrondan_sovereignty_formed }
		localization_key = dhrondan_event_detail_clause
	}
	text = { localization_key = dhrondan_event_detail_clause_blank }
}

# Landing reservation length for the reader: Human Signal Teams shorten it for
# D'Rhonda (dhrondan_adjust_landing_reservation).
defined_text = {
	name = GetAlienInfantryLandingReservationDays
	text = {
		trigger = {
			tag = DHR
			has_country_flag = dhrondan_human_signal_teams_trained
		}
		localization_key = alien_infantry_landing_reservation_days_signal
	}
	text = { localization_key = alien_infantry_landing_reservation_days_standard }
}

# COUNTRY scope on the recipient. Preserve the compact's original label while

	name = GetDhrondanDiplomaticAcceptLabel
	text = {

		localization_key = chaosx.nr16.49.partner.a
	}
	text = { localization_key = chaosx.nr16.49.a }
}
