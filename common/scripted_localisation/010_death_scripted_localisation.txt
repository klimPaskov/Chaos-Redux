defined_text = {
	name = GetDeathAtlasLineStatus

	text = {
		trigger = {
			check_variable = {
				var = global.death_front_readiness
				value = constant:death_decision_tuning.atlas_front_held_threshold
				compare = greater_than_or_equals
			}
		}
		localization_key = death_black_atlas_line_held
	}
	text = {
		trigger = {
			check_variable = {
				var = global.death_front_readiness
				value = constant:death_decision_tuning.atlas_front_thin_threshold
				compare = greater_than_or_equals
			}
		}
		localization_key = death_black_atlas_line_thin
	}
	text = {
		trigger = { always = yes }
		localization_key = death_black_atlas_line_broken
	}
}
