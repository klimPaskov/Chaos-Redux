defined_text = {
	name = GetBlackFridayEventDetailStatusLine
	text = {
		trigger = {
			check_variable = { var = event_id value = constant:black_friday_event.id compare = equals }
			black_friday_is_active = yes
		}
		localization_key = black_friday.event_detail.status.active
	}
	text = {
		trigger = {
			check_variable = { var = event_id value = constant:black_friday_event.id compare = equals }
			black_friday_is_reserved = yes
		}
		localization_key = black_friday.event_detail.status.reserved
	}
	text = {
		trigger = {
			check_variable = { var = event_id value = constant:black_friday_event.id compare = equals }
			black_friday_is_fired = yes
		}
		localization_key = black_friday.event_detail.status.expired
	}
	text = {
		trigger = {
			check_variable = { var = event_id value = constant:black_friday_event.id compare = equals }
			black_friday_is_disabled = yes
		}
		localization_key = black_friday.event_detail.status.disabled
	}
	text = {
		trigger = {
			check_variable = { var = event_id value = constant:black_friday_event.id compare = equals }
			check_variable = {
				var = global.chaos_meter_value
				value = constant:black_friday_event.required_chaos
				compare = less_than
			}
		}
		localization_key = black_friday.event_detail.status.unavailable
	}
	text = {
		trigger = {
			check_variable = { var = event_id value = constant:black_friday_event.id compare = equals }
			black_friday_can_reserve = yes
		}
		localization_key = black_friday.event_detail.status.eligible
	}
	text = {
		trigger = { always = yes }
		localization_key = black_friday.event_detail.status.none
	}
}
