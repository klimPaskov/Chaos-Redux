# ============================================================================
# EVENT 016 - PORTAL TRANSIT BATCH COST PRESENTATION
# ============================================================================
# One fresh payer quote owns the three direct-cost outputs across its return.
# Nested component getters use the same affordability predicates as admission.
# Integer amounts stay compact; fractional amounts retain five-digit precision.
# This presentation changes temporary scratch only and never records payment.
# ============================================================================

defined_text = {
	name = GetPortalTransitBatchCost
	text = {
		trigger = {
			set_temp_variable = { krg_portal_batch_quote_pp = constant:brilliant_scientist_krg_value.zero }
			set_temp_variable = { krg_portal_batch_quote_support = constant:brilliant_scientist_krg_value.zero }
			set_temp_variable = { krg_portal_batch_quote_fuel = constant:brilliant_scientist_krg_value.zero }
			set_temp_variable = { krg_portal_batch_quote_valid = constant:brilliant_scientist_krg_value.zero }
			brilliant_scientist_krg_quote_portal_transit_batch = yes
			check_variable = { var = krg_portal_batch_quote_valid value = constant:brilliant_scientist_krg_value.one compare = equals }
		}
		localization_key = portal_transit_batch_cost_values
	}
	text = { localization_key = portal_transit_batch_quote_unavailable }
}

# Recovery quotes are amounts owed to the payer, not a new purchase quote.
defined_text = {
	name = GetPortalTransitBatchRefundAmounts
	text = {
		trigger = {
			brilliant_scientist_krg_portal_batch_can_recover_refund = yes
			set_temp_variable = { portal_batch_refund_display_index = constant:portal_transit_batch.pp_index }
			set_temp_variable = { portal_batch_refund_display_pp = krg_portal_batch_refund_remaining^portal_batch_refund_display_index }
			set_temp_variable = { portal_batch_refund_display_index = constant:portal_transit_batch.support_index }
			set_temp_variable = { portal_batch_refund_display_support = krg_portal_batch_refund_remaining^portal_batch_refund_display_index }

			set_temp_variable = { portal_batch_refund_display_fuel = krg_portal_batch_refund_remaining^portal_batch_refund_display_index }
		}
		localization_key = portal_transit_batch_refund_amounts
	}
