# ============================================================================
# CHAOS REDUX - EVENT 006 FOREIGN REACTION LOCALISATION
# Overview:
# - Names the facts behind a foreign government's interest in a new state:
#   the strongest interest first, then one secondary interest from another
#   family, then a war or settled-relationship note.
# - Shows whether a recognition or guarantee settles a mission the new state
#   has already paid for, and the terms of a patron offer.
# - The profile events read the actor's own flags and variables (ROOT is the
#   foreign actor); the consent event reads the released country's offer.
# No entry changes state.
# ============================================================================

defined_text = {
	name = GetIndependenceWaveForeignHostClause
	text = {
		trigger = {
			has_variable = independence_wave_foreign_reaction_host
			var:independence_wave_foreign_reaction_host = { exists = yes }
		}
		localization_key = independence_wave_foreign_host_clause
	}
	text = { localization_key = independence_wave_foreign_empty }
}

# The next family after the primary one, when there is one.
defined_text = {
	name = GetIndependenceWaveForeignSecondaryReason
	text = {
		trigger = {
			has_country_flag = independence_wave_foreign_family_direct
			has_country_flag = independence_wave_foreign_family_relationship
		}
		localization_key = independence_wave_foreign_secondary_relationship
	}
	text = {
		trigger = {
			has_country_flag = independence_wave_foreign_family_direct
			has_country_flag = independence_wave_foreign_family_strategic
		}
		localization_key = independence_wave_foreign_secondary_strategic
	}
	text = {
		trigger = {
			has_country_flag = independence_wave_foreign_family_direct
			has_country_flag = independence_wave_foreign_family_political
		}
		localization_key = independence_wave_foreign_secondary_political
	}
	text = {
		trigger = {
			NOT = { has_country_flag = independence_wave_foreign_family_direct }
			has_country_flag = independence_wave_foreign_family_relationship
			has_country_flag = independence_wave_foreign_family_strategic
		}
		localization_key = independence_wave_foreign_secondary_strategic
	}
	text = {
		trigger = {
			NOT = { has_country_flag = independence_wave_foreign_family_direct }
			has_country_flag = independence_wave_foreign_family_relationship
			has_country_flag = independence_wave_foreign_family_political
		}
		localization_key = independence_wave_foreign_secondary_political
	}
	text = {
		trigger = {
			NOT = { has_country_flag = independence_wave_foreign_family_direct }
			NOT = { has_country_flag = independence_wave_foreign_family_relationship }
			has_country_flag = independence_wave_foreign_family_strategic
			has_country_flag = independence_wave_foreign_family_political
		}
		localization_key = independence_wave_foreign_secondary_political
	}
	text = { localization_key = independence_wave_foreign_empty }
}
