# ============================================================================
# CHAOS REDUX - ALL SETTINGS WINDOW SCRIPTED LOCALISATION
# ============================================================================
# Checkbox sprites for chaosx_all_settings_window and the Air Cleanliness Fallout
# toggle. Every checkbox there is checked when its feature is on, so these getters
# invert the settings-page sprites that are checked when a feature is disabled.
# ============================================================================

defined_text = {
	name = GetAllSettingsEventSystemSprite
	text = {
		trigger = { has_country_flag = events_activated }
		localization_key = "GFX_chaosx_checkbox_checked"
	}
	text = {
		localization_key = "GFX_chaosx_checkbox_unchecked"
	}
}

defined_text = {
	name = GetAllSettingsChaosMeterSprite
	text = {
		trigger = { NOT = { has_global_flag = settings_chaos_meter_disabled } }
		localization_key = "GFX_chaosx_checkbox_checked"
	}
	text = {
		localization_key = "GFX_chaosx_checkbox_unchecked"
	}
}

defined_text = {
	name = GetAllSettingsWorldEndSprite
	text = {
		trigger = { NOT = { has_global_flag = world_end_disabled } }
		localization_key = "GFX_chaosx_checkbox_checked"
	}
	text = {
		localization_key = "GFX_chaosx_checkbox_unchecked"
	}
}

# Fallout stays checked once it has begun, because it can no longer be turned off.
defined_text = {
	name = GetFalloutAllowedCheckboxSprite
	text = {
		trigger = {
			OR = {
				fallout_consequence_enabled = yes
				has_global_flag = fallout_transition_active
				has_global_flag = fallout_active
				has_global_flag = fallout_air_contamination_permanent_99
			}
		}
		localization_key = "GFX_chaosx_checkbox_checked"
	}
	text = {
		localization_key = "GFX_chaosx_checkbox_unchecked"
	}
}
