# ============================================================================
# CHAOS REDUX - EVENT 039 SCRIPTED LOCALISATION
# ============================================================================
# Overview:
# - Office names for scheduled incident reports, read from the country's
#   murder_mystery_last_incident_role variable.
# - Names of the institutional bodies that replace a murdered ruler.
# - Country names of the Assassin State (from the government it broke away
#   from) and of each derivative (from the government it rose against).
# ============================================================================

defined_text = {
	name = GetMurderMysteryEmergencyCouncilName
	text = {
		trigger = { always = yes }
		localization_key = MURDER_MYSTERY_EMERGENCY_COUNCIL_NAME
	}
}

defined_text = {
	name = GetMurderMysteryActingCabinetName
	text = {
		trigger = { always = yes }
		localization_key = MURDER_MYSTERY_ACTING_CABINET_NAME
	}
}

defined_text = {
	name = GetMurderMysteryIncidentRole
	text = {
		trigger = { check_variable = { var = murder_mystery_last_incident_role value = constant:murder_mystery_target_role.investigator compare = equals } }
		localization_key = murder_mystery_incident_role_investigator
	}
	text = {
		trigger = { check_variable = { var = murder_mystery_last_incident_role value = constant:murder_mystery_target_role.judiciary_administration compare = equals } }
		localization_key = murder_mystery_incident_role_judiciary
	}
	text = {
		trigger = { check_variable = { var = murder_mystery_last_incident_role value = constant:murder_mystery_target_role.cabinet compare = equals } }
		localization_key = murder_mystery_incident_role_cabinet
	}
	text = {
		trigger = { check_variable = { var = murder_mystery_last_incident_role value = constant:murder_mystery_target_role.intelligence compare = equals } }
		localization_key = murder_mystery_incident_role_intelligence
	}
	text = {
		trigger = { always = yes }
		localization_key = murder_mystery_incident_role_unknown
	}
}

defined_text = {
	name = GetMurderMysteryAssassinStateName
	text = {
		trigger = { has_variable = murder_mystery_origin_government }
		localization_key = MURDER_MYSTERY_ASSASSIN_STATE_NAME
	}
	text = { localization_key = MURDER_MYSTERY_ASSASSIN_STATE_NAME_PLAIN }
}

defined_text = {
	name = GetMurderMysteryAssassinStateNameDef
	text = {
		trigger = { has_variable = murder_mystery_origin_government }
		localization_key = MURDER_MYSTERY_ASSASSIN_STATE_NAME_DEF
	}
	text = { localization_key = MURDER_MYSTERY_ASSASSIN_STATE_NAME_PLAIN_DEF }
}

# The derivative's origin word: the adjective of the government it rose
# against, or "Free" for a derivative without one.
defined_text = {
	name = GetMurderMysteryDerivativeOrigin
	text = {
		trigger = { has_variable = murder_mystery_derivative_parent_government }
		localization_key = MURDER_MYSTERY_DERIVATIVE_ORIGIN
	}
	text = { localization_key = MURDER_MYSTERY_DERIVATIVE_ORIGIN_FREE }
}
