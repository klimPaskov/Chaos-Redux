# ============================================================================
# CHAOS REDUX - CONDEMNATION RESPONSE COST LOCALISATION
# ============================================================================
# Every row calculates the visible payment with the same score multiplier used
# by the response effects. Integer equipment and experience amounts use the
# shared rounded-up quote; fuel remains a fixed-point fraction.
# ============================================================================

defined_text = {
	name = GetCondemnationCompensationCost
	text = {
		trigger = {
			set_temp_variable = { condemnation_cost_tier = 0 }
			if = { limit = { has_variable = condemnation_tier } set_temp_variable = { condemnation_cost_tier = condemnation_tier } }
			set_temp_variable = { condemnation_unscaled_cost = condemnation_cost_tier }
			multiply_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_convoys_per_tier }
			add_to_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_convoys_base }
			condemnation_prepare_scaled_integer_cost = yes
			set_temp_variable = { condemnation_compensation_convoys_cost = condemnation_scaled_cost }
			set_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_infantry_equipment }
			condemnation_prepare_scaled_integer_cost = yes
			set_temp_variable = { condemnation_compensation_equipment_cost = condemnation_scaled_cost }
			set_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_fuel_fraction }
			condemnation_prepare_scaled_fraction_cost = yes
			set_temp_variable = { condemnation_compensation_fuel_cost = condemnation_scaled_cost }
		}
		localization_key = condemnation.cost.compensation.row
	}
}

# Participant costs quote the condemned target's score through FROM. If the
# target is not carrying a condemnation score, the shared helper remains at 1x.
defined_text = {
	name = GetCondemnationJoinCost
	text = {
		trigger = {
			set_temp_variable = { condemnation_cost_score_override = 0 }
			if = { limit = { FROM = { has_variable = condemnation_total } } set_temp_variable = { condemnation_cost_score_override = FROM.condemnation_total } }
			set_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.participant_convoys_base }
			condemnation_prepare_scaled_integer_cost = yes
			set_temp_variable = { condemnation_join_convoys_cost = condemnation_scaled_cost }
			set_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.participant_fuel_fraction }
			condemnation_prepare_scaled_fraction_cost = yes
			set_temp_variable = { condemnation_join_fuel_cost = condemnation_scaled_cost }
		}
		localization_key = condemnation.cost.join.row
	}
}
