			NOT = { capital_scope = { supply_node > 0 } }
		}
		localization_key = independence_wave_cost_provisional_capital_viable_isolated
	}
	text = {
		trigger = {
			OR = {
				check_variable = { var = independence_wave_force_level value = constant:liberation_force_level.armed compare = equals }
