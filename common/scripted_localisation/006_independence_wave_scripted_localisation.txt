defined_text = {
	name = GetIndependenceWavePresentationArmedText
	text = {
		trigger = {
			check_variable = {
				var = global.independence_wave_presentation_armed_count
				value = constant:independence_wave_planner.no_candidates
				compare = greater_than
			}
		}
		localisation_key = independence_wave_presentation_armed_present
	}
	text = { localisation_key = independence_wave_presentation_armed_absent }
}
