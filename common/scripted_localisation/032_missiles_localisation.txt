defined_text = {
	name = GetMissilesGuidanceGradeName
	text = {
		trigger = { NOT = { has_variable = missiles_guidance_score } }
		localization_key = missiles_guidance_grade_unknown
	}
	text = {
		trigger = { check_variable = { var = missiles_guidance_score value = constant:missiles_program_value.guidance_poor compare = less_than_or_equals } }
		localization_key = missiles_guidance_grade_poor
	}
	text = {
		trigger = { check_variable = { var = missiles_guidance_score value = constant:missiles_program_value.guidance_uncertain compare = less_than_or_equals } }
		localization_key = missiles_guidance_grade_uncertain
	}
	text = {
		trigger = { check_variable = { var = missiles_guidance_score value = constant:missiles_program_value.guidance_serviceable compare = less_than_or_equals } }
		localization_key = missiles_guidance_grade_serviceable
	}
	text = {
		trigger = { check_variable = { var = missiles_guidance_score value = constant:missiles_program_value.guidance_accurate compare = less_than_or_equals } }
		localization_key = missiles_guidance_grade_accurate
	}
	text = { localization_key = missiles_guidance_grade_precise }
}
