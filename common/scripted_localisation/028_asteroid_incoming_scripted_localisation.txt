# Preparedness event: whether the locked state is the target country's current
# capital. Empty once the lock pointers are cleared after the impact.
defined_text = {
	name = GetAsteroidIncomingLockedCapitalClause
	text = {
		trigger = {
			has_event_target = asteroid_incoming_locked_state
			has_event_target = asteroid_incoming_locked_country

				check_variable = { var = capital value = event_target:asteroid_incoming_locked_state compare = equals }
