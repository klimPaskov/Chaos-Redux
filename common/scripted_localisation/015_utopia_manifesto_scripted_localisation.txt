# Manual-fire and debug availability explanation. The ordering mirrors the
# candidate diagnostic effect so only the first actionable reason is shown.
defined_text = {
	name = GetUtopiaManifestoCandidateAvailabilityReason
	text = {
		trigger = {
			OR = {
				has_country_flag = utopia_manifesto_accepted
				has_country_flag = utopia_manifesto_rejected
				has_country_flag = utopia_manifesto_focus_tree_loaded
			}
		}
		localization_key = utopia_manifesto.availability.already_handled
	}
	text = { trigger = { utopia_manifesto_candidate_has_terminal_identity = yes } localization_key = utopia_manifesto.availability.terminal_identity }
	text = { trigger = { NOT = { uses_normal_civilian_systems = yes } } localization_key = utopia_manifesto.availability.nonstandard_country }
	text = { trigger = { is_major = yes } localization_key = utopia_manifesto.availability.major }
	text = { trigger = { utopia_manifesto_candidate_has_active_event_package = yes } localization_key = utopia_manifesto.availability.active_event_package }
	text = { trigger = { utopia_manifesto_candidate_has_protected_package = yes } localization_key = utopia_manifesto.availability.protected_package }
	text = { trigger = { utopia_manifesto_candidate_has_mature_generic_tree = yes } localization_key = utopia_manifesto.availability.mature_tree }
	text = {
		trigger = {
			NOT = {
				OR = {
					has_focus_tree = generic_focus
					utopia_manifesto_candidate_has_approved_light_tree = yes
				}
			}
		}
		localization_key = utopia_manifesto.availability.non_generic_tree
	}
	text = { trigger = { has_civil_war = yes } localization_key = utopia_manifesto.availability.civil_war }
	text = { trigger = { has_offensive_war = yes } localization_key = utopia_manifesto.availability.offensive_war }
	text = { trigger = { NOT = { surrender_progress < constant:utopia_manifesto_candidate_thresholds.maximum_surrender_progress } } localization_key = utopia_manifesto.availability.near_capitulation }
	text = { trigger = { NOT = { utopia_manifesto_candidate_capital_is_secure = yes } } localization_key = utopia_manifesto.availability.insecure_capital }
	text = { trigger = { utopia_manifesto_candidate_is_dominant_faction_leader = yes } localization_key = utopia_manifesto.availability.faction_leader }
	text = { trigger = { num_of_factories > constant:utopia_manifesto_candidate_limits.maximum_total_factories } localization_key = utopia_manifesto.availability.industrial_power }
	text = { trigger = { NOT = { num_subjects < constant:utopia_manifesto_candidate_limits.extensive_subject_count } } localization_key = utopia_manifesto.availability.subject_empire }
	text = { trigger = { NOT = { num_occupied_states < constant:utopia_manifesto_candidate_limits.extensive_occupation_count } } localization_key = utopia_manifesto.availability.extensive_occupation }
	text = {
		trigger = {
			is_subject = yes
			NOT = { utopia_manifesto_subject_candidate_is_safe = yes }
		}
		localization_key = utopia_manifesto.availability.unsafe_subject
	}
	text = { localization_key = utopia_manifesto.availability.available }
}
