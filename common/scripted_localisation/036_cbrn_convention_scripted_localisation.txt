# ============================================================================
# CHAOS REDUX - EVENT 036 CHEMICAL AND BIOLOGICAL WEAPONS CONVENTION
# SCRIPTED LOCALISATION
# ============================================================================
# Overview:
# - Category header: posture with its permission line, Standing, and the
#   agenda or program line.
# - Family, project, stage, sponsor and ratification wording for the agenda,
#   resolution and program reports.
# - Projected Condemnation tiers for the opening option tooltips.
# Report text drawn on parchment never uses colour codes. Names that appear in
# the dark category header and in tooltips are coloured by their yml wrappers.
# Numeric output goes through real yml keys, because an inline formatted key in
# localization_key is tokenised as a dynamic key.
# ============================================================================

# COUNTRY scope. What the country's ratified clauses permit.
defined_text = {
	name = GetCbrnConventionPermissionLine
	text = {
		trigger = { cbrn_convention_is_opponent = yes }
		localization_key = cbrn_convention.permission.none
	}
	text = {
		trigger = {
			has_country_flag = cbrn_convention_ratified_9
			NOT = { has_country_flag = cbrn_convention_reserved_9 }
		}
		localization_key = cbrn_convention.permission.unrestricted
	}
	text = {
		trigger = {
			has_country_flag = cbrn_convention_clause_chemical
			has_country_flag = cbrn_convention_clause_biological
		}
		localization_key = cbrn_convention.permission.chemical_biological
	}
	text = {
		trigger = { has_country_flag = cbrn_convention_clause_chemical }
		localization_key = cbrn_convention.permission.chemical
	}
	text = { localization_key = cbrn_convention.permission.retaliation }
}

# Days left on the agenda under negotiation.
defined_text = {
	name = GetCbrnConventionDaysLeft
	text = {
		trigger = {
			set_temp_variable = { cbrn_convention_days_left_display = global.cbrn_convention_agenda_deadline }
			subtract_from_temp_variable = { cbrn_convention_days_left_display = global.num_days }
			clamp_temp_variable = { var = cbrn_convention_days_left_display min = constant:cbrn_convention.zero }
		}
		localization_key = cbrn_convention.header.days_left_value
	}
}
