# ============================================================================
# CHAOS REDUX - EVENT 012 SPONSORSHIP DUTY TEXT
# ============================================================================
# Country scope on a sponsored actor (call as [FROM.GetAfricaSponsorship...]).
# One duty decision and one obligation mission cover all four sponsorship
# modes; these functions name the mode the actor accepted.
# ============================================================================

defined_text = {
	name = GetAfricaSponsorshipDutyName
	text = {
		trigger = { check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.diplomatic } }
		localization_key = africa_sponsorship_duty_name_diplomatic
	}
	text = {
		trigger = { check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.material } }
		localization_key = africa_sponsorship_duty_name_material
	}
	text = {
		trigger = { check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.military } }
		localization_key = africa_sponsorship_duty_name_military
	}
	text = {
		trigger = { check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.ideological } }
		localization_key = africa_sponsorship_duty_name_ideological
	}
	text = { localization_key = africa_sponsorship_unset }
}
