# Event 012 industrial repair terms and numeric presentation.
# ACTOR country scope. Reads frozen normal terms, owner-prepared day caches and
# frozen response receipts. Only the caller-owned family classifier temp is set.
# Formatting and texticons belong to the selected localisation keys.

	name = GetGodsOfAfricaIndustryTermsLine
	text = {
		trigger = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.industry_repair }
			has_country_flag = gods_of_africa_industry_terms_bound
			gods_of_africa_industry_completion_is_proven = yes
		}
		localization_key = gods_of_africa_industry_terms_completed
	}
	text = {
		trigger = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.industry_repair }

		}
		localization_key = gods_of_africa_industry_terms

}

defined_text = {
	name = GetGodsOfAfricaIndustryResponseLine
	text = {
		trigger = {
			has_country_flag = gods_of_africa_last_response_industry
			check_variable = { gods_of_africa_last_response_kind = constant:gods_of_africa_history_kind.lapsed }
		}
		localization_key = gods_of_africa_industry_response_retired
	}
	text = {
		trigger = {
			has_country_flag = gods_of_africa_last_response_industry
			has_country_flag = gods_of_africa_last_response_industry_completed
		}
		localization_key = gods_of_africa_industry_response_completed
	}
	text = {
		trigger = {
			has_country_flag = gods_of_africa_last_response_industry
			check_variable = { gods_of_africa_last_response_industry_new_days > constant:gods_of_africa_value.zero }
			OR = {
				check_variable = { gods_of_africa_last_response_kind = constant:gods_of_africa_history_kind.compliance }
				check_variable = { gods_of_africa_last_response_kind = constant:gods_of_africa_history_kind.partial }
			}
		}
		localization_key = gods_of_africa_industry_response_paid
	}
	text = {
		trigger = {
			has_country_flag = gods_of_africa_last_response_industry
			check_variable = { gods_of_africa_last_response_kind = constant:gods_of_africa_history_kind.substitute }
		}
		localization_key = gods_of_africa_industry_response_substitute
	}
	text = {
		trigger = { has_country_flag = gods_of_africa_last_response_industry }
		localization_key = gods_of_africa_industry_response_unchanged
	}
	text = { localization_key = gods_of_africa_industry_text_empty }
}
