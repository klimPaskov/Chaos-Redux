# ============================================================================
# CHAOS REDUX - CONDEMNATION SANCTION REVIEW DURATIONS
# ============================================================================
# Scripted-localisation triggers expose the same tier review constants used by
# the sanction-pair effect. They quote nominal review periods, not the days
# remaining on an individual bilateral record.
# ============================================================================

defined_text = {
	name = GetCondemnationArmsReviewDays
	text = {
		trigger = {
			set_temp_variable = { condemnation_review_arms_days_quote = constant:condemnation_duration_days.arms_embargo_review }
			always = yes
		}
		localization_key = condemnation_review_arms_days_value
	}
}
