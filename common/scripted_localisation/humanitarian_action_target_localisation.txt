# Actor-country names for shared native decision and humanitarian GUI requirements.
# GUI text reads the persisted selected-state pointer; native decisions retain FROM.
defined_text = {
	name = GetFamineActionTargetName
	text = {
		trigger = {
			has_country_flag = famine_panel_open
			famine_panel_has_selected_state = yes
		}
		localization_key = famine_action_panel_target_name
	}
	text = {
		trigger = { has_country_flag = famine_panel_open }
		localization_key = famine_action_unselected_target_name
	}
	text = { localization_key = famine_action_native_target_name }
}
