# Event 002 news count phrase. The caller is a country/news event scope.
# Input: global.zombie_outbreak_news_site_count is reset at campaign start and incremented only after a selected state is zombie-controlled.
# Output: a grammatically singular or plural phrase for chaosx.news.3.d.
defined_text = {
	name = GetZombieOutbreakNewsSiteCount
	text = {
		trigger = {
			check_variable = {
				var = global.zombie_outbreak_news_site_count
				value = 1
				compare = equals
			}
		}
		localisation_key = zombie_outbreak_news_site_count_one
	}
	text = {
		localisation_key = zombie_outbreak_news_site_count_multiple
	}
}
