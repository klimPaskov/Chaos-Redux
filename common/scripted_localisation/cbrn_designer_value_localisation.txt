# CBRN medical designer tooltip values. Read-only percentages from authored
# reduction multipliers. Each getter owns a distinct temporary variable.

defined_text = {
	name = GetCbrnValueRespiratoryCareReduction
	text = {
		trigger = {
			set_temp_variable = { cbrn_loc_respiratory_care_reduction = constant:cbrn_math.one }
			subtract_from_temp_variable = { cbrn_loc_respiratory_care_reduction = constant:cbrn_medical_designer_effect.respiratory_care_choking_death_mult }
			multiply_temp_variable = { cbrn_loc_respiratory_care_reduction = constant:cbrn_math.percent_divisor }
			always = yes
		}
		localization_key = cbrn_value_respiratory_care_reduction
	}
}
