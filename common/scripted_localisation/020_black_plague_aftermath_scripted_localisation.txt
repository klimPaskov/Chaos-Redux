# ============================================================================
# CHAOS REDUX - EVENT 020 DEFEAT AFTERMATH LOCALISATION
# ============================================================================
# The defeat actor is optional because a Rat King can collapse through a broad
# coalition or an AI state transfer. The fallback keeps every report readable
# when no country can be attributed safely.
# ============================================================================

defined_text = {
	name = GetBlackPlagueRatKingDefeatActorName
	text = {
		trigger = { has_event_target = black_plague_rat_king_defeat_actor }
		localization_key = black_plague_rat_king_defeat_actor_name
	}
	text = {
		trigger = { always = yes }
		localization_key = black_plague_rat_king_defeat_actor_unknown
	}
}

# The shared Rat category is visible to both carriers. Keep its description
# readable for a human RTA player while retaining a separate royal summary for
# RTX instead of exposing stale base-carrier meters after coronation.
defined_text = {
	name = GetBlackPlagueRatBroodCategoryDescription
	text = {
		trigger = { black_plague_rat_country_is_king = yes }
		localization_key = black_plague_rat_brood_category_desc_king
	}
	text = {
		trigger = { always = yes }
		localization_key = black_plague_rat_brood_category_desc_rta
	}
}
