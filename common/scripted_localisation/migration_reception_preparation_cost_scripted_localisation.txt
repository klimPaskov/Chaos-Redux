# Migration reception preparation cost selectors.
# Central constants govern readiness; the existing country phase provider
# projects display-only amounts. These selectors do not mutate or pay costs.

defined_text = {
	name = GetMigrationReceptionPreparationPpCost
	text = {
		trigger = { NOT = { has_political_power < constant:migration_reception_preparation_cost.political_power } }
		localization_key = migration_reception_preparation_pp_ready
	}
	text = { trigger = { always = yes } localization_key = migration_reception_preparation_pp_blocked }
}

defined_text = {
	name = GetMigrationReceptionPreparationSupportCost
	text = {
		trigger = { NOT = { has_equipment = { support_equipment < constant:migration_reception_preparation_cost.support_equipment } } }
		localization_key = migration_reception_preparation_support_ready
	}
	text = { trigger = { always = yes } localization_key = migration_reception_preparation_support_blocked }
}
