# ============================================================================
# CHAOS REDUX - EVENT 020 SHARED RESPONSE COST LOCALISATION
# ============================================================================
# State-targeted response decisions pay the standard material bundle at one
# of three population bands. These helpers expose the exact rounded stockpile
# amounts to custom_cost_text from the same Black Friday price table entries
# that black_plague_begin_shared_state_action pays and the band triggers check.
# ============================================================================

defined_text = {
	name = GetBlackPlagueSharedSupportAmount
	text = {
		trigger = {
			check_variable = { var = state_population_k value = constant:black_plague_shared_scale.metropolitan_population_k compare = greater_than_or_equals }
			set_temp_variable = { black_plague_shared_support_quote = global.black_friday_price_black_plague_shared_cost_scaled_metropolitan_support_equipment }
		}
		localization_key = black_plague_shared_support_amount_value
	}
	text = {
		trigger = {
			check_variable = { var = state_population_k value = constant:black_plague_shared_scale.large_population_k compare = greater_than_or_equals }
			set_temp_variable = { black_plague_shared_support_quote = global.black_friday_price_black_plague_shared_cost_scaled_large_support_equipment }
		}
		localization_key = black_plague_shared_support_amount_value
	}
	text = {
		trigger = {
			set_temp_variable = { black_plague_shared_support_quote = global.black_friday_price_black_plague_shared_cost_scaled_min_support_equipment }
			always = yes
		}
		localization_key = black_plague_shared_support_amount_value
	}
}
