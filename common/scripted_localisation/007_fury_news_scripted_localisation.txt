# Event 007 opening news describes the actor count frozen by its entry batch.
# Country scope. The regular effect-chain target travels into the immediate news event.
defined_text = {
	name = GetFuryOpeningNewsTitle
	text = {
		trigger = {
			has_event_target = fury_opening_news_actor
			event_target:fury_opening_news_actor = {
				check_variable = { fury_opening_actor_count > 1 }
			}
		}
		localization_key = chaosx.news.7008.t.multiple
	}
	text = { localization_key = chaosx.news.7008.t.single }
}
