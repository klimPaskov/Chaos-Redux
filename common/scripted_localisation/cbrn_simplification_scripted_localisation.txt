# training status from current technology and equipment without duplicate flags.

defined_text = {
	name = GetCBRNPolicyStatus
	text = {
		trigger = { cbrn_policy_allows_extreme_use = yes }
		localization_key = cbrn_policy_status_unrestricted
	}
	text = {
		trigger = { cbrn_policy_allows_battlefield_use = yes }
		localization_key = cbrn_policy_status_battlefield
	}
	text = {
		trigger = { always = yes }
		localization_key = cbrn_policy_status_retaliation
	}
}

# These quotes share the same script constants as the policy payment and the
# reassessment flag. Scripted-localisation triggers set temporary values before
# their localisation keys are rendered, so this form always resolves.
# It is not established that raw constant interpolation is invalid: roughly 7000
# [?constant:category.key|format] reads exist across this mod's localisation and
# one in vanilla TAOG_decisions_l_english.yml. Settle that in the live game
# before using this comment to justify rewriting those reads.
defined_text = {
	name = GetCBRNPolicyChangeCostQuote
	text = {
		trigger = {
			set_temp_variable = { cbrn_policy_change_cost_quote = constant:cbrn_doctrine_policy_command_power_cost.policy_change }
			always = yes
		}
		localization_key = cbrn_policy_change_cost_quote_value
	}
}

# Warfare help page Condemnation paragraph. The scale reads the clamp bounds of
# condemnation_total, the concealment ceiling reads the shared response-price
# multiplier at its highest score (base + maximum score / divisor), and the
# inspection figures read the same duration and consumer-goods band that
# GetCondemnationAcceptInternationalInspectionsTooltip quotes.
defined_text = {
	name = GetCBRNHelpCondemnationSummary
	text = {
		trigger = {
			set_temp_variable = { cbrn_help_condemnation_scale_minimum = constant:condemnation_decay.minimum_total }
			set_temp_variable = { cbrn_help_condemnation_scale_maximum = constant:condemnation_threshold.total_cap }
			set_temp_variable = { cbrn_help_concealment_price_ceiling = constant:condemnation_cost.cost_multiplier_score_maximum }
			divide_temp_variable = { cbrn_help_concealment_price_ceiling = constant:condemnation_cost.cost_multiplier_score_divisor }
			add_to_temp_variable = { cbrn_help_concealment_price_ceiling = constant:condemnation_cost.cost_multiplier_base }
			set_temp_variable = { cbrn_help_inspection_days = constant:condemnation_duration_days.inspection }
			set_temp_variable = { cbrn_help_inspection_consumer_goods_minimum = constant:condemnation_cost.inspection_consumer_goods_minimum }
			set_temp_variable = { cbrn_help_inspection_consumer_goods_maximum = constant:condemnation_cost.inspection_consumer_goods_maximum }
			always = yes
		}
		localization_key = cbrn_help_condemnation_summary
	}
}

defined_text = {
	name = GetCBRNOperationalStatus
	text = {
		trigger = {
			OR = {
				NOT = { has_variable = cbrn_military_mask_requirement }
				check_variable = { cbrn_military_mask_requirement < constant:cbrn_math.one }
			}
		}
		localization_key = cbrn_operational_status_no_field_force
	}
	text = {
		trigger = { NOT = { has_tech = basic_gas_masks } }
		localization_key = cbrn_operational_status_training_needed
	}
	text = {
		trigger = { check_variable = { cbrn_military_mask_shortage_crates > constant:cbrn_math.zero } }
		localization_key = cbrn_operational_status_equipment_short
	}
	text = {
		trigger = { check_variable = { cbrn_military_mask_filter_condition < constant:cbrn_mask_filters.condition_full } }
		localization_key = cbrn_operational_status_filter_replacement
	}
	text = {
		trigger = { NOT = { has_tech = field_decontamination_kits } }
		localization_key = cbrn_operational_status_cleanup_training_needed
	}
	text = {
		trigger = { check_variable = { cbrn_military_decontamination_shortage_kits > constant:cbrn_math.zero } }
		localization_key = cbrn_operational_status_cleanup_needed
	}
	text = {
		localization_key = cbrn_operational_status_supported
	}
}

# Civil Defence category status: one named cleanup state instead of kit counts.
defined_text = {
	name = GetCBRNCivilCleanupStatus
	text = {
		trigger = {
			OR = {
				NOT = { has_variable = cbrn_civilian_decontamination_demand_kits }
				check_variable = { cbrn_civilian_decontamination_demand_kits < constant:cbrn_math.one }
			}
		}
		localization_key = cbrn_civil_cleanup_status_not_needed
	}
	text = {
		trigger = { NOT = { has_tech = field_decontamination_kits } }
		localization_key = cbrn_civil_cleanup_status_research
	}
	text = {
		trigger = { check_variable = { cbrn_civilian_decontamination_shortage_kits > constant:cbrn_math.zero } }
		localization_key = cbrn_civil_cleanup_status_short
	}
	text = {
		localization_key = cbrn_civil_cleanup_status_covered
	}
}
