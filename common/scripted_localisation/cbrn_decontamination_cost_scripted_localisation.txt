# CBRN decontamination assignment cost colors each payment independently.
# The decision cost row and its blocked state use these country-scope bindings.

defined_text = {
	name = GetCbrnCleanupCommandCost
	text = {
		trigger = {
			black_friday_quote_cbrn_theater_assignment = yes
			OR = {
				check_variable = { var = black_friday_cbrn_theater_quote_valid value = constant:universal_cost_framework.zero compare = equals }
				check_variable = { var = black_friday_cbrn_theater_command_power_affordable value = constant:universal_cost_framework.zero compare = equals }
			}
		}
		localization_key = cbrn_cleanup_command_cost_short
	}
	text = {
		trigger = { black_friday_quote_cbrn_theater_assignment = yes }
		localization_key = cbrn_cleanup_command_cost_ready
	}
}
