# CBRN decontamination assignment cost colors each payment independently.
# The decision cost row and its blocked state use these country-scope bindings.

defined_text = {
	name = GetCbrnCleanupCommandCost
	text = {
		trigger = { command_power < constant:cbrn_doctrine_cleanup_assignment_cost.command_power }
		localization_key = cbrn_cleanup_command_cost_short
	}
	text = { localization_key = cbrn_cleanup_command_cost_ready }
}

defined_text = {
	name = GetCbrnCleanupEquipmentCost
	text = {
		trigger = {
			has_equipment = {
				decontamination_equipment < constant:cbrn_doctrine_cleanup_assignment_cost.decontamination_equipment
			}
		}
		localization_key = cbrn_cleanup_equipment_cost_short
	}
	text = { localization_key = cbrn_cleanup_equipment_cost_ready }
}

defined_text = {
	name = GetCbrnCleanupFuelCost
	text = {
		trigger = { has_fuel < constant:cbrn_doctrine_cleanup_assignment_cost.fuel }
		localization_key = cbrn_cleanup_fuel_cost_short
	}
	text = { localization_key = cbrn_cleanup_fuel_cost_ready }
}
