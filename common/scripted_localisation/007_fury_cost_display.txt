# Country-scope per-component getters for decision custom-cost rows and
# tooltips. Each colour branch recomputes its quote and caller-owned outputs.

# participates in a cost component's colour.

defined_text = {
	name = GetFuryTargetCommandPowerCostDisplay
	text = {
		trigger = {
			set_temp_variable = {
				fury_cost_display_ordinary_cost = constant:fury_decision_cost.target_cp
			}
			set_temp_variable = {
				fury_cost_display_family_mask = constant:universal_cost_family.command_power
			}
			set_temp_variable = {
				fury_cost_display_resource_kind = constant:universal_cost_resource_kind.command_power
			}
			# Initialize caller-owned outputs before the helper call.
			set_temp_variable = {
				fury_cost_display_quoted_amount = fury_cost_display_ordinary_cost
			}
			set_temp_variable = {
				fury_cost_display_quote_valid = constant:universal_cost_framework.zero
			}
			set_temp_variable = {
				fury_cost_display_affordable = constant:universal_cost_framework.zero
			}
			fury_cost_display_quote_component = yes
			set_temp_variable = {
				fury_cost_display_target_command_power = fury_cost_display_quoted_amount
			}
			check_variable = {
				var = fury_cost_display_quote_valid
				value = constant:universal_cost_framework.one
				compare = equals
			}
