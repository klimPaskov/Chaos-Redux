# Country-scope quote fragments color each resource by its own inclusive
# affordability check. The owning cost-family trigger still governs payment.
# Values and formatting live in localisation, not this trigger-only selector.

defined_text = {
	name = GetGreatDepressionCostPoliticalPower
	text = {
		trigger = {
			has_variable = great_depression_cost_political_power
			NOT = { has_political_power < great_depression_cost_political_power }
		}
		localization_key = great_depression_quote_political_power_available
	}
	text = { localization_key = great_depression_quote_political_power_blocked }
}
