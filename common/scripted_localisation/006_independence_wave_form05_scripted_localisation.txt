# country pointer.

defined_text = {
	name = GetIndependenceWaveForm05Ledger
	text = {
		trigger = {
			COR = {
				has_country_flag = independence_wave_form05_league_carrier
				OR = {
					has_country_flag = independence_wave_form05_charter_active
					has_country_flag = independence_wave_form05_post_formation_active
				}
			}
		}
		localization_key = independence_wave_form05_ledger_cor
	}
	text = {
		trigger = {
			ARX = {
				has_country_flag = independence_wave_form05_league_carrier
				OR = {
					has_country_flag = independence_wave_form05_charter_active
					has_country_flag = independence_wave_form05_post_formation_active
				}
			}
		}
		localization_key = independence_wave_form05_ledger_arx
	}
	text = {
		trigger = {
			ASX = {
				has_country_flag = independence_wave_form05_league_carrier
				OR = {
					has_country_flag = independence_wave_form05_charter_active
					has_country_flag = independence_wave_form05_post_formation_active
				}
			}
		}
		localization_key = independence_wave_form05_ledger_asx
	}
	text = {
		trigger = { has_idea = independence_wave_form05_charter_breakdown }
		localization_key = independence_wave_form05_ledger_recovery
	}
	text = {
		trigger = { always = yes }
		localization_key = independence_wave_form05_ledger_dormant
	}
}
