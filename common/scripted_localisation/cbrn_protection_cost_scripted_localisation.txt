# Procurement cost text colors each spendable payment independently.
# Sale-adapted PP/support rows use Event 026 quotes; factory use stays in tooltips.

defined_text = {
	name = GetCbrnOccupiedMaskReservationQuote
	text = {
		trigger = {
			set_temp_variable = { cbrn_mask_reservation_quote = constant:cbrn_math.zero }
			cbrn_quote_occupied_program_mask_reservation = yes
		}
		localization_key = cbrn_occupied_mask_reservation_quote
	}
}

defined_text = {
	name = GetCbrnReconditionPoliticalCost
	text = {
		trigger = {
			black_friday_quote_cbrn_recondition_masks = yes
			OR = {
				check_variable = { var = black_friday_cbrn_recondition_masks_quote_valid value = constant:universal_cost_framework.zero compare = equals }
				check_variable = { var = black_friday_cbrn_recondition_masks_pp_affordable value = constant:universal_cost_framework.zero compare = equals }
			}
		}
		localization_key = cbrn_recondition_political_cost_short
	}
	text = {
		trigger = { black_friday_quote_cbrn_recondition_masks = yes }
		localization_key = cbrn_recondition_political_cost_ready
	}
}
