defined_text = {
	name = GetCbrnOccupationValueOccupationDecisionCostProtectiveAidFactories
	text = {
		trigger = {
			set_temp_variable = { cbrn_occupation_loc_occupation_decision_cost_protective_aid_factories = constant:cbrn_occupation_decision_cost.protective_aid_factories }
			if = {
				limit = { FROM = { cbrn_protective_aid_state_holds_sale_delivery = yes } }
				set_temp_variable = { cbrn_occupation_loc_occupation_decision_cost_protective_aid_factories = constant:cbrn_occupation_decision_cost.protective_aid_sale_factories }
			}
			else_if = {
				limit = { FROM = { NOT = { cbrn_occupation_state_has_donor_external_aid_reservation = yes } } }
				set_temp_variable = { cbrn_protective_aid_factory_quote = constant:cbrn_occupation_decision_cost.protective_aid_factories }
				set_temp_variable = { cbrn_protective_aid_factory_quote_valid = constant:universal_cost_framework.zero }
				cbrn_protective_aid_quote_factory_cost = yes
				set_temp_variable = { cbrn_occupation_loc_occupation_decision_cost_protective_aid_factories = cbrn_protective_aid_factory_quote }
			}
		}
		localization_key = cbrn_occupation_value_occupation_decision_cost_protective_aid_factories
	}
}

defined_text = {
	name = GetCbrnOccupationSealStatePpQuote
	text = {
		trigger = {
			set_temp_variable = { cbrn_occupation_coverup_target = FROM }

			cbrn_occupation_coverup_can_retry_refund = yes
		}
		localization_key = cbrn_occupation_coverup_recovery_pending
	}
	text = {
		trigger = {
			set_temp_variable = { cbrn_occupation_coverup_target = FROM }
			NOT = { has_variable = cbrn_occupation_coverup_open_receipt_@var:cbrn_occupation_coverup_target }
			set_temp_variable = { cbrn_occupation_coverup_pp_quote = constant:universal_cost_framework.zero }
			set_temp_variable = { cbrn_occupation_coverup_pp_ordinary = constant:universal_cost_framework.zero }
			set_temp_variable = { cbrn_occupation_coverup_pp_quote_valid = constant:universal_cost_framework.zero }
			set_temp_variable = { cbrn_occupation_coverup_pp_source_id = constant:universal_cost_framework.zero }
			cbrn_occupation_prepare_seal_state_pp_quote = yes
			check_variable = { var = cbrn_occupation_coverup_pp_quote_valid value = constant:universal_cost_framework.one compare = equals }
		}
		localization_key = cbrn_occupation_seal_state_pp_quote_value
	}
	text = { localization_key = cbrn_occupation_coverup_price_unavailable }
}

# Preserve represented fractional ordinary prices rather than rounding their
# display to two decimals. Sale prices use the registered integer quantum.
defined_text = {
	name = GetCbrnOccupationCoverupPpAmount
	text = {
		trigger = {
			set_temp_variable = { cbrn_occupation_coverup_pp_display_remainder = cbrn_occupation_coverup_pp_quote }
			modulo_temp_variable = { cbrn_occupation_coverup_pp_display_remainder = constant:universal_cost_framework.one }
			check_variable = { var = cbrn_occupation_coverup_pp_display_remainder value = constant:universal_cost_framework.zero compare = equals }
		}
		localization_key = cbrn_occupation_coverup_pp_amount_integer
	}
	text = { localization_key = cbrn_occupation_coverup_pp_amount_fractional }
}
