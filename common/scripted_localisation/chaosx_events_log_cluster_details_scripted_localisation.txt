# ============================================================================
# CHAOS REDUX - EVENT LOGS CLUSTER DETAILS FACT CELLS
# ============================================================================
# The Cluster Details window shows six fact cells in two columns. Cells 1 and 5
# read the same in both views; cells 2, 3, 4 and 6 switch between the catalogue
# view (opened from the Clusters tab) and the history view (opened from a
# logged firing, where events_log_selected_cluster_sequence is set).
# ============================================================================

defined_text = {
	name = GetEventsLogSelectedClusterFact2
	text = {
		trigger = { check_variable = { events_log_selected_cluster_sequence > 0 } }
		localization_key = chaosx.events_log.window.cluster_details.fact_2.history
	}
	text = {
		localization_key = chaosx.events_log.window.cluster_details.fact_2.catalogue
	}
}
