# ============================================================================
# CHAOS REDUX - EVENT 006 PRE-WAVE CRISIS SCRIPTED LOCALISATION
# ============================================================================
# The crisis history row stores a compact cause code. Keep the cause resolver
# in an Event 006-owned file so the shared events-log localisation surface can
# evolve independently of the crisis package.
# ============================================================================

defined_text = {
	name = GetIndependenceWaveCrisisHistoryCause
	text = {
		trigger = { check_variable = { var = events_log_history_selected_payload value = constant:independence_wave_crisis_history.occupation compare = equals } }
		localization_key = independence_wave.history.crisis.cause.occupation
	}
	text = {
		trigger = { check_variable = { var = events_log_history_selected_payload value = constant:independence_wave_crisis_history.stability compare = equals } }
		localization_key = independence_wave.history.crisis.cause.stability
	}
	text = {
		trigger = { check_variable = { var = events_log_history_selected_payload value = constant:independence_wave_crisis_history.combined compare = equals } }
		localization_key = independence_wave.history.crisis.cause.combined
	}
	text = {
		trigger = { always = yes }
		localization_key = independence_wave.history.crisis.cause.unknown
	}
}
