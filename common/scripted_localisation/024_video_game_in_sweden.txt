			has_country_flag = video_game_in_sweden_reality_audit_active

		}
		localization_key = video_game_review_narrow_audit

		trigger = { has_country_flag = video_game_in_sweden_reality_audit_active }

		trigger = { has_country_flag = video_game_in_sweden_dual_track_active }
		localization_key = video_game_review_dual_track
	}
	text = {
		trigger = { has_country_flag = video_game_in_sweden_restriction_active }
		localization_key = video_game_review_restriction
	}
	# Baseline: a high-Reliance trial that cannot evolve meets a containment
	# review; otherwise it can be concluded and closes on its own.
	text = {
		trigger = {
			check_variable = { var = video_game_in_sweden_phase value = constant:video_game_in_sweden_value.phase_baseline compare = equals }
			has_variable = video_game_in_sweden_trial_close_date
			check_variable = { var = simulation_reliance value = constant:video_game_in_sweden_reliance.doctrine_min compare = greater_than_or_equals }
			NOT = { video_game_in_sweden_evolution_i_eligible = yes }
		}
		localization_key = video_game_review_containment
	}
	text = {
		trigger = {
			check_variable = { var = video_game_in_sweden_phase value = constant:video_game_in_sweden_value.phase_baseline compare = equals }
			has_variable = video_game_in_sweden_trial_review_date
			check_variable = { var = simulation_reliance value = constant:video_game_in_sweden_reliance.doctrine_min compare = less_than }
			check_variable = { var = video_game_in_sweden_program_days value = constant:video_game_in_sweden_timing.baseline_review_days compare = less_than }
		}
		localization_key = video_game_review_baseline_pending
	}
	text = {
		trigger = {
			check_variable = { var = video_game_in_sweden_phase value = constant:video_game_in_sweden_value.phase_baseline compare = equals }
			has_variable = video_game_in_sweden_trial_close_date
			check_variable = { var = simulation_reliance value = constant:video_game_in_sweden_reliance.doctrine_min compare = less_than }
		}
		localization_key = video_game_review_baseline_open
	}
	# Idle Evolution III counts toward the final review.
	text = {
		trigger = {
			video_game_in_sweden_evolution_iii_is_idle = yes
			has_variable = video_game_in_sweden_final_review_date
			check_variable = { var = video_game_in_sweden_final_review_days value = constant:video_game_in_sweden_value.zero compare = greater_than }
		}
		localization_key = video_game_review_final
