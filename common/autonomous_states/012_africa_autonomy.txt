@africa_federal_min_freedom = 0.72

@africa_integrated_manpower_influence = 0.35

@africa_integrated_trade_share = 0.25
@africa_federal_trade_discount = -0.15
@africa_integrated_trade_discount = -0.30
@africa_federal_research_bonus = 0.03
@africa_integrated_research_bonus = 0.06
@africa_integrated_factory_share = 0.10

autonomy_state = {
	id = autonomy_africa_federal_member
	is_puppet = yes
	min_freedom_level = @africa_federal_min_freedom
	manpower_influence = @africa_federal_manpower_influence
	use_overlord_color = no

	rule = {
		desc = "RULE_DESC_IS_A_SUBJECT"
		can_not_declare_war = yes
		can_decline_call_to_war = yes
		units_deployed_to_overlord = no
		can_be_spymaster = yes
		contributes_operatives = yes
		can_create_collaboration_government = no
	}

	modifier = {
		autonomy_manpower_share = 0
		can_master_build_for_us = 1
		extra_trade_to_overlord_factor = @africa_federal_trade_share
		overlord_trade_cost_factor = @africa_federal_trade_discount
		peace_score_ratio_transferred_to_overlord = 0
		research_sharing_per_country_bonus_factor = @africa_federal_research_bonus
	}

	ai_subject_wants_higher = { factor = 0 }
	ai_overlord_wants_lower = { factor = 0 }
	ai_overlord_wants_garrison = { always = no }

	allowed = {
		has_country_flag = africa_member_autonomy_statute_accepted
		has_country_flag = africa_member_continental_constitution_accepted
		africa_member_host_generation_is_current = yes
		OR = {
			OVERLORD = { has_country_flag = africa_unifier_host }
			AND = {
				is_subject = no
				has_event_target = africa_host
				event_target:africa_host = { has_country_flag = africa_unifier_host }
			}
		}
	}

	can_take_level = { always = no }
	can_lose_level = { always = no }
}
