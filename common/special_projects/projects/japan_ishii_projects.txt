# ============================================================================
# CHAOS REDUX - JAPAN ISHII / PINGFANG SPECIAL PROJECTS
# ============================================================================
# Overview:
# - Adds the five accepted research-and-risk projects for Japan's Pingfang
#   branch without adding an operational biological attack recipe.
# - Availability follows the canonical Ishii, occupation-record, outbreak,
#   experiment-site, and desperate-war values from the repression network.
# - Only one project in this branch may be researched at a time.
# ============================================================================

@japan_ishii_project_breakthrough_low = 1
@japan_ishii_project_breakthrough_high = 2
@japan_ishii_project_steel_cost = 2
@japan_ishii_project_rubber_cost = 2
@japan_ishii_project_chromium_cost = 1
@japan_ishii_project_ai_base = 2
@japan_ishii_project_ai_war_factor = 2
@japan_ishii_project_ai_risk_factor = 2

sp_japan_pingfang_records_office = {
	specialization = specialization_biowarfare
	icon = GFX_sp_japan_pingfang_records_office
	allowed = { original_tag = JAP }
	visible = { FROM = { has_country_flag = japan_pingfang_records_office_available } }
	available = {
		FROM = {
			japan_ishii_project_branch_available = yes
			NOT = {
				OR = {
					is_special_project_being_researched = sp:sp_japan_kwantung_medical_intelligence
					is_special_project_being_researched = sp:sp_japan_occupation_test_ledger
					is_special_project_being_researched = sp:sp_japan_epidemic_mapping_bureau
					is_special_project_being_researched = sp:sp_japan_cherry_blossom_dossier
				}
			}
		}
	}
	breakthrough_cost = { specialization_biowarfare = @japan_ishii_project_breakthrough_low }
	prototype_time = sp_time.prototype.short
	resource_cost = { resources = { steel = @japan_ishii_project_steel_cost rubber = @japan_ishii_project_rubber_cost } }
	complexity = sp_complexity.small
	ai_will_do = {
		base = @japan_ishii_project_ai_base
		modifier = { factor = @japan_ishii_project_ai_war_factor FROM = { has_war = yes } }
	}
	project_output = {
		country_effects = {
			custom_effect_tooltip = sp_japan_pingfang_records_office_completed_tt
			hidden_effect = { camp_rework_japan_complete_pingfang_records_project = yes }
		}
	}
}
