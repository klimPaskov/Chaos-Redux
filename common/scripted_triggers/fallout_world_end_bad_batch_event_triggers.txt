fallout_event_pilot_bad_batch_state_is_current = {
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_variable = fallout_pretransition_air_winter_reclamation
	has_variable = fallout_pretransition_air_winter_adaptation
	check_variable = { fallout_pretransition_air_winter_reclamation > constant:fallout_event_204_threshold.minimum_reclamation }
	check_variable = { fallout_pretransition_air_winter_adaptation > constant:fallout_event_204_threshold.minimum_adaptation }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	OR = {
		AND = {
			has_state_flag = fallout_pretransition_air_winter_greenhouse_provenance_recorded
			check_variable = { fallout_pretransition_air_winter_greenhouse_generation value = global.fallout_transition_generation compare = equals }
		}
		has_state_flag = air_winter_memory_seed_vaults
		has_state_flag = air_winter_memory_seed_vaults_flourished
		has_state_flag = fallout_event_188_memory_national_archive
		has_state_flag = fallout_event_188_memory_farmers_covenant
		has_state_flag = fallout_event_188_memory_alliance_compact
		has_state_flag = fallout_event_188_memory_market_access
	}
	owner = {
		check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_204_threshold.minimum_food compare = greater_than_or_equals }
		OR = {
			AND = {
				check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_204_cost.destroy_medicine compare = greater_than_or_equals }
				check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_204_cost.destroy_scrap compare = greater_than_or_equals }
			}
			check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_204_cost.plant_food compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_204_cost.isolate_filters compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_204_cost.share_recognition compare = greater_than_or_equals }
		}
	}
	NOT = { has_state_flag = fallout_event_204_batch_registry_committed }
}
