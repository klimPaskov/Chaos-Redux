# ============================================================================
# CHAOS REDUX - WORLD-END SCENARIO ENABLEMENT HELPERS
# ============================================================================
# Persistent per-scenario selection gates shared by natural terminal branches.
# Manual Triggerable Scenario launches remain separate from automatic world-end
# selection and continue to obey their own explicit launch contract.

world_end_zombie_apocalypse_scenario_enabled = {
	NOT = { is_in_array = { global.disabled_world_end_scenarios = constant:world_end_scenario_id.zombie_apocalypse } }
}

world_end_black_plague_scenario_enabled = {
	NOT = { is_in_array = { global.disabled_world_end_scenarios = constant:world_end_scenario_id.black_plague } }
}

# Fallout is the replacement world-end consequence route. It is represented in
# the world-end selector, but it never creates an ordinary event-log or
# evolution entry.
fallout_consequence_enabled = {
	NOT = { is_in_array = { global.disabled_world_end_scenarios = constant:fallout_consequence_id.settings_ledger } }
	NOT = { is_in_array = { global.disabled_world_end_scenarios = constant:world_end_scenario_id.fallout } }
}

world_end_fallout_scenario_enabled = {
	fallout_consequence_enabled = yes
}

# now gates the Fallout-owned consequence route and does not enable a Final
# Silence world-end implementation.
