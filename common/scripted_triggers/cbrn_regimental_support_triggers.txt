# Country-level HQ readiness for native chemical operations and AI formations.
# The native raid validates its selected state and assigned formation separately.
cbrn_country_has_deployed_chemical_hq_section = {
	any_army_leader = {
		cbrn_hq_command_is_deployed = yes
		cbrn_hq_has_operations_section = yes
	}
}

# This is a country-level formation-readiness gate. The current 1.19
# country-scope AI/template surface exposes neither the selected native-raid
# target nor an Army-HQ-to-target-state pointer, so this helper cannot prove a
# per-raid target. It therefore requires a real CBRN programme or war state and
# strategic-agent stock; each caller adds its complete standing bill and the
# native operation adapter retains exact target validation before release.
cbrn_country_has_verified_chemical_force_target = {
	cbrn_country_has_any_chemical_payload_stock = yes
	OR = {
		cbrn_country_has_program = yes
		has_war = yes
	}
}
