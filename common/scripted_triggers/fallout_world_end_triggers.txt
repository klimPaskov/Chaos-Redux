# Supported scope: COUNTRY. Automatic air collapse respects the public
# Fallout catalog gate and the ordinary world-end disable switch.
fallout_air_request_is_valid = {
	fallout_request_source_is_air = yes
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = world_end_disabled }
	world_end_fallout_scenario_enabled = yes
	check_variable = {
		var = global.air_contamination_bp
		value = constant:air_contamination_threshold_bp.irreversible_100
		compare = greater_than_or_equals
	}
}

# Supported scope: COUNTRY. Final Silence is the one accepted predecessor that
# can already own the generic world_end flag when it hands control to Fallout.
fallout_final_silence_request_is_valid = {
	fallout_request_source_is_final_silence = yes
	has_global_flag = world_end_final_silence
	has_global_flag = world_end_final_silence_completed
	NOT = { has_global_flag = world_end_zombies }
	NOT = { has_global_flag = world_end_fury }
	NOT = { has_global_flag = world_end_death }
	NOT = { has_global_flag = world_end_cannibalism }
	NOT = { has_global_flag = world_end_cannibalism_wendigo }
}

# Supported scope: COUNTRY. Explicit terminal sources do not require an Air
# Contamination or Chaos threshold, but they cannot replace another terminal.
fallout_explicit_terminal_request_is_valid = {
	OR = {
		fallout_request_source_is_chemical = yes
		fallout_request_source_is_biological = yes
		fallout_request_source_is_mixed = yes
	}
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = world_end_disabled }
}

# Supported scope: COUNTRY. The manual scenario is an explicit player action.
# Its own reservation and completed-sweep checks are performed by the caller.
fallout_manual_request_is_valid = {
	fallout_request_source_is_manual = yes
	NOT = { has_global_flag = world_end }
	has_global_flag = fallout_manual_native_sweep_complete
}

		fallout_final_silence_request_is_valid = yes

		fallout_manual_request_is_valid = yes

# Supported scope: STATE. This capture-only proof binds a produced frozen row
# to the live Air Winter receipt. N/A is accepted only for an invalid state.
fallout_pretransition_air_winter_source_matches_capture = {
	fallout_pretransition_air_winter_source_is_structurally_current = yes
	OR = {
		AND = {
			check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
			air_winter_fallout_snapshot_source_receipt_is_current = yes
			check_variable = { var = fallout_pretransition_air_winter_source_schema_version value = air_winter_fallout_snapshot_producer_schema_version compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_source_generation value = air_winter_fallout_snapshot_producer_generation compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_phase value = air_winter_phase compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_exposure value = air_winter_exposure compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_recovery value = air_winter_recovery compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_adaptation value = air_winter_adaptation compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_food_reserve value = air_winter_food_reserve compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_shelter_capacity value = air_winter_shelter_capacity compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_reclamation value = air_winter_reclamation compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_water_security value = air_winter_water_security compare = equals }
			check_variable = { var = fallout_pretransition_air_winter_original_category value = air_winter_original_category compare = equals }
		}
		AND = {
			check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.not_applicable }
			NOT = { air_winter_state_is_valid = yes }
		}
	}
}

# Supported scope: COUNTRY
fallout_pretransition_country_snapshot_row_is_current = {
	has_country_flag = fallout_pretransition_country
	has_variable = fallout_pretransition_country_scope
	has_variable = fallout_pretransition_snapshot_generation
	var:fallout_pretransition_country_scope = { tag = PREV }
	check_variable = {
		var = fallout_pretransition_snapshot_generation
		value = global.fallout_snapshot_epoch_generation
		compare = equals
	}
	is_in_array = { array = global.fallout_pretransition_countries value = THIS }
	OR = {
		AND = {
			has_country_flag = fallout_pretransition_government_democratic
			NOT = { has_country_flag = fallout_pretransition_government_communism }
			NOT = { has_country_flag = fallout_pretransition_government_fascism }
			NOT = { has_country_flag = fallout_pretransition_government_neutrality }
		}
		AND = {
			has_country_flag = fallout_pretransition_government_communism
			NOT = { has_country_flag = fallout_pretransition_government_democratic }
			NOT = { has_country_flag = fallout_pretransition_government_fascism }
			NOT = { has_country_flag = fallout_pretransition_government_neutrality }
		}
		AND = {
			has_country_flag = fallout_pretransition_government_fascism
			NOT = { has_country_flag = fallout_pretransition_government_democratic }
			NOT = { has_country_flag = fallout_pretransition_government_communism }
			NOT = { has_country_flag = fallout_pretransition_government_neutrality }
		}
		AND = {
			has_country_flag = fallout_pretransition_government_neutrality
			NOT = { has_country_flag = fallout_pretransition_government_democratic }
			NOT = { has_country_flag = fallout_pretransition_government_communism }
			NOT = { has_country_flag = fallout_pretransition_government_fascism }
