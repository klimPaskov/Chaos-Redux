# COUNTRY scope. A priced decision is shown only while its preview ledger still
# describes the exact country and selected-field inputs used by the shared
# calculator. The refresh control remains outside this gate so stale ledgers
# always have a visible recovery path.
resources_found_decision_cost_preview_context_current = {
	has_country_flag = resources_found_decision_cost_preview_ready
	OR = {
		AND = {
			num_of_factories < constant:resources_found_country_scale.small_factory_ceiling
			check_variable = { resources_found_decision_preview_country_factor = constant:resources_found_country_factor.small_cost }
		}
		AND = {
			NOT = { num_of_factories < constant:resources_found_country_scale.small_factory_ceiling }
			is_major = yes
			check_variable = { resources_found_decision_preview_country_factor = constant:resources_found_country_factor.major_cost }
		}
		AND = {
			NOT = { num_of_factories < constant:resources_found_country_scale.small_factory_ceiling }
			NOT = { is_major = yes }
			num_of_factories > constant:resources_found_country_scale.large_factory_floor
			check_variable = { resources_found_decision_preview_country_factor = constant:resources_found_country_factor.large_cost }
		}
		AND = {
			NOT = { num_of_factories < constant:resources_found_country_scale.small_factory_ceiling }
			NOT = { is_major = yes }
			NOT = { num_of_factories > constant:resources_found_country_scale.large_factory_floor }
			check_variable = { resources_found_decision_preview_country_factor = constant:resources_found_country_factor.standard_cost }
		}
	}
	OR = {
		AND = { has_war = yes has_country_flag = resources_found_decision_preview_war_context }
		AND = { NOT = { has_war = yes } NOT = { has_country_flag = resources_found_decision_preview_war_context } }
	}
	OR = {
		AND = {
			NOT = { resources_found_country_has_valid_selected_field = yes }
			NOT = { has_variable = resources_found_decision_preview_field }
		}
		AND = {
			resources_found_country_has_valid_selected_field = yes
			has_variable = resources_found_decision_preview_field
			check_variable = { var = resources_found_selected_field value = resources_found_decision_preview_field compare = equals }
			var:resources_found_selected_field = {
				check_variable = { var = resources_found_excavation_depth value = ROOT.resources_found_decision_preview_excavation_depth compare = equals }
				check_variable = { var = resources_found_breach_pressure value = ROOT.resources_found_decision_preview_breach_pressure compare = equals }
				check_variable = { var = resources_found_subsurface_disturbance value = ROOT.resources_found_decision_preview_subsurface_disturbance compare = equals }
				check_variable = { var = resources_found_workforce_safety value = ROOT.resources_found_decision_preview_workforce_safety compare = equals }
				OR = {
					AND = { resources_found_contract_is_active_and_valid = yes ROOT = { has_country_flag = resources_found_decision_preview_contract_context } }
					AND = { NOT = { resources_found_contract_is_active_and_valid = yes } ROOT = { NOT = { has_country_flag = resources_found_decision_preview_contract_context } } }
				}
				OR = {
					AND = {
						NOT = { infrastructure > constant:resources_found_country_scale.poor_infrastructure_maximum }
						ROOT = { has_country_flag = resources_found_decision_preview_poor_infrastructure_context }
					}
					AND = {
						NOT = { infrastructure < constant:resources_found_country_scale.strong_infrastructure_minimum }
						ROOT = { has_country_flag = resources_found_decision_preview_strong_infrastructure_context }
					}
					AND = {
						infrastructure > constant:resources_found_country_scale.poor_infrastructure_maximum
						infrastructure < constant:resources_found_country_scale.strong_infrastructure_minimum
						ROOT = {
							NOT = { has_country_flag = resources_found_decision_preview_poor_infrastructure_context }
							NOT = { has_country_flag = resources_found_decision_preview_strong_infrastructure_context }
						}
					}
				}
			}
		}
	}
}
