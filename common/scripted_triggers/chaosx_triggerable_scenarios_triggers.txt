# ============================================================================
# CHAOS REDUX - TRIGGERABLE SCENARIO TRIGGERS
# ============================================================================
# Shared gates for manual sandbox/challenge scenarios opened through the
# triggerable scenario window.
# ============================================================================

triggerable_scenario_can_spawn_zombie_state = {
	NOT = { is_zombie_controlled_state = yes }
	controller = {
		exists = yes
		uses_normal_civilian_systems = yes
	}
}

triggerable_scenario_cannibalism_can_be_coastal_origin = {
	cannibalism_can_be_origin = yes
	any_owned_state = {
		is_controlled_by = ROOT
		is_coastal = yes
	}
}

triggerable_scenario_cannibalism_hannibal_network_unlocked = {
	OR = {
		cannibalism_hannibal_or_unifier_exists = yes
		has_global_flag = cannibalism_triggerable_hannibal_network_test_bypass
	}
}

triggerable_scenario_can_launch_selected = {
	OR = {
		check_variable = {
			var = triggerable_scenarios_selected
			value = constant:triggerable_scenario_id.zombie_apocalypse
		}
		check_variable = {
			var = triggerable_scenarios_selected
			value = constant:triggerable_scenario_id.mengele_clones
		}
		AND = {
			check_variable = {
				var = triggerable_scenarios_selected
				value = constant:triggerable_scenario_id.soviet_collapse
			}
			SOV = { exists = yes }
		}
		AND = {
			check_variable = {
				var = triggerable_scenarios_selected
				value = constant:triggerable_scenario_id.final_silence
			}
			NOT = { has_global_flag = world_end }
		}
		AND = {
			check_variable = {
				var = triggerable_scenarios_selected
				value = constant:triggerable_scenario_id.world_in_fury
			}
			any_country = { fury_can_be_selected = yes }
		}
		AND = {
			check_variable = {
				var = triggerable_scenarios_selected
				value = constant:triggerable_scenario_id.death
			}
			NOT = { has_global_flag = world_end }
			NOT = { has_global_flag = death_active }
			NOT = { has_global_flag = death_defeated }
			any_state = { death_is_valid_origin_state_outer = yes }
			any_state = { death_is_valid_triggerable_mainland_reveal_target = yes }
		}
		check_variable = {
			var = triggerable_scenarios_selected
			value = constant:triggerable_scenario_id.disaster_barrage
		}
		check_variable = {
			var = triggerable_scenarios_selected
			value = constant:triggerable_scenario_id.africa_is_one
		}
		AND = {
			check_variable = {
				var = triggerable_scenarios_selected
				value = constant:triggerable_scenario_id.cannibalism
			}
			NOT = { has_global_flag = world_end }
			NOT = { has_global_flag = cannibalism_world_end_started }
			OR = {
				AND = {
					check_variable = { triggerable_scenarios_cannibalism_type = constant:triggerable_scenario_cannibalism_type.war_horror_opening }
					cannibalism_has_valid_origin_available = yes
				}
				AND = {
					check_variable = { triggerable_scenarios_cannibalism_type = constant:triggerable_scenario_cannibalism_type.cult_seeds }
					cannibalism_has_valid_origin_available = yes
				}
				AND = {
					check_variable = { triggerable_scenarios_cannibalism_type = constant:triggerable_scenario_cannibalism_type.silent_islands }
					any_country = { triggerable_scenario_cannibalism_can_be_coastal_origin = yes }
				}
				AND = {
					check_variable = { triggerable_scenarios_cannibalism_type = constant:triggerable_scenario_cannibalism_type.cannibal_commune }
					cannibalism_has_valid_origin_available = yes
				}
				AND = {
					check_variable = { triggerable_scenarios_cannibalism_type = constant:triggerable_scenario_cannibalism_type.hannibal_network }
					triggerable_scenario_cannibalism_hannibal_network_unlocked = yes
					OR = {
						cannibalism_has_valid_origin_available = yes
						country_exists = CBL
					}
				}
			}
		}
	}
}

is_triggerable_scenario_zombie_country = {
	exists = yes
	has_country_flag = zombie_outbreak_dynamic_country
	num_of_controlled_states > 0
	NOT = { has_capitulated = yes }
	OR = {
		has_country_flag = weaponized_zombie_triggerable_scenario
		has_country_flag = zombie_triggerable_scenario_standard
	}
}

triggerable_scenario_zombie_same_type_as_actor = {
	OR = {
		AND = {
			OR = {
				has_country_flag = zombie_triggerable_scenario_standard
				has_country_flag = weaponized_zombie_type_infected
			}
			event_target:triggerable_zombie_relation_actor = {
				OR = {
					has_country_flag = zombie_triggerable_scenario_standard
					has_country_flag = weaponized_zombie_type_infected
				}
			}
		}
		AND = {
			has_country_flag = weaponized_zombie_type_rabid
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_rabid }
		}
		AND = {
			has_country_flag = weaponized_zombie_type_parasitic
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_parasitic }
		}
		AND = {
			has_country_flag = weaponized_zombie_type_mutant
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_mutant }
		}
		AND = {
			has_country_flag = weaponized_zombie_type_undead
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_undead }
		}
		AND = {
			has_country_flag = weaponized_zombie_type_necrotic
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_necrotic }
		}
		AND = {
			has_country_flag = weaponized_zombie_type_demonic
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_demonic }
		}
		AND = {
			has_country_flag = weaponized_zombie_type_wendigo
			event_target:triggerable_zombie_relation_actor = { has_country_flag = weaponized_zombie_type_wendigo }
		}
	}
}
