# ============================================================================
# CHAOS REDUX - EVENT 025 / EVENT 036 ALIEN RECOVERY LEDGER TRIGGERS
# ============================================================================
# Overview:
# - These read-only triggers expose the shared recovery ledger contract to
#   Event 025, Event 036, and their parent integration effects.
# - Country-scoped result triggers are idempotency gates; overlap triggers
#   accept only the exact aircraft or propulsion codes one and two.
# - Fragment input validation is bounded to the six configured fragment slots.
# - Owner validity checks the persistent global event target without iterating
#   over countries. Cleanup remains an explicit effect call, never a scan.
# Public trigger names include event025_can_resolve, event036_can_resolve,
# event025_has_exact_overlap, event036_has_exact_overlap,
# fragment_input_is_valid, and has_loser_evidence with the full
# chaosx_alien_recovery_ledger_ prefix.
# ============================================================================

# Any scope. True only when the persistent Event 025 owner target is present and
# still points to the historical country's identity, regardless of sovereignty.
chaosx_alien_recovery_ledger_event025_owner_target_is_valid = {
	has_event_target = chaosx_alien_recovery_event025_owner
	event_target:chaosx_alien_recovery_event025_owner = {
		exists = yes
	}
}

# Any scope. True when the global owner marker and its target agree.
chaosx_alien_recovery_ledger_event025_owner_is_recorded = {
	has_global_flag = chaosx_alien_recovery_event025_owner_recorded
	has_global_flag = chaosx_alien_recovery_event025_claimed
	chaosx_alien_recovery_ledger_event025_owner_target_is_valid = yes
}

# Country scope. A result is recorded once and must never call the external
# Event 016 owner API again.
chaosx_alien_recovery_ledger_event025_result_is_recorded = {
	has_country_flag = chaosx_alien_recovery_event025_recorded
	has_variable = chaosx_alien_recovery_event025_result
	check_variable = {
		var = chaosx_alien_recovery_event025_result
		value = constant:chaosx_nr25_result.none
		compare = greater_than
	}
}

# Country scope. Event 036 uses this as its one-shot ledger gate.
chaosx_alien_recovery_ledger_event036_result_is_recorded = {
	has_country_flag = chaosx_alien_recovery_event036_recorded
	has_variable = chaosx_alien_recovery_event036_result
	check_variable = {
		var = chaosx_alien_recovery_event036_result
		value = constant:chaosx_nr25_result.none
		compare = greater_than
	}
}

# Country scope. The stored overlap code is zero, one, or two, where one is an
# exact aircraft-family collision and two is an exact propulsion collision.
chaosx_alien_recovery_ledger_event025_has_exact_overlap = {
	has_country_flag = chaosx_alien_recovery_event025_recorded
	has_variable = chaosx_alien_recovery_event025_overlap
	check_variable = {
		var = chaosx_alien_recovery_event025_overlap
		value = constant:chaosx_nr25_value.one
		compare = greater_than_or_equals
	}
	check_variable = {
		var = chaosx_alien_recovery_event025_overlap
		value = constant:chaosx_nr25_value.two
		compare = less_than_or_equals
	}
}

# Country scope. Event 036 collides only with an Event 025 result explicitly
# classified as aircraft or propulsion by the neutral recovery ledger.
chaosx_alien_recovery_ledger_event025_has_exact_field_overlap = {
	has_country_flag = chaosx_alien_recovery_event025_recorded
	has_country_flag = chaosx_alien_recovery_event025_reward_applied
	has_variable = chaosx_alien_recovery_event025_field
	OR = {
		check_variable = {
			var = chaosx_alien_recovery_event025_field
			value = constant:chaosx_nr25_alien_field.aircraft
			compare = equals
		}
		check_variable = {
			var = chaosx_alien_recovery_event025_field
			value = constant:chaosx_nr25_alien_field.propulsion
			compare = equals
		}
	}
}

# Country scope. Event 036 has an exact overlap only when the shared Event 025
# record carries the exact aircraft or propulsion classification.
chaosx_alien_recovery_ledger_event036_has_exact_overlap = {
	has_country_flag = chaosx_alien_recovery_event036_recorded
	has_variable = chaosx_alien_recovery_event036_overlap
	check_variable = {
		var = chaosx_alien_recovery_event036_overlap

	}
	check_variable = {
		var = chaosx_alien_recovery_event036_overlap
		value = constant:chaosx_nr25_value.two
		compare = less_than_or_equals
	}
}

# Country scope. The Event 025 effect may claim the single global owner only
# when neither a country claim nor a persistent owner target already exists.
chaosx_alien_recovery_ledger_event025_can_resolve = {
	NOT = { has_country_flag = chaosx_alien_recovery_event025_recorded }
	NOT = { has_global_flag = chaosx_alien_recovery_event025_claimed }
	NOT = { has_global_flag = chaosx_alien_recovery_event025_owner_recorded }
	NOT = { has_event_target = chaosx_alien_recovery_event025_owner }
}

# Country scope. Event 036 is independently idempotent from Event 025.
chaosx_alien_recovery_ledger_event036_can_resolve = {
	exists = yes
	NOT = { has_country_flag = chaosx_alien_recovery_event036_recorded }
	OR = {
		NOT = { has_variable = chaosx_alien_recovery_event036_requested_field }
		check_variable = { var = chaosx_alien_recovery_event036_requested_field value = constant:chaosx_nr25_alien_field.aircraft compare = equals }
		check_variable = { var = chaosx_alien_recovery_event036_requested_field value = constant:chaosx_nr25_alien_field.propulsion compare = equals }
	}
}

# Country scope. Public fragment assignment requires a positive bounded slot ID
# and a positive bounded role code. The temporary names are part of the public
# effect contract and are supplied before calling the assignment effect.
chaosx_alien_recovery_ledger_fragment_input_is_valid = {
	chaosx_alien_recovery_ledger_custody_country_is_valid = yes
	has_variable = chaosx_alien_recovery_fragment_id
	has_variable = chaosx_alien_recovery_fragment_role
	OR = {
		check_variable = { var = chaosx_alien_recovery_fragment_id value = constant:chaosx_nr25_value.one compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_id value = constant:chaosx_nr25_value.two compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_id value = constant:chaosx_nr25_value.three compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_id value = constant:chaosx_nr25_value.four compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_id value = constant:chaosx_nr25_value.five compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_id value = constant:chaosx_nr25_value.six compare = equals }
	}
	OR = {
		check_variable = { var = chaosx_alien_recovery_fragment_role value = constant:chaosx_nr25_value.one compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_role value = constant:chaosx_nr25_value.two compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_role value = constant:chaosx_nr25_value.three compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_role value = constant:chaosx_nr25_value.four compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_role value = constant:chaosx_nr25_value.five compare = equals }
		check_variable = { var = chaosx_alien_recovery_fragment_role value = constant:chaosx_nr25_value.six compare = equals }
	}
	meta_trigger = {
		text = { NOT = { has_global_flag = chaosx_alien_recovery_fragment_[SLOT]_destroyed } }
		SLOT = "[?chaosx_alien_recovery_fragment_id|.0]"
	}
}
# Country scope. This is the bounded loser-result evidence gate. The effects
# still choose fragment over survey when both are present.
chaosx_alien_recovery_ledger_has_loser_evidence = {
	has_country_flag = chaosx_nr25_participant
	OR = {
		AND = {
			has_variable = chaosx_nr25_fragment_count
			check_variable = {
				var = chaosx_nr25_fragment_count
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
		}
		AND = {
			has_variable = chaosx_nr25_survey_evidence
			check_variable = {
				var = chaosx_nr25_survey_evidence
				value = constant:chaosx_nr25_value.zero
				compare = greater_than
			}
		}
		has_country_flag = chaosx_alien_recovery_verified_survey
	}
}

# Country scope. This names the standalone Event 036 path that preserves the
# legacy reward caller when Event 025 has no exact overlap record.
chaosx_alien_recovery_ledger_event036_standalone_legacy_path = {
	NOT = { has_country_flag = chaosx_alien_recovery_event036_recorded }
	NOT = { chaosx_alien_recovery_ledger_event025_has_exact_field_overlap = yes }

# Country scope. This names the later exact-overlap conversion path.
chaosx_alien_recovery_ledger_event036_overlap_path = {
	NOT = { has_country_flag = chaosx_alien_recovery_event036_recorded }
	has_country_flag = chaosx_alien_recovery_event025_recorded
	chaosx_alien_recovery_ledger_event025_has_exact_field_overlap = yes
}

# ============================================================================
# EVENT 025 EXPEDITION ACTION GATES
# ============================================================================
# These gates keep the decision surface and the runtime transaction contract
# aligned. Every resource threshold is paired with the signed delta in the
# Event 025 cost constants, so an action cannot be displayed as affordable and
# then fail because its hidden effect uses a different amount.
# ============================================================================

# Any country scope. This is the bounded AI roster candidate gate. It keeps
# ordinary AI selection to viable majors with a narrow active Kruger State
# exception and leaves every human entrant outside the probability surface.
chaosx_nr25_ai_candidate_available = {
	is_ai = yes
	OR = {
		AND = {
			is_major = yes
			NOT = { is_special_chaos_country = yes }
			NOT = { is_actual_nonhuman_country = yes }
		}
		brilliant_scientist_is_active_kruger_state = yes
	}
	NOT = { has_country_flag = chaosx_nr25_participant }
	NOT = { is_subject = yes }
	NOT = { has_capitulated = yes }
	has_political_power > constant:chaosx_nr25_cost.ai_reserve_pp
	has_fuel > constant:chaosx_nr25_cost.ai_reserve_fuel
	has_equipment = { convoy > constant:chaosx_nr25_cost.ai_reserve_convoys }
	has_equipment = { support_equipment > constant:chaosx_nr25_cost.ai_reserve_support }
}

# Country scope. Shared gate for every primary, rival, and evolution action.
chaosx_nr25_base_action_available = {
	hidden_trigger = {
		has_country_flag = chaosx_nr25_participant
		has_global_flag = chaosx_nr25_active
		NOT = { has_country_flag = chaosx_nr25_terminal }
	}
	NOT = { has_global_flag = chaosx_nr25_final_recovery_resolved }
	NOT = { has_country_flag = chaosx_nr25_evacuation_requested }
	chaosx_nr25_order_ready = yes
}

# Country scope. The expedition's shared action cooldown has run out.
chaosx_nr25_order_ready = {
	custom_trigger_tooltip = {
		tooltip = chaosx_nr25_order_ready_tt
		check_variable = {
			var = chaosx_nr25_action_cooldown
			value = constant:chaosx_nr25_value.zero
			compare = less_than_or_equals
		}
	}
}

	NOT = { has_country_flag = chaosx_nr25_terminal }
	NOT = { is_subject = yes }
	NOT = { has_capitulated = yes }
	NOT = { tag = PREV }
}

# Country scope (AI). Spec 009 withdrawal. The AI leaves only when its route
# has collapsed twice, its Logistics Readiness has stayed critical, it has
# gone a long time without a new phase or objective while unable to pay for
# its next phase, its homeland is close to surrender, or a rival already
# works the wreck while this expedition is still off the survey ground with
# nothing recovered. It stays while it holds major fragments, is in or ready
# for the final recovery, or is gambling on the technology in a losing war.
chaosx_nr25_ai_should_withdraw = {
	NOT = { has_global_flag = chaosx_nr25_final_recovery_resolved }
	NOT = { check_variable = { var = chaosx_nr25_fragment_count value = constant:chaosx_nr25_cost.fragment_major_threshold compare = greater_than_or_equals } }
	NOT = { check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.final_recovery compare = equals } }
	NOT = { has_country_flag = chaosx_nr25_final_mission_active }
	NOT = {
		AND = {
			check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.survey compare = equals }
			has_country_flag = chaosx_nr25_primary_site_confirmed
		}
	}
	NOT = {
		AND = {
			has_war = yes
			surrender_progress > constant:chaosx_nr25_ai.desperate_surrender
			NOT = { surrender_progress > constant:chaosx_nr25_ai.withdraw_home_defeat_surrender }
		}
	}
	OR = {
		check_variable = { var = chaosx_nr25_route_collapses value = constant:chaosx_nr25_ai.withdraw_route_collapses compare = greater_than_or_equals }
		check_variable = { var = chaosx_nr25_critical_readiness_pulses value = constant:chaosx_nr25_ai.withdraw_critical_pulses compare = greater_than_or_equals }
		AND = {
			NOT = { has_country_flag = chaosx_nr25_phase_mission_active }
			has_variable = chaosx_nr25_withdraw_review_date
			check_variable = { var = global.date value = chaosx_nr25_withdraw_review_date compare = greater_than }
			NOT = { chaosx_nr25_phase_gate_cost_available = yes }
		}
		AND = {
			has_war = yes
			surrender_progress > constant:chaosx_nr25_ai.withdraw_home_defeat_surrender
		}
		AND = {
			check_variable = { var = global.chaosx_nr25_leading_phase value = constant:chaosx_nr25_phase.final_recovery compare = greater_than_or_equals }
			check_variable = { var = chaosx_nr25_phase value = constant:chaosx_nr25_phase.survey compare = less_than }
			NOT = { check_variable = { var = chaosx_nr25_fragment_count value = constant:chaosx_nr25_value.zero compare = greater_than } }
			NOT = { check_variable = { var = chaosx_nr25_survey_evidence value = constant:chaosx_nr25_value.zero compare = greater_than } }
			NOT = { chaosx_nr25_wreck_breaking_apart_evolution_in_force = yes }
		}
	}
}
