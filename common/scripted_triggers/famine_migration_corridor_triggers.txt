# ============================================================================
# CHAOS REDUX - FAMINE AND MIGRATION CORRIDOR CONTRACT TRIGGERS
# ============================================================================
# Package-owned, fail-closed predicates for the exact humanitarian-corridor
# contract. State and country database IDs are persisted in normal variables;
# regular event targets are used only inside the caller's transaction chain.
# These predicates never discover a destination, mutate a ledger, create an
# event, or perform a whole-world scan.
# ============================================================================

# STATE scope. The requester must submit a trapped population/cohort state
# before a corridor can be prepared. The active-contract flag prevents a
# second request from overwriting an exact counterpart binding.
famine_migration_corridor_origin_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		has_state_flag = famine_migration_population_trapped
		has_variable = famine_migration_trapped_population
		check_variable = { var = famine_migration_trapped_population > constant:famine_migration_population.zero }
		has_variable = famine_migration_current_cohort_id
		NOT = { has_state_flag = famine_migration_cohort_selection_ambiguous }
		OWNER = { famine_migration_country_is_valid = yes }
		CONTROLLER = { famine_migration_country_is_valid = yes }
		NOT = { has_state_flag = famine_migration_corridor_contract_active }
		NOT = { has_state_flag = famine_migration_corridor_cleanup_pending }
	}
}

# STATE scope while every_neighbor_state is selecting one front. The origin
# event target is mandatory, so this helper cannot silently select a state for
# another request. Relation is supplied by the requester and is deliberately
# explicit rather than inferred from a generic any_enemy_state query.
famine_migration_corridor_front_candidate_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		has_event_target = famine_migration_corridor_origin
		NOT = { state = event_target:famine_migration_corridor_origin }
		is_neighbor_of = event_target:famine_migration_corridor_origin
		CONTROLLER = { famine_migration_country_is_valid = yes }
		NOT = { CONTROLLER = { tag = ROOT } }
		event_target:famine_migration_corridor_origin = { has_variable = famine_migration_corridor_relation }
		OR = {
			AND = {
				event_target:famine_migration_corridor_origin = { check_variable = { var = famine_migration_corridor_relation value = constant:famine_migration_corridor_relation.enemy_front compare = equals } }
				CONTROLLER = { has_war_with = ROOT }
			}
			AND = {
				event_target:famine_migration_corridor_origin = { check_variable = { var = famine_migration_corridor_relation value = constant:famine_migration_corridor_relation.neutral_border compare = equals } }
				CONTROLLER = {
					NOT = { has_war_with = ROOT }
					NOT = { is_in_faction_with = ROOT }
				}
			}
			AND = {
				event_target:famine_migration_corridor_origin = { check_variable = { var = famine_migration_corridor_relation value = constant:famine_migration_corridor_relation.faction_partner compare = equals } }
				CONTROLLER = {
					NOT = { has_war_with = ROOT }
					is_in_faction_with = ROOT
				}
			}
			AND = {
				event_target:famine_migration_corridor_origin = { check_variable = { var = famine_migration_corridor_relation value = constant:famine_migration_corridor_relation.neighbor compare = equals } }
				CONTROLLER = { NOT = { has_war_with = ROOT } }
			}
		}
	}
}

# STATE scope. A counterpart is a country ID captured from the selected
# adjacent state's controller, never from an unrelated any_country query.
famine_migration_corridor_counterpart_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		has_variable = famine_migration_corridor_requester_country_id
		has_variable = famine_migration_corridor_counterpart_country_id
		check_variable = { var = famine_migration_corridor_requester_country_id > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_counterpart_country_id > constant:famine_migration_corridor_runtime.zero }
		NOT = { check_variable = { var = famine_migration_corridor_requester_country_id value = famine_migration_corridor_counterpart_country_id compare = equals } }
		var:famine_migration_corridor_requester_country_id = { famine_migration_country_is_valid = yes }
		var:famine_migration_corridor_counterpart_country_id = { famine_migration_country_is_valid = yes }
	}
}

# STATE scope. This is the authoritative state/country counterpart test after
# preparation. The stored front must remain adjacent and controlled by the
# stored counterpart; a changed controller therefore invalidates the contract.
famine_migration_corridor_front_state_is_valid = {
	hidden_trigger = {
	famine_migration_corridor_counterpart_is_valid = yes
		has_variable = famine_migration_corridor_front_state_id
		has_variable = famine_migration_corridor_origin_controller_id
		has_variable = famine_migration_corridor_front_controller_id
		check_variable = { var = famine_migration_corridor_front_state_id > constant:famine_migration_corridor_runtime.zero }
		NOT = { check_variable = { var = famine_migration_corridor_origin_controller_id value = constant:famine_migration_corridor_runtime.zero compare = equals } }
		check_variable = { var = famine_migration_corridor_front_controller_id value = famine_migration_corridor_counterpart_country_id compare = equals }
		is_controlled_by = var:famine_migration_corridor_origin_controller_id
		var:famine_migration_corridor_front_state_id = {
			famine_migration_state_is_valid = yes
			is_neighbor_of = PREV
			is_controlled_by = var:PREV.famine_migration_corridor_counterpart_country_id
		}
	}
}

# STATE scope. Route geometry is caller-owned evidence. Every dimension must
# be present and positive; missing proof is not replaced by an inferred border
# or by the existing generic destination selector.
famine_migration_corridor_route_geometry_is_valid = {
	hidden_trigger = {
		famine_migration_corridor_front_state_is_valid = yes
		NOT = { has_state_flag = famine_migration_route_unsafe }
		NOT = { has_state_flag = famine_migration_corridor_route_invalid }
		NOT = { has_state_flag = famine_migration_corridor_attack_received }
		has_variable = famine_migration_corridor_route_geometry_proof
		has_variable = famine_migration_corridor_route_border_proof
		has_variable = famine_migration_corridor_route_transport_proof
		has_variable = famine_migration_corridor_route_safety_proof
		has_variable = famine_migration_corridor_route_actor_proof
		has_variable = famine_migration_corridor_route_ceasefire_proof
		check_variable = { var = famine_migration_corridor_route_geometry_proof > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_route_border_proof > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_route_transport_proof > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_route_safety_proof > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_route_actor_proof > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_route_ceasefire_proof > constant:famine_migration_corridor_runtime.zero }
	}
}

# STATE scope. The requester may call this on a prepared contract to make a
# pending offer. No acceptance is implied by preparation.
famine_migration_corridor_offer_request_is_valid = {
	hidden_trigger = {
		famine_migration_corridor_origin_is_valid = yes
			OR = {
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.prepared compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.offer_pending compare = equals }
		}
		famine_migration_corridor_route_geometry_is_valid = yes
		NOT = { has_state_flag = famine_migration_corridor_offer_pending }
	}
}

# STATE scope. This is called after entering the exact origin from the
# counterpart country. It cross-checks the stored counterpart and the response
# deadline before any operation state is changed.
famine_migration_corridor_acceptance_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		famine_migration_corridor_counterpart_is_valid = yes
		famine_migration_corridor_route_geometry_is_valid = yes
		has_state_flag = famine_migration_corridor_offer_pending
		check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.offer_pending compare = equals }
		has_variable = famine_migration_corridor_offer_deadline
		check_variable = { var = global.date value = famine_migration_corridor_offer_deadline compare = less_than }
	}
}

# COUNTRY scope. The counterpart response surface is exact and sparse: the
# country carries only one pending offer mirror and no state scan is needed.
famine_migration_corridor_offer_is_valid = {
	hidden_trigger = {
		famine_migration_country_is_valid = yes
		has_country_flag = famine_migration_corridor_offer_pending
		has_variable = famine_migration_corridor_offer_origin_state_id
		has_variable = famine_migration_corridor_offer_requester_country_id
		has_variable = famine_migration_corridor_offer_front_state_id
		has_variable = famine_migration_corridor_offer_deadline
		check_variable = { var = famine_migration_corridor_offer_origin_state_id > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_offer_requester_country_id > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_offer_front_state_id > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = global.date value = famine_migration_corridor_offer_deadline compare = less_than }
		var:famine_migration_corridor_offer_origin_state_id = {
			famine_migration_corridor_acceptance_is_valid = yes
			check_variable = { var = famine_migration_corridor_counterpart_country_id value = PREV.id compare = equals }
			check_variable = { var = famine_migration_corridor_front_state_id value = PREV.famine_migration_corridor_offer_front_state_id compare = equals }
			check_variable = { var = famine_migration_corridor_requester_country_id value = PREV.famine_migration_corridor_offer_requester_country_id compare = equals }
		}
	}
}

# STATE scope. Accepted state is still only a contract; population or reserve
# mutation is performed by the exact transaction owner after route binding.
famine_migration_corridor_operation_is_valid = {
	hidden_trigger = {
		famine_migration_corridor_route_geometry_is_valid = yes
		OR = {
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.accepted compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.operation_pending compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.mission_active compare = equals }
		}
		has_variable = famine_migration_corridor_operation
		OR = {
			check_variable = { var = famine_migration_corridor_operation value = constant:famine_migration_corridor_operation.evacuation compare = equals }
			check_variable = { var = famine_migration_corridor_operation value = constant:famine_migration_corridor_operation.relief compare = equals }
		}
	}
}

# STATE scope. This mission predicate requires a recorded exact transaction,
# not merely accepted diplomatic intent. The mission deadline is separate from
# the response deadline and is retained until terminal cleanup.
famine_migration_corridor_mission_is_valid = {
	hidden_trigger = {
		famine_migration_corridor_operation_is_valid = yes
		has_state_flag = famine_migration_corridor_mission_active
		has_variable = famine_migration_corridor_mission_deadline
		check_variable = { var = global.date value = famine_migration_corridor_mission_deadline compare = greater_than_or_equals }
		OR = {
			AND = {
				check_variable = { var = famine_migration_corridor_operation value = constant:famine_migration_corridor_operation.evacuation compare = equals }
				has_variable = famine_migration_corridor_evacuation_proven
				check_variable = { var = famine_migration_corridor_evacuation_proven > constant:famine_migration_corridor_runtime.zero }
			}
			AND = {
				check_variable = { var = famine_migration_corridor_operation value = constant:famine_migration_corridor_operation.relief compare = equals }
				has_variable = famine_migration_corridor_relief_proven
				check_variable = { var = famine_migration_corridor_relief_proven > constant:famine_migration_corridor_runtime.zero }
			}
		}
	}
}

# STATE scope. The route binder uses this predicate before exposing the
# existing generic destination/route contracts to an exact transfer call.
famine_migration_corridor_route_binding_is_valid = {
	hidden_trigger = {
		famine_migration_corridor_operation_is_valid = yes
		OR = {
			AND = {
				check_variable = { var = famine_migration_corridor_operation value = constant:famine_migration_corridor_operation.evacuation compare = equals }
				has_variable = famine_migration_corridor_route_food_proof
				has_variable = famine_migration_corridor_route_reception_proof
				check_variable = { var = famine_migration_corridor_route_food_proof > constant:famine_migration_corridor_runtime.zero }
				check_variable = { var = famine_migration_corridor_route_reception_proof > constant:famine_migration_corridor_runtime.zero }
			}
			AND = {
				check_variable = { var = famine_migration_corridor_operation value = constant:famine_migration_corridor_operation.relief compare = equals }
				has_variable = famine_migration_corridor_route_food_proof
				check_variable = { var = famine_migration_corridor_route_food_proof > constant:famine_migration_corridor_runtime.zero }
			}
		}
	}
}

# STATE scope. Attack disqualification is receipt-driven. An engine callback or
# parent-owned war/control hook must submit attacker and proof; this helper does
# not guess an attacker from the current war graph.
famine_migration_corridor_attack_receipt_is_valid = {
	hidden_trigger = {
		famine_migration_corridor_counterpart_is_valid = yes
		famine_migration_corridor_front_state_is_valid = yes
		has_state_flag = famine_migration_corridor_contract_active
		has_variable = famine_migration_corridor_attack_receipt_proven
		has_variable = famine_migration_corridor_attack_attacker_country_id
		check_variable = { var = famine_migration_corridor_attack_receipt_proven > constant:famine_migration_corridor_runtime.zero }
		check_variable = { var = famine_migration_corridor_attack_attacker_country_id > constant:famine_migration_corridor_runtime.zero }
		var:famine_migration_corridor_attack_attacker_country_id = { famine_migration_country_is_valid = yes }
	}
}

# COUNTRY scope. Sparse maintenance is opt-in. The caller sets a pulse marker
# when an exact offer mirror needs revalidation; this trigger never iterates
# any_owned_state or any_country.
famine_migration_corridor_country_pulse_is_valid = {
	hidden_trigger = {
		famine_migration_country_is_valid = yes
		OR = {
			has_country_flag = famine_migration_corridor_country_pulse_pending
			AND = {
				has_variable = famine_migration_corridor_country_pulse_due
				check_variable = { var = global.date value = famine_migration_corridor_country_pulse_due compare = greater_than_or_equals }
			}
		}
	}
}

# STATE scope. Cleanup may be called repeatedly from a terminal branch or an
# invalidation hook; it is needed only while active identity/response/mission
# state remains on the origin.
famine_migration_corridor_cleanup_needed = {
	hidden_trigger = {
		OR = {
			has_state_flag = famine_migration_corridor_contract_active
			has_state_flag = famine_migration_corridor_offer_pending
			has_state_flag = famine_migration_corridor_mission_armed
			has_variable = famine_migration_corridor_origin_state_id
		}
		OR = {
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.completed compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.rejected compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.expired compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.invalidated compare = equals }
			check_variable = { var = famine_migration_corridor_status value = constant:famine_migration_corridor_status.disqualified compare = equals }
		}
	}
}
