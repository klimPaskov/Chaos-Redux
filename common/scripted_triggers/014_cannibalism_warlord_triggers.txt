# ============================================================================
# CHAOS REDUX - EVENT 014 WARLORD DECISION TRIGGERS
# ============================================================================
# Validates the cannibal command category, exact-population recruitment states,
# unit caps, Larder reserves, and equipment stores. Normal recruitment remains
# disabled by Closed Muster Rolls and every scripted unit uses these gates.
# ============================================================================

cannibalism_warlord_command_is_open = {
	exists = yes
	OR = {
		AND = {
			is_cannibalism_warlord_country = yes
			has_country_flag = cannibalism_warlord_decisions_open
		}
		AND = {
			has_global_flag = cannibalism_reveal_complete
			has_country_flag = cannibalism_inherited_warlord_recruitment_open
			OR = {
				cannibalism_unified_decisions_are_open = yes
				cannibalism_wendigo_command_is_open = yes
			}
		}
	}
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = cannibalism_cleanup_complete }
}

cannibalism_state_is_warlord_consumption_territory = {
	is_controlled_by = ROOT
	OR = {
		AND = {
			OR = {
				has_state_flag = cannibalism_warlord_state
				cannibalism_state_has_active_cell = yes
			}
		}
		AND = {
			is_owned_by = ROOT
			ROOT = {
				has_country_flag = cannibalism_inherited_warlord_recruitment_open
				OR = {
					cannibalism_unified_decisions_are_open = yes
					cannibalism_wendigo_command_is_open = yes
				}
			}
		}
	}
	NOT = { cannibalism_state_is_unusable_larder = yes }
	NOT = { cannibalism_state_is_liberated_recovery = yes }
	NOT = { has_state_flag = cannibalism_recovery_active }
}

cannibalism_state_can_be_abandoned_by_current_warlord = {
	is_controlled_by = ROOT
	has_state_flag = cannibalism_warlord_state
	has_state_flag = cannibalism_active_node
	NOT = { is_capital = yes }
	NOT = { has_state_flag = cannibalism_recovery_active }
	check_variable = {
		var = cannibalism_consumption_actions
		value = constant:cannibalism_consumption.state_exhausted_actions
		compare = greater_than_or_equals
	}
	ROOT = {
		cannibalism_warlord_command_is_open = yes
		num_of_controlled_states > constant:cannibalism_slot.minimum
		has_variable = cannibalism_warlord_former_owner
		var:cannibalism_warlord_former_owner = {
			exists = yes
			NOT = { tag = ROOT }
		}
	}
}
