# has_resources_amount does not accept script-constant tokens in amount
# comparisons. This mirror matches natural_disaster_threshold.resource_presence_amount.
@ND_RESOURCE_PRESENCE_AMOUNT = 0

natural_disaster_is_valid_impact_state = {
	impassable = no
	check_variable = { state_population_k > 0 }
	OWNER = { exists = yes }
	NOT = { has_state_flag = natural_disaster_impact_guard }
	NOT = { has_state_flag = natural_disaster_impact_scheduled }
	NOT = { has_state_flag = natural_disaster_aftermath_active }
}

natural_disaster_is_valid_repeat_impact_state = {
	impassable = no
	check_variable = { state_population_k > 0 }
	OWNER = { exists = yes }
	has_state_flag = natural_disaster_aftermath_active
	NOT = { has_state_flag = natural_disaster_impact_guard }
	NOT = { has_state_flag = natural_disaster_impact_scheduled }
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.heat_wave } }
		OWNER = {
			NOT = { has_idea = heat_wave }
			NOT = { has_country_flag = natural_disaster_heat_exclusion }
		}
	}
}

natural_disaster_is_valid_coastal_impact_state = {
	natural_disaster_is_valid_impact_state = yes
	is_coastal = yes
}

natural_disaster_is_valid_heat_impact_state = {

natural_disaster_scheduled_state_is_still_valid = {
	impassable = no
	check_variable = { state_population_k > 0 }
	OWNER = { exists = yes }
	has_state_flag = natural_disaster_impact_scheduled

		limit = { check_variable = { natural_disaster_family = constant:natural_disaster_family.heat_wave } }

natural_disaster_is_valid_family_target = {
	natural_disaster_is_valid_impact_state = yes
	if = {
		limit = {
			OR = {
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.tropical_cyclone }
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.tsunami }
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.storm_surge }
			}
		}
		is_coastal = yes
	}
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.heat_wave } }
		natural_disaster_is_valid_heat_impact_state = yes
	}
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.dust_and_sandstorm } }
		OR = {
			natural_disaster_state_has_low_infrastructure = yes
			natural_disaster_state_has_agricultural_proxy = yes
			natural_disaster_state_has_resource_proxy = yes
			natural_disaster_state_has_dry_hazard_history = yes
		}
	}
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.dry_mass_movement } }
		OR = {
			natural_disaster_state_has_low_infrastructure = yes
			natural_disaster_state_has_strategic_transport = yes
			natural_disaster_state_has_resource_proxy = yes
			natural_disaster_state_has_dry_hazard_history = yes
		}
	}
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.wet_mass_movement } }
		OR = {
			natural_disaster_state_has_low_infrastructure = yes
			natural_disaster_state_has_strategic_transport = yes
			natural_disaster_state_has_resource_proxy = yes
			natural_disaster_state_has_wet_hazard_history = yes
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.volcanic_eruption }
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.massive_eruption }
			}
		}
		OR = {
			natural_disaster_state_has_resource_proxy = yes
			natural_disaster_state_has_volcanic_hazard_history = yes
			is_coastal = yes
		}
	}
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.ashfall } }
		OR = {
			natural_disaster_state_has_air_assets = yes
			natural_disaster_state_has_agricultural_proxy = yes
			natural_disaster_state_has_volcanic_hazard_history = yes
		}
	}
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.lahar } }
		OR = {
			natural_disaster_state_has_resource_proxy = yes
			natural_disaster_state_has_volcanic_hazard_history = yes
			natural_disaster_state_has_wet_hazard_history = yes
		}
	}
}

# State-scope filter used by the bounded priority refill loops. A zero phase
# keeps the selector reusable for relief and GUI callers.
natural_disaster_state_matches_priority_phase = {
	OR = {
		check_variable = { natural_disaster_priority_phase < 1 }
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.early_rescue }
			check_variable = { natural_disaster_phase = constant:natural_disaster_phase.early_rescue }
			NOT = { natural_disaster_state_has_rescue_slot = yes }
		}
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.middle_stabilization }
			check_variable = { natural_disaster_phase = constant:natural_disaster_phase.middle_stabilization }
			NOT = { natural_disaster_state_has_stabilization_slot = yes }
		}
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.late_reconstruction }
			check_variable = { natural_disaster_phase = constant:natural_disaster_phase.late_reconstruction }
			NOT = { natural_disaster_state_has_reconstruction_slot = yes }
		}
	}
}
