natural_disaster_state_can_take_disaster = {
	OWNER = { exists = yes }
	CONTROLLER = { exists = yes }
	state_population_k > 1
	NOT = { has_state_flag = natural_disaster_recently_hit }
}

natural_disaster_state_has_open_aftermath = {
	has_state_flag = natural_disaster_aftermath_active
	NOT = { has_state_flag = natural_disaster_reconstruction_complete }
}

natural_disaster_state_needs_general_response = {
	natural_disaster_state_has_open_aftermath = yes
	OR = {
		has_state_flag = natural_disaster_family_earthquake
		has_state_flag = natural_disaster_family_flood
		has_state_flag = natural_disaster_family_tropical_cyclone
		has_state_flag = natural_disaster_family_extreme_wind
		has_state_flag = natural_disaster_family_tsunami
		has_state_flag = natural_disaster_family_global_rupture
		has_state_flag = natural_disaster_family_massive_eruption
		has_state_flag = natural_disaster_family_meteor_shower
		has_state_flag = natural_disaster_family_storm_corridor
	}
}

	OR = {

			natural_disaster_state_has_open_aftermath = yes

	command_power > constant:natural_disaster_decision_cost.rescue_cp
	has_manpower > constant:natural_disaster_decision_cost.rescue_manpower
