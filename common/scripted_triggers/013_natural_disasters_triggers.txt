# Temp-variable contract used by the bounded random-family resolver. A family
# may be tried only after its evolution has unlocked it; this turns the final
# search into a complete pass over the live pool instead of another random
# draw with replacement.

	OR = {
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.earthquake }
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.flood }
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.extreme_wind }
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.thunderstorm }
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.hailstorm }

		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.heat_wave }
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.wildfire }
		check_variable = { natural_disaster_current_family = constant:natural_disaster_family.dry_mass_movement }
		AND = {
			check_variable = { natural_disaster_current_evolution > constant:natural_disaster_event.evolution_i_stage }
			OR = {
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.tornado_outbreak }

			OR = {
				check_variable = { natural_disaster_current_family = constant:natural_disaster_family.tropical_cyclone }

# State scope. Warning choices mitigate a matching hazard after the family has
# independently resolved; they never choose, replace, or manufacture the
# follow-up route.

	has_state_flag = natural_disaster_warning_action_taken
	OR = {
		AND = {
			check_variable = { natural_disaster_warning_protection = constant:natural_disaster_warning_protection.population }
			OR = {
				check_variable = { natural_disaster_chain_risk = constant:natural_disaster_chain.aftershock }

		}
		AND = {
			check_variable = { natural_disaster_warning_protection = constant:natural_disaster_warning_protection.transport }
			OR = {
				check_variable = { natural_disaster_chain_risk = constant:natural_disaster_chain.aftershock }
				check_variable = { natural_disaster_chain_risk = constant:natural_disaster_chain.supply_collapse }
				check_variable = { natural_disaster_chain_risk = constant:natural_disaster_chain.lahar }

# State scope. Generic recovery packages take the sea route only when a
# coastal card actually carries coastal/water damage. All other cards use the
# rail route. This is a physical route decision, never an inventory fallback.

	is_coastal = yes
	OR = {

# State-scope filter used by the bounded priority refill loops. A zero phase
# keeps the selector reusable for relief and GUI callers.
natural_disaster_state_matches_priority_phase = {
	OR = {
		check_variable = { natural_disaster_priority_phase < 1 }
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.early_rescue }
			check_variable = { natural_disaster_phase = constant:natural_disaster_phase.early_rescue }
			NOT = { natural_disaster_state_has_rescue_slot = yes }
			check_variable = { natural_disaster_rescue_score < constant:natural_disaster_aftermath.rescue_success_score }
			NOT = { has_state_flag = natural_disaster_phase_partial }
			NOT = { has_state_flag = natural_disaster_phase_failed }
		}
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.middle_stabilization }
			check_variable = { natural_disaster_phase = constant:natural_disaster_phase.middle_stabilization }
			NOT = { natural_disaster_state_has_stabilization_slot = yes }
			check_variable = { natural_disaster_stabilization_score < constant:natural_disaster_aftermath.stabilization_success_score }
			NOT = { has_state_flag = natural_disaster_phase_partial }
			NOT = { has_state_flag = natural_disaster_phase_failed }
		}
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.late_reconstruction }
			check_variable = { natural_disaster_phase = constant:natural_disaster_phase.late_reconstruction }
			NOT = { natural_disaster_state_has_reconstruction_slot = yes }
			check_variable = { natural_disaster_reconstruction_score < constant:natural_disaster_aftermath.reconstruction_success_score }
			NOT = { has_state_flag = natural_disaster_phase_partial }
			NOT = { has_state_flag = natural_disaster_phase_failed }
		}
	}
}

# State scope. Uses the same phase eligibility as the normal priority selector,
# but narrows the candidate pool to severe, catastrophic, and abnormal cards.
natural_disaster_state_matches_severe_reservation_phase = {
	check_variable = {
		var = natural_disaster_severity
		value = constant:natural_disaster_severity.severe
		compare = greater_than_or_equals
	}
	natural_disaster_state_matches_priority_phase = yes
}

# Country scope. A phase needs at most one protected severe slot; remaining
# capacity continues to use the ordinary priority order.
natural_disaster_country_has_severe_reserved_slot_for_phase = {
	OR = {
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.early_rescue }
			any_controlled_state = {
				check_variable = {
					var = natural_disaster_severity
					value = constant:natural_disaster_severity.severe
					compare = greater_than_or_equals
				}
				check_variable = { natural_disaster_phase = constant:natural_disaster_phase.early_rescue }
				natural_disaster_state_has_rescue_slot = yes
			}
		}
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.middle_stabilization }
			any_controlled_state = {
				check_variable = {
					var = natural_disaster_severity
					value = constant:natural_disaster_severity.severe
					compare = greater_than_or_equals
				}
				check_variable = { natural_disaster_phase = constant:natural_disaster_phase.middle_stabilization }
				natural_disaster_state_has_stabilization_slot = yes
			}
		}
		AND = {
			check_variable = { natural_disaster_priority_phase = constant:natural_disaster_phase.late_reconstruction }
			any_controlled_state = {
				check_variable = {
					var = natural_disaster_severity
					value = constant:natural_disaster_severity.severe
					compare = greater_than_or_equals
				}
				check_variable = { natural_disaster_phase = constant:natural_disaster_phase.late_reconstruction }
				natural_disaster_state_has_reconstruction_slot = yes
			}
		}
	}
}
