@natural_disaster_population_sparse_target = 50
@natural_disaster_population_low_target = 100
@natural_disaster_population_settled_target = 250
@natural_disaster_population_dense_target = 500
@natural_disaster_strategic_profile_minor = 5
@natural_disaster_strategic_profile_major = 10
@natural_disaster_strategic_profile_extreme = 15

natural_disaster_state_can_take_disaster = {
	OWNER = { exists = yes }
	CONTROLLER = { exists = yes }
	state_population_k > 1
	NOT = { has_state_flag = natural_disaster_recently_hit }
}

natural_disaster_state_has_open_aftermath = {
	has_state_flag = natural_disaster_aftermath_active
	NOT = { has_state_flag = natural_disaster_reconstruction_complete }
}

natural_disaster_state_needs_general_response = {
	natural_disaster_state_has_open_aftermath = yes
	OR = {
		has_state_flag = natural_disaster_family_earthquake
		has_state_flag = natural_disaster_family_flood
		has_state_flag = natural_disaster_family_tropical_cyclone
		has_state_flag = natural_disaster_family_extreme_wind
		has_state_flag = natural_disaster_family_tsunami
		has_state_flag = natural_disaster_family_global_rupture
		has_state_flag = natural_disaster_family_massive_eruption
		has_state_flag = natural_disaster_family_meteor_shower
		has_state_flag = natural_disaster_family_storm_corridor
	}
}

natural_disaster_state_needs_evacuation = {
	OR = {
		has_state_flag = natural_disaster_corridor_warning
		has_state_flag = natural_disaster_corridor_path_state
		has_state_flag = natural_disaster_tsunami_warning
		AND = {
			natural_disaster_state_has_open_aftermath = yes
			has_state_flag = natural_disaster_abnormal_aftermath
		}
	}

natural_disaster_country_has_recovery_state = {
	any_controlled_state = { natural_disaster_state_has_open_aftermath = yes }
}

natural_disaster_state_ai_priority_high = {
	check_variable = {
		var = natural_disaster_ai_recovery_priority
		value = constant:natural_disaster_ai_priority.high
		compare = greater_than_or_equals
	}
}

natural_disaster_state_ai_priority_critical = {

		value = constant:natural_disaster_ai_priority.critical

	check_variable = {
		var = natural_disaster_ai_recovery_priority

	}

	command_power > constant:natural_disaster_decision_cost.rescue_cp
