# ============================================================================
# CHAOS REDUX - EVENT 027 DOCTRINE RESEARCH ADAPTER TRIGGERS
# ============================================================================
# Fail-closed domain, track, branch, evolution, and transaction predicates.
# Ordinary doctrine rows are static and explicit so an invalid custom row can
# never widen the adapter to an arbitrary folder or executable callback.
# The built-in adapters are verified against the native add_mastery and
# set_sub_doctrine contract; future custom rows remain disabled unless they add
# their own explicit adapter and postcondition.
# ============================================================================

}

# Event 027's AI strategy signal is derived from native, observable AI output
# and current doctrine state. The vanilla strategy system exposes no generic
# targetless trigger for reading its internal plan entries, so these signals
# fail closed to zero when the corresponding native force, plan, or owner
# readiness evidence is absent.
doctrine_research_ai_has_army_strategy_signal = {
	OR = {
		has_doctrine = new_mobile_warfare
		has_doctrine = superior_firepower
		has_doctrine = grand_battleplan
		has_doctrine = mass_assault
		ai_has_role_template = armor
		ai_has_role_template = heavy_armor
		ai_has_role_template = mechanized_assault
		ai_has_role_template = mobile
		ai_has_role_template = motorized
		ai_has_role_template = infantry
		ai_has_role_template = anti_armor
		has_offensive_war = yes
		has_defensive_war = yes
	}
}
doctrine_research_ai_has_navy_strategy_signal = {
	OR = {
		has_doctrine = new_fleet_in_being
		has_doctrine = new_convoy_raiding
		has_doctrine = new_base_strike
		has_navy_size = { unit = submarine size > 0 }
		has_navy_size = { unit = destroyer size > 0 }
		OR = { has_navy_size = { unit = light_cruiser size > 0 } has_navy_size = { unit = heavy_cruiser size > 0 } }
		has_navy_size = { unit = battleship size > 0 }
		has_navy_size = { unit = carrier size > 0 }
		has_navy_size = { type = convoy size > 0 }
		num_of_naval_factories > 0
	}
}
doctrine_research_ai_has_air_strategy_signal = {
	OR = {
		has_doctrine = new_strategic_destruction
		has_doctrine = new_battlefield_support
		has_doctrine = new_operational_integrity
		has_deployed_air_force_size = { type = fighter size > 0 }
		has_deployed_air_force_size = { type = cas size > 0 }
		has_deployed_air_force_size = { type = naval_bomber size > 0 }
		has_deployed_air_force_size = { type = tactical_bomber size > 0 }
		has_deployed_air_force_size = { type = strategic_bomber size > 0 }
	}
}
doctrine_research_ai_has_special_forces_strategy_signal = {
	OR = {
		has_doctrine = special_forces_quantity
		has_doctrine = special_forces_quality
		ai_has_role_template = mountaineers
		ai_has_role_template = marines
		ai_has_role_template = paratroopers
		ai_has_role_template = rangers_role
		has_template_containing_unit = mountaineers
		has_template_containing_unit = marine
		has_template_containing_unit = paratrooper
		has_template_containing_unit = ranger_battalion
	}
}
doctrine_research_ai_has_chaos_warfare_strategy_signal = {
	OR = {
		has_doctrine = chaos_warfare
		cbrn_chaos_warfare_ai_has_viable_program = yes
	}
}

	OR = {
		any_controlled_state = { is_coastal = yes }
		num_of_naval_factories > 0
		has_navy_size = { unit = submarine size > 0 }
		has_navy_size = { unit = destroyer size > 0 }
		OR = { has_navy_size = { unit = light_cruiser size > 0 } has_navy_size = { unit = heavy_cruiser size > 0 } }

	OR = {
		AND = {
			check_variable = { var = doctrine_research_selected_domain value = constant:doctrine_research_domain.army compare = equals }
			doctrine_research_army_has_valid_action = yes
		}
		AND = {
			check_variable = { var = doctrine_research_selected_domain value = constant:doctrine_research_domain.navy compare = equals }

			check_variable = { var = doctrine_research_selected_domain value = constant:doctrine_research_domain.special_forces compare = equals }

		}

			check_variable = { var = doctrine_research_selected_domain value = constant:doctrine_research_domain.chaos_warfare compare = equals }
			cbrn_chaos_warfare_ai_has_viable_program = yes
		}
	}
}

doctrine_research_country_is_live = {
	exists = yes
	# A government in exile remains a valid owner even without controlled
	# territory. A capitulated country that has not entered exile is no longer a
	# live Event 027 participant.
	OR = {
		NOT = { has_capitulated = yes }
		is_government_in_exile = yes
	}
}
