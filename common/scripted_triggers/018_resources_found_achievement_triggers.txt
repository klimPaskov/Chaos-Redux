# ============================================================================
# CHAOS REDUX - EVENT 018 ACHIEVEMENT TRIGGERS

# Final-state predicates for the fifteen Event 018 achievements. Historical
# route facts are recorded by field, cave, decision, and cleanup effects; these
# triggers never infer a past choice from an unrelated current modifier.

resources_found_achievement_is_ordinary_field_country = {
	NOT = { is_resources_found_cave_country = yes }
	has_country_flag = resources_found_achievement_started_ordinary_country
	NOT = { has_country_flag = resources_found_cave_player_continuation_chosen }
}

resources_found_achievement_one_vein_market = {
	resources_found_achievement_is_ordinary_field_country = yes
	any_owned_state = {
		resources_found_is_active_field = yes
		is_owned_and_controlled_by = ROOT
		OR = {
			check_variable = { var = resources_found_added_oil value = constant:resources_found_achievement.one_vein_minimum compare = greater_than_or_equals }
			check_variable = { var = resources_found_added_aluminium value = constant:resources_found_achievement.one_vein_minimum compare = greater_than_or_equals }
			check_variable = { var = resources_found_added_rubber value = constant:resources_found_achievement.one_vein_minimum compare = greater_than_or_equals }
			check_variable = { var = resources_found_added_tungsten value = constant:resources_found_achievement.one_vein_minimum compare = greater_than_or_equals }
			check_variable = { var = resources_found_added_steel value = constant:resources_found_achievement.one_vein_minimum compare = greater_than_or_equals }
			check_variable = { var = resources_found_added_chromium value = constant:resources_found_achievement.one_vein_minimum compare = greater_than_or_equals }
		}
	}
}

# STATE scope. This is the live predicate sampled for the entire immutable
# long-term export review. Historical failures are latched by the matching
# refresh effect, so restoring a value before expiry cannot recover the term.
resources_found_contract_century_conditions_met = {
	resources_found_is_active_field = yes
	has_state_flag = resources_found_contract_century_qualifying
	has_state_flag = resources_found_long_term_export_contract
	has_state_flag = resources_found_field_contract_active
	has_variable = resources_found_contract_century_owner
	is_owned_and_controlled_by = var:resources_found_contract_century_owner
	var:resources_found_contract_century_owner = {
		exists = yes
		is_subject = no
		NOT = { is_resources_found_cave_country = yes }
	}
	resources_found_contract_is_active_and_valid = yes
	check_variable = {
		var = resources_found_developed_yield
		value = constant:resources_found_achievement.industrial_yield
		compare = greater_than_or_equals
	}
	check_variable = {
		var = resources_found_workforce_safety
		value = constant:resources_found_achievement.managed_safety
		compare = greater_than_or_equals
	}
	check_variable = { resources_found_foreign_pressure < constant:resources_found_achievement.pressure_below_crisis }
	NOT = { has_state_flag = resources_found_public_breach }
	NOT = { has_state_flag = resources_found_contract_century_border_war }
	NOT = { has_state_flag = resources_found_contract_century_disqualified }
	NOT = { has_state_flag = resources_found_field_suspended }
	NOT = { has_state_flag = resources_found_field_closing }
	NOT = { has_state_flag = resources_found_field_control_disrupted }
	NOT = { has_state_flag = resources_found_field_converted_to_cave_origin }
}

# STATE scope. Both facts are historical and claimant-specific; a severe field
# dispute cannot be manufactured by starting an agreement after a quiet claim.
resources_found_dispute_has_required_history = {
	resources_found_is_active_field = yes
	has_variable = resources_found_border_claimant
	has_variable = resources_found_dispute_history_owner
	is_owned_by = var:resources_found_dispute_history_owner
	var:resources_found_border_claimant = {
		exists = yes
		NOT = { is_special_chaos_country = yes }
		NOT = { is_actual_nonhuman_country = yes }
	}
	has_state_flag = resources_found_dispute_reached_crisis_pressure
	has_state_flag = resources_found_dispute_reached_armed_stage
	NOT = { has_state_flag = resources_found_dispute_border_war_fought }
	NOT = { has_state_flag = resources_found_dispute_border_transfer }
	NOT = { has_state_flag = resources_found_dispute_history_owner_changed }
	NOT = { has_state_flag = resources_found_border_claimant_removed }
}
