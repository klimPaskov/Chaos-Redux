# Exact adapter and pre-release history proofs for playable Event 006 packages.

# Input: temporary `independence_wave_execution_package_id`.
has_independence_wave_runtime_package_adapter_for_execution_id = {
	OR = {
		check_variable = { independence_wave_execution_package_id = constant:independence_wave_package_id.iw_006 }
		check_variable = { independence_wave_execution_package_id = constant:independence_wave_package_id.iw_007 }
	}
}

# COUNTRY scope for the frozen target tag. Input: temporary
# `independence_wave_execution_package_id`. This proof runs before release, so
# every item checked here must come from the dormant country history or the
# audited package registry rather than from post-release setup.
is_independence_wave_runtime_package_preflight_ready = {
	exists = no
	has_country_flag = independence_wave_package_content_ready
	has_independence_wave_runtime_package_adapter_for_execution_id = yes
	OR = {
		AND = {
			check_variable = { independence_wave_execution_package_id = constant:independence_wave_package_id.iw_006 }
			original_tag = AFX
			has_idea = civilian_economy
			has_idea = export_focus
			has_idea = volunteer_only
			has_independence_wave_afx_command_roster = yes
		}
		AND = {
			check_variable = { independence_wave_execution_package_id = constant:independence_wave_package_id.iw_007 }
			original_tag = AGX
			has_idea = civilian_economy
			has_idea = export_focus
			has_idea = volunteer_only
			has_independence_wave_agx_command_roster = yes
		}
	}
}
