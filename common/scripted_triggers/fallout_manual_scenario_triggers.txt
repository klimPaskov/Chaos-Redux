	fallout_manual_population_contract_preflight_is_current = yes
	has_state_flag = fallout_manual_population_contract_preflight_current
	has_variable = fallout_manual_population_contract_preflight_generation
	has_variable = fallout_manual_population_contract_preflight_prestrike_k
	has_variable = fallout_manual_population_contract_preflight_prestrike_people
	has_variable = fallout_manual_population_contract_preflight_first_week_loss_people
	has_variable = fallout_manual_population_contract_preflight_first_week_after_people
	has_variable = fallout_manual_population_contract_preflight_frozen_current_people
	has_variable = fallout_manual_population_contract_preflight_target_survivors

	check_variable = { var = fallout_manual_population_contract_preflight_prestrike_k value = fallout_manual_prestrike_population_k compare = equals }
	check_variable = { var = fallout_manual_population_contract_preflight_prestrike_people value = fallout_manual_prestrike_population_people compare = equals }
	check_variable = { var = fallout_manual_population_contract_preflight_first_week_loss_people value = fallout_manual_first_week_population_loss_people compare = equals }
