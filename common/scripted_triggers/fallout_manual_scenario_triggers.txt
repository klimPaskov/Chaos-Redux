	has_variable = global.fallout_manual_selected_intensity
	check_variable = {
		var = global.fallout_manual_selected_intensity
		value = constant:fallout_manual_sweep.minimum_intensity
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.fallout_manual_selected_intensity
		value = constant:fallout_manual_sweep.maximum_intensity
		compare = less_than_or_equals
	}
}
