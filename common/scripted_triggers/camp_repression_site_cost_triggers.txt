# Repression-owned deterministic quote and inclusive affordability predicates.
# Country scope. Required input: unscoped temporary camp_site_cost_state_id.
# Temporary quote outputs are unscoped. Callers retaining outputs initialize them first.
# Every payment amount is the Event 026 sale price: the site quote applies the
# ratio the sale price table holds through camp_rework_sale_quote, and the fixed
# order stock below reads the table's camp_rework_cost and
# camp_rework_equipment_cost prices. Contract: camp_repression_site_cost_effects.md.

# Any scope. Trigger-side mirror of black_friday_price_table_quote
# (026_black_friday_price_table_effects.txt) for camp quotes computed inside
# triggers. Inputs: temp camp_sale_ordinary and optional temp camp_sale_quantum
# (zero or unset means one whole unit). Reads the ratio the sale price table
# holds, global.black_friday_price_table_ratio, or the full price before the
# table is first built. Output: temp camp_sale_payable. Outside a sale it is the
# ordinary amount; during a sale it is the amount times the payment ratio,
# rounded up to whole steps, at least one step and never above the ordinary
# amount. Zero and negative amounts are unchanged. Always true.
camp_rework_sale_quote = {
	set_temp_variable = { camp_sale_payable = camp_sale_ordinary }
	set_temp_variable = { camp_sale_ratio = constant:black_friday_event.ratio_basis }
	if = {
		limit = { has_variable = global.black_friday_price_table_ratio }
		set_temp_variable = { camp_sale_ratio = global.black_friday_price_table_ratio }
	}
	if = {
		limit = {
			check_variable = { camp_sale_ordinary > constant:black_friday_event.zero }
			check_variable = { camp_sale_ratio > constant:black_friday_event.zero }
			check_variable = { var = camp_sale_ratio value = constant:black_friday_event.ratio_basis compare = less_than }
		}
		set_temp_variable = { camp_sale_step = constant:black_friday_event.one }
		if = {
			limit = { check_variable = { camp_sale_quantum > constant:black_friday_event.zero } }
			set_temp_variable = { camp_sale_step = camp_sale_quantum }
		}
		# The ratio becomes a fraction first, as in the production table quote.
		set_temp_variable = { camp_sale_fraction = camp_sale_ratio }
		divide_temp_variable = { camp_sale_fraction = constant:black_friday_event.ratio_basis }
		multiply_temp_variable = { camp_sale_payable = camp_sale_fraction }
		divide_temp_variable = { camp_sale_payable = camp_sale_step }
		set_temp_variable = { camp_sale_whole = camp_sale_payable }
		round_temp_variable = camp_sale_whole
		if = {
			limit = { check_variable = { var = camp_sale_whole value = camp_sale_payable compare = less_than } }
			add_to_temp_variable = { camp_sale_whole = constant:black_friday_event.one }
		}
		clamp_temp_variable = { var = camp_sale_whole min = constant:black_friday_event.one }
		set_temp_variable = { camp_sale_payable = camp_sale_whole }
		multiply_temp_variable = { camp_sale_payable = camp_sale_step }
		if = {
			limit = { check_variable = { camp_sale_payable > camp_sale_ordinary } }
			set_temp_variable = { camp_sale_payable = camp_sale_ordinary }
		}
	}
	always = yes
}
