# Shared native/panel operation gates. Internal identity and phase checks stay hidden.
# Visible requirement clauses retain the native decision custom tooltips.

migration_prepare_evacuation_action_can_start = {
	set_temp_variable = {
		humanitarian_action_actor = ROOT
	}
	hidden_trigger = {
		has_variable = humanitarian_action_target
		var:humanitarian_action_target = {
			humanitarian_state_is_valid = yes
		}
		set_temp_variable = {
			humanitarian_action_target_id = constant:humanitarian_runtime.zero
		}
		var:humanitarian_action_target = {
			set_temp_variable = {
				humanitarian_action_target_id = id
			}
		}
		NOT = {
			has_variable = migration_action_due_prepare_evacuation_@var:humanitarian_action_target_id
		}
		OR = {
			NOT = {
				has_variable = migration_action_cooldown_prepare_evacuation_@var:humanitarian_action_target_id
			}
			check_variable = {
				var = global.date
				value = migration_action_cooldown_prepare_evacuation_@var:humanitarian_action_target_id
				compare = greater_than_or_equals
			}
		}
		var:humanitarian_action_target = {
			NOT = {
				has_state_flag = migration_evacuation_prepared
			}
		}
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
			migration_state_has_real_migration_issue = yes
		}
		NOT = {
			has_country_flag = humanitarian_corridor_offer_pending
		}
		var:humanitarian_action_target = {
			humanitarian_state_is_valid = yes
			OR = {
				AND = {
					famine_to_migration_food_safety_is_valid = yes
					check_variable = {
						var = famine_to_migration_food_stage
						value = constant:famine_food_stage.supply_strain
						compare = greater_than_or_equals
					}
					OR = {
						check_variable = {
							var = famine_to_migration_food_exposure_days
							value = constant:migration_decision_threshold.category_food_exposure_days
							compare = greater_than_or_equals
						}
						check_variable = {
							var = famine_to_migration_food_incident_count
							value = constant:migration_decision_threshold.category_food_incidents
							compare = greater_than_or_equals
						}
					}
				}
				AND = {
					has_state_flag = migration_displacement_active
					OR = {
						check_variable = {
							var = migration_flight_pressure
							value = constant:migration_decision_threshold.category_flight_people
							compare = greater_than_or_equals
						}
						check_variable = {
							var = migration_trapped_population
							value = constant:migration_decision_threshold.category_trapped_people
							compare = greater_than_or_equals
						}
					}
				}
			}
		}
		OR = {
			NOT = {
				has_variable = migration_panel_selected_cohort_id
			}
			check_variable = {
				migration_panel_selected_cohort_id = constant:humanitarian_runtime.zero
			}
			var:humanitarian_action_target = {
				check_variable = {
					migration_current_cohort_id = ROOT.migration_panel_selected_cohort_id
				}
			}
		}
	}
	var:humanitarian_action_target = {
		NOT = {
			has_state_flag = migration_evacuation_prepared
		}
	}
	custom_trigger_tooltip = {
		tooltip = migration_real_pressure_tt
		hidden_trigger = {
			var:humanitarian_action_target = {
				is_controlled_by = ROOT
				migration_state_has_real_migration_issue = yes
			}
		}
	}
	custom_trigger_tooltip = {
		tooltip = migration_prepare_evacuation_target_tt
		var:humanitarian_action_target = {
			humanitarian_state_is_valid = yes
			OR = {
				AND = {
					famine_to_migration_food_safety_is_valid = yes
					check_variable = {
						var = famine_to_migration_food_stage
						value = constant:famine_food_stage.supply_strain
						compare = greater_than_or_equals
					}
					OR = {
						check_variable = {
							var = famine_to_migration_food_exposure_days
							value = constant:migration_decision_threshold.category_food_exposure_days
							compare = greater_than_or_equals
						}
						check_variable = {
							var = famine_to_migration_food_incident_count
							value = constant:migration_decision_threshold.category_food_incidents
							compare = greater_than_or_equals
						}
					}
				}
				AND = {
					has_state_flag = migration_displacement_active
					OR = {
						check_variable = {
							var = migration_flight_pressure
							value = constant:migration_decision_threshold.category_flight_people
							compare = greater_than_or_equals
						}
						check_variable = {
							var = migration_trapped_population
							value = constant:migration_decision_threshold.category_trapped_people
							compare = greater_than_or_equals
						}
					}
				}
			}
		}
	}
	hidden_trigger = {
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
		}
	}
	hidden_trigger = {
		migration_prepare_evacuation_cost_is_affordable_in_context = yes
	}
	hidden_trigger = {
		NOT = {
			var:humanitarian_action_target = {
				has_state_flag = migration_prepare_evacuation_cost_paid
			}
		}
	}
}

migration_prepare_evacuation_action_pending_is_current = {
	hidden_trigger = {
		has_variable = migration_action_generation_prepare_evacuation_@var:humanitarian_action_target_id
		check_variable = {
			migration_action_generation_prepare_evacuation_@var:humanitarian_action_target_id = humanitarian_action_row_id
		}
		var:humanitarian_action_target = {
			has_variable = migration_prepare_evacuation_operation_generation
			check_variable = {
				migration_prepare_evacuation_operation_generation = humanitarian_action_row_id
			}
			has_variable = migration_prepare_evacuation_operation_actor
			var:migration_prepare_evacuation_operation_actor = {
				exists = yes
				tag = ROOT
			}
		}
	}
}
