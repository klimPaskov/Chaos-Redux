# COUNTRY scope on the host. The initializer snapshots the current playbook so
# a stale profile cannot admit actions after host succession or reassignment.
africa_first_proof_profile_is_current = {
	africa_is_current_host = yes
	has_country_flag = africa_first_proof_profile_initialized
	check_variable = {
		var = africa_first_proof_profile_playbook
		value = africa_host_playbook
		compare = equals
	}
}

# COUNTRY scope on the host. Input: temporary africa_requested_action_id.
# The profile lists one exact row action, one optional route alternative, and
# the additional actions that may supply compound proof roles.
africa_first_proof_requested_action_is_allowed = {
	OR = {
		AND = {
			has_country_flag = africa_first_proof_active
			NOT = { has_country_flag = africa_first_proof_failed }
			africa_first_proof_profile_is_current = yes
			check_variable = { africa_requested_action_id > constant:africa_action.none }
			OR = {
				check_variable = {
					var = africa_requested_action_id
					value = africa_first_proof_primary_action_id
					compare = equals
				}
				check_variable = {
					var = africa_requested_action_id
					value = africa_first_proof_alternate_primary_action_id
					compare = equals
				}
				check_variable = {
					var = africa_requested_action_id
					value = africa_first_proof_additional_action_id_1
					compare = equals
				}
				check_variable = {
					var = africa_requested_action_id
					value = africa_first_proof_additional_action_id_2
					compare = equals
				}
				check_variable = {
					var = africa_requested_action_id
					value = africa_first_proof_additional_action_id_3
					compare = equals
				}
				check_variable = {
					var = africa_requested_action_id
					value = africa_first_proof_additional_action_id_4
					compare = equals
				}
			}
		}
		AND = {
			africa_first_proof_recovery_host_can_select_action = yes
			check_variable = { africa_requested_action_id = constant:africa_action.recover_failed_host_proof }
		}
	}
}

# COUNTRY scope on the host after the exact target and any state cursor have
# been prepared. This is the opening-proof semantic gate; it admits no action
# outside the current 51-row profile and keeps war, route, consent, and partner
# requirements visible instead of inheriting later-phase relationship gates.
africa_first_proof_current_action_semantics_are_valid = {
	africa_first_proof_requested_action_is_allowed = yes
	has_event_target = africa_action_target
	event_target:africa_action_target = { africa_action_target_is_usable = yes }
	africa_first_proof_playbook_action_target_is_exact = yes
	OR = {
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.guarantee_sovereignty }
			event_target:africa_action_target = { NOT = { africa_is_current_host = yes } }
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.open_aid_corridor }
			event_target:africa_action_target = { NOT = { africa_is_current_host = yes } }
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.dispatch_charter_mission }
			event_target:africa_action_target = { NOT = { africa_is_current_host = yes } }
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.deploy_volunteers }
			event_target:africa_action_target = { has_war = yes }
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.offer_defence_charter }
			event_target:africa_action_target = { NOT = { africa_is_current_host = yes } }
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.hold_accession_referendum }
			event_target:africa_action_target = { NOT = { africa_is_current_host = yes } }
		}
		AND = {
			OR = {
				check_variable = { africa_requested_action_id = constant:africa_action.convene_regional_congress }
				check_variable = { africa_requested_action_id = constant:africa_action.settle_overlapping_claims }
				check_variable = { africa_requested_action_id = constant:africa_action.guarantee_regional_representation }
				check_variable = { africa_requested_action_id = constant:africa_action.build_regional_rail_spine }
				check_variable = { africa_requested_action_id = constant:africa_action.expand_river_transport }
				check_variable = { africa_requested_action_id = constant:africa_action.modernise_continental_port }
				check_variable = { africa_requested_action_id = constant:africa_action.resource_sovereignty_review }
			}
			africa_has_selected_action_state_target = yes
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.connect_member_capitals }
			event_target:africa_action_target = { NOT = { africa_is_current_host = yes } }
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.offer_rival_arbitration }
			event_target:africa_action_target = {
				OR = {
					has_war = yes
					has_country_flag = africa_first_proof_outward_partner_eligible
					africa_relationship_is_rival_bloc = yes
					africa_relationship_is_leaving = yes
					africa_relationship_is_resistant = yes
				}
			}
		}
		AND = {
			check_variable = { africa_requested_action_id = constant:africa_action.contain_regional_secession_war }
			event_target:africa_action_target = { has_war = yes }
		}
	}
}

# a named inland council, while Ruanda-Urundi may credit its other polity. The
# controller must still be one of that dossier's exact named geographic actors.
africa_first_proof_region_state_evidence_is_live_and_exact = {
	OR = {
		africa_first_proof_state_is_current_playbook_homeland = yes
		AND = {
			event_target:africa_host = { check_variable = { africa_host_playbook = constant:africa_host_playbook.senegal_fwa } }
			OR = { state = 296 state = 556 state = 557 state = 701 state = 778 state = 782 state = 786 state = 898 state = 899 }
			CONTROLLER = { africa_first_proof_country_is_current_playbook_geographic_partner = yes }
		}
		AND = {
			event_target:africa_host = { check_variable = { africa_host_playbook = constant:africa_host_playbook.french_equatorial_africa } }
			OR = { state = 539 state = 773 state = 774 state = 775 }
			CONTROLLER = { africa_first_proof_country_is_current_playbook_geographic_partner = yes }
		}
		AND = {
			event_target:africa_host = { check_variable = { africa_host_playbook = constant:africa_host_playbook.ruanda_urundi } }
			OR = { state = 768 state = 769 }
			CONTROLLER = {
				OR = {
					africa_is_current_host = yes
					AND = {
						africa_first_proof_country_is_current_playbook_geographic_partner = yes
						OR = { original_tag = RWA original_tag = BRD }
					}
				}
			}
		}
	}
}

# COUNTRY scope on a credited corridor partner with ROOT equal to the host.
# Action 32 creates a real capital-to-capital railway; its proof marker remains
# valid only while that route exists. The other country-mode corridor actions
# carry their own delivered convoy, charter, or war-resolution semantics.
africa_first_proof_corridor_actor_route_is_live = {
	OR = {
		NOT = { has_country_flag = africa_first_proof_capital_rail_route }
		AND = {
			has_country_flag = africa_first_proof_capital_rail_route
			has_railway_connection = {
				start_state = ROOT.capital
				target_state = THIS.capital
			}
		}
	}
}
