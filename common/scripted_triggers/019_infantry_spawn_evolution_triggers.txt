# ============================================================================
# CHAOS REDUX - EVENT 019 INFANTRY SPAWN EVOLUTION TRIGGERS
# ============================================================================
# These gates consume a score assembled from several world-state signals.
# They deliberately avoid a single threshold deciding an evolution by itself.
# ============================================================================

infantry_spawn_evolution_i_is_eligible = {
	has_global_flag = infantry_spawn_event_has_fired
	NOT = { has_global_flag = infantry_spawn_evolution_i_active }
	check_variable = { global.infantry_spawn_generation > 0 }
	check_variable = {
		var = infantry_spawn_global_evolution_score
		value = constant:infantry_spawn_evolution_requirement.evolution_i_score
		compare = greater_than_or_equals
	}
}
