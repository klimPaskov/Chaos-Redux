# ============================================================================
# CHAOS REDUX - EVENT 042 EQUIPMENT FROM HEAVENS SCRIPTED TRIGGERS
# ============================================================================
# Recipient, landing-state, evolution-stage, and family-availability
# predicates for Equipment from Heavens. Recipient eligibility is identical
# for players, AI, majors, minors, subjects, and civil-war participants: the
# only gates are the shared special-country and nonhuman classifiers, the test
# fixture, one owned and controlled landing state, and the temporary exclusion
# of a country whose own landing-report chain is still open.
# ============================================================================

# STATE scope. A valid landing site is a passable state that its owner also
# controls.
equipment_from_heavens_state_is_valid_landing = {
	impassable = no
	is_owned_and_controlled_by = OWNER
}

# COUNTRY scope. Ordinary-country classification shared with other systems.
equipment_from_heavens_country_is_ordinary = {
	exists = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_country_flag = chaosx_test_country_initialized }
}

# COUNTRY scope. Full recipient contract. Player status, major status,
# industry, manpower, army, fuel, supply, geography, ideology, war state, and
# prior deliveries are deliberately absent.
equipment_from_heavens_country_is_valid_recipient = {
	equipment_from_heavens_country_is_ordinary = yes
	NOT = { hidden_trigger = { has_country_flag = efh_chain_active } }
	any_owned_state = {
		equipment_from_heavens_state_is_valid_landing = yes
	}
}

# Any scope. True when at least one country can receive a delivery.
equipment_from_heavens_has_any_valid_recipient = {
	any_country = {
		equipment_from_heavens_country_is_valid_recipient = yes
	}
}

# COUNTRY scope. True while this country's landing-report chain is open.
equipment_from_heavens_chain_is_open = {
	has_country_flag = efh_chain_active
}

# Any scope. Evolution stage records. Recorded flags are only ever set after
# the shared enable gate passed, so a disabled stage never reaches them.
equipment_from_heavens_evolution_i_recorded = {
	has_global_flag = efh_evolution_i_recorded
}

equipment_from_heavens_evolution_ii_recorded = {
	has_global_flag = efh_evolution_ii_recorded
}

equipment_from_heavens_evolution_iii_recorded = {
	has_global_flag = efh_evolution_iii_recorded
}

# Any scope. Current disable state of each evolution row, read directly from
# the shared Event Log toggle flag for Event 42 type 42.
equipment_from_heavens_evolution_i_disabled = {
	has_global_flag = events_log_disabled_evolution_42_42_1
}

equipment_from_heavens_evolution_ii_disabled = {
	has_global_flag = events_log_disabled_evolution_42_42_2
}

equipment_from_heavens_evolution_iii_disabled = {
	has_global_flag = events_log_disabled_evolution_42_42_3
}

# Any scope. Ordered active stages. A later stage is active only while every
# earlier stage is recorded and enabled.
equipment_from_heavens_stage_i_active = {
	equipment_from_heavens_evolution_i_recorded = yes
	NOT = { equipment_from_heavens_evolution_i_disabled = yes }
}

equipment_from_heavens_stage_ii_active = {
	equipment_from_heavens_stage_i_active = yes
	equipment_from_heavens_evolution_ii_recorded = yes
	NOT = { equipment_from_heavens_evolution_ii_disabled = yes }
}

equipment_from_heavens_stage_iii_active = {
	equipment_from_heavens_stage_ii_active = yes
	equipment_from_heavens_evolution_iii_recorded = yes
	NOT = { equipment_from_heavens_evolution_iii_disabled = yes }
}

# Any scope. The event is registered, enabled in Event Logs, and not already
# resolving a world-end transition that halts event processing.
equipment_from_heavens_event_is_enabled = {
	NOT = {
		is_in_array = {
			array = global.disabled_events
			value = constant:equipment_from_heavens_event.id
		}
	}
}

# COUNTRY scope. Ordinary nuclear production access, used for the Chaos
# milestone split and the Nuclear Lottery eligibility snapshot.
equipment_from_heavens_has_nuclear_production_access = {
	OR = {
		has_tech = nukes
		is_special_project_completed = sp:sp_nuclear_bomb
	}
}

# COUNTRY scope. Delivery families present in the current manifest.
equipment_from_heavens_manifest_has_armor = {
	has_country_flag = efh_manifest_has_armor
}

equipment_from_heavens_manifest_has_air = {
	has_country_flag = efh_manifest_has_air
}

equipment_from_heavens_manifest_has_logistics = {
	has_country_flag = efh_manifest_has_logistics
}

equipment_from_heavens_manifest_has_infantry = {
	has_country_flag = efh_manifest_has_infantry
}

equipment_from_heavens_manifest_has_nuclear = {
	check_variable = { efh_manifest_nuclear_amount > 0 }
}

equipment_from_heavens_manifest_has_special = {
	check_variable = { efh_manifest_special_entries^num > 0 }
}

# COUNTRY scope. AI assessment enablers read ordinary country state only.
equipment_from_heavens_ai_has_fuel_margin = {
	fuel_ratio > 0.25
}

equipment_from_heavens_ai_has_manpower_margin = {
	has_manpower > 20000
}

equipment_from_heavens_ai_has_air_bases = {
	any_owned_state = {
		is_owned_and_controlled_by = PREV
		air_base > 0
	}
}

equipment_from_heavens_ai_has_naval_bases = {
	any_owned_state = {
		is_owned_and_controlled_by = PREV
		naval_base > 0
	}
}
