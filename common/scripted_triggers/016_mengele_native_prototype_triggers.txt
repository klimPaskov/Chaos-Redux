# Read-only bridge proof. Both slots must exist before either value is read.
# The bridge retains its own strict provider, Theory, and private Prototype gates.
# Slot 7 is schema metadata: existing Cloning family ID 8 minus one, matching
# its family receipt predicate. A literal preserves this trigger's no-write
# contract without introducing unscoped scratch state or an unverified index token.
brilliant_scientist_mengele_native_cloning_fresh_receipt_is_present = {
	check_variable = { var = mengele_event016_native_prototype_source_entries^num value = constant:brilliant_scientist_project_family.cloning compare = greater_than_or_equals }
	check_variable = { var = mengele_event016_native_prototype_mode_entries^num value = constant:brilliant_scientist_project_family.cloning compare = greater_than_or_equals }
	check_variable = { var = mengele_event016_native_prototype_source_entries^7 value = constant:mengele_event016_native_prototype_source.cloning compare = equals }
	check_variable = { var = mengele_event016_native_prototype_mode_entries^7 value = constant:mengele_event016_native_prototype_mode.fresh_output compare = equals }
}

# Country equivalent of the installed sp_air_jet_engine.available.FROM gate.
brilliant_scientist_mengele_native_jet_engine_technology_gate = {
	OR = {
		AND = {
			has_dlc = "By Blood Alone"
			has_tech = engines_2
			OR = {
				has_tech = improved_small_airframe
				has_tech = improved_medium_airframe
			}
		}
		AND = {
			NOT = { has_dlc = "By Blood Alone" }
			OR = {
				has_tech = fighter2
				has_tech = tactical_bomber2
			}
		}
	}
}
