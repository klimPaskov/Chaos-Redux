# ============================================================================
# CHAOS REDUX - EVENT 007 FURY COST DISPLAY TRIGGERS
# ============================================================================
# Read-only quote and per-resource affordability for Event 007 custom-cost
# localisation. The decision owner keeps payment and requirement ownership.
#
# fury_cost_display_quote_component
# Scope: COUNTRY (the cost payer).
# Inputs: positive fury_cost_display_ordinary_cost, the universal-cost family
# mask in fury_cost_display_family_mask (zero means fixed price), and the exact
# universal-cost resource kind in fury_cost_display_resource_kind.
# Outputs: caller-owned fury_cost_display_quoted_amount,
# fury_cost_display_quote_valid, and fury_cost_display_affordable.
# Callers must initialize all three outputs before this trigger.
# Side effects: temporary quote/affordability inputs only; it does not change
# any persistent variable, stockpile, resource, flag, or decision state.
# A family-priced component is freshly quoted through Event 026 whenever that
# sale is active. Failed active-sale quotation preserves the ordinary input as
# an invalid diagnostic amount and marks the component unaffordable.
# ============================================================================
fury_cost_display_quote_component = {
	set_temp_variable = {
		fury_cost_display_quoted_amount = fury_cost_display_ordinary_cost
	}
	set_temp_variable = {
		fury_cost_display_quote_valid = constant:universal_cost_framework.zero
	}
	set_temp_variable = {
		fury_cost_display_affordable = constant:universal_cost_framework.zero
	}
	if = {
		limit = {
			check_variable = {
				var = fury_cost_display_ordinary_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
		}
		if = {
			limit = {
				check_variable = {
					var = fury_cost_display_family_mask
					value = constant:universal_cost_framework.zero
					compare = equals
				}
			}
			set_temp_variable = {
				fury_cost_display_quote_valid = constant:universal_cost_framework.one
			}
		}
		else_if = {
			limit = { NOT = { black_friday_is_active = yes } }
			set_temp_variable = {
				fury_cost_display_quote_valid = constant:universal_cost_framework.one
			}
		}
		else = {
			set_temp_variable = {
				black_friday_quote_ordinary_cost = fury_cost_display_ordinary_cost
			}
			set_temp_variable = {
				black_friday_quote_family_mask = fury_cost_display_family_mask
			}
			# Caller-owned shared quote outputs, initialized before the nested API.
			set_temp_variable = {
				black_friday_quote_final_cost = fury_cost_display_ordinary_cost
			}
			set_temp_variable = {
				black_friday_quote_valid = constant:universal_cost_framework.zero
			}
			set_temp_variable = {
				black_friday_quote_composed_ratio = constant:universal_cost_framework.zero
			}
			set_temp_variable = {
				black_friday_quote_rounding_quantum = constant:universal_cost_framework.default_rounding_quantum
			}
			set_temp_variable = {
				black_friday_quote_requested_quantum = constant:universal_cost_framework.zero
			}
			if = {
				limit = { black_friday_quote_custom_cost = yes }
				set_temp_variable = {
					fury_cost_display_quoted_amount = black_friday_quote_final_cost
				}
				set_temp_variable = {
					fury_cost_display_quote_valid = constant:universal_cost_framework.one
				}
			}
		}
	}
	set_temp_variable = {
		universal_cost_affordability_resource_kind = fury_cost_display_resource_kind
	}
	set_temp_variable = {
		universal_cost_affordability_amount = fury_cost_display_quoted_amount
	}
	if = {
		limit = {
			check_variable = {
				var = fury_cost_display_quote_valid
				value = constant:universal_cost_framework.one
				compare = equals
			}
			universal_cost_resource_kind_is_native = yes
			universal_cost_component_is_affordable = yes
		}
		set_temp_variable = {
			fury_cost_display_affordable = constant:universal_cost_framework.one
		}
	}
	always = yes
}
