# ============================================================================
# CHAOS REDUX - FALLOUT ANIMAL FEED TRIGGERS
# ============================================================================
# The Animal Feed Debate uses one native state food-reserve snapshot. Missing
# state identity, source provenance, ownership, or generation proof refuses the
# candidate rather than selecting a generic agricultural target.
# ============================================================================

fallout_event_pilot_animal_feed_state_is_current = {
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_variable = fallout_pretransition_air_winter_food_reserve
	check_variable = { fallout_pretransition_air_winter_food_reserve > constant:fallout_event_164_threshold.feed_minimum }
	check_variable = { fallout_pretransition_air_winter_food_reserve < constant:fallout_event_164_threshold.feed_pressure }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	owner = {
		check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_164_cost.minimum_food compare = greater_than_or_equals }
	}
	NOT = { has_state_flag = fallout_event_164_feed_registry_committed }
}

fallout_event_164_opening_feed_target_is_current = {
	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.state }
	check_variable = { fallout_event_dispatch_issued_target > constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_164_target_to_check = fallout_event_dispatch_issued_target }
	set_temp_variable = { fallout_event_164_target_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_164_target_to_check = {
		set_temp_variable = { fallout_event_164_target_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		owner = { tag = ROOT }
		fallout_event_pilot_animal_feed_state_is_current = yes
	}
	check_variable = { fallout_event_164_target_context_loaded = constant:fallout_event_runtime.one }
}
