# COUNTRY scope.
africa_world_package_candidate_base_is_valid = {
	exists = yes
	has_capitulated = no
	NOT = { has_country_flag = africa_unifier_host }
	NOT = { has_country_flag = africa_world_package_actor }
	NOT = { has_country_flag = africa_world_package_candidate }
	NOT = { has_country_flag = africa_terminal_world_actor }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_government_in_exile = yes }
	OR = {
		has_focus_tree = generic_focus
		has_country_flag = africa_world_package_focus_replacement_approved
	}
	capital_scope = {
		is_controlled_by = PREV
		NOT = { is_on_continent = africa }
	}
	NOT = { has_global_flag = world_end }
}

# COUNTRY scope on a nominated actor.
africa_world_package_is_candidate = {
	has_country_flag = africa_world_package_candidate
	has_variable = africa_world_continent_id
	check_variable = { africa_world_continent_id > constant:africa_world_continent.none }
	check_variable = {
		var = africa_world_continent_id
		value = constant:africa_world_continent.oceania
		compare = less_than_or_equals
	}
	NOT = { has_country_flag = africa_world_package_installed }
	NOT = { has_global_flag = world_end }
}

# COUNTRY scope on a nominated actor. The long comparison form is used because
# the package registry includes the final Oceania identifier.
africa_world_package_has_valid_identifier = {
	has_variable = africa_world_continent_id
	check_variable = { africa_world_continent_id > constant:africa_world_continent.none }
	check_variable = {
		var = africa_world_continent_id
		value = constant:africa_world_continent.oceania
		compare = less_than_or_equals
	}
}

africa_world_package_is_installed = {
	has_country_flag = africa_world_package_actor
	has_country_flag = africa_world_package_installed
	africa_world_package_has_valid_identifier = yes
	NOT = { has_global_flag = world_end }
}

africa_world_package_is_sovereign_complete = {
	africa_world_package_is_installed = yes
	has_country_flag = africa_world_package_sovereign_complete

	africa_world_package_is_installed = yes

		has_country_flag = africa_world_package_sovereign_complete
		has_country_flag = africa_world_union_partner
		has_country_flag = africa_world_submission_settlement
		has_country_flag = africa_world_package_terminal_absorbed
	}
}

	africa_is_current_host = yes
	has_global_flag = africa_is_one

	check_variable = {

		value = africa_scramble_recognition_target
		compare = greater_than_or_equals
	}
}

africa_scramble_sanctions_requirement_met = {
	OR = {
		check_variable = { africa_scramble_sanction_count = constant:africa_value.zero }
		has_country_flag = africa_anti_sanctions_network_ready
		has_country_flag = africa_scramble_sanctions_lifted
	}
}

africa_scramble_ultimatum_requirement_met = {
	OR = {
		check_variable = { africa_scramble_ultimatum_count = constant:africa_value.zero }
		has_country_flag = africa_foreign_ultimatum_answered
		has_country_flag = africa_scramble_ultimata_withdrawn
	}
}

africa_scramble_intervention_requirement_met = {
	OR = {
		check_variable = { africa_scramble_expedition_count = constant:africa_value.zero }
		AND = {
			has_country_flag = africa_continental_defence_mobilised
			OR = {
				has_country_flag = africa_expedition_plan_disrupted
				has_country_flag = africa_intervention_coalition_withdrawn
				has_country_flag = africa_scramble_armistice_signed
				has_country_flag = africa_scramble_defensive_war_won
			}
		}
	}
}

# COUNTRY scope on another continent actor. ROOT remains the host when called

	africa_world_package_is_installed = yes

	NOT = { has_country_flag = africa_world_package_rival }
	NOT = { has_country_flag = africa_world_submission_settlement }
	NOT = { has_war_with = ROOT }

	check_variable = {
		var = africa_world_package_legitimacy
		value = constant:africa_world_order.minimum_world_union_compatibility
		compare = greater_than_or_equals
	}
}

	africa_world_package_is_installed = yes
	NOT = { has_country_flag = africa_world_union_partner }
	NOT = { has_country_flag = africa_world_submission_settlement }
	NOT = { has_war_with = ROOT }
	OR = {
		has_country_flag = africa_world_package_rival
		has_country_flag = africa_world_order_rivalry_intensified
