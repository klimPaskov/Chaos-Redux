# COUNTRY scope.
africa_world_package_candidate_base_is_valid = {
	exists = yes
	has_capitulated = no
	NOT = { has_country_flag = africa_unifier_host }
	NOT = { has_country_flag = africa_world_package_actor }
	NOT = { has_country_flag = africa_world_package_candidate }
	NOT = { has_country_flag = africa_world_package_successor_candidate }
	NOT = { has_country_flag = africa_terminal_world_actor }
	is_subject = no
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_government_in_exile = yes }
	OR = {
		has_focus_tree = generic_focus
		has_country_flag = africa_world_package_focus_replacement_approved
	}
	capital_scope = {
		is_controlled_by = PREV
		NOT = { is_on_continent = africa }
	}
	NOT = { has_global_flag = world_end }
}

# COUNTRY scope on a nominated actor.
africa_world_package_is_candidate = {
	has_country_flag = africa_world_package_candidate
	has_variable = africa_world_continent_id
	check_variable = { africa_world_continent_id > constant:africa_world_continent.none }
	check_variable = {
		var = africa_world_continent_id
		value = constant:africa_world_continent.oceania
		compare = less_than_or_equals
	}
	NOT = { has_country_flag = africa_world_package_installed }
	NOT = { has_global_flag = world_end }
}

# COUNTRY scope on the Event 12 host. Every required continent has an explicit
# disposition (installed, pending, or documented absent) before this is set.

	check_variable = {

	}

# COUNTRY scope on an installed package actor. This is a real heartland proof,
# not a cosmetic continent flag: the actor must still control the recorded

	check_variable = {
		var = africa_world_package_heartland_state_count
		value = constant:africa_world_roster.minimum_heartland_states
		compare = greater_than_or_equals
	}
}

africa_world_package_constituent_settlement_is_proven = {
	africa_world_package_is_installed = yes
	has_country_flag = africa_world_package_polity_foundation_initialised
	check_variable = {
		var = africa_world_package_constituent_count
		value = constant:africa_world_roster.minimum_constituents
		compare = greater_than_or_equals
	}
	africa_world_package_protocol_pending_is_clear = yes
	OR = {
		AND = {
			check_variable = {
				var = africa_world_package_external_eligible_count
				value = constant:africa_world_package_protocol.sparse_external_cutoff
				compare = greater_than_or_equals
			}
			check_variable = {
				var = africa_world_package_voluntary_external_count
				value = constant:africa_world_package_protocol.voluntary_external_minimum
				compare = greater_than_or_equals
			}
		}
		AND = {
			check_variable = {
				var = africa_world_package_external_eligible_count
				value = constant:africa_world_package_protocol.sparse_external_cutoff
				compare = less_than
			}
			check_variable = {
				var = africa_world_package_voluntary_external_count
				value = constant:africa_value.one
				compare = greater_than_or_equals
			}
		}
	}
}

# COUNTRY scope on an installed package actor. This is the narrow gate used
# by the six constituent decisions and by the package capstone. The roster is
# frozen at installation; no recurring world scan can add a new constituent.

	check_variable = {
		var = africa_world_package_pending_response_count
		value = constant:africa_value.zero

				has_country_flag = africa_world_constituent_counterterms_pending
				has_country_flag = africa_world_constituent_withdrawal_notice
			}
		}
	}
}
