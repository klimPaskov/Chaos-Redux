# Input contract for the deliberately unwired exact-division combat effect.
infantry_spawn_achievement_significant_battle_inputs_are_valid = {
	has_event_target = infantry_spawn_achievement_battle_division
	check_variable = { var = infantry_spawn_achievement_battle_enemy_strength_ratio value = constant:infantry_spawn_achievement_battle.minimum_enemy_strength_ratio compare = greater_than_or_equals }
	check_variable = { var = infantry_spawn_achievement_battle_duration_hours value = constant:infantry_spawn_achievement_battle.minimum_duration_hours compare = greater_than_or_equals }
	check_variable = { var = infantry_spawn_achievement_battle_enemy_casualties value = constant:infantry_spawn_achievement_battle.minimum_enemy_casualties compare = greater_than_or_equals }
}

infantry_spawn_achievement_rail_connection_is_live = {
	has_variable = infantry_spawn_achievement_rail_target_state
	var:infantry_spawn_achievement_rail_target_state = {
		is_controlled_by = ROOT
		ROOT = {
			has_railway_connection = {
				start_state = ROOT.capital
				target_state = PREV
			}
		}
	}
}

infantry_spawn_achievement_scenario_history_is_maximum_crisis = {
	has_country_flag = infantry_spawn_achievement_scenario_launched
	OR = {
		check_variable = { var = infantry_spawn_achievement_scenario_type value = constant:infantry_spawn_scenario.type_general_mutiny compare = equals }
		check_variable = { var = infantry_spawn_achievement_scenario_type value = constant:infantry_spawn_scenario.type_anomalous_rising compare = equals }
	}
	check_variable = { var = infantry_spawn_achievement_scenario_intensity value = constant:infantry_spawn_scenario.maximum_intensity compare = equals }
	check_variable = { var = infantry_spawn_achievement_scenario_start_country_id value = THIS.id compare = equals }
	is_ai = no
	has_capitulated = no
	NOT = { has_global_flag = world_end }
	NOT = { has_country_flag = infantry_spawn_achievement_scenario_world_end_seen }
	NOT = { has_country_flag = infantry_spawn_achievement_scenario_intensity_changed }
	NOT = { has_country_flag = infantry_spawn_achievement_scenario_tag_switched }
}
