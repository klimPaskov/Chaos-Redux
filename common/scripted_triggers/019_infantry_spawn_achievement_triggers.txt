# State scope. No division of any owner may be present before the exact trial
# detachment is created, which also makes one-state opponents safe participants.

# allied formations are excluded by the exact country count and owner check.

	impassable = no
	has_border_war = no
	is_owned_by = ROOT
	is_controlled_by = ROOT

		divisions_in_state = { state = PREV size > constant:infantry_spawn_achievement_value.zero }
