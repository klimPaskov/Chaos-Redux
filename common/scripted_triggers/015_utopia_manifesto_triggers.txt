	is_subject = yes
	num_subjects < constant:utopia_manifesto_candidate_limits.subject_candidate_subject_ceiling

	has_civil_war = no
	utopia_manifesto_candidate_capital_is_secure = yes
	utopia_manifesto_candidate_has_valid_ordinary_overlord = yes

utopia_manifesto_candidate_passes_absolute_automatic_gates = {
	exists = yes
	uses_normal_civilian_systems = yes
	is_major = no
	OR = {
		has_focus_tree = generic_focus
		utopia_manifesto_candidate_has_approved_light_tree = yes
	}
	NOT = { utopia_manifesto_candidate_is_dominant_faction_leader = yes }
	has_capitulated = no
	has_civil_war = no
	has_offensive_war = no
	utopia_manifesto_candidate_capital_is_secure = yes
	NOT = { utopia_manifesto_candidate_has_terminal_identity = yes }
	NOT = { utopia_manifesto_candidate_has_active_event_package = yes }
	NOT = { utopia_manifesto_candidate_has_protected_package = yes }
	NOT = { utopia_manifesto_candidate_has_mature_generic_tree = yes }
	NOT = { has_country_flag = utopia_manifesto_accepted }
	NOT = { has_country_flag = utopia_manifesto_rejected }
	NOT = { has_country_flag = utopia_manifesto_focus_tree_loaded }
	NOT = { num_of_factories > constant:utopia_manifesto_candidate_limits.maximum_total_factories }
	num_subjects < constant:utopia_manifesto_candidate_limits.extensive_subject_count
	num_occupied_states < constant:utopia_manifesto_candidate_limits.extensive_occupation_count
	surrender_progress < constant:utopia_manifesto_candidate_thresholds.maximum_surrender_progress

		is_subject = no
		utopia_manifesto_subject_candidate_is_safe = yes
	}
