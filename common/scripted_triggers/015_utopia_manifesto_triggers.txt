utopia_manifesto_has_valid_target_available = {
	any_country = {
		OR = {
			is_valid_utopia_manifesto_player_target = yes
			is_valid_utopia_manifesto_automatic_target = yes
		}
	}
}

is_valid_utopia_manifesto_target = {
	exists = yes
	uses_normal_civilian_systems = yes
	is_major = no
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = world_end_death }
	NOT = { has_country_flag = utopia_manifesto_accepted }
	NOT = { has_country_flag = utopia_manifesto_rejected }
	NOT = { has_country_flag = utopia_manifesto_focus_tree_loaded }
	NOT = { has_country_flag = fury_actor }
	NOT = { has_country_flag = death_country }
	NOT = { has_country_flag = soviet_collapse_regional_focus_tree_loaded }
	NOT = { num_of_factories > constant:utopia_manifesto_target_gate.max_total_factories }
	NOT = { num_of_military_factories > constant:utopia_manifesto_target_gate.max_military_factories }
	NOT = { num_of_naval_factories > constant:utopia_manifesto_target_gate.max_naval_factories }
	NOT = { num_of_controlled_states > constant:utopia_manifesto_target_gate.max_controlled_states }
	NOT = { num_divisions > constant:utopia_manifesto_target_gate.max_divisions }
	num_of_controlled_states > constant:utopia_manifesto_target_gate.dead_on_arrival_state_floor
	capital_scope = { is_controlled_by = PREV }
}

is_valid_utopia_manifesto_player_target = {
	is_ai = no
	is_valid_utopia_manifesto_target = yes
}

is_valid_utopia_manifesto_automatic_target = {
	is_ai = yes
	is_valid_utopia_manifesto_target = yes
	NOT = { any_enemy_country = { is_ai = no } }
	OR = {
		is_subject = no
		AND = {
			is_subject = yes
			NOT = { has_war = yes }
			has_stability > constant:utopia_manifesto_stability_threshold.low
		}
	}
}
