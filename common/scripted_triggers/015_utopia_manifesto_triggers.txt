	OR = {
		AND = {
			has_war = yes
			num_occupied_states > constant:utopia_manifesto_math.zero
		}
		AND = {
			OR = {
				has_country_flag = utopia_manifesto_district_refugee_municipality
				has_country_flag = utopia_manifesto_emergency_housing_required
			}
			NOT = { has_country_flag = utopia_manifesto_voluntary_migration_corridors }
			NOT = { any_owned_state = { has_state_flag = utopia_manifesto_refugee_municipality } }
		}

# State scope. All roles use engine-visible proxies: population and building
# capacity for settlement pressure, infrastructure/rail/supply buildings for
# transport, and the owner's live Ledger/calling state for the named deficit.
utopia_manifesto_state_is_open_district_candidate = {
	is_owned_by = ROOT
	is_controlled_by = ROOT
	is_core_of = ROOT
	NOT = { has_state_flag = utopia_manifesto_district_built }
	NOT = { has_state_flag = utopia_manifesto_district_project_refused }
	NOT = { has_state_flag = utopia_manifesto_district_build_active }
}

utopia_manifesto_state_is_suitable_for_market_garden = {
	utopia_manifesto_state_is_open_district_candidate = yes
	state_population_k > constant:utopia_manifesto_district_threshold.market_population_floor_k
	infrastructure > constant:utopia_manifesto_district_threshold.viable_infrastructure_floor
	infrastructure < constant:utopia_manifesto_district_threshold.market_infrastructure_ceiling
	ROOT = {
		OR = {
			utopia_manifesto_need_is_high = yes
			NOT = { utopia_manifesto_reserve_is_secure = yes }
			has_country_flag = utopia_manifesto_calling_provisioning_shortage
		}
	}
}

utopia_manifesto_state_is_suitable_for_industrial_housing = {
	utopia_manifesto_state_is_open_district_candidate = yes
	state_population_k > constant:utopia_manifesto_district_threshold.housing_population_floor_k
	infrastructure > constant:utopia_manifesto_district_threshold.viable_infrastructure_floor
	OR = {
		industrial_complex > constant:utopia_manifesto_district_threshold.industry_factory_floor
		arms_factory > constant:utopia_manifesto_district_threshold.industry_factory_floor
	}
	free_building_slots = {
		building = industrial_complex
		size > constant:utopia_manifesto_district_threshold.free_shared_slot_floor
		include_locked = yes
	}
	ROOT = {
		utopia_manifesto_has_live_housing_pressure = yes
	}
}

utopia_manifesto_state_is_suitable_for_rail_junction = {
	utopia_manifesto_state_is_open_district_candidate = yes
	state_population_k > constant:utopia_manifesto_district_threshold.rail_population_floor_k
	infrastructure > constant:utopia_manifesto_district_threshold.viable_infrastructure_floor
	infrastructure < constant:utopia_manifesto_district_threshold.rail_infrastructure_ceiling
	OR = {
		infrastructure < constant:utopia_manifesto_district_threshold.secure_transport_floor
		supply_node > constant:utopia_manifesto_district_threshold.industry_factory_floor
		rail_way > constant:utopia_manifesto_district_threshold.industry_factory_floor
	}
	ROOT = {
		OR = {

# State scope. These three obligations are recorded at project start and then
# re-evaluated from live population, slots, infrastructure, stockpiles, and
# reserves at the terminal mission tick.
utopia_manifesto_district_housing_obligation_met = {
	has_state_flag = utopia_manifesto_district_housing_commitment
	OR = {
		AND = {
			has_state_flag = utopia_manifesto_district_role_market_garden_active
			state_population_k > constant:utopia_manifesto_district_threshold.market_population_floor_k
			infrastructure > constant:utopia_manifesto_district_threshold.viable_infrastructure_floor
		}
		AND = {
			has_state_flag = utopia_manifesto_district_role_industrial_housing_active
			state_population_k > constant:utopia_manifesto_district_threshold.housing_population_floor_k
			free_building_slots = {
				building = industrial_complex
				size > constant:utopia_manifesto_district_threshold.free_shared_slot_floor
				include_locked = yes
			}
		}
		AND = {
			has_state_flag = utopia_manifesto_district_role_rail_junction_active
			state_population_k > constant:utopia_manifesto_district_threshold.rail_population_floor_k
		}
		AND = {
			has_state_flag = utopia_manifesto_district_role_refugee_municipality_active
			state_population_k > constant:utopia_manifesto_district_threshold.refugee_population_floor_k

	has_country_flag = utopia_manifesto_league_initialized
	any_of_scopes = {
		array = utopia_manifesto_league_members
		exists = yes
		uses_normal_civilian_systems = yes
		is_subject = no
		has_capitulated = no
		has_civil_war = no
		OR = {
			ROOT = { NOT = { has_country_flag = utopia_manifesto_league_formal_defense } }
			is_in_faction_with = ROOT
		}
	}
}
