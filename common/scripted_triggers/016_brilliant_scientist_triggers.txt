# ============================================================================
# CHAOS REDUX - EVENT 016 BRILLIANT SCIENTIST TRIGGERS
# ============================================================================
# Overview:
# - Validates Event 016 host, facility, transfer, foreign, formation, temporal,
#   origin, evolution, project-ledger, and terminal inputs.
# - Reads persistent country arrays maintained by the Event 016 effects file.
# - Keeps exact Independent Capacity and Grievance values script-only; only the
#   derived Government Control flags are intended for player-facing output.
# - Never performs all-country work from an on_action. Pool triggers may be
#   evaluated explicitly by the fire-once prefire resolver or a decision.
# ============================================================================

# COUNTRY scope.
brilliant_scientist_is_eligible_host = {
	exists = yes
	has_capitulated = no
	is_subject = no
	capital_scope = { is_controlled_by = ROOT }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_country_flag = brilliant_scientist_host_rejected_permanently }
	NOT = { has_country_flag = brilliant_scientist_host_history }
	NOT = { has_country_flag = brilliant_scientist_former_host }
	NOT = { has_country_flag = brilliant_scientist_kruger_expelled }
	NOT = { has_country_flag = brilliant_scientist_kruger_killed }
	NOT = { has_country_flag = brilliant_scientist_kruger_transferred_away }
	NOT = { has_country_flag = brilliant_scientist_kruger_lost }
	NOT = { has_country_flag = brilliant_scientist_kruger_departed }
	NOT = { has_country_flag = brilliant_scientist_kruger_received_by_transfer }
	NOT = { has_country_flag = brilliant_scientist_kruger_state }
	NOT = { has_global_flag = world_end }
	any_owned_state = {
		count = constant:brilliant_scientist_host_selection.minimum_core_states
		is_owned_and_controlled_by = PREV
		is_core_of = PREV
	}
	any_owned_state = {
		is_owned_and_controlled_by = PREV
		is_core_of = PREV
		brilliant_scientist_is_valid_primary_facility_state = yes
	}
}

# Any scope. This is an explicit availability query, not a periodic on_action.
brilliant_scientist_has_any_eligible_host = {
	any_country = {
		brilliant_scientist_is_eligible_host = yes
	}
}

# Any scope. The shared fire-once dispatcher additionally checks its disabled
# and already-fired arrays; this helper owns Event 016-specific availability.
brilliant_scientist_automatic_event_is_available = {
	NOT = { has_global_flag = brilliant_scientist_event_resolved }
	NOT = { has_global_flag = world_end }
	NOT = {
		any_country = {
			has_character = KRG_warren_kruger
		}
	}
	brilliant_scientist_has_any_eligible_host = yes
}

# COUNTRY scope.
brilliant_scientist_is_current_host = {
	has_country_flag = brilliant_scientist_current_host
	has_character = KRG_warren_kruger
}

# COUNTRY scope. Event 060 may be salvaged once by the active host. The gate
# keeps the choice tied to the live Directorate rather than a stale character
# receipt or a country that merely met Kruger in an earlier chain.
brilliant_scientist_can_prevent_research_failure = {
	brilliant_scientist_is_current_host = yes
	NOT = { has_global_flag = world_end }
	NOT = { has_country_flag = brilliant_scientist_research_failure_prevented }
	check_variable = { var = political_power value = constant:brilliant_scientist_research_failure_salvage.political_power_gate compare = greater_than }
}

	}
