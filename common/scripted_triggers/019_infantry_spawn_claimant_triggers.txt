infantry_spawn_selected_claimant_index_is_valid = {
	has_variable = infantry_spawn_selected_claimant_index
	check_variable = { infantry_spawn_selected_claimant_index > constant:infantry_spawn_event.invalid_index }
	check_variable = { var = infantry_spawn_selected_claimant_index value = infantry_spawn_claimant_uid_entries^num compare = less_than }
	check_variable = {
		var = infantry_spawn_claimant_status_entries^infantry_spawn_selected_claimant_index
		value = constant:infantry_spawn_claimant_status.retired
		compare = less_than
	}
}

# A country can expose several claimant rows, but only the immutable UID that
# opened the current demand may resolve it. The selected row remains a movable
# UI cursor and is never accepted as demand ownership by itself.
infantry_spawn_selected_claimant_owns_pending_demand = {
	has_country_flag = infantry_spawn_claimant_demand_pending
	has_variable = infantry_spawn_claimant_demand_owner_uid
	infantry_spawn_selected_claimant_index_is_valid = yes
	check_variable = {
		var = infantry_spawn_claimant_uid_entries^infantry_spawn_selected_claimant_index
		value = infantry_spawn_claimant_demand_owner_uid
		compare = equals
	}
	check_variable = {
		var = infantry_spawn_claimant_demand_entries^infantry_spawn_selected_claimant_index
		value = constant:infantry_spawn_claimant_demand.none
		compare = greater_than
	}
}

# Event 67 stores the receiving country ID in global.generalissimo_country and
# the selected character ID in global.generalissimo. The country and character
# checks together reject stale globals, dead or retired characters, and leaders
# who no longer belong to the recorded country.
infantry_spawn_country_has_valid_event67_generalissimo = {
	has_variable = global.generalissimo_country
	has_variable = global.generalissimo
	check_variable = {
		var = global.generalissimo_country
		value = THIS.id
		compare = equals
	}
	any_unit_leader = {
		meta_trigger = {
			text = { has_id = [GENERALISSIMO_ID] }
			GENERALISSIMO_ID = "[?global.generalissimo|.0]"
		}
		is_army_leader = yes
	}
}

infantry_spawn_selected_claimant_is_event67_generalissimo = {
	infantry_spawn_country_has_valid_event67_generalissimo = yes
	infantry_spawn_selected_claimant_index_is_valid = yes
	check_variable = {
		var = infantry_spawn_claimant_uid_entries^infantry_spawn_selected_claimant_index
		value = global.generalissimo
		compare = equals
	}
}

infantry_spawn_selected_claimant_rivals_event67_generalissimo = {
