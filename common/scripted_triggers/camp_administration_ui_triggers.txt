# Administration command eligibility. Display, decisions and execution share these gates.
camp_admin_ui_action_two_visible = {
	OR = { camp_admin_country_action_2_visible = yes camp_admin_generic_reform_visible = yes }
}
camp_admin_ui_action_two_available = {
	OR = {
		AND = { camp_admin_country_action_2_visible = yes camp_admin_country_action_2_available = yes }
		AND = { NOT = { camp_admin_country_action_2_visible = yes } camp_admin_generic_reform_available = yes }
	}
}
camp_admin_has_open_picker = {
	OR = {
		has_country_flag = camp_admin_picker_priority
		has_country_flag = camp_admin_picker_budget
		has_country_flag = camp_admin_picker_mandate
		has_country_flag = camp_admin_picker_role
		has_country_flag = camp_admin_picker_filter
	}
}
camp_admin_can_set_priority = {
	exists = yes
	has_country_flag = camp_admin_initialized
	camp_rework_country_is_eligible = yes
	NOT = { has_country_flag = camp_admin_reforming }
	OR = {
		check_variable = { camp_admin_control_requested = constant:camp_admin_priority.industry }
		check_variable = { camp_admin_control_requested = constant:camp_admin_priority.works }
		check_variable = { camp_admin_control_requested = constant:camp_admin_priority.extraction }
	}
	NOT = { check_variable = { camp_admin_priority = camp_admin_control_requested } }
	has_political_power > camp_admin_policy_quote_floor
}
camp_admin_can_set_budget = {
	exists = yes
	has_country_flag = camp_admin_initialized
	camp_rework_country_is_eligible = yes
	OR = {
		check_variable = { camp_admin_control_requested = constant:camp_admin_budget.maintenance }
		check_variable = { camp_admin_control_requested = constant:camp_admin_budget.development }
		check_variable = { camp_admin_control_requested = constant:camp_admin_budget.priority }
	}
	OR = {
		AND = {
			check_variable = { camp_admin_budget = camp_admin_control_requested }
			has_country_flag = camp_admin_development_paused
		}
		AND = {
			NOT = { check_variable = { camp_admin_budget = camp_admin_control_requested } }
			has_political_power > camp_admin_policy_quote_floor
		}
	}
}
camp_admin_can_set_mandate = {
	exists = yes
	has_country_flag = camp_admin_initialized
	camp_rework_country_is_eligible = yes
	NOT = { has_country_flag = camp_admin_reforming }
	OR = {
		check_variable = { camp_admin_control_requested = constant:camp_admin_mandate.custody }
		check_variable = { camp_admin_control_requested = constant:camp_admin_mandate.coercive }
		check_variable = { camp_admin_control_requested = constant:camp_admin_mandate.radical }
	}
	NOT = { check_variable = { camp_admin_mandate = camp_admin_control_requested } }
	has_political_power > camp_admin_mandate_quote_floor
	OR = {
		NOT = { check_variable = { camp_admin_control_requested = constant:camp_admin_mandate.radical } }
		camp_admin_can_use_radical_mandate = yes
	}
}

camp_admin_killing_role_offer_visible = {
	camp_admin_can_use_radical_mandate = yes
	OR = { camp_admin_country_nazi_authority = yes camp_rework_country_can_use_radicalized_route = yes }
}
camp_admin_killing_role_mandate_ready = {
	check_variable = { camp_admin_mandate = constant:camp_admin_mandate.radical }
}
camp_admin_killing_role_existing = {
	has_variable = camp_selected_state_id
	var:camp_selected_state_id = {
		OR = {
			check_variable = { camp_admin_role = constant:camp_admin_role.killing }
			extermination_camp > 0
		}
	}
}
camp_admin_killing_role_level_ready = {
	has_variable = camp_selected_state_id
	var:camp_selected_state_id = { concentration_camp > 0 }
}
camp_admin_killing_role_funds_ready = {
	has_political_power > camp_admin_role_quote_floor
}
camp_admin_can_change_selected_role = {
	camp_admin_has_selected_proven_operating_site = yes
	has_political_power > camp_admin_role_quote_floor
	check_variable = { var = camp_admin_control_requested value = constant:camp_admin_role.custody compare = greater_than_or_equals }
	check_variable = { var = camp_admin_control_requested value = constant:camp_admin_role.killing compare = less_than_or_equals }
	var:camp_selected_state_id = {
		NOT = { check_variable = { camp_admin_role = camp_admin_control_requested } }
		NOT = { extermination_camp > 0 }
	}
	OR = {
		check_variable = { camp_admin_control_requested = constant:camp_admin_role.custody }
		AND = {
			check_variable = { camp_admin_control_requested = constant:camp_admin_role.labor }
			check_variable = { var = camp_admin_mandate value = constant:camp_admin_mandate.coercive compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { camp_admin_control_requested = constant:camp_admin_role.penal }
			original_tag = SOV
			has_government = communism
		}
		AND = {
			check_variable = { camp_admin_control_requested = constant:camp_admin_role.killing }
			camp_admin_can_use_radical_mandate = yes
			check_variable = { camp_admin_mandate = constant:camp_admin_mandate.radical }
			OR = {
				camp_admin_country_nazi_authority = yes
				camp_rework_country_can_use_radicalized_route = yes
			}
			var:camp_selected_state_id = {
				concentration_camp > 0
				check_variable = { building_level@extermination_camp < constant:genocide_camp_level.maximum }
			}
		}
	}
}

camp_admin_gui_host_records_uncertain = {
	OR = {
		has_state_flag = camp_origin_ambiguous
		has_state_flag = camp_origin_records_unresolved
		has_state_flag = camp_origin_baseline_unverified
		has_state_flag = camp_origin_transaction_unresolved
		has_state_flag = camp_origin_counter_regressed
		has_state_flag = camp_origin_single_origin_reconciliation_pending
		AND = {
			# Exact aligned zero-live history has no held claimant for a later
			# unrelated territorial loss. Sticky reasons above remain visible.
			NOT = {
				AND = {
					camp_admin_known_site_ready_for_reform = yes
					check_variable = { camp_admin_detainees_k = constant:camp_admin_value.zero }
					check_variable = { camp_occ_remaining_k = constant:camp_admin_value.zero }
				}
			}
			OR = {
				check_variable = { camp_admin_detainees_k > constant:camp_admin_value.zero }
				check_variable = { camp_occ_remaining_k > constant:camp_admin_value.zero }
				check_variable = { camp_admin_row_count > constant:camp_admin_value.zero }
			}
			NOT = { camp_origin_host_provenance_ready = yes }
		}
	}
}
