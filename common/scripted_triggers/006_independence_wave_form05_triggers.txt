# ============================================================================
# CHAOS REDUX - EVENT 006 FORM-05 MEDITERRANEAN ISLAND LEAGUE TRIGGERS
# ============================================================================
# Overview:
# - Restricts the founding carrier to the live COR, ARX, or ASX Event 006
#   package while admitting future compatible FORM-05 island packages.
# - Proves sovereign consent, a usable member port, and a bilateral diplomatic
#   connection without scanning every country or transferring member states.
# - Gates every visible surface behind the dedicated FORM-05 asset attestation.
# ============================================================================

is_independence_wave_form05_member_candidate = {
	is_independence_wave_active_country = yes
	is_subject = no
	has_country_flag = independence_wave_formable_family_registered
	has_country_flag = independence_wave_formable_family_selected
	check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.mediterranean_island_league }
	has_variable = independence_wave_anchor_state
	var:independence_wave_anchor_state = {
		is_owned_and_controlled_by = ROOT
		naval_base > constant:independence_wave_formable_tuning.no_entries
	}
}

has_independence_wave_form05_light_convoy_capacity = {
	has_equipment = { convoy > constant:independence_wave_decision_cost.convoy_light }
}

has_independence_wave_form05_standard_convoy_capacity = {
	has_equipment = { convoy > constant:independence_wave_decision_cost.convoy_standard }
}

has_independence_wave_form05_light_factory_capacity = {
	num_of_civilian_factories_available_for_projects > constant:independence_wave_form05_decision_cost.minimum_available_factory_threshold
}

# FORM-05 separates the shared material cost from its island-scaled factory
# assignment. The shared administration trigger otherwise demands two spare
# factories before a one-factory project may begin.
can_pay_independence_wave_form05_administration_light_cost = {
	command_power > constant:independence_wave_decision_cost.command_power_light
	has_manpower > constant:independence_wave_decision_cost.manpower_light
}

# Opening, customs, and proclamation each debit both the light administrative
# palette and the standard diplomatic palette. The combined gate proves the
# full Command Power total before either helper can spend it.
can_pay_independence_wave_form05_combined_administrative_diplomatic_cost = {
	command_power > constant:independence_wave_form05_decision_cost.combined_command_power
	has_manpower > constant:independence_wave_decision_cost.manpower_light
	has_independence_wave_form05_standard_convoy_capacity = yes
}

# The shared strategic trigger carries a multi-factory gate. FORM-05 retains the
# full strategic stability, war-support, Command Power, and convoy expenditure
# while its decision supplies the one-factory island assignment separately.
can_pay_independence_wave_form05_strategic_cost = {
	has_stability > constant:independence_wave_decision_cost.stability_standard
	has_war_support > constant:independence_wave_decision_cost.war_support_minor
	can_pay_independence_wave_diplomatic_standard_cost = yes
	has_independence_wave_form05_standard_convoy_capacity = yes
}

can_pay_independence_wave_form05_proclamation_cost = {
	has_stability > constant:independence_wave_decision_cost.stability_standard
	has_war_support > constant:independence_wave_decision_cost.war_support_minor
	can_pay_independence_wave_form05_combined_administrative_diplomatic_cost = yes
}

# Country scope. Before proclamation the MIX identity must still be free. Once
# the league exists, the exact carrier and its recorded sovereign members keep
# access through their live, diplomatically connected post-formation carrier.
# This avoids a mutable "assets ready" switch while keeping every visible
# FORM-05 surface tied to the same audited family-readiness contract installed
# by the package adapter.

	has_country_flag = independence_wave_form05_autonomous_member

		AND = {
			has_country_flag = independence_wave_form05_member_of_cor_carrier

				is_independence_wave_form05_post_formation_carrier = yes
				has_independence_wave_form05_maritime_connection_to_root = yes
			}
		}
		AND = {
			has_country_flag = independence_wave_form05_member_of_arx_carrier
			ARX = {
				is_independence_wave_form05_post_formation_carrier = yes
				has_independence_wave_form05_maritime_connection_to_root = yes

			has_country_flag = independence_wave_form05_member_of_asx_carrier
