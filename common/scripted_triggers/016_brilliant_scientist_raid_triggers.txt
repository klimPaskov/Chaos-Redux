# ============================================================================
# CHAOS REDUX - EVENT 016 RAID TRIGGERS
# ============================================================================
# Native raid integration for countries holding Kruger's raid technologies and
# for Warren Kruger's active host and sovereign state.
# Existing biological raids retain their native payload resolver. These triggers
# add the Kruger-directed AI preference and make the portal raid a direct
# weaponization payoff: the hidden weaponization technology and its
# already-created Portal Raider template are enough to use it. Kruger's active
# presence changes AI willingness; it is not a player-access gate.
# ============================================================================

# COUNTRY scope. Kruger actively directs this country and therefore prefers
# native biological operations more often. This never authorizes a raid by
# itself; native policy, target, payload, staging, aircraft or formation, and
# outcome checks still have to pass.
brilliant_scientist_kruger_raid_ai_is_active = {
	OR = {
		brilliant_scientist_is_current_host = yes
		brilliant_scientist_is_active_kruger_state = yes
	}
	has_character = KRG_warren_kruger
	KRG_warren_kruger = {
		has_character_flag = brilliant_scientist_personal_active
	}
}

# COUNTRY scope. The native project-force rebuild creates the locked formation
# template when the operational bridge is granted. The weaponization technology
# then authorizes the raid for any country that holds both pieces. No separate
# network, terminal, authority, or staging decision is required.
brilliant_scientist_portal_raid_actor_is_ready = {
	has_tech = brilliant_scientist_portal_warfare_weaponization_tech
	has_template = "Quantum Transit Raiders"
}

# COUNTRY/raid target context. State targets cover state-level factories,
# reactors, and rocket sites. Provincial special-project facilities use the
# dedicated facility target below so removal always addresses the exact building
# province selected by the native raid UI.
brilliant_scientist_portal_raid_target_is_valid = {
	FROM = {
		exists = yes
		has_war_with = ROOT
	}
	var:target_state = {
		is_owned_by = FROM
		is_controlled_by = FROM
		OR = {
			industrial_complex > 0
			arms_factory > 0
			dockyard > 0
			AND = {
				rocket_site > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT free_building_slots = { building = rocket_site size > 0 } } }
			}
			AND = {
				nuclear_reactor > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT free_building_slots = { building = nuclear_reactor size > 0 } } }
			}
			AND = {
				commercial_nuclear_reactor > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT free_building_slots = { building = commercial_nuclear_reactor size > 0 } } }
			}
		}
	}
}

# COUNTRY/raid target context. The building-target raid supplies the exact
# facility province through var:target_province. Destination checks happen by
# facility family so the source is never removed unless it can be rebuilt.
brilliant_scientist_portal_facility_raid_target_is_valid = {
	FROM = {
		exists = yes
		has_war_with = ROOT
	}
	var:target_state = {
		is_owned_by = FROM
		is_controlled_by = FROM
		OR = {
			AND = {
				land_facility > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT land_facility < 1 air_facility < 1 naval_facility < 1 nuclear_facility < 1 } }
			}
			AND = {
				air_facility > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT land_facility < 1 air_facility < 1 naval_facility < 1 nuclear_facility < 1 } }
			}
			AND = {
				naval_facility > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT is_coastal = yes land_facility < 1 air_facility < 1 naval_facility < 1 nuclear_facility < 1 } }
			}
			AND = {
				nuclear_facility > 0
				ROOT = { any_owned_state = { is_controlled_by = ROOT land_facility < 1 air_facility < 1 naval_facility < 1 nuclear_facility < 1 } }
			}
		}
	}
	var:target_province = {
		OR = {
			land_facility > 0
			air_facility > 0
			naval_facility > 0
			nuclear_facility > 0
		}
	}
}
