# ============================================================================
# CHAOS REDUX - EVENT 029 RICHES FOUND ACHIEVEMENT TRIGGERS
# ============================================================================
# Historical, route-aware predicates for the seven Event 029 achievements.
# These triggers read state-persistent flags and bounded date/count values;
# they never infer a historical route from a mine's final presentation alone.
# ============================================================================

riches_found_achievement_player_eligible = {
	is_ai = no
	NOT = { has_country_flag = chaosx_test_country_initialized }
}

riches_found_mine_is_open_or_regulated = {
	riches_found_mine_is_registered = yes
	OR = {
		check_variable = { var = riches_found_operating_state value = constant:riches_found_operating_state.open_expanding compare = equals }
		check_variable = { var = riches_found_operating_state value = constant:riches_found_operating_state.open_regulated compare = equals }
		check_variable = { var = riches_found_operating_state value = constant:riches_found_operating_state.open_concession compare = equals }
		check_variable = { var = riches_found_operating_state value = constant:riches_found_operating_state.open_military compare = equals }
		check_variable = { var = riches_found_operating_state value = constant:riches_found_operating_state.open_predatory compare = equals }
	}
	check_variable = { var = riches_found_mine_contribution_score value = constant:riches_found_value.zero compare = greater_than }
}

riches_found_achievement_original_discoverer_is_controller = {
	has_variable = riches_found_discoverer
	CONTROLLER = { exists = yes }
	check_variable = { var = riches_found_discoverer value = CONTROLLER compare = equals }
}

riches_found_achievement_account_closure_route = {
	riches_found_mine_is_registered = yes
	has_state_flag = riches_found_achievement_account_history
	has_state_flag = riches_found_account_closure_finalized
	OR = {
		AND = {
			has_state_flag = riches_found_mine_permanent_seal
			has_state_flag = riches_found_achievement_permanent_seal_recorded
		}
		has_state_flag = riches_found_account_bound_terms_closure
	}
	OR = {
		has_state_flag = riches_found_achievement_evacuation_recorded
		NOT = { has_variable = riches_found_harm_total_deaths }
		check_variable = { var = riches_found_harm_total_deaths value = constant:riches_found_achievement_limit.account_containment_death_ceiling compare = less_than_or_equals }
	}
	NOT = { has_state_flag = riches_found_account_predatory_prosperity }
	NOT = { has_state_flag = riches_found_mine_collapsed }
	NOT = { has_state_flag = riches_found_achievement_mass_death_recorded }
	NOT = { has_state_flag = riches_found_account_breach_imminent }
	check_variable = { var = riches_found_supernatural_pressure value = constant:riches_found_value.zero compare = equals }
	CONTROLLER = {
		exists = yes
		is_ai = no
		has_stability > constant:riches_found_achievement_threshold.account_stability_floor
		check_variable = { var = PREV.riches_found_account_closure_controller value = THIS compare = equals }
	}
	check_variable = { var = global.date value = riches_found_achievement_account_deadline_date compare = greater_than_or_equals }
}
