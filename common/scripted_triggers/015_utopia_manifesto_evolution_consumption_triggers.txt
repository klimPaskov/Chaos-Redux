utopia_manifesto_can_start_evolution_policy_action = {
	has_country_flag = utopia_manifesto_accepted
	NOT = { has_country_flag = utopia_manifesto_kernel_disabled }
	NOT = { has_country_flag = utopia_manifesto_evolution_policy_action_active }
	NOT = { utopia_manifesto_has_active_constitutional_crisis = yes }
}

	has_variable = utopia_manifesto_evolution_policy_action
	NOT = { utopia_manifesto_has_active_constitutional_crisis = yes }

		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.glosses_open_schools compare = equals }
			has_country_flag = utopia_manifesto_interpretation_schools_open
			utopia_manifesto_glosses_delivery_is_enabled = yes
			utopia_manifesto_has_resolved_route = yes
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.glosses_chartered_schools compare = equals }

			utopia_manifesto_glosses_delivery_is_enabled = yes
			utopia_manifesto_has_resolved_route = yes
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.glosses_censored_margins compare = equals }
			has_country_flag = utopia_manifesto_conduct_margins_censored
			utopia_manifesto_glosses_delivery_is_enabled = yes
			utopia_manifesto_has_resolved_route = yes
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.shores_domestic_alternative compare = equals }
			has_country_flag = utopia_manifesto_necessary_shores_domestic_alternative
			utopia_manifesto_necessary_shores_delivery_is_enabled = yes
			utopia_manifesto_necessary_ground_decisions_are_unlocked = yes
			has_country_flag = utopia_manifesto_need_case_active
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.shores_peaceful_priority compare = equals }
			has_country_flag = utopia_manifesto_necessary_shores_peaceful_priority
			utopia_manifesto_necessary_shores_delivery_is_enabled = yes
			utopia_manifesto_necessary_ground_decisions_are_unlocked = yes
			has_country_flag = utopia_manifesto_need_case_active
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.shores_emergency_powers compare = equals }
			has_country_flag = utopia_manifesto_necessary_shores_emergency_powers
			utopia_manifesto_necessary_shores_delivery_is_enabled = yes
			utopia_manifesto_necessary_ground_decisions_are_unlocked = yes
			has_country_flag = utopia_manifesto_need_case_active
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.cities_local_charters compare = equals }
			has_country_flag = utopia_manifesto_cities_local_charters
			utopia_manifesto_cities_delivery_is_enabled = yes

		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.cities_linked_standard compare = equals }
			has_country_flag = utopia_manifesto_cities_linked_standard
			utopia_manifesto_cities_delivery_is_enabled = yes
			utopia_manifesto_district_decisions_are_unlocked = yes
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.cities_foreign_circle compare = equals }
			has_country_flag = utopia_manifesto_cities_foreign_circle
			utopia_manifesto_cities_delivery_is_enabled = yes
			utopia_manifesto_district_decisions_are_unlocked = yes
			utopia_manifesto_league_decisions_are_unlocked = yes
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.nowhere_plural_law compare = equals }
			has_country_flag = utopia_manifesto_nowhere_law_plural
			utopia_manifesto_nowhere_law_delivery_is_enabled = yes
			utopia_manifesto_league_decisions_are_unlocked = yes
		}
		AND = {
			check_variable = { var = utopia_manifesto_evolution_policy_action value = constant:utopia_manifesto_evolution_choice.nowhere_sponsored_law compare = equals }
			has_country_flag = utopia_manifesto_nowhere_law_sponsored

utopia_manifesto_ai_prefers_open_schools_evolution = {
	OR = {
		utopia_manifesto_is_consent_of_households_route = yes
		utopia_manifesto_is_joke_understood_route = yes
		utopia_manifesto_ai_prefers_consent_route = yes
		utopia_manifesto_ai_prefers_joke_understood_route = yes
		has_country_flag = utopia_manifesto_public_debate_proven
	}
}

utopia_manifesto_ai_prefers_chartered_schools_evolution = {
	OR = {
		utopia_manifesto_is_common_table_route = yes
		utopia_manifesto_is_guardians_of_measure_route = yes
		utopia_manifesto_ai_prefers_common_table_route = yes
		utopia_manifesto_ai_prefers_guardians_route = yes
	}
}

utopia_manifesto_ai_prefers_censored_margins_evolution = {
	OR = {
		utopia_manifesto_is_closed_island_route = yes
		utopia_manifesto_ai_prefers_closed_island_route = yes
		has_country_flag = utopia_manifesto_conduct_censorship
		AND = { has_war = yes NOT = { utopia_manifesto_concord_is_cooperative = yes } }
		num_occupied_states > constant:utopia_manifesto_math.zero
	}
}

utopia_manifesto_ai_prefers_domestic_alternative_evolution = {
	OR = {
		utopia_manifesto_need_is_high = yes
		utopia_manifesto_country_landlocked = yes
		NOT = { utopia_manifesto_plenty_is_stable = yes }
	}
}

utopia_manifesto_ai_prefers_peaceful_priority_evolution = {
	OR = {
		utopia_manifesto_is_consent_of_households_route = yes
		utopia_manifesto_is_joke_understood_route = yes
		AND = { has_war = no utopia_manifesto_concord_is_cooperative = yes }
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.purchase }
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.lease }
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.settlement_agreement }
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.joint_administration }
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.association }
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.autonomy_pact }
	}
}

utopia_manifesto_ai_prefers_emergency_powers_evolution = {
	OR = {
		utopia_manifesto_is_closed_island_route = yes
		AND = { has_war = yes utopia_manifesto_need_is_high = yes }
		has_country_flag = utopia_manifesto_case_hardline_review
		has_country_flag = utopia_manifesto_conduct_skipped_peaceful_case_step
		check_variable = { utopia_manifesto_case_method = constant:utopia_manifesto_case_method.ultimatum }
		num_occupied_states > constant:utopia_manifesto_math.zero
	}
}

utopia_manifesto_ai_prefers_local_charters_evolution = {

		utopia_manifesto_is_consent_of_households_route = yes
		utopia_manifesto_is_common_table_route = yes
		AND = { is_in_faction = no utopia_manifesto_concord_is_cooperative = yes }
	}
}

utopia_manifesto_ai_prefers_linked_standard_evolution = {

utopia_manifesto_ai_prefers_plural_law_evolution = {

		utopia_manifesto_is_consent_of_households_route = yes
		utopia_manifesto_is_joke_understood_route = yes

		has_country_flag = utopia_manifesto_partner_autonomy_retained
	}

utopia_manifesto_ai_prefers_sponsored_law_evolution = {
	OR = {
		has_country_flag = utopia_manifesto_league_sponsorship_accepted
		AND = { is_in_faction = yes utopia_manifesto_is_common_table_route = yes }
		AND = { is_in_faction = yes utopia_manifesto_is_guardians_of_measure_route = yes }
		has_country_flag = utopia_manifesto_conduct_league_aid_conditioned
		check_variable = { var = utopia_manifesto_league_cohesion value = constant:utopia_manifesto_league_limits.stable_cohesion compare = less_than }

		utopia_manifesto_is_common_table_route = yes
		utopia_manifesto_country_landlocked = yes

utopia_manifesto_ai_prefers_closed_order_evolution = {
	utopia_manifesto_is_closed_island_route = yes
	OR = {
		utopia_manifesto_ai_prefers_closed_island_route = yes
		has_war = yes
		NOT = { utopia_manifesto_concord_is_cooperative = yes }
		num_occupied_states > constant:utopia_manifesto_math.zero
		surrender_progress > constant:utopia_manifesto_route_ai_threshold.severe_surrender_pressure
	}
