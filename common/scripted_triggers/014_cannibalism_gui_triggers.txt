# ============================================================================
# CHAOS REDUX - EVENT 014 SCRIPTED GUI FILTERS
# ============================================================================
# These filters run only while a human player refreshes the Evolution II
# network ledger. ROOT is the viewing country and THIS is the candidate scope.
# No event target is read by the scripted GUI.
# ============================================================================

cannibalism_gui_country_matches_selected_tab = {
	OR = {
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.domestic_cells } }
			tag = ROOT
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.islands_and_communes } }
			is_cannibalism_warlord_country = yes
			has_country_flag = cannibalism_origin_island_host
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.foreign_links } }
			NOT = { tag = ROOT }
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.warlord_states } }
			is_cannibalism_warlord_country = yes
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.countermeasures } }
			NOT = { tag = ROOT }
			OR = {
				is_cannibalism_warlord_country = yes
				is_cannibalism_unified_country = yes
				is_cannibalism_wendigo_hannibal_country = yes
			}
		}
	}
}

cannibalism_gui_state_matches_selected_tab = {
	OR = {
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.domestic_cells } }
			OR = { OWNER = { tag = ROOT } CONTROLLER = { tag = ROOT } }
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.islands_and_communes } }
			OR = {
				check_variable = { cannibalism_node_type = constant:cannibalism_node_type.island_commune }
				has_state_flag = cannibalism_commune_state
			}
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.foreign_links } }
			NOT = { OWNER = { tag = ROOT } }
			NOT = { CONTROLLER = { tag = ROOT } }
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.warlord_states } }
			OR = {
				OWNER = { is_cannibalism_warlord_country = yes }
				CONTROLLER = { is_cannibalism_warlord_country = yes }
			}
		}
		AND = {
			ROOT = { check_variable = { cannibalism_network_gui_tab = constant:cannibalism_gui_tab.countermeasures } }
			NOT = { CONTROLLER = { tag = ROOT } }
		}
	}
}
