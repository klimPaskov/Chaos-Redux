# STATE scope; ROOT is the responding country.
black_plague_shared_state_target_is_usable = {
	black_plague_response_state_can_accept_actions = yes
	NOT = { has_state_flag = black_plague_shared_action_active }
	NOT = { has_state_flag = black_plague_shared_action_cooldown }
}

black_plague_shared_state_is_prepared_or_exposed = {
	OR = {
		black_plague_state_is_threatened = yes
		black_plague_state_is_incubating = yes
	}
}

black_plague_shared_state_is_infected_or_worse = {
	OR = {
		black_plague_state_is_infected = yes
		black_plague_state_is_severe = yes
		black_plague_state_is_collapsed = yes
		black_plague_state_is_contained = yes
		black_plague_state_is_recovering = yes
	}
}

black_plague_shared_state_is_contained_or_recovery = {
	OR = {
		black_plague_state_is_contained = yes
		black_plague_state_is_recovering = yes
	}
}

black_plague_shared_state_is_recovery = {
	black_plague_state_is_recovering = yes
}

black_plague_shared_state_is_cured_or_recovery = {
	OR = {
		black_plague_state_is_cured = yes
		black_plague_state_is_recovering = yes
	}
}

black_plague_shared_state_is_coastal_hub = {
	OR = {
		AND = { is_coastal = yes naval_base > 0 }
		black_plague_state_has_strategic_transport = yes
		supply_node > constant:black_plague_response_threshold.response_access_hub_floor
	}
}

black_plague_shared_state_has_military_route = {
	OR = {
		black_plague_state_has_army_presence = yes
		any_neighbor_state = {
			is_controlled_by = ROOT
			black_plague_state_has_army_presence = yes
		}
	}
}

black_plague_shared_state_can_start_action = {
	black_plague_shared_state_target_is_usable = yes
	black_plague_response_state_has_access = yes
	black_plague_shared_state_can_pay_scaled_materials = yes
	ROOT = { black_plague_country_can_direct_response = yes }
}

# The common floor prevents an action from draining an unavailable stockpile;
# the action effect applies the population scale before payment.
black_plague_shared_country_can_pay_standard_materials = {
	has_equipment = { support_equipment > constant:black_plague_shared_cost.standard_support_equipment }
	has_equipment = { motorized_equipment > constant:black_plague_shared_cost.standard_motorized_equipment }
	has_fuel > constant:black_plague_shared_cost.standard_fuel
}

black_plague_shared_country_can_pay_country_materials = {
	has_equipment = { support_equipment > constant:black_plague_shared_cost.country_support_equipment }
	has_equipment = { motorized_equipment > constant:black_plague_shared_cost.country_motorized_equipment }
	has_fuel > constant:black_plague_shared_cost.country_fuel
}

black_plague_shared_country_can_pay_royal_node = {
	black_plague_shared_country_can_pay_country_materials = yes
	has_equipment = { infantry_equipment > constant:black_plague_shared_cost.country_infantry_equipment }
	has_manpower > constant:black_plague_shared_cost.country_manpower
	command_power > constant:black_plague_shared_cost.country_command_power
}

# STATE scope; ROOT is the responding country. The begin effect scales the
# standard payment by population, so the target gate mirrors all four bands.
black_plague_shared_state_can_pay_scaled_materials = {
	OR = {
		AND = {
			check_variable = { var = state_population_k value = constant:black_plague_shared_scale.large_population_k compare = less_than }
			ROOT = {
				has_equipment = { support_equipment > constant:black_plague_shared_cost.scaled_min_support_equipment }
				has_equipment = { motorized_equipment > constant:black_plague_shared_cost.scaled_min_motorized_equipment }
				has_fuel > constant:black_plague_shared_cost.scaled_min_fuel
			}
		}
		AND = {
			check_variable = { var = state_population_k value = constant:black_plague_shared_scale.large_population_k compare = greater_than_or_equals }
			check_variable = { var = state_population_k value = constant:black_plague_shared_scale.metropolitan_population_k compare = less_than }
			ROOT = {
				has_equipment = { support_equipment > constant:black_plague_shared_cost.scaled_large_support_equipment }
				has_equipment = { motorized_equipment > constant:black_plague_shared_cost.scaled_large_motorized_equipment }
				has_fuel > constant:black_plague_shared_cost.scaled_large_fuel
			}
		}
		AND = {
			check_variable = { var = state_population_k value = constant:black_plague_shared_scale.metropolitan_population_k compare = greater_than_or_equals }
			ROOT = {
				has_equipment = { support_equipment > constant:black_plague_shared_cost.scaled_metropolitan_support_equipment }
				has_equipment = { motorized_equipment > constant:black_plague_shared_cost.scaled_metropolitan_motorized_equipment }
				has_fuel > constant:black_plague_shared_cost.scaled_metropolitan_fuel
			}
		}
	}
}

# COUNTRY scope. The selected-state board owns the target pointer for human
# responders, while AI countries evaluate their state target through the normal
# availability trigger. This keeps custom-cost colouring aligned with the
# population-scaled payment without blocking AI actions that have no board row.
black_plague_shared_country_can_pay_scaled_selected_state = {
	OR = {
		is_ai = yes
		AND = {
			has_variable = black_plague_selected_response_state
			var:black_plague_selected_response_state = {
				black_plague_shared_state_can_pay_scaled_materials = yes
			}
		}
	}
}

# STATE scope. The Crown Strike is a second, earned operation against the
# sovereign basin, not another Royal Node raid. It remains visible only after
# Evolution IV, enough successful node strikes, and a real military route.
black_plague_shared_crown_target_is_valid = {
	black_plague_state_is_rat_controlled = yes
	OR = {
		has_state_flag = black_plague_royal_basin
		AND = { is_capital = yes CONTROLLER = { tag = RTX } }
	}
	CONTROLLER = { tag = RTX }
}

black_plague_shared_crown_target_can_continue = {
	OR = {
		is_controlled_by = RTX
		CONTROLLER = { black_plague_country_is_human_host = yes }
	}
	NOT = { has_global_flag = black_plague_terminal_takeover_active }
	NOT = { has_global_flag = black_plague_rat_king_defeated }
}

black_plague_shared_crown_mission_target_can_continue = {
	any_state = {
		has_state_flag = black_plague_shared_crown_mission_target
		has_variable = black_plague_shared_crown_mission_owner
		check_variable = { var = black_plague_shared_crown_mission_owner value = ROOT compare = equals }
		black_plague_shared_crown_target_can_continue = yes
	}
}

# COUNTRY scope. A Crown Strike succeeds as soon as the marked royal target is
# no longer under RTX control and a human host controls the state. This keeps
# coalition liberation valid without treating an uncontrolled or rat-held
# target as a success.
black_plague_shared_crown_mission_target_success = {
	any_state = {
		has_state_flag = black_plague_shared_crown_mission_target
		has_variable = black_plague_shared_crown_mission_owner
		check_variable = { var = black_plague_shared_crown_mission_owner value = ROOT compare = equals }
		NOT = { is_controlled_by = RTX }
		CONTROLLER = { black_plague_country_is_human_host = yes }
		NOT = { has_global_flag = black_plague_terminal_takeover_active }
		NOT = { has_global_flag = black_plague_rat_king_defeated }
	}
}

	has_equipment = { motorized_equipment > constant:black_plague_shared_result.crown_strike_motorized_equipment }
	has_equipment = { infantry_equipment > constant:black_plague_shared_result.crown_strike_infantry_equipment }
	has_manpower > constant:black_plague_shared_result.crown_strike_manpower
	has_fuel > constant:black_plague_shared_result.crown_strike_fuel
	command_power > constant:black_plague_shared_result.crown_strike_command_power
	NOT = { num_of_available_civilian_factories < constant:black_plague_shared_result.crown_strike_civilian_factories }
	black_plague_country_has_response_capacity_five = yes
}

black_plague_shared_can_strike_crown = {
	black_plague_shared_crown_target_is_valid = yes
	black_plague_shared_crown_target_can_continue = yes
	NOT = { has_state_flag = black_plague_shared_action_active }
	NOT = { has_state_flag = black_plague_shared_action_cooldown }
	black_plague_shared_state_has_military_route = yes
	ROOT = {
		black_plague_country_can_direct_response = yes
		has_war_with = RTX
		has_country_flag = black_plague_royal_node_strike_contributor
		NOT = { has_country_flag = black_plague_shared_crown_mission_active }
		black_plague_shared_country_can_pay_crown_strike = yes
	}
	has_global_flag = black_plague_evolution_iv_active
	NOT = { has_global_flag = black_plague_terminal_takeover_active }
	NOT = { has_global_flag = black_plague_rat_king_defeated }
	check_variable = { global.black_plague_rat_king_royal_nodes_struck > constant:black_plague_value.one }
	RTX = {
		exists = yes
		has_country_flag = black_plague_rat_king_country
		OR = {
			NOT = { has_variable = black_plague_rat_king_route }
			NOT = { check_variable = { var = black_plague_rat_king_route value = constant:black_plague_rat_king_route.brood_council compare = equals } }
			check_variable = { global.black_plague_rat_king_royal_nodes_struck > constant:black_plague_value.two }
		}
	}
}
