# ============================================================================
# CHAOS REDUX - EVENT 020 HUMAN LAST-RESPONSE TRIGGERS
# ============================================================================
# Human defensive missions that open only after the earned Evolution V route.
# The country must still hold established Black Plague ground, and every
# terminal outcome rechecks the live world before it changes royal pressure.
# ============================================================================

black_plague_human_last_response_country_is_eligible = {
	black_plague_country_can_direct_response = yes
	is_ai = no
	NOT = { has_country_flag = black_plague_rat_country }
	NOT = { has_country_flag = black_plague_rat_king_country }
	has_global_flag = black_plague_evolution_v_route_open
	has_global_flag = black_plague_rat_king_active
	NOT = { has_global_flag = black_plague_terminal_takeover_active }
	NOT = { has_global_flag = black_plague_terminal_takeover_complete }
	any_owned_state = {
		black_plague_state_is_established = yes
		is_controlled_by = ROOT
		NOT = { black_plague_state_is_rat_controlled = yes }
	}
}

black_plague_human_last_response_hold_can_start = {
	black_plague_human_last_response_country_is_eligible = yes
	NOT = { has_country_flag = black_plague_human_last_response_hold_active }
	NOT = { has_country_flag = black_plague_human_last_response_hold_completed }
	NOT = { has_country_flag = black_plague_human_last_response_hold_cooldown }
	NOT = { num_of_civilian_factories_available_for_projects < constant:black_plague_last_response.hold_civilian_factories }
	set_temp_variable = { black_plague_sale_price_check = global.black_friday_price_black_plague_last_response_hold_manpower } NOT = { has_manpower < black_plague_sale_price_check }
	set_temp_variable = { black_plague_sale_price_check = global.black_friday_price_black_plague_last_response_hold_support_equipment } NOT = { has_equipment = { support_equipment < black_plague_sale_price_check } }
	set_temp_variable = { black_plague_sale_price_check = global.black_friday_price_black_plague_last_response_hold_infantry_equipment } NOT = { has_equipment = { infantry_equipment < black_plague_sale_price_check } }
	NOT = { has_stability < constant:black_plague_last_response.hold_stability_floor }
	NOT = { has_war_support < constant:black_plague_last_response.hold_war_support_floor }
}

black_plague_human_last_response_refuge_can_start = {
	black_plague_human_last_response_country_is_eligible = yes
	NOT = { has_country_flag = black_plague_human_last_response_refuge_active }
	NOT = { has_country_flag = black_plague_human_last_response_refuge_completed }
	NOT = { has_country_flag = black_plague_human_last_response_refuge_cooldown }
	black_plague_human_last_response_refuge_target_is_held = yes
	NOT = { num_of_civilian_factories_available_for_projects < constant:black_plague_last_response.refuge_civilian_factories }
	set_temp_variable = { black_plague_sale_price_check = global.black_friday_price_black_plague_last_response_refuge_manpower } NOT = { has_manpower < black_plague_sale_price_check }
	set_temp_variable = { black_plague_sale_price_check = global.black_friday_price_black_plague_last_response_refuge_motorized_equipment } NOT = { has_equipment = { motorized_equipment < black_plague_sale_price_check } }
	set_temp_variable = { black_plague_sale_price_check = global.black_friday_price_black_plague_last_response_refuge_fuel } NOT = { has_fuel < black_plague_sale_price_check }
	NOT = { has_stability < constant:black_plague_last_response.refuge_stability_floor }
	NOT = { has_war_support < constant:black_plague_last_response.refuge_war_support_floor }
}

black_plague_human_last_response_refuge_target_is_held = {
	any_owned_state = {
		black_plague_state_is_established = yes
		is_controlled_by = ROOT
		NOT = { black_plague_state_is_rat_controlled = yes }
		OR = {
			has_state_flag = black_plague_rat_terminal_capital_target
			has_state_flag = black_plague_rat_terminal_refuge_node
			has_state_category = city
		}
	}
}

black_plague_human_last_response_hold_success = {
	black_plague_human_last_response_country_is_eligible = yes
	has_country_flag = black_plague_human_last_response_hold_active
	custom_trigger_tooltip = {
		tooltip = black_plague_human_last_response_hold_progress_tt
		has_variable = black_plague_human_last_response_hold_progress
		check_variable = {
			var = black_plague_human_last_response_hold_progress
			value = constant:black_plague_last_response.success_progress
			compare = greater_than_or_equals
		}
	}
	any_owned_state = {
		black_plague_state_is_established = yes
		is_controlled_by = ROOT
		NOT = { black_plague_state_is_rat_controlled = yes }
	}
}

black_plague_human_last_response_refuge_success = {
	black_plague_human_last_response_country_is_eligible = yes
	has_country_flag = black_plague_human_last_response_refuge_active
	custom_trigger_tooltip = {
		tooltip = black_plague_human_last_response_refuge_progress_tt
		has_variable = black_plague_human_last_response_refuge_progress
		check_variable = {
			var = black_plague_human_last_response_refuge_progress
			value = constant:black_plague_last_response.success_progress
			compare = greater_than_or_equals
		}
	}
	black_plague_human_last_response_refuge_target_is_held = yes
}

black_plague_human_last_response_hold_is_invalid = {
	OR = {
		NOT = { has_country_flag = black_plague_human_last_response_hold_active }
		NOT = { black_plague_human_last_response_country_is_eligible = yes }
	}
}

black_plague_human_last_response_refuge_is_invalid = {
	OR = {
		NOT = { has_country_flag = black_plague_human_last_response_refuge_active }
		NOT = { black_plague_human_last_response_country_is_eligible = yes }
		NOT = { black_plague_human_last_response_refuge_target_is_held = yes }
	}
}
