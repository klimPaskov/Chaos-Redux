infantry_spawn_unit_registry_current_candidate_is_eligible = {
	check_variable = {
		var = infantry_spawn_unit_registry_current_slot_kind
		value = infantry_spawn_unit_registry_requested_slot_kind
		compare = equals
	}
	infantry_spawn_unit_registry_current_gate_is_open = yes
	OR = {
		AND = {
			check_variable = { var = infantry_spawn_unit_registry_current_slot_kind value = constant:infantry_spawn_unit_registry_slot_kind.combat compare = equals }
			check_variable = { var = infantry_spawn_unit_registry_selected_combat_count value = constant:infantry_spawn_unit_registry_cardinality.combat_maximum compare = less_than }
			check_variable = { var = infantry_spawn_unit_registry_current_group_count value = constant:infantry_spawn_unit_registry_cardinality.combat_group_row_maximum compare = less_than }
			OR = {
				check_variable = { var = infantry_spawn_unit_registry_current_group_index value = constant:infantry_spawn_unit_registry_cardinality.missing_index compare = greater_than }
				check_variable = { var = infantry_spawn_unit_registry_selected_combat_group_ids^num value = constant:infantry_spawn_unit_registry_cardinality.combat_group_maximum compare = less_than }
			}
		}
		AND = {
			check_variable = { var = infantry_spawn_unit_registry_current_slot_kind value = constant:infantry_spawn_unit_registry_slot_kind.support compare = equals }
			check_variable = { var = infantry_spawn_unit_registry_selected_support_count value = constant:infantry_spawn_unit_registry_cardinality.support_maximum compare = less_than }
			NOT = {
				is_in_array = {
					array = infantry_spawn_unit_registry_selected_support_token_ids
					value = infantry_spawn_unit_registry_current_token_id
				}
			}
			OR = {
				check_variable = { var = infantry_spawn_unit_registry_current_same_support_type_primary value = constant:infantry_spawn_unit_registry_same_support_type.none compare = equals }
				NOT = {
					is_in_array = {
						array = infantry_spawn_unit_registry_selected_support_type_ids
						value = infantry_spawn_unit_registry_current_same_support_type_primary
					}
				}
			}
			OR = {
				check_variable = { var = infantry_spawn_unit_registry_current_same_support_type_secondary value = constant:infantry_spawn_unit_registry_same_support_type.none compare = equals }
				NOT = {
					is_in_array = {
						array = infantry_spawn_unit_registry_selected_support_type_ids
						value = infantry_spawn_unit_registry_current_same_support_type_secondary
					}
				}
			}
			OR = {
				check_variable = { var = infantry_spawn_unit_registry_current_same_support_type_tertiary value = constant:infantry_spawn_unit_registry_same_support_type.none compare = equals }
				NOT = {
					is_in_array = {
						array = infantry_spawn_unit_registry_selected_support_type_ids
						value = infantry_spawn_unit_registry_current_same_support_type_tertiary
					}
				}
			}
		}
	}
}
