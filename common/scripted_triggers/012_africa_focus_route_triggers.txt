# Route-body focus AI pressure. This is intentionally live-state driven: a
# committed constitution selects one route, and low constitutional axes,
# unresolved proof/action contracts, or a documented crisis make that route's
# next focus more attractive. It does not add a new tag or scan other nations.
africa_focus_ai_route_pressure = {
	OR = {
		has_country_flag = africa_focus_route_action_ai_contract_pending
		has_country_flag = africa_route_viability_under_review
		has_country_flag = africa_constitutional_crisis_escalated
		has_country_flag = africa_postwar_constitutional_review_pending
		AND = {
			has_variable = africa_focus_pending_route_action
			africa_focus_pending_route_action_is_unresolved = yes
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.federal_union }
			OR = {
				AND = { has_variable = africa_constitutional_representation check_variable = { africa_constitutional_representation < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_resource_governance check_variable = { africa_constitutional_resource_governance < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_command_integration check_variable = { africa_constitutional_command_integration < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_withdrawal_protection check_variable = { africa_constitutional_withdrawal_protection < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
			}
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.continental_republic }
			OR = {
				AND = { has_variable = africa_constitutional_representation check_variable = { africa_constitutional_representation < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_executive_power check_variable = { africa_constitutional_executive_power < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_resource_governance check_variable = { africa_constitutional_resource_governance < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_post_unification_rule check_variable = { africa_constitutional_post_unification_rule < constant:africa_constitution_axis.low } }
			}
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.council_of_crowns }
			OR = {
				AND = { has_variable = africa_constitutional_representation check_variable = { africa_constitutional_representation < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_executive_power check_variable = { africa_constitutional_executive_power < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_withdrawal_protection check_variable = { africa_constitutional_withdrawal_protection < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_post_unification_rule check_variable = { africa_constitutional_post_unification_rule < constant:africa_constitution_axis.low } }
			}
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.peoples_union }
			OR = {
				AND = { has_variable = africa_constitutional_representation check_variable = { africa_constitutional_representation < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_executive_power check_variable = { africa_constitutional_executive_power < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_resource_governance check_variable = { africa_constitutional_resource_governance < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_command_integration check_variable = { africa_constitutional_command_integration < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
			}
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.military_continentalism }
			OR = {
				AND = { has_variable = africa_constitutional_command_integration check_variable = { africa_constitutional_command_integration < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_executive_power check_variable = { africa_constitutional_executive_power < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_representation check_variable = { africa_constitutional_representation < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_emergency_rule_debt check_variable = { africa_emergency_rule_debt > constant:africa_focus_route_threshold.maximum_capstone_emergency_debt } }
				AND = { has_variable = africa_continental_war_readiness check_variable = { africa_continental_war_readiness < constant:africa_measure.low } }
			}
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.continental_confederation }
			OR = {
				AND = { has_variable = africa_constitutional_representation check_variable = { africa_constitutional_representation < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_withdrawal_protection check_variable = { africa_constitutional_withdrawal_protection < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_resource_governance check_variable = { africa_constitutional_resource_governance < constant:africa_constitution_axis.low } }
			}
		}
		AND = {
			check_variable = { africa_constitution = constant:africa_constitution.ancestral_covenant }
			OR = {
				AND = { has_variable = africa_ecological_wrath check_variable = { africa_ecological_wrath > constant:africa_focus_route_threshold.minimum_covenant_pressure } }
				AND = { has_variable = africa_constitutional_command_integration check_variable = { africa_constitutional_command_integration < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_resource_governance check_variable = { africa_constitutional_resource_governance < constant:africa_constitution_axis.low } }
				AND = { has_variable = africa_constitutional_crisis_resilience check_variable = { africa_constitutional_crisis_resilience < constant:africa_constitution_axis.low } }
			}
		}
	}
}

# The focus tree is the sole authority that opens actions 093-100. This
# predicate prevents a selector from exposing an unopened contract or replaying
# one after its full, partial, or failure disposition has been recorded.

	OR = {
		AND = {
			check_variable = { africa_focus_pending_route_action = constant:africa_action.convene_federal_deadlock_conference }

				OR = {
					has_country_flag = africa_action_convene_federal_deadlock_conference_full_recorded
					has_country_flag = africa_action_convene_federal_deadlock_conference_partial_recorded
					has_country_flag = africa_action_convene_federal_deadlock_conference_failure_recorded
				}
			}
		}
		AND = {
			check_variable = { africa_focus_pending_route_action = constant:africa_action.conduct_first_continental_election }
			NOT = {
				OR = {
					has_country_flag = africa_action_conduct_first_continental_election_full_recorded
					has_country_flag = africa_action_conduct_first_continental_election_partial_recorded
					has_country_flag = africa_action_conduct_first_continental_election_failure_recorded
				}
			}
		}
		AND = {
			check_variable = { africa_focus_pending_route_action = constant:africa_action.arbitrate_continental_succession }

		}
		AND = {

			}
		}
	}

# COUNTRY scope on a member. Federal and integrated settlements are genuine
# Event 12 autonomy tiers, accepted by the member and locked for the current
# constitutional term. A leaving, rival, resistant, or civil-war government
# never counts as settled.
africa_country_counts_as_federal_settlement = {
	africa_member_host_generation_is_current = yes
	has_country_flag = africa_member_continental_constitution_accepted
	has_country_flag = africa_member_settlement_term_locked
	NOT = { has_civil_war = yes }
	NOT = { has_war_with = event_target:africa_host }
	NOT = { africa_relationship_is_leaving = yes }
	NOT = { africa_relationship_is_resistant = yes }
	NOT = { africa_relationship_is_rival_bloc = yes }
	is_subject = yes
	OVERLORD = { tag = event_target:africa_host }
	OR = {
		AND = {
			africa_relationship_is_autonomous_federal = yes
			has_autonomy_state = autonomy_africa_federal_member
		}
		AND = {
			africa_relationship_is_integrated_region = yes
			has_autonomy_state = autonomy_africa_integrated_region
			has_country_flag = africa_member_final_integration_consent
		}
	}
}

# COUNTRY scope on an independent confederal member. This path exists only for
# the confederation constitution and requires that government's own recorded
# recognition of the common settlement.
africa_country_counts_as_confederal_settlement = {
	africa_member_host_generation_is_current = yes
	is_subject = no
	africa_relationship_is_chartered = yes
	has_country_flag = africa_member_continental_constitution_accepted
	has_country_flag = africa_member_confederal_unity_recognised
	NOT = { has_civil_war = yes }
	NOT = { has_war_with = event_target:africa_host }
	event_target:africa_host = {
		check_variable = { africa_constitution = constant:africa_constitution.continental_confederation }
		has_country_flag = africa_confederation_route_capstone
	}
}

# COUNTRY scope on a restored or priority polity whose own settlement accepted
# the continental constitution. This does not infer consent from alliance or
# opinion: the explicit package flags are mandatory.
africa_country_counts_as_allied_restored_settlement = {
	africa_member_host_generation_is_current = yes
	has_country_flag = africa_member_continental_constitution_accepted
	has_country_flag = africa_member_allied_settlement_accepted
	NOT = { has_civil_war = yes }
	NOT = { has_war_with = event_target:africa_host }
	NOT = { africa_relationship_is_leaving = yes }
	NOT = { africa_relationship_is_resistant = yes }
	NOT = { africa_relationship_is_rival_bloc = yes }
	OR = {
		has_country_flag = africa_restored_polity_member
		has_country_flag = africa_priority_member_package_active
		africa_rsa_allied_settlement_is_complete = yes
	}
	OR = {
		africa_relationship_is_associate = yes
		africa_relationship_is_chartered = yes
		is_in_faction_with = event_target:africa_host
	}
}

africa_country_counts_as_continental_settlement = {
	OR = {
		africa_country_counts_as_federal_settlement = yes
		africa_country_counts_as_confederal_settlement = yes
		africa_country_counts_as_allied_restored_settlement = yes
	}
}

# STATE scope. Ownership and control must agree. Direct host land must be a
# real core; all other cases point to an accepted country settlement or a
# time-bounded congress exemption with a live completion mission.
africa_state_counts_as_continentally_settled = {
	is_on_continent = africa
	is_controlled_by = OWNER
	OR = {
		AND = {
			is_owned_by = event_target:africa_host
			OR = {
				is_core_of = event_target:africa_host
				has_state_flag = africa_local_settlement_complete
			}
		}
		owner = { africa_country_counts_as_continental_settlement = yes }
		AND = {
			has_state_flag = africa_continental_peace_exemption_active
			owner = { africa_continental_peace_exemption_is_current = yes }
		}
	}
}

africa_all_african_states_are_settled = {
	all_state = {
		OR = {
			NOT = { is_on_continent = africa }
			africa_state_counts_as_continentally_settled = yes
		}
	}
}
