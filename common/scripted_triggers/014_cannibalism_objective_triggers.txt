cannibalism_investigation_target_is_valid = {
	has_variable = cannibalism_investigation_target_state
	var:cannibalism_investigation_target_state = {
		is_controlled_by = ROOT
		impassable = no
		OR = {
			ROOT = { NOT = { has_country_flag = cannibalism_investigation_target_has_node_identity } }
			AND = {
				has_variable = cannibalism_node_id
				has_variable = cannibalism_node_generation
				check_variable = { var = cannibalism_node_id value = ROOT.cannibalism_investigation_target_node_id compare = equals }
				check_variable = { var = cannibalism_node_generation value = ROOT.cannibalism_investigation_target_node_generation compare = equals }
			}
		}
	}
}

cannibalism_investigation_objective_complete = {
	has_country_flag = cannibalism_investigation_mission_active
	check_variable = {
		var = cannibalism_investigation_progress
		value = cannibalism_investigation_goal
		compare = greater_than_or_equals
	}
}

cannibalism_prison_objective_secured = {
	cannibalism_prison_target_is_valid = yes
	NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_objective_cost.prison_support_reserve } }
	OR = {
		var:cannibalism_prison_target_state = { is_island_state = yes }
		NOT = { has_equipment = { train_equipment_1 < constant:cannibalism_objective_cost.prison_train_reserve } }
	}
	OR = {
		var:cannibalism_prison_target_state = { is_island_state = no }
		NOT = { has_equipment = { convoy_1 < constant:cannibalism_objective_cost.prison_convoy_reserve } }
	}
	meta_trigger = {
		text = {
			divisions_in_state = {
				state = var:cannibalism_prison_target_state
				size > [DIVISION_FLOOR]
			}
		}
		DIVISION_FLOOR = "[?cannibalism_prison_division_floor|.0]"
	}
}

cannibalism_prison_objective_complete = {
	has_country_flag = cannibalism_hold_prison_mission_active
	check_variable = {
		var = cannibalism_prison_hold_days
		value = cannibalism_prison_required_hold_days
		compare = greater_than_or_equals
	}
}

cannibalism_reach_island_objective_complete = {
	has_country_flag = cannibalism_reach_island_mission_active
	OR = {
		check_variable = {
			var = cannibalism_reach_island_progress
			value = cannibalism_reach_island_goal
			compare = greater_than_or_equals
		}
		AND = {
			check_variable = { cannibalism_reach_island_progress > constant:cannibalism_objective_value.zero }
			cannibalism_reach_island_target_identity_matches = yes
			var:cannibalism_reach_island_target_country = { has_capitulated = yes }
		}
	}
}

cannibalism_stop_unification_objective_is_maintained = {
	cannibalism_stop_unification_target_identity_matches = yes
	NOT = { has_global_flag = cannibalism_reveal_complete }
	var:cannibalism_stop_unification_target_country = {
		has_war_with = ROOT
		NOT = { has_capitulated = yes }
	}
}

cannibalism_stop_unification_objective_complete = {
	has_country_flag = cannibalism_stop_unification_mission_active
	OR = {
		check_variable = {
			var = cannibalism_stop_unification_hold_days
			value = cannibalism_stop_unification_goal_days
			compare = greater_than_or_equals
		}
		AND = {
			check_variable = { cannibalism_stop_unification_hold_days > constant:cannibalism_objective_value.zero }
			has_global_flag = cannibalism_convergence_broken
			NOT = { has_global_flag = cannibalism_reveal_complete }
		}
		AND = {
			check_variable = { cannibalism_stop_unification_hold_days > constant:cannibalism_objective_value.zero }
			cannibalism_stop_unification_target_identity_matches = yes
			var:cannibalism_stop_unification_target_country = { has_capitulated = yes }
		}
	}
}
