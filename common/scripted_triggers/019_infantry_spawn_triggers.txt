infantry_spawn_country_has_eligible_state = {
	any_controlled_state = { infantry_spawn_state_is_eligible = yes }
}

infantry_spawn_country_can_receive_generation = {
	exists = yes
	infantry_spawn_scenario_transaction_is_idle = yes
	NOT = { has_country_flag = infantry_spawn_scenario_actor }
	NOT = { has_country_flag = infantry_spawn_processing_deferred }
	NOT = { has_country_flag = infantry_spawn_defeat_cleanup_in_progress }
	NOT = { has_country_flag = infantry_spawn_annex_cleanup_invariant_failure }
	NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
	infantry_spawn_country_has_eligible_state = yes
}

infantry_spawn_state_is_eligible = {
	impassable = no
	CONTROLLER = { exists = yes }
	NOT = { has_state_flag = infantry_spawn_selected_for_current_generation }
}

infantry_spawn_country_is_active = {
	has_country_flag = infantry_spawn_participant
	NOT = { has_country_flag = infantry_spawn_country_cleanup_complete }
	NOT = { has_country_flag = infantry_spawn_defeat_cleanup_in_progress }
	NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
}

# Country scope. Direct-scenario actors keep complete private Event 19 ledgers,
# but their units, decisions, claimants, and management outcomes must never
# advance the ordinary manifestation or evolution history.
infantry_spawn_contributes_to_ordinary_evolution_history = {
	infantry_spawn_scenario_transaction_is_idle = yes
	NOT = { is_infantry_spawn_derivative_country = yes }
	NOT = {
		OR = {
			has_country_flag = infantry_spawn_scenario_actor
			has_country_flag = infantry_spawn_scenario_dynamic_breakaway
			has_country_flag = infantry_spawn_scenario_takeover_actor
			has_country_flag = infantry_spawn_scenario_setup_bypass
		}
	}
}

# Stable country membership for the maintained global evolution population.
# The transient scenario transaction gate remains on scoring callers: a setup
# transaction pauses evolution, while its persistent actor flags remove only
# the exact country from ordinary history.
infantry_spawn_is_ordinary_evolution_counter_country = {
	exists = yes
	NOT = { is_infantry_spawn_derivative_country = yes }
	NOT = { has_country_flag = infantry_spawn_derivative_country }
	NOT = {
		OR = {
			has_country_flag = infantry_spawn_scenario_actor
			has_country_flag = infantry_spawn_scenario_dynamic_breakaway
			has_country_flag = infantry_spawn_scenario_takeover_actor
			has_country_flag = infantry_spawn_scenario_setup_bypass
		}
	}
}

# Ordinary-country decision surface lifecycle. Evolution III and IV capability
# flags deliberately do not appear here: they remain after the government has
# suppressed further draws and settled every live Event 19 obligation.
infantry_spawn_ordinary_management_crisis_is_active = {
	OR = {
		infantry_spawn_has_unresolved_generation = yes
		check_variable = { infantry_spawn_active_lot_count > constant:infantry_spawn_weight_result.zero }
		AND = {
			has_country_flag = infantry_spawn_unaccounted_formation_detected
			check_variable = { infantry_spawn_unaccounted_division_count > constant:infantry_spawn_weight_result.zero }
		}
		infantry_spawn_has_active_claimant = yes
		infantry_spawn_has_deferred_event19_action = yes
		has_country_flag = infantry_spawn_prefire_opening_pending
		has_country_flag = infantry_spawn_audit_decision_running
		has_country_flag = infantry_spawn_standardization_running
		has_country_flag = infantry_spawn_demobilization_running
		has_country_flag = infantry_spawn_training_cycle_running
		has_country_flag = infantry_spawn_muster_districts_running
		has_country_flag = infantry_spawn_integration_staff_search_running
		has_country_flag = infantry_spawn_specialist_preservation_running
		has_country_flag = infantry_spawn_prototype_trial_running
		has_country_flag = infantry_spawn_rail_corridor_running
		has_country_flag = infantry_spawn_request_cooldown
	}
}

	NOT = { has_country_flag = infantry_spawn_scenario_takeover_actor }

	exists = yes
	infantry_spawn_scenario_transaction_is_idle = yes
	OR = {
		AND = {
			infantry_spawn_country_is_active = yes
			OR = {
				check_variable = { infantry_spawn_active_lot_count > 0 }
				has_country_flag = infantry_spawn_claimant_system_active
				has_country_flag = infantry_spawn_anomalous_registry_active
			}
		}
		infantry_spawn_derivative_package_is_active = yes
		AND = {
			NOT = { has_country_flag = infantry_spawn_ledger_invariant_failure }
			infantry_spawn_has_deferred_event19_action = yes
		}
	}
}

# Generic registry contract used by direct scenarios and natural Evolution IV
# pressure. A future family participates through its one aligned registry row;
# neither caller maintains a family enumeration.
infantry_spawn_current_registry_row_can_form_derivative = {
	chaos_unit_family_registry_is_aligned = yes
	chaos_unit_family_current_row_is_enabled = yes
	chaos_unit_family_current_row_allows_event19_family_lot = yes
	chaos_unit_family_current_row_can_spawn = yes
	chaos_unit_family_current_row_has_supported_event19_visual_contract = yes
	check_variable = {
		var = global.chaos_unit_family_derivative_profile_entries^chaos_unit_family_registry_index
		value = constant:infantry_spawn_weight_result.zero
		compare = greater_than
	}
	check_variable = {
		var = global.chaos_unit_family_parent_isolation_profile_entries^chaos_unit_family_registry_index
		value = constant:infantry_spawn_weight_result.zero
		compare = greater_than
	}
	check_variable = { global.chaos_unit_family_cleanup_profile_entries^chaos_unit_family_registry_index > constant:infantry_spawn_weight_result.zero }
	check_variable = {
		var = global.chaos_unit_family_contract_version_entries^chaos_unit_family_registry_index
		value = constant:chaos_unit_family_registry.contract_version
		compare = equals
	}
}

# Country scope. The first reception freezes row identity, not availability.
# This pure trigger can therefore protect a delayed event without re-running a
# provider eligibility callback or substituting another family after dispatch.

	chaos_unit_family_registry_is_aligned = yes
	has_variable = infantry_spawn_first_family_reception_registry_index
	has_variable = infantry_spawn_first_family_reception_family_id
	has_variable = infantry_spawn_first_family_reception_provider_id
	has_variable = infantry_spawn_first_family_reception_visual_profile
	has_variable = infantry_spawn_first_family_reception_nonce
	check_variable = {
		var = infantry_spawn_first_family_reception_registry_index
		value = constant:chaos_unit_family_registry_index.invalid
		compare = greater_than
	}
	check_variable = {
		var = infantry_spawn_first_family_reception_registry_index
		value = global.chaos_unit_family_id_entries^num
		compare = less_than
	}
	check_variable = {
		var = global.chaos_unit_family_id_entries^infantry_spawn_first_family_reception_registry_index
		value = infantry_spawn_first_family_reception_family_id
		compare = equals
	}
	check_variable = {
		var = global.chaos_unit_family_provider_id_entries^infantry_spawn_first_family_reception_registry_index
		value = infantry_spawn_first_family_reception_provider_id
		compare = equals
	}
	check_variable = {
		var = global.chaos_unit_family_visual_profile_entries^infantry_spawn_first_family_reception_registry_index
		value = infantry_spawn_first_family_reception_visual_profile
		compare = equals
	}
	check_variable = {
		var = infantry_spawn_first_family_reception_family_id
		value = constant:chaos_unit_family_registry.minimum_family_id
		compare = greater_than_or_equals
	}
	check_variable = {
		var = infantry_spawn_first_family_reception_provider_id
		value = constant:chaos_unit_family_registry.minimum_provider_id
		compare = greater_than_or_equals
	}
	check_variable = {
		var = infantry_spawn_first_family_reception_nonce
		value = constant:infantry_spawn_first_family_reception.initial_nonce
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.chaos_unit_family_cleanup_profile_entries^infantry_spawn_first_family_reception_registry_index
		value = constant:infantry_spawn_weight_result.zero
		compare = greater_than
	}
	check_variable = {
		var = global.chaos_unit_family_parent_isolation_profile_entries^infantry_spawn_first_family_reception_registry_index
		value = constant:infantry_spawn_weight_result.zero
		compare = greater_than
	}
	check_variable = {
		var = global.chaos_unit_family_contract_version_entries^infantry_spawn_first_family_reception_registry_index
		value = constant:chaos_unit_family_registry.contract_version
		compare = equals
	}
	OR = {
		AND = {
			check_variable = { infantry_spawn_first_family_reception_family_id = constant:event19_zombie_family.family_id }
			check_variable = { infantry_spawn_first_family_reception_provider_id = constant:event19_zombie_family.provider_id }
			check_variable = { infantry_spawn_first_family_reception_visual_profile = constant:event19_zombie_family.visual_profile }
		}
		AND = {

			check_variable = { infantry_spawn_first_family_reception_provider_id = constant:event19_ghost_family.provider_id }
			check_variable = { infantry_spawn_first_family_reception_visual_profile = constant:event19_ghost_family.visual_profile }
		}
		AND = {
			check_variable = { infantry_spawn_first_family_reception_family_id = constant:event19_golem_family.family_id }
			check_variable = { infantry_spawn_first_family_reception_provider_id = constant:event19_golem_family.provider_id }
			check_variable = { infantry_spawn_first_family_reception_visual_profile = constant:event19_golem_family.visual_profile }

			check_variable = { infantry_spawn_first_family_reception_visual_profile = constant:chaos_unit_family_visual_profile.provider_neutral_army }

# Ordinary and anomalous claimant releases are multi-state transactions. The
# ordinary mode needs Evolution III but has no anomalous-family prerequisite.
infantry_spawn_natural_release_has_preferred_mainland_state = {
	any_owned_state = {
		is_controlled_by = PREV
		impassable = no
		NOT = { is_capital = yes }
		NOT = { is_island_state = yes }
	}
}

# An island is viable only when no controlled, passable, non-capital mainland
# state exists. This is the specification's only-viable-base rule, not a random
# geographic substitute.
infantry_spawn_natural_release_has_viable_region_state = {
	OR = {
		infantry_spawn_natural_release_has_preferred_mainland_state = yes
		any_owned_state = {
			is_controlled_by = PREV
			impassable = no
			NOT = { is_capital = yes }
			is_island_state = yes
		}
	}
}

infantry_spawn_natural_claimant_release_base_is_safe = {
	infantry_spawn_natural_release_common_is_safe = yes
	infantry_spawn_has_evolution_iii = yes
	num_of_controlled_states > constant:infantry_spawn_weight_result.one
	infantry_spawn_natural_release_has_viable_region_state = yes
}

# Inputs are the current registry index plus the temporary family-state values
# produced by infantry_spawn_calculate_current_family_state.

	infantry_spawn_current_registry_row_can_form_derivative = yes
	check_variable = { chaos_unit_family_candidate_eligible > constant:infantry_spawn_weight_result.zero }
	check_variable = {
		var = infantry_spawn_family_live_division_count
		value = constant:infantry_spawn_saturation.derivative_minimum_family_divisions

	}
	OR = {
		check_variable = {
