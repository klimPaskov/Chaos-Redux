is_secret_alliance_target = {
	has_country_flag = secret_alliance_target
}

is_secret_alliance_member = {
	has_country_flag = secret_alliance_member
}

is_secret_alliance_hidden_member = {
	has_country_flag = secret_alliance_member
	NOT = { has_country_flag = secret_alliance_public_member }
}

is_secret_alliance_public_member = {
	has_country_flag = secret_alliance_public_member
}

secret_alliance_is_active = {
	has_global_flag = secret_alliance_active
	has_event_target = secret_alliance_target
}

secret_alliance_is_public = {
	has_global_flag = secret_alliance_public_revealed
	has_event_target = secret_alliance_target
}

secret_alliance_can_be_target = {
	exists = yes
	NOT = { has_capitulated = yes }
	NOT = { is_subject = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_country_flag = secret_alliance_target }
	NOT = { has_country_flag = secret_alliance_member }
	num_of_controlled_states > 1
}

secret_alliance_can_be_founder_for_target = {
	exists = yes
	is_major = no
	NOT = { has_capitulated = yes }
	NOT = { is_subject = yes }
	NOT = { is_in_faction = yes }
	NOT = { tag = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { is_subject_of = ROOT }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_country_flag = secret_alliance_target }
	NOT = { has_country_flag = secret_alliance_member }
	num_of_controlled_states > 0
}

secret_alliance_member_valid_for_target_root = {
	exists = yes
	has_country_flag = secret_alliance_member
	NOT = { has_capitulated = yes }
	NOT = { tag = ROOT }
	NOT = { is_subject_of = ROOT }
	NOT = { is_in_faction_with = ROOT }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
}

secret_alliance_has_three_founder_candidates = {
	any_country = {
		count = constant:secret_alliance_foundation.founder_count
		secret_alliance_can_be_founder_for_target = yes
	}
}

secret_alliance_can_start_for_target = {
	secret_alliance_can_be_target = yes
	secret_alliance_has_three_founder_candidates = yes
}

is_secret_alliance_automatic_event_available = {
	NOT = { has_global_flag = secret_alliance_active }
	any_country = { secret_alliance_can_start_for_target = yes }
}

secret_alliance_member_valid_for_public_faction = {
	exists = yes
	has_country_flag = secret_alliance_member
	NOT = { has_capitulated = yes }
	NOT = { is_subject = yes }
	NOT = { is_in_faction_with = event_target:secret_alliance_target }

secret_alliance_has_valid_public_member = {
	any_country = { secret_alliance_member_valid_for_public_faction = yes }
}

secret_alliance_has_neighboring_hidden_member = {
	any_country = {
		exists = yes
		has_country_flag = secret_alliance_member
		NOT = { has_country_flag = secret_alliance_public_member }
		is_neighbor_of = ROOT
	}
}

secret_alliance_has_neighboring_member = {
	any_country = {
		exists = yes
		has_country_flag = secret_alliance_member
		is_neighbor_of = ROOT
	}
}

secret_alliance_has_neighboring_public_member = {
	any_country = {
		exists = yes
		has_country_flag = secret_alliance_member
		has_country_flag = secret_alliance_public_member
		is_neighbor_of = ROOT
	}
}

secret_alliance_has_neighboring_exposed_member = {
	any_country = {
		exists = yes
		has_country_flag = secret_alliance_member
		has_country_flag = secret_alliance_exposed
		NOT = { has_country_flag = secret_alliance_public_member }
		is_neighbor_of = ROOT
	}
}

secret_alliance_has_active_mission = {
	OR = {
		has_country_flag = secret_alliance_defense_mission_active
		has_country_flag = secret_alliance_investigation_mission_active
		has_country_flag = secret_alliance_diplomacy_mission_active
		has_country_flag = secret_alliance_border_mission_active
	}
}

secret_alliance_target_can_open_dossier = {
	is_secret_alliance_target = yes
	secret_alliance_is_active = yes
	OR = {
		has_country_flag = secret_alliance_dossier_open
		check_variable = {
			var = pact_suspicion
			value = constant:secret_alliance_threshold.open_category_suspicion
			compare = greater_than_or_equals
		}
		check_variable = {
			var = pact_evidence
			value = constant:secret_alliance_threshold.evidence_medium
			compare = greater_than_or_equals
		}
		has_country_flag = secret_alliance_evolution_ii
		has_country_flag = secret_alliance_serious_incident
	}
}

secret_alliance_target_can_use_hidden_actions = {
	secret_alliance_target_can_open_dossier = yes
	NOT = { secret_alliance_is_public = yes }
}

secret_alliance_target_can_use_counterplay = {
	is_secret_alliance_target = yes
	has_country_flag = secret_alliance_counterplay_open
}

secret_alliance_state_has_route_asset = {
	OR = {
		rail_way > 0
		supply_node > 0
		naval_base > 0
		air_base > 0
	}
}

secret_alliance_state_borders_hidden_member = {
	any_neighbor_state = {
		OR = {
			OWNER = { is_secret_alliance_hidden_member = yes }
			CONTROLLER = { is_secret_alliance_hidden_member = yes }
		}
	}
}

secret_alliance_state_borders_public_member = {
	any_neighbor_state = {
		OR = {
			OWNER = { is_secret_alliance_public_member = yes }
			CONTROLLER = { is_secret_alliance_public_member = yes }
		}
	}
}

secret_alliance_has_foreign_route_objective_state = {
	any_owned_state = {
		is_controlled_by = ROOT
		infrastructure > constant:secret_alliance_mission.route_infrastructure_gate
		secret_alliance_state_has_route_asset = yes
		OR = {
			secret_alliance_state_borders_hidden_member = yes
			secret_alliance_state_borders_public_member = yes
			naval_base > 0
			air_base > 0
		}
	}
}

secret_alliance_has_guarded_foreign_route_state = {
	any_owned_state = {
		is_controlled_by = ROOT
		infrastructure > constant:secret_alliance_mission.route_infrastructure_gate
		secret_alliance_state_has_route_asset = yes
		OR = {
			secret_alliance_state_borders_hidden_member = yes
			secret_alliance_state_borders_public_member = yes
			naval_base > 0
			air_base > 0
		}
		ROOT = {
			divisions_in_state = {
				state = PREV
				size > constant:secret_alliance_mission.route_state_division_floor
			}
		}
	}
}

secret_alliance_has_guarded_public_border_state = {
	any_owned_state = {
		is_controlled_by = ROOT
		is_core_of = ROOT
		secret_alliance_state_borders_public_member = yes
		ROOT = {
			divisions_in_state = {
				state = PREV
				size > constant:secret_alliance_mission.border_state_division_floor
			}
		}
	}
}

secret_alliance_guard_capital_network_ready = {
	capital_scope = {
		is_controlled_by = ROOT
		infrastructure > constant:secret_alliance_mission.route_infrastructure_gate
		OR = {
			secret_alliance_state_has_route_asset = yes
			has_state_category = city
			has_state_category = metropolis
			has_state_category = megalopolis
		}
		ROOT = {
			divisions_in_state = {
				state = PREV
				size > constant:secret_alliance_mission.capital_state_division_floor
			}
		}
	}
	OR = {
		has_country_flag = secret_alliance_guard_nodes_hardened
		has_country_flag = secret_alliance_ministries_secured
		has_country_flag = secret_alliance_command_threat_vetted
	}
}

secret_alliance_industrial_belt_ready = {
	num_of_military_factories > 0
	any_owned_state = {
		is_controlled_by = ROOT
		infrastructure > constant:secret_alliance_mission.route_infrastructure_gate
		OR = {
			arms_factory > 0
			industrial_complex > 0
		}
		ROOT = {
			divisions_in_state = {
				state = PREV
				size > constant:secret_alliance_mission.industrial_state_division_floor
			}
		}
	}
	OR = {
		has_country_flag = secret_alliance_plants_hardened
		has_country_flag = secret_alliance_sabotage_bonus_blocked
	}
}

secret_alliance_foreign_route_watched = {
	has_equipment = { train_equipment > constant:secret_alliance_decision_cost.pass_trains_gate }
	OR = {
		secret_alliance_has_guarded_foreign_route_state = yes
		NOT = { secret_alliance_has_foreign_route_objective_state = yes }
	}
	OR = {
		has_country_flag = secret_alliance_courier_pass_sealed
		has_country_flag = secret_alliance_sabotage_bonus_blocked
		NOT = {
			check_variable = {
				var = pact_infiltration
				value = constant:secret_alliance_threshold.infiltration_high
				compare = greater_than_or_equals
			}
		}
	}
}
