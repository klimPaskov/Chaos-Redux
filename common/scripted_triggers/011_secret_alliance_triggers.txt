		}

				value = constant:secret_alliance_evolution_iii_state.minimum_members_with_major

			}
		}
	}
	OR = {
		AND = {
			check_variable = {
				var = global.secret_alliance_readiness
				value = constant:secret_alliance_evolution_iii_state.controlled_readiness
				compare = greater_than_or_equals
			}

				var = global.secret_alliance_cohesion
				value = constant:secret_alliance_evolution_iii_state.controlled_cohesion

			}
		}
		check_variable = {
			var = global.secret_alliance_evidence
			value = constant:secret_alliance_evolution_iii_state.forced_evidence
			compare = greater_than_or_equals
		}

	}
