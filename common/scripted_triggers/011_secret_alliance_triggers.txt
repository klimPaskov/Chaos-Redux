	NOT = { is_special_chaos_country = yes }
	NOT = { has_capitulated = yes }

secret_alliance_candidate_matches_scenario_type = {
	OR = {
		NOT = { has_global_flag = secret_alliance_scenario_origin }
		AND = {
			has_global_flag = secret_alliance_scenario_origin
			event_target:secret_alliance_target = {
				check_variable = { secret_alliance_scenario_type_snapshot = constant:secret_alliance_scenario_type.regional_ring }
			}
			is_on_same_continent_as = event_target:secret_alliance_target
			secret_alliance_has_operational_reach_to_target = yes
		}
		AND = {
			has_global_flag = secret_alliance_scenario_origin
			event_target:secret_alliance_target = {
				check_variable = { secret_alliance_scenario_type_snapshot = constant:secret_alliance_scenario_type.ideological_front }
			}
			NOT = { has_same_ideology = event_target:secret_alliance_target }
		}
		AND = {
			has_global_flag = secret_alliance_scenario_origin
			event_target:secret_alliance_target = {
				check_variable = { secret_alliance_scenario_type_snapshot = constant:secret_alliance_scenario_type.great_power_sponsor }
			}
			secret_alliance_has_operational_reach_to_target = yes
		}
		AND = {
			has_global_flag = secret_alliance_scenario_origin
			event_target:secret_alliance_target = {
				OR = {
					check_variable = { secret_alliance_scenario_type_snapshot = constant:secret_alliance_scenario_type.unlikely_coalition }
					check_variable = { secret_alliance_scenario_type_snapshot = constant:secret_alliance_scenario_type.random_coalition }
				}
			}
		}
	}
}

secret_alliance_scenario_can_launch = {
	secret_alliance_target_is_valid = yes
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = secret_alliance_active }
	NOT = { has_global_flag = secret_alliance_postwar_bloc_active }
	any_country = {
		count = constant:secret_alliance_event.normal_founders
		secret_alliance_is_valid_scenario_minor_for_root = yes
	}
	secret_alliance_scenario_type_has_valid_composition = yes
}

secret_alliance_is_valid_scenario_minor_for_root = {
	secret_alliance_is_valid_founder_for_root = yes
	OR = {
		AND = {
			ROOT = { check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.regional_ring } }
			is_on_same_continent_as = ROOT
			secret_alliance_has_operational_reach_to_root = yes
		}
		AND = {
			ROOT = { check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.ideological_front } }
			NOT = { has_same_ideology = ROOT }
		}
		AND = {
			ROOT = { check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.great_power_sponsor } }
			secret_alliance_has_operational_reach_to_root = yes
		}
		ROOT = {
			OR = {
				check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.unlikely_coalition }
				check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.random_coalition }
			}
		}
	}
}

secret_alliance_is_valid_major_sponsor_for_root = {
	exists = yes
	is_ai = yes
	is_major = yes
	NOT = { tag = ROOT }
	is_subject = no
	has_civil_war = no
	uses_normal_civilian_systems = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { has_capitulated = yes }
	NOT = { has_country_flag = secret_alliance_active_member }
	NOT = { has_war_with = ROOT }
	NOT = { is_in_faction_with = ROOT }
	NOT = { is_subject_of = ROOT }
	ROOT = { NOT = { is_subject_of = PREV } }
	NOT = { has_guaranteed = ROOT }
	NOT = { is_guaranteed_by = ROOT }
	secret_alliance_scenario_major_matches_type_for_root = yes
	OR = {
		is_in_faction = no
		AND = {
			is_in_faction = yes
			is_faction_leader = no
		}
	}
}

secret_alliance_scenario_type_has_valid_composition = {
	OR = {
		AND = {
			check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.regional_ring }
			any_country = {
				count = constant:secret_alliance_scenario_identity.minimum_regional_approaches
				secret_alliance_is_valid_scenario_minor_for_root = yes
			}
		}
		AND = {
			check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.ideological_front }
			any_country = {
				count = constant:secret_alliance_event.normal_founders
				secret_alliance_is_valid_scenario_minor_for_root = yes
			}
		}
		AND = {
			check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.great_power_sponsor }
			any_country = {
				count = constant:secret_alliance_scenario_identity.minimum_sponsor_partners
				secret_alliance_is_valid_scenario_minor_for_root = yes
			}
			any_country = {
				count = constant:secret_alliance_state.true
				secret_alliance_is_valid_major_sponsor_for_root = yes
			}
		}
		AND = {
			check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.unlikely_coalition }
			any_country = {
				secret_alliance_is_valid_scenario_minor_for_root = yes
				has_same_ideology = ROOT
			}
			any_country = {
				secret_alliance_is_valid_scenario_minor_for_root = yes
				NOT = { has_same_ideology = ROOT }
			}
		}
		check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.random_coalition }
	}
}

secret_alliance_scenario_type_has_valid_major = {
	OR = {
		AND = {
			NOT = { check_variable = { triggerable_scenarios_secret_alliance_type = constant:secret_alliance_scenario_type.great_power_sponsor } }
			NOT = { check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.high } }
			NOT = { check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.maximum } }
		}
		AND = {
			any_country = {
				count = constant:secret_alliance_event.evolution_ii_major_cap
				secret_alliance_is_valid_major_sponsor_for_root = yes
			}
			NOT = { check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.maximum } }
		}
		AND = {
			check_variable = { triggerable_scenarios_intensity = constant:triggerable_scenario_intensity.maximum }
			any_country = {
				count = constant:secret_alliance_event.evolution_iii_major_cap
				secret_alliance_is_valid_major_sponsor_for_root = yes
			}
		}
	}
}
