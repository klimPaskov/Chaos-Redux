secret_alliance_second_major_can_join_public_crisis = {
	has_global_flag = secret_alliance_active
	NOT = { has_global_flag = secret_alliance_revealed }
	NOT = { has_global_flag = secret_alliance_resolved }
	check_variable = {
		var = secret_alliance_stage
		value = constant:secret_alliance_stage.public_crisis
		compare = greater_than_or_equals
	}
	NOT = { has_country_flag = secret_alliance_member_split_success }
	NOT = {
		AND = {
			check_variable = {
				var = secret_alliance_known_members
				value = constant:secret_alliance_threshold.second_major_known_members_deterrent
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_preparedness
				value = constant:secret_alliance_threshold.preparedness_high
				compare = greater_than_or_equals
			}
		}
	}
	NOT = {
		AND = {
			secret_alliance_has_strong_faction_backing = yes
			check_variable = {
				var = secret_alliance_evidence
				value = constant:secret_alliance_threshold.evidence_full_dossier
				compare = greater_than_or_equals
			}
		}
	}
	OR = {
		count_triggers = {
			amount = 2
			check_variable = {
				var = secret_alliance_cohesion
				value = constant:secret_alliance_threshold.cohesion_dissolution_vulnerable
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_readiness
				value = constant:secret_alliance_threshold.readiness_public_crisis
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_hostility
				value = constant:secret_alliance_threshold.hostility_sabotage
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_preparedness
				value = constant:secret_alliance_threshold.ai_target_preparedness_action
				compare = less_than
			}
			NOT = { secret_alliance_has_strong_faction_backing = yes }
		}
		AND = {
			is_current_evolution_enabled_for_tier_3 = yes
			check_variable = {
				var = secret_alliance_hostility
				value = constant:secret_alliance_threshold.readiness_public_crisis
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_preparedness
				value = constant:secret_alliance_threshold.preparedness_border_ready
				compare = less_than
			}
		}
	}
}
