# ============================================================================
# CHAOS REDUX - EVENT 011 SECRET ALLIANCE TRIGGERS
# ============================================================================
# Target, founder, member, reveal, evolution, and AI safety gates for the
# Secret Alliance helper layer. Founder and member candidate triggers are
# evaluated in the candidate country scope with ROOT as the target country.
# ============================================================================

secret_alliance_can_be_target = {
	exists = yes
	uses_normal_civilian_systems = yes
	num_of_controlled_states > constant:secret_alliance_requirement.min_controlled_states
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = secret_alliance_unavailable }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_global_flag = secret_alliance_active }
	NOT = { has_country_flag = secret_alliance_target }
	any_country = {
		count = constant:secret_alliance_requirement.founders
		secret_alliance_is_valid_founder_for_root_target = yes
	}
}

secret_alliance_has_valid_target_available = {
	any_country = {
		secret_alliance_can_be_target = yes
	}
}

secret_alliance_target_has_active_pact = {
	has_country_flag = secret_alliance_target
	has_global_flag = secret_alliance_active
	has_event_target = secret_alliance_target
}

secret_alliance_is_valid_founder_for_root_target = {
	exists = yes
	uses_normal_civilian_systems = yes
	num_of_controlled_states > constant:secret_alliance_requirement.min_controlled_states
	NOT = { tag = ROOT }
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { has_war_with = ROOT }
	NOT = { is_in_faction_with = ROOT }
	NOT = { is_subject = yes }
	NOT = { is_subject_of = ROOT }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_in_array = { array = global.secret_alliance_members value = THIS } }
	NOT = { has_country_flag = secret_alliance_member }
	NOT = { has_country_flag = secret_alliance_diplomacy_locked }
}

secret_alliance_is_valid_member_candidate_for_root_target = {
	secret_alliance_is_valid_founder_for_root_target = yes
	OR = {
		is_major = no
		AND = {
			is_major = yes
			NOT = { has_global_flag = secret_alliance_major_patron_joined }
			OR = {
				NOT = { is_in_faction = yes }
				is_faction_leader = yes
			}
		}
	}
	ROOT = {
		OR = {
			check_variable = {
				var = secret_alliance_member_count
				value = secret_alliance_member_cap
				compare = less_than
			}
			NOT = { has_variable = secret_alliance_member_count }
		}
	}
}

secret_alliance_member_is_live_for_root_target = {
	exists = yes
	has_country_flag = secret_alliance_member
	is_in_array = { array = global.secret_alliance_members value = THIS }
	NOT = { tag = ROOT }
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { is_subject_of = ROOT }
	NOT = { is_in_faction_with = ROOT }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
}

secret_alliance_member_should_leave_for_root_target = {
	OR = {
		NOT = { exists = yes }
		has_capitulated = yes
		has_global_flag = world_end
		is_subject_of = ROOT
		is_in_faction_with = ROOT
		is_special_chaos_country = yes
		is_actual_nonhuman_country = yes
	}
}

secret_alliance_has_live_members = {
	any_of_scopes = {
		array = global.secret_alliance_members
		secret_alliance_member_is_live_for_root_target = yes
	}
}

secret_alliance_has_minimum_pre_reveal_members = {
	check_variable = {
		var = secret_alliance_member_count
		value = constant:secret_alliance_requirement.active_member_floor
		compare = greater_than_or_equals
	}
}

secret_alliance_can_publicly_reveal = {
	secret_alliance_target_has_active_pact = yes
	NOT = { has_country_flag = secret_alliance_publicly_revealed }
	check_variable = {
		var = secret_alliance_evidence
		value = constant:secret_alliance_reveal_threshold.public_evidence
		compare = greater_than_or_equals
	}
}

secret_alliance_can_self_reveal = {
	secret_alliance_target_has_active_pact = yes
	NOT = { has_country_flag = secret_alliance_publicly_revealed }
	check_variable = {
		var = global.secret_alliance_readiness
		value = constant:secret_alliance_reveal_threshold.self_reveal_readiness
		compare = greater_than_or_equals
	}
}

secret_alliance_counterplay_open = {
	secret_alliance_target_has_active_pact = yes
	OR = {
		has_global_flag = secret_alliance_evolution_ii_unlocked
		has_country_flag = secret_alliance_dossier_open
		has_country_flag = secret_alliance_publicly_revealed
	}
}

secret_alliance_public_crisis_open = {
	secret_alliance_target_has_active_pact = yes
	has_country_flag = secret_alliance_publicly_revealed
	NOT = { has_country_flag = secret_alliance_war_revealed }
}

secret_alliance_war_counterplay_open = {
	secret_alliance_target_has_active_pact = yes
	has_country_flag = secret_alliance_war_revealed
}

secret_alliance_has_neighboring_live_member = {
	secret_alliance_target_has_active_pact = yes
	any_neighbor_country = {
		secret_alliance_member_is_live_for_root_target = yes
	}
}

secret_alliance_can_unlock_evolution_i = {
	secret_alliance_target_has_active_pact = yes
	NOT = { has_global_flag = secret_alliance_evolution_i_unlocked }
	OR = {
		check_variable = {
			var = secret_alliance_member_count
			value = constant:secret_alliance_evolution_threshold.evolution_i_member_count
			compare = greater_than_or_equals
		}
		check_variable = {
			var = secret_alliance_suspicion
			value = constant:secret_alliance_evolution_threshold.evolution_i_suspicion
			compare = greater_than_or_equals
		}
		check_variable = {
			var = global.secret_alliance_cohesion
			value = constant:secret_alliance_evolution_threshold.evolution_i_cohesion
			compare = greater_than_or_equals
		}
	}
}

secret_alliance_can_unlock_evolution_ii = {
	secret_alliance_target_has_active_pact = yes
	has_global_flag = secret_alliance_evolution_i_unlocked
	NOT = { has_global_flag = secret_alliance_evolution_ii_unlocked }
	OR = {
		check_variable = {
			var = secret_alliance_member_count
			value = constant:secret_alliance_evolution_threshold.evolution_ii_member_count
			compare = greater_than_or_equals
		}
		check_variable = {
			var = global.secret_alliance_readiness
			value = constant:secret_alliance_evolution_threshold.evolution_ii_readiness
			compare = greater_than_or_equals
		}
		check_variable = {
			var = global.secret_alliance_cohesion
			value = constant:secret_alliance_evolution_threshold.evolution_ii_cohesion
			compare = greater_than_or_equals
		}
	}
}

secret_alliance_can_unlock_evolution_iii = {
	secret_alliance_target_has_active_pact = yes
	has_global_flag = secret_alliance_evolution_ii_unlocked
	NOT = { has_global_flag = secret_alliance_evolution_iii_unlocked }
	NOT = { has_country_flag = secret_alliance_prefire_evolution_iii_pending }
	OR = {
		check_variable = {
			var = secret_alliance_member_count
			value = constant:secret_alliance_evolution_threshold.evolution_iii_member_count
			compare = greater_than_or_equals
		}
		AND = {
			check_variable = {
				var = global.secret_alliance_readiness
				value = constant:secret_alliance_evolution_threshold.evolution_iii_readiness
				compare = greater_than_or_equals
			}
			check_variable = {
				var = secret_alliance_evidence
				value = constant:secret_alliance_evolution_threshold.evolution_iii_evidence
				compare = greater_than_or_equals
			}
		}
	}
}

secret_alliance_known_member_target = {
	secret_alliance_member_is_live_for_root_target = yes
	OR = {
		has_country_flag = secret_alliance_member_suspected
		has_country_flag = secret_alliance_member_confirmed
		has_country_flag = secret_alliance_public_leader
		ROOT = { has_country_flag = secret_alliance_publicly_revealed }
	}
}

secret_alliance_neighbor_member_target = {
	secret_alliance_known_member_target = yes
	any_neighbor_country = { tag = ROOT }
}

secret_alliance_ai_can_use_counterplay = {
	is_ai = yes
	secret_alliance_target_has_active_pact = yes
	OR = {
		check_variable = {
			var = secret_alliance_suspicion
			value = constant:secret_alliance_meter.mid
			compare = greater_than_or_equals
		}
		has_country_flag = secret_alliance_dossier_open
		has_country_flag = secret_alliance_publicly_revealed
	}
}

secret_alliance_ai_should_prepare_for_war = {
	secret_alliance_ai_can_use_counterplay = yes
	check_variable = {
		var = global.secret_alliance_readiness
		value = constant:secret_alliance_meter.high
		compare = greater_than_or_equals
	}
}
