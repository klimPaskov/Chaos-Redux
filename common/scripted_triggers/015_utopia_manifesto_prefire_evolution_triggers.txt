# Country scope, with ROOT as the Event 15 actor.

	exists = yes
	uses_normal_civilian_systems = yes
	is_major = no
	is_subject = no
	has_capitulated = no
	has_civil_war = no
	NOT = { tag = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { utopia_manifesto_candidate_has_terminal_identity = yes }
}

	any_of_scopes = {
		array = utopia_manifesto_prefire_district_candidate_states
		is_owned_and_controlled_by = ROOT
	}
}

		exists = yes

	}
}
