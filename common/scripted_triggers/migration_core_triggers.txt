# Same-chain rollback is deliberately narrower than terminal cleanup. It may
# remove only a row just staged at this physical origin before any population,
# reception, bind, history-visit beyond origin, projection, or finalize receipt.

		check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
		OR = {
			NOT = { has_variable = civilian_transfer_result }
			NOT = { check_variable = { var = civilian_transfer_result value = constant:civilian_transfer_route_result.valid compare = equals } }
		}
		OR = {
			NOT = { has_variable = civilian_transfer_actual_origin_debit }
			check_variable = { var = civilian_transfer_actual_origin_debit value = constant:humanitarian_population.zero compare = equals }
		}
		OR = {
			NOT = { has_variable = civilian_transfer_destination_actual_credit }
			check_variable = { var = civilian_transfer_destination_actual_credit value = constant:humanitarian_population.zero compare = equals }
		}
		OR = {
			NOT = { has_variable = civilian_transfer_source_reception_applied }
			check_variable = { var = civilian_transfer_source_reception_applied value = constant:humanitarian_population.zero compare = equals }
		}

			NOT = { has_variable = civilian_transfer_destination_reception_applied }

	}
}

# Safe third-country resettlement rebinds an already received, nonterminal row
# from the actual new destination state. It requires the resolver's persisted
# host and owner targets plus explicit food, reception, route, and actor proof.
# The effect changes only aligned host/destination identity; it never creates,
# debits, credits, or duplicates population.
migration_cohort_resettlement_rebind_request_is_valid = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		check_variable = { var = migration_cohort_id_request value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = migration_cohort_resettlement_rebind_request_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = migration_cohort_resettlement_food_safe_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = migration_cohort_resettlement_reception_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = migration_cohort_resettlement_route_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		check_variable = { var = migration_cohort_resettlement_actor_proven value = constant:humanitarian_runtime.zero compare = greater_than }
		has_event_target = migration_cohort_host
		event_target:migration_cohort_host = {
			humanitarian_state_is_valid = yes
			OWNER = { humanitarian_country_is_valid = yes }
		}
		has_event_target = migration_cohort_owner
		event_target:migration_cohort_owner = { humanitarian_country_is_valid = yes }
		NOT = { state = event_target:migration_cohort_host }
	}
}
