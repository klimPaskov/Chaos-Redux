# CBRN-owned chemical reservation identity and storage predicates.
# COUNTRY scope. Read-only consumers initialize the unscoped lookup handle,
# consumer and nonce. Launch authorization additionally uses the current
# CBRN action policy; a structural receipt does not authorize release by itself.

cbrn_payload_count_phase_is_bound = {
	OR = {
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.held }
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.return_pending }
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.recovery_pending }
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.quarantined }
	}
}

cbrn_payload_count_phase_is_captured = {
	OR = {
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.captured }
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.captured_quarantined }
	}
}

cbrn_payload_count_phase_is_quarantined = {
	OR = {
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.quarantined }
		check_variable = { cbrn_payload_count_phase = constant:cbrn_payload_receipt_phase.captured_quarantined }
	}
}

cbrn_payload_receipt_identity_matches = {
	check_variable = { cbrn_payload_lookup_handle > constant:cbrn_math.zero }
	has_variable = cbrn_payload_receipt_holder_id_@var:cbrn_payload_lookup_handle
	has_variable = cbrn_payload_receipt_consumer_@var:cbrn_payload_lookup_handle
	has_variable = cbrn_payload_receipt_nonce_@var:cbrn_payload_lookup_handle
	check_variable = { var = cbrn_payload_receipt_holder_id_@var:cbrn_payload_lookup_handle value = id compare = equals }
	check_variable = { var = cbrn_payload_receipt_consumer_@var:cbrn_payload_lookup_handle value = cbrn_payload_lookup_consumer compare = equals }
	check_variable = { var = cbrn_payload_receipt_nonce_@var:cbrn_payload_lookup_handle value = cbrn_payload_lookup_nonce compare = equals }
}

cbrn_payload_receipt_has_full_hold = {
	cbrn_payload_receipt_identity_matches = yes
	has_variable = cbrn_payload_receipt_required_@var:cbrn_payload_lookup_handle
	has_variable = cbrn_payload_receipt_held_@var:cbrn_payload_lookup_handle
	check_variable = { var = cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle value = constant:cbrn_payload_receipt_phase.held compare = equals }
	check_variable = { cbrn_payload_receipt_required_@var:cbrn_payload_lookup_handle > constant:cbrn_math.zero }
	check_variable = { var = cbrn_payload_receipt_held_@var:cbrn_payload_lookup_handle value = cbrn_payload_receipt_required_@var:cbrn_payload_lookup_handle compare = equals }
}

cbrn_payload_receipt_matches_requested_binding = {
	cbrn_payload_receipt_identity_matches = yes
	check_variable = { var = cbrn_payload_receipt_agent_@var:cbrn_payload_lookup_handle value = cbrn_payload_request_agent compare = equals }
	check_variable = { var = cbrn_payload_receipt_route_@var:cbrn_payload_lookup_handle value = cbrn_payload_request_route compare = equals }
	has_variable = cbrn_payload_request_site
	has_variable = cbrn_payload_request_target
	var:cbrn_payload_request_site = {
		check_variable = { var = id value = PREV.cbrn_payload_receipt_site_id_@var:cbrn_payload_lookup_handle compare = equals }
	}
	var:cbrn_payload_request_target = {
		check_variable = { var = id value = PREV.cbrn_payload_receipt_target_id_@var:cbrn_payload_lookup_handle compare = equals }
	}
}

cbrn_payload_receipt_is_settled = {
	OR = {
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.consumed }
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.returned }
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.destroyed }
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.transferred }
	}
}

cbrn_payload_receipt_can_return = {
	OR = {
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.held }
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.return_pending }
		check_variable = { cbrn_payload_receipt_phase_@var:cbrn_payload_lookup_handle = constant:cbrn_payload_receipt_phase.recovery_pending }
	}
}

# Exact profile identity still matters after equipment has left free stock.
# This checks current technology/project and profile readiness, not a second
# free-stock payment. The action's policy/target checks remain a separate gate.
cbrn_payload_receipt_technology_and_profile_are_ready = {
	set_temp_variable = { cbrn_action_agent = cbrn_payload_receipt_agent_@var:cbrn_payload_lookup_handle }
	set_temp_variable = { cbrn_action_agent_class = cbrn_payload_receipt_class_@var:cbrn_payload_lookup_handle }
	set_temp_variable = { cbrn_action_delivery_route = cbrn_payload_receipt_route_@var:cbrn_payload_lookup_handle }
	cbrn_action_agent_is_unlocked = yes
	cbrn_action_payload_profile_is_ready = yes
}

cbrn_payload_receipt_source_is_controlled = {
	has_variable = cbrn_payload_receipt_site_@var:cbrn_payload_lookup_handle
	has_variable = cbrn_payload_receipt_site_id_@var:cbrn_payload_lookup_handle
	var:cbrn_payload_receipt_site_@var:cbrn_payload_lookup_handle = {
		is_controlled_by = PREV
		check_variable = { var = id value = PREV.cbrn_payload_receipt_site_id_@var:cbrn_payload_lookup_handle compare = equals }
	}
}

# Sealed captured receipts do not bind the captor's production profile until
# integration. Original holds, pending returns and native-mismatch records do.
cbrn_air_payload_profile_has_no_reserved_stock = {
	OR = {
		NOT = { has_variable = cbrn_payload_profile_bound_count }
		check_variable = { cbrn_payload_profile_bound_count = constant:cbrn_math.zero }
	}
}

cbrn_country_has_reserved_chemical_payload = {
	OR = {
		check_variable = { cbrn_payload_held_total > constant:cbrn_math.zero }
		check_variable = { cbrn_payload_quarantined_count > constant:cbrn_math.zero }
	}
}

# COUNTRY, read-only preparation quote. Input: caller's temporary exact
# cbrn_action_agent. Stock/model/class knowledge stays with CBRN and the same
# strategic route cost used by the reservation constructor.
cbrn_air_payload_can_fund_strategic_action = {
	set_temp_variable = { cbrn_action_delivery_route = constant:cbrn_delivery_route.strategic_chemical_raid }
	set_temp_variable = { cbrn_action_payload_required = constant:cbrn_payload_route_cost.strategic_chemical_raid }
	cbrn_action_agent_is_unlocked = yes
	cbrn_action_payload_profile_is_ready = yes
	OR = {
		AND = {
			set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.choking }
			cbrn_action_agent_matches_class = yes
			cbrn_action_payload_stock_is_sufficient = yes
		}
		AND = {
			set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.blister }
			cbrn_action_agent_matches_class = yes
			cbrn_action_payload_stock_is_sufficient = yes
		}
		AND = {
			set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.nerve }
			cbrn_action_agent_matches_class = yes
			cbrn_action_payload_stock_is_sufficient = yes
		}
		AND = {
			set_temp_variable = { cbrn_action_agent_class = constant:cbrn_agent_class.incapacitating }
			cbrn_action_agent_matches_class = yes
			cbrn_action_payload_stock_is_sufficient = yes
		}
	}
}

# COUNTRY; caller supplies the temporary cbrn_payload_lookup_consumer.
# The constructor discards inherited foreign bookkeeping before use, so that
# same foreign marker must not falsely prevent an otherwise new preparation.
cbrn_payload_consumer_has_no_open_receipt = {
	OR = {
		NOT = { has_variable = cbrn_payload_ledger_owner_id }
		NOT = { check_variable = { var = cbrn_payload_ledger_owner_id value = id compare = equals } }
		NOT = { has_variable = cbrn_payload_consumer_handle_@var:cbrn_payload_lookup_consumer }
	}
}
