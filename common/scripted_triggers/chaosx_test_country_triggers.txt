# ============================================================================
# CHAOS REDUX - TEST COUNTRY TRIGGERS
# ============================================================================
# Overview:
# - Identifies states already occupied by a special-project facility.
# - Keeps each of the six CXT specializations in a distinct legal facility
#   state, including one coastal campus for the naval specialization.
# ============================================================================

chaosx_test_country_state_has_special_facility = {
	OR = {
		naval_facility > 0
		nuclear_facility > 0
		air_facility > 0
		land_facility > 0
		biowarfare_facility > 0
		cw_facility > 0
	}
}

chaosx_test_country_state_can_host_special_facility = {
	is_controlled_by = ROOT
	NOT = { chaosx_test_country_state_has_special_facility = yes }
}

# State scope. FACILITY is the actual installed building type, and ROOT must
# be the CXT country. COASTAL_REQUIRED is yes for naval, no for other types.
# Native legality checks whether ROOT and this state
# can build this exact type; the trigger does not manufacture a legal site.
chaosx_test_country_state_can_host_facility_type = {
	is_owned_by = ROOT
	OR = {
		NOT = { always = $COASTAL_REQUIRED$ }
		is_coastal = yes
	}
	chaosx_test_country_state_can_host_special_facility = yes
	can_construct_building = $FACILITY$
}
