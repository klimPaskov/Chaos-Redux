# ============================================================================
# CHAOS REDUX - TEST COUNTRY TRIGGERS
# ============================================================================
# Overview:
# - Identifies states already occupied by a special-project facility.
# - Keeps each of the six CXT specializations in a distinct legal facility
#   state, including one coastal campus for the naval specialization.
# ============================================================================

chaosx_test_country_state_has_special_facility = {
	OR = {
		naval_facility > 0
		nuclear_facility > 0
		air_facility > 0
		land_facility > 0
		biowarfare_facility > 0
		cw_facility > 0
	}
}

chaosx_test_country_state_can_host_special_facility = {
	is_controlled_by = ROOT
	NOT = { chaosx_test_country_state_has_special_facility = yes }
}

# State scope, ROOT = CXT. Input: temporary building token
# chaosx_test_country_current_facility, assigned before the helper call.
# These are shared placement preconditions. Every caller must also check
# native can_construct_building for its literal type in the outer template.
chaosx_test_country_state_meets_facility_requirements = {
	is_owned_by = ROOT
	OR = {
		NOT = { check_variable = { chaosx_test_country_current_facility = token:naval_facility } }
		is_coastal = yes
	}
	chaosx_test_country_state_can_host_special_facility = yes
}
