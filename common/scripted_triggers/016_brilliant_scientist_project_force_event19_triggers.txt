# ============================================================================
# CHAOS REDUX - EVENT 016 GENERIC EVENT 019 PROVIDER TRIGGERS
# ============================================================================
# These country-scope gates keep provider eligibility tied to the completed
# Event 016 history/runtime package, while allowing an Event 019 derivative to
# continue using the same provider after the parent identity is isolated.
# ============================================================================

brilliant_scientist_event19_clone_provider_unlocked = {
	OR = {
		AND = {
			has_country_flag = brilliant_scientist_project_force_package_state_current
			has_country_flag = brilliant_scientist_project_force_cloning_active
			brilliant_scientist_has_cloning_force_history_prototype = yes
		}
		AND = {
			has_country_flag = infantry_spawn_event16_provider_derivative
			check_variable = { infantry_spawn_derivative_family_id = constant:chaos_unit_family_event16_clone_infantry.family_id }
			check_variable = { infantry_spawn_derivative_provider_id = constant:chaos_unit_family_event16_clone_infantry.provider_id }
		}
	}
}
