# ============================================================================
# CHAOS REDUX - EVENT 038 MALTA CRUSADERS TRIGGERS
# ============================================================================
# Read-only triggers for the Malta Crusaders campaign. COUNTRY scope means the
# actor (tag MLT, carrying cosmetic identities) unless stated otherwise.
# Sections: actor and origin, anchors and Command Standing, value bands,
# council (orders, demands, incidents), expedition state, the Eleventh
# Crusade, visibility rules of the decision categories and the payment checks
# of the actions. Payment checks read the costs published by
# malta_publish_costs (global.malta_cost_*), so a displayed cost and the
# checked cost are the same number.
# ============================================================================

# ---------------------------------------------------------------------------
# Actor and origin
# ---------------------------------------------------------------------------

# COUNTRY. A live country that carries the Event 38 origin proof.
malta_is_actor = {
	exists = yes
	has_country_flag = malta_origin
}

# COUNTRY. The same, and the player or AI can still take actions.
malta_is_active_actor = {
	malta_is_actor = yes
	has_capitulated = no
}

# ANY. The event has fired and the actor is alive.
malta_campaign_is_live = {
	has_global_flag = malta_event_fired
	MLT = { malta_is_actor = yes }
}

# ---------------------------------------------------------------------------
# Anchors and Command Standing
# ---------------------------------------------------------------------------

# COUNTRY. Malta itself, held with its harbour.
malta_anchor_malta_held = {
	owns_state = 116
	controls_state = 116
}

# COUNTRY. Jerusalem is held and has been for the proof period. The counter
# lives on the actor and is advanced by the weekly pulse.
malta_anchor_jerusalem_held = {
	454 = { malta_state_is_held_by_realm = yes }
	check_variable = { var = malta_jerusalem_days value = constant:malta_standing.jerusalem_days compare = greater_than_or_equals }
}

# COUNTRY. Jerusalem is held now, whatever the counter says.
malta_holds_jerusalem_now = {
	454 = { malta_state_is_held_by_realm = yes }
}

# COUNTRY. At least one bridge state links Malta to the Holy Land by sea.
malta_anchor_sea_road_held = {
	malta_anchor_malta_held = yes
	OR = {
		any_owned_state = {
			malta_state_is_bridge_link = yes
			is_controlled_by = MLT
		}
		any_subject_country = {
			malta_is_chartered_principality = yes
			any_owned_state = {
				malta_state_is_bridge_link = yes
				is_controlled_by = PREV
			}
		}
	}
}

# COUNTRY. A port of the Holy Land command.
malta_anchor_port_held = {
	OR = {
		454 = { malta_state_is_held_by_realm = yes }
		455 = { malta_state_is_held_by_realm = yes }
	}
}

# COUNTRY. Every state of the Holy Land command is held.
malta_holds_holy_land = {
	454 = { malta_state_is_held_by_realm = yes }
	455 = { malta_state_is_held_by_realm = yes }
}

# COUNTRY. The actor holds no state of the Holy Land command.
malta_holy_land_lost = {
	NOT = {
		OR = {
			454 = { malta_state_is_held_by_realm = yes }
			455 = { malta_state_is_held_by_realm = yes }
		}
	}
}

# COUNTRY. The command is at the given step or better. Steps are the
# malta_standing constants (1 Fractured to 5 Supreme).
malta_standing_is_functional_or_better = {
	check_variable = { var = malta_standing value = constant:malta_standing.functional compare = greater_than_or_equals }
}

malta_standing_is_recognized_or_better = {
	check_variable = { var = malta_standing value = constant:malta_standing.recognized compare = greater_than_or_equals }
}

malta_standing_is_fractured = {
	check_variable = { var = malta_standing value = constant:malta_standing.fractured compare = less_than_or_equals }
}

# The half payment of the compromise answer, whatever the kind.
malta_can_pay_open_demand_half = {
	OR = {
		AND = {
			check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.treasury compare = equals }
			malta_can_pay_demand_treasury_half = yes
		}
		AND = {
			check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.command compare = equals }
			malta_can_pay_demand_command_half = yes
		}
		AND = {
			check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.levy compare = equals }
			malta_can_pay_demand_levy_half = yes
		}
		AND = {
			check_variable = { var = malta_demand_kind value = constant:malta_demand_kind.headquarters compare = equals }
			malta_can_pay_demand_headquarters_half = yes
		}
	}
}

# not the actor, not a displaced owner and not a special actor.

	NOT = { tag = MLT }
	NOT = { is_in_array = { array = global.malta_displaced_owners value = THIS } }
	is_special_chaos_country = no
	is_actual_nonhuman_country = no
	has_capitulated = no
}
