fallout_event_pilot_filters_fail_state_is_current = {
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_variable = fallout_pretransition_air_winter_shelter_capacity
	has_variable = fallout_pretransition_air_winter_exposure
	has_variable = fallout_pretransition_air_winter_adaptation
	has_variable = fallout_pretransition_air_winter_reclamation
	check_variable = { fallout_pretransition_air_winter_shelter_capacity > constant:fallout_event_217_threshold.minimum_shelter }
	check_variable = { fallout_pretransition_air_winter_exposure > constant:fallout_event_217_threshold.minimum_exposure }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	owner = {
		check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_217_threshold.minimum_filters compare = greater_than_or_equals }
		check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_217_threshold.maximum_filters compare = less_than }
		OR = {
			AND = {
				check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_217_cost.seal_filters compare = greater_than_or_equals }
				check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_217_cost.seal_medicine compare = greater_than_or_equals }
			}
			AND = {
				check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_217_cost.rotate_filters compare = greater_than_or_equals }
				check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_217_cost.rotate_food compare = greater_than_or_equals }
			}
			AND = {
				check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_217_cost.vent_scrap compare = greater_than_or_equals }
				check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_217_cost.vent_recognition compare = greater_than_or_equals }
			}
			AND = {
				check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_217_cost.household_filters compare = greater_than_or_equals }
				check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_217_cost.household_recognition compare = greater_than_or_equals }
			}
		}
	}
	NOT = { has_state_flag = fallout_event_217_filters_registry_committed }
}
