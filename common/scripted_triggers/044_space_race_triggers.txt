# ============================================================================
# CHAOS REDUX - EVENT 044 SPACE RACE TRIGGERS
# ============================================================================
# Milestone funding: a technology is an open milestone when the country can
# research it now, is not already researching it, and has no research bonus on
# it yet.  The funding decision always takes the first open milestone of the
# ladder in `space_race_fund_next_milestone`.
# ============================================================================

space_race_milestone_open_radio = {
	NOT = { has_tech = radio }
	NOT = { is_researching_technology = radio }
	can_research = radio
	NOT = { has_tech_bonus = { technology = radio } }
}
