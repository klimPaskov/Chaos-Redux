# ============================================================================
# CHAOS REDUX - CBRN ARMY HEADQUARTERS TRIGGERS
# ============================================================================
# Overview:
# - Detects exact Army HQ company composition from HQ-only subunit counts.
# - Gates abilities by affected-battalion command-power bands, policy, context,
#   and real national operating stock without estimating deployed-HQ fill.
# - Gates weekly upkeep and baseline AI HQ templates with fail-closed checks.
# - Contains no persistent effects and never selects or changes a state.
# ============================================================================

cbrn_hq_force_is_light = {
	check_variable = {
		var = num_battalions
		value = constant:cbrn_hq_force.standard_battalion_threshold
		compare = less_than
	}
}

cbrn_hq_force_is_standard = {
	check_variable = {
		var = num_battalions
		value = constant:cbrn_hq_force.standard_battalion_threshold
		compare = greater_than_or_equals
	}
	check_variable = {
		var = num_battalions
		value = constant:cbrn_hq_force.mass_battalion_threshold
		compare = less_than
	}
}

	}
}

	NOT = {
		OR = {
			has_variable = cbrn_hq_operation_code
			has_trait = cbrn_hq_preparing_operation
			has_trait = cbrn_hq_active_chemical_offensive
			has_trait = cbrn_hq_active_protective_posture

		has_war = yes
		cbrn_country_has_full_readiness = yes
		cbrn_policy_allows_battlefield_use = yes
		OR = {
			cbrn_country_has_any_chemical_payload_stock = yes
			bio_battlefield_country_has_any_agent_reserve = yes
			PREV = {
				has_trait = cbrn_hq_active_combined_overmatch
				check_variable = {
					var = cbrn_hq_operation_code
					value = constant:cbrn_hq_operation.combined_overmatch
					compare = equals
				}
			}
		}
		has_tech = theater_cbrn_headquarters
	}
}
