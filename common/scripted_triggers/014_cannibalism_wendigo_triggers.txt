# ============================================================================
# CHAOS REDUX - EVENT 014 WENDIGO AND UNIFICATION DESTINATION TRIGGERS
# ============================================================================
# Defines the exact live Wendigo identity discovered in Event 2, the durable
# selected-host contract, and a route-neutral destination trigger used by
# post-reveal warlord submission. These checks preserve the structural ZZZ
# survivor instead of attempting to reconstruct its country-scoped state.
# ============================================================================

cannibalism_is_valid_wendigo_merge_host = {
	exists = yes
	original_tag = ZZZ
	has_country_flag = zombie_outbreak_dynamic_country
	has_country_flag = weaponized_zombie_outbreak_country
	has_country_flag = weaponized_zombie_independent_outbreak
	has_country_flag = weaponized_zombie_type_wendigo
	has_country_flag = weaponized_zombie_archetype_wendigo
	NOT = { has_country_flag = cannibalism_wendigo_hannibal_country }
	NOT = { has_capitulated = yes }
	NOT = { has_idea = weaponized_zombie_wendigo_world_end }
	OR = {
		num_of_controlled_states > 0
		num_divisions > 0
	}
}

cannibalism_selected_wendigo_merge_host_is_valid = {
	has_event_target = cannibalism_wendigo_merge_host
	has_global_flag = cannibalism_wendigo_transformation_pending
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = world_end_wendigo }
	event_target:cannibalism_wendigo_merge_host = {
		cannibalism_is_valid_wendigo_merge_host = yes
	}
}

cannibalism_valid_wendigo_merge_host_exists = {
	any_country = { cannibalism_is_valid_wendigo_merge_host = yes }
}

cannibalism_is_valid_unification_destination = {
	exists = yes
	NOT = { has_capitulated = yes }
	has_global_flag = cannibalism_reveal_complete
	OR = {
		is_cannibalism_unified_country = yes
		is_cannibalism_wendigo_hannibal_country = yes
	}
}

cannibalism_has_valid_unification_destination = {
	has_event_target = cannibalism_unification_host
	event_target:cannibalism_unification_host = {
		cannibalism_is_valid_unification_destination = yes
	}
}

cannibalism_current_warlord_can_submit_without_displacing_player = {
	is_cannibalism_warlord_country = yes
	has_country_flag = cannibalism_pending_absorption
	cannibalism_has_valid_unification_destination = yes
	OR = {
		is_ai = yes
		event_target:cannibalism_unification_host = { is_ai = yes }
	}
}

cannibalism_current_warlord_can_challenge_active_unifier = {
	is_cannibalism_warlord_country = yes
	has_country_flag = cannibalism_pending_absorption
	has_country_flag = cannibalism_hierarchy_personal_tyranny
	NOT = { has_capitulated = yes }
	num_divisions > constant:cannibalism_unification.minimum_challenge_divisions
	cannibalism_has_valid_unification_destination = yes
}

cannibalism_current_wendigo_anchor_is_live = {
	has_state_flag = cannibalism_wendigo_transformation_anchor
	has_variable = cannibalism_wendigo_anchor_strength
	has_variable = cannibalism_wendigo_anchor_source_country
	has_variable = cannibalism_wendigo_anchor_source_generation
	check_variable = {
		var = cannibalism_wendigo_anchor_strength
		value = constant:cannibalism_wendigo_anchor.minimum_live_strength
		compare = greater_than_or_equals
	}
	check_variable = {
		var = cannibalism_wendigo_anchor_source_country
		value = event_target:cannibalism_wendigo_merge_host
		compare = equals
	}
	check_variable = {
		var = cannibalism_wendigo_anchor_source_generation
		value = event_target:cannibalism_wendigo_merge_host.cannibalism_actor_generation
		compare = equals
	}
	OWNER = { tag = event_target:cannibalism_wendigo_merge_host }
	CONTROLLER = { tag = event_target:cannibalism_wendigo_merge_host }
}

cannibalism_wendigo_can_start_countdown = {
	is_cannibalism_wendigo_hannibal_country = yes
	has_country_flag = cannibalism_wendigo_winter_network_active
	has_country_flag = cannibalism_wendigo_countdown_route_complete
	NOT = { has_country_flag = cannibalism_wendigo_transformation_locked }
	NOT = { has_country_flag = cannibalism_wendigo_transformation_broken }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = world_end_disabled }
	check_variable = {
		var = global.chaos_meter_value
		value = constant:cannibalism_evolution_threshold.world_end_chaos
		compare = greater_than
	}
	check_variable = {
		var = global.cannibalism_wendigo_active_anchor_count
		value = constant:cannibalism_wendigo_anchor.minimum_for_countdown
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.cannibalism_network_reach
		value = constant:cannibalism_wendigo_countdown.network_reach_gate
		compare = greater_than_or_equals
	}
	num_of_controlled_states > constant:cannibalism_wendigo_countdown.controlled_states_gate
	check_variable = {
		var = global.cannibalism_consumed_population_k
		value = constant:cannibalism_wendigo_countdown.consumed_population_k_gate
		compare = greater_than_or_equals
	}
	check_variable = {
		var = cannibalism_wendigo_winter_victories
		value = constant:cannibalism_wendigo_countdown.minimum_winter_victories
		compare = greater_than_or_equals
	}
	check_variable = {
		var = cannibalism_unified_authority
		value = constant:cannibalism_wendigo_countdown.minimum_authority
		compare = greater_than_or_equals
	}
	check_variable = {
		var = cannibalism_larder
		value = constant:cannibalism_wendigo_countdown.minimum_larder
		compare = greater_than_or_equals
	}
}

cannibalism_wendigo_can_lock_terminal_form = {
	cannibalism_wendigo_can_start_countdown = yes
	world_end_cannibalism_wendigo_scenario_enabled = yes
	has_country_flag = cannibalism_wendigo_countdown_active
	has_country_flag = cannibalism_wendigo_terminal_route_complete
	check_variable = {
		var = cannibalism_wendigo_transformation_progress
		value = constant:cannibalism_wendigo_countdown.progress_max
		compare = greater_than_or_equals
	}
}
