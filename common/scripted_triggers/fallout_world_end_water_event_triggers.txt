# The water pilot never invents an intake in event text. A candidate must carry

# source still agree with the country receiving the ordinary receipt.

fallout_event_107_opening_intake_target_is_current = {
	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.state }
	check_variable = { fallout_event_dispatch_issued_target > constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_107_target_to_check = fallout_event_dispatch_issued_target }
	set_temp_variable = { fallout_event_107_target_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_107_target_to_check = {
		set_temp_variable = { fallout_event_107_target_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		owner = { tag = ROOT }
		fallout_survival_state_identity_row_is_current = yes
		fallout_survival_state_resource_row_is_durable = yes
		has_variable = air_winter_water_security
		check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
		check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
		NOT = { has_state_flag = fallout_event_107_intake_registry_committed }
	}
	check_variable = { fallout_event_107_target_context_loaded = constant:fallout_event_runtime.one }
}

fallout_event_107_intake_registry_is_current = {
	has_variable = fallout_event_107_intake_state_id
	has_variable = fallout_event_107_intake_registry_generation
	has_variable = fallout_event_107_intake_registry_owner
	check_variable = { fallout_event_107_intake_state_id > constant:fallout_event_runtime.zero }
	check_variable = { fallout_event_107_intake_registry_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_event_107_intake_registry_owner value = ROOT compare = equals }
	set_temp_variable = { fallout_event_107_registry_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_107_intake_state_id = {
		set_temp_variable = { fallout_event_107_registry_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		has_state_flag = fallout_event_107_intake_registry_committed
		fallout_survival_state_identity_row_is_current = yes
		fallout_survival_state_resource_row_is_durable = yes
		check_variable = { fallout_survival_state_owner value = ROOT compare = equals }
		has_variable = air_winter_water_security
		check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
		check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	}
	check_variable = { fallout_event_107_registry_context_loaded = constant:fallout_event_runtime.one }
}
