# ============================================================================
# CHAOS REDUX - EVENT 021 RANDOM CIVIL WAR SUCCESSOR TRIGGERS
# ============================================================================
# These predicates prove the explicit same-crisis successor transaction and
# validate actor rows already present in the bounded Event 021 front ledger.
# They have no side effects, world iteration, random choice, or war mutation.
# ============================================================================

# COUNTRY scope on the ordinary opposition successor before white peace and
# annexation. The caller supplies event021_history_predecessor as a regular
# chain-local event target. A global host pointer or history-only receipt is
# deliberately insufficient for this proof.
event021_successor_adoption_context_valid = {
	random_civil_war_is_normal_human_country = yes
	NOT = { has_capitulated = yes }
	has_country_flag = random_civil_war_active
	has_country_flag = random_civil_war_opposition_side
	has_country_flag = random_civil_war_opposition_actor
	has_country_flag = random_civil_war_successor_claimed
	NOT = { has_country_flag = random_civil_war_event6_front_active }
	NOT = { has_event_target = event021_successor_roster_adoption_prepared }
	NOT = { has_country_flag = random_civil_war_successor_roster_pending }
	NOT = { has_country_flag = random_civil_war_successor_government_role_pending }
	has_variable = random_civil_war_crisis_id
	has_variable = random_civil_war_host_country_scope
	has_event_target = event021_history_predecessor
	var:random_civil_war_host_country_scope = {
		exists = yes
		tag = event_target:event021_history_predecessor
	}
	event_target:event021_history_predecessor = {
		exists = yes
		random_civil_war_is_normal_human_country = yes
		NOT = { has_capitulated = yes }
		NOT = { tag = PREV }
		has_country_flag = random_civil_war_active
		has_country_flag = random_civil_war_government_side
		NOT = { has_country_flag = random_civil_war_opposition_actor }
		NOT = { has_country_flag = random_civil_war_event6_host_side }
		NOT = { has_country_flag = random_civil_war_event6_secondary_host_side }
		has_variable = random_civil_war_crisis_id
		check_variable = {
			var = random_civil_war_crisis_id
			value = PREV.random_civil_war_crisis_id
			compare = equals
		}
		has_variable = random_civil_war_total_belligerent_count
		has_variable = random_civil_war_belligerent_cap
		has_variable = random_civil_war_route_evidence_count
	}
}

# COUNTRY scope on ROOT after FROM has run event021_handle_annexed_country.
# The lifecycle target is a current ROOT/FROM proof witness, not a mere saved
# successor target. The pending receipts and crisis receipt tie it to this
# adoption transaction and make a stale target from an earlier annex unusable.
event021_successor_adoption_after_annex_valid = {
	random_civil_war_is_normal_human_country = yes
	NOT = { has_capitulated = yes }
	has_country_flag = random_civil_war_active
	has_country_flag = random_civil_war_successor_claimed
	NOT = { has_country_flag = random_civil_war_event6_front_active }
	has_country_flag = random_civil_war_successor_roster_pending
	has_country_flag = random_civil_war_successor_government_role_pending
	NOT = { has_event_target = event021_successor_roster_adoption_complete }
	check_variable = {
		var = event021_lifecycle_successor_transfer
		value = constant:random_civil_war_value.one
		compare = equals
	}
	has_variable = random_civil_war_crisis_id
	has_variable = random_civil_war_successor_adoption_crisis_id
	check_variable = {
		var = random_civil_war_crisis_id
		value = random_civil_war_successor_adoption_crisis_id
		compare = equals
	}
	has_event_target = event021_successor_roster_adoption_successor
	event_target:event021_successor_roster_adoption_successor = {
		exists = yes
		tag = PREV
	}
	has_event_target = event021_successor_roster_adoption_prepared
	event_target:event021_successor_roster_adoption_prepared = {
		exists = yes
		tag = PREV
	}
	has_event_target = event021_lifecycle_successor
	event_target:event021_lifecycle_successor = {
		exists = yes
		tag = PREV
		has_variable = random_civil_war_crisis_id
		check_variable = {
			var = random_civil_war_crisis_id
			value = PREV.random_civil_war_crisis_id
			compare = equals
		}
	}
}

# COUNTRY scope on a row actor from global.random_civil_war_front_actors. The
# caller already proved that the row's crisis id and host link match; this
# trigger enforces the actor's live role and registry receipt. Event 006 actors
# are eligible when valid and retain their Event 006 identity unchanged.
event021_successor_roster_actor_valid = {
	exists = yes
	random_civil_war_is_normal_human_country = yes
	NOT = { has_capitulated = yes }
	check_variable = { num_owned_states > constant:random_civil_war_value.zero }
	has_country_flag = random_civil_war_active
	has_country_flag = random_civil_war_opposition_side
	has_country_flag = random_civil_war_opposition_actor
	NOT = { tag = event_target:event021_successor_roster_adoption_successor }
	has_variable = random_civil_war_crisis_id
	check_variable = {
		var = random_civil_war_crisis_id
		value = event021_successor_roster_crisis_id
		compare = equals
	}
	has_country_flag = random_civil_war_front_registered
	has_variable = random_civil_war_front_id
	check_variable = {
		var = random_civil_war_front_id
		value = event021_successor_roster_row_front_id
		compare = equals
	}
	random_civil_war_front_id_is_registered = yes
}
