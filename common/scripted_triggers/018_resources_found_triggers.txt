# ============================================================================
# CHAOS REDUX - EVENT 018 RESOURCES FOUND TRIGGERS
# ============================================================================
# Reusable state and country validation for resource-field selection, exact
# ledgers, country-selected fields, transaction locks, suspension and closure,
# contracts, commissions, border stages, evolution settings, cave emergence,
# captured-state capacity, and world-end verification inputs.
#
# Scope contracts:
# - Field, ledger, closure, border, capacity, and cave-origin triggers use STATE.
# - Owner, selected-field, cave-strength-output, and terminal triggers use COUNTRY.
# - Evolution setting and chronology-visibility triggers can run in any scope.
#   They mirror shared flags and the chaos-tier contract without mutating temp
#   state.
# ============================================================================

# ---------------------------------------------------------------------------
# Field identity, selection, and ledger integrity
# ---------------------------------------------------------------------------

resources_found_is_active_field = {
	has_state_flag = resources_found_field_active
	has_variable = resources_found_field_sequence
	NOT = { has_state_flag = resources_found_field_closed_permanently }
	NOT = { has_state_flag = resources_found_field_converted_to_cave_origin }
}

resources_found_has_event_resource_ledger = {
	OR = {
		check_variable = { resources_found_added_oil > constant:resources_found_value.zero }
		check_variable = { resources_found_added_aluminium > constant:resources_found_value.zero }
		check_variable = { resources_found_added_rubber > constant:resources_found_value.zero }
		check_variable = { resources_found_added_tungsten > constant:resources_found_value.zero }
		check_variable = { resources_found_added_steel > constant:resources_found_value.zero }
		check_variable = { resources_found_added_chromium > constant:resources_found_value.zero }
	}
}

resources_found_resource_ledger_is_exactly_reversible = {
	OR = {
		NOT = { has_variable = resources_found_added_oil }
		check_variable = { var = resources_found_added_oil value = constant:resources_found_value.zero compare = greater_than_or_equals }
	}
	OR = {
		NOT = { has_variable = resources_found_added_oil }
		check_variable = { var = resource@oil value = resources_found_added_oil compare = greater_than_or_equals }
	}
	OR = {
		NOT = { has_variable = resources_found_added_aluminium }
		check_variable = { var = resources_found_added_aluminium value = constant:resources_found_value.zero compare = greater_than_or_equals }

		NOT = { has_variable = resources_found_added_rubber }
		check_variable = { var = resources_found_added_rubber value = constant:resources_found_value.zero compare = greater_than_or_equals }
	}
	OR = {
		NOT = { has_variable = resources_found_added_rubber }
		check_variable = { var = resource@rubber value = resources_found_added_rubber compare = greater_than_or_equals }
	}
	OR = {
		NOT = { has_variable = resources_found_added_tungsten }
		check_variable = { var = resources_found_added_tungsten value = constant:resources_found_value.zero compare = greater_than_or_equals }
	}
	OR = {
		NOT = { has_variable = resources_found_added_tungsten }
		check_variable = { var = resource@tungsten value = resources_found_added_tungsten compare = greater_than_or_equals }
	}
	OR = {
		NOT = { has_variable = resources_found_added_steel }
		check_variable = { var = resources_found_added_steel value = constant:resources_found_value.zero compare = greater_than_or_equals }
	}

		NOT = { has_variable = resources_found_added_steel }

		NOT = { has_variable = resources_found_added_chromium }

resources_found_is_valid_new_field_state = {
	impassable = no
	OWNER = {
		exists = yes
		NOT = { is_special_chaos_country = yes }
		NOT = { is_actual_nonhuman_country = yes }
	}
	CONTROLLER = { exists = yes }
	is_owned_and_controlled_by = OWNER
	NOT = { has_global_flag = world_end }
	NOT = { has_state_flag = resources_found_field_closed_permanently }
	NOT = { has_state_flag = resources_found_field_converted_to_cave_origin }
	NOT = { has_state_flag = resources_found_incompatible_terminal_site }
	NOT = { has_state_flag = resources_found_field_transaction_lock }
	NOT = { resources_found_is_active_field = yes }
	OR = {
		check_variable = { state_population_k > constant:resources_found_threshold.minimum_state_population_thousands }
		resources_found_state_has_standard_resource_presence = yes
		is_coastal = yes
	}
}
