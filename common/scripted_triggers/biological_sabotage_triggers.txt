# ============================================================================
# Overview:
# - Gates exact-state covert contamination of public food, water, and medical
#   supply chains by project, readiness, use policy, payload, support equipment,
#   Command Power, relationship, target profile, and operation ledger.
# - Keeps every helper private to the ordinary biological sabotage subsystem.
# - Gives route-aware AI a fail-closed safety gate and exact selected-state
#   evidence without introducing a periodic country or state scan.

bio_sabotage_actor_cooldown_is_ready = {
	OR = {
		NOT = { has_variable = bio_sabotage_next_allowed_day }
		check_variable = {
			var = bio_sabotage_next_allowed_day
			value = global.num_days
			compare = less_than_or_equals
		}
	}
}

# State scope with the acting country as ROOT. Belligerent targets require
# strategic authority. Neutral targets require unrestricted authority.
bio_sabotage_state_relationship_is_valid = {
	CONTROLLER = {
		exists = yes
		NOT = { tag = ROOT }
		NOT = { is_subject_of = ROOT }
		ROOT = {
			NOT = { is_in_faction_with = PREV }
			NOT = { is_subject_of = PREV }
			PREV = { set_temp_variable = { cbrn_retaliation_target_id = id } }
			OR = {
				AND = {
					has_war_with = PREV
					cbrn_policy_allows_strategic_use_against_target_id = yes
				}
				AND = {
					NOT = { has_war_with = PREV }
					cbrn_policy_allows_extreme_use = yes
				}
			}
		}
	}
}
