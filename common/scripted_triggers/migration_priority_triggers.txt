# Cohort-priority validation for organized evacuation receipts.

migration_cohort_priority_is_valid = {
	OR = {
		check_variable = { var = migration_cohort_record_priority value = constant:migration_cohort_priority.general compare = equals }
		check_variable = { var = migration_cohort_record_priority value = constant:migration_cohort_priority.vulnerable compare = equals }
		check_variable = { var = migration_cohort_record_priority value = constant:migration_cohort_priority.industrial compare = equals }
		check_variable = { var = migration_cohort_record_priority value = constant:migration_cohort_priority.administrative compare = equals }
	}
}


migration_evacuate_administration_cost_is_affordable = {
	OR = {
		AND = {
			check_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one }
			migration_evacuate_administration_cost_is_affordable_in_context = yes
		}
		AND = {
			NOT = { check_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one } }
			set_temp_variable = { humanitarian_action_actor = ROOT }
			set_temp_variable = { humanitarian_action_target = FROM }
			migration_evacuate_administration_cost_is_affordable_in_context = yes
		}
	}
}

migration_evacuate_administration_cost_is_affordable_in_context = {
	hidden_trigger = {
			set_temp_variable = { migration_transport_fuel_land = constant:migration_transport_cost.standard_fuel_land }
			set_temp_variable = { migration_transport_fuel_sea = constant:migration_transport_cost.standard_fuel_sea }
			set_temp_variable = { migration_transport_convoys_sea = constant:migration_transport_cost.standard_convoys_sea }
			set_temp_variable = { migration_transport_target = var:humanitarian_action_target }
			migration_transport_state_cost_is_affordable = yes
			NOT = {
				set_temp_variable = { migration_transport_target = var:humanitarian_action_target }
				set_temp_variable = { migration_transport_receipt = token:migration_evacuate_administration }
				migration_transport_state_receipt_is_paid = yes
			}
			humanitarian_cost_train_light_is_affordable = yes
		}
}

migration_evacuate_civilians_can_start = {
	OR = {
		AND = {
			check_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one }
			migration_evacuate_civilians_can_start_in_context = yes
		}
		AND = {
			NOT = { check_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one } }
			set_temp_variable = { humanitarian_action_actor = ROOT }
			set_temp_variable = { humanitarian_action_target = FROM }
			migration_evacuate_civilians_can_start_in_context = yes
		}
	}
}

migration_evacuate_civilians_can_start_in_context = {
	hidden_trigger = { var:humanitarian_action_target = { is_controlled_by = var:humanitarian_action_actor migration_state_has_real_migration_issue = yes } }
		hidden_trigger = {
			NOT = { has_country_flag = humanitarian_corridor_offer_pending }
			migration_reception_has_room_for_evacuation = yes
			var:humanitarian_action_target = {
				is_controlled_by = var:humanitarian_action_actor
				humanitarian_state_is_valid = yes
				has_state_flag = migration_displacement_active
				has_state_flag = migration_evacuation_prepared
			}
			humanitarian_cost_train_light_is_affordable = yes
			migration_evacuate_civilians_cost_is_affordable_in_context = yes
			NOT = { var:humanitarian_action_target = { has_state_flag = migration_evacuate_civilians_cost_paid } }
		}
}
