# Supported scope: COUNTRY.
fallout_country_technology_regression_is_current = {
	has_country_flag = fallout_technology_regressed
	has_variable = fallout_technology_regression_generation
	has_variable = fallout_technology_regression_schema_version
	check_variable = {
		var = fallout_technology_regression_generation
		value = global.fallout_transition_generation
		compare = equals
	}
	check_variable = {
		var = fallout_technology_regression_schema_version
		value = constant:fallout_technology_regression.version
		compare = equals
	}
}

# Supported scope: any. The world rewrite cannot reveal the map until every
# live country has received the same generation-bound regression receipt.
fallout_all_live_country_technology_rows_are_current = {
	NOT = {
		any_country = {
			exists = yes
			NOT = { fallout_country_technology_regression_is_current = yes }
		}
	}
}
