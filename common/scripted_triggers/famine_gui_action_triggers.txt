# Shared native/panel operation gates. Internal identity and phase checks stay hidden.
# Visible requirement clauses retain the native decision custom tooltips.

famine_release_reserves_action_can_start = {
	set_temp_variable = { humanitarian_action_actor = ROOT }
	hidden_trigger = {
		has_variable = humanitarian_action_target
		var:humanitarian_action_target = {
			humanitarian_state_is_valid = yes
		}
		set_temp_variable = {
			humanitarian_action_target_id = constant:humanitarian_runtime.zero
		}
		var:humanitarian_action_target = {
			set_temp_variable = {
				humanitarian_action_target_id = id
			}
		}
		NOT = {
			has_variable = famine_action_due_release_reserves_@var:humanitarian_action_target_id
		}
		OR = {
			NOT = {
				has_variable = famine_action_cooldown_release_reserves_@var:humanitarian_action_target_id
			}
			check_variable = {
				var = global.date
				value = famine_action_cooldown_release_reserves_@var:humanitarian_action_target_id
				compare = greater_than_or_equals
			}
		}
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
			famine_state_has_visible_food_problem = yes
		}
		NOT = {
			has_country_flag = humanitarian_corridor_offer_pending
		}
		famine_decision_problem_is_active = yes
		OR = {
			NOT = {
				has_country_flag = famine_decision_resolution
			}
			famine_decision_problem_is_active = yes
		}
		var:humanitarian_action_target = {
			famine_state_can_release_reserves = yes
		}
	}
	hidden_trigger = {
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
			famine_state_has_visible_food_problem = yes
		}
	}
	custom_trigger_tooltip = {
		tooltip = famine_release_reserves_target_tt
		var:humanitarian_action_target = {
			famine_state_can_release_reserves = yes
		}
	}
	hidden_trigger = {
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
		}
	}
	hidden_trigger = {
		famine_release_reserves_cost_is_affordable_in_context = yes
	}
	hidden_trigger = {
		NOT = {
			var:humanitarian_action_target = {
				if = {
					limit = { humanitarian_state_is_valid = yes }
					has_state_flag = famine_release_reserves_cost_paid
				}
				else = { always = no }
			}
		}
	}
}

famine_release_reserves_action_pending_is_current = {
	hidden_trigger = {
		has_variable = famine_action_generation_release_reserves_@var:humanitarian_action_target_id
		check_variable = {
			famine_action_generation_release_reserves_@var:humanitarian_action_target_id = humanitarian_action_row_id
		}
		var:humanitarian_action_target = {
			has_variable = famine_release_reserves_operation_generation

				famine_release_reserves_operation_generation = humanitarian_action_row_id
			}
			has_variable = famine_release_reserves_operation_actor
			var:famine_release_reserves_operation_actor = {
				exists = yes
				tag = ROOT
			}
		}
	}

famine_conceal_crisis_action_can_start = {
	set_temp_variable = { humanitarian_action_actor = ROOT }
	hidden_trigger = {
		has_variable = humanitarian_action_target
		var:humanitarian_action_target = {
			humanitarian_state_is_valid = yes
		}
		set_temp_variable = {
			humanitarian_action_target_id = constant:humanitarian_runtime.zero
		}
		var:humanitarian_action_target = {
			set_temp_variable = {
				humanitarian_action_target_id = id
			}
		}
		NOT = {
			has_variable = famine_action_due_conceal_crisis_@var:humanitarian_action_target_id
		}
		OR = {
			NOT = {
				has_variable = famine_action_cooldown_conceal_crisis_@var:humanitarian_action_target_id
			}
			check_variable = {
				var = global.date
				value = famine_action_cooldown_conceal_crisis_@var:humanitarian_action_target_id
				compare = greater_than_or_equals
			}
		}
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
			famine_state_has_visible_food_problem = yes
		}
		NOT = {
			has_country_flag = humanitarian_corridor_offer_pending
		}
		famine_decision_problem_is_active = yes
		OR = {
			NOT = {
				has_country_flag = famine_decision_resolution
			}
			famine_decision_problem_is_active = yes
		}
		OR = {
			has_government = fascism
			has_government = communism
			has_government = neutrality
		}
		var:humanitarian_action_target = {
			has_state_flag = famine_food_security_active
			NOT = {
				has_state_flag = famine_crisis_concealed
			}
			OR = {
				NOT = {
					has_variable = famine_profile_extraction
				}
				NOT = {
					check_variable = {
						var = famine_profile_extraction
						value = constant:famine_food_component.zero
						compare = greater_than
					}
				}
			}
		}
	}
	hidden_trigger = {
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
			famine_state_has_visible_food_problem = yes
		}
	}
	custom_trigger_tooltip = {
		tooltip = famine_conceal_crisis_target_tt
		var:humanitarian_action_target = {
			humanitarian_state_is_valid = yes
			has_state_flag = famine_food_security_active
			NOT = {
				has_state_flag = famine_crisis_concealed
			}
		}
	}
	OR = {
		has_government = fascism
		has_government = communism
		has_government = neutrality
	}
	hidden_trigger = {
		var:humanitarian_action_target = {
			is_controlled_by = ROOT
		}
	}
	hidden_trigger = {
		famine_conceal_crisis_cost_is_affordable_in_context = yes
	}
	hidden_trigger = {
		NOT = {
			var:humanitarian_action_target = {
				if = {
					limit = { humanitarian_state_is_valid = yes }
					has_state_flag = famine_conceal_crisis_cost_paid
				}
				else = { always = no }
			}
		}
	}
}
