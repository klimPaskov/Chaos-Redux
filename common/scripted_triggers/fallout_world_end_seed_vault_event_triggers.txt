# ============================================================================
# CHAOS REDUX - FALLOUT SEED VAULT TRIGGERS
# ============================================================================
# The Seed Vault Custody candidate requires an existing Air Winter seed memory,
# a current state survival row, and a real produced snapshot. It never invents
# a state target after the transition.
# ============================================================================

fallout_event_pilot_seed_vault_state_is_current = {
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_variable = fallout_pretransition_air_winter_reclamation
	check_variable = { fallout_pretransition_air_winter_reclamation > constant:fallout_event_188_threshold.minimum_reclamation }
	check_variable = { fallout_pretransition_air_winter_adaptation > constant:fallout_event_188_threshold.minimum_adaptation }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	OR = {
		has_state_flag = air_winter_memory_seed_vaults
		has_state_flag = air_winter_memory_seed_vaults_flourished
		has_state_flag = air_winter_memory_seed_vaults_failed
	}
	owner = {
		check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_188_threshold.minimum_food compare = greater_than_or_equals }
		check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_188_cost.minimum_scrap compare = greater_than_or_equals }
	}
	NOT = { has_state_flag = fallout_event_188_seed_registry_committed }
}

fallout_event_188_opening_seed_target_is_current = {
	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.state }
	check_variable = { fallout_event_dispatch_issued_target > constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_188_target_to_check = fallout_event_dispatch_issued_target }
	set_temp_variable = { fallout_event_188_target_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_188_target_to_check = {
		set_temp_variable = { fallout_event_188_target_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		owner = { tag = ROOT }
		fallout_event_pilot_seed_vault_state_is_current = yes
	}
	check_variable = { fallout_event_188_target_context_loaded = constant:fallout_event_runtime.one }
}

fallout_event_188_seed_registry_is_current = {
	has_variable = fallout_event_188_seed_state_id
	has_variable = fallout_event_188_seed_registry_generation
	has_variable = fallout_event_188_seed_registry_owner
	check_variable = { fallout_event_188_seed_state_id > constant:fallout_event_runtime.zero }
	check_variable = { fallout_event_188_seed_registry_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_event_188_seed_registry_owner value = ROOT compare = equals }
	set_temp_variable = { fallout_event_188_seed_registry_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_188_seed_state_id = {
		set_temp_variable = { fallout_event_188_seed_registry_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		has_state_flag = fallout_event_188_seed_registry_committed
		fallout_survival_state_identity_row_is_current = yes
		fallout_survival_state_resource_row_is_durable = yes
		check_variable = { fallout_survival_state_owner value = ROOT compare = equals }
		has_variable = fallout_event_188_seed_reserve
		check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
		check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	}
	check_variable = { fallout_event_188_seed_registry_context_loaded = constant:fallout_event_runtime.one }
}
