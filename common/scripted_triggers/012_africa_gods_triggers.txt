# ============================================================================
# CHAOS REDUX - EVENT 012 GODS OF AFRICA SCRIPTED TRIGGERS
# ============================================================================
# These triggers are intentionally owner-local. Event 012 supplies the
# authoritative global event target `africa_host`; this file only validates the
# Gods-of-Africa generation, bounded participant registry, frozen demand
# contracts, and punishment safety gates used by the owner and its decisions.
# ============================================================================

# COUNTRY scope on any country. The active host pointer is shared with the
# existing Event 012 Charter layer and is never replaced by this subsystem.
gods_of_africa_has_current_host = {
	has_event_target = africa_host
	event_target:africa_host = {
		exists = yes
		has_country_flag = africa_unifier_host
		has_country_flag = africa_host_commit_completed
		has_variable = africa_host_generation
	}
}

# COUNTRY scope on a participant. Unlike the host-only active-host trigger,
# this predicate validates the authoritative host through its event target and
# is therefore safe to reuse in participant decisions and missions.
gods_of_africa_participant_host_is_active = {
	gods_of_africa_has_current_host = yes
	has_global_flag = africa_event_active
	has_global_flag = africa_is_one
	has_global_flag = africa_evolution_i_logged
	event_target:africa_host = {
		has_country_flag = gods_of_africa_active
		has_country_flag = gods_of_africa_activation_committed
		NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
		NOT = { has_capitulated = yes }
	}
}

# COUNTRY scope on the committed Event 012 host.
gods_of_africa_active_host_is_valid = {
	africa_is_current_host = yes
	NOT = { has_capitulated = yes }
	has_global_flag = africa_event_active
	has_global_flag = africa_is_one
	has_global_flag = africa_evolution_i_logged
	NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
	has_country_flag = gods_of_africa_active
	has_country_flag = gods_of_africa_activation_committed
	has_variable = gods_of_africa_restoration_generation
	check_variable = {
		var = gods_of_africa_restoration_generation
		value = africa_host_generation
		compare = equals
	}
}

# COUNTRY scope on the host. The one-time consolidation clock must be started
# by the Event 012 evolution-I owner path; no generic world tick owns it.
gods_of_africa_system_can_activate = {
	africa_is_current_host = yes
	NOT = { has_capitulated = yes }
	has_global_flag = africa_event_active
	has_global_flag = africa_is_one
	has_global_flag = africa_evolution_i_logged
	NOT = { has_country_flag = gods_of_africa_active }
	NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
	NOT = { has_country_flag = gods_of_africa_activation_committed }
	# A defeated unifier's delayed activation must remain terminal for this
	# generation.  A successor host receives a new generation and the
	# initializer explicitly clears the old loss receipt before restarting.
	NOT = { has_country_flag = gods_of_africa_cancelled_on_unifier_loss }
	has_variable = gods_of_africa_restoration_generation
	check_variable = {
		var = gods_of_africa_restoration_generation
		value = africa_host_generation
		compare = equals
	}
}

gods_of_africa_consolidation_ready = {
	gods_of_africa_system_can_activate = yes
	has_country_flag = gods_of_africa_consolidation_pending
	has_variable = gods_of_africa_consolidation_due_day
	check_variable = {
		var = global.num_days

# COUNTRY scope on a possible participant. Only majors and the player can
# enter the registry; the host and special/non-human actors remain excluded.
# Geography is deliberately not part of enrollment: Event 012's participant
# pool is global, so a valid major/player outside Africa can still receive a
# tribute contract and carry the relationship into final settlement.
gods_of_africa_participant_is_valid = {
	exists = yes
	NOT = { has_capitulated = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { has_country_flag = africa_unifier_host }
	OR = {
		is_major = yes
		is_ai = no
	}
}

# Once enrolled, a country remains in the relationship ledger after losing
# major status. Only capitulation, deletion, special/non-human conversion, or
# replacement by the unifier invalidates the live entry.
gods_of_africa_participant_is_live = {
	exists = yes
	NOT = { has_capitulated = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { has_country_flag = africa_unifier_host }
}

gods_of_africa_participant_generation_is_current = {
	gods_of_africa_participant_is_live = yes
	has_country_flag = gods_of_africa_participant
	has_variable = gods_of_africa_participant_generation
	has_event_target = africa_host
	event_target:africa_host = {
		has_variable = africa_host_generation
	}
	check_variable = {
		var = gods_of_africa_participant_generation
		value = event_target:africa_host.africa_host_generation
		compare = equals
	}
}

gods_of_africa_participant_can_register = {
	gods_of_africa_participant_is_valid = yes
	gods_of_africa_has_current_host = yes
	event_target:africa_host = {
		has_country_flag = gods_of_africa_active
		NOT = { has_country_flag = gods_of_africa_final_settlement_committed }
		# The host scope is nested here, so PREV remains the country being tested.
		NOT = { is_in_array = { array = gods_of_africa_participants value = PREV } }
		OR = {
			NOT = { has_variable = gods_of_africa_participant_count }
			check_variable = { gods_of_africa_participant_count < constant:gods_of_africa_timing.max_participants }
		}
	}
}

gods_of_africa_participant_is_registered = {
	gods_of_africa_participant_generation_is_current = yes
	event_target:africa_host = {
		is_in_array = { array = gods_of_africa_participants value = PREV }
	}
}

		value = event_target:africa_host.africa_host_generation
		compare = equals
	}

	}
}

gods_of_africa_demand_contract_is_current = {
	# A participant may still retain the old generation pointer for the short
	# interval before the capitulation/settlement cleanup runs.  Validate the
	# host through its event target rather than the host-only
	# `gods_of_africa_active_host_is_valid` trigger, whose ROOT comparison would
	# (correctly) reject a participant scope.
	gods_of_africa_participant_host_is_active = yes
	gods_of_africa_participant_has_active_demand = yes
	has_variable = gods_of_africa_demand_family
	has_variable = gods_of_africa_demand_amount
	has_variable = gods_of_africa_demand_deadline
	check_variable = {
		var = gods_of_africa_demand_deadline
		value = global.num_days
		compare = greater_than
	}
}

# Same immutable contract check without a future-deadline requirement. The
# owner-local expiry pass and mission timeout use this resolver so an expired
# demand can still record its failure and punishment exactly once.
gods_of_africa_demand_contract_can_resolve = {
	gods_of_africa_participant_has_active_demand = yes
	has_variable = gods_of_africa_demand_family
	has_variable = gods_of_africa_demand_amount
	has_variable = gods_of_africa_demand_deadline
}

gods_of_africa_participant_can_receive_demand = {
	gods_of_africa_participant_host_is_active = yes
	gods_of_africa_participant_generation_is_current = yes
	# Tribute is a political compact, not a wartime tax. A country at war must
	# finish its current conflict before a fresh ordinary demand is issued.
	NOT = { has_war = yes }
	NOT = { has_country_flag = gods_of_africa_defiant }
	NOT = { has_country_flag = gods_of_africa_reconciliation_pending }
	NOT = { has_country_flag = gods_of_africa_final_outcome_committed }
	NOT = { has_country_flag = gods_of_africa_demand_active }
	OR = {
		NOT = { has_variable = gods_of_africa_next_demand_date }
		check_variable = {
			var = global.num_days
			value = gods_of_africa_next_demand_date
			compare = greater_than_or_equals
		}
	}
}

			OR = {

		}

# COUNTRY scope on the host. Family is supplied in the host variable
# `gods_of_africa_payment_family`; this trigger is used for late payment and
# punishment dispatch validation.

# COUNTRY scope on a participant. A voluntary offer is available only outside
# an active demand, when the owner has a current material need and the sender
# can make the fixed shipment while keeping its protected operating reserve.
# The two branches mirror the deterministic choice in the submit effect so a
# stale UI quote can only fail closed.
gods_of_africa_participant_can_offer_aid = {
	gods_of_africa_participant_generation_is_current = yes
	NOT = { has_country_flag = gods_of_africa_demand_active }
	NOT = { has_country_flag = gods_of_africa_defiant }
	NOT = { has_country_flag = gods_of_africa_reconciliation_pending }
	NOT = { has_country_flag = gods_of_africa_final_outcome_committed }
	NOT = { has_war_with = event_target:africa_host }
	has_event_target = africa_host
	OR = {
		NOT = { has_variable = gods_of_africa_next_aid_day }
		check_variable = {
			var = global.num_days
			value = gods_of_africa_next_aid_day
			compare = greater_than_or_equals
		}
	}
	event_target:africa_host = {
		gods_of_africa_active_host_is_valid = yes
		OR = {
			check_variable = { gods_of_africa_need_land_equipment > constant:gods_of_africa_value.zero }
			check_variable = { gods_of_africa_need_fuel > constant:gods_of_africa_value.zero }
		}
	}
	OR = {
		AND = {
			event_target:africa_host = {
				check_variable = { gods_of_africa_need_land_equipment > constant:gods_of_africa_value.zero }
			}
			NOT = { has_equipment = { infantry_equipment < constant:gods_of_africa_transaction.voluntary_aid_equipment_capacity } }
		}
		AND = {
			event_target:africa_host = {
				check_variable = { gods_of_africa_need_fuel > constant:gods_of_africa_value.zero }
			}
			NOT = { has_fuel < constant:gods_of_africa_transaction.voluntary_aid_fuel_capacity }
		}
	}
}

# COUNTRY scope on a participant. Partial payment is a measured half-shipment
# of the frozen contract and is legal only while a positive balance remains.
# The amount is precomputed when the contract is created, so the decision
# surface never mutates state merely to evaluate affordability.
gods_of_africa_partial_payment_is_affordable = {
	gods_of_africa_demand_contract_is_current = yes
	has_variable = gods_of_africa_partial_payment_amount
	has_variable = gods_of_africa_partial_capacity_requirement
	check_variable = { gods_of_africa_partial_payment_amount > constant:gods_of_africa_value.zero }
	check_variable = {
		var = gods_of_africa_partial_payment_amount
		value = gods_of_africa_demand_remaining
		compare = less_than
	}
	OR = {
		NOT = { has_variable = gods_of_africa_next_partial_payment_day }
		check_variable = {
			var = global.num_days
			value = gods_of_africa_next_partial_payment_day
			compare = greater_than_or_equals
		}
	}
	OR = {
		AND = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.land_equipment }
			NOT = { has_equipment = { infantry_equipment < gods_of_africa_partial_capacity_requirement } }
		}
		AND = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.support_equipment }
			NOT = { has_equipment = { support_equipment < gods_of_africa_partial_capacity_requirement } }
		}
		AND = {
			OR = {
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.mobile_artillery }
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.armor }
			}
			NOT = { has_equipment = { motorized_equipment < gods_of_africa_partial_capacity_requirement } }
		}
		AND = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.aircraft }
			NOT = { has_equipment = { fighter_equipment_1 < gods_of_africa_partial_capacity_requirement } }
		}
		AND = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.fuel }
			NOT = { has_fuel < gods_of_africa_partial_capacity_requirement }
		}
		AND = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.transport }
			OR = {
				NOT = { has_equipment = { convoy_1 < gods_of_africa_partial_capacity_requirement } }
				NOT = { has_equipment = { train_equipment < gods_of_africa_partial_capacity_requirement } }
			}
		}
		AND = {
			check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.manpower }
			NOT = { has_manpower < gods_of_africa_partial_capacity_requirement }
		}
		AND = {
			OR = {
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.industry_repair }
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.resources }
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.recognition }
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.territory }
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.alliance_access }
				check_variable = { gods_of_africa_demand_family = constant:gods_of_africa_demand_family.emergency_relief }
			}
			NOT = { has_political_power < gods_of_africa_partial_capacity_requirement }
		}
	}
}
