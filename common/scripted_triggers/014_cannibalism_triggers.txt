is_cannibalism_warlord_country = {
	has_country_flag = cannibalism_event_created_warlord
	OR = {
		tag = CBA
		tag = CBB
		tag = CBC
		tag = CBD
		tag = CBE
		tag = CBF
		tag = CBG
		tag = CBH
	}
}

is_cannibalism_reusable_warlord_slot_country = {
	OR = {
		original_tag = CBA
		original_tag = CBB
		original_tag = CBC
		original_tag = CBD
		original_tag = CBE
		original_tag = CBF
		original_tag = CBG
		original_tag = CBH
	}
}

is_cannibalism_unified_country = {
	tag = CBL
	has_country_flag = cannibalism_unified_country
}

is_cannibalism_wendigo_hannibal_country = {
	has_country_flag = cannibalism_wendigo_hannibal_country
	has_country_flag = weaponized_zombie_type_wendigo
	original_tag = ZZZ
}

is_cannibalism_country = {
	OR = {
		is_cannibalism_warlord_country = yes
		is_cannibalism_unified_country = yes
		is_cannibalism_wendigo_hannibal_country = yes
	}
}

is_cannibalism_active_ordinary_country = {
	has_country_flag = cannibalism_country_active
	NOT = { is_cannibalism_country = yes }
	NOT = { has_country_flag = cannibalism_local_victory }
	NOT = { has_country_flag = cannibalism_country_cleanup_complete }
}

is_cannibalism_active_country = {
	OR = {
		is_cannibalism_active_ordinary_country = yes
		is_cannibalism_country = yes
	}
	NOT = { has_country_flag = cannibalism_country_cleanup_complete }
}

cannibalism_country_can_continue_pulse = {
	exists = yes
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = cannibalism_cleanup_complete }
	NOT = { has_country_flag = cannibalism_retired_pulse_callback }
	NOT = { is_cannibalism_reusable_warlord_slot_country = yes }
	OR = {
		is_cannibalism_active_country = yes
		has_country_flag = cannibalism_recovery_coordinator
	}
}

cannibalism_has_valid_pulse_anchor = {
	has_event_target = cannibalism_pulse_anchor
	event_target:cannibalism_pulse_anchor = {
		exists = yes
		has_country_flag = cannibalism_pulse_anchor

	impassable = no
	check_variable = {
		var = state_population_k
		value = constant:cannibalism_state_threshold.minimum_population_k
		compare = greater_than_or_equals
	}
	NOT = { has_state_category = wasteland }
	NOT = { death_is_consumed_state = yes }
	NOT = { has_dynamic_modifier = { modifier = nuclear_fallout_state } }
	NOT = { has_state_flag = bioweapon_contaminated }
	NOT = { has_state_flag = cannibalism_state_stabilized }
	OWNER = {
		exists = yes
		NOT = { is_actual_nonhuman_country = yes }
	}
}

	NOT = { cannibalism_is_ineligible_special_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_cannibalism_active_country = yes }
	NOT = { has_country_flag = cannibalism_local_victory }
	NOT = { has_country_flag = cannibalism_external_reinfection_protected }

cannibalism_country_has_isolated_theater = {
	any_controlled_state = {
		cannibalism_state_is_valid_origin = yes
		OR = {
			is_island_state = yes
			NOT = { is_in_home_area = yes }
		}
	}
}

cannibalism_country_has_damaged_supply_theater = {
	any_controlled_state = {
		cannibalism_state_is_valid_origin = yes
		OR = {
			infrastructure < constant:cannibalism_host_threshold.low_infrastructure
			AND = {
				naval_base > 0
				NOT = { is_in_home_area = yes }
			}
			AND = {
				any_state_division = { OWNER = { tag = event_target:cannibalism_scored_host } }
				NOT = { supply_node > 0 }
				infrastructure < constant:cannibalism_host_threshold.critical_infrastructure
			}
		}
	}
}

cannibalism_country_has_remote_army = {
	any_controlled_state = {
		any_state_division = { OWNER = { tag = event_target:cannibalism_scored_host } }
		OR = {
			is_island_state = yes
			NOT = { is_in_home_area = yes }
		}
	}
}

	any_controlled_state = {

		cannibalism_state_is_valid_origin = yes
	}
}

	OR = {
		check_variable = {
			var = convoy_threat
			value = constant:cannibalism_host_threshold.convoy_threat_medium
			compare = greater_than_or_equals
		}
		has_convoys_war_support < constant:cannibalism_host_threshold.convoy_war_support_loss
	}
}

cannibalism_state_has_prison_or_camp = {
	OR = {
		has_state_flag = cannibalism_scenario_prison_compound
		extermination_camp > 0
		gulag_labor_camp_network > 0
		has_state_flag = genocide_active_atrocity_site
		has_state_flag = camp_rework_active_site
		has_state_flag = camp_rework_expanded_labor_site
		has_state_flag = genocide_auschwitz_experiment_site
		has_state_flag = genocide_japanese_biowarfare_atrocity_site
	}
}

cannibalism_state_has_army_presence = {
	OR = {
		AND = {
			has_event_target = cannibalism_state_host
			any_state_division = { OWNER = { tag = event_target:cannibalism_state_host } }
		}
		AND = {
			NOT = { has_event_target = cannibalism_state_host }
			has_event_target = cannibalism_scored_host
			any_state_division = { OWNER = { tag = event_target:cannibalism_scored_host } }
		}
		AND = {
			NOT = { has_event_target = cannibalism_state_host }
			NOT = { has_event_target = cannibalism_scored_host }
			any_state_division = { always = yes }
		}
	}
}

cannibalism_state_is_isolated = {
	OR = {
		is_island_state = yes
		NOT = { is_in_home_area = yes }
	}
}

cannibalism_state_has_logistics_pressure = {
	OR = {
		infrastructure < constant:cannibalism_state_threshold.low_infrastructure
		AND = {
			cannibalism_state_has_army_presence = yes
			NOT = { supply_node > 0 }
		}
		AND = {
			cannibalism_state_is_isolated = yes
			naval_base > 0
		}
	}
}
