cannibalism_has_valid_origin_available = {
	any_country = { cannibalism_can_be_origin = yes }
}

cannibalism_can_be_origin = {
	exists = yes
	uses_normal_civilian_systems = yes
	has_war = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { has_country_flag = cannibalism_country_defeated_outbreak }
	NOT = { has_country_flag = cannibalism_outbreak_active }
	capital_scope = { is_controlled_by = ROOT }
}

cannibalism_can_receive_spread = {
	exists = yes
	uses_normal_civilian_systems = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_capitulated = yes }
	NOT = { has_global_flag = world_end }
	NOT = { has_country_flag = cannibalism_country_defeated_outbreak }
	NOT = { has_country_flag = cannibalism_outbreak_active }
	OR = {
		has_war = yes
		has_stability < 0.45
		any_controlled_state = { is_coastal = yes }
	}
}

is_cannibal_commune_country = {
	OR = {
		tag = CBL
		original_tag = CBL
		has_country_flag = cannibalism_commune_country
	}
}

cannibalism_cbl_is_island_origin = {
	is_cannibal_commune_country = yes
	has_country_flag = cannibal_origin_island
}

cannibalism_cbl_is_coastal_origin = {
	is_cannibal_commune_country = yes
	OR = {
		has_country_flag = cannibal_origin_island
		has_country_flag = cannibal_origin_coastal
		any_controlled_state = {
			is_controlled_by = ROOT
			is_coastal = yes
		}
	}
}

cannibalism_cbl_is_inland_origin = {
	is_cannibal_commune_country = yes
	OR = {
		has_country_flag = cannibal_origin_inland
		NOT = {
			any_controlled_state = {
				is_controlled_by = ROOT
				is_coastal = yes
			}
		}
	}
}

cannibalism_cbl_has_hunting_ground_project = {
	is_cannibal_commune_country = yes
	check_variable = {
		var = cannibalism_cbl_hunting_ground_projects_completed
		value = constant:cannibalism_last_table_requirement.hunting_ground_projects
		compare = greater_than_or_equals
	}
}

cannibalism_cbl_has_available_hunting_ground_state = {
	is_cannibal_commune_country = yes
	any_controlled_state = {
		is_controlled_by = ROOT
		NOT = { has_state_flag = cannibalism_commune }
		NOT = { has_dynamic_modifier = { modifier = cannibalism_hunting_ground_state } }
	}
}

cannibalism_cbl_has_map_projection = {
	is_cannibal_commune_country = yes
	OR = {
		any_controlled_state = {
			is_controlled_by = ROOT
			is_coastal = yes
		}
		any_controlled_state = {
			is_controlled_by = ROOT
			rail_way > 0
		}
	}
}

cannibalism_cbl_has_neighbor_target = {
	is_cannibal_commune_country = yes
	any_controlled_state = {
		is_controlled_by = ROOT
		any_neighbor_state = {
			NOT = { is_claimed_by = ROOT }
			owner = {
				exists = yes
				NOT = { tag = ROOT }
				NOT = { is_cannibal_commune_country = yes }
				NOT = { has_capitulated = yes }
				NOT = { has_war_with = ROOT }
				can_ROOT_get_wargoal_on_THIS = yes
			}
		}
	}
}

cannibalism_hannibal_or_unifier_exists = {
	OR = {
		has_global_flag = hannibal_exists
		has_global_flag = cannibalism_later_unifier_accepted
		has_global_flag = cannibalism_hannibal_proxy_unifier
		AND = {
			has_global_flag = cannibalism_triggerable_scenario_launch_active
			has_global_flag = cannibalism_triggerable_hannibal_network_test_bypass
			check_variable = {
				var = global.cannibalism_triggerable_scenario_type
				value = constant:triggerable_scenario_cannibalism_type.hannibal_network
				compare = equals
			}
		}
	}
}
