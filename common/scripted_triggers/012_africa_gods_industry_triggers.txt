# ACTOR private arithmetic/preflight. Caller preinitializes amount and forwarded
# entitlement/increment temps. Uses the legacy generation + CPP reserve floor.
gods_of_africa_industry_amount_can_execute = {
	gods_of_africa_industry_host_can_receive_payment = yes
	gods_of_africa_demand_payment_is_open = yes
	gods_of_africa_participant_generation_is_current = yes
	check_variable = { gods_of_africa_industry_amount > constant:gods_of_africa_value.zero }
	check_variable = { var = gods_of_africa_industry_amount value = gods_of_africa_demand_remaining compare = less_than_or_equals }
	set_temp_variable = { gods_of_africa_industry_capacity = gods_of_africa_industry_amount }
	add_to_temp_variable = { gods_of_africa_industry_capacity = constant:gods_of_africa_value.capacity_floor_industry }
	NOT = { has_political_power < gods_of_africa_industry_capacity }
	set_temp_variable = { gods_of_africa_industry_funding_next = var:gods_of_africa_industry_host.gods_of_africa_industry_program_funded }
	add_to_temp_variable = { gods_of_africa_industry_funding_next = gods_of_africa_industry_amount }
	set_temp_variable = { gods_of_africa_industry_entitlement = gods_of_africa_industry_service_days }
	multiply_temp_variable = { gods_of_africa_industry_entitlement = gods_of_africa_industry_funding_next }
	divide_temp_variable = { gods_of_africa_industry_entitlement = gods_of_africa_industry_quote }
	set_temp_variable = { gods_of_africa_industry_remainder = gods_of_africa_industry_entitlement }
	modulo_temp_variable = { gods_of_africa_industry_remainder = constant:gods_of_africa_value.one }
	subtract_from_temp_variable = { gods_of_africa_industry_entitlement = gods_of_africa_industry_remainder }
	# Truncated fixed-point division cannot award the final day on short funding.
	set_temp_variable = { gods_of_africa_industry_ceiling = gods_of_africa_industry_service_days }
	subtract_from_temp_variable = { gods_of_africa_industry_ceiling = constant:gods_of_africa_value.one }
	if = {
		limit = { check_variable = { var = gods_of_africa_industry_funding_next value = gods_of_africa_industry_quote compare = less_than } }
		clamp_temp_variable = { var = gods_of_africa_industry_entitlement min = constant:gods_of_africa_value.zero max = gods_of_africa_industry_ceiling }
	}
	else = { set_temp_variable = { gods_of_africa_industry_entitlement = gods_of_africa_industry_service_days } }
	set_temp_variable = { gods_of_africa_industry_increment = gods_of_africa_industry_entitlement }
	subtract_from_temp_variable = { gods_of_africa_industry_increment = var:gods_of_africa_industry_host.gods_of_africa_industry_program_granted_days }
	OR = {
		gods_of_africa_industry_completion_is_proven = yes
		check_variable = { gods_of_africa_industry_increment > constant:gods_of_africa_value.zero }
	}
}
