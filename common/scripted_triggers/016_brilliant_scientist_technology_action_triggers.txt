brilliant_scientist_can_pay_light_technology_action = {
	NOT = { command_power < constant:brilliant_scientist_technology_action.light_command_power }
	NOT = { num_of_civilian_factories_available_for_projects < constant:brilliant_scientist_technology_action.light_civilian_factories }
	has_equipment = { support_equipment > constant:brilliant_scientist_technology_action.light_support_exclusive_gate }
	NOT = { has_fuel < constant:brilliant_scientist_technology_action.light_fuel_refund }
}

brilliant_scientist_has_strategic_resource_shortage = {
	OR = {
		has_resources_in_country = { resource = aluminium amount < constant:brilliant_scientist_technology_action.resource_shortage_exclusive_threshold }
		has_resources_in_country = { resource = tungsten amount < constant:brilliant_scientist_technology_action.resource_shortage_exclusive_threshold }
		has_resources_in_country = { resource = chromium amount < constant:brilliant_scientist_technology_action.resource_shortage_exclusive_threshold }
		has_resources_in_country = { resource = rubber amount < constant:brilliant_scientist_technology_action.resource_shortage_exclusive_threshold }
	}
}

# STATE scope. Synthesis works may be constructed once in an owned industrial
# core with real building capacity. The state becomes the permanent receipt.
brilliant_scientist_materials_synthesis_target_is_valid = {
	is_owned_and_controlled_by = ROOT
	is_core_of = ROOT
	impassable = no
	NOT = { has_state_flag = brilliant_scientist_materials_synthesis_works }
	OR = {
		industrial_complex > 0
		arms_factory > 0
		synthetic_refinery > 0
	}
}

# STATE scope. Sensor saturation is anchored on a controlled core containing
# radar, air, or anti-air infrastructure so the map choice is meaningful.
brilliant_scientist_sensor_saturation_target_is_valid = {
	is_owned_and_controlled_by = ROOT
	is_core_of = ROOT
	impassable = no
	OR = {
		radar_station > 0
		air_base > 0
		anti_air_building > 0
	}
}

# STATE scope. The delivery network attacks a state controlled by a current war
# opponent and cannot repeatedly strike the same industrial target.
brilliant_scientist_long_range_strike_target_is_valid = {
	impassable = no
	CONTROLLER = {
		exists = yes
		has_war_with = ROOT
	}
	NOT = { has_state_flag = brilliant_scientist_long_range_strike_target_cooldown }
	OR = {
		industrial_complex > 0
		arms_factory > 0
		infrastructure > 0
		air_base > 0
		radar_station > 0
		anti_air_building > 0
	}
}
