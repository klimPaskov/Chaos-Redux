# ============================================================================
# CHAOS REDUX - FALLOUT ASHLINE FIREBREAK TRIGGERS
# ============================================================================
# A state-targeted chain may enter only through an open, owner-controlled
# natural-disaster aftermath card. Wildfire and ash-heavy volcanic families
# are admitted through their native history triggers, so the chain never
# invents a province or duplicates the capped Air Contamination source writer.

fallout_event_554_state_is_current = {
	owner = { fallout_event_timeline_phase_receipt_is_current = yes }
	is_controlled_by = ROOT
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_state_flag = natural_disaster_aftermath_active
	natural_disaster_state_has_open_card = yes
	OR = {
		natural_disaster_state_has_dry_hazard_history = yes
		natural_disaster_state_has_volcanic_hazard_history = yes
	}
	has_variable = natural_disaster_family
	has_variable = natural_disaster_severity
	has_variable = fallout_pretransition_air_winter_adaptation
	has_variable = fallout_pretransition_air_winter_exposure
	has_variable = fallout_pretransition_air_winter_reclamation
	has_variable = fallout_state_supply_access_current
	check_variable = { state_population_k value = constant:fallout_event_554_threshold.minimum_state_population_k compare = greater_than_or_equals }
	check_variable = { fallout_pretransition_air_winter_exposure value = constant:fallout_event_554_threshold.minimum_exposure compare = greater_than_or_equals }
	check_variable = { fallout_pretransition_air_winter_exposure value = constant:fallout_event_554_threshold.maximum_exposure compare = less_than }
	check_variable = { fallout_pretransition_air_winter_reclamation value = constant:fallout_event_554_threshold.minimum_reclamation compare = less_than }
	check_variable = { fallout_state_supply_access_current value = constant:fallout_event_554_threshold.maximum_supply compare = less_than }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	NOT = { has_state_flag = fallout_event_554_state_chain_seen }
}

fallout_event_554_country_is_current = {
	fallout_event_registry_country_row_is_current = yes
	fallout_survival_country_resource_row_is_durable = yes
	has_variable = fallout_event_554_candidate_state_id
	has_variable = fallout_event_554_ash_pressure
	check_variable = { fallout_event_554_candidate_state_id > constant:fallout_event_runtime.zero }
	check_variable = { fallout_event_554_ash_pressure value = constant:fallout_event_runtime.one compare = greater_than_or_equals }
	check_variable = { var = global.num_days value = constant:fallout_event_554_threshold.minimum_campaign_day compare = greater_than_or_equals }
	check_variable = { var = global.num_days value = constant:fallout_event_554_threshold.maximum_campaign_day compare = less_than }
	check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_554_cost.cut_scrap compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^5 value = constant:fallout_event_554_cost.cut_fuel compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^8 value = constant:fallout_event_554_cost.seal_shelter compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_554_cost.cordon_power compare = greater_than_or_equals }
	NOT = { has_country_flag = fallout_event_554_chain_committed }
	NOT = { has_country_flag = fallout_event_554_memory_closed }
	any_owned_state = {
		check_variable = { var = THIS.id value = ROOT.fallout_event_554_candidate_state_id compare = equals }
		fallout_event_554_state_is_current = yes
	}
}

	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.state }
	check_variable = { fallout_event_dispatch_issued_target > constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_554_target_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_dispatch_issued_target = {
		set_temp_variable = { fallout_event_554_target_context_loaded = constant:fallout_event_runtime.one }

	}

}
