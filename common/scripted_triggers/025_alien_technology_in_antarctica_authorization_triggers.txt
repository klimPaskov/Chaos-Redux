# Event 025 entry authorization costs. Inclusive resource checks are shared
# by decision affordability and each resource's cost-display state.
chaosx_nr25_authorize_convoys_available = {
	check_variable = { var = num_equipment@convoy value = constant:chaosx_nr25_cost.convoy_authorize_required compare = greater_than_or_equals }
}

chaosx_nr25_authorize_fuel_available = {
	NOT = { has_fuel < constant:chaosx_nr25_cost.fuel_authorize_required }
}

chaosx_nr25_authorize_support_available = {
	check_variable = { var = num_equipment@support_equipment value = constant:chaosx_nr25_cost.support_authorize_required compare = greater_than_or_equals }
}
