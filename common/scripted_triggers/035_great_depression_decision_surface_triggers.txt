# CHAOS REDUX - EVENT 035 DECISION SURFACE TRIGGERS
# ============================================================================
# These predicates bind the ordinary decision category to the parent-owned
# Severity lifecycle. State and partner targets are validated in their own
# scopes, while the country surface remains bounded and fail-closed.
# ============================================================================


great_depression_decision_surface_is_ready = {
	great_depression_active_crisis_is_coherent = yes
	NOT = { has_country_flag = great_depression_transition_pending }
}

great_depression_action_slot_is_open = {
	great_depression_decision_surface_is_ready = yes
	NOT = { has_country_flag = great_depression_action_lock }
}

# Countries carrying only an exposure or worldwide-pressure row receive a
# compact response surface without being given a national Depression Severity.
great_depression_exposure_response_surface_is_ready = {
	great_depression_has_viable_economy = yes
	has_country_flag = great_depression_contagion_exposure_open
	NOT = { has_country_flag = great_depression_active }
	NOT = { has_country_flag = great_depression_contagion_converted }
	has_variable = great_depression_contagion_source_countries^num
	check_variable = { var = great_depression_contagion_source_countries^num value = constant:great_depression_api.zero compare = greater_than }
}

great_depression_global_supplier_surface_is_ready = {
	great_depression_has_viable_economy = yes
	has_global_flag = great_depression_worldwide_episode_active
	has_country_flag = great_depression_worldwide_pressure_active
	check_variable = { var = great_depression_worldwide_pressure_episode_id value = global.great_depression_global_episode_id compare = equals }
	NOT = { check_variable = { var = global.great_depression_global_stage value = constant:great_depression_global_stage.recovery compare = greater_than_or_equals } }
	NOT = { has_country_flag = great_depression_active }
	NOT = { has_country_flag = great_depression_transition_pending }
	NOT = { has_country_flag = great_depression_incompatible_economic_state }
	NOT = { has_country_flag = great_depression_contagion_converted }
}

great_depression_global_action_surface_is_ready = {
	OR = {
		great_depression_second_great_depression_surface_is_active = yes
		great_depression_global_supplier_surface_is_ready = yes
	}
}

great_depression_external_action_slot_is_open = {
	OR = {
		great_depression_action_slot_is_open = yes
		AND = {
			OR = {
				great_depression_exposure_response_surface_is_ready = yes
				great_depression_global_supplier_surface_is_ready = yes
			}
			NOT = { has_country_flag = great_depression_action_lock }
		}
	}
}

# Candidate partner scope with ROOT as the decision owner. A target must have
# an exact diplomatic, geographic, subject, or registered contagion link and a
# viable land/faction connection or convoy route. This keeps the public target
# list sparse and prevents an unrelated major from becoming a universal bank.
great_depression_has_proved_partner_relationship = {
	exists = yes
	NOT = { tag = ROOT }
	NOT = { has_war_with = ROOT }
	OR = {
		is_in_faction_with = ROOT
		is_ally_with = ROOT
		is_neighbor_of = ROOT
		is_subject_of = ROOT
		ROOT = { is_subject_of = PREV }
		is_in_array = { array = great_depression_contagion_source_countries value = ROOT }
		ROOT = { is_in_array = { array = great_depression_contagion_exposure_targets value = PREV } }
	}
	OR = {
		is_in_faction_with = ROOT
		is_ally_with = ROOT
		is_neighbor_of = ROOT
		is_subject_of = ROOT
		ROOT = { is_subject_of = PREV }
		AND = {
			has_equipment = { convoy_1 > constant:great_depression_api.zero }
			ROOT = { has_equipment = { convoy_1 > constant:great_depression_api.zero } }
		}
	}
}

# STATE scope. Incident state pointers are valid only while the state remains
# usable, owned, controlled, and economically meaningful for the active
# country. This prevents a stale center or transferred state from receiving a
# later incident receipt.
great_depression_incident_state_target_is_valid = {
	exists = yes
	impassable = no
	is_owned_by = ROOT
	is_controlled_by = ROOT
	state_population_k > constant:great_depression_api.minimum_viable_state_population_k
	OR = {
		industrial_complex > constant:great_depression_api.zero
		arms_factory > constant:great_depression_api.zero
		dockyard > constant:great_depression_api.zero
	}
}
