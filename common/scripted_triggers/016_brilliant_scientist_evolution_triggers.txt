# COUNTRY scope. Evolution III's disabled-state safety always permits this
# settlement. With the full chain enabled, the host must earn it by making
# Kruger's work independently reproducible before forbidden weaponization.
brilliant_scientist_evolution_iv_supervised_compact_is_available = {
	OR = {
		brilliant_scientist_evolution_iii_safe_resolution_is_required = yes
		AND = {
			brilliant_scientist_control_is_stable = yes
			brilliant_scientist_has_high_mandate = yes
			brilliant_scientist_has_low_dependence = yes
			OR = {
				has_country_flag = brilliant_scientist_evolution_iii_policy_safe_public_science
				has_country_flag = brilliant_scientist_evolution_iii_policy_negotiated_limits
			}
			check_variable = {
				var = brilliant_scientist_independently_replicated_family_count
				value = constant:brilliant_scientist_evolution_runtime.supervised_compact_replication_families
				compare = greater_than_or_equals
			}
			NOT = { brilliant_scientist_has_forbidden_weaponization = yes }
			NOT = { brilliant_scientist_institutional_capture_is_proven = yes }
		}
	}
}
