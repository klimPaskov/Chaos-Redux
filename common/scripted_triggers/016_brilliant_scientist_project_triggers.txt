brilliant_scientist_project_board_is_ready = {
	brilliant_scientist_is_current_host = yes
	brilliant_scientist_primary_facility_is_valid = yes
	NOT = { has_country_flag = brilliant_scientist_project_stage_in_progress }
	NOT = { has_country_flag = brilliant_scientist_project_incident_active }
}

brilliant_scientist_has_capacity_to_resume_selected_project = {
	OR = {
		AND = {
			check_variable = { brilliant_scientist_directorate_selected_project_stage = constant:brilliant_scientist_project_stage.theory }
			check_variable = { var = brilliant_scientist_project_capacity value = constant:brilliant_scientist_project_resume_capacity.theory compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_directorate_selected_project_stage = constant:brilliant_scientist_project_stage.prototype }
			check_variable = { var = brilliant_scientist_project_capacity value = constant:brilliant_scientist_project_resume_capacity.prototype compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_directorate_selected_project_stage = constant:brilliant_scientist_project_stage.deployment }
			check_variable = { var = brilliant_scientist_project_capacity value = constant:brilliant_scientist_project_resume_capacity.deployment compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_directorate_selected_project_stage = constant:brilliant_scientist_project_stage.weaponization }
			check_variable = { var = brilliant_scientist_project_capacity value = constant:brilliant_scientist_project_resume_capacity.weaponization compare = greater_than_or_equals }
		}
	}
}

brilliant_scientist_selected_project_can_be_dismantled_safely = {
	custom_trigger_tooltip = {
		tooltip = brilliant_scientist_singularity_dismantle_guard_tt
		OR = {
			NOT = { check_variable = { brilliant_scientist_directorate_selected_project_family = constant:brilliant_scientist_project_family.singularity } }
			check_variable = {
				var = brilliant_scientist_directorate_selected_project_stage
				value = constant:brilliant_scientist_project_stage.deployment
				compare = less_than
			}
		}
	}
}

brilliant_scientist_has_required_singularity_facilities = {
	custom_trigger_tooltip = {
		tooltip = brilliant_scientist_singularity_facility_requirement_tt
		any_owned_state = {
			count = constant:brilliant_scientist_singularity.minimum_facilities
			is_controlled_by = ROOT
			OR = {
				land_facility > 0
				air_facility > 0
				nuclear_facility > 0
				biowarfare_facility > 0
			}
		}
	}
}
