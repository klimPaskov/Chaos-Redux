# CHAOS REDUX - EVENT 012 AFRICA AI PROFILE TRIGGERS
# ============================================================================
# Country-scope activation predicates for the exact sixty-four-row Event 12 AI
# registry. Every predicate reads implemented state: host overlay, constitution,
# relationship graph, action evidence, foreign posture, capital continent, or
# full host playbook. No opinion-only accession test and no periodic scan lives
# here. Callers refresh profiles only from an action, focus, event, or bounded
# AI decision evaluation.

# Uses the prepared temporary action profile without writing a player quote.
africa_ai_prepared_action_phase_is_available = {
	OR = {
		check_variable = { africa_profile_phase = constant:africa_action_phase.any }
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.opening }
			africa_action_phase_has_reached_opening = yes
		}
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.regional }
			africa_action_phase_has_reached_regional = yes
		}
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.late }
			africa_action_phase_has_reached_late = yes
		}
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.high_chaos }
			has_global_flag = africa_evolution_iii_logged
			check_variable = { var = africa_current_phase value = constant:africa_action_phase.high_chaos compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.post_unification }
			has_global_flag = africa_is_one
			check_variable = { var = africa_current_phase value = constant:africa_action_phase.post_unification compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.terminal_approach }
			has_global_flag = africa_world_order_open
			check_variable = { var = africa_current_phase value = constant:africa_action_phase.terminal_approach compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { africa_profile_phase = constant:africa_action_phase.terminal }
			OR = {
				AND = {
					has_global_flag = africa_continental_wars_active
					check_variable = { var = africa_current_phase value = constant:africa_action_phase.terminal compare = greater_than_or_equals }
				}
				AND = {
					has_global_flag = world_end_africa_the_world
					OR = {
						check_variable = { africa_requested_action_id = constant:africa_action.administer_world_regions }
						check_variable = { africa_requested_action_id = constant:africa_action.contain_terminal_high_chaos }
					}
				}
			}
		}
	}
	OR = {
		check_variable = { africa_profile_required_constitution = constant:africa_constitution.uncommitted }
		check_variable = {
			var = africa_constitution
			value = africa_profile_required_constitution
			compare = equals
		}
	}
}

# The target's immutable last-action ledger makes retry policy observable.
africa_ai_selected_target_retry_policy_allows_action = {
	has_event_target = africa_action_target
	OR = {
		event_target:africa_action_target = { NOT = { has_country_flag = africa_has_action_history } }
		event_target:africa_action_target = {
			NOT = {
				check_variable = {
					var = africa_last_action_id
					value = africa_requested_action_id
					compare = equals
				}
			}
		}
		AND = {
			event_target:africa_action_target = { check_variable = { africa_last_action_outcome = constant:africa_action_outcome.partial } }
			NOT = { africa_ai_accepts_partial_outcome = yes }
			africa_ai_retry_stance_allows_current_target = yes
		}
		AND = {
			event_target:africa_action_target = {
				OR = {
					check_variable = { africa_last_action_outcome = constant:africa_action_outcome.failure }
					check_variable = { africa_last_action_outcome = constant:africa_action_outcome.cancelled }
				}
			}
			africa_ai_retry_stance_allows_current_target = yes
		}
	}
}
