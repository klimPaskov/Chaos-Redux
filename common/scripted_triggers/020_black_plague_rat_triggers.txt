black_plague_rat_country_is_active = {
	exists = yes
	has_country_flag = black_plague_rat_country
	NOT = { has_country_flag = black_plague_rat_country_retired }
}

# Rat immunity is a country-level army contract. It does not suppress the
# civilian population loss applied to plague states, including rat-controlled
# occupied states; it gates the non-human force surface instead.
black_plague_rat_country_is_plague_immune = {
	exists = yes
	OR = {
		has_country_flag = black_plague_rat_disease_immune
		has_country_flag = black_plague_rat_plague_immune
	}
}

}

# STATE scope. The Rat King can only project a royal strike into a real enemy
# human state. The state remains eligible while clear or cured so a successful
# operation can reopen the shared disease lifecycle, but rat-controlled land
# and a recent target cooldown are excluded from the selector.
black_plague_rat_king_royal_strike_target_is_valid = {
	exists = yes
	CONTROLLER = {
		exists = yes
		black_plague_country_is_human_host = yes
		has_war_with = ROOT
	}
	black_plague_state_can_receive_exposure = yes
	NOT = { black_plague_state_is_rat_controlled = yes }
	NOT = { has_state_flag = black_plague_rat_king_royal_strike_cooldown }
}

# A surviving RTA carrier can attempt one constrained restoration after the
# RTX crown falls. The route needs a living human basin for the new Royal
# Basin, enough sentience to remember the old court, and enough coherence to
# keep the carrier from splitting while the crown is rebuilt.
black_plague_rat_king_restoration_can_start = {
	black_plague_rat_country_is_active = yes
	has_global_flag = black_plague_rat_king_defeated
	has_global_flag = black_plague_evolution_iv_active
	NOT = { has_global_flag = black_plague_rat_king_restoration_used }
	NOT = { has_global_flag = black_plague_rat_king_restored }
	NOT = { has_global_flag = black_plague_evolution_v_active }
	NOT = { has_global_flag = black_plague_evolution_v_recorded }
	NOT = { has_global_flag = black_plague_evolution_v_route_open }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = black_plague_terminal_takeover_active }
	NOT = { has_country_flag = black_plague_rat_country_retired }
	NOT = { has_country_flag = black_plague_rat_king_restoration_cooldown }
	check_variable = {
		var = black_plague_rat_brood_mass
		value = constant:black_plague_rat_decision.king_restoration_mass_cost
		compare = greater_than_or_equals
	}
	check_variable = {
		var = black_plague_rat_sentience
		value = constant:black_plague_rat_decision.king_restoration_sentience_gate
		compare = greater_than_or_equals
	}
	check_variable = {
		var = black_plague_rat_coherence
		value = constant:black_plague_rat_decision.king_restoration_cohesion_gate
		compare = greater_than_or_equals
	}
	num_of_controlled_states > 1
	any_state = {
		black_plague_state_is_rat_controlled = yes
		CONTROLLER = { black_plague_rat_country_is_active = yes }
	}
	any_state = {
		black_plague_state_is_established = yes
		NOT = { black_plague_state_is_rat_controlled = yes }
		NOT = { has_state_flag = black_plague_rat_royal_basin_reserved }
		CONTROLLER = { black_plague_country_is_human_host = yes }
	}
}

# STATE scope. Evolution III needs an uncontrolled connected basin rather than
# one isolated collapse. Infected, Severe Crisis, and Collapsed states qualify;
# containment and Rat-Controlled ground do not. The nested neighbour walk proves
# a three-state path, and the final comparison prevents the walk from counting
# the origin state again when a two-state border is traversed in both directions.
black_plague_rat_state_is_connected_basin_member = {
	black_plague_state_is_established = yes
	OR = {
		black_plague_state_is_infected = yes
		black_plague_state_is_severe = yes
		black_plague_state_is_collapsed = yes
	}
	check_variable = {
		var = black_plague_rat_infestation
		value = constant:black_plague_threshold.relapse_exposure
		compare = greater_than_or_equals
	}
	OWNER = { black_plague_country_is_human_host = yes }
	CONTROLLER = { black_plague_country_is_human_host = yes }
	NOT = { black_plague_state_is_rat_controlled = yes }
}

black_plague_rat_state_has_connected_basin_signal = {
	black_plague_rat_state_is_connected_basin_member = yes
	any_neighbor_state = {
		black_plague_rat_state_is_connected_basin_member = yes
		any_neighbor_state = {
			black_plague_rat_state_is_connected_basin_member = yes
			NOT = { state = PREV.PREV }
		}
	}
}
