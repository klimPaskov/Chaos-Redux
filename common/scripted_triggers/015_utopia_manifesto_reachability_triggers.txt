# Country scope, with ROOT as the Event 15 actor selecting a real recipient.
utopia_manifesto_foreign_reaction_candidate_is_safe = {
	exists = yes
	uses_normal_civilian_systems = yes
	is_subject = no
	has_capitulated = no
	has_civil_war = no
	NOT = { tag = ROOT }
	NOT = { has_country_flag = utopia_manifesto_accepted }
	NOT = { utopia_manifesto_candidate_has_terminal_identity = yes }
	NOT = { utopia_manifesto_candidate_has_active_event_package = yes }
	NOT = { has_war_with = ROOT }
	surrender_progress < constant:utopia_manifesto_candidate_thresholds.maximum_surrender_progress
	OR = {
		is_neighbor_of = ROOT
		is_on_same_continent_as = ROOT
		has_opinion = { target = ROOT value > constant:utopia_manifesto_decision_threshold.positive_opinion }
	}
}

	NOT = { tag = FROM }
