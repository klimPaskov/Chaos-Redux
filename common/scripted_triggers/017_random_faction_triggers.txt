# has_opinion value does not accept script constants, so this parser-static threshold is file scoped.
@random_faction_positive_opinion = 20

random_faction_country_base_valid = {
	exists = yes
	uses_normal_civilian_systems = yes
	num_of_controlled_states > constant:random_faction_requirement.min_controlled_states
	capital_scope = { is_controlled_by = PREV }
	NOT = { has_capitulated = yes }
	NOT = { is_government_in_exile = yes }
	NOT = { has_global_flag = world_end }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
}

is_random_faction_eligible_country = {
	random_faction_country_base_valid = yes
	NOT = { is_major = yes }
	NOT = { is_subject = yes }
	NOT = { is_in_faction = yes }
	NOT = { has_war = yes }
	NOT = { has_country_flag = random_faction_recent_alignment }
	NOT = { has_country_flag = random_faction_selection_cooldown }
	NOT = { has_country_flag = random_faction_selection_pending }
	any_country = {
		random_faction_is_valid_faction_leader_for_prev_target = yes
	}
}

is_random_faction_wartime_candidate = {
	random_faction_country_base_valid = yes
	NOT = { is_major = yes }
	NOT = { is_subject = yes }
	NOT = { is_in_faction = yes }
	NOT = { has_country_flag = random_faction_recent_alignment }
	NOT = { has_country_flag = random_faction_selection_cooldown }
	NOT = { has_country_flag = random_faction_selection_pending }
	OR = {
		has_war = yes
		any_neighbor_country = {
			is_in_faction = yes
			has_war = yes
		}
		random_faction_has_active_pressure = yes
	}
	any_country = {
		random_faction_is_valid_faction_leader_for_prev_target = yes
	}
}

random_faction_has_valid_target_available = {
	any_country = {
		OR = {
			is_random_faction_eligible_country = yes
			AND = {
				has_global_flag = random_faction_evolution_ii_unlocked
				is_random_faction_wartime_candidate = yes
			}
		}
	}
}

random_faction_has_dispatchable_context = {
	random_faction_has_valid_target_available = yes
	any_country = { is_random_faction_allowed_faction_leader = yes }
}

is_random_faction_allowed_faction_leader = {
	exists = yes
	uses_normal_civilian_systems = yes
	num_of_controlled_states > constant:random_faction_requirement.min_controlled_states
	is_faction_leader = yes
	is_in_faction = yes
	NOT = { has_capitulated = yes }
	NOT = { is_government_in_exile = yes }
	NOT = { has_global_flag = world_end }
	NOT = { is_subject = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
}

random_faction_target_requires_wartime_reach = {
	has_global_flag = random_faction_evolution_ii_unlocked
	OR = {
		has_war = yes
		any_neighbor_country = {
			is_in_faction = yes
			has_war = yes
		}
		random_faction_has_active_pressure = yes
	}
}

random_faction_current_leader_has_wartime_reach_to_root = {
	OR = {
		is_neighbor_of = ROOT
		is_on_same_continent_as = ROOT
		any_country = {
			is_in_faction_with = PREV
			OR = {
				is_neighbor_of = ROOT
				is_on_same_continent_as = ROOT
			}
		}
		AND = {
			ROOT = {
				any_controlled_state = { is_coastal = yes }
				NOT = { has_equipment = { convoy < constant:random_faction_requirement.wartime_overseas_convoys } }
			}
			OR = {
				any_controlled_state = { is_coastal = yes }
				any_country = {
					is_in_faction_with = PREV
					any_controlled_state = { is_coastal = yes }
				}
			}
		}
		any_country = {
			has_war_with = ROOT
			PREV = { has_war_with = THIS }
		}
	}
}

random_faction_current_leader_has_wartime_reach_to_prev = {
	OR = {
		is_neighbor_of = PREV
		is_on_same_continent_as = PREV
		any_country = {
			is_in_faction_with = PREV
			OR = {
				is_neighbor_of = PREV.PREV
				is_on_same_continent_as = PREV.PREV
			}
		}
		AND = {
			PREV = {
				any_controlled_state = { is_coastal = yes }
				NOT = { has_equipment = { convoy < constant:random_faction_requirement.wartime_overseas_convoys } }
			}
			OR = {
				any_controlled_state = { is_coastal = yes }
				any_country = {
					is_in_faction_with = PREV
					any_controlled_state = { is_coastal = yes }
				}
			}
		}
		any_country = {
			has_war_with = PREV.PREV
			PREV = { has_war_with = THIS }
		}
	}
}

random_faction_is_valid_faction_leader_for_root_target = {
	is_random_faction_allowed_faction_leader = yes
	NOT = { tag = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { is_in_faction_with = ROOT }
	NOT = {
		any_country = {
			is_in_faction_with = PREV
			has_war_with = ROOT
		}
	}
	ROOT = {
		random_faction_country_base_valid = yes
		NOT = { is_subject = yes }
		NOT = { is_in_faction = yes }
	}
	OR = {
		NOT = { has_global_flag = random_faction_evolution_ii_unlocked }
		AND = {
			NOT = { has_war = yes }
			ROOT = { NOT = { random_faction_target_requires_wartime_reach = yes } }
		}
		random_faction_current_leader_has_wartime_reach_to_root = yes
	}
}

random_faction_is_valid_faction_leader_for_prev_target = {
	is_random_faction_allowed_faction_leader = yes
	NOT = { tag = PREV }
	NOT = { has_war_with = PREV }
	NOT = { is_in_faction_with = PREV }
	NOT = {
		any_country = {
			is_in_faction_with = PREV
			has_war_with = PREV.PREV
		}
	}
	PREV = {
		random_faction_country_base_valid = yes
		NOT = { is_subject = yes }
		NOT = { is_in_faction = yes }
	}
	OR = {
		NOT = { has_global_flag = random_faction_evolution_ii_unlocked }
		AND = {
			NOT = { has_war = yes }
			PREV = { NOT = { random_faction_target_requires_wartime_reach = yes } }
		}
		random_faction_current_leader_has_wartime_reach_to_prev = yes
	}
}
