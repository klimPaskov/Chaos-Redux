@random_faction_border_post_divisions = 1

random_faction_country_base_valid = {
	exists = yes
	uses_normal_civilian_systems = yes
	num_of_controlled_states > constant:random_faction_requirement.min_controlled_states
	capital_scope = { is_controlled_by = PREV }
	NOT = { has_capitulated = yes }
	NOT = { is_government_in_exile = yes }
	NOT = { has_global_flag = world_end }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
}

is_random_faction_eligible_country = {
	random_faction_country_base_valid = yes
	NOT = { is_major = yes }
	NOT = { is_subject = yes }
	NOT = { is_in_faction = yes }
	NOT = { has_country_flag = random_faction_recent_alignment }
	NOT = { has_country_flag = random_faction_selection_cooldown }
	any_country = {
		random_faction_is_valid_faction_leader_for_root_target = yes
	}
}

is_random_faction_wartime_candidate = {
	random_faction_country_base_valid = yes
	NOT = { is_major = yes }
	NOT = { is_subject = yes }
	NOT = { is_in_faction = yes }
	NOT = { has_country_flag = random_faction_recent_alignment }
	NOT = { has_country_flag = random_faction_selection_cooldown }
	OR = {
		has_war = yes
		any_neighbor_country = { is_in_faction = yes }
	}
	any_country = {
		random_faction_is_valid_faction_leader_for_root_target = yes
	}
}

random_faction_has_valid_target_available = {
	any_country = {
		OR = {
			is_random_faction_eligible_country = yes
			AND = {
				has_global_flag = random_faction_evolution_ii_unlocked
				is_random_faction_wartime_candidate = yes
			}
		}
	}
}

random_faction_has_dispatchable_context = {
	random_faction_has_valid_target_available = yes
	any_country = { is_random_faction_allowed_faction_leader = yes }
}

is_random_faction_allowed_faction_leader = {
	exists = yes
	uses_normal_civilian_systems = yes
	num_of_controlled_states > constant:random_faction_requirement.min_controlled_states
	is_faction_leader = yes
	is_in_faction = yes
	NOT = { has_capitulated = yes }
	NOT = { is_government_in_exile = yes }
	NOT = { has_global_flag = world_end }
	NOT = { is_subject = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
}

random_faction_is_valid_faction_leader_for_root_target = {
	is_random_faction_allowed_faction_leader = yes
	NOT = { tag = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { is_in_faction_with = ROOT }
	NOT = {
		any_country = {
			is_in_faction_with = PREV
			has_war_with = ROOT
		}
	}
	ROOT = {
		random_faction_country_base_valid = yes
		NOT = { is_subject = yes }
		NOT = { is_in_faction = yes }
	}
}
