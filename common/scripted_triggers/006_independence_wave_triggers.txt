# ============================================================================
# CHAOS REDUX - EVENT 006 INDEPENDENCE WAVE TRIGGERS
# ============================================================================
# Public value bands, founding phases, origin checks, patron thresholds, host
# settlement state, network access, and league gates. No trigger here scans the
# world or changes state.
# ============================================================================

is_independence_wave_active_country = {
	exists = yes
	has_country_flag = independence_wave_active_origin
	check_variable = { liberation_origin = constant:liberation_origin.independence_wave }
	NOT = { has_country_flag = independence_wave_origin_ended }
}

# Country-local Event 006 decisions, categories, and focus content are
# player-facing for a real active Event 006 origin or a complete, human Event
# 021 adapter package. The adapter receives the same package-owned content,
# while its separate origin receipt keeps Event 006 firing, evolution, league,
# and network lifecycle state untouched.
is_independence_wave_event6_local_content_active = {
	exists = yes
	OR = {
		is_independence_wave_active_country = yes
		is_independence_wave_event021_package_country = yes
	}
}

is_independence_wave_package_content_active = {
	exists = yes
	OR = {
		is_independence_wave_active_country = yes
		AND = {
			# Event 021's adapter owns this bounded setup window. It may consume
			# package-local setup predicates, but it must not satisfy Event 006's
			# lifecycle classifier before its origin-neutral receipts are proven.
			has_country_flag = random_civil_war_event6_adapter_preparing
			has_country_flag = independence_wave_event6_adapter_package_ready
			has_valid_independence_wave_setup_input = yes
			NOT = { has_country_flag = independence_wave_active_origin }
			NOT = { has_country_flag = independence_wave_origin_ended }
		}
		AND = {
			has_country_flag = random_civil_war_event6_adapter_complete
			has_country_flag = independence_wave_event021_adapter_setup_proven
			has_country_flag = random_civil_war_event6_origin_recorded
			has_country_flag = random_civil_war_origin_adapter_complete
			has_variable = independence_wave_package_id
			NOT = { has_country_flag = independence_wave_active_origin }
			NOT = { has_country_flag = independence_wave_origin_ended }
		}
	}
}

# Package-local identity gate. A real Event 006 origin keeps its historical
# origin value. The Event 021 adapter receives an origin-neutral compatibility
# receipt during setup and after completion, so the reused package roots can
# use internal package checks without being misclassified as an Event 006
# origin or publishing an Event 006 player surface.
is_independence_wave_package_origin_compatible = {
	OR = {
		check_variable = { liberation_origin = constant:liberation_origin.independence_wave }
		AND = {
			random_civil_war_is_normal_human_country = yes
			OR = {
				AND = {
					has_country_flag = random_civil_war_event6_adapter_preparing
					has_country_flag = independence_wave_event6_adapter_package_ready
					has_valid_independence_wave_setup_input = yes
				}
				is_independence_wave_event021_package_country = yes
			}
		}
	}
}

# Player-facing Event 006 categories and decisions stay closed during the
# Event 021 adapter's transient setup window. Once the complete human package
# proof is recorded, its package-owned categories and decisions become
# available without setting Event 006's fired, evolution, league, or network
# lifecycle state.
is_independence_wave_event6_player_surface_allowed = {
	OR = {
		is_independence_wave_active_country = yes
		is_independence_wave_event021_package_country = yes
	}
}

# Country scope. Empty fixed-tag shells do not have a capital until their
# package states are transferred. Use an owned-state query for current-capital
# control checks so dormant shells fail closed without evaluating an invalid
# capital_scope target. PREV is the checked country, so the proof stays correct
# when a foreign stance, League decision, host response or the executor
# refreshes this country from another country's effect.

	any_owned_state = {
		is_capital = yes

	}

# Initialization reads regular event targets and unscoped temporary setup
# values. Every input is validated before the first origin flag or variable is
# mutated, so a malformed package cannot leave a partial active generation.
has_valid_independence_wave_setup_input = {
	exists = yes
	NOT = { has_country_flag = independence_wave_active_origin }
	NOT = { is_soviet_collapse_active_origin_country = yes }
	has_event_target = independence_wave_setup_former_host
	has_event_target = independence_wave_setup_anchor_state
	event_target:independence_wave_setup_former_host = {
		exists = yes
		NOT = { tag = PREV }
	}
	event_target:independence_wave_setup_anchor_state = {
		# This trigger is evaluated inside the target country during package
		# setup. PREV is that country; ROOT is still the country that fired the
		# event and may be the former host.
		is_owned_by = PREV
	}
	check_variable = { var = independence_wave_setup_package_id value = constant:independence_wave_plan.first_package_id compare = greater_than_or_equals }
	check_variable = { var = independence_wave_setup_package_id value = constant:independence_wave_registry.package_count compare = less_than_or_equals }
	check_variable = { var = independence_wave_setup_region value = constant:independence_wave_region.northern_western_europe compare = greater_than_or_equals }

# ---------------------------------------------------------------------------
# Liberations cluster member capacity
# ---------------------------------------------------------------------------
#
# The ordinary random-event pool and the Liberations cluster both reach this
# proof through an explicit member-availability context. It constructs one
# exact, deterministic witness from the currently registered runtime
# package adapters. It uses temporary arrays only and does not reserve a tag,
# state, host, or package in the shared transaction ledger.
#
# The witness is deliberately conservative. Every accepted Event 006 package
# is required to be disjoint from every possible Event 005 opening tag, core
# state, and host footprint before the joint allocator runs. A false negative
# can skip the optional member. A false positive could admit a joint incident
# that cannot reach the exact automatic count, so the proof never relies on a
# raw ready-row count.

# The automatic package readiness wrappers below evaluate their exact carrier
# inside the durable plan possible-country registry. The registry is populated
# with every_possible_country by both the pre-selection capacity probe and the
# live allocator, which keeps absent carriers in scope even after a nested
# scripted effect returns while preserving package identity and preflight
# checks.
