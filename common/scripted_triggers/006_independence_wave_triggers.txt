# Initialization reads regular event targets and unscoped temporary setup
# values. Every input is validated before the first origin flag or variable is
# mutated, so a malformed package cannot leave a partial active generation.
has_valid_independence_wave_setup_input = {
	exists = yes
	NOT = { has_country_flag = independence_wave_active_origin }
	NOT = { is_soviet_collapse_active_origin_country = yes }
	has_event_target = independence_wave_setup_former_host
	has_event_target = independence_wave_setup_anchor_state
	event_target:independence_wave_setup_former_host = {
		exists = yes
		NOT = { tag = PREV }
	}
	event_target:independence_wave_setup_anchor_state = {
		# This trigger is evaluated inside the target country during package
		# setup. PREV is that country; ROOT is still the country that fired the
		# event and may be the former host.
		is_owned_by = PREV
	}
	check_variable = { var = independence_wave_setup_package_id value = constant:independence_wave_plan.first_package_id compare = greater_than_or_equals }
	check_variable = { var = independence_wave_setup_package_id value = constant:independence_wave_registry.package_count compare = less_than_or_equals }
	check_variable = { var = independence_wave_setup_region value = constant:independence_wave_region.northern_western_europe compare = greater_than_or_equals }
	check_variable = { var = independence_wave_setup_region value = constant:independence_wave_region.americas_caribbean compare = less_than_or_equals }

	check_variable = { var = independence_wave_setup_depth value = constant:independence_wave_package_depth.signature compare = less_than_or_equals }
	check_variable = { var = independence_wave_setup_territory_level value = constant:liberation_territory_level.anchor compare = greater_than_or_equals }
	check_variable = { var = independence_wave_setup_territory_level value = constant:liberation_territory_level.extended compare = less_than_or_equals }
