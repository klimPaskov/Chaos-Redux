# Receiver-front intent predicates. Identity metadata is not population,
# reception, capacity, a bound destination or route permission. Map readers
# use only local scalar/cache checks and one exact selected-origin lookup.

migration_reception_front_intent_arrays_are_aligned = {
	hidden_trigger = {
		has_variable = migration_reception_front_intent_schema_version
		check_variable = { migration_reception_front_intent_schema_version = constant:migration_reception_front_intent.schema_version }
		check_variable = { migration_reception_front_intent_origins^num = migration_reception_front_intent_generations^num }
		check_variable = { migration_reception_front_intent_origins^num = migration_reception_front_intent_requesters^num }
		check_variable = { migration_reception_front_intent_origins^num = migration_reception_front_intent_receivers^num }
		check_variable = { migration_reception_front_intent_origins^num = migration_reception_front_intent_cohorts^num }
		check_variable = { migration_reception_front_intent_origins^num = migration_reception_front_intent_phases^num }
	}
}

# Receiving COUNTRY. This is the existing reveal threshold, with the exact
# validated origin amount rather than a pending-status dependency. The caller
# still supplies its pre-transition reveal receipt for successful consent.
migration_reception_front_intent_problem_qualifies = {
	hidden_trigger = {
		humanitarian_country_is_valid = yes
		has_event_target = migration_reception_front_intent_origin
		OR = {
			has_country_flag = migration_decision_emerging
			has_country_flag = migration_decision_active
			has_country_flag = migration_decision_resolution
			check_variable = { var = migration_incident_count value = constant:migration_decision_threshold.category_incidents compare = greater_than_or_equals }
			event_target:migration_reception_front_intent_origin = {
				check_variable = { var = humanitarian_corridor_transfer_people value = constant:migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }
			}
		}
	}
}

# Current origin STATE. All six identity_* values are unscoped temporary
# inputs, declared by the transaction walker or the pure cached reader.
migration_reception_front_intent_identity_is_current = {
	hidden_trigger = {
		if = {
			limit = { humanitarian_state_is_valid = yes }
			check_variable = { id = migration_reception_front_intent_identity_origin_id }
			has_state_flag = humanitarian_corridor_contract_active
			NOT = { has_state_flag = humanitarian_corridor_cleanup_pending }
			NOT = { has_state_flag = humanitarian_corridor_route_invalid }
			NOT = { has_state_flag = humanitarian_corridor_attack_received }
			NOT = { has_variable = humanitarian_corridor_evacuation_proven }
			check_variable = { humanitarian_corridor_origin_state_id = id }
			check_variable = { humanitarian_corridor_front_state_id = migration_reception_front_intent_identity_front_id }
			check_variable = { humanitarian_corridor_route_generation = migration_reception_front_intent_identity_generation }
			check_variable = { humanitarian_corridor_requester_country_id = migration_reception_front_intent_identity_requester }
			check_variable = { humanitarian_corridor_counterpart_country_id = migration_reception_front_intent_identity_receiver }
			check_variable = { humanitarian_corridor_cohort_id = migration_reception_front_intent_identity_cohort }
			check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation }
			is_owned_and_controlled_by = var:migration_reception_front_intent_identity_requester
			any_neighbor_state = { check_variable = { id = migration_reception_front_intent_identity_front_id } }
			NOT = { has_state_flag = migration_cohort_selection_ambiguous }
			NOT = { has_state_flag = migration_cohort_registry_misaligned }
			has_variable = migration_state_live_cohort_count
			check_variable = { migration_state_live_cohort_count = constant:humanitarian_runtime.one }
			check_variable = { migration_current_cohort_id = migration_reception_front_intent_identity_cohort }
			check_variable = { migration_state_live_cohort_amount = humanitarian_corridor_transfer_people }
			check_variable = { humanitarian_corridor_transfer_people > constant:humanitarian_population.zero }
			check_variable = { var = migration_flight_pressure value = humanitarian_corridor_transfer_people compare = greater_than_or_equals }
			check_variable = { var = migration_state_flight_population value = humanitarian_corridor_transfer_people compare = greater_than_or_equals }
			OR = {
				AND = {
					OR = {
						check_variable = { migration_reception_front_intent_identity_phase = constant:migration_reception_front_intent.requested }
						check_variable = { migration_reception_front_intent_identity_phase = constant:migration_reception_front_intent.prepared }
					}
					has_state_flag = humanitarian_corridor_offer_pending
					check_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.offer_pending }
					has_variable = humanitarian_corridor_offer_deadline
					check_variable = { global.date < humanitarian_corridor_offer_deadline }
				}
				AND = {
					check_variable = { migration_reception_front_intent_identity_phase = constant:migration_reception_front_intent.agreed }
					has_state_flag = humanitarian_corridor_response_accepted
					check_variable = { humanitarian_corridor_status = constant:humanitarian_corridor_status.accepted }
					has_variable = humanitarian_corridor_operation_deadline
					check_variable = { global.date < humanitarian_corridor_operation_deadline }
				}
			}
		}
		else = { always = no }
	}
}

# FRONT STATE. Prepared means actual still-proven services, not a guarantee
# of current admission headroom. Agreed means consent only.
migration_reception_front_intent_identity_front_is_current = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		check_variable = { id = migration_reception_front_intent_identity_front_id }
		is_owned_and_controlled_by = var:migration_reception_front_intent_identity_receiver
		var:migration_reception_front_intent_identity_requester = { humanitarian_country_is_valid = yes }
		var:migration_reception_front_intent_identity_receiver = { humanitarian_country_is_valid = yes }
		OR = {
			NOT = { check_variable = { migration_reception_front_intent_identity_phase = constant:migration_reception_front_intent.prepared } }
			AND = {
				check_variable = { migration_reception_shelter_service_proven = constant:migration_capacity.one }
				check_variable = { migration_reception_medical_service_proven = constant:migration_capacity.one }
				check_variable = { migration_reception_shelter_capacity > constant:humanitarian_population.zero }
				check_variable = { migration_reception_medical_capacity > constant:humanitarian_population.zero }
			}
		}
	}
}

migration_reception_front_intent_cached_identity_is_current = {
	hidden_trigger = {
		migration_reception_front_intent_arrays_are_aligned = yes
		has_state_flag = migration_reception_front_intent_cache_active
		check_variable = { migration_reception_front_intent_cache_origin_id > constant:migration_reception_front_intent.zero }
		check_variable = { migration_reception_front_intent_cache_generation > constant:migration_reception_front_intent.zero }
		check_variable = { migration_reception_front_intent_cache_requester > constant:migration_reception_front_intent.zero }
		check_variable = { migration_reception_front_intent_cache_receiver > constant:migration_reception_front_intent.zero }
		check_variable = { migration_reception_front_intent_cache_cohort > constant:migration_reception_front_intent.zero }
		set_temp_variable = { migration_reception_front_intent_identity_front_id = id }
		set_temp_variable = { migration_reception_front_intent_identity_origin_id = migration_reception_front_intent_cache_origin_id }
		set_temp_variable = { migration_reception_front_intent_identity_generation = migration_reception_front_intent_cache_generation }
		set_temp_variable = { migration_reception_front_intent_identity_requester = migration_reception_front_intent_cache_requester }
		set_temp_variable = { migration_reception_front_intent_identity_receiver = migration_reception_front_intent_cache_receiver }
		set_temp_variable = { migration_reception_front_intent_identity_cohort = migration_reception_front_intent_cache_cohort }
		set_temp_variable = { migration_reception_front_intent_identity_phase = migration_reception_front_intent_cache_phase }
		migration_reception_front_intent_identity_front_is_current = yes
		var:migration_reception_front_intent_cache_origin_id = { migration_reception_front_intent_identity_is_current = yes }
	}
}

migration_reception_front_intent_is_visible_to_player = {
	hidden_trigger = {
		OR = { is_owned_by = ROOT is_controlled_by = ROOT }
		migration_reception_front_intent_cached_identity_is_current = yes
	}
}

# Existing real detail contexts take precedence. This does not create a role,
# modify a ledger, inspect local receipts or scan a global registry.
migration_reception_front_intent_has_actual_map_context = {
	hidden_trigger = {
		OR = {
			has_state_flag = migration_population_trapped
			has_state_flag = migration_displacement_active
			has_state_flag = migration_reception_context_active
			has_state_flag = migration_overcrowded_context_active
			has_state_flag = migration_evacuation_prepared
			has_state_flag = migration_mission_evacuation_subject
			has_state_flag = migration_return_context_active
			has_state_flag = migration_state_resettlement_context_active
			has_variable = migration_current_cohort_id
			AND = { has_variable = migration_state_reception_load check_variable = { migration_state_reception_load > constant:humanitarian_population.zero } }
			AND = {
				has_state_flag = humanitarian_corridor_contract_active
				check_variable = { humanitarian_corridor_operation = constant:humanitarian_corridor_operation.evacuation }
			}
			AND = {
				has_variable = humanitarian_corridor_last_terminal_status
				check_variable = { humanitarian_corridor_last_terminal_operation = constant:humanitarian_corridor_operation.evacuation }
			}
		}
	}
}
