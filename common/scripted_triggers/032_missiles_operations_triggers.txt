missiles_selected_target_is_valid = {
	has_variable = missiles_selected_target
	var:missiles_selected_target = {
		missiles_target_is_valid = yes
		OR = {
			missiles_ordinary_target_is_valid = yes
			ROOT = {
				OR = {
					has_country_flag = missiles_rogue_operation_bypass_active
					has_country_flag = missiles_operation_rogue_origin
					check_variable = { var = missiles_operation_type value = constant:missiles_operation.retaliation compare = equals }
				}
			}
		}
	}
}

missiles_target_country_supports_operation = {
	missiles_ordinary_target_is_valid = yes
	OR = {
		ROOT = { NOT = { check_variable = { var = missiles_operation_type value = constant:missiles_operation.counterforce compare = equals } } }
		any_controlled_state = { missiles_counterforce_target_is_proven = yes }
	}
}

# COUNTRY scope. Maintenance stays reachable during war: in the operations
# phase it shows while no strike can be prepared, while a human has not begun
# choosing a target, or for the AI, so a worn-down program is never stuck
# between hidden preparation and hidden maintenance. Target selection and
# preparation replace it once a human starts choosing a target.
missiles_maintenance_actions_are_visible = {
	OR = {
		has_country_flag = missiles_program_phase_maintenance
		AND = {
			has_country_flag = missiles_program_phase_operations
			OR = {
				is_ai = yes
				NOT = { missiles_country_can_prepare_operation = yes }
				AND = {
					NOT = { has_country_flag = missiles_target_selected }
					NOT = { has_country_flag = missiles_target_country_page_open }
				}
			}
		}
	}
}

missiles_ordinary_preparation_action_is_visible = {
	has_country_flag = missiles_program_phase_operations
	missiles_country_can_prepare_operation = yes
	OR = {
		is_ai = yes
		missiles_human_target_is_complete = yes
	}
}

# STATE scope, ROOT is the operating country. A facility must be real, with
# adequate country intelligence; an arbitrary radar alone is not a launch site.
missiles_counterforce_target_is_proven = {
	OR = {
		has_state_flag = missiles_site_active
		has_state_flag = missiles_site_hardened
		has_state_flag = missiles_site_special_payload_custody
		rocket_site > constant:missiles_program_value.zero
		nuclear_reactor > constant:missiles_program_value.zero
		nuclear_facility > constant:missiles_program_value.zero
		cw_facility > constant:missiles_program_value.zero
		biowarfare_facility > constant:missiles_program_value.zero
	}
	CONTROLLER = {
		ROOT = { intel_level_over = { target = PREV civilian_intel > constant:missiles_tuning.operation_counterforce_intel } }
	}
}

# An existing receipt uses escrow, not the gate for starting another operation.
# Every participating site must still hold this receipt and its physical share.
missiles_operation_command_is_operational = {
	check_variable = { var = missiles_launch_readiness value = constant:missiles_tuning.operation_minimum_readiness compare = greater_than_or_equals }
	check_variable = { var = missiles_command_control value = constant:missiles_tuning.operation_minimum_control compare = greater_than_or_equals }
	OR = {
		NOT = { check_variable = { var = missiles_operation_type value = constant:missiles_operation.precision compare = equals } }
		check_variable = { var = missiles_launch_readiness value = constant:missiles_program_value.precision_readiness_gate compare = greater_than_or_equals }
	}
}

missiles_operation_sites_remain_valid = {
	missiles_operation_has_valid_site = yes
	any_of_scopes = { array = missiles_operation_sites is_controlled_by = ROOT }
	NOT = {
		any_of_scopes = {
			array = missiles_operation_sites
			NOT = {
				AND = {
					is_controlled_by = ROOT
					missiles_site_can_launch = yes
					check_variable = { var = missiles_site_operation_id value = ROOT.missiles_active_operation_id compare = equals }
					has_variable = missiles_site_operation_allocation
					check_variable = { var = missiles_site_operation_allocation value = constant:missiles_program_value.zero compare = greater_than }
					check_variable = { var = missiles_site_operation_allocation value = missiles_site_capacity compare = less_than_or_equals }
				}
			}
		}
	}
}

missiles_operation_tracks_are_enabled = {
	hidden_trigger = {
		OR = {
			NOT = { check_variable = { var = missiles_operation_strike_profile value = constant:missiles_strike_profile.saturation compare = equals } }
			missiles_saturation_active = yes
		}
		OR = {
			NOT = { check_variable = { var = missiles_operation_type value = constant:missiles_operation.retaliation compare = equals } }
			missiles_automatic_retaliation_active = yes
		}
		OR = {
			AND = {
				NOT = { has_country_flag = missiles_rogue_operation_bypass_active }
				NOT = { has_country_flag = missiles_operation_rogue_origin }
			}
			missiles_rogue_commands_active = yes
		}
		OR = {
			NOT = { missiles_operation_has_special_payload = yes }
			missiles_special_warheads_active = yes
		}
	}
}

# STATE scope with ROOT as the operating country. Family flags allow one secure
# site to hold several real payload families without overwriting another
# integration record.
missiles_site_has_selected_payload_custody = {
	has_state_flag = missiles_site_special_payload_custody
	OR = {
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.chemical compare = equals } } has_state_flag = missiles_site_custody_chemical }
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.biological compare = equals } } has_state_flag = missiles_site_custody_biological }
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.nuclear compare = equals } } has_state_flag = missiles_site_custody_nuclear }
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.thermonuclear compare = equals } } has_state_flag = missiles_site_custody_thermonuclear }
	}
}
