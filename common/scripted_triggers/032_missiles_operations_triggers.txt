# ============================================================================
# CHAOS REDUX - EVENT 032 OPERATION AND INCIDENT TRIGGERS
# ============================================================================
# Read-only gates for the prepared missile operation, payload, incident, and
# retaliation state machines. Persistent targets are checked before every
# transition; the companion effects file owns all mutations and receipts.
# ============================================================================

missiles_selected_target_is_valid = {
	has_variable = missiles_selected_target
	var:missiles_selected_target = {
		missiles_target_is_valid = yes
	}
}

missiles_selected_target_state_is_valid = {
	has_variable = missiles_selected_target
	has_variable = missiles_operation_target_state
	var:missiles_selected_target = {
		missiles_target_is_valid = yes
	}
	var:missiles_operation_target_state = {
		missiles_target_state_is_valid = yes
	}
}

missiles_operation_has_valid_site = {
	has_variable = missiles_operation_launch_site
	var:missiles_operation_launch_site = {
		is_controlled_by = ROOT
		missiles_site_can_launch = yes
	}
}

missiles_operation_receipt_is_active = {
	has_country_flag = missiles_operation_active
	has_variable = missiles_active_operation_id
	check_variable = {
		var = missiles_active_operation_id
		value = constant:missiles_program_value.zero
		compare = greater_than
	}
	has_variable = missiles_operation_state
	check_variable = {
		var = missiles_operation_state
		value = constant:missiles_operation_state.none
		compare = greater_than
	}
}

missiles_operation_before_commitment = {
	missiles_operation_receipt_is_active = yes
	NOT = { has_country_flag = missiles_operation_committed }
	OR = {
		check_variable = { var = missiles_operation_state value = constant:missiles_operation_state.selecting compare = equals }
		check_variable = { var = missiles_operation_state value = constant:missiles_operation_state.preparing compare = equals }
		check_variable = { var = missiles_operation_state value = constant:missiles_operation_state.resources_reserved compare = equals }
		check_variable = { var = missiles_operation_state value = constant:missiles_operation_state.ready compare = equals }
	}
}

missiles_operation_is_committed = {
	missiles_operation_receipt_is_active = yes
	has_country_flag = missiles_operation_committed
	OR = {
		check_variable = { var = missiles_operation_state value = constant:missiles_operation_state.committed compare = equals }
		check_variable = { var = missiles_operation_state value = constant:missiles_operation_state.resolving compare = equals }
	}
}

}

missiles_can_pay_reserve = {
	has_variable = missiles_reserve_payment_request
	check_variable = { var = missiles_reserve_payment_request value = constant:missiles_program_value.zero compare = greater_than }
	check_variable = { var = missiles_operational_reserve value = missiles_reserve_payment_request compare = greater_than_or_equals }

missiles_operation_can_be_committed = {
	missiles_operation_before_commitment = yes
	missiles_selected_target_state_is_valid = yes
	missiles_operation_has_valid_site = yes
	has_country_flag = missiles_operation_ready
	NOT = { has_country_flag = missiles_incident_active }
	has_country_flag = missiles_operation_reserve_receipt_active
	has_variable = missiles_reserved_reserve
	check_variable = { var = missiles_reserved_reserve value = constant:missiles_program_value.zero compare = greater_than }
	check_variable = { var = missiles_operation_missile_cost value = missiles_reserved_reserve compare = less_than_or_equals }
	OR = {
		NOT = { has_variable = missiles_next_launch_date }
		check_variable = { var = global.date value = missiles_next_launch_date compare = greater_than_or_equals }
	}
}

		check_variable = { var = missiles_operation_payload value = constant:missiles_payload.thermonuclear compare = equals }
	}
}

missiles_operation_target_has_strategic_value = {
	OR = {
		arms_factory > constant:missiles_program_value.zero
		industrial_complex > constant:missiles_program_value.zero
		dockyard > constant:missiles_program_value.zero
		air_base > constant:missiles_program_value.zero
		radar_station > constant:missiles_program_value.zero
		naval_base > constant:missiles_program_value.zero
		supply_node > constant:missiles_program_value.zero
		rail_way > constant:missiles_program_value.zero
		rocket_site > constant:missiles_program_value.zero
		is_capital = yes
	}
}

missiles_operation_target_has_profile_value = {
	OR = {
		AND = {
			check_variable = { var = ROOT.missiles_operation_target_profile value = constant:missiles_target_profile.logistics compare = equals }
			OR = { supply_node > constant:missiles_program_value.zero rail_way > constant:missiles_program_value.zero infrastructure > constant:missiles_tuning.site_min_infrastructure naval_base > constant:missiles_program_value.zero }
		}
		AND = {
			check_variable = { var = ROOT.missiles_operation_target_profile value = constant:missiles_target_profile.industrial compare = equals }
			OR = { arms_factory > constant:missiles_program_value.zero industrial_complex > constant:missiles_program_value.zero dockyard > constant:missiles_program_value.zero synthetic_refinery > constant:missiles_program_value.zero fuel_silo > constant:missiles_program_value.zero }
		}
		AND = {
			check_variable = { var = ROOT.missiles_operation_target_profile value = constant:missiles_target_profile.air_coastal compare = equals }
			OR = { air_base > constant:missiles_program_value.zero radar_station > constant:missiles_program_value.zero anti_air_building > constant:missiles_program_value.zero naval_base > constant:missiles_program_value.zero coastal_bunker > constant:missiles_program_value.zero }
		}
		AND = {
			check_variable = { var = ROOT.missiles_operation_target_profile value = constant:missiles_target_profile.command compare = equals }
			OR = { is_capital = yes rocket_site > constant:missiles_program_value.zero radar_station > constant:missiles_program_value.zero }
		}
		AND = {
			check_variable = { var = ROOT.missiles_operation_target_profile value = constant:missiles_target_profile.counterforce compare = equals }
			OR = { rocket_site > constant:missiles_program_value.zero radar_station > constant:missiles_program_value.zero nuclear_reactor > constant:missiles_program_value.zero nuclear_facility > constant:missiles_program_value.zero }
		}
		AND = {
			check_variable = { var = ROOT.missiles_operation_target_profile value = constant:missiles_target_profile.broad_strategic compare = equals }
			missiles_operation_target_has_strategic_value = yes
		}
	}
}

# STATE scope with ROOT as the operating country. Family flags allow one secure
# site to hold several real payload families without overwriting another
# integration record.
missiles_site_has_selected_payload_custody = {
	has_state_flag = missiles_site_special_payload_custody
	OR = {
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.chemical compare = equals } } has_state_flag = missiles_site_custody_chemical }
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.biological compare = equals } } has_state_flag = missiles_site_custody_biological }
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.nuclear compare = equals } } has_state_flag = missiles_site_custody_nuclear }
		AND = { ROOT = { check_variable = { var = missiles_operation_payload value = constant:missiles_payload.thermonuclear compare = equals } } has_state_flag = missiles_site_custody_thermonuclear }
	}
}

# STATE scope. Natural-disaster mutation must have enough current-controller
# custody for every reserve ledger that the bridge will settle.
missiles_natural_disaster_reserve_settlement_is_valid = {
	OR = {
		NOT = { has_variable = missiles_site_reserve }
		check_variable = { var = missiles_site_reserve value = constant:missiles_program_value.zero compare = less_than_or_equals }
		CONTROLLER = {
			has_variable = missiles_operational_reserve
			check_variable = { var = missiles_operational_reserve value = PREV.missiles_site_reserve compare = greater_than_or_equals }
		}
	}
	OR = {
		NOT = { has_variable = missiles_site_captured_reserve }
		check_variable = { var = missiles_site_captured_reserve value = constant:missiles_program_value.zero compare = less_than_or_equals }
		CONTROLLER = {
			has_variable = missiles_captured_reserve_held
			check_variable = { var = missiles_captured_reserve_held value = PREV.missiles_site_captured_reserve compare = greater_than_or_equals }
		}
	}
}

		check_variable = { var = missiles_site_security value = constant:missiles_reserve.site_secure_security_threshold compare = greater_than_or_equals }
	}
}
