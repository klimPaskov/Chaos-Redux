# ============================================================================
# CHAOS REDUX - EVENT 020 BLACK PLAGUE RESPONSE TRIGGERS
# ============================================================================
# Country and state predicates for the selected-state Black Plague response
# package. Human decisions are bound to the state selected on the shared
# disease board; AI countries evaluate every eligible controlled state. Costs
# use state-owned population-scaled quotes and country-owned capacity values.
# ============================================================================

black_plague_country_can_direct_response = {
	exists = yes
	black_plague_country_is_human_host = yes
	has_global_flag = black_plague_system_started
	NOT = { has_global_flag = black_plague_terminal_takeover_complete }
	NOT = { has_global_flag = world_end }
}

black_plague_response_decisions_are_visible = {
	black_plague_country_can_direct_response = yes
	OR = {
		is_ai = yes
		disease_containment_board_is_black_plague_selected = yes
	}
}

# STATE scope; ROOT is the responding country.
black_plague_response_state_is_selected_or_ai_target = {
	is_controlled_by = ROOT
	OR = {
		ROOT = { is_ai = yes }
		AND = {
			ROOT = {
				is_ai = no
				has_variable = black_plague_selected_response_state
			}
			check_variable = {
				var = id
				value = ROOT.black_plague_selected_response_state
				compare = equals
			}
		}
	}
}

# STATE scope; ROOT is the responding country.
black_plague_response_state_can_accept_actions = {
	black_plague_state_is_tracked = yes
	black_plague_response_state_is_selected_or_ai_target = yes
	NOT = { has_state_flag = black_plague_response_failed_mobilization }
	NOT = { black_plague_state_is_rat_controlled = yes }
	NOT = { black_plague_controller_is_rat_country = yes }
	ROOT = { black_plague_country_can_direct_response = yes }
}

black_plague_response_state_is_urban_or_dense = {
	OR = {
		has_state_category = city
		has_state_category = large_city
		has_state_category = metropolis
		has_state_category = megalopolis
		is_capital = yes
		check_variable = {
			var = state_population_k
			value = constant:black_plague_response_threshold.dense_state_population_k
			compare = greater_than_or_equals
		}
	}
}

# A state-level supply/access proof. It is deliberately local and route based;
# national stockpiles alone do not make an isolated state reachable.
black_plague_response_state_has_access = {
	OR = {
		infrastructure > constant:black_plague_response_threshold.response_access_infrastructure_floor
		supply_node > constant:black_plague_response_threshold.response_access_hub_floor
		naval_base > constant:black_plague_response_threshold.response_access_hub_floor
		any_neighbor_state = {
			is_controlled_by = ROOT
			impassable = no
			infrastructure > constant:black_plague_response_threshold.response_access_infrastructure_floor
		}
	}
}

# STATE scope; ROOT is the responding country.
black_plague_response_state_has_containment_force = {
	black_plague_response_state_has_access = yes
	any_state_division = {
		OWNER = { tag = ROOT }
		unit_strength > constant:black_plague_response_penalty.military_force_strength
		unit_organization > constant:black_plague_response_penalty.military_force_organization
	}
}
