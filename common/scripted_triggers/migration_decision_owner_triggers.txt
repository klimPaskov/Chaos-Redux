# Migration reveal evidence must be sustained or large. Merely initializing a
# cohort, registry, or country does not expose the category.
migration_pending_reception_issue_is_valid = {
	hidden_trigger = {
		humanitarian_corridor_offer_is_valid = yes
		check_variable = { var = humanitarian_corridor_offer_operation value = constant:humanitarian_corridor_operation.evacuation compare = equals }
		OR = {
			has_country_flag = migration_decision_emerging
			has_country_flag = migration_decision_active
			has_country_flag = migration_decision_resolution
			check_variable = { var = migration_incident_count value = constant:migration_decision_threshold.category_incidents compare = greater_than_or_equals }
			var:humanitarian_corridor_offer_origin_state_id = {
				# Offer validity binds this amount to the exact live origin row.
				# Unrelated national pressure is not the receiver's intake amount.
				check_variable = { var = humanitarian_corridor_transfer_people value = constant:migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }
			}
		}
	}
}

migration_decision_reveal_threshold_is_met = {
		hidden_trigger = {
		OR = {
			check_variable = { var = migration_incident_count value = constant:migration_decision_threshold.category_incidents compare = greater_than_or_equals }
			AND = {
				migration_pending_reception_issue_is_valid = yes
			}
			AND = {
				has_variable = migration_reception_load
				check_variable = { var = migration_reception_load value = constant:migration_decision_threshold.category_reception_load compare = greater_than }
			}
			check_variable = { var = migration_flight_pressure value = constant:migration_decision_threshold.category_flight_people compare = greater_than_or_equals }
			check_variable = { var = migration_trapped_population value = constant:migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }
			any_owned_state = {
				OR = {
					check_variable = { var = migration_state_flight_population value = constant:migration_decision_threshold.category_flight_people compare = greater_than_or_equals }
					check_variable = { var = migration_state_trapped_population value = constant:migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }
					check_variable = { var = migration_state_reception_load value = constant:migration_decision_threshold.category_reception_load compare = greater_than_or_equals }
				}
			}
		}
	}
}

# Country/state seams used by famine-owned evacuation decisions. They expose
# only migration-owned capacity and preparation availability; the famine
# decision separately proves its own food-security cause.
migration_decision_famine_evacuation_country_is_available = {
	migration_reception_has_room_for_evacuation = yes
}

# COUNTRY scope. An organized evacuation may start while the country hosts no
# arrivals, or while its measured reception load is still below its current
# capacity. Only an overloaded host must first distribute, integrate, or
# resettle the people it already shelters.
migration_reception_has_room_for_evacuation = {
	hidden_trigger = {
		OR = {
			NOT = { has_variable = migration_reception_load }
			NOT = { check_variable = { var = migration_reception_load value = constant:humanitarian_population.zero compare = greater_than } }
			AND = {
				has_variable = migration_reception_capacity
				NOT = { check_variable = { var = migration_reception_load value = migration_reception_capacity compare = greater_than } }
			}
		}
	}
}

# STATE scope. A state needs an organized evacuation when its displacement is
# active and its flight pressure or trapped population has reached the
# sustained-issue threshold.
migration_state_needs_organized_evacuation = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		has_state_flag = migration_displacement_active
		OR = {
			check_variable = { var = migration_flight_pressure value = constant:migration_decision_threshold.category_flight_people compare = greater_than_or_equals }
			check_variable = { var = migration_trapped_population value = constant:migration_decision_threshold.category_trapped_people compare = greater_than_or_equals }
		}
	}
}

# The acting state must have exactly one aligned, live positive cohort row.  The
# core reconciler derives these counters from the persisted host state, so an
# old selector, stale origin, or ambiguous row fails closed.
migration_decision_exact_hosted_cohort_is_valid = {
	hidden_trigger = {
		if = {
			limit = {
				humanitarian_state_is_valid = yes
			}
			migration_cohort_arrays_are_aligned = yes
			NOT = { has_state_flag = migration_cohort_selection_ambiguous }
			has_variable = migration_current_cohort_id
			has_variable = migration_state_live_cohort_count
			check_variable = { var = migration_state_live_cohort_count value = constant:humanitarian_runtime.one compare = equals }
			has_variable = migration_state_live_cohort_amount
			check_variable = { var = migration_state_live_cohort_amount value = constant:humanitarian_population.zero compare = greater_than }
		}
		else = { always = no }
	}
}

# A host action needs a valid current country capacity revision and a positive
# measured load.  Capacity validity is intentionally separate from the load
# check: a stale or dirty revision cannot authorize a transfer.
migration_decision_reception_host_is_valid = {
	hidden_trigger = {
		humanitarian_country_is_valid = yes
		migration_reception_capacity_is_valid = yes
		has_variable = migration_reception_load
		check_variable = { var = migration_reception_load value = constant:humanitarian_population.zero compare = greater_than }
		has_variable = migration_reception_capacity
		check_variable = { var = migration_reception_load value = migration_reception_capacity compare = less_than }
	}
}

# Outbound relief lanes may be needed precisely while the host is above its

		has_variable = migration_reception_capacity
	}
}

# Durable receipt for the Hungry, Not Contagious outcome.  The receipt is
# written only by the decision-owned arrival adapter after the exact transfer,
# bind, reception delta, and history visit all succeed.  It binds one live
# cohort to its host and preserves the arrival-time origin/exposure snapshot;
# current host infection or a policy flag cannot satisfy it.
migration_decision_reception_exposure_receipt_is_valid = {
	hidden_trigger = {
		humanitarian_state_is_valid = yes
		has_state_flag = migration_mission_reception_exposure_receipt
		has_variable = migration_mission_reception_receipt_cohort_id
		check_variable = { var = migration_mission_reception_receipt_cohort_id value = constant:humanitarian_runtime.zero compare = greater_than }
		has_variable = migration_mission_reception_receipt_origin_state_id
		check_variable = { var = migration_mission_reception_receipt_origin_state_id value = constant:humanitarian_runtime.zero compare = greater_than }
		has_variable = migration_mission_reception_receipt_host_state_id
		check_variable = { var = migration_mission_reception_receipt_host_state_id value = THIS.id compare = equals }
		has_variable = migration_mission_reception_receipt_generation
		check_variable = { var = migration_mission_reception_receipt_generation value = constant:humanitarian_runtime.zero compare = greater_than }
		has_variable = migration_mission_reception_receipt_transaction
		check_variable = { var = migration_mission_reception_receipt_transaction value = migration_mission_reception_receipt_generation compare = equals }
		has_variable = migration_mission_reception_receipt_exposure_class
		check_variable = { var = migration_mission_reception_receipt_exposure_class value = constant:humanitarian_pressure_source.outbreak compare = equals }
		migration_cohort_arrays_are_aligned = yes
		migration_decision_exact_hosted_cohort_is_valid = yes
		check_variable = { var = migration_current_cohort_id value = migration_mission_reception_receipt_cohort_id compare = equals }
	}
}
