# ============================================================================
# CHAOS REDUX - EVENT 020 ACHIEVEMENT CONTRACTS
# ============================================================================
# Player-facing completion contracts for the fourteen Black Plague
# achievements. The event-owned pulse writes the historical flags and
# variables consumed below; no achievement is unlocked merely because Event
# 020 fired.
# ============================================================================

black_plague_achievement_is_player = {
	is_ai = no
	NOT = { has_country_flag = black_plague_achievement_forced_setup_used }
	NOT = { has_country_flag = black_plague_achievement_debug_completion_used }
}

black_plague_achievement_natural_campaign_is_valid = {
	black_plague_achievement_is_player = yes
	NOT = { has_global_flag = black_plague_scenario_shortcut_disqualifier }
	NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap }
	NOT = { has_global_flag = black_plague_triggerable_scenario_launched }
	NOT = { has_global_flag = black_plague_triggerable_scenario_history_recorded }
}

black_plague_achievement_rat_actor_is_player = {
	black_plague_achievement_natural_campaign_is_valid = yes
	OR = {
		has_country_flag = black_plague_rat_country
		has_country_flag = black_plague_rat_king_country
	}
}

black_plague_achievement_human_actor_is_player = {
	black_plague_achievement_natural_campaign_is_valid = yes
	black_plague_country_is_human_host = yes
}

black_plague_achievement_first_cordon_is_complete = {
	black_plague_achievement_natural_campaign_is_valid = yes
	check_variable = {
		var = THIS.id
		value = global.black_plague_achievement_origin_owner_id
		compare = equals
	}
	has_global_flag = black_plague_achievement_origin_cured
	NOT = { has_global_flag = black_plague_achievement_second_infection_seen }
	NOT = { has_global_flag = black_plague_achievement_rat_emergence_seen }
	NOT = { has_global_flag = black_plague_rat_king_active }
	NOT = { has_country_flag = black_plague_achievement_weaponization_started }
	has_variable = global.black_plague_achievement_origin_death_limit_people
	check_variable = {
		var = global.black_plague_achievement_origin_deaths_people
		value = global.black_plague_achievement_origin_death_limit_people
		compare = less_than
	}
}

	black_plague_achievement_rat_actor_is_player = yes
	has_country_flag = black_plague_rat_king_country
	# This achievement is deliberately pre-Evolution-V. The earned terminal
	# route is gated by Evolution V itself, so the reachable pre-terminal route
	# marker is the completed government/core route.
	has_global_flag = black_plague_rat_king_core_route_completed
	NOT = { has_global_flag = black_plague_evolution_v_recorded }
	NOT = { has_global_flag = black_plague_terminal_takeover_complete }
	NOT = { has_global_flag = world_end }
	has_global_flag = black_plague_rat_king_target_continent_selected
	check_variable = {
		var = global.black_plague_rat_king_target_continent_ratio
		value = constant:black_plague_achievement_threshold.crown_continent_ratio
		compare = greater_than_or_equals
	}
