# ============================================================================
# CHAOS REDUX - EVENT 006 PRE-WAVE CRISIS TRIGGERS
# ============================================================================
# These country-scoped triggers expose a crisis only for a low-stability host,
# an enemy-occupied owned state, or a severely resistant foreign-owned state
# under the country's control. No world-wide periodic scan is used. The
# release gate delegates to the shared Liberations coordinator and never
# bypasses its frozen-plan contract.
# ============================================================================

has_independence_wave_crisis_occupation_pressure = {
	OR = {
		any_owned_state = {
			NOT = { is_controlled_by = ROOT }
			resistance > constant:independence_wave_crisis.occupied_resistance_threshold
		}
		any_controlled_state = {
			NOT = { is_owned_by = ROOT }
			resistance > constant:independence_wave_crisis.occupied_resistance_threshold
		}
	}
}

has_independence_wave_crisis_stability_pressure = {
	has_stability < constant:independence_wave_crisis.stability_threshold
}

has_independence_wave_crisis_pressure = {
	OR = {
		has_independence_wave_crisis_occupation_pressure = yes
		has_independence_wave_crisis_stability_pressure = yes
	}
}

can_independence_wave_crisis_release_barrier = {
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = independence_wave_joint_presentation_pending }
	NOT = { SOV = { has_country_flag = soviet_collapse_joint_opening_presentation_pending } }
	can_liberation_release_reset_plan = yes
}

can_independence_wave_open_crisis = {
	has_independence_wave_crisis_pressure = yes
	NOT = { has_country_flag = independence_wave_crisis_active }
	NOT = { has_country_flag = independence_wave_crisis_cooldown }
	NOT = { has_global_flag = independence_wave_crisis_release_queued }
	can_independence_wave_crisis_release_barrier = yes
}

can_pay_independence_wave_crisis_cost = {
	can_pay_independence_wave_security_standard_cost = yes
	command_power > constant:independence_wave_decision_cost.command_power_standard
}
