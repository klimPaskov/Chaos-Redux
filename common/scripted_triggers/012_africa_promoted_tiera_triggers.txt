	OR = {
		AND = { africa_promoted_tiera_carrier_is_pan = yes africa_promoted_tiera_state_pan_is_eligible = yes }
		AND = { africa_promoted_tiera_carrier_is_gorilla_kingdom = yes africa_promoted_tiera_state_gorilla_kingdom_is_eligible = yes }
		AND = { africa_promoted_tiera_carrier_is_the_green = yes africa_promoted_tiera_state_the_green_is_eligible = yes }
		AND = { africa_promoted_tiera_carrier_is_living_rivers = yes africa_promoted_tiera_state_living_rivers_is_eligible = yes }
		AND = { africa_promoted_tiera_carrier_is_stoneborn = yes africa_promoted_tiera_state_stoneborn_is_eligible = yes }
		AND = { africa_promoted_tiera_carrier_is_ancient_hosts = yes africa_promoted_tiera_state_ancient_hosts_is_eligible = yes }
	}
}

africa_promoted_tiera_activation_is_open = {
	AND = {
		has_global_flag = africa_event_active
		has_global_flag = africa_evolution_iii_logged
		has_event_target = africa_host
		event_target:africa_host = {
			has_country_flag = africa_host_commit_completed
			check_variable = { africa_current_phase value = constant:africa_action_phase.high_chaos compare = greater_than_or_equals }
		}
	}
}
