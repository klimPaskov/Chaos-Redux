# Internal trigger. The row scanners stage a positive living row's immutable
# identity in camp_origin_candidate_* and call this in their current scope.
camp_origin_accumulate_inspection_row = {
	if = {
		limit = { check_variable = { camp_origin_candidate_amount_k > constant:camp_admin_value.zero } }
		if = {
			limit = {
				check_variable = { camp_origin_live_row_count > constant:camp_admin_value.zero }
				OR = {
					NOT = { check_variable = { camp_origin_candidate_family = camp_origin_first_family } }
					NOT = { check_variable = { camp_origin_candidate_origin = camp_origin_first_origin } }
					NOT = { check_variable = { camp_origin_candidate_authority = camp_origin_first_authority } }
					NOT = { check_variable = { camp_origin_candidate_root = camp_origin_first_root } }
					AND = {
						check_variable = { camp_origin_candidate_root = constant:camp_admin_value.zero }
						NOT = { check_variable = { camp_origin_candidate_row_id = camp_origin_first_row_id } }
					}
				}
			}
			set_temp_variable = { camp_origin_mixed_lineages = constant:camp_admin_value.one }
		}
		if = {
			limit = { check_variable = { camp_origin_live_row_count = constant:camp_admin_value.zero } }
			set_temp_variable = { camp_origin_first_family = camp_origin_candidate_family }
			set_temp_variable = { camp_origin_first_origin = camp_origin_candidate_origin }
			set_temp_variable = { camp_origin_first_authority = camp_origin_candidate_authority }
			set_temp_variable = { camp_origin_first_root = camp_origin_candidate_root }
			set_temp_variable = { camp_origin_first_row_id = camp_origin_candidate_row_id }
			set_temp_variable = { camp_origin_first_amount_k = camp_origin_candidate_amount_k }
			set_temp_variable = { camp_origin_first_removed_k = camp_origin_candidate_removed_k }
			set_temp_variable = { camp_origin_first_deaths_k = camp_origin_candidate_deaths_k }
			set_temp_variable = { camp_origin_first_released_k = camp_origin_candidate_released_k }
		}
		add_to_temp_variable = { camp_origin_live_row_count = constant:camp_admin_value.one }
		if = {
			limit = { check_variable = { camp_origin_candidate_family = constant:camp_origin_family.custody } }
			add_to_temp_variable = { camp_origin_inspect_custody_k = camp_origin_candidate_amount_k }
		}
		if = {
			limit = { check_variable = { camp_origin_candidate_family = constant:camp_origin_family.occupation } }
			add_to_temp_variable = { camp_origin_inspect_occupation_k = camp_origin_candidate_amount_k }
		}
	}
	# Always return true so the caller's bounded read-only scan can finish.
	check_variable = { camp_origin_candidate_amount_k = camp_origin_candidate_amount_k }
}

	NOT = { has_state_flag = camp_origin_ambiguous }
	NOT = { has_state_flag = camp_origin_baseline_unverified }

	NOT = { has_state_flag = camp_origin_counter_regressed }

		has_variable = camp_occ_responsible_country

	OR = {
		AND = {

			check_variable = { camp_origin_inspect_custody_k = constant:camp_admin_value.zero }
			NOT = { has_state_flag = camp_origin_single_origin_reconciliation_pending }
		}
		camp_origin_host_provenance_ready = yes
	}
}
