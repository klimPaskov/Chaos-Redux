# Country groups/collections provide the reusable array API; these triggers
# provide safe country-scope predicates for gameplay call sites. Membership is
# never sufficient to activate content: callers must also require the creating
# event's package/origin markers.

# Event 012 may use a carrier only after it has recorded its own package origin
# and neither Event 006 nor Event 005 owns the live origin.

	has_country_flag = africa_priority_member_package_active
	has_country_flag = africa_priority_member_origin_validated
	NOT = { is_independence_wave_registry_event6_origin = yes }
	NOT = { is_independence_wave_registry_soviet_origin = yes }
}
