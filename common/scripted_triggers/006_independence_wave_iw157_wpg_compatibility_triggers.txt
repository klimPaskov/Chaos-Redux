# Dormant, package-local preservation predicates for IW-157. The package is
# a specific-community variant only and remains unbound. WPG is a registered
# vanilla carrier; these predicates never turn the broad West Papua label into
# a country candidate and never replace the registered identity.
#
# Every predicate below is gated by the living country's original WPG tag, an
# active Event 006 origin, the exact IW-157 package id, and a future named
# community selector. The selector is reserved for a later source-backed
# named people-and-district package and is not set in this file.
# ============================================================================

# COUNTRY scope. This is the shared context gate for the dormant adapter.
# is_independence_wave_active_country already requires the Event 006 origin
# flag, liberation_origin value, and an un-ended origin. The explicit checks
# remain here so this package-local boundary cannot be weakened by a broader
# registry membership test.
is_independence_wave_iw_157_wpg_compatibility_context = {
	exists = yes
	original_tag = WPG
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_active_origin
	check_variable = { liberation_origin = constant:liberation_origin.independence_wave }
	check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_157 }
	has_country_flag = independence_wave_iw_157_named_community_selected
}

# COUNTRY scope. The Event 006 route must own and control the exact WPG
# research anchor and keep state 669 as its capital. The state core witness
# preserves the installed vanilla carrier history without mutating it.
has_independence_wave_iw_157_wpg_anchor_surface = {
	is_independence_wave_iw_157_wpg_compatibility_context = yes
	owns_state = 669
	controls_state = 669
	capital_scope = { state = 669 }
	669 = { is_core_of = WPG }
}

# COUNTRY scope. Read-only witnesses for the remaining installed vanilla WPG
# core and Indonesian releasable row. No adapter effect adds, removes, or
# rewrites either source surface.
has_independence_wave_iw_157_wpg_vanilla_surface = {
	is_independence_wave_iw_157_wpg_compatibility_context = yes
	1057 = { is_core_of = WPG }
	INS = {
		exists = yes
		is_in_array = { array = INS_releasables value = WPG }
	}
}

# COUNTRY scope. Strict preservation contract for a future named-community
# consumer. It is intentionally not a generic West Papua, Papuan, or
# pan-Melanesian release trigger.
has_independence_wave_iw_157_wpg_compatibility_contract = {
	is_independence_wave_iw_157_wpg_compatibility_context = yes
	has_independence_wave_iw_157_wpg_anchor_surface = yes
	has_independence_wave_iw_157_wpg_vanilla_surface = yes
}
