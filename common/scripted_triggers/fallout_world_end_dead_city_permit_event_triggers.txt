# ============================================================================
# CHAOS REDUX - FALLOUT DEAD CITY PERMIT TRIGGERS
# ============================================================================
# Country-owned expedition chain with one deterministic dead-city state id.
# Every receipt checks the Fallout registry, owner, generation, and target.
# ============================================================================

fallout_event_338_dead_city_state_is_current = {
	OR = {
		AND = {
			has_state_category = city
			check_variable = { fallout_state_grade = constant:fallout_state_grade.dead_city }
		}
		AND = {
			has_state_category = large_city
			check_variable = { fallout_state_grade = constant:fallout_state_grade.dead_city }
		}
	}
	check_variable = { fallout_pretransition_air_winter_exposure value = constant:fallout_event_338_threshold.minimum_city_exposure compare = greater_than_or_equals }
	check_variable = { state_population_k value = constant:fallout_event_338_threshold.minimum_city_population_k compare = greater_than_or_equals }
}

fallout_event_338_country_is_current = {
	fallout_event_registry_country_row_is_current = yes
	fallout_survival_country_resource_row_is_durable = yes
	has_variable = fallout_event_338_candidate_state_id
	check_variable = { fallout_event_338_candidate_state_id > constant:fallout_event_runtime.zero }
	check_variable = { var = global.num_days value = constant:fallout_event_338_threshold.minimum_campaign_day compare = greater_than_or_equals }
	check_variable = { var = global.num_days value = constant:fallout_event_338_threshold.maximum_campaign_day compare = less_than }
	check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_338_threshold.minimum_scrap compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_338_threshold.minimum_medicine compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^5 value = constant:fallout_event_338_threshold.minimum_fuel compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_338_threshold.minimum_recognition compare = greater_than_or_equals }
	check_variable = { fallout_survival_cohesion_current value = constant:fallout_event_338_threshold.minimum_cohesion compare = greater_than_or_equals }
	has_country_flag = fallout_event_331_memory_closed
	NOT = { has_country_flag = fallout_event_338_chain_committed }
	NOT = { has_country_flag = fallout_event_338_memory_closed }
	OR = {
		AND = {
			check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_338_cost.survey_scrap compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^5 value = constant:fallout_event_338_cost.survey_fuel compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_338_cost.survey_medicine compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_338_cost.license_scrap compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_338_cost.license_recognition compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^5 value = constant:fallout_event_338_cost.license_fuel compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_338_cost.military_scrap compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^5 value = constant:fallout_event_338_cost.military_fuel compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_338_cost.military_power compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { fallout_survival_current_entries^8 value = constant:fallout_event_338_cost.forbid_shelter compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_338_cost.forbid_filters compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_338_cost.forbid_scrap compare = greater_than_or_equals }
		}
	}
	any_owned_state = {
		check_variable = { var = THIS.id value = ROOT.fallout_event_338_candidate_state_id compare = equals }
		fallout_event_338_dead_city_state_is_current = yes
	}
}

fallout_event_338_registry_is_current = {
	has_variable = fallout_event_338_registry_generation
	has_variable = fallout_event_338_registry_owner
	has_variable = fallout_event_338_candidate_id
	has_variable = fallout_event_338_registry_target
	has_variable = fallout_event_338_target_state_id
	has_variable = fallout_event_338_frozen_scrap
	has_variable = fallout_event_338_frozen_medicine
	has_variable = fallout_event_338_frozen_fuel
	has_variable = fallout_event_338_frozen_cohesion
	has_variable = fallout_event_338_frozen_recognition
	check_variable = { fallout_event_338_registry_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_event_338_registry_owner value = ROOT compare = equals }
	check_variable = { fallout_event_338_candidate_id value = constant:fallout_event_candidate_pilot.dead_city_permit_candidate_id compare = equals }
	check_variable = { fallout_event_338_registry_target value = constant:fallout_event_runtime.zero compare = equals }
	has_country_flag = fallout_event_338_chain_committed
	fallout_event_registry_country_row_is_current = yes
	fallout_survival_country_resource_row_is_durable = yes
	any_owned_state = { check_variable = { var = THIS.id value = ROOT.fallout_event_338_target_state_id compare = equals } }
}

fallout_event_338_opening_target_is_current = {
	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.none }
	check_variable = { fallout_event_dispatch_issued_target value = constant:fallout_event_runtime.zero compare = equals }
	fallout_event_338_country_is_current = yes
}

	check_variable = { fallout_event_dispatch_issued_mode = constant:fallout_event_dispatch_mode.human_visible }
	fallout_event_338_opening_target_is_current = yes
}
