# CHAOS REDUX - EVENT 006 FORM-07 IBERIAN FEDERATION TRIGGERS
# ============================================================================
# Overview:
# - Binds the reserved Iberian family to the exact CAT/NAV/GLC package IDs,
#   installed anchors, and live Event 006 member records.
# - Keeps the founding carrier and route gates separate from identity review;
#   a carrier can never pass the transaction without the explicit, reviewed
#   FORM-07 X-tag and flag contract.
# - Reuses the shared frozen invitation ledger and bounded active-country
#   registry. No trigger performs an all-country or periodic scan.
# ============================================================================

# Country scope. This is the only member-candidate allowlist for FORM-07.
# Package IDs and original tags are both required so a reused tag or a stale
# package variable cannot silently enter the Iberian ledger.
is_independence_wave_form07_member_candidate = {
	exists = yes
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_formable_family_selected
	check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.iberian_federation }
	has_country_flag = independence_wave_package_setup_complete
	NOT = { has_country_flag = independence_wave_form07_autonomous_member }
	OR = {
		AND = {
			original_tag = CAT
			check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_014 }
		}
		AND = {
			original_tag = NAV
			check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_013 }
		}
		AND = {
			original_tag = GLC
			check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_015 }
		}
	}
}

# Country scope. The anchor crosswalk is exact and remains ownership/control
# authoritative. Capital movement is deliberately left to the future identity
# adapter, because no reviewed FORM-07 X identity currently exists.
has_independence_wave_form07_exact_member_anchor = {
	is_independence_wave_form07_member_candidate = yes
	OR = {
		AND = {
			original_tag = CAT
			check_variable = { independence_wave_anchor_state = @FORM07_CAT_ANCHOR }
			owns_state = @FORM07_CAT_ANCHOR
			controls_state = @FORM07_CAT_ANCHOR
		}
		AND = {
			original_tag = NAV
			check_variable = { independence_wave_anchor_state = @FORM07_NAV_ANCHOR }
			owns_state = @FORM07_NAV_ANCHOR
			controls_state = @FORM07_NAV_ANCHOR
		}
		AND = {
			original_tag = GLC
			check_variable = { independence_wave_anchor_state = @FORM07_GLC_ANCHOR }
			owns_state = @FORM07_GLC_ANCHOR
			controls_state = @FORM07_GLC_ANCHOR
		}
	}
}

# Country scope. FORM-07 admits only the two researched civic routes; the
# traditional, emergency, patron-client, and radical route families remain
# outside this adapter.
has_independence_wave_form07_compatible_route = {
	OR = {
		has_independence_wave_constitutional_route = yes
		has_independence_wave_popular_council_route = yes
	}
}

# Candidate-country scope with ROOT as the carrier. Route compatibility is a
# bilateral gate for invitations and integration, not a culture or geography
# inference.
has_independence_wave_form07_route_compatible_with_root = {
	has_independence_wave_form07_compatible_route = yes
	ROOT = { has_independence_wave_form07_compatible_route = yes }
}

# Country scope. This is the exact carrier proof used by the generic founding
# invitation and proposal validators.
is_independence_wave_form07_registered_carrier = {
	is_independence_wave_form07_member_candidate = yes
	has_independence_wave_form07_exact_member_anchor = yes
	independence_wave_formable_profile_matches_selected_family = yes
	has_country_flag = independence_wave_formable_profile_loaded
	is_independence_wave_regional_power = yes
	NOT = { has_country_flag = independence_wave_formable_active }
	has_independence_wave_form07_compatible_route = yes
}

has_independence_wave_form07_exact_carrier_anchor = {
	is_independence_wave_form07_registered_carrier = yes
	NOT = { has_global_flag = independence_wave_form07_identity_in_use }
}

# Country scope. The family registry mask is the source of method support;
# this trigger narrows it to FORM-07's negotiated and revolutionary methods.
has_independence_wave_form07_method_compatibility = {
	check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.iberian_federation }
	OR = {
		AND = {
			check_variable = { independence_wave_formable_method = constant:independence_wave_formable_method.negotiated_federation }
			has_country_flag = independence_wave_formable_supports_negotiated_method
		}
		AND = {
			check_variable = { independence_wave_formable_method = constant:independence_wave_formable_method.revolutionary_union }
			has_country_flag = independence_wave_formable_supports_revolutionary_method
		}
	}
}

# Country scope. Revolutionary union is voluntary-only; negotiated federation
# may use voluntary membership or a unanimous compact.
has_independence_wave_form07_consent_compatibility = {
	check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.iberian_federation }
	OR = {
		AND = {
			check_variable = { independence_wave_formable_method = constant:independence_wave_formable_method.negotiated_federation }
			OR = {
				check_variable = { independence_wave_formable_consent_rule = constant:independence_wave_formable_consent_rule.voluntary_membership }
				check_variable = { independence_wave_formable_consent_rule = constant:independence_wave_formable_consent_rule.unanimous_compact }
			}
		}
		AND = {
			check_variable = { independence_wave_formable_method = constant:independence_wave_formable_method.revolutionary_union }
			check_variable = { independence_wave_formable_consent_rule = constant:independence_wave_formable_consent_rule.voluntary_membership }
		}
	}
}

# Candidate-country scope with ROOT as the proposing carrier. The frozen
# invitation snapshot is the only accepted consent proof for a FORM-07 member.
has_independence_wave_form07_declared_consent = {
	is_independence_wave_form07_member_candidate = yes
	has_independence_wave_form07_exact_member_anchor = yes
	has_frozen_accepted_independence_wave_formable_invitation_from_root = yes
	check_variable = { independence_wave_formable_invitation_family = constant:independence_wave_formable_family.iberian_federation }
}

# Candidate-country scope with ROOT as the carrier. This mirrors the shared
# Event 006 diplomatic connection contract and prevents a disconnected anchor
# from being staged as an integrated or autonomous member.
has_independence_wave_form07_connection_to_root = {
	exists = yes
	OR = {
		tag = ROOT
		is_neighbor_of = ROOT
		is_in_faction_with = ROOT
		has_non_aggression_pact_with = ROOT
		has_military_access_to = ROOT
		gives_military_access_to = ROOT
		has_guaranteed = ROOT
		is_guaranteed_by = ROOT
	}
	NOT = { has_war_with = ROOT }
}

# Country scope. CAT, NAV, and GLC are the reviewed installed-map corridor.
# Requiring all three exact anchors keeps the family row fail-closed while NAV
# and GLC package adapters remain absent; future package additions must extend
# this explicit corridor rather than weakening it to a region scan.
has_independence_wave_form07_corridor_proof = {
	CAT = {
		is_independence_wave_form07_member_candidate = yes
		has_independence_wave_form07_exact_member_anchor = yes
	}
	NAV = {
		is_independence_wave_form07_member_candidate = yes
		has_independence_wave_form07_exact_member_anchor = yes
	}
	GLC = {
		is_independence_wave_form07_member_candidate = yes
		has_independence_wave_form07_exact_member_anchor = yes
	}
}

# Family identity contract. These three flags are intentionally not set by
# this tranche: no source-approved X tag or complete FORM-07 flag triplet is
# present. The generic readiness gate therefore remains false.
has_independence_wave_form07_identity_runtime_contract = {
	has_country_flag = independence_wave_form07_identity_attested
	has_country_flag = independence_wave_form07_x_tag_reserved
	has_country_flag = independence_wave_form07_flag_package_ready
}

can_independence_wave_form07_register_readiness = {
	has_country_flag = independence_wave_formable_family_selected
	check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.iberian_federation }
	has_country_flag = independence_wave_formable_profile_loaded
	has_independence_wave_form07_identity_runtime_contract = yes
}

	has_country_flag = independence_wave_formable_flag_package_ready
	has_country_flag = independence_wave_formable_identity_adapter_ready
	has_country_flag = independence_wave_formable_integration_adapter_ready
	has_country_flag = independence_wave_formable_member_policy_audited
	has_country_flag = independence_wave_form07_readiness_attested
	has_independence_wave_form07_exact_carrier_anchor = yes
	has_independence_wave_form07_method_compatibility = yes
	has_independence_wave_form07_consent_compatibility = yes
	has_independence_wave_form07_corridor_proof = yes
