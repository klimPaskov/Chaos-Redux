# ============================================================================
# CHAOS REDUX - FORM-16 TRANSCAUCASIAN FEDERATION TRIGGERS
# ============================================================================

has_independence_wave_form16_runtime_family = {
	check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.transcaucasian_federation }
}

# Candidate-country scope with ROOT as the inviting carrier. FORM-16 AI
# consent is never inferred from regional proximity alone: the member must
# have a controlled-border or an explicit diplomatic/access connection to the
# carrier and must be at peace with it.
has_independence_wave_form16_connection_to_root = {
	exists = yes
	NOT = { has_war_with = ROOT }
	OR = {
		tag = ROOT
		is_neighbor_of = ROOT
		is_in_faction_with = ROOT
		has_non_aggression_pact_with = ROOT
		has_military_access_to = ROOT
		gives_military_access_to = ROOT
		has_guaranteed = ROOT
		is_guaranteed_by = ROOT
	}
}

# Carrier-country scope. The receipt is recorded before formation and is then
# reused after the member countries have been absorbed, when their tags are no
# longer addressable as independent scopes.
has_independence_wave_form16_member_connection_proof = {
	ARM = { has_independence_wave_form16_connection_to_root = yes }
	GEO = { has_independence_wave_form16_connection_to_root = yes }
	AZR = { has_independence_wave_form16_connection_to_root = yes }
}
# Family-specific readiness, discovery, member-policy, and commit gates for
# the late-binding Transcaucasian federation. The generic formable registry
# supplies the congress transaction; these predicates keep the member set,
# identity, anchors, generation, and rollback receipt exact.
# ============================================================================

is_independence_wave_form16_carrier = {
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_formable_family_selected
	check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.transcaucasian_federation }
	has_country_flag = independence_wave_formable_profile_loaded
	has_independence_wave_transcaucasus_focus_contract = yes
	has_independence_wave_form16_all_members_live = yes
	has_independence_wave_form16_member_peace = yes
	has_independence_wave_form16_identity_clearance = yes
	has_independence_wave_form16_route_compatibility = yes
	has_independence_wave_form16_arbitration_receipt = yes
	NOT = { has_country_flag = independence_wave_formable_active }
	NOT = { has_country_flag = independence_wave_formable_founding_proposal_active }
}

has_independence_wave_form16_member_policy = {
	check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.transcaucasian_federation }
	OR = {
		check_variable = { independence_wave_formable_method = constant:independence_wave_formable_method.negotiated_federation }
		check_variable = { independence_wave_formable_method = constant:independence_wave_formable_method.league_transformation }
	}
	check_variable = { independence_wave_formable_consent_rule = constant:independence_wave_formable_consent_rule.voluntary_membership }
	ARM = { has_independence_wave_form16_member_candidate = yes }
	GEO = { has_independence_wave_form16_member_candidate = yes }
	AZR = { has_independence_wave_form16_member_candidate = yes }
}

has_independence_wave_form16_discovery_gate = {
	is_independence_wave_form16_carrier = yes
	has_country_flag = independence_wave_unlock_formable_discovery
	has_country_flag = independence_wave_formable_family_registered
	has_country_flag = independence_wave_formable_profile_loaded
	has_independence_wave_form16_member_policy = yes
}

has_independence_wave_form16_current_receipts = {

	has_variable = independence_wave_formable_proposal_sequence
	has_variable = independence_wave_formable_profile_family

has_independence_wave_form16_commit_readiness = {

has_independence_wave_form16_runtime_commit_prevalidation = {
	independence_wave_formable_profile_matches_selected_family = yes
	check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.transcaucasian_federation }
	has_independence_wave_form16_all_members_live = yes
	has_independence_wave_form16_member_peace = yes
	OR = {
		has_independence_wave_form16_identity_clearance = yes
		has_country_flag = independence_wave_form16_identity_committed
	}
	has_independence_wave_form16_route_compatibility = yes
	has_independence_wave_form16_arbitration_receipt = yes
	has_country_flag = independence_wave_formable_mutation_prevalidated
	has_country_flag = independence_wave_formable_territory_adapter_ready
	has_country_flag = independence_wave_formable_member_policy_audited
	NOT = { has_country_flag = independence_wave_formable_active }
}

has_independence_wave_form16_runtime_commit_proof = {
	has_independence_wave_form16_runtime_commit_prevalidation = yes
	has_independence_wave_form16_commit_readiness = yes
	has_country_flag = independence_wave_formable_transaction_ready
	check_variable = { independence_wave_formable_transaction_state = constant:independence_wave_formable_transaction.formation_ready }
	can_independence_wave_formable_pass_congress_vote = yes
}

has_independence_wave_form16_ai_gate = {
	OR = {
		AND = {
			is_independence_wave_form16_carrier = yes
			has_independence_wave_form16_member_connection_proof = yes
			has_independence_wave_form16_member_peace = yes
			has_independence_wave_form16_route_compatibility = yes
			has_independence_wave_form16_arbitration_receipt = yes
		}
		AND = {
			has_country_flag = independence_wave_formable_active
			has_independence_wave_form16_runtime_family = yes
			has_country_flag = independence_wave_form16_postformation_initialized
			has_country_flag = independence_wave_form16_ai_connection_receipt
			check_variable = { var = independence_wave_form16_ai_connection_generation value = independence_wave_generation_id compare = equals }
			has_independence_wave_form16_route_compatibility = yes
			has_independence_wave_form16_arbitration_receipt = yes
		}
	}
}
