cannibalism_wendigo_can_pay_terminal_hunt_launch_cost = {
	cannibalism_wendigo_command_is_open = yes
	has_country_flag = cannibalism_wendigo_terminal_hunt_open
	has_country_flag = cannibalism_wendigo_countdown_active
	NOT = { has_country_flag = cannibalism_wendigo_transformation_broken }
	NOT = { has_country_flag = cannibalism_wendigo_terminal_hunt_active }
	NOT = { has_country_flag = cannibalism_wendigo_terminal_hunt_launch_cooldown }
	check_variable = {
		var = cannibalism_larder
		value = constant:cannibalism_wendigo_terminal_hunt.launch_larder
		compare = greater_than_or_equals
	}
	command_power > constant:cannibalism_wendigo_terminal_hunt.launch_command_gate
	NOT = { has_equipment = { infantry_equipment < constant:cannibalism_wendigo_terminal_hunt.launch_infantry } }
	NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_wendigo_terminal_hunt.launch_support } }
}

cannibalism_wendigo_ai_can_pay_terminal_hunt_launch_cost = {
	cannibalism_wendigo_can_pay_terminal_hunt_launch_cost = yes
	set_temp_variable = { cannibalism_wendigo_ai_larder_cost = constant:cannibalism_wendigo_terminal_hunt.launch_larder }
	cannibalism_wendigo_ai_preserves_countdown_larder_after_cost = yes
}

		cannibalism_target_country_is_ordinary_human_target = yes
		has_country_flag = cannibalism_wendigo_terminal_hunt_defender

	}
}

cannibalism_wendigo_can_pay_terminal_hunt_press_cost = {
	cannibalism_wendigo_command_is_open = yes
	cannibalism_wendigo_has_valid_active_terminal_hunt_target = yes
	NOT = { has_country_flag = cannibalism_wendigo_terminal_hunt_press_cooldown }
	check_variable = {
		var = cannibalism_larder
		value = constant:cannibalism_wendigo_terminal_hunt.press_larder
		compare = greater_than_or_equals
	}
	command_power > constant:cannibalism_wendigo_terminal_hunt.press_command_gate
	NOT = { has_equipment = { infantry_equipment < constant:cannibalism_wendigo_terminal_hunt.press_infantry } }
	NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_wendigo_terminal_hunt.press_support } }
	NOT = { has_fuel < constant:cannibalism_wendigo_terminal_hunt.press_fuel }
}

cannibalism_wendigo_ai_should_press_terminal_hunt = {
	cannibalism_wendigo_can_pay_terminal_hunt_press_cost = yes
	set_temp_variable = { cannibalism_wendigo_ai_larder_cost = constant:cannibalism_wendigo_terminal_hunt.press_larder }
	cannibalism_wendigo_ai_preserves_countdown_larder_after_cost = yes
	set_temp_variable = { cannibalism_wendigo_terminal_hunt_ai_pressure_gate = constant:cannibalism_wendigo_terminal_hunt.starting_pressure }
	add_to_temp_variable = { cannibalism_wendigo_terminal_hunt_ai_pressure_gate = constant:cannibalism_wendigo_terminal_hunt.press_pressure }
	check_variable = {
		var = global.cannibalism_wendigo_active_anchor_count
		value = constant:cannibalism_wendigo_anchor.minimum_for_countdown
		compare = greater_than_or_equals
	}
	OR = {
		check_variable = {
			var = cannibalism_wendigo_terminal_hunt_pressure
			value = cannibalism_wendigo_terminal_hunt_ai_pressure_gate
			compare = greater_than_or_equals
		}
		event_target:cannibalism_wendigo_terminal_hunt_target = {
			capital_scope = { is_controlled_by = ROOT }
		}
	}
}

cannibalism_wendigo_terminal_hunt_defender_is_open = {
	has_country_flag = cannibalism_wendigo_terminal_hunt_defender
	has_event_target = cannibalism_wendigo_terminal_hunt_target
	tag = event_target:cannibalism_wendigo_terminal_hunt_target
	has_event_target = cannibalism_wendigo_merge_host
	event_target:cannibalism_wendigo_merge_host = {
		is_cannibalism_wendigo_hannibal_country = yes
		has_country_flag = cannibalism_wendigo_terminal_hunt_active
		NOT = { has_country_flag = cannibalism_wendigo_transformation_locked }
		NOT = { has_country_flag = cannibalism_wendigo_transformation_broken }
		has_war_with = ROOT
	}
	NOT = { has_capitulated = yes }
}

cannibalism_can_pay_wendigo_terminal_hunt_break_cost = {
	cannibalism_wendigo_terminal_hunt_defender_is_open = yes
	capital_scope = { is_controlled_by = ROOT }
	has_manpower > constant:cannibalism_wendigo_terminal_hunt.defender_manpower_gate
	command_power > constant:cannibalism_wendigo_terminal_hunt.defender_command_gate
	NOT = { has_equipment = { infantry_equipment < constant:cannibalism_wendigo_terminal_hunt.defender_infantry } }
	NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_wendigo_terminal_hunt.defender_support } }
	event_target:cannibalism_wendigo_merge_host = {
		check_variable = {
			var = cannibalism_wendigo_terminal_hunt_counterpressure
			value = constant:cannibalism_wendigo_terminal_hunt.counterpressure_goal
			compare = less_than
		}
	}
}

cannibalism_wendigo_terminal_hunt_has_succeeded = {
	has_country_flag = cannibalism_wendigo_terminal_hunt_active
	has_event_target = cannibalism_wendigo_terminal_hunt_target
	event_target:cannibalism_wendigo_terminal_hunt_target = {
		OR = {
			has_capitulated = yes
			AND = {
				capital_scope = { is_controlled_by = ROOT }
				ROOT = {
					check_variable = {
						var = cannibalism_wendigo_terminal_hunt_pressure
						value = constant:cannibalism_wendigo_terminal_hunt.pressure_goal
						compare = greater_than_or_equals
					}
				}
			}
		}
	}
}

cannibalism_wendigo_terminal_hunt_has_failed = {
	has_country_flag = cannibalism_wendigo_terminal_hunt_active
	NOT = { cannibalism_wendigo_terminal_hunt_has_succeeded = yes }
	OR = {
		has_country_flag = cannibalism_wendigo_transformation_broken
		check_variable = { global.cannibalism_wendigo_active_anchor_count < 1 }
		check_variable = {
			var = cannibalism_wendigo_terminal_hunt_counterpressure
			value = constant:cannibalism_wendigo_terminal_hunt.counterpressure_goal
			compare = greater_than_or_equals
		}
		NOT = { exists = yes }
		NOT = { has_event_target = cannibalism_wendigo_terminal_hunt_target }
		event_target:cannibalism_wendigo_terminal_hunt_target = { NOT = { exists = yes } }
		event_target:cannibalism_wendigo_terminal_hunt_target = { NOT = { has_war_with = ROOT } }
		event_target:cannibalism_wendigo_terminal_hunt_target = { NOT = { cannibalism_target_country_is_ordinary_human_target = yes } }
	}
}
