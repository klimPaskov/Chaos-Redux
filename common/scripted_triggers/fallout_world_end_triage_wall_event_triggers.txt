# ============================================================================
# CHAOS REDUX - FALLOUT TRIAGE WALL TRIGGERS
# ============================================================================
# The Triage Wall is a dormant first-season medicine chain. It may only use a
# current state identity and a produced Air Winter snapshot. Missing provenance,
# ownership, or shelter capacity refuses the candidate instead of falling back
# to a generic country event.
# ============================================================================

fallout_event_pilot_triage_wall_state_is_current = {
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_variable = fallout_pretransition_air_winter_shelter_capacity
	check_variable = { fallout_pretransition_air_winter_shelter_capacity > constant:fallout_event_175_threshold.state_shelter_minimum }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	owner = {
		check_variable = { fallout_survival_current_entries^3 value = constant:fallout_event_175_cost.minimum_medicine compare = greater_than_or_equals }
		check_variable = { fallout_survival_current_entries^3 < constant:fallout_event_175_threshold.medicine_pressure }
	}
	NOT = { has_state_flag = fallout_event_175_triage_registry_committed }
}

fallout_event_175_opening_triage_target_is_current = {
	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.state }
	check_variable = { fallout_event_dispatch_issued_target > constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_175_target_to_check = fallout_event_dispatch_issued_target }
	set_temp_variable = { fallout_event_175_target_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_175_target_to_check = {
		set_temp_variable = { fallout_event_175_target_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		owner = { tag = ROOT }
		fallout_event_pilot_triage_wall_state_is_current = yes
	}
	check_variable = { fallout_event_175_target_context_loaded = constant:fallout_event_runtime.one }
}

fallout_event_175_triage_registry_is_current = {
	has_variable = fallout_event_175_triage_state_id
	has_variable = fallout_event_175_triage_registry_generation
	has_variable = fallout_event_175_triage_registry_owner
	check_variable = { fallout_event_175_triage_state_id > constant:fallout_event_runtime.zero }
	check_variable = { fallout_event_175_triage_registry_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_event_175_triage_registry_owner value = ROOT compare = equals }
	set_temp_variable = { fallout_event_175_triage_registry_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_175_triage_state_id = {
		set_temp_variable = { fallout_event_175_triage_registry_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		has_state_flag = fallout_event_175_triage_registry_committed
		fallout_survival_state_identity_row_is_current = yes
		fallout_survival_state_resource_row_is_durable = yes
		check_variable = { fallout_survival_state_owner value = ROOT compare = equals }
		has_variable = fallout_event_175_triage_shelter
		check_variable = { fallout_event_175_triage_shelter > constant:fallout_event_175_threshold.state_shelter_minimum }
		check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
		check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	}
	check_variable = { fallout_event_175_triage_registry_context_loaded = constant:fallout_event_runtime.one }
}

	fallout_event_country_can_receive_ordinary_event = yes

}
