# ============================================================================
# CHAOS REDUX - EVENT 014 WENDIGO DECISION TRIGGERS
# ============================================================================
# Defines click-time resource, state, capacity, and counterwar checks for the
# revealed Wendigo route. Counterwar availability iterates only the registered
# transformation-anchor array; it never scans every state on a recurring tick.
# ============================================================================

cannibalism_wendigo_command_is_open = {
	has_global_flag = cannibalism_reveal_complete
	has_global_flag = cannibalism_wendigo_route_active
	NOT = { has_global_flag = world_end }
	is_cannibalism_wendigo_hannibal_country = yes
	has_country_flag = cannibalism_wendigo_transformation_prelock
	NOT = { has_country_flag = cannibalism_wendigo_transformation_locked }
	NOT = { has_capitulated = yes }
	has_event_target = cannibalism_wendigo_merge_host
	tag = event_target:cannibalism_wendigo_merge_host
}

cannibalism_wendigo_can_pay_designate_anchor_cost = {
	cannibalism_wendigo_command_is_open = yes
	NOT = { has_country_flag = cannibalism_wendigo_transformation_broken }
	check_variable = {
		var = cannibalism_larder
		value = constant:cannibalism_wendigo_decision.designate_anchor_larder
		compare = greater_than_or_equals
	}
	command_power > constant:cannibalism_wendigo_decision.designate_anchor_command_gate
	check_variable = {
		var = global.cannibalism_wendigo_active_anchor_count
		value = constant:cannibalism_wendigo_anchor.maximum
		compare = less_than
	}
}

cannibalism_state_can_become_wendigo_anchor = {
	OWNER = { tag = ROOT }
	CONTROLLER = { tag = ROOT }
	NOT = { has_state_flag = cannibalism_wendigo_transformation_anchor }
	NOT = { has_state_flag = cannibalism_wendigo_anchor_population_committed }
	NOT = { has_state_flag = cannibalism_wendigo_anchor_broken }
	NOT = { has_state_flag = cannibalism_recovery_active }
	NOT = { has_state_flag = cannibalism_consumption_cooldown }
	NOT = { is_in_array = { array = global.cannibalism_transformation_anchor_states value = THIS } }
	cannibalism_can_consume_current_state = yes
	check_variable = {
		var = state_population_k
		value = constant:cannibalism_wendigo_anchor.minimum_creation_population_k
		compare = greater_than_or_equals
	}
}

cannibalism_state_can_freeze_wendigo_supply_corridor = {
	OWNER = { tag = ROOT }
	CONTROLLER = { tag = ROOT }
	NOT = { has_state_flag = cannibalism_wendigo_frozen_supply_corridor_active }
	OR = {
		supply_node > 0
		rail_way > 0
		naval_base > 0
	}
	any_neighbor_state = {
		CONTROLLER = {
			exists = yes
			has_war_with = ROOT
		}
	}
}

cannibalism_current_wendigo_anchor_is_reachable_by_root = {
	cannibalism_current_wendigo_anchor_is_live = yes
	OR = {
		CONTROLLER = { has_war_with = ROOT }
		any_neighbor_state = { CONTROLLER = { tag = ROOT } }
	}
}

cannibalism_wendigo_counterwar_is_open = {
	has_global_flag = cannibalism_reveal_complete
	has_global_flag = cannibalism_wendigo_route_active
	NOT = { has_global_flag = world_end }
	NOT = { is_cannibalism_wendigo_hannibal_country = yes }
	NOT = { has_capitulated = yes }
	has_event_target = cannibalism_wendigo_merge_host
	event_target:cannibalism_wendigo_merge_host = {
		is_cannibalism_wendigo_hannibal_country = yes
		has_country_flag = cannibalism_wendigo_transformation_prelock
		NOT = { has_country_flag = cannibalism_wendigo_transformation_locked }
		NOT = { has_country_flag = cannibalism_wendigo_transformation_broken }
	}
	any_of_scopes = {
		array = global.cannibalism_transformation_anchor_states
		cannibalism_current_wendigo_anchor_is_reachable_by_root = yes
	}
}

cannibalism_can_identify_any_wendigo_anchor = {
	cannibalism_wendigo_counterwar_is_open = yes
	NOT = { has_country_flag = cannibalism_wendigo_identification_cooldown }
	has_political_power > constant:cannibalism_wendigo_decision.identify_political_power_gate
	any_of_scopes = {
		array = global.cannibalism_transformation_anchor_states
		cannibalism_current_wendigo_anchor_is_reachable_by_root = yes
		NOT = { has_state_flag = cannibalism_wendigo_anchor_identified }
	}
}
