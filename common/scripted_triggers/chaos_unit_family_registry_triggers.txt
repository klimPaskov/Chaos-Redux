# ============================================================================
# CHAOS REDUX - OPT-IN CHAOS UNIT FAMILY REGISTRY TRIGGERS
# ============================================================================
# The registry is an aligned global table. Providers append one row through
# chaos_unit_family_register_current_provider and consumers dispatch through
# the stored provider id. A malformed row is an invariant failure, never a
# reason to substitute another family.
# ============================================================================

chaos_unit_family_registry_is_aligned = {
	NOT = { has_global_flag = chaos_unit_family_registry_invariant_failure }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_provider_id_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_source_event_id_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_availability_mode_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_event19_lot_policy_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_derivative_profile_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_sustainment_profile_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_containment_profile_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_ai_class_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_visual_profile_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_cleanup_profile_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_parent_isolation_profile_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_spawn_weight_entries^num compare = equals }
	check_variable = { var = global.chaos_unit_family_id_entries^num value = global.chaos_unit_family_contract_version_entries^num compare = equals }
}

chaos_unit_family_current_row_is_enabled = {
	check_variable = {
		var = global.chaos_unit_family_availability_mode_entries^chaos_unit_family_registry_index
		value = constant:chaos_unit_family_availability.disabled
		compare = greater_than
	}
}

chaos_unit_family_current_row_allows_event19_family_lot = {
	OR = {
		check_variable = {
			var = global.chaos_unit_family_event19_lot_policy_entries^chaos_unit_family_registry_index
			value = constant:chaos_unit_family_event19_lot_policy.family_only
			compare = equals
		}
		check_variable = {
			var = global.chaos_unit_family_event19_lot_policy_entries^chaos_unit_family_registry_index
			value = constant:chaos_unit_family_event19_lot_policy.both
			compare = equals
		}

# Registration-time visual ownership. Profiles 1-3 are reserved for the three
# initial provider packages and are valid only with their exact family/provider
# identities. A future provider opts into the neutral army/host presentation
# explicitly with profile 999; an unknown positive number is never a display
# substitute.
chaos_unit_family_registration_has_supported_event19_visual_contract = {
	OR = {
		AND = {
			check_variable = { chaos_unit_family_register_id = constant:event19_zombie_family.family_id }
			check_variable = { chaos_unit_family_register_provider_id = constant:event19_zombie_family.provider_id }
			check_variable = { chaos_unit_family_register_visual_profile = constant:event19_zombie_family.visual_profile }
		}
		AND = {
			check_variable = { chaos_unit_family_register_id = constant:event19_ghost_family.family_id }
			check_variable = { chaos_unit_family_register_provider_id = constant:event19_ghost_family.provider_id }
			check_variable = { chaos_unit_family_register_visual_profile = constant:event19_ghost_family.visual_profile }
		}
		AND = {
			check_variable = { chaos_unit_family_register_id = constant:event19_golem_family.family_id }
			check_variable = { chaos_unit_family_register_provider_id = constant:event19_golem_family.provider_id }
			check_variable = { chaos_unit_family_register_visual_profile = constant:event19_golem_family.visual_profile }
		}
		AND = {
			check_variable = { chaos_unit_family_register_visual_profile = constant:chaos_unit_family_visual_profile.provider_neutral_army }
			NOT = {
				OR = {
					check_variable = { chaos_unit_family_register_id = constant:event19_zombie_family.family_id }
					check_variable = { chaos_unit_family_register_id = constant:event19_ghost_family.family_id }
					check_variable = { chaos_unit_family_register_id = constant:event19_golem_family.family_id }
					check_variable = { chaos_unit_family_register_provider_id = constant:event19_zombie_family.provider_id }
					check_variable = { chaos_unit_family_register_provider_id = constant:event19_ghost_family.provider_id }
					check_variable = { chaos_unit_family_register_provider_id = constant:event19_golem_family.provider_id }
				}
			}
		}
	}
}
