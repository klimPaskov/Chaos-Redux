# COUNTRY. The rival threat that makes British aid attractive: a war or a
# major power on the border.
lawrence_target_has_rival_threat = {
	OR = {
		has_war = yes
		any_neighbor_country = {
			is_major = yes
			NOT = { tag = ENG }
			NOT = { is_subject_of = ENG }
			NOT = { is_in_faction_with = ENG }
		}
	}
}

# COUNTRY, scoped to a candidate sponsor with ROOT as the target. A real
# patron with a reason and a route: an enemy of Britain with access, or a
# major power of the same ideology on the border or in the same faction.
lawrence_is_alternative_sponsor = {
	is_major = yes
	exists = yes
	has_capitulated = no
	NOT = { tag = ENG }
	NOT = { is_subject_of = ENG }
	NOT = { is_in_faction_with = ENG }
	is_actual_nonhuman_country = no
	OR = {
		has_war_with = ENG
		AND = {
			ideology_group_matches_target = yes
			OR = {
				is_neighbor_of = ROOT
				is_in_faction_with = ROOT
			}
		}
	}
	OR = {
		has_military_access_to = ROOT
		ROOT = { has_military_access_to = PREV }
		is_neighbor_of = ROOT
		is_in_faction_with = ROOT
	}
}

# COUNTRY, the target. At least one alternative sponsor exists.
lawrence_target_has_alternative_sponsor = {
	any_other_country = { lawrence_is_alternative_sponsor = yes }
}

# COUNTRY. Same broad ideology group as the target (ROOT).
ideology_group_matches_target = {
	OR = {
		AND = {
			has_government = democratic
			ROOT = { has_government = democratic }
		}
		AND = {
			has_government = fascism
			ROOT = { has_government = fascism }
		}
		AND = {
			has_government = communism
			ROOT = { has_government = communism }
		}
	}
}
