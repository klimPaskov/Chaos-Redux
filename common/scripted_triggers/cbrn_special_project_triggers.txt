# Completion and iteration rewards must not reuse research-admission predicates:
# the Mengele admission triggers reject a project as soon as native completion
# is recorded. These country-scope predicates use durable story-stage evidence.
cbrn_project_biomedical_uses_event016_story = {
	OR = {
		AND = {
			brilliant_scientist_is_current_host = yes
			check_variable = { var = brilliant_scientist_project_stage_entries^5 value = constant:brilliant_scientist_project_stage.theory compare = greater_than_or_equals }
		}
		AND = {
			brilliant_scientist_mengele_project_stage_provider_is_valid = yes
			has_country_flag = mengele_event016_biomedical_theory_completed
		}
	}
}
