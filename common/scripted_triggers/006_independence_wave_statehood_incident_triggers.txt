# ============================================================================
# CHAOS REDUX - EVENT 006 STATEHOOD INCIDENT TRIGGERS
# ============================================================================
# Eligibility of the recurring statehood incidents. Every trigger reads the
# checked country's own live state: founding phase, the five Statehood values,
# the former-host relation, League membership, formable progress and the
# evolutions that are active and enabled. Array checks walk only the bounded
# Event 006 active-country and League-member registries.
#
# The incident triggers are evaluated by the pulse in chaosx.nr6.150 and by
# the incident events themselves. In both, ROOT is the incident country, so the
# registry checks compare members with ROOT.
# ============================================================================

# ---------------------------------------------------------------------------
# Pulse and open-incident gates
# ---------------------------------------------------------------------------

is_independence_wave_statehood_incident_generation_current = {
	has_variable = independence_wave_statehood_incident_generation
	has_variable = independence_wave_generation_id
	check_variable = { var = independence_wave_statehood_incident_generation value = independence_wave_generation_id compare = equals }
}

# COUNTRY scope. A pulse event runs only for the live generation that queued it,
# and only on or after the stored due date, so a pulse left over from an
# earlier release of the same tag does nothing.
can_run_independence_wave_statehood_incident_pulse = {
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_statehood_incident_pulse_queued
	is_independence_wave_statehood_incident_generation_current = yes
	has_variable = independence_wave_statehood_incident_due_date
	check_variable = { var = global.date value = independence_wave_statehood_incident_due_date compare = greater_than_or_equals }
}

# COUNTRY scope. Each incident event adds its own id check after this gate.
can_fire_independence_wave_statehood_incident = {
	is_independence_wave_active_country = yes
	is_independence_wave_statehood_incident_generation_current = yes
	has_variable = independence_wave_statehood_incident_open
}

has_independence_wave_statehood_league_member_at_war = {
	any_of_scopes = {
		array = global.independence_wave_league_member_country_entries
		exists = yes
		NOT = { tag = ROOT }
		has_war = yes
	}
}

has_independence_wave_statehood_league_member_invaded = {
	any_of_scopes = {
		array = global.independence_wave_league_member_country_entries
		exists = yes
		NOT = { tag = ROOT }
		has_independence_wave_living_former_host = yes
		var:independence_wave_former_host = { has_war_with = PREV }
	}
}

has_independence_wave_statehood_radical_league_voice = {
	OR = {
		check_variable = { global.independence_wave_league_route = constant:independence_wave_league_route.armed_liberation }
		check_variable = { global.independence_wave_league_route = constant:independence_wave_league_route.radical_revisionist }
		any_of_scopes = {
			array = global.independence_wave_league_member_country_entries
			exists = yes
			NOT = { tag = ROOT }
			has_independence_wave_radical_sovereignty_route = yes
		}
	}
}
