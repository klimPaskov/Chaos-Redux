has_independence_wave_severe_host_threat = {
	is_independence_wave_active_country = yes
	has_independence_wave_living_former_host = yes
	OR = {
		check_variable = { independence_wave_host_outcome = constant:independence_wave_host_outcome.reclamation_conflict }
		check_variable = {
			var = independence_wave_reconquest_fear
			value = constant:independence_wave_host_relation.conflict_gate
			compare = greater_than_or_equals
		}
		var:independence_wave_former_host = { has_war_with = ROOT }
	}
}

is_valid_independence_wave_depot_state = {
	is_controlled_by = ROOT
	is_owned_by = ROOT
	NOT = {
		AND = {
			has_state_flag = independence_wave_dm18_depot_secured_by@ROOT
			check_variable = {
				var = independence_wave_dm18_generation@ROOT
				value = ROOT.independence_wave_generation_id
				compare = equals
			}
		}
	}
	OR = {
		supply_node > 0
		arms_factory > 0
		infrastructure > 2
	}
	ROOT = {
		divisions_in_state = {
			state = PREV
			size > constant:independence_wave_decision_gate.minimum_divisions
		}
	}
}

is_valid_independence_wave_ambition_state = {
	owner = { exists = yes NOT = { tag = ROOT } }
	controller = { exists = yes }

		is_claimed_by = ROOT
		any_neighbor_state = { is_owned_by = ROOT }
	}
}
