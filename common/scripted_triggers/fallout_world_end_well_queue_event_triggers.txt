# ============================================================================
# CHAOS REDUX - FALLOUT WELL QUEUE TRIGGERS
# ============================================================================
# The Well Queue uses a real post-Fallout state row and an Air Winter water
# source. A missing target, owner, or generation rejects the chain.
# ============================================================================

fallout_event_pilot_well_queue_state_is_current = {
	fallout_survival_state_identity_row_is_current = yes
	fallout_survival_state_resource_row_is_durable = yes
	has_variable = air_winter_water_security
	check_variable = { air_winter_water_security < constant:fallout_event_153_threshold.state_water_success }
	check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	owner = {
		OR = {
			check_variable = { fallout_survival_current_entries^2 value = constant:fallout_event_153_cost.public_water compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^7 value = constant:fallout_event_153_cost.ration_filters compare = greater_than_or_equals }
			NOT = { command_power < constant:fallout_event_153_cost.guard_command_power_minimum }
		}
	}
	NOT = { has_state_flag = fallout_event_153_well_registry_committed }
}

fallout_event_153_opening_well_target_is_current = {
	check_variable = { fallout_event_dispatch_issued_target_type = constant:fallout_event_subject_type.state }
	check_variable = { fallout_event_dispatch_issued_target > constant:fallout_event_runtime.zero }
	set_temp_variable = { fallout_event_153_target_to_check = fallout_event_dispatch_issued_target }
	set_temp_variable = { fallout_event_153_target_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_153_target_to_check = {
		set_temp_variable = { fallout_event_153_target_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		owner = { tag = ROOT }
		fallout_event_pilot_well_queue_state_is_current = yes
	}
	check_variable = { fallout_event_153_target_context_loaded = constant:fallout_event_runtime.one }
}

fallout_event_153_well_registry_is_current = {
	has_variable = fallout_event_153_well_state_id
	has_variable = fallout_event_153_well_registry_generation
	has_variable = fallout_event_153_well_registry_owner
	check_variable = { fallout_event_153_well_state_id > constant:fallout_event_runtime.zero }
	check_variable = { fallout_event_153_well_registry_generation value = global.fallout_transition_generation compare = equals }
	check_variable = { fallout_event_153_well_registry_owner value = ROOT compare = equals }
	set_temp_variable = { fallout_event_153_registry_context_loaded = constant:fallout_event_runtime.zero }
	var:fallout_event_153_well_state_id = {
		set_temp_variable = { fallout_event_153_registry_context_loaded = constant:fallout_event_runtime.one }
		exists = yes
		has_state_flag = fallout_event_153_well_registry_committed
		fallout_survival_state_identity_row_is_current = yes
		fallout_survival_state_resource_row_is_durable = yes
		check_variable = { fallout_survival_state_owner value = ROOT compare = equals }
		has_variable = air_winter_water_security
		check_variable = { fallout_pretransition_snapshot_generation value = global.fallout_transition_generation compare = equals }
		check_variable = { fallout_pretransition_air_winter_source_kind = constant:air_winter_fallout_snapshot_source_kind.produced }
	}
	check_variable = { fallout_event_153_registry_context_loaded = constant:fallout_event_runtime.one }
}

fallout_event_153_opening_event_can_enter = {
	is_ai = no
	fallout_event_country_can_receive_ordinary_event = yes

	check_variable = { fallout_event_dispatch_issued_mode = constant:fallout_event_dispatch_mode.human_visible }
	fallout_event_153_opening_well_target_is_current = yes
}
