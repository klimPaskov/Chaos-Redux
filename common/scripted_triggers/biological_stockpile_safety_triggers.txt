# ============================================================================
# CHAOS REDUX - BIOLOGICAL STOCKPILE SAFETY SCRIPTED TRIGGERS
# ============================================================================
# Inputs: country-scoped pointer, actual stockpile snapshot, exact-state
#   condition snapshot, incident-kind input, and caller-owned runtime flags.
# Outputs: fail-closed validity, risk-band, agent, loss, incident-kind, and
#   highest-safety ordinary-accident immunity triggers.
# Side effects: none. These triggers never select an alternate state, debit
#   equipment, dispatch a raid, or enter the biological lifecycle.
# ============================================================================

# Country scope. The pointer is the only accepted arsenal state. The caller
# must set the designation flag and source after an exact UI selection or an
# exact special-project facility_state proof. No completed-project lookup is
# attempted here.
bio_stockpile_safety_arsenal_pointer_source_is_valid = {
	has_variable = bio_stockpile_safety_arsenal_pointer_source
	OR = {
		check_variable = {
			var = bio_stockpile_safety_arsenal_pointer_source
			value = constant:bio_stockpile_safety_pointer_source.selected_state
			compare = equals
		}
		check_variable = {
			var = bio_stockpile_safety_arsenal_pointer_source
			value = constant:bio_stockpile_safety_pointer_source.verified_special_project_facility
			compare = equals
		}
	}
}

# State scope, called from a state-target decision or exact special-project
# facility_state_effects. ROOT remains the selecting country.
bio_stockpile_safety_state_is_valid_designation_candidate = {
	exists = yes
	impassable = no
	is_owned_by = ROOT
	is_controlled_by = ROOT
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	biowarfare_facility > constant:bio_stockpile_safety_risk.facility_level_trigger_floor
	check_variable = {
		var = non_damaged_building_level@biowarfare_facility
		value = constant:bio_stockpile_safety_risk.facility_operational_minimum_level
		compare = greater_than_or_equals
	}
}

bio_stockpile_safety_arsenal_pointer_is_valid = {
	exists = yes
	has_country_flag = bio_stockpile_safety_arsenal_designated
	has_variable = bio_stockpile_safety_arsenal_state
	bio_stockpile_safety_arsenal_pointer_source_is_valid = yes
	var:bio_stockpile_safety_arsenal_state = {
		exists = yes
		impassable = no
		is_owned_by = PREV
		is_controlled_by = PREV
		bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
		biowarfare_facility > constant:bio_stockpile_safety_risk.facility_level_trigger_floor
	}
}

# Explicit sabotage is accepted only when the caller's private exact-state flag
# is set or the existing exact-state sabotage operation is active on this same
# arsenal.
bio_stockpile_safety_explicit_sabotage_is_active = {
	bio_stockpile_safety_arsenal_pointer_is_valid = yes
	OR = {
		var:bio_stockpile_safety_arsenal_state = {
			has_state_flag = bio_stockpile_safety_explicit_sabotage_active
		}
		AND = {
			has_country_flag = bio_sabotage_operation_in_progress
			has_variable = bio_sabotage_active_state
			var:bio_sabotage_active_state = {
				state = var:bio_stockpile_safety_arsenal_state
			}
		}
	}
}

# A caller sets bio_stockpile_safety_incident_kind before invoking this trigger.
# Only an ordinary accident can be blocked. Sabotage, bombing, captured
# facility release, and doomsday release explicitly return false here.
bio_stockpile_safety_highest_safety_ordinary_accident_immunity = {
	bio_stockpile_safety_incident_is_ordinary_accident = yes
	bio_stockpile_safety_arsenal_pointer_is_valid = yes
	bio_stockpile_safety_exact_condition_snapshot_is_valid = yes
	bio_stockpile_safety_handling_context_is_valid = yes
	has_tech = fail_safe_containment_facilities
	NOT = { bio_stockpile_safety_explicit_sabotage_is_active = yes }
	check_variable = {
		var = bio_stockpile_safety_recent_bombing_active
		value = constant:bio_lifecycle_math.zero
		compare = equals
	}
}
