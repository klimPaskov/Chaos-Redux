# AI-only gates. The regular decisions remain player-available at the exact
# material costs above. The AI may take the ledger and government actions
# required to settle a valid foundation, but stops after a foundation failure.
# Post-spend reserve gates still protect the package material floor. Ledger
# preference is deliberately package-local so Karelia and Crimea do not share
# or overwrite one another's regional priorities.
independence_wave_kc_ai_foundation_ready = {
	is_independence_wave_karelia_crimea_package = yes
	NOT = { has_country_flag = independence_wave_kc_foundation_failed }
	OR = {
		has_country_flag = independence_wave_kc_foundation_settled
		AND = {
			is_independence_wave_kar_package = yes
			has_country_flag = independence_wave_iw_033_setup_complete
		}
		AND = {
			is_independence_wave_cri_package = yes
			has_country_flag = independence_wave_iw_041_setup_complete
		}
	}
}
