# ============================================================================
# CHAOS REDUX - JAPAN-CHINA BIOLOGICAL CAMPAIGN SCRIPTED TRIGGERS
# ============================================================================
# Overview:
# - Keeps every helper private to Japan's historically specific China campaign.
# - Validates the explicit offensive program route, shared CBRN authorization,
#   exact Chinese enemy state, and occupation-linked native raid reach.
# - No helper infers a state, substitutes another target, or creates a fallback.
# ============================================================================

japan_bio_campaign_actor_cooldown_is_ready = {
	OR = {
		NOT = { has_variable = japan_bio_campaign_next_allowed_day }
		check_variable = {
			var = japan_bio_campaign_next_allowed_day
			value = global.num_days
			compare = less_than_or_equals
		}
	}
}

japan_bio_campaign_route_is_open = {
	exists = yes
	original_tag = JAP
	NOT = { has_capitulated = yes }
	has_war_with_chinese_country = yes
	has_country_flag = japan_pingfang_route_unlocked
	has_country_flag = japan_ishii_program_active
	has_country_flag = japan_ishii_authority_route
	NOT = { has_country_flag = japan_biowarfare_containment_route }
	NOT = { has_country_flag = japan_biowarfare_reform_route }
	NOT = { has_country_flag = japan_prisoner_experiments_shutdown }
	cbrn_country_has_program = yes
	cbrn_country_has_operational_readiness = yes
	cbrn_policy_allows_strategic_use = yes
	has_variable = cbrn_biological_security
	has_variable = cbrn_attribution_control
}

japan_bio_campaign_actor_can_operate = {
	japan_bio_campaign_route_is_open = yes
	japan_bio_campaign_actor_cooldown_is_ready = yes
}

	OR = {
		is_core_of = CHI
		is_core_of = PRC
		is_core_of = SHX
		is_core_of = GXC
		is_core_of = YUN
		is_core_of = XSM
		is_core_of = SIK
	}
}

		tag = CHI

		tag = SHX
		tag = GXC
		tag = YUN
		tag = XSM
		tag = SIK
	}
}

# State scope from native show_target, available, or launchable, where ROOT is
# the acting country. Do not use this in raid-instance outcome effects.
# The target must remain under a Chinese belligerent's control.
japan_bio_campaign_state_relationship_is_valid = {
	CONTROLLER = {
		exists = yes
		japan_bio_campaign_country_is_recognized_chinese_belligerent = yes
		NOT = { tag = ROOT }
		NOT = { is_subject_of = ROOT }
		ROOT = {
			has_war_with = PREV
			NOT = { is_in_faction_with = PREV }
			NOT = { is_subject_of = PREV }
		}
	}
}

# State scope from native pre-raid triggers, where ROOT is the acting country.
# A neighboring Japanese or Japanese-subject occupation zone supplies the link.
japan_bio_campaign_state_has_occupation_link = {
	any_neighbor_state = {
		CONTROLLER = {
			OR = {
				tag = ROOT
				is_subject_of = ROOT
			}
		}
	}
}

japan_bio_campaign_common_target_state = {

	japan_bio_campaign_state_relationship_is_valid = yes

japan_bio_campaign_anthrax_target_state = {
	japan_bio_campaign_common_target_state = yes
	NOT = { bio_lifecycle_state_has_anthrax_episode = yes }
	OR = {
		supply_node > constant:japan_bio_campaign_target.building_floor
		naval_base > constant:japan_bio_campaign_target.building_floor
		industrial_complex > constant:japan_bio_campaign_target.anthrax_industry_threshold
		arms_factory > constant:japan_bio_campaign_target.anthrax_industry_threshold
		infrastructure > constant:japan_bio_campaign_target.anthrax_infrastructure_threshold
		any_state_division = { always = yes }
	}
}

japan_bio_campaign_plague_target_state = {
	japan_bio_campaign_common_target_state = yes
	NOT = { bio_lifecycle_state_has_plague_episode = yes }
	OR = {
		is_capital = yes
		has_state_category = city

		state_population_k > constant:japan_bio_campaign_target.plague_population_threshold_k
		naval_base > constant:japan_bio_campaign_target.building_floor
	}
}

# Actor-country hidden-event scope. The callback saved the exact target and
# belligerents as chain-local event targets before entering this scope.
japan_bio_campaign_native_context_is_valid = {
	has_event_target = japan_bio_campaign_actor_target
	has_event_target = japan_bio_campaign_victim_target
	has_event_target = japan_bio_campaign_state_target
	event_target:japan_bio_campaign_actor_target = {
		japan_bio_campaign_route_is_open = yes
		has_war_with = event_target:japan_bio_campaign_victim_target
		cbrn_policy_allows_strategic_use_against_target_id = yes
		OR = {
			AND = {
				check_variable = { var = japan_bio_campaign_agent value = constant:bio_lifecycle_agent.anthrax compare = equals }
				is_special_project_completed = sp:anthrax_bomb
			}
			AND = {
				check_variable = { var = japan_bio_campaign_agent value = constant:bio_lifecycle_agent.plague compare = equals }
				is_special_project_completed = sp:plague_bomb
			}
		}
	}
	event_target:japan_bio_campaign_victim_target = {
		exists = yes
		japan_bio_campaign_country_is_recognized_chinese_belligerent = yes
	}
	event_target:japan_bio_campaign_state_target = {
		is_controlled_by = event_target:japan_bio_campaign_victim_target
		bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
		impassable = no
		japan_bio_campaign_state_is_chinese_homeland = yes
		japan_bio_campaign_state_cooldown_is_ready = yes
		state_population_k > constant:japan_bio_campaign_target.minimum_population_k
		any_neighbor_state = {
			CONTROLLER = {
				OR = {
					tag = event_target:japan_bio_campaign_actor_target
					is_subject_of = event_target:japan_bio_campaign_actor_target
				}
			}
		}
		OR = {
			AND = {
				check_variable = { var = japan_bio_campaign_agent value = constant:bio_lifecycle_agent.anthrax compare = equals }
				NOT = { bio_lifecycle_state_has_anthrax_episode = yes }
				OR = {
					supply_node > constant:japan_bio_campaign_target.building_floor
					naval_base > constant:japan_bio_campaign_target.building_floor
					industrial_complex > constant:japan_bio_campaign_target.anthrax_industry_threshold
					arms_factory > constant:japan_bio_campaign_target.anthrax_industry_threshold
					infrastructure > constant:japan_bio_campaign_target.anthrax_infrastructure_threshold
					any_state_division = { always = yes }
				}
			}
			AND = {
				check_variable = { var = japan_bio_campaign_agent value = constant:bio_lifecycle_agent.plague compare = equals }
				NOT = { bio_lifecycle_state_has_plague_episode = yes }
				OR = {
					is_capital = yes
					has_state_category = city
					has_state_category = large_city
					has_state_category = metropolis
					has_state_category = megalopolis
					state_population_k > constant:japan_bio_campaign_target.plague_population_threshold_k
					naval_base > constant:japan_bio_campaign_target.building_floor
				}
			}
		}
	}
}

	NOT = { black_friday_is_active = yes }
}

	black_friday_is_active = yes
	check_variable = { var = global.black_friday_discount_ratio value = constant:black_friday_event.baseline_payment_ratio compare = equals }
}

	black_friday_is_active = yes
	check_variable = { var = global.black_friday_discount_ratio value = constant:black_friday_event.evolution_payment_ratio compare = equals }
}

# These marker targets are saved in the native RAID_INSTANCE callback and
# carried only into the immediate hidden country event. Each family must
# contain exactly one marker before event-local temporary inputs are rebuilt.
japan_cbrn_raid_bridge_context_is_valid = {
	has_event_target = japan_cbrn_raid_actor
	has_event_target = japan_cbrn_raid_victim
	has_event_target = japan_cbrn_raid_state
	event_target:japan_cbrn_raid_actor = { tag = ROOT }
	event_target:japan_cbrn_raid_victim = { exists = yes }
	event_target:japan_cbrn_raid_state = { is_controlled_by = event_target:japan_cbrn_raid_victim }
}

	OR = {

		AND = { has_event_target = japan_cbrn_raid_price_sale NOT = { has_event_target = japan_cbrn_raid_price_normal } NOT = { has_event_target = japan_cbrn_raid_price_evolution } }

}
