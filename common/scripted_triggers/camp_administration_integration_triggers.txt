# Bounded registry retention and material record visibility for administration.
camp_admin_integration_has_continuing_work = {
	OR = {
		check_variable = { camp_admin_civilian_states^num > 0 }
		check_variable = { camp_occ_states^num > 0 }
		camp_occ_country_has_records = yes
		has_country_flag = camp_admin_civil_policy_authorized
		has_country_flag = camp_admin_civil_review_active
		has_country_flag = camp_admin_civil_claims_active
		has_country_flag = camp_admin_institution_slot_active
		has_country_flag = camp_admin_economy_slot_active
		has_country_flag = bel_congo_quota_economy_project_pending
		has_country_flag = bel_congo_corridor_economy_project_pending
		has_country_flag = camp_admin_country_archive_work_active
		has_country_flag = camp_admin_country_oversight_active
	}
}

# COUNTRY. Record availability is separate from periodic work registration.
camp_admin_integration_has_site_records = {
	check_variable = { camp_admin_known_site_states^num > constant:camp_admin_value.zero }
}
camp_admin_integration_is_records_only = {
	camp_admin_integration_has_site_records = yes
	NOT = { check_variable = { camp_active_site_states^num > constant:camp_admin_value.zero } }
	NOT = { camp_admin_integration_has_continuing_work = yes }
}

# COUNTRY. True for the current chain's dispatch event. A save without the
# stored earliest date treats its outstanding timer as current.
camp_admin_dispatch_timer_is_due = {
	OR = {
		NOT = { has_variable = camp_admin_dispatch_timer_earliest }
		check_variable = { var = camp_admin_dispatch_timer_earliest value = global.date compare = less_than_or_equals }
	}
}

# Country quote mirrors the bounded mandate price from the shared command constants.
# Replacement equipment is funded over time and never prevents immediate release.
camp_admin_generic_reform_quote = {
	set_temp_variable = { camp_admin_reform_quote_pp = camp_active_site_count }
	multiply_temp_variable = { camp_admin_reform_quote_pp = constant:camp_admin_command.policy_pp_per_site }
	add_to_temp_variable = { camp_admin_reform_quote_pp = constant:camp_admin_command.policy_pp }
	clamp_temp_variable = { var = camp_admin_reform_quote_pp min = constant:camp_admin_command.policy_pp max = constant:camp_admin_command.policy_pp_max }
	multiply_temp_variable = { camp_admin_reform_quote_pp = constant:camp_admin_command.mandate_pp_multiplier }
	# The order charges the Event 026 sale price at the ratio the sale price table holds.
	set_temp_variable = { camp_sale_ordinary = camp_admin_reform_quote_pp }
	camp_rework_sale_quote = yes
	set_temp_variable = { camp_admin_reform_quote_pp = camp_sale_payable }
	always = yes
}
camp_admin_generic_reform_can_pay = {
	camp_admin_generic_reform_quote = yes
	check_variable = { var = political_power value = camp_admin_reform_quote_pp compare = greater_than_or_equals }
}

camp_admin_generic_reform_visible = {
	has_country_flag = camp_admin_initialized
	camp_rework_country_uses_generic_kit = yes
	NOT = { camp_administration_civil_country_supported = yes }
	NOT = { has_country_flag = camp_admin_reforming }
	check_variable = { camp_active_site_states^num > 0 }
	camp_rework_country_can_reform = yes
}

camp_admin_generic_reform_available = {
	camp_admin_generic_reform_visible = yes
	camp_admin_country_sites_ready_for_reform = yes
	NOT = { has_country_flag = camp_rework_site_closure_active }
	NOT = { has_country_flag = camp_rework_gui_dismantlement_active }
	# Every site must be releasable now; the effect checks rows again before payment.
	all_of_scopes = {
		array = camp_active_site_states
		has_state_flag = camp_admin_initialized
		NOT = { has_state_flag = camp_admin_closed }
		camp_admin_site_authority_is_valid = yes
		OR = {
			camp_admin_site_rows_reconciled = yes
			# Closes with incomplete records and no survivor claim.
			camp_origin_host_records_incomplete = yes
		}
		is_valid_camp_dismantlement_state = yes
		var:camp_admin_responsible_country = { tag = PREV.PREV }
	}
	camp_admin_generic_reform_can_pay = yes
}
