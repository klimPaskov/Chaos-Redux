# ============================================================================
# CHAOS REDUX - CBRN DESIGNER TRIGGERS
# ============================================================================
# Overview:
# - Exposes completed CBRN MIO traits to country-scope logistics, exposure,
#   consequence, and AI logic.
# - Uses the installed 1.19 country-to-MIO iterator and exact organization and
#   trait tokens; merely owning an uncompleted trait never grants its effect.
# - Invisible organizations remain eligible after completion so later route or
#   policy changes cannot silently remove an earned institutional capability.
# - Every trigger is read-only and defaults false when the organization or
#   completed trait is absent.
# ============================================================================

	any_military_industrial_organization = {
		include_invisible = yes
		is_military_industrial_organization = cbrn_chemical_munitions_combine
		is_mio_trait_completed = cbrn_munitions_stable_choking_fill
	}
}
