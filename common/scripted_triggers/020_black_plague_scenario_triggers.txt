# ============================================================================
# CHAOS REDUX - BLACK PLAGUE TRIGGERABLE SCENARIO BRIDGE
# ============================================================================
# SCN-012 is available only when the complete Rat Nation/Rat King package and
# the atomic transaction are present. It never substitutes a disease-only
# launch and never treats the ordinary Event 020 start as a scenario launch.
# ============================================================================

black_plague_scenario_rat_country_package_is_ready = {
	RTA = { exists = yes }
}

black_plague_scenario_rat_king_package_is_ready = {
	RTX = {
		exists = yes
		OR = {
			has_country_flag = black_plague_rat_king_country
			AND = {
				NOT = { has_country_flag = black_plague_rat_country }
				NOT = { num_of_controlled_states > constant:black_plague_value.zero }
			}
		}
	}
}

black_plague_scenario_commit_provider_is_ready = {
	has_global_flag = black_plague_scenario_commit_provider_ready
}

black_plague_scenario_valid_human_state = {
	impassable = no
	NOT = { has_state_category = wasteland }
	NOT = { has_state_category = enclave }
	check_variable = { state_population_k > 0 }
	OWNER = { black_plague_country_is_human_host = yes }
	CONTROLLER = { black_plague_country_is_human_host = yes }
}

black_plague_scenario_has_europe = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		is_on_continent = europe
	}
}

black_plague_scenario_has_asia = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		is_on_continent = asia
	}
}

black_plague_scenario_has_africa = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		is_on_continent = africa
	}
}

black_plague_scenario_has_north_america = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		is_on_continent = north_america
	}
}

black_plague_scenario_has_south_america = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		is_on_continent = south_america
	}
}

black_plague_scenario_has_australia = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		is_on_continent = australia
	}
}

black_plague_scenario_has_two_eligible_continents = {
	OR = {
		AND = { black_plague_scenario_has_europe = yes black_plague_scenario_has_asia = yes }
		AND = { black_plague_scenario_has_europe = yes black_plague_scenario_has_africa = yes }
		AND = { black_plague_scenario_has_europe = yes black_plague_scenario_has_north_america = yes }
		AND = { black_plague_scenario_has_europe = yes black_plague_scenario_has_south_america = yes }
		AND = { black_plague_scenario_has_europe = yes black_plague_scenario_has_australia = yes }
		AND = { black_plague_scenario_has_asia = yes black_plague_scenario_has_africa = yes }
		AND = { black_plague_scenario_has_asia = yes black_plague_scenario_has_north_america = yes }
		AND = { black_plague_scenario_has_asia = yes black_plague_scenario_has_south_america = yes }
		AND = { black_plague_scenario_has_asia = yes black_plague_scenario_has_australia = yes }
		AND = { black_plague_scenario_has_africa = yes black_plague_scenario_has_north_america = yes }
		AND = { black_plague_scenario_has_africa = yes black_plague_scenario_has_south_america = yes }
		AND = { black_plague_scenario_has_africa = yes black_plague_scenario_has_australia = yes }
		AND = { black_plague_scenario_has_north_america = yes black_plague_scenario_has_south_america = yes }
		AND = { black_plague_scenario_has_north_america = yes black_plague_scenario_has_australia = yes }
		AND = { black_plague_scenario_has_south_america = yes black_plague_scenario_has_australia = yes }
	}
}

# Evolution II is forced by SCN-012, so the commit must have at least one real
# human-controlled port that can own the overseas-spread milestone. The
# scenario effect may seed that existing port after the ordinary continent
# anchors are chosen, but it never fabricates a naval base on a portless map.
black_plague_scenario_has_human_port = {
	any_state = {
		black_plague_scenario_valid_human_state = yes
		black_plague_rat_state_has_dock_archetype_signal = yes
	}
}

black_plague_scenario_has_free_rat_slot = {
	NOT = { has_global_flag = black_plague_rat_slot_rta_in_use }
	RTA = {
		exists = yes
		OR = {
			NOT = { has_variable = black_plague_rat_retirement_date }
			check_variable = {
				var = global.date
				value = black_plague_rat_retirement_date
				compare = greater_than_or_equals
			}
		}
	}
}

black_plague_scenario_has_usable_rat_slot = {
	OR = {
		black_plague_scenario_has_free_rat_slot = yes
		RTA = { black_plague_rat_country_is_active = yes }
	}
}

black_plague_scenario_has_existing_crisis = {
	OR = {
		has_global_flag = black_plague_system_started
		check_variable = { var = global.black_plague_established_state_count value = constant:black_plague_value.zero compare = greater_than }
	}
}

black_plague_scenario_has_minimum_state_candidates = {
	any_state = { black_plague_scenario_valid_human_state = yes }
}

# Called after intensity tuning, candidate collection, and continent-anchor
# selection. This is the fail-closed gate for the SCN-012 commit block: if it
# cannot reserve the requested geography, no disease, country, or evolution
# mutation is attempted.
black_plague_scenario_setup_preflight_is_ready = {
	check_variable = {
		var = global.black_plague_scenario_selected_continents
		value = global.black_plague_scenario_continent_target
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.black_plague_scenario_established_capacity
		value = global.black_plague_scenario_established_target
		compare = greater_than_or_equals
	}
	check_variable = { global.black_plague_scenario_anchor_states^num > constant:black_plague_value.zero }
	# On a clean map the infected candidate array is empty before seeding. The
	# reserved anchors become infected-or-worse candidates during the commit, so
	# accept either source while still failing closed when geography is empty.
	OR = {
		check_variable = { global.black_plague_scenario_candidate_states^num > constant:black_plague_value.zero }
		check_variable = { global.black_plague_scenario_anchor_states^num > constant:black_plague_value.zero }
	}
	black_plague_scenario_has_minimum_state_candidates = yes
	black_plague_scenario_has_human_port = yes
	black_plague_scenario_has_usable_rat_slot = yes
	# A stale Evolution IV flag without a living King cannot be repaired safely
	# inside the commit block. Reject it before disease or country mutation.
	OR = {
		NOT = { has_global_flag = black_plague_evolution_iv_active }
		AND = {
			has_global_flag = black_plague_evolution_iv_active
			RTX = { black_plague_rat_country_is_king = yes }
		}
	}
	OR = {
		RTX = { black_plague_rat_country_is_king = yes }
		any_state = {
			black_plague_scenario_valid_human_state = yes
			NOT = { has_state_flag = black_plague_scenario_reserved }
			NOT = { has_state_flag = black_plague_scenario_anchor }
			NOT = { has_state_flag = black_plague_scenario_king_reserved }
		}
	}
}

# SCN-012 is a bootstrap for the I-IV crisis. Once the natural campaign has
# recorded the terminal evolution or the separate King has been resolved as
# defeated, re-entering the bootstrap would reuse historical state that cannot
# be rolled back safely. Fail closed before the launch effect mutates anything.
black_plague_scenario_is_terminally_closed = {
	OR = {
		has_global_flag = world_end
		has_global_flag = black_plague_evolution_v_recorded
		has_global_flag = black_plague_terminal_takeover_complete
		has_global_flag = black_plague_rat_king_defeated
	}
}

black_plague_scenario_can_reconcile_from_triggerable_scenarios = {
	has_global_flag = black_plague_triggerable_scenario_launched
	NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap }
	NOT = { black_plague_scenario_is_terminally_closed = yes }
}

black_plague_scenario_can_start_from_triggerable_scenarios = {
	NOT = { black_plague_scenario_is_terminally_closed = yes }
	NOT = { has_global_flag = black_plague_triggerable_scenario_launched }
	NOT = { has_global_flag = black_plague_triggerable_scenario_bootstrap }
	black_plague_scenario_rat_country_package_is_ready = yes
	black_plague_scenario_rat_king_package_is_ready = yes
	black_plague_scenario_commit_provider_is_ready = yes
	black_plague_scenario_has_two_eligible_continents = yes
	OR = {
		black_plague_scenario_has_minimum_state_candidates = yes
		black_plague_scenario_has_existing_crisis = yes
	}
	black_plague_scenario_has_usable_rat_slot = yes
}

black_plague_scenario_can_launch_from_triggerable_scenarios = {
	OR = {
		black_plague_scenario_can_reconcile_from_triggerable_scenarios = yes
		black_plague_scenario_can_start_from_triggerable_scenarios = yes
	}
}
