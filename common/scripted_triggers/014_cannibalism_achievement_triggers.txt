# ============================================================================
# CHAOS REDUX - EVENT 014 ACHIEVEMENT, COUNTERPLAY, AND AFTERMATH TRIGGERS
# ============================================================================
# Defines the eighteen exact achievement contracts and the reusable gates for
# international suppression, convergence interdiction, island rescue, and
# post-defeat reconstruction. Manual scenario launches are intentionally
# excluded from the campaign achievements, but never from gameplay systems.
# ============================================================================

cannibalism_achievement_campaign_is_eligible = {
	has_global_flag = cannibalism_system_started
	NOT = { has_global_flag = cannibalism_manual_scenario_ever_launched }
}

cannibalism_achievement_player_is_valid = {
	is_ai = no
	has_capitulated = no
	cannibalism_achievement_campaign_is_eligible = yes
}

cannibalism_international_response_category_is_visible = {
	exists = yes
	NOT = { is_cannibalism_country = yes }
	NOT = { has_global_flag = world_end }
	OR = {
		AND = {
			has_global_flag = cannibalism_system_active
			has_global_flag = cannibalism_evolution_ii_active
		}
		AND = {
			has_global_flag = cannibalism_reconstruction_system_active
			has_country_flag = cannibalism_reconstruction_participant
		}
	}
}

cannibalism_can_pay_joint_suppression_cost = {
	has_manpower > constant:cannibalism_international_response_cost.joint_manpower_gate
	command_power > constant:cannibalism_international_response_cost.joint_command_gate
	NOT = { has_equipment = { infantry_equipment < constant:cannibalism_international_response_cost.joint_infantry } }
	NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_international_response_cost.joint_support } }
	NOT = { has_equipment = { motorized_equipment_1 < constant:cannibalism_international_response_cost.joint_trucks } }
}

	has_state_flag = cannibalism_former_island_host_origin
	NOT = { has_state_flag = cannibalism_island_survivors_rescued }
	is_controlled_by = ROOT
	OR = {
		has_state_flag = cannibalism_recovery_active
		has_state_flag = cannibalism_state_stabilized
	}
	NOT = { cannibalism_state_has_active_cell = yes }
	check_variable = {
		var = state_population_k
		value = constant:cannibalism_international_response_cost.rescue_population_floor_k
		compare = greater_than_or_equals
	}
}

cannibalism_state_recovery_supply_is_maintained = {
	OR = { is_island_state = yes is_coastal = yes }
	CONTROLLER = {
		exists = yes
		has_capitulated = no
		NOT = { has_country_flag = cannibalism_supply_corridor_failed }
		NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_decision_cost.logistics_support } }
		OR = {
			AND = {
				PREV = { OR = { is_island_state = yes is_coastal = yes } }
				NOT = { has_equipment = { convoy_1 < constant:cannibalism_decision_cost.logistics_convoy_reserve } }
			}
			AND = {
				PREV = { is_island_state = no }
				NOT = { has_equipment = { train_equipment_1 < constant:cannibalism_decision_cost.logistics_train_reserve } }
			}
		}
	}
	OR = { infrastructure > 0 naval_base > 0 supply_node > 0 }
}
