africa_achievement_normal_host_is_eligible = {
	has_country_flag = africa_achievement_host_actor_eligible

# exclusions therefore remain documentary invariants rather than live checks.

# HOST scope for the bounded Archive Disposition decision category. The source
# roster is the maintained relationship array; this never opens a world scan.
africa_achievement_archive_disposition_window_is_open = {
	africa_is_current_host = yes
	NOT = { has_global_flag = africa_is_one }
	any_of_scopes = {
		array = africa_relationship_countries
		africa_achievement_archive_disposition_target_is_valid = yes
	}
}

# COUNTRY scope on a selected evacuation target. A target may be disposed only
# once, while it remains in the current host generation and has not capitulated.

	exists = yes
	africa_is_valid_relationship_target = yes
	africa_member_host_generation_is_current = yes
	has_country_flag = africa_achievement_archive_evacuated
	NOT = { has_country_flag = africa_achievement_archive_disposition_recorded }
}
