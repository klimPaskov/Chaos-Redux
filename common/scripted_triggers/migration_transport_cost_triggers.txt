# Pure inclusive physical transport affordability in COUNTRY payer scope.
# Inputs: temporary migration_transport_target STATE pointer and the three
# migration_transport_fuel_land/fuel_sea/convoys_sea amounts from constants.
# Geography is shared by quoting, affordability and display in payer scope.

migration_transport_state_has_mainland_connection = {
	hidden_trigger = {
		OR = {
			var:migration_transport_target = {
				is_owned_by = PREV
				is_in_home_area = yes
			}
			AND = {
				var:migration_transport_target = { NOT = { is_owned_by = PREV } }
				# Native railway pathing starts at the paying country's capital.
				# An existing maximum-level route still qualifies if no further
				# construction can be requested on it.
				OR = {
					has_railway_connection = {
						start_state = THIS.capital
						target_state = var:migration_transport_target
						build_only_on_allied = no
					}
					can_build_railway = {
						start_state = THIS.capital
						target_state = var:migration_transport_target
						build_only_on_allied = no
					}
				}
			}
		}
	}
}

migration_transport_state_fuel_is_affordable = {
	hidden_trigger = {
		OR = {
			AND = {
				migration_transport_state_has_mainland_connection = yes
				NOT = { has_fuel < migration_transport_fuel_land }
			}
			AND = {
				NOT = { migration_transport_state_has_mainland_connection = yes }
				NOT = { has_fuel < migration_transport_fuel_sea }
			}
		}
	}
}

migration_transport_state_convoys_are_affordable = {
	hidden_trigger = {
		OR = {
			migration_transport_state_has_mainland_connection = yes
			AND = {
				NOT = { migration_transport_state_has_mainland_connection = yes }
				NOT = { has_equipment = { convoy < migration_transport_convoys_sea } }
			}
		}
	}
}

migration_transport_state_cost_is_affordable = {
	hidden_trigger = {
		var:migration_transport_target = { state_population_k > constant:migration_transport_cost.zero }
		migration_transport_state_fuel_is_affordable = yes
		migration_transport_state_convoys_are_affordable = yes
	}
}

migration_transport_policy_cost_is_affordable = {
	hidden_trigger = { NOT = { has_fuel < migration_transport_fuel_land } }
}

# COUNTRY payer with exact STATE target. Aircraft carry this shipment, so
# the state-keyed payment uses fuel without a sea-convoy requirement.
migration_transport_air_cost_is_affordable = {
	hidden_trigger = {
		var:migration_transport_target = { state_population_k > constant:migration_transport_cost.zero }
		migration_transport_policy_cost_is_affordable = yes
	}
}

migration_transport_state_receipt_is_paid = {
	hidden_trigger = {
		set_temp_variable = { migration_transport_target_id = constant:migration_transport_cost.zero }
		var:migration_transport_target = { set_temp_variable = { migration_transport_target_id = id } }
		check_variable = { migration_transport_target_id > constant:migration_transport_cost.zero }
		OR = {
			AND = {
				check_variable = { migration_transport_receipt = token:migration_open_reception }
				has_variable = migration_reception_preparation_fuel_paid_@var:migration_transport_target_id
				check_variable = { migration_reception_preparation_fuel_paid_@var:migration_transport_target_id > constant:migration_transport_cost.zero }
				has_variable = migration_reception_preparation_convoys_paid_@var:migration_transport_target_id
				NOT = { check_variable = { migration_reception_preparation_convoys_paid_@var:migration_transport_target_id < constant:migration_transport_cost.zero } }
			}
			AND = {
				NOT = { check_variable = { migration_transport_receipt = token:migration_open_reception } }
				meta_trigger = {
					text = {
						has_variable = [RECEIPT]_fuel_paid_@var:migration_transport_target_id
						check_variable = { [RECEIPT]_fuel_paid_@var:migration_transport_target_id > constant:migration_transport_cost.zero }
						has_variable = [RECEIPT]_convoys_paid_@var:migration_transport_target_id
						NOT = { check_variable = { [RECEIPT]_convoys_paid_@var:migration_transport_target_id < constant:migration_transport_cost.zero } }
					}
					RECEIPT = "[?migration_transport_receipt.GetTokenKey]"
				}
			}
		}
	}
}

migration_transport_state_receipt_is_clear = {
	hidden_trigger = {
		OR = {
			AND = {
				check_variable = { migration_transport_receipt = token:migration_open_reception }
				NOT = { has_variable = migration_reception_preparation_fuel_paid_@var:migration_transport_target_id }
				NOT = { has_variable = migration_reception_preparation_convoys_paid_@var:migration_transport_target_id }
			}
			AND = {
				NOT = { check_variable = { migration_transport_receipt = token:migration_open_reception } }
				meta_trigger = {
					text = {
						NOT = { has_variable = [RECEIPT]_fuel_paid_@var:migration_transport_target_id }
						NOT = { has_variable = [RECEIPT]_convoys_paid_@var:migration_transport_target_id }
					}
					RECEIPT = "[?migration_transport_receipt.GetTokenKey]"
				}
			}
		}
	}
}

migration_transport_policy_receipt_is_paid = {
	hidden_trigger = {
		meta_trigger = {
			text = {
				has_country_flag = [RECEIPT]_transport_paid
				has_variable = [RECEIPT]_fuel_paid
				check_variable = { [RECEIPT]_fuel_paid > constant:migration_transport_cost.zero }
			}
			RECEIPT = "[?migration_transport_receipt.GetTokenKey]"
		}
	}
}

migration_transport_policy_receipt_is_clear = {
	hidden_trigger = {
		meta_trigger = {
			text = {
				NOT = { has_country_flag = [RECEIPT]_transport_paid }
				NOT = { has_variable = [RECEIPT]_fuel_paid }
			}
			RECEIPT = "[?migration_transport_receipt.GetTokenKey]"
		}
	}
}
