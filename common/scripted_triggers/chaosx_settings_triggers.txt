has_disabled_current_evolution = {
    meta_trigger = {
        text = {
            has_global_flag = {
                flag = events_log_disabled_evolution_[EVOLUTION_EVENT_ID]_[EVOLUTION_TYPE]_[EVOLUTION_STAGE]
            }
        }
        EVOLUTION_EVENT_ID = "[?events_log_evolution_event_id|0]"
        EVOLUTION_TYPE = "[?events_log_evolution_type|0]"
        EVOLUTION_STAGE = "[?events_log_evolution_stage|0]"
    }
}

is_current_evolution_enabled = {
    NOT = { has_disabled_current_evolution = yes }
	has_reached_current_evolution_tier = yes
}

has_reached_current_evolution_tier = {
	OR = {
		NOT = { has_variable = events_log_evolution_tier }
		check_variable = {
			var = events_log_evolution_tier
			value = 1
			compare = less_than
		}
		AND = {
			check_variable = { var = events_log_evolution_tier value = 1 compare = equals }
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_1.min
				compare = greater_than_or_equals
			}
		}
		AND = {
			check_variable = { var = events_log_evolution_tier value = 2 compare = equals }
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_2.min
				compare = greater_than_or_equals
			}
		}
		AND = {
			check_variable = { var = events_log_evolution_tier value = 3 compare = equals }
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_3.min
				compare = greater_than_or_equals
			}
		}
		AND = {
			check_variable = { var = events_log_evolution_tier value = 4 compare = equals }
			check_variable = {
				var = global.chaos_meter_value
				value = constant:chaos_meter_tier_range.tier_4.min
				compare = greater_than_or_equals
			}
		}
		AND = {

# Input: variable `events_log_fire_candidate_id` holding an event ID.
# Mirrors the availability rules of the single event-detail trigger button:
# the event must be selected (enabled), unique events must not have fired yet,
# and special events need a valid manual fire context.
events_log_fire_candidate_is_available = {
	NOT = { is_in_array = { global.disabled_events = events_log_fire_candidate_id } }
	OR = {
		AND = {
			check_variable = { events_log_fire_candidate_id = constant:africa_event.id }
			africa_manual_event_is_available = yes
		}
		AND = {
			check_variable = { events_log_fire_candidate_id = constant:random_faction_event.id }
			random_faction_has_manual_dispatchable_context = yes
		}
		AND = {
			NOT = { check_variable = { events_log_fire_candidate_id = constant:africa_event.id } }
			NOT = { check_variable = { events_log_fire_candidate_id = constant:random_faction_event.id } }
			OR = {
				NOT = { is_in_array = { global.fired_events = events_log_fire_candidate_id } }
				NOT = {
					OR = {
						is_in_array = { global.major_events = events_log_fire_candidate_id }
						is_in_array = { global.fire_once_events = events_log_fire_candidate_id }
					}
				}
			}
		}
	}
}

# True when at least one selected (enabled) event in the Event Logs events
# view can currently be fired manually.
events_log_any_selected_event_is_fireable = {
	any_of = {
		array = global.all_events
		value = events_log_fire_candidate_id
		events_log_fire_candidate_is_available = yes
	}
}
