# Supported scope: COUNTRY. This proves a current-transition map return with an
# exact calendar date and an arithmetic day receipt. Legacy completed saves do
# not receive a fabricated reveal day.
fallout_event_map_return_receipts_are_current = {
	has_global_flag = fallout_active
	has_global_flag = fallout_transition_complete
	has_global_flag = fallout_transition_map_return_applied
	NOT = { has_global_flag = fallout_transition_active }
	has_variable = global.fallout_transition_generation
	has_variable = global.fallout_event_timeline_generation
	has_variable = global.fallout_event_timeline_start_date
	has_variable = global.fallout_event_timeline_start_day
	check_variable = {
		var = global.fallout_event_timeline_generation
		value = global.fallout_transition_generation
		compare = equals
	}
	NOT = {
		check_variable = {
			var = global.fallout_event_timeline_start_date
			value = global.date
			compare = greater_than
		}
	}
	NOT = {
		check_variable = {
			var = global.fallout_event_timeline_start_day
			value = global.num_days
			compare = greater_than
		}
	}
}

# Supported scope: COUNTRY
fallout_event_timeline_phase_receipt_is_current = {
	fallout_event_map_return_receipts_are_current = yes
	has_variable = global.fallout_event_timeline_elapsed_days
	has_variable = global.fallout_event_current_phase
	has_variable = global.fallout_event_phase_generation
	has_variable = global.fallout_event_phase_last_update_day
	check_variable = {
		var = global.fallout_event_phase_generation
		value = global.fallout_transition_generation
		compare = equals
	}
	check_variable = {
		var = global.fallout_event_phase_last_update_day
		value = global.num_days
		compare = equals
	}
	NOT = { check_variable = { global.fallout_event_timeline_elapsed_days < constant:fallout_event_runtime.zero } }
	check_variable = { global.fallout_event_current_phase > constant:fallout_event_phase.none }
	check_variable = { global.fallout_event_current_phase < constant:fallout_event_phase.upper_bound }
	OR = {
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.ash_week }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.ash_week_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.first_season }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.ash_week_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.first_season_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.first_winter_year }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.first_season_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.first_winter_year_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.consolidation }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.first_winter_year_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.consolidation_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.rival_orders }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.consolidation_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.rival_orders_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.new_states }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.rival_orders_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.new_states_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.soot_retreat }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.new_states_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.soot_retreat_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.second_world }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.soot_retreat_end_days }
			NOT = { check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.second_world_end_days } }
		}
		AND = {
			check_variable = { global.fallout_event_current_phase = constant:fallout_event_phase.open_continuation }
			check_variable = { global.fallout_event_timeline_elapsed_days > constant:fallout_event_timeline.second_world_end_days }
		}
	}
}

fallout_event_registry_country_identity_is_current = {
	has_country_flag = fallout_event_registry_member
	has_country_flag = fallout_event_runtime_initialized
	has_variable = global.fallout_event_registry_generation
	has_variable = global.fallout_event_registry_count
	has_variable = fallout_event_registry_generation
	has_variable = fallout_event_registry_index
	has_variable = fallout_event_runtime_generation
	check_variable = {
		var = fallout_event_registry_generation
		value = global.fallout_event_registry_generation
		compare = equals
	}
	check_variable = {
		var = fallout_event_runtime_generation
		value = global.fallout_event_registry_generation
		compare = equals
	}
	NOT = { check_variable = { fallout_event_registry_index < constant:fallout_event_runtime.first_array_index } }
	check_variable = {
		var = fallout_event_registry_index
		value = global.fallout_event_registry_count
		compare = less_than
	}
	set_temp_variable = { fallout_event_registry_back_reference_context_loaded = constant:fallout_event_runtime.zero }
	var:global.fallout_event_registry_countries^fallout_event_registry_index = {
		set_temp_variable = { fallout_event_registry_back_reference_context_loaded = constant:fallout_event_runtime.one }
		tag = PREV
	}
	check_variable = { fallout_event_registry_back_reference_context_loaded = constant:fallout_event_runtime.one }
	check_variable = {
		var = global.fallout_event_registry_generation_entries^fallout_event_registry_index
		value = fallout_event_registry_generation
		compare = equals
	}
	check_variable = {
		var = global.fallout_event_registry_index_entries^fallout_event_registry_index
		value = fallout_event_registry_index
		compare = equals
	}
	fallout_event_registry_arrays_are_aligned = yes

# Supported scope: COUNTRY. This validates the complete uncommitted payload
# before the ready flag is written. Initial registry equality is frozen from
# the final allocation array. Later annexation does not delete the identity row.
fallout_event_registry_payload_is_current = {
	fallout_event_registry_header_payload_is_current = yes
	all_of = {
		array = global.fallout_event_registry_index_entries
		value = fallout_event_registry_stored_index
		index = fallout_event_registry_payload_index
		set_temp_variable = { fallout_event_registry_payload_country_context_loaded = constant:fallout_event_runtime.zero }
		check_variable = {
			var = fallout_event_registry_stored_index
			value = fallout_event_registry_payload_index
			compare = equals
		}
		var:global.fallout_event_registry_countries^fallout_event_registry_payload_index = {
			set_temp_variable = { fallout_event_registry_payload_country_context_loaded = constant:fallout_event_runtime.one }
			check_variable = {
				var = fallout_event_registry_index
				value = fallout_event_registry_payload_index
				compare = equals
			}
			is_in_array = { array = global.fallout_successor_assigned_countries value = THIS }
			fallout_event_registry_country_row_is_current = yes
		}
		check_variable = { fallout_event_registry_payload_country_context_loaded = constant:fallout_event_runtime.one }
	}
	all_of = {
		array = global.fallout_successor_assigned_countries
		value = fallout_event_registry_source_country
		set_temp_variable = { fallout_event_registry_source_country_context_loaded = constant:fallout_event_runtime.zero }
		var:fallout_event_registry_source_country = {
			set_temp_variable = { fallout_event_registry_source_country_context_loaded = constant:fallout_event_runtime.one }
			is_in_array = { array = global.fallout_event_registry_countries value = THIS }
		}
		check_variable = { fallout_event_registry_source_country_context_loaded = constant:fallout_event_runtime.one }
	}
}

# Supported scope: COUNTRY. Pending rows may hold any exact subset of the five
# receipts. A flag without its generation, or a generation without its flag,
# invalidates the row.
fallout_event_orientation_receipts_are_consistent = {
	has_variable = fallout_event_orientation_status
	OR = {
		fallout_event_orientation_is_complete = yes
		AND = {
			check_variable = { fallout_event_orientation_status = constant:fallout_event_orientation_status.pending }
			NOT = { has_country_flag = fallout_event_orientation_complete }
			NOT = { has_variable = fallout_event_orientation_generation }
			OR = {
				fallout_event_orientation_national_receipt_is_current = yes
				AND = {
					NOT = { has_country_flag = fallout_event_orientation_national_complete }
					NOT = { has_variable = fallout_event_orientation_national_generation }
				}
			}
			OR = {
				fallout_event_orientation_capital_receipt_is_current = yes
				AND = {
					NOT = { has_country_flag = fallout_event_orientation_capital_complete }
					NOT = { has_variable = fallout_event_orientation_capital_generation }
				}
			}
			OR = {
				fallout_event_orientation_resource_receipt_is_current = yes
				AND = {
					NOT = { has_country_flag = fallout_event_orientation_resource_complete }
					NOT = { has_variable = fallout_event_orientation_resource_generation }
				}
			}
			OR = {
				fallout_event_orientation_archetype_receipt_is_current = yes
				AND = {
					NOT = { has_country_flag = fallout_event_orientation_archetype_complete }
					NOT = { has_variable = fallout_event_orientation_archetype_generation }
				}
			}
			OR = {
				fallout_event_orientation_character_receipt_is_current = yes
				AND = {
					NOT = { has_country_flag = fallout_event_orientation_character_complete }
					NOT = { has_variable = fallout_event_orientation_character_generation }
				}
			}
		}
	}
}

# Supported scope: COUNTRY. Inputs: fallout_event_subject_type_to_check and
# fallout_event_subject_value_to_check. This proves only the stored type/value
# shape. Live actor and target validity is checked separately for cancellation.
fallout_event_subject_shape_is_valid = {
	check_variable = { fallout_event_subject_type_to_check > constant:fallout_event_subject_type.none }
	check_variable = { fallout_event_subject_type_to_check < constant:fallout_event_subject_type.upper_bound }
	check_variable = { fallout_event_subject_value_to_check > constant:fallout_event_runtime.zero }
}

# Supported scope: COUNTRY. Inputs are the same as the shape trigger.
fallout_event_subject_is_current = {
	fallout_event_subject_shape_is_valid = yes
	OR = {
		AND = {
			check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.country }
			set_temp_variable = { fallout_event_subject_country_context_loaded = constant:fallout_event_runtime.zero }
			var:fallout_event_subject_value_to_check = {
				set_temp_variable = { fallout_event_subject_country_context_loaded = constant:fallout_event_runtime.one }
				exists = yes
				fallout_event_registry_country_row_is_current = yes
			}
			check_variable = { fallout_event_subject_country_context_loaded = constant:fallout_event_runtime.one }
		}
		AND = {
			check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.state }
			set_temp_variable = { fallout_event_subject_state_context_loaded = constant:fallout_event_runtime.zero }
			var:fallout_event_subject_value_to_check = {
				set_temp_variable = { fallout_event_subject_state_context_loaded = constant:fallout_event_runtime.one }
				has_variable = fallout_pretransition_snapshot_generation
				check_variable = {
					var = fallout_pretransition_snapshot_generation
					value = global.fallout_transition_generation
					compare = equals
				}
			}
			check_variable = { fallout_event_subject_state_context_loaded = constant:fallout_event_runtime.one }
		}
		AND = {
			check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.character }
			has_character = var:fallout_event_subject_value_to_check
		}
		AND = {
			check_variable = { fallout_event_subject_type_to_check = constant:fallout_event_subject_type.institution }
			# No generic engine liveness surface exists for an opaque institution
			# token. Future content must add a generation-bound registry first.
			always = no
		}
	}
}
