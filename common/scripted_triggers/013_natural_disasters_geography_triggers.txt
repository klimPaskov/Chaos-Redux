		AND = { check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.indian_tsunami } natural_disaster_state_is_indian_tsunami_basin = yes }
		AND = { check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.atlantic_tsunami } natural_disaster_state_is_atlantic_tsunami_basin = yes }
		AND = { check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.mediterranean_tsunami } natural_disaster_state_is_mediterranean_tsunami_basin = yes }
		AND = { check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.black_sea_tsunami } natural_disaster_state_is_black_sea_tsunami_basin = yes }
		AND = { check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.tornado } natural_disaster_state_is_tornado_zone = yes }
		AND = { check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.windstorm } natural_disaster_state_is_windstorm_zone = yes }
	}
}

natural_disaster_target_matches_locked_path_basin = {
	OR = {
		NOT = { has_variable = natural_disaster_sequence_path_basin }
		check_variable = { natural_disaster_sequence_path_basin = constant:natural_disaster_path_basin.none }
		NOT = {
			check_variable = {
				var = natural_disaster_current_family
				value = natural_disaster_sequence_path_family
				compare = equals
			}
		}
		natural_disaster_state_matches_path_basin = yes
	}
}

natural_disaster_state_is_active_origin_or_neighbor = {
	check_variable = { natural_disaster_active_origin_saved > 0 }
	has_event_target = natural_disaster_active_origin_state
	OR = {
		state = event_target:natural_disaster_active_origin_state
		any_neighbor_state = { state = event_target:natural_disaster_active_origin_state }
	}
}

natural_disaster_target_matches_active_origin_tsunami_basin = {
	check_variable = { natural_disaster_active_origin_saved > 0 }
	has_event_target = natural_disaster_active_origin_state
	OR = {
		AND = { natural_disaster_state_is_pacific_tsunami_basin = yes event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_pacific_tsunami_basin = yes } }
		AND = { natural_disaster_state_is_indian_tsunami_basin = yes event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_indian_tsunami_basin = yes } }
		AND = { natural_disaster_state_is_atlantic_tsunami_basin = yes event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_atlantic_tsunami_basin = yes } }
		AND = { natural_disaster_state_is_mediterranean_tsunami_basin = yes event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_mediterranean_tsunami_basin = yes } }
		AND = { natural_disaster_state_is_black_sea_tsunami_basin = yes event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_black_sea_tsunami_basin = yes } }
	}
}

natural_disaster_active_origin_is_volcanic = {
	check_variable = { natural_disaster_active_origin_saved > 0 }
	has_event_target = natural_disaster_active_origin_state
	OR = {
		check_variable = { natural_disaster_active_origin_medium = constant:natural_disaster_origin_medium.none }
		check_variable = { natural_disaster_active_origin_medium = constant:natural_disaster_origin_medium.land_impact }
	}
	OR = {
		AND = {
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.volcanic_eruption }
			event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_volcanic_vent_zone = yes }
		}
		AND = {
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.massive_eruption }
			event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_massive_eruption_zone = yes }
		}
	}
}

natural_disaster_active_origin_can_generate_tsunami = {
	check_variable = { natural_disaster_active_origin_saved > 0 }
	has_event_target = natural_disaster_active_origin_state
	OR = {
		AND = {
			OR = {
				check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.earthquake }
				check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.whole_earth_rupture }
			}
			OR = {
				check_variable = { natural_disaster_active_origin_medium = constant:natural_disaster_origin_medium.none }
				check_variable = { natural_disaster_active_origin_medium = constant:natural_disaster_origin_medium.land_impact }
			}
			event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_seismic_zone = yes }
		}
		AND = {
			natural_disaster_active_origin_is_volcanic = yes
		}
		AND = {
			OR = {
				check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.meteor_impact }
				check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.meteor_shower }
			}
			check_variable = { natural_disaster_active_origin_medium = constant:natural_disaster_origin_medium.ocean_impact }
			event_target:natural_disaster_active_origin_state = { natural_disaster_state_is_ocean_exposed_coast = yes }
		}
	}
}

# State scope. Origin-dependent families always require a supplied or inherited
# physical origin. The receptor can be the origin state itself, but the target
# can never manufacture the origin proof that authorizes its own impact.
natural_disaster_family_origin_context_is_valid = {
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.ashfall } }
		AND = {
			natural_disaster_active_origin_is_volcanic = yes
			natural_disaster_state_is_active_origin_or_neighbor = yes
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.lahar } }
		AND = {
			natural_disaster_active_origin_is_volcanic = yes
			natural_disaster_state_is_active_origin_or_neighbor = yes
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.tsunami } }
		AND = {
			natural_disaster_active_origin_can_generate_tsunami = yes
			natural_disaster_target_matches_active_origin_tsunami_basin = yes
		}
	}
}

# Physical initiating context is separate from immutable terrain. It prevents
# barren desert wildfire and flat-land mass movement while still permitting
# lightning-driven Siberian forest fires without making Siberia heat-capable.
natural_disaster_family_initiating_context_is_valid = {
	if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.wildfire } }
		OR = {
			natural_disaster_state_is_heat_zone = yes
			natural_disaster_state_is_drought_zone = yes
			natural_disaster_state_is_arid_zone = yes
			natural_disaster_state_is_windstorm_zone = yes
			natural_disaster_state_is_convective_storm_zone = yes
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.meteor_impact }
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.meteor_shower }
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.thunderstorm }
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.wildfire }
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.dry_mass_movement } }
		OR = {
			natural_disaster_state_is_seismic_zone = yes
			natural_disaster_state_is_drought_zone = yes
			natural_disaster_state_is_heat_zone = yes
			natural_disaster_state_is_arid_zone = yes
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.earthquake }
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.whole_earth_rupture }
		}
	}
	else_if = {
		limit = { check_variable = { natural_disaster_current_family = constant:natural_disaster_family.wet_mass_movement } }
		OR = {
			natural_disaster_state_is_flood_exposed_zone = yes
			natural_disaster_state_is_tropical_storm_basin = yes
			natural_disaster_state_is_convective_storm_zone = yes
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.flood }
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.tropical_cyclone }
			check_variable = { natural_disaster_active_origin_family = constant:natural_disaster_family.thunderstorm }
		}
	}
}
