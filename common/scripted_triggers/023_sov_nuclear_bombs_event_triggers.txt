# Country scope. Event 23's scripted release goes through launch_nuke behind
# this preflight, which mirrors the vanilla nuclear_strike raid: atomic
# weapons, at least ten deployed strategic bombers (the strategic_bomber
# sub-unit type also covers intercontinental bombers), a controlled air base
# and fuel. Exact-endpoint range and war with the target are checked by the
# targeted wrappers and target validators.
sov_nuclear_bombs_event_has_delivery_route = {
	has_tech = atomic_research
	has_tech = nukes
	OR = {
		has_tech = strategic_bomber1
		AND = {
			has_dlc = "By Blood Alone"
			has_tech = iw_large_airframe
		}
	}
	# Numeric fields are injected because direct constant support is undocumented.
	set_temp_variable = { sov_nuclear_bombs_delivery_bomber_floor = constant:sov_nuclear_bombs_delivery.deployed_bomber_floor }
	set_temp_variable = { sov_nuclear_bombs_delivery_airbase_floor = constant:sov_nuclear_bombs_delivery.airbase_level_floor }
	set_temp_variable = { sov_nuclear_bombs_delivery_fuel = constant:sov_nuclear_bombs_delivery.minimum_fuel }
	meta_trigger = {
		text = {
			has_deployed_air_force_size = { size > [BOMBERS] type = strategic_bomber }
			NOT = { has_fuel < [FUEL] }
			any_owned_state = {
				is_controlled_by = PREV air_base > [AIRBASE] }
		}
		BOMBERS = "[?sov_nuclear_bombs_delivery_bomber_floor|.0]"
		FUEL = "[?sov_nuclear_bombs_delivery_fuel|.0]"
		AIRBASE = "[?sov_nuclear_bombs_delivery_airbase_floor|.0]"
	}
}

# State scope, PREV is the launcher country. Checks this exact endpoint.
# No persistent outputs. Range is an event ceiling, not an air-wing range query.
sov_nuclear_bombs_event_state_has_delivery_access = {
	scope_exists = yes
	impassable = no
	OR = { owner = { exists = yes } controller = { exists = yes } }
	PREV = {
		sov_nuclear_bombs_event_has_delivery_route = yes
		set_temp_variable = { sov_nuclear_bombs_delivery_range = constant:sov_nuclear_bombs_delivery.maximum_state_distance }
		any_owned_state = {
			is_controlled_by = PREV
			meta_trigger = {
				text = {
					air_base > [AIRBASE]
					distance_to = { target = PREV.PREV value < [RANGE] }
				}
				AIRBASE = "[?sov_nuclear_bombs_delivery_airbase_floor|.0]"
				RANGE = "[?sov_nuclear_bombs_delivery_range|.0]"
			}
		}
	}
}

# State scope. Evolution IV first use may select a lower-consequence military
# endpoint while the severe emergency gate is active. Capital and heavily
# populated states remain excluded unless the enemy has already used a nuclear
# weapon against a populated Soviet state, which is recorded by the neutral
# on_nuke_drop hook.

	NOT = { is_capital = yes }
	state_population_k < constant:sov_nuclear_bombs_tuning.first_use_target_population_max_k

		air_base > 0
		arms_factory > 0
		dockyard > 0
		naval_base > 0
		radar_station > 0
		rail_way > 0
	}
}

# Country scope, the Soviet actor. Inputs of the weekly war assessment in
# sov_nuclear_bombs_assess_strategic_losses: the home capital (Moscow) is
# occupied or held by an enemy of the Union, or the current capital is not
# Soviet-controlled or borders a state held by a country at war with the Union.
# A Moscow left to a breakaway at peace with the Union does not count.
sov_nuclear_bombs_event_home_capital_is_lost = {
	NOT = { controls_state = constant:sov_nuclear_bombs_tuning.home_capital_state }
	OR = {
		owns_state = constant:sov_nuclear_bombs_tuning.home_capital_state
		any_enemy_country = { controls_state = constant:sov_nuclear_bombs_tuning.home_capital_state }
	}
}

sov_nuclear_bombs_event_capital_is_threatened = {
	OR = {
		sov_nuclear_bombs_event_home_capital_is_lost = yes
		capital_scope = {
			OR = {
				NOT = { is_controlled_by = SOV }
				any_neighbor_state = {
					controller = { has_war_with = SOV }
				}
			}
		}
	}
}

# Country scope. The Evolution IV emergency shared by a staged first use and
# the prospective first-use target selection. It never reads a staged action.
# The strategic-loss count and its three flags are written only by the weekly
# war assessment, so this gate reads the last assessed war situation.
sov_nuclear_bombs_event_first_use_emergency = {
	sov_nuclear_bombs_event_has_evolution_iv = yes
	check_variable = { var = global.chaos_meter_value value = constant:sov_nuclear_bombs_evolution.first_use_chaos compare = greater_than_or_equals }
	has_war = yes
	check_variable = { var = sov_nuclear_bombs_strategic_losses value = constant:sov_nuclear_bombs_evolution.first_use_min_strategic_loss compare = greater_than_or_equals }
	has_country_flag = sov_nuclear_bombs_capital_threat_active
	has_country_flag = sov_nuclear_bombs_front_collapsed
	has_country_flag = sov_nuclear_bombs_reserve_exhausted
	OR = {
		has_country_flag = sov_nuclear_bombs_confirmed_enemy_nuclear_use
		has_country_flag = sov_nuclear_bombs_enemy_launch_preparation_verified
	}
	NOT = { has_country_flag = sov_nuclear_bombs_atomic_moratorium }
	NOT = { has_country_flag = sov_nuclear_bombs_soviet_major_first_use_recorded }
	sov_nuclear_bombs_event_has_delivery_route = yes
	sov_nuclear_bombs_event_has_delivery_crew_certification = yes
	check_variable = { var = sov_nuclear_bombs_readiness value = constant:sov_nuclear_bombs_tuning.first_use_profile_readiness compare = greater_than_or_equals }
	check_variable = { var = sov_nuclear_bombs_integrity value = constant:sov_nuclear_bombs_tuning.minimum_integrity_for_release compare = greater_than_or_equals }

# Country scope. Evolution IV first use against a nuclear major before any
# action is staged: the full emergency, one line of release authority, an
# unreserved device, no suspended orders and no verified stand-down. It opens
# target selection and preparation only; the staged severe gate, technical
# certification, a separate final authorization and the runtime first-use
# guard still decide release.
sov_nuclear_bombs_event_prospective_first_use_gate = {
	sov_nuclear_bombs_event_actor_is_valid = yes
	sov_nuclear_bombs_event_first_use_emergency = yes
	NOT = { has_country_flag = sov_nuclear_bombs_release_orders_suspended }
	NOT = { has_country_flag = sov_nuclear_bombs_major_exchange_stood_down }
	OR = {
		has_country_flag = sov_nuclear_bombs_authority_centralized
		has_country_flag = sov_nuclear_bombs_retaliation_delegated
	}
	sov_nuclear_bombs_event_has_unreserved_device = yes
}

# Country scope. The authority half of the prospective gate, used for its own

# State scope, ROOT is the Soviet actor. A state of a nuclear major at war with
# ROOT that is a lower-consequence military or logistics endpoint. Capitals and

	is_major = no
	NOT = { has_capitulated = yes }

# Country scope. This is the final settlement proof used by Event 23's
# verification event. Every accepted or partial response must leave a real,
# named term in the world before it can be recorded as durable compliance.
sov_nuclear_bombs_event_settlement_terms_are_valid = {
	sov_nuclear_bombs_event_actor_is_valid = yes
	has_country_flag = sov_nuclear_bombs_settlement_terms_applied
	has_variable = sov_nuclear_bombs_demand_kind
	has_variable = sov_nuclear_bombs_demand_state_id
	has_event_target = sov_nuclear_bombs_coercion_target
	has_event_target = sov_nuclear_bombs_selected_state
	event_target:sov_nuclear_bombs_coercion_target = {
		exists = yes
		NOT = { tag = SOV }
		is_subject = no
		NOT = { has_capitulated = yes }
	}
	event_target:sov_nuclear_bombs_selected_state = {
		check_variable = { var = THIS.id value = ROOT.sov_nuclear_bombs_demand_state_id compare = equals }
	}
	if = {
		limit = { has_country_flag = sov_nuclear_bombs_partial_settlement_accepted }
		OR = {
			AND = {
				OR = {
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.cease_hostilities compare = equals }
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals }
				}
				has_country_flag = sov_nuclear_bombs_settlement_ceasefire_recorded
				event_target:sov_nuclear_bombs_coercion_target = { NOT = { has_war_with = SOV } }
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals } }
					has_country_flag = sov_nuclear_bombs_settlement_surrender_recorded
				}
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.withdraw_territory compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_withdrawal_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_demilitarized_zone }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.return_devices compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_return_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_site_joint_custody }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.soviet_protection compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_protection_recorded
				has_guaranteed = event_target:sov_nuclear_bombs_coercion_target
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.demilitarize_zone compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_demilitarization_recorded
				event_target:sov_nuclear_bombs_selected_state = { has_state_flag = sov_nuclear_bombs_demilitarized_zone }
			}
		}
	}
	else = {
		OR = {
			AND = {
				OR = {
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.cease_hostilities compare = equals }
					check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals }
				}
				has_country_flag = sov_nuclear_bombs_settlement_ceasefire_recorded
				event_target:sov_nuclear_bombs_coercion_target = { NOT = { has_war_with = SOV } }
				OR = {
					NOT = { check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.surrender_collapsing_war compare = equals } }
					has_country_flag = sov_nuclear_bombs_settlement_surrender_recorded
				}
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.withdraw_territory compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_withdrawal_recorded
				event_target:sov_nuclear_bombs_selected_state = { OR = { is_owned_by = SOV is_controlled_by = SOV } }
			}
			AND = {
				check_variable = { var = sov_nuclear_bombs_demand_kind value = constant:sov_nuclear_bombs_demand.return_devices compare = equals }
				has_country_flag = sov_nuclear_bombs_settlement_return_recorded
