acid_rain_state_is_excluded = {
	OR = {
		has_state_category = wasteland
		has_state_category = enclave
		has_state_flag = acid_rain_shared_civilian_loss_excluded
		has_state_flag = fallout_shared_civilian_loss_excluded
		has_state_flag = air_winter_excluded
	}
}

acid_rain_state_has_supported_region = {
	OR = {
		is_on_continent = europe
		is_on_continent = middle_east
		is_on_continent = asia
		is_on_continent = africa
		is_on_continent = australia
		is_on_continent = north_america
		is_on_continent = south_america
	}
}

acid_rain_state_is_valid_target = {
	scope_exists = yes
	impassable = no
	NOT = { acid_rain_state_is_excluded = yes }
	acid_rain_state_has_supported_region = yes
	check_variable = { state_population_k > constant:acid_rain_mortality.minimum_population_k }
	OWNER = {
		exists = yes
		is_actual_nonhuman_country = no
	}
	CONTROLLER = {
		exists = yes
		is_actual_nonhuman_country = no
	}
}

# STATE SCOPE. True while this state's controller runs an evacuation here
# that started in the current generation and has run at least one full day.
acid_rain_state_has_partial_evacuation = {
	has_state_flag = acid_rain_action_lock
	has_variable = acid_rain_action_type
	OR = {
		check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
		check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
	}
	has_variable = acid_rain_action_owner_generation
	check_variable = { var = acid_rain_action_owner_generation value = global.acid_rain_generation_id compare = equals }
	has_variable = acid_rain_action_owner_country_id
	check_variable = { var = acid_rain_action_owner_country_id value = CONTROLLER.id compare = equals }
	has_variable = acid_rain_evacuation_partial_day
	check_variable = { var = global.date value = acid_rain_evacuation_partial_day compare = greater_than_or_equals }
}

# A later generation may begin only after every paid commitment and state-level
# recovery obligation from the resolved generation has settled. This keeps the
# reset path idempotent without cancelling factories, manpower, or target proofs.
acid_rain_previous_generation_is_settled = {
	OR = {
		NOT = { has_global_flag = acid_rain_runtime_resolved }
		AND = {
			has_global_flag = acid_rain_runtime_resolved
			all_of_scopes = {
				array = global.acid_rain_participant_countries
				NOT = { check_variable = { acid_rain_active_projects > constant:acid_rain_runtime.zero } }
				NOT = { has_country_flag = acid_rain_national_action_active }
				NOT = { has_country_flag = acid_rain_recovery_tail_active }
			}
			all_of_scopes = {
				array = global.acid_rain_frozen_states
				NOT = { has_state_flag = acid_rain_aftercare_unresolved }
				NOT = { has_state_flag = acid_rain_action_lock }
			}
		}
	}
}

# COUNTRY SCOPE. The caller computes these exact pre-spend requirements from
# the currently loaded quote. Meta trigger injection is needed because stockpile
# triggers do not accept variable right-hand values directly.
acid_rain_country_ai_reserves_allow_loaded_quote = {
	set_temp_variable = { acid_rain_motorized_available = constant:acid_rain_runtime.zero }
	acid_rain_prepare_motorized_stock = yes
	meta_trigger = {
		text = {
			NOT = { has_equipment = { support_equipment < [SUPPORT_REQUIRED] } }
			NOT = { has_equipment = { train_equipment < [TRAIN_REQUIRED] } }
			NOT = { has_equipment = { convoy_1 < [CONVOY_REQUIRED] } }
			check_variable = { var = acid_rain_motorized_available value = [MOTORIZED_REQUIRED] compare = greater_than_or_equals }
			NOT = { has_manpower < [MANPOWER_REQUIRED] }
			NOT = { has_fuel < [FUEL_REQUIRED] }
			num_of_civilian_factories_available_for_projects > [CIV_TRIGGER_FLOOR]
		}
		SUPPORT_REQUIRED = "[?acid_rain_ai_required_support|.0]"
		TRAIN_REQUIRED = "[?acid_rain_ai_required_train|.0]"
		CONVOY_REQUIRED = "[?acid_rain_ai_required_convoy|.0]"
		MOTORIZED_REQUIRED = "[?acid_rain_ai_required_motorized|.0]"

		has_variable = acid_rain_global_cap_population_people
		has_variable = acid_rain_global_episode_loss
		set_temp_variable = { acid_rain_ai_cap_room = acid_rain_global_cap_population_people }
		multiply_temp_variable = { acid_rain_ai_cap_room = acid_rain_ai_cap_fraction }
		subtract_from_temp_variable = { acid_rain_ai_cap_room = acid_rain_global_episode_loss }
	}
	check_variable = { var = acid_rain_ai_cap_room value = constant:acid_rain_runtime.one compare = greater_than_or_equals }
	set_temp_variable = { acid_rain_ai_share = CONTROLLER.acid_rain_preparedness }
	multiply_temp_variable = { acid_rain_ai_share = constant:acid_rain_mortality.protection_coefficient }
	multiply_temp_variable = { acid_rain_ai_share = constant:acid_rain_aftermath.negative_one }
	add_to_temp_variable = { acid_rain_ai_share = constant:acid_rain_mortality.one }
	if = {
		limit = { check_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.one } }
		set_temp_variable = { acid_rain_ai_component = CONTROLLER.acid_rain_shelter_tier }
		multiply_temp_variable = { acid_rain_ai_component = constant:acid_rain_mortality.shelter_opening_tier_reduction }
	}
	else = {
		set_temp_variable = { acid_rain_ai_component = CONTROLLER.acid_rain_medical_tier }
		multiply_temp_variable = { acid_rain_ai_component = constant:acid_rain_mortality.medical_sustained_tier_reduction }
	}
	subtract_from_temp_variable = { acid_rain_ai_share = acid_rain_ai_component }
	if = {
		limit = {
			check_variable = { acid_rain_ai_class = constant:acid_rain_mortality_class.ordinary }
			check_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero }
			check_variable = { acid_rain_ai_known_ordinary_episode = constant:acid_rain_runtime.one }
		}
		set_temp_variable = { acid_rain_ai_exposure_days = acid_rain_window_days }
		if = {
			limit = { has_state_flag = acid_rain_consequence_interval_active has_variable = acid_rain_consequence_clock_day }
			set_temp_variable = { acid_rain_ai_elapsed_days = acid_rain_ai_candidate_day }
			subtract_from_temp_variable = { acid_rain_ai_elapsed_days = acid_rain_consequence_clock_day }
			multiply_temp_variable = { acid_rain_ai_elapsed_days = constant:chaosx_date_units.hours_per_date_unit }
			divide_temp_variable = { acid_rain_ai_elapsed_days = constant:chaosx_date_units.hours_per_day }
			clamp_temp_variable = { var = acid_rain_ai_elapsed_days min = constant:acid_rain_runtime.zero }
			add_to_temp_variable = { acid_rain_ai_exposure_days = acid_rain_ai_elapsed_days }
		}
		if = {
			limit = { check_variable = { var = acid_rain_ai_exposure_days value = constant:acid_rain_aftermath.exposure_block_days compare = greater_than_or_equals } }
			set_temp_variable = { acid_rain_ai_component = CONTROLLER.acid_rain_water_tier }
			multiply_temp_variable = { acid_rain_ai_component = constant:acid_rain_mortality.water_sustained_tier_reduction }
			subtract_from_temp_variable = { acid_rain_ai_share = acid_rain_ai_component }
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = { acid_rain_ai_class = constant:acid_rain_mortality_class.severe }
				check_variable = { acid_rain_ai_class = constant:acid_rain_mortality_class.superstorm }
			}
		}
		set_temp_variable = { acid_rain_ai_component = CONTROLLER.acid_rain_transport_tier }
		multiply_temp_variable = { acid_rain_ai_component = constant:acid_rain_mortality.transport_response_tier_reduction }
		subtract_from_temp_variable = { acid_rain_ai_share = acid_rain_ai_component }
		if = { limit = { has_state_flag = acid_rain_evacuated } subtract_from_temp_variable = { acid_rain_ai_share = constant:acid_rain_mortality.evacuation_protection } }
	}
	# Current CBRN can change before readiness. It cannot prove future zero gain.
	clamp_temp_variable = { var = acid_rain_ai_share min = acid_rain_ai_floor max = constant:acid_rain_mortality.one }
	set_temp_variable = { acid_rain_ai_after_share = acid_rain_ai_share }
	set_temp_variable = { acid_rain_ai_added = constant:acid_rain_runtime.zero }
	if = { limit = { check_variable = { acid_rain_ai_row = constant:acid_rain_action.urgent_shelter } } set_temp_variable = { acid_rain_ai_added = constant:acid_rain_response.shelter_protection } }
	else_if = {
		limit = { OR = { check_variable = { acid_rain_ai_row = constant:acid_rain_action.urgent_water_land } check_variable = { acid_rain_ai_row = constant:acid_rain_action.urgent_water_island } } }
		set_temp_variable = { acid_rain_ai_added = constant:acid_rain_response.water_protection }
	}
	else_if = { limit = { check_variable = { acid_rain_ai_row = constant:acid_rain_action.urgent_medical } } set_temp_variable = { acid_rain_ai_added = constant:acid_rain_response.medical_protection } }
	else = {
		set_temp_variable = { acid_rain_ai_added = constant:acid_rain_mortality.evacuation_protection }
		if = {
			limit = { check_variable = { acid_rain_ai_candidate_day < acid_rain_ai_full_day } }
			multiply_temp_variable = { acid_rain_ai_added = constant:acid_rain_mortality.evacuation_partial_share }
		}
	}
	subtract_from_temp_variable = { acid_rain_ai_after_share = acid_rain_ai_added }
	clamp_temp_variable = { var = acid_rain_ai_after_share min = acid_rain_ai_floor max = constant:acid_rain_mortality.one }
	check_variable = { acid_rain_ai_after_share < acid_rain_ai_share }
	acid_rain_ai_whole_request_may_improve = yes
}

# STATE. Source-specific fields above are initialized before this nested call.
# Candidate is always initialized in the caller, including active/opening cases.
acid_rain_ai_forecast_window_may_help = {
	check_variable = { acid_rain_ai_source_start > constant:acid_rain_runtime.zero }
	check_variable = { acid_rain_ai_source_end > acid_rain_ai_source_start }
	check_variable = { acid_rain_ai_source_end > global.date }
	OR = { check_variable = { acid_rain_ai_source_status = constant:acid_rain_runtime.zero } check_variable = { acid_rain_ai_source_status = constant:acid_rain_runtime.one } }
	if = {
		limit = { check_variable = { acid_rain_ai_is_evacuation = constant:acid_rain_runtime.one } }
		check_variable = { acid_rain_ai_source_status = constant:acid_rain_runtime.zero }
		check_variable = { acid_rain_ai_source_start > global.date }
		NOT = { has_state_flag = acid_rain_action_lock }
		NOT = { has_state_flag = acid_rain_evacuated }
	}
	set_temp_variable = { acid_rain_ai_known_ordinary_episode = constant:acid_rain_runtime.zero }
	if = {
		limit = { check_variable = { acid_rain_ai_source_status = constant:acid_rain_runtime.one } check_variable = { acid_rain_ai_recorded > constant:acid_rain_runtime.zero } }
		check_variable = { var = acid_rain_ai_source_last value = acid_rain_ai_source_start compare = greater_than_or_equals }
		check_variable = { var = acid_rain_ai_source_last value = global.date compare = less_than_or_equals }
		check_variable = { acid_rain_ai_source_last < acid_rain_ai_source_end }
		set_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_source_last }
		add_to_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_cadence_offset }
		set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero }
	}
	else = {
		# Earliest legal opening, not a synthetic stored clock or catch-up pulse.
		set_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_source_start }
		clamp_temp_variable = { var = acid_rain_ai_candidate_day min = global.date }
		set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.one }
	}
	set_temp_variable = { acid_rain_ai_first_day = acid_rain_ai_candidate_day }
	set_temp_variable = { acid_rain_ai_seek_day = acid_rain_ai_ready_day }
	acid_rain_ai_advance_candidate = yes
	if = { limit = { check_variable = { acid_rain_ai_candidate_day > acid_rain_ai_first_day } } set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero } }
	check_variable = { acid_rain_ai_candidate_day < acid_rain_ai_source_end }
	OR = {
		acid_rain_ai_candidate_may_help = yes
		AND = {
			# An opening at its floor does not prove later sustained pulses useless.
			check_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.one }
			add_to_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_cadence_offset }
			set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero }
			check_variable = { acid_rain_ai_candidate_day < acid_rain_ai_source_end }
			acid_rain_ai_candidate_may_help = yes
		}
		AND = {
			# Partial rounding can hide a later full evacuation improvement.
			check_variable = { acid_rain_ai_is_evacuation = constant:acid_rain_runtime.one }
			set_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_first_day }
			set_temp_variable = { acid_rain_ai_seek_day = acid_rain_ai_full_day }
			acid_rain_ai_advance_candidate = yes
			set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero }
			if = { limit = { check_variable = { acid_rain_ai_candidate_day = acid_rain_ai_first_day } } set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.one } }
			check_variable = { acid_rain_ai_candidate_day < acid_rain_ai_source_end }
			acid_rain_ai_candidate_may_help = yes
		}
	}
}

# STATE. Latest possible global pulse derives from the current queued check and
# the authored eight-check ladder, aligned to the existing three-day clock.
# It is not a fabricated fixed global lifetime or a future success guarantee.
acid_rain_ai_global_window_is_useful = {
	set_temp_variable = { acid_rain_ai_fresh_cap = constant:acid_rain_runtime.zero }
	if = {
		limit = { check_variable = { global.acid_rain_phase = constant:acid_rain_phase.global } }
		acid_rain_state_has_global_layer = yes
		has_variable = acid_rain_last_pulse_day
		check_variable = { acid_rain_last_pulse_day > constant:acid_rain_runtime.zero }
		check_variable = { var = acid_rain_last_pulse_day value = global.date compare = less_than_or_equals }
		has_variable = acid_rain_global_cap_generation
		check_variable = { var = acid_rain_global_cap_generation value = global.acid_rain_generation_id compare = equals }
		has_variable = global.acid_rain_next_dissipation_check_day
		has_variable = global.acid_rain_dissipation_check_index
		check_variable = { var = global.acid_rain_dissipation_check_index value = constant:acid_rain_runtime.zero compare = greater_than_or_equals }
		check_variable = { global.acid_rain_dissipation_check_index < constant:acid_rain_dissipation.global_step_count }
		set_temp_variable = { acid_rain_ai_global_anchor = acid_rain_last_pulse_day }
		set_temp_variable = { acid_rain_ai_guarantee_day = global.acid_rain_next_dissipation_check_day }
		set_temp_variable = { acid_rain_ai_remaining_checks = constant:acid_rain_dissipation.global_step_count }
		subtract_from_temp_variable = { acid_rain_ai_remaining_checks = global.acid_rain_dissipation_check_index }
		subtract_from_temp_variable = { acid_rain_ai_remaining_checks = constant:acid_rain_runtime.one }
	}
	else = {
		check_variable = { global.acid_rain_phase = constant:acid_rain_phase.global_warning }
		has_global_flag = acid_rain_global_warning_pending
		has_variable = global.acid_rain_global_warning_day
		check_variable = { global.acid_rain_global_warning_day > constant:acid_rain_runtime.zero }
		acid_rain_global_evolution_is_not_disabled = yes
		set_temp_variable = { acid_rain_ai_global_anchor = global.acid_rain_global_warning_day }
		set_temp_variable = { acid_rain_ai_guarantee_day = constant:acid_rain_evolution.global_minimum_days }
		multiply_temp_variable = { acid_rain_ai_guarantee_day = constant:chaosx_date_units.hours_per_day }
		divide_temp_variable = { acid_rain_ai_guarantee_day = constant:chaosx_date_units.hours_per_date_unit }
		add_to_temp_variable = { acid_rain_ai_guarantee_day = acid_rain_ai_global_anchor }
		set_temp_variable = { acid_rain_ai_remaining_checks = constant:acid_rain_dissipation.global_step_count }
		subtract_from_temp_variable = { acid_rain_ai_remaining_checks = constant:acid_rain_runtime.one }
		set_temp_variable = { acid_rain_ai_fresh_cap = constant:acid_rain_runtime.one }
	}
	check_variable = { acid_rain_ai_global_anchor > constant:acid_rain_runtime.zero }
	check_variable = { acid_rain_ai_guarantee_day > acid_rain_ai_global_anchor }
	set_temp_variable = { acid_rain_ai_check_hours = acid_rain_ai_guarantee_day }
	subtract_from_temp_variable = { acid_rain_ai_check_hours = acid_rain_ai_global_anchor }
	multiply_temp_variable = { acid_rain_ai_check_hours = constant:chaosx_date_units.hours_per_date_unit }
	round_temp_variable = acid_rain_ai_check_hours
	set_temp_variable = { acid_rain_ai_check_remainder = acid_rain_ai_check_hours }
	modulo_temp_variable = { var = acid_rain_ai_check_remainder value = acid_rain_ai_cadence_hours }
	if = {
		limit = { check_variable = { acid_rain_ai_check_remainder > constant:acid_rain_runtime.zero } }
		subtract_from_temp_variable = { acid_rain_ai_check_hours = acid_rain_ai_check_remainder }
		add_to_temp_variable = { acid_rain_ai_check_hours = acid_rain_ai_cadence_hours }
	}
	divide_temp_variable = { acid_rain_ai_check_hours = constant:chaosx_date_units.hours_per_date_unit }
	set_temp_variable = { acid_rain_ai_guarantee_day = acid_rain_ai_global_anchor }
	add_to_temp_variable = { acid_rain_ai_guarantee_day = acid_rain_ai_check_hours }
	set_temp_variable = { acid_rain_ai_check_spacing_hours = constant:acid_rain_dissipation.check_interval_days }
	multiply_temp_variable = { acid_rain_ai_check_spacing_hours = constant:chaosx_date_units.hours_per_day }
	set_temp_variable = { acid_rain_ai_check_remainder = acid_rain_ai_check_spacing_hours }
	modulo_temp_variable = { var = acid_rain_ai_check_remainder value = acid_rain_ai_cadence_hours }
	if = {
		limit = { check_variable = { acid_rain_ai_check_remainder > constant:acid_rain_runtime.zero } }
		subtract_from_temp_variable = { acid_rain_ai_check_spacing_hours = acid_rain_ai_check_remainder }
		add_to_temp_variable = { acid_rain_ai_check_spacing_hours = acid_rain_ai_cadence_hours }
	}
	multiply_temp_variable = { acid_rain_ai_check_spacing_hours = acid_rain_ai_remaining_checks }
	divide_temp_variable = { acid_rain_ai_check_spacing_hours = constant:chaosx_date_units.hours_per_date_unit }
	add_to_temp_variable = { acid_rain_ai_guarantee_day = acid_rain_ai_check_spacing_hours }
	check_variable = { var = acid_rain_ai_guarantee_day value = global.date compare = greater_than_or_equals }
	set_temp_variable = { acid_rain_ai_class = constant:acid_rain_mortality_class.global }
	set_temp_variable = { acid_rain_ai_intensity = constant:acid_rain_intensity.standard }
	set_temp_variable = { acid_rain_ai_floor = constant:acid_rain_mortality.ordinary_protection_floor }
	set_temp_variable = { acid_rain_ai_cap_fraction = constant:acid_rain_mortality.global_episode_cap }
	set_temp_variable = { acid_rain_ai_known_ordinary_episode = constant:acid_rain_runtime.zero }
	set_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_global_anchor }
	add_to_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_cadence_offset }
	set_temp_variable = { acid_rain_ai_first_day = acid_rain_ai_candidate_day }
	set_temp_variable = { acid_rain_ai_seek_day = acid_rain_ai_ready_day }
	acid_rain_ai_advance_candidate = yes
	check_variable = { var = acid_rain_ai_candidate_day value = acid_rain_ai_guarantee_day compare = less_than_or_equals }
	set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero }
	if = { limit = { NOT = { has_state_flag = acid_rain_global_opening_pulse_recorded } check_variable = { acid_rain_ai_candidate_day = acid_rain_ai_first_day } } set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.one } }
	OR = {
		acid_rain_ai_candidate_may_help = yes
		AND = {
			check_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.one }
			add_to_temp_variable = { acid_rain_ai_candidate_day = acid_rain_ai_cadence_offset }
			set_temp_variable = { acid_rain_ai_opening = constant:acid_rain_runtime.zero }
			check_variable = { var = acid_rain_ai_candidate_day value = acid_rain_ai_guarantee_day compare = less_than_or_equals }
			acid_rain_ai_candidate_may_help = yes
		}
	}
}

	multiply_temp_variable = { acid_rain_ai_full_day = constant:chaosx_date_units.hours_per_day }
	divide_temp_variable = { acid_rain_ai_full_day = constant:chaosx_date_units.hours_per_date_unit }

	set_temp_variable = { acid_rain_ai_ready_day = acid_rain_ai_full_day }
	set_temp_variable = { acid_rain_ai_benefit_end_day = acid_rain_ai_effect_duration }
	multiply_temp_variable = { acid_rain_ai_benefit_end_day = constant:chaosx_date_units.hours_per_day }
	divide_temp_variable = { acid_rain_ai_benefit_end_day = constant:chaosx_date_units.hours_per_date_unit }

	multiply_temp_variable = { acid_rain_ai_cadence_hours = constant:chaosx_date_units.hours_per_day }
	set_temp_variable = { acid_rain_ai_cadence_offset = acid_rain_ai_cadence_hours }
	divide_temp_variable = { acid_rain_ai_cadence_offset = constant:chaosx_date_units.hours_per_date_unit }

		if = {
			limit = { check_variable = { acid_rain_ai_use_saved_target = constant:acid_rain_runtime.one } }

			var:acid_rain_selected_action_state = { acid_rain_ai_state_has_useful_window = yes }
		}
		else = { FROM = { acid_rain_ai_state_has_useful_window = yes } }
	}
	else = { any_controlled_state = { acid_rain_ai_state_has_useful_window = yes } }
