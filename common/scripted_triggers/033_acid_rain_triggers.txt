acid_rain_state_is_excluded = {
	OR = {
		has_state_category = wasteland
		has_state_category = enclave
		has_state_flag = acid_rain_shared_civilian_loss_excluded
		has_state_flag = fallout_shared_civilian_loss_excluded
		has_state_flag = air_winter_excluded
	}
}

acid_rain_state_has_supported_region = {
	OR = {
		is_on_continent = europe
		is_on_continent = middle_east
		is_on_continent = asia
		is_on_continent = africa
		is_on_continent = australia
		is_on_continent = north_america
		is_on_continent = south_america
	}
}

acid_rain_state_is_valid_target = {
	scope_exists = yes
	impassable = no
	NOT = { acid_rain_state_is_excluded = yes }
	acid_rain_state_has_supported_region = yes
	check_variable = { state_population_k > constant:acid_rain_mortality.minimum_population_k }
	OWNER = {
		exists = yes
		is_actual_nonhuman_country = no
	}
	CONTROLLER = {
		exists = yes
		is_actual_nonhuman_country = no
	}
}

# STATE SCOPE. True while this state's controller runs an evacuation here
# that started in the current generation and has run at least one full day.
acid_rain_state_has_partial_evacuation = {
	has_state_flag = acid_rain_action_lock
	has_variable = acid_rain_action_type
	OR = {
		check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
		check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
	}
	has_variable = acid_rain_action_owner_generation
	check_variable = { var = acid_rain_action_owner_generation value = global.acid_rain_generation_id compare = equals }
	has_variable = acid_rain_action_owner_country_id
	check_variable = { var = acid_rain_action_owner_country_id value = CONTROLLER.id compare = equals }
	has_variable = acid_rain_evacuation_partial_day
	check_variable = { var = global.date value = acid_rain_evacuation_partial_day compare = greater_than_or_equals }
}

	hidden_trigger = {

		check_variable = { var = acid_rain_pulse_key value = constant:acid_rain_aftermath.zero compare = greater_than }

			AND = { check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.ordinary compare = equals } acid_rain_state_has_ordinary_layer = yes }
			AND = { check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.severe compare = equals } acid_rain_state_has_severe_layer = yes }
			AND = { check_variable = { var = acid_rain_pulse_class value = constant:acid_rain_mortality_class.global compare = equals } acid_rain_state_has_global_layer = yes }

# A later generation may begin only after every paid commitment and state-level
# recovery obligation from the resolved generation has settled. This keeps the
# reset path idempotent without cancelling factories, manpower, or target proofs.
acid_rain_previous_generation_is_settled = {
	OR = {
		NOT = { has_global_flag = acid_rain_runtime_resolved }
		AND = {
			has_global_flag = acid_rain_runtime_resolved
			all_of_scopes = {
				array = global.acid_rain_participant_countries
				NOT = { check_variable = { acid_rain_active_projects > constant:acid_rain_runtime.zero } }
				NOT = { has_country_flag = acid_rain_national_action_active }
				NOT = { has_country_flag = acid_rain_recovery_tail_active }
			}
			all_of_scopes = {
				array = global.acid_rain_frozen_states
				NOT = { has_state_flag = acid_rain_aftercare_unresolved }
				NOT = { has_state_flag = acid_rain_action_lock }
			}
		}
	}
}

# COUNTRY SCOPE. The caller computes these exact pre-spend requirements from
# the currently loaded quote. Meta trigger injection is needed because stockpile
# triggers do not accept variable right-hand values directly.
acid_rain_country_ai_reserves_allow_loaded_quote = {
	set_temp_variable = { acid_rain_motorized_available = constant:acid_rain_runtime.zero }
	acid_rain_prepare_motorized_stock = yes
	meta_trigger = {
		text = {
			NOT = { has_equipment = { support_equipment < [SUPPORT_REQUIRED] } }
			NOT = { has_equipment = { train_equipment < [TRAIN_REQUIRED] } }
			NOT = { has_equipment = { convoy_1 < [CONVOY_REQUIRED] } }
			check_variable = { var = acid_rain_motorized_available value = [MOTORIZED_REQUIRED] compare = greater_than_or_equals }
			NOT = { has_manpower < [MANPOWER_REQUIRED] }
			NOT = { has_fuel < [FUEL_REQUIRED] }
			num_of_civilian_factories_available_for_projects > [CIV_TRIGGER_FLOOR]
		}
		SUPPORT_REQUIRED = "[?acid_rain_ai_required_support|.0]"
		TRAIN_REQUIRED = "[?acid_rain_ai_required_train|.0]"
		CONVOY_REQUIRED = "[?acid_rain_ai_required_convoy|.0]"
		MOTORIZED_REQUIRED = "[?acid_rain_ai_required_motorized|.0]"
		MANPOWER_REQUIRED = "[?acid_rain_ai_required_manpower|.0]"
		FUEL_REQUIRED = "[?acid_rain_ai_required_fuel|.0]"
		CIV_TRIGGER_FLOOR = "[?acid_rain_ai_required_civ_trigger_floor|.0]"
	}
}

# COUNTRY SCOPE. The family or cap of acid_rain_selected_action has room.
acid_rain_selected_operation_can_open = {
	if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_shelter compare = equals } }
		acid_rain_shelter_operation_available = yes
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_water_land compare = equals }
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_water_island compare = equals }
			}
		}
		acid_rain_water_operation_available = yes
	}
	else_if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_medical compare = equals } }
		acid_rain_medical_operation_available = yes
	}
	else_if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_transport compare = equals } }
		acid_rain_transport_operation_available = yes
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
			}
		}
		acid_rain_country_can_open_evacuation = yes
	}
	else_if = {
		limit = {
			check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_transport compare = greater_than_or_equals }
			check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_industry compare = less_than_or_equals }
		}
		acid_rain_country_can_open_recovery = yes
	}
	else_if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_demobilize compare = equals } }
		acid_rain_demobilize_operation_available = yes
	}
	else = { always = no }
}

# COUNTRY SCOPE, ROOT is the acting country. The state stored in
# acid_rain_selected_action_state suits acid_rain_selected_action.
acid_rain_selected_state_accepts_action = {
	has_variable = acid_rain_selected_action_state
	if = {
		limit = {
			OR = {
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
				check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
			}
		}
		var:acid_rain_selected_action_state = { acid_rain_state_is_evacuation_target = yes }
	}
	else_if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_transport compare = equals } }
		var:acid_rain_selected_action_state = { acid_rain_state_is_transport_recovery_target = yes }
	}
	else_if = {
		limit = { check_variable = { var = acid_rain_selected_action value = constant:acid_rain_action.recovery_industry compare = equals } }
		var:acid_rain_selected_action_state = { acid_rain_state_is_industry_recovery_target = yes }
	}
	else = {
		var:acid_rain_selected_action_state = { acid_rain_state_is_decontamination_target = yes }
	}
}

# STATE SCOPE. The operation running on this state can still complete.
acid_rain_state_operation_still_valid = {

	has_state_flag = acid_rain_action_lock
	check_variable = { var = acid_rain_action_owner_generation value = global.acid_rain_generation_id compare = equals }
	check_variable = { var = acid_rain_action_owner_country_id value = CONTROLLER.id compare = equals }
	OR = {
		AND = {
			OR = {
				check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.urgent_evacuate_land compare = equals }
				check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.urgent_evacuate_island compare = equals }
			}
			OR = {
				has_state_flag = acid_rain_severe_cell_target
				has_state_flag = acid_rain_superstorm_target
			}
		}
		AND = {
			check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.recovery_transport compare = greater_than_or_equals }
			check_variable = { var = acid_rain_action_type value = constant:acid_rain_action.recovery_industry compare = less_than_or_equals }
			has_state_flag = acid_rain_aftercare_unresolved
		}
	}
}
