# ============================================================================
# CHAOS REDUX - EVENT 033 ACID RAIN ACHIEVEMENT TRIGGERS
# ============================================================================
# These triggers read only the generation-scoped achievement ledger. Runtime
# modifiers, current ownership, and UI state are not completion evidence.
# Completion effects set one permanent receipt flag at a bounded runtime edge;
# the achievement definitions read the corresponding *_is_complete trigger.
# ============================================================================

# The input form is used before a country has received its first achievement
# record. It deliberately does not accept a missing or stale generation.
acid_rain_033_achievement_input_generation_is_current = {
	has_variable = acid_rain_033_input_generation
	has_variable = global.acid_rain_generation_id
	check_variable = { var = acid_rain_033_input_generation value = global.acid_rain_generation_id compare = equals }
}

acid_rain_033_achievement_generation_is_current = {
	has_country_flag = acid_rain_033_achievement_tracking_initialized
	has_variable = acid_rain_033_achievement_generation_id
	has_variable = global.acid_rain_generation_id
	check_variable = { var = acid_rain_033_achievement_generation_id value = global.acid_rain_generation_id compare = equals }
}
