# CBRN occupation coverup refund preflight. COUNTRY payer with FROM STATE.
# Pure normal-state read. Only temporary target and receipt-ID outputs change.
# See the matching markdown for caller-owned output storage and failure rules.
cbrn_occupation_coverup_can_retry_refund = {
	set_temp_variable = { cbrn_occupation_coverup_target = FROM }
	set_temp_variable = { cbrn_occupation_coverup_receipt_id = constant:universal_cost_framework.zero }
	check_variable = { cbrn_occupation_coverup_open_receipt_@var:cbrn_occupation_coverup_target > constant:universal_cost_framework.zero }
	set_temp_variable = { cbrn_occupation_coverup_receipt_id = cbrn_occupation_coverup_open_receipt_@var:cbrn_occupation_coverup_target }
	has_variable = cbrn_occupation_coverup_receipt_target_@var:cbrn_occupation_coverup_receipt_id
	has_variable = cbrn_occupation_coverup_receipt_paid_@var:cbrn_occupation_coverup_receipt_id
	has_variable = cbrn_occupation_coverup_receipt_phase_@var:cbrn_occupation_coverup_receipt_id
	check_variable = { cbrn_occupation_coverup_receipt_target_@var:cbrn_occupation_coverup_receipt_id = cbrn_occupation_coverup_target }
	check_variable = { cbrn_occupation_coverup_receipt_paid_@var:cbrn_occupation_coverup_receipt_id > constant:universal_cost_framework.zero }
	OR = {
		check_variable = { cbrn_occupation_coverup_receipt_phase_@var:cbrn_occupation_coverup_receipt_id = constant:cbrn_occupation_coverup_receipt_phase.unrecorded_refund }
		check_variable = { cbrn_occupation_coverup_receipt_phase_@var:cbrn_occupation_coverup_receipt_id = constant:cbrn_occupation_coverup_receipt_phase.recorded_refund }
	}
}

# CBRN occupation coverup payment recovery. COUNTRY payer, no FROM required.
# True while any paid cover-up receipt still waits for its refund or settlement.
# The recovery decision uses it so a payer whose every cover-up target is blocked
# can still retry the refund without a new purchase.
cbrn_occupation_coverup_has_pending_receipt = {
	check_variable = { cbrn_occupation_coverup_pending_receipts^num > constant:universal_cost_framework.zero }
}
