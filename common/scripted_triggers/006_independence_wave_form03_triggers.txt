# ============================================================================
# CHAOS REDUX - EVENT 006 FORM-03 LOW COUNTRIES TRIGGERS
# ============================================================================
# Overview:
# - Binds FORM-03 exclusively to the AFX or AGX Event 006 carrier packages.
# - Requires a second consenting core delegation and a verified border,
#   faction, access, guarantee, or non-aggression connection.
# - Restricts the carrier to the accepted constitutional or Development
#   Compact route family.
# - Admits only living BEL with the BEL_flanders identity to the founding
#   congress; HOL and LUX remain post-charter autonomous associates.
# - Never authorizes state 6, 7, 8, 35, 977, or 980 for transfer or coring.
# ============================================================================

is_independence_wave_form03_registered_carrier = {
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_formable_family_selected
	check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.low_countries_federation }
	has_country_flag = independence_wave_low_countries_federation_candidate
	OR = {
		AND = {
			original_tag = AFX
			check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_006 }
		}
		AND = {
			original_tag = AGX
			check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_007 }
		}
	}
}

has_independence_wave_form03_exact_carrier_anchor = {
	is_independence_wave_form03_registered_carrier = yes
	has_variable = independence_wave_anchor_state
	OR = {
		AND = {
			original_tag = AFX
			var:independence_wave_anchor_state = {
				state = 34
				is_owned_and_controlled_by = PREV
			}
			capital_scope = { state = 34 }
		}
		AND = {
			original_tag = AGX
			var:independence_wave_anchor_state = {
				state = 36
				is_owned_and_controlled_by = PREV
			}
			capital_scope = { state = 36 }
		}
	}
}

has_independence_wave_form03_belgian_declared_consent = {
	is_independence_wave_form03_eligible_belgian_delegation = yes
	has_independence_wave_form03_declared_consent = yes
}

# Country scope. ROOT is the prospective carrier. This verifies an actual
# controlled border or a bilateral treaty link; mere regional proximity is not
# enough to move an integrated anchor or admit an autonomous member.
has_independence_wave_form03_connection_to_root = {
	exists = yes
	NOT = { has_war_with = ROOT }
	OR = {
		tag = ROOT
		is_neighbor_of = ROOT
		is_in_faction_with = ROOT
		has_non_aggression_pact_with = ROOT
		has_military_access_to = ROOT
		gives_military_access_to = ROOT
		has_guaranteed = ROOT
		is_guaranteed_by = ROOT
	}
}

has_independence_wave_form03_connection_to_inviting_carrier = {
	has_variable = independence_wave_formable_inviting_carrier
	NOT = { has_war_with = var:independence_wave_formable_inviting_carrier }
	OR = {
		is_neighbor_of = var:independence_wave_formable_inviting_carrier
		is_in_faction_with = var:independence_wave_formable_inviting_carrier
		has_non_aggression_pact_with = var:independence_wave_formable_inviting_carrier
		has_military_access_to = var:independence_wave_formable_inviting_carrier
		gives_military_access_to = var:independence_wave_formable_inviting_carrier
		has_guaranteed = var:independence_wave_formable_inviting_carrier
		is_guaranteed_by = var:independence_wave_formable_inviting_carrier
	}
}

# State scope. The exact owner and controller checks make the project legal in
# sovereign territory without permitting state 980 or any generic target.
is_independence_wave_form03_associate_corridor_state = {
	OR = {
		AND = {
			OR = { state = 6 state = 977 }
			owner = { exists = yes original_tag = BEL has_country_flag = independence_wave_form03_autonomous_member }
			controller = { original_tag = BEL }
		}
		AND = {
			OR = { state = 7 state = 35 }
			owner = { exists = yes original_tag = HOL has_country_flag = independence_wave_form03_autonomous_member }
			controller = { original_tag = HOL }
		}
		AND = {
			state = 8
			owner = { exists = yes original_tag = LUX has_country_flag = independence_wave_form03_autonomous_member }
			controller = { original_tag = LUX }
		}
	}
}
