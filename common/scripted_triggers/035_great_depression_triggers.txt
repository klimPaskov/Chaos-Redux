# ============================================================================
# CHAOS REDUX - EVENT 035 GREAT DEPRESSION API AND ACHIEVEMENT TRIGGERS
# ============================================================================
# Event 35 achievement campaign and runtime eligibility. The runtime predicate
# is repeated in the achievement registry because `possible` is cached at
# campaign start while force/test state can change later.
# ============================================================================

great_depression_achievement_possible_campaign_is_valid = {
	exists = yes
	is_ai = no
	is_special_chaos_country = no
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_debug = yes }
	NOT = { has_country_flag = chaosx_test_country_initialized }
}

great_depression_achievement_player_is_valid = {
	great_depression_achievement_possible_campaign_is_valid = yes
	uses_normal_civilian_systems = yes
	NOT = { has_country_flag = force_trigger_mode_enabled }
}

# ============================================================================
# EVENT 035 GREAT DEPRESSION 2.0 START AND DEEPEN API
# ============================================================================
# Country-scope validation for the reusable great_depression_start_or_deepen
# gateway. These predicates are deliberately fail-closed: callers provide
# source, proof, and severity selectors; this file never guesses a country,
# source, state, evolution, or receipt.
# ============================================================================

great_depression_has_viable_economy = {
	exists = yes
	uses_normal_civilian_systems = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	num_of_factories > constant:great_depression_api.zero
	any_controlled_state = {
		state_population_k > constant:great_depression_api.minimum_viable_state_population_k
		OR = {
			industrial_complex > constant:great_depression_api.zero
			arms_factory > constant:great_depression_api.zero
			dockyard > constant:great_depression_api.zero
		}
	}
}

great_depression_has_active_crisis = {
	has_country_flag = great_depression_active
}

great_depression_active_crisis_is_coherent = {
	has_country_flag = great_depression_active
	has_variable = great_depression_active_episode_id
	check_variable = {
		var = great_depression_active_episode_id
		value = constant:great_depression_api.minimum_episode_id
		compare = greater_than_or_equals
	}
	has_variable = great_depression_depression_severity
	check_variable = {
		var = great_depression_depression_severity
		value = constant:great_depression_event.public_value_minimum
		compare = greater_than_or_equals
	}
	check_variable = {
		var = great_depression_depression_severity
		value = constant:great_depression_event.public_value_maximum
		compare = less_than_or_equals
	}
	has_variable = great_depression_evolution_floor
	check_variable = {
		var = great_depression_evolution_floor
		value = constant:great_depression_evolution.baseline
		compare = greater_than_or_equals
	}
	check_variable = {
		var = great_depression_evolution_floor
		value = constant:great_depression_evolution.second_great_depression
		compare = less_than_or_equals
	}
}

great_depression_has_stale_active_registry_entry = {

# relationship helper and remain independent of ordinary random-event pacing.

	NOT = { has_country_flag = industrial_boom_snapshot_frozen }

}

great_depression_contagion_conversion_vulnerability_is_high = {
	great_depression_is_valid_contagion_network_target = yes
	OR = {
		has_stability < constant:great_depression_thresholds.social_peace_stability_floor
		is_subject = yes
		has_war = yes
		has_country_flag = great_depression_worldwide_pressure_active
		has_country_flag = great_depression_recovery_hollow
		has_country_flag = great_depression_supplier_contract_at_risk
		num_of_civilian_factories_available_for_projects < constant:great_depression_scaling.large_economy_civilian_limit
	}
}

	NOT = { has_country_flag = great_depression_active }
	NOT = { has_country_flag = industrial_boom_active }
	NOT = { has_country_flag = industrial_boom_snapshot_frozen }
	NOT = { has_country_flag = great_depression_transition_pending }
