# ============================================================================
# EVENT 035 GREAT DEPRESSION 2.0 START AND DEEPEN API
# ============================================================================
# Country-scope validation for the reusable great_depression_start_or_deepen
# gateway. These predicates are deliberately fail-closed: callers provide
# source, proof, and severity selectors; this file never guesses a country,
# source, state, evolution, or receipt.
# ============================================================================

great_depression_has_viable_economy = {
	exists = yes
	uses_normal_civilian_systems = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	num_of_factories > constant:great_depression_api.zero
	any_controlled_state = {
		state_population_k > constant:great_depression_api.minimum_viable_state_population_k
		OR = {
			industrial_complex > constant:great_depression_api.zero
			arms_factory > constant:great_depression_api.zero
			dockyard > constant:great_depression_api.zero
		}
	}
}

great_depression_has_active_crisis = {
	has_country_flag = great_depression_active
}

# Country preflight. A new crisis needs a real state that the center selector
# can use, not only factories held through occupation or an impassable core.
# This runs before any national setup, opening shock, or source-ledger reset.
great_depression_has_viable_center_pool = {
	any_controlled_state = {
		great_depression_new_center_state_is_permitted = yes
		is_owned_by = PREV
		is_controlled_by = PREV
		is_core_of = PREV
		impassable = no
		state_population_k > constant:great_depression_api.minimum_viable_state_population_k
		OR = {
			industrial_complex > constant:great_depression_api.zero
			arms_factory > constant:great_depression_api.zero
			dockyard > constant:great_depression_api.zero
		}
	}
}

# State directly below the receiving country. A fresh episode cannot overwrite
# an existing physical center or a live/frozen source conversion prohibition.
# Registry membership is checked separately during selection after its reset.
great_depression_new_center_state_is_permitted = {
	NOT = { has_state_flag = great_depression_center }
	NOT = { has_state_flag = industrial_boom_region_conversion_forbidden }
	NOT = {
		AND = {
			PREV = {
				check_variable = { var = great_depression_call_caller_type value = constant:great_depression_caller.industrial_boom_collapse compare = equals }
				has_country_flag = industrial_boom_snapshot_region_conversion_forbidden
				has_variable = industrial_boom_snapshot_region_state_id
			}
			check_variable = { var = id value = PREV.industrial_boom_snapshot_region_state_id compare = equals }
		}
	}
}

great_depression_active_crisis_is_coherent = {
	has_country_flag = great_depression_active
	has_variable = great_depression_active_episode_id
	check_variable = {
		var = great_depression_active_episode_id
		value = constant:great_depression_api.minimum_episode_id
		compare = greater_than_or_equals
	}
	has_variable = great_depression_depression_severity
	check_variable = {
		var = great_depression_depression_severity
		value = constant:great_depression_event.public_value_minimum
		compare = greater_than_or_equals
	}
	check_variable = {
		var = great_depression_depression_severity
		value = constant:great_depression_event.public_value_maximum
		compare = less_than_or_equals
	}
	has_variable = great_depression_evolution_floor
	check_variable = {
		var = great_depression_evolution_floor
		value = constant:great_depression_evolution.baseline
		compare = greater_than_or_equals
	}
	check_variable = {
		var = great_depression_evolution_floor
		value = constant:great_depression_evolution.second_great_depression
		compare = less_than_or_equals
	}
}

# Any scope. Event-pool availability: the random pool and the Negative Economy
# cluster offer Event 35 only while one country could start an independent
# depression, so a firing never ends without a target. The cheap major/player

	any_country = {

			is_major = yes

	}
}

	}

# Recipient country. Prove the aligned incoming and outgoing pair rows against
# the supplied exact source, episode, receipt and qualitative exposure stage.
# any_of does not change scope. The two var: entries do, so PREV.PREV is the
# recipient in the outgoing-target scope even if ROOT is an unrelated sender.
great_depression_call_contagion_rows_are_proven = {
	has_country_flag = great_depression_contagion_exposure_open
	NOT = { has_country_flag = great_depression_contagion_contained }
	has_variable = great_depression_contagion_exposure_start_day
	set_temp_variable = { great_depression_call_proven_exposure_days = global.num_days }
	subtract_from_temp_variable = { great_depression_call_proven_exposure_days = great_depression_contagion_exposure_start_day }
	check_variable = { var = great_depression_call_proven_exposure_days value = constant:great_depression_event.global_pressure_days compare = greater_than_or_equals }
	has_variable = great_depression_contagion_source_countries^num
	check_variable = { var = great_depression_contagion_source_countries^num value = constant:great_depression_event.contagion_source_cap compare = less_than_or_equals }
	any_of = {
		array = great_depression_contagion_source_countries
		value = great_depression_call_validate_source_country
		index = great_depression_call_validate_source_row
		check_variable = { var = great_depression_call_validate_source_row value = great_depression_contagion_source_episode_entries^num compare = less_than }
		check_variable = { var = great_depression_call_validate_source_row value = great_depression_contagion_source_receipt_entries^num compare = less_than }
		check_variable = { var = great_depression_call_validate_source_row value = great_depression_contagion_source_stage_entries^num compare = less_than }
		check_variable = { var = great_depression_contagion_source_episode_entries^great_depression_call_validate_source_row value = great_depression_call_source_episode_id compare = equals }
		check_variable = { var = great_depression_contagion_source_receipt_entries^great_depression_call_validate_source_row value = great_depression_call_receipt_id compare = equals }
		check_variable = { var = great_depression_contagion_source_stage_entries^great_depression_call_validate_source_row value = great_depression_call_contagion_exposure_stage compare = equals }
		var:great_depression_call_validate_source_country = {
			exists = yes
			tag = event_target:great_depression_call_source_country
			NOT = { tag = PREV }
			great_depression_active_crisis_is_coherent = yes
			has_country_flag = great_depression_evolution_financial_contagion
			check_variable = { var = great_depression_active_episode_id value = PREV.great_depression_call_source_episode_id compare = equals }
			check_variable = { var = great_depression_contagion_exposure_targets^num value = constant:great_depression_event.contagion_source_cap compare = less_than_or_equals }
			any_of = {
