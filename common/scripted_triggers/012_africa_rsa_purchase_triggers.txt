# CHAOS REDUX - SOUTH AFRICA PURCHASE QUOTES
# Country scope. africa_rsa_purchase_action selects one existing owner action.
# Quotation initializes every display field on each call. Affordability checks
# the payable amounts and leaves route, target, and objective gates to the owner.

africa_rsa_prepare_purchase_costs = {
	set_temp_variable = { africa_rsa_purchase_quote_valid = constant:universal_cost_framework.one }
	set_temp_variable = { africa_rsa_purchase_pp_ordinary = constant:universal_cost_framework.zero }
	set_temp_variable = { africa_rsa_purchase_cp_ordinary = constant:universal_cost_framework.zero }
	set_temp_variable = { africa_rsa_purchase_trains_ordinary = constant:universal_cost_framework.zero }
	set_temp_variable = { africa_rsa_purchase_support_ordinary = constant:universal_cost_framework.zero }
	set_temp_variable = { africa_rsa_purchase_stability_ordinary = constant:universal_cost_framework.zero }
	if = {
		limit = { check_variable = { var = africa_rsa_purchase_action value = constant:africa_rsa_purchase.relief compare = equals } }
		set_temp_variable = { africa_rsa_purchase_pp_ordinary = constant:africa_rsa_cost.relief_pp }
		set_temp_variable = { africa_rsa_purchase_cp_ordinary = constant:africa_rsa_cost.relief_cp }
		set_temp_variable = { africa_rsa_purchase_trains_ordinary = constant:africa_rsa_cost.relief_trains }
		set_temp_variable = { africa_rsa_purchase_support_ordinary = constant:africa_rsa_cost.relief_support_equipment }
	}
	else_if = {
		limit = { check_variable = { var = africa_rsa_purchase_action value = constant:africa_rsa_purchase.regional_request compare = equals } }
		set_temp_variable = { africa_rsa_purchase_pp_ordinary = constant:africa_rsa_cost.regional_request_pp }
	}
	else_if = {
		limit = { check_variable = { var = africa_rsa_purchase_action value = constant:africa_rsa_purchase.citizenship compare = equals } }
		set_temp_variable = { africa_rsa_purchase_pp_ordinary = constant:africa_rsa_cost.citizenship_pp }
		set_temp_variable = { africa_rsa_purchase_stability_ordinary = constant:africa_rsa_cost.citizenship_stability }
	}
	else_if = {
		limit = { check_variable = { var = africa_rsa_purchase_action value = constant:africa_rsa_purchase.sovereignty compare = equals } }
		set_temp_variable = { africa_rsa_purchase_pp_ordinary = constant:africa_rsa_cost.sovereignty_guarantee_pp }
		set_temp_variable = { africa_rsa_purchase_cp_ordinary = constant:africa_rsa_cost.sovereignty_guarantee_cp }
	}
	else_if = {
		limit = { check_variable = { var = africa_rsa_purchase_action value = constant:africa_rsa_purchase.exile_recovery compare = equals } }
		set_temp_variable = { africa_rsa_purchase_pp_ordinary = constant:africa_rsa_cost.exile_recovery_pp }
		set_temp_variable = { africa_rsa_purchase_cp_ordinary = constant:africa_rsa_cost.exile_recovery_cp }
	}
	else = { set_temp_variable = { africa_rsa_purchase_quote_valid = constant:universal_cost_framework.zero } }
	always = yes
}

# Inputs: africa_rsa_component_ordinary and africa_rsa_component_family.
# Output: africa_rsa_component_quote. Invalid active quotations reject payment.
africa_rsa_quote_purchase_component = {
	set_temp_variable = { africa_rsa_component_quote = africa_rsa_component_ordinary }
	if = {
		limit = {
			black_friday_is_active = yes
			check_variable = { africa_rsa_component_ordinary > constant:universal_cost_framework.zero }
		}
		set_temp_variable = { universal_cost_source_query_id = constant:universal_cost_framework.black_friday_source_id }
		set_temp_variable = { black_friday_quote_final_cost = constant:universal_cost_framework.zero }
		if = {
			limit = {
				universal_cost_source_is_active = yes
				set_temp_variable = { black_friday_quote_ordinary_cost = africa_rsa_component_ordinary }
				set_temp_variable = { black_friday_quote_family_mask = africa_rsa_component_family }
				black_friday_quote_custom_cost = yes
			}
			set_temp_variable = { africa_rsa_component_quote = black_friday_quote_final_cost }
		}
		else = { set_temp_variable = { africa_rsa_purchase_quote_valid = constant:universal_cost_framework.zero } }
	}
	always = yes
}
