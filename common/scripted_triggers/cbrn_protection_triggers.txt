# ============================================================================
# CHAOS REDUX - CBRN PROTECTIVE-EQUIPMENT AND CIVIL-DEFENCE TRIGGERS
# ============================================================================
# Overview:
# - Gates national respirator programs, field issue, state distribution,
#   procurement, allied transfers, filter service, and reconditioning.
# - Requires verified alerts for pre-raid responses; hospital care also serves
#   controlled states with actual chemical contamination.
# - Reads real equipment and factory capacity. It never infers air activity or
#   creates contamination from aircraft presence.
# ============================================================================

# any_province_building_level needs a building bound. Bunker levels are never
# negative, so this bound admits every province the filter selects.
@CBRN_PROTECTION_ANY_BUNKER_LEVEL = -1

cbrn_protection_program_category_visible = {
	NOT = { is_actual_nonhuman_country = yes }
	OR = {
		cbrn_protection_decision_surface_is_unlocked = yes
		cbrn_country_has_public_chemical_emergency = yes
		has_country_flag = cbrn_allied_protection_request_received
		cbrn_country_has_mask_replacement_backlog = yes
	}
}

cbrn_civil_defence_category_visible = {
	NOT = { is_actual_nonhuman_country = yes }
	OR = {
		cbrn_country_has_current_chemical_need = yes
		has_active_biowarfare_outbreak = yes
		has_country_flag = cbrn_civilian_distribution_program_active
	}
}

# Current civil protection is distinct from durable attack history and a
# retaliation permission. These pure predicates never clear either record.

# Pure state-scope quote for the two timed occupied-population distributors.
# ROOT is the paying country. Temporary output is the rounded state need;
# callers sum state needs and cap the total against real model stock.
cbrn_quote_occupied_state_mask_reservation = {
	set_temp_variable = { cbrn_reservation_quote_state_need = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_reservation_quote_target = constant:cbrn_protection_program_effect.occupied_distribution_target }
	set_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_protection_program_effect.occupied_distribution_cost_mult }
	set_temp_variable = { cbrn_reservation_quote_effectiveness = constant:cbrn_math.one }
	set_temp_variable = { cbrn_reservation_quote_new_filter = constant:cbrn_mask_filters.condition_full }
	cbrn_quote_state_mask_reservation_internal = yes
}

cbrn_quote_external_state_mask_reservation = {
	set_temp_variable = { cbrn_reservation_quote_state_need = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_mask_reservation_available_stock = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_mask_reservation_quote = constant:cbrn_math.zero }
	cbrn_quote_occupied_state_mask_reservation = yes
	ROOT = { cbrn_quote_available_mask_stock_for_reservation = yes }
	set_temp_variable = { cbrn_mask_reservation_quote = cbrn_reservation_quote_state_need }
	clamp_temp_variable = { var = cbrn_mask_reservation_quote min = constant:cbrn_math.zero max = cbrn_mask_reservation_available_stock }
}

# State scope; the same live coverage and delivery calculation as occupied
# distribution, with the emergency order's target, wastage, and filter quality.
# The uncapped state need drives target visibility; the stock-capped quote is
# the actual one-click respirator payment shown in the cost row.
cbrn_quote_emergency_state_mask_reservation = {
	set_temp_variable = { cbrn_reservation_quote_state_need = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_mask_reservation_available_stock = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_mask_reservation_quote = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_reservation_quote_target = constant:cbrn_mask_distribution.emergency_target }
	set_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.emergency_cost_mult }
	multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.emergency_wastage_mult }
	set_temp_variable = { cbrn_reservation_quote_effectiveness = constant:cbrn_mask_distribution.emergency_effectiveness_mult }
	set_temp_variable = { cbrn_reservation_quote_new_filter = constant:cbrn_mask_filters.emergency_condition }
	cbrn_quote_state_mask_reservation_internal = yes
	ROOT = { cbrn_quote_available_mask_stock_for_reservation = yes }
	set_temp_variable = { cbrn_mask_reservation_quote = cbrn_reservation_quote_state_need }
	clamp_temp_variable = { var = cbrn_mask_reservation_quote min = constant:cbrn_math.zero max = cbrn_mask_reservation_available_stock }
	always = yes
}

# Country scope, selected decision state in FROM. The exact quoted PP fee and
# any positive stock-capped physical crate issue suffice; the cost row uses this gate.

cbrn_quote_state_mask_reservation_internal = {
	set_temp_variable = { cbrn_reservation_quote_state_need = constant:cbrn_math.zero }

	multiply_temp_variable = { cbrn_reservation_quote_requirement = constant:cbrn_mask_scale.civilian_crates_per_population_k }
	set_temp_variable = { cbrn_reservation_quote_existing_crates = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_reservation_quote_model = constant:cbrn_math.zero }
	if = { limit = { has_variable = cbrn_civilian_mask_basic_crates } set_temp_variable = { cbrn_reservation_quote_model = cbrn_civilian_mask_basic_crates } }
	clamp_temp_variable = { var = cbrn_reservation_quote_model min = constant:cbrn_math.zero }
	add_to_temp_variable = { cbrn_reservation_quote_existing_crates = cbrn_reservation_quote_model }
	set_temp_variable = { cbrn_reservation_quote_model = constant:cbrn_math.zero }
	if = { limit = { has_variable = cbrn_civilian_mask_reconditioned_crates } set_temp_variable = { cbrn_reservation_quote_model = cbrn_civilian_mask_reconditioned_crates } }
	clamp_temp_variable = { var = cbrn_reservation_quote_model min = constant:cbrn_math.zero }
	add_to_temp_variable = { cbrn_reservation_quote_existing_crates = cbrn_reservation_quote_model }
	set_temp_variable = { cbrn_reservation_quote_model = constant:cbrn_math.zero }
	if = { limit = { has_variable = cbrn_civilian_mask_improved_crates } set_temp_variable = { cbrn_reservation_quote_model = cbrn_civilian_mask_improved_crates } }
	clamp_temp_variable = { var = cbrn_reservation_quote_model min = constant:cbrn_math.zero }
	add_to_temp_variable = { cbrn_reservation_quote_existing_crates = cbrn_reservation_quote_model }
	set_temp_variable = { cbrn_reservation_quote_model = constant:cbrn_math.zero }
	if = { limit = { has_variable = cbrn_civilian_mask_advanced_crates } set_temp_variable = { cbrn_reservation_quote_model = cbrn_civilian_mask_advanced_crates } }
	clamp_temp_variable = { var = cbrn_reservation_quote_model min = constant:cbrn_math.zero }
	add_to_temp_variable = { cbrn_reservation_quote_existing_crates = cbrn_reservation_quote_model }
	set_temp_variable = { cbrn_reservation_quote_model = constant:cbrn_math.zero }
	if = { limit = { has_variable = cbrn_civilian_mask_sealed_crates } set_temp_variable = { cbrn_reservation_quote_model = cbrn_civilian_mask_sealed_crates } }
	clamp_temp_variable = { var = cbrn_reservation_quote_model min = constant:cbrn_math.zero }
	add_to_temp_variable = { cbrn_reservation_quote_existing_crates = cbrn_reservation_quote_model }
	set_temp_variable = { cbrn_reservation_quote_existing_coverage = constant:cbrn_math.zero }
	if = {
		limit = {
			check_variable = { cbrn_reservation_quote_requirement > constant:cbrn_math.zero }
			check_variable = { cbrn_reservation_quote_existing_crates > constant:cbrn_math.zero }
		}
		set_temp_variable = { cbrn_reservation_quote_fit = constant:cbrn_math.zero }
		if = { limit = { has_variable = cbrn_civilian_mask_fit_points } set_temp_variable = { cbrn_reservation_quote_fit = cbrn_civilian_mask_fit_points } }
		divide_temp_variable = { cbrn_reservation_quote_fit = cbrn_reservation_quote_existing_crates }
		clamp_temp_variable = { var = cbrn_reservation_quote_fit min = constant:cbrn_math.zero max = constant:cbrn_math.one }
		set_temp_variable = { cbrn_reservation_quote_filter = constant:cbrn_mask_filters.condition_full }
		if = { limit = { has_variable = cbrn_civilian_mask_filter_condition } set_temp_variable = { cbrn_reservation_quote_filter = cbrn_civilian_mask_filter_condition } }
		clamp_temp_variable = { var = cbrn_reservation_quote_filter min = constant:cbrn_mask_filters.condition_minimum max = constant:cbrn_mask_filters.condition_maximum }
		divide_temp_variable = { cbrn_reservation_quote_filter = constant:cbrn_math.percent_divisor }
		set_temp_variable = { cbrn_reservation_quote_existing_coverage = cbrn_reservation_quote_existing_crates }
		multiply_temp_variable = { cbrn_reservation_quote_existing_coverage = constant:cbrn_mask_scale.percent_multiplier }
		divide_temp_variable = { cbrn_reservation_quote_existing_coverage = cbrn_reservation_quote_requirement }
		multiply_temp_variable = { cbrn_reservation_quote_existing_coverage = cbrn_reservation_quote_fit }
		multiply_temp_variable = { cbrn_reservation_quote_existing_coverage = cbrn_reservation_quote_filter }
		clamp_temp_variable = { var = cbrn_reservation_quote_existing_coverage min = constant:cbrn_meter.minimum max = constant:cbrn_meter.maximum }
	}

	clamp_temp_variable = { var = cbrn_reservation_quote_cost min = constant:cbrn_action_input_bounds.condition_mult_min max = constant:cbrn_action_input_bounds.condition_mult_max }
	clamp_temp_variable = { var = cbrn_reservation_quote_effectiveness min = constant:cbrn_math.zero max = constant:cbrn_math.one }
	clamp_temp_variable = { var = cbrn_reservation_quote_new_filter min = constant:cbrn_mask_filters.condition_minimum max = constant:cbrn_mask_filters.condition_maximum }
	if = { limit = { OR = { is_capital = yes has_state_category = city has_state_category = large_city has_state_category = metropolis has_state_category = megalopolis } } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.urban_or_capital_cost_mult } }
	if = { limit = { infrastructure < 4 } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.low_infrastructure_cost_mult } }
	if = { limit = { has_state_flag = cbrn_active_distribution_disruption } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.active_combat_cost_mult } }
	if = { limit = { NOT = { is_core_of = ROOT } } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.occupied_non_core_cost_mult } }
	ROOT = {
		if = { limit = { has_country_flag = cbrn_civil_defence_established } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.civil_defence_cost_mult } }
		if = { limit = { has_country_flag = cbrn_population_registered_and_fitted } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_mask_distribution.registration_cost_mult } }
		if = { limit = { has_tech = civil_defence_fitting_and_registration } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_protection_technology_effect.civil_registration_distribution_cost_mult } }
		if = { limit = { has_country_flag = cbrn_national_respirator_reserve } multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_protection_program_effect.reserve_distribution_cost_mult } }
		if = {
			limit = { has_country_flag = cbrn_simplified_filter_program_active }
			multiply_temp_variable = { cbrn_reservation_quote_cost = constant:cbrn_protection_program_effect.simplified_distribution_cost_mult }
			multiply_temp_variable = { cbrn_reservation_quote_effectiveness = constant:cbrn_protection_program_effect.simplified_distribution_effectiveness_mult }
			clamp_temp_variable = { var = cbrn_reservation_quote_new_filter max = constant:cbrn_protection_program_effect.simplified_filter_condition }
		}
		if = {
			limit = { cbrn_designer_has_mass_civilian_pattern = yes }
			multiply_temp_variable = { cbrn_reservation_quote_effectiveness = constant:cbrn_protection_designer_effect.mass_civilian_distribution_efficiency_mult }
			clamp_temp_variable = { var = cbrn_reservation_quote_effectiveness min = constant:cbrn_math.zero max = constant:cbrn_math.one }
		}
	}
	clamp_temp_variable = { var = cbrn_reservation_quote_cost min = constant:cbrn_action_input_bounds.condition_mult_min max = constant:cbrn_action_input_bounds.condition_mult_max }
	clamp_temp_variable = { var = cbrn_reservation_quote_target min = constant:cbrn_math.zero max = constant:cbrn_math.one }
	set_temp_variable = { cbrn_reservation_quote_effective_needed = cbrn_reservation_quote_requirement }
	multiply_temp_variable = { cbrn_reservation_quote_effective_needed = cbrn_reservation_quote_target }
	set_temp_variable = { cbrn_reservation_quote_existing_effective = cbrn_reservation_quote_requirement }
	multiply_temp_variable = { cbrn_reservation_quote_existing_effective = cbrn_reservation_quote_existing_coverage }
	divide_temp_variable = { cbrn_reservation_quote_existing_effective = constant:cbrn_math.percent_divisor }
	subtract_from_temp_variable = { cbrn_reservation_quote_effective_needed = cbrn_reservation_quote_existing_effective }
	clamp_temp_variable = { var = cbrn_reservation_quote_effective_needed min = constant:cbrn_math.zero }
	set_temp_variable = { cbrn_reservation_quote_new_effective_fraction = cbrn_reservation_quote_new_filter }
	divide_temp_variable = { cbrn_reservation_quote_new_effective_fraction = constant:cbrn_math.percent_divisor }
	multiply_temp_variable = { cbrn_reservation_quote_new_effective_fraction = cbrn_reservation_quote_effectiveness }
	if = {
		limit = { check_variable = { cbrn_reservation_quote_new_effective_fraction > constant:cbrn_math.zero } }
		set_temp_variable = { cbrn_reservation_quote_state_need = cbrn_reservation_quote_effective_needed }
		divide_temp_variable = { cbrn_reservation_quote_state_need = cbrn_reservation_quote_new_effective_fraction }
		multiply_temp_variable = { cbrn_reservation_quote_state_need = cbrn_reservation_quote_cost }
		round_temp_variable = cbrn_reservation_quote_state_need
		clamp_temp_variable = { var = cbrn_reservation_quote_state_need min = constant:cbrn_math.zero }
	}
	always = yes
}
