# ============================================================================
# CHAOS REDUX - CBRN PROTECTIVE-EQUIPMENT AND CIVIL-DEFENCE TRIGGERS
# ============================================================================
# Overview:
# - Gates national respirator programs, field issue, state distribution,
#   procurement, allied transfers, filter service, and reconditioning.
# - Keeps exact-state chemical-raid responses unavailable unless a verified
#   raid/operation adapter has set cbrn_chemical_raid_alert_active.
# - Reads real equipment and factory capacity. It never infers air activity or
#   creates contamination from aircraft presence.
# ============================================================================

cbrn_protection_program_category_visible = {
	OR = {
		cbrn_country_has_program = yes
		has_country_flag = cbrn_national_respirator_reserve
		has_tech = basic_gas_masks
		has_equipment = { gas_mask_equipment > constant:cbrn_math.zero }
		has_enemy_used_chemical_weapons = yes
		any_controlled_state = {
			has_state_flag = cbrn_chemical_raid_alert_active
		}
	}
}

cbrn_civil_defence_category_visible = {
	cbrn_protection_program_category_visible = yes
	OR = {
		has_country_flag = cbrn_national_respirator_reserve
		has_country_flag = cbrn_population_registered_and_fitted
		has_enemy_used_chemical_weapons = yes
		any_controlled_state = {
			OR = {
				has_state_flag = cbrn_chemical_raid_alert_active
				cbrn_state_has_chemical_contamination = yes
				check_variable = { cbrn_civilian_mask_crates > constant:cbrn_math.zero }
			}
		}
	}
}

cbrn_country_below_mask_ai_target = {
	OR = {
		NOT = { has_variable = cbrn_military_mask_coverage }
		AND = {
			has_variable = cbrn_ai_military_mask_coverage_target
			check_variable = {
				var = cbrn_military_mask_coverage
				value = cbrn_ai_military_mask_coverage_target
				compare = less_than
			}
		}
		AND = {
			has_variable = cbrn_ai_mask_reserve_target_crates
			check_variable = {
				var = cbrn_mask_reserve_crates
				value = cbrn_ai_mask_reserve_target_crates
				compare = less_than
			}
		}
		cbrn_country_has_mask_replacement_backlog = yes
		AND = {
			has_country_flag = cbrn_population_registered_and_fitted
			OR = {
				cbrn_profile_is_mass_civil_defence = yes
				cbrn_profile_is_prepared_power = yes
				cbrn_profile_is_civil_defence_network = yes
			}
			cbrn_country_has_full_civilian_protection_gap = yes
		}
		AND = {
			cbrn_country_has_public_chemical_emergency = yes
			cbrn_country_has_priority_civilian_protection_gap = yes
		}
	}
}

cbrn_state_has_hospital_response_resources = {
	has_variable = cbrn_response_hospital_mask_cost
	has_variable = cbrn_response_hospital_support_cost
	meta_trigger = {
		text = {
			ROOT = {
				NOT = { has_equipment = { gas_mask_equipment < [MASK_COST] } }
				NOT = { has_equipment = { support_equipment < [SUPPORT_COST] } }
			}
		}
		MASK_COST = "[?cbrn_response_hospital_mask_cost|.0]"
		SUPPORT_COST = "[?cbrn_response_hospital_support_cost|.0]"
	}
}
