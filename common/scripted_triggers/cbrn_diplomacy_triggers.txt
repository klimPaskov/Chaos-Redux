# ============================================================================
# CHAOS REDUX - CBRN DIPLOMACY SCRIPTED TRIGGERS
# ============================================================================
# Overview:
# - Keeps CBRN diplomatic target and cost checks in the CBRN subsystem.
# - Forensic publication requires one exact unshared state record with a live
#   recorded actor; it never infers responsibility from country history.
# - International decontamination requires an exact contaminated state and
#   prior allied, subject, inspection, or observer consent.
# ============================================================================

cbrn_country_can_pay_inspection_demand_cost = {
	NOT = { has_equipment = { cbrn_instrument_equipment < constant:cbrn_diplomacy_cost.inspection_instruments } }
	NOT = { has_equipment = { support_equipment < constant:cbrn_diplomacy_cost.inspection_support } }
}

cbrn_country_can_pay_forensic_publication_cost = {
	NOT = { has_equipment = { cbrn_instrument_equipment < constant:cbrn_diplomacy_cost.forensic_instruments } }
	NOT = { has_equipment = { support_equipment < constant:cbrn_diplomacy_cost.forensic_support } }
	NOT = { num_of_civilian_factories_available_for_projects < constant:cbrn_diplomacy_factory_use.forensic_publication }
}

cbrn_country_can_pay_international_decon_cost = {
	NOT = { has_equipment = { gas_mask_equipment < constant:cbrn_diplomacy_cost.decon_masks } }
	NOT = { has_equipment = { decontamination_equipment < constant:cbrn_diplomacy_cost.decon_equipment } }
	NOT = { has_equipment = { cbrn_instrument_equipment < constant:cbrn_diplomacy_cost.decon_instruments } }
	NOT = { has_equipment = { support_equipment < constant:cbrn_diplomacy_cost.decon_support } }
	NOT = { has_equipment = { motorized_equipment < constant:cbrn_diplomacy_cost.decon_trucks } }
	NOT = { has_equipment = { convoy_1 < constant:cbrn_diplomacy_cost.decon_convoys } }
	NOT = { num_of_civilian_factories_available_for_projects < constant:cbrn_diplomacy_factory_use.decon_mission }
}

# Country scope. The caller must set temporary cbrn_retaliation_target_id from
# one exact live country scope before calling this trigger.
cbrn_country_has_active_retaliation_right_against_target_id = {
	has_variable = cbrn_retaliation_target_id
	check_variable = { cbrn_retaliation_target_id > constant:cbrn_math.zero }
	cbrn_policy_is_retaliation_authority = yes
	meta_trigger = {
		text = {
			has_variable = cbrn_retaliation_[TARGET_ID]_expiry_day
			check_variable = {
				var = global.num_days
				value = cbrn_retaliation_[TARGET_ID]_expiry_day
				compare = less_than_or_equals
			}
		}
		TARGET_ID = "[?cbrn_retaliation_target_id|.0]"
	}
}

# Country scope with FROM as the exact proposed target.
cbrn_policy_allows_battlefield_use_against_from = {
	FROM = {
		exists = yes
		set_temp_variable = { cbrn_retaliation_target_id = id }
	}
	has_war_with = FROM
	OR = {
		check_variable = {
			var = cbrn_use_policy
			value = constant:cbrn_use_policy.limited_battlefield_authority
			compare = greater_than_or_equals
		}
		cbrn_country_has_active_retaliation_right_against_target_id = yes
	}
}

# Country scope with the exact target state already supplied by the shared
# chemical action record. This gate runs before payload debit.
cbrn_chemical_action_policy_allows_exact_target = {
	has_event_target = cbrn_action_target_state
	event_target:cbrn_action_target_state = {
		CONTROLLER = {
			exists = yes
			NOT = { tag = ROOT }
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}
	}
	event_target:cbrn_action_target_state = {
		CONTROLLER = { has_war_with = ROOT }
	}
	OR = {
		AND = {
			OR = {
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.cylinder_release compare = equals }
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.projector_barrage compare = equals }
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.artillery_fire_plan compare = equals }
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.armored_delivery compare = equals }
			}
			check_variable = {
				var = cbrn_use_policy
				value = constant:cbrn_use_policy.limited_battlefield_authority
				compare = greater_than_or_equals
			}
		}
		AND = {
			OR = {
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.chemical_air_raid compare = equals }
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.strategic_chemical_raid compare = equals }
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.covert_operation compare = equals }
			}
			check_variable = {
				var = cbrn_use_policy
				value = constant:cbrn_use_policy.strategic_release_authority
				compare = greater_than_or_equals
			}
		}
		AND = {
			OR = {
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.nerve_suppression compare = equals }
				check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.doomsday_release compare = equals }
			}
			cbrn_policy_allows_extreme_use = yes
		}
		AND = {
			NOT = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.nerve_suppression compare = equals } }
			NOT = { check_variable = { var = cbrn_action_delivery_route value = constant:cbrn_delivery_route.doomsday_release compare = equals } }
			NOT = { check_variable = { var = cbrn_action_severity value = constant:cbrn_operation_severity.doomsday compare = equals } }
			cbrn_country_has_active_retaliation_right_against_target_id = yes
		}
	}
}

# Country scope. ROOT is the demanding participant.
cbrn_target_can_receive_inspection_demand_from_root = {
	exists = yes
	NOT = { tag = ROOT }
	NOT = { is_actual_nonhuman_country = yes }
	condemnation_is_formally_censured = yes
	NOT = { has_country_flag = condemnation_inspections_active }
	NOT = { has_country_flag = condemnation_refused_inspection }
	NOT = { has_country_flag = cbrn_inspection_demand_active }
	NOT = { has_war_with = ROOT }
}

cbrn_state_has_shareable_chemical_evidence = {
	has_variable = cbrn_evidence_quality
	check_variable = {
		var = cbrn_evidence_quality
		value = constant:cbrn_evidence_threshold.suspected
		compare = greater_than_or_equals
	}
	has_variable = cbrn_last_chemical_actor
	var:cbrn_last_chemical_actor = { exists = yes }
	has_variable = cbrn_last_chemical_action_date
	has_variable = cbrn_last_chemical_condemnation_record_date
	check_variable = {
		var = cbrn_last_chemical_condemnation_record_date
		value = cbrn_last_chemical_action_date
		compare = equals
	}
	has_variable = cbrn_last_chemical_condemnation_total
	has_variable = cbrn_last_chemical_condemnation_paid
	has_variable = cbrn_last_chemical_condemnation_unpaid
	check_variable = { cbrn_last_chemical_condemnation_unpaid > constant:cbrn_math.zero }
	has_variable = cbrn_last_chemical_evidence_applied
	check_variable = {
		var = cbrn_last_chemical_evidence_applied
		value = constant:cbrn_diplomacy_evidence.chemical_sample_minimum
		compare = greater_than_or_equals
	}
	OR = {
		NOT = { has_variable = cbrn_last_chemical_evidence_shared_action_date }
		check_variable = {
			var = cbrn_last_chemical_evidence_shared_action_date
			value = cbrn_last_chemical_action_date
			compare = less_than
		}
	}
}

}

# State scope. ROOT is the foreign mission sponsor.
cbrn_state_can_receive_international_decontamination_from_root = {
	cbrn_state_has_chemical_contamination = yes
	NOT = { has_state_flag = cbrn_international_decon_mission_active }
	NOT = { has_state_flag = cbrn_international_decon_mission_cooldown }
	CONTROLLER = {
		exists = yes
		NOT = { tag = ROOT }
		NOT = { is_actual_nonhuman_country = yes }
		NOT = { has_war_with = ROOT }
		OR = {
			is_in_faction_with = ROOT
			is_subject_of = ROOT
			has_country_flag = condemnation_inspections_active
			has_country_flag = condemnation_observers_active
		}
	}
}
