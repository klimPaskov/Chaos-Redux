is_africa_valid_unifier_candidate = {
	exists = yes
	capital_scope = { is_on_continent = africa }
	NOT = { has_capitulated = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_country_flag = africa_unifier_active }
	NOT = { has_country_flag = africa_continental_formable_complete }
	NOT = { has_country_flag = africa_world_end_actor }
	NOT = { has_country_flag = chaosx_broken_country }
	OR = {
		any_owned_state = { is_on_continent = africa }
		any_controlled_state = { is_on_continent = africa }
	}
	OR = {
		is_subject = no
		has_country_flag = africa_subject_unifier_allowed
		AND = {
			is_subject = yes
			has_war = yes
			any_enemy_country = {
				NOT = { capital_scope = { is_on_continent = africa } }
			}
		}
	}
}
