# COUNTRY scope. The mapped playbook is itself the African-rooted historical
# package; the capital/core checks prevent relocated metropoles or occupation
# authorities from borrowing that legitimacy.
africa_is_eligible_host = {
	exists = yes
	africa_has_mapped_host_playbook = yes
	has_capitulated = no
	capital_scope = {
		is_on_continent = africa
		is_controlled_by = ROOT
	}
	any_core_state = {
		is_on_continent = africa
	}
	OR = {
		has_civil_war = no
		original_tag = SAF
	}
	OR = {
		is_government_in_exile = no
		any_controlled_state = {
			is_on_continent = africa
		}
	}
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { has_country_flag = africa_unifier_host }
	NOT = { has_country_flag = africa_unified_identity }
	NOT = { has_country_flag = africa_terminal_world_actor }
	NOT = { has_country_flag = event_012_incompatible_terminal_actor }
	NOT = { has_global_flag = africa_event_active }
	NOT = { has_global_flag = africa_is_one }
	NOT = { has_global_flag = world_end }
}

# COUNTRY scope. Foreign colonial holders, coalition members, continent-scale
# partners, and bounded high-chaos actors can be explicit action targets without
# becoming Charter relationship records.

	exists = yes
	NOT = { has_country_flag = africa_unifier_host }
	OR = {
		has_country_flag = africa_registered_high_chaos_actor
		AND = {
			NOT = { is_actual_nonhuman_country = yes }
			NOT = { is_special_chaos_country = yes }
			OR = {
				any_owned_state = { is_on_continent = africa }
				has_country_flag = africa_scramble_coalition_member
				has_country_flag = africa_world_order_partner
			}
		}
	}
}

	OR = {
		africa_is_valid_relationship_target = yes
		africa_is_valid_external_action_target = yes
	}
}

# COUNTRY scope on a contact candidate. The current host's overlay selects a
# controlled regional pool; this prevents an opening in one end of Africa from
# presenting arbitrary distant governments as its first practical contact.

			OR = { original_tag = MOR original_tag = ALG original_tag = TUN original_tag = LBA original_tag = MRT original_tag = EGY }
		}
		AND = {
			event_target:africa_host = { check_variable = { africa_regional_overlay = constant:africa_overlay.west_atlantic } }
			OR = { original_tag = LIB original_tag = NGA original_tag = GHA original_tag = SEN original_tag = SIE original_tag = GNB original_tag = CBV original_tag = GAM original_tag = IVO original_tag = DAH original_tag = TOG original_tag = MLI original_tag = VOL }
		}
		AND = {
			event_target:africa_host = { check_variable = { africa_regional_overlay = constant:africa_overlay.sahel_lake_chad } }
			OR = { original_tag = SUD original_tag = SEN original_tag = MLI original_tag = MRT original_tag = NGR original_tag = VOL original_tag = CHA original_tag = NGA original_tag = CMR }
		}
		AND = {

			OR = { original_tag = EGY original_tag = SUD original_tag = ETH original_tag = SOM original_tag = ERI original_tag = DJI original_tag = UGA original_tag = KEN }

# COUNTRY scope on a member. Integration needs an explicit accepted agreement,
# representation, administration, access, settlement, confidence, and a route;
# opinion by itself can never satisfy this gate.

	OR = {
		africa_relationship_is_chartered = yes
		africa_relationship_is_autonomous_federal = yes
		africa_relationship_is_occupied_settlement = yes
	}
	africa_member_host_generation_is_current = yes
	has_country_flag = africa_member_integration_consent
	has_country_flag = africa_member_representation_settled
	has_country_flag = africa_member_administrative_dossier_complete
	has_country_flag = africa_member_transport_connection_complete
	has_country_flag = africa_member_local_settlement_complete
	check_variable = {
		var = africa_member_confidence
		value = constant:africa_measure.high
		compare = greater_than_or_equals
	}
	event_target:africa_host = {
		check_variable = {
			var = africa_integration_burden
			value = constant:africa_measure.high
			compare = less_than
		}
		check_variable = { africa_constitution > constant:africa_constitution.uncommitted }
	}
}

# COUNTRY scope on the host. The stored quote is the only player-facing cost
# record and is regenerated whenever either the action or target changes.
africa_has_current_action_quote = {
	africa_is_current_host = yes
	has_country_flag = africa_action_quote_ready
	check_variable = { africa_selected_action_id > constant:africa_action.none }
	check_variable = {
		var = africa_quote_action_id
		value = africa_selected_action_id
		compare = equals
	}
}

# COUNTRY scope on the host.
africa_can_pay_action_quote = {
	africa_has_current_action_quote = yes
	has_political_power > africa_quote_cost_pp
	command_power > africa_quote_cost_cp
	has_manpower > africa_quote_cost_manpower
	has_fuel > africa_quote_cost_fuel
	has_stability > africa_quote_cost_stability
	war_support > africa_quote_cost_war_support
	check_variable = {
		var = africa_civilian_capacity_available
		value = africa_quote_cost_civilian
		compare = greater_than_or_equals
	}
	check_variable = {
		var = africa_intelligence_capacity_available
		value = africa_quote_cost_intelligence
		compare = greater_than_or_equals
	}
	NOT = { has_equipment = { infantry_equipment < africa_quote_cost_infantry } }
	NOT = { has_equipment = { support_equipment < africa_quote_cost_support } }
	NOT = { has_equipment = { motorized_equipment < africa_quote_cost_motorized } }
	NOT = { has_equipment = { train_equipment < africa_quote_cost_trains } }
	NOT = { has_equipment = { convoy_1 < africa_quote_cost_convoys } }
}

			}
		}
		check_variable = { africa_quote_phase = constant:africa_action_phase.any }
		AND = {
			check_variable = { africa_quote_phase = constant:africa_action_phase.opening }
			check_variable = { africa_current_phase = constant:africa_action_phase.opening }
		}
		AND = {
			check_variable = { africa_quote_phase = constant:africa_action_phase.regional }
			OR = {
				check_variable = { africa_current_phase = constant:africa_action_phase.regional }
				check_variable = { africa_current_phase = constant:africa_action_phase.late }
				check_variable = { africa_current_phase = constant:africa_action_phase.high_chaos }
			}
		}
		AND = {
			check_variable = { africa_quote_phase = constant:africa_action_phase.late }
			OR = {
				check_variable = { africa_current_phase = constant:africa_action_phase.late }
				check_variable = { africa_current_phase = constant:africa_action_phase.post_unification }
			}
		}
		AND = {
			check_variable = { africa_quote_phase = constant:africa_action_phase.high_chaos }
			has_global_flag = africa_evolution_iii_logged
			OR = {
				check_variable = { africa_current_phase = constant:africa_action_phase.high_chaos }
				check_variable = { africa_current_phase = constant:africa_action_phase.post_unification }
				check_variable = { africa_current_phase = constant:africa_action_phase.terminal_approach }
				check_variable = { africa_current_phase = constant:africa_action_phase.terminal }
