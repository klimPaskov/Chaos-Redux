# COUNTRY scope. The mapped playbook is itself the African-rooted historical
# package; the capital/core checks prevent relocated metropoles or occupation
# authorities from borrowing that legitimacy. The release-candidate package
# only takes over the generic tree. Countries with meaningful national trees
# remain eligible as members but cannot lose their existing content to Event 12.
africa_is_eligible_host = {
	exists = yes
	africa_has_mapped_host_playbook = yes
	africa_has_prefire_opening_contact = yes
	has_focus_tree = generic_focus
	has_capitulated = no
	capital_scope = {
		is_on_continent = africa
		is_controlled_by = PREV
	}
	any_core_state = {
		is_on_continent = africa
	}
	has_civil_war = no
	OR = {
		NOT = { original_tag = SAF }
		tag = SAF
	}
	OR = {
		is_government_in_exile = no
		any_controlled_state = {
			is_on_continent = africa
		}
	}
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { has_country_flag = africa_unifier_host }
	NOT = { has_country_flag = africa_unified_identity }
	NOT = { has_country_flag = africa_terminal_world_actor }
	NOT = { has_country_flag = event_012_incompatible_terminal_actor }
	NOT = { has_global_flag = africa_event_active }
	NOT = { has_global_flag = africa_is_one }
	NOT = { has_global_flag = world_end }
}

# opinion check. It binds the mapped opening weakness to a successful proof,
# public obligation, provisional congress, compact identity, territorial

# COUNTRY scope on an active target. Objective completion is deliberately
# narrow: war-end rows can resolve when the target is no longer at war, while
# response rows resolve through their shared-mission timeout event and all other
# rows use their contract timeout or existing action callbacks.

			check_variable = { africa_active_action_objective = constant:africa_action_objective.war_preparation }
			has_war_with = ROOT
		}
	}
}
