# ============================================================================
# CHAOS REDUX - EVENT 021 RANDOM CIVIL WAR TRIGGERS
# ============================================================================
# Public predicates for Event 021. Country predicates run in COUNTRY scope,
# state predicates run in STATE scope, and the bounded scheduler predicates may
# be called from any scope while reading the global registry.
#
# Parent contracts:
# - route-ready flags are asserted by the Event 021 planner after an actual
#   actor, territory, remnant, and identity proof exists;
# - Event 006 admission runs in the dormant package-country scope with a
#   temporary `independence_wave_execution_package_id`;
# - global review is driven by a bounded caller, never by an unrestricted
#   daily or monthly any_country hook.
#
# Actual non-human countries are rejected in every target, queue, scenario,
# exposure, and Event 006 adapter predicate. A broader Chaos state is not an
# immunity bypass. No trigger in this file mutates state.
# ============================================================================

# COUNTRY scope. Internal Event 021 compatibility identity used by the Event
# 006 package-content and player-surface gates after the adapter is complete.
# The predicate remains owned by Event 021 and does not publish Event 006
# lifecycle, league, network, firing, or evolution state.
is_independence_wave_event021_package_country = {
	exists = yes
	random_civil_war_is_normal_human_country = yes
	has_country_flag = random_civil_war_event6_adapter_complete
	has_country_flag = independence_wave_event021_adapter_setup_proven
	has_country_flag = random_civil_war_event6_origin_recorded
	has_country_flag = random_civil_war_origin_adapter_complete
	has_variable = independence_wave_package_id
	NOT = { has_country_flag = independence_wave_active_origin }
	NOT = { has_country_flag = independence_wave_origin_ended }
}

# COUNTRY scope. A successor grace receipt blocks every automatic, queue, and
# manual-scenario opening until its recorded date has passed. The expiry branch
# keeps a bounded target review from depending on a separate maintenance tick.
random_civil_war_successor_grace_clear = {
	OR = {
		NOT = { has_country_flag = random_civil_war_successor_grace }
		AND = {
			has_variable = random_civil_war_successor_grace_until
			check_variable = {
				var = global.date
				value = random_civil_war_successor_grace_until
				compare = greater_than_or_equals
			}
		}
	}
}

# COUNTRY scope. Baseline automatic selection uses this predicate. Subject
# status is intentionally not an exclusion because the eligible route planner
# is allowed to supply a valid subject crisis under the active evolution.
random_civil_war_country_can_be_target = {
	random_civil_war_is_normal_human_country = yes
	# The public automatic target receipt must remain empty until the owner
	# deliberately publishes the rework-ready flag.
	has_global_flag = random_civil_war_rework_ready
	# Event 021 is a bounded individual crisis. Do not admit a country that is
	# already carrying the shared package cap, even when its own route remains
	# otherwise valid.
	individual_crisis_load_is_below_cap = yes
	random_civil_war_major_targeting_allowed = yes
	num_of_controlled_states > constant:random_civil_war_value.zero
	NOT = { random_civil_war_has_terminal_lock = yes }
	random_civil_war_successor_grace_clear = yes
	NOT = { random_civil_war_has_incompatible_bespoke_route = yes }
	NOT = { has_country_flag = random_civil_war_active }
	NOT = { has_country_flag = random_civil_war_target_reserved }
	OR = {
		NOT = { has_country_flag = random_civil_war_recent_target }
		AND = {
			has_variable = random_civil_war_recent_target_until
			check_variable = {
				var = global.date
				value = random_civil_war_recent_target_until
				compare = greater_than_or_equals
			}
		}
	}
	random_civil_war_has_valid_opposition_route = yes
}

# STATE scope. The first state is selected from the caller's seed array. Later
# states must touch a reserved state, unless the host explicitly selected the
# same-tag all-island route. Ownership and control are both required so a
# third-party occupied state cannot enter the transaction.
random_civil_war_state_can_be_reserved = {
	has_event_target = random_civil_war_host_country
	is_owned_by = event_target:random_civil_war_host_country
	is_controlled_by = event_target:random_civil_war_host_country
	NOT = { is_in_array = { array = global.random_civil_war_reserved_states value = THIS } }
	NOT = { is_in_array = { array = global.random_civil_war_plan_states value = THIS } }
	OR = {
		has_global_flag = random_civil_war_plan_seed_pending
		event_target:random_civil_war_host_country = { has_country_flag = random_civil_war_all_island_route_ready }
		any_neighbor_state = {
			is_in_array = { array = global.random_civil_war_plan_states value = THIS }
		}
	}
}
