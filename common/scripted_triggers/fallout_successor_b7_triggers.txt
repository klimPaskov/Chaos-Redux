fallout_successor_b7_transition_owner_is_current = {
	check_variable = {
		var = global.fallout_transition_phase
		value = constant:fallout_transition_phase.survivor_allocation
		compare = equals
	}
	NOT = { has_global_flag = fallout_transition_error }
	NOT = { has_global_flag = fallout_successor_allocation_complete }
	OR = {
		NOT = { has_global_flag = fallout_successor_conflict_ledger_built }
		fallout_successor_conflict_ledger_is_current = yes
	}
	OR = {
		NOT = { has_global_flag = fallout_successor_allocation_initialized }
		fallout_successor_allocation_source_inventory_is_current = yes
	}
}
