# CHAOS REDUX - CBRN OCCUPATION SCRIPTED TRIGGERS
# ============================================================================
# Overview:
# - Exact predicates for occupation law, external protective aid, stored
#   chemical records, disclosure, trauma, and discovery.
# - Legacy coercive-law AI gates remain for existing policy records; they
#   cannot prepare or execute a nerve-agent decision attack.

#   or opens an unrelated crisis surface.

cbrn_occupation_state_is_coercive_security = {
	occupation_law = cbrn_coercive_security_occupation
}

cbrn_occupation_state_is_protected_administration = {
	occupation_law = cbrn_protected_occupation_administration
}

cbrn_occupation_state_has_allied_force_risk = {
	any_state_division = {
		OWNER = {
			NOT = { tag = ROOT }
			is_in_faction_with = ROOT
		}
	}
}

cbrn_occupation_coercive_security_authorization_requirements = {
	has_country_flag = cbrn_armor_mastery_three
	has_tech = armored_agent_delivery
	OR = {
		is_special_project_completed = sp:sp_cw_sarin_program
		is_special_project_completed = sp:sp_cw_soman_program
	}
	cbrn_chaos_warfare_has_advanced_protection = yes
}

cbrn_occupation_protected_administration_authorization_requirements = {
	has_tech = basic_gas_masks
	cbrn_chaos_warfare_has_advanced_protection = yes
}

cbrn_occupation_category_visible = {
	NOT = { is_actual_nonhuman_country = yes }
	OR = {
		# Existing compatibility records may still need disclosure or cleanup.
		has_country_flag = cbrn_occupation_nerve_suppression_history
		check_variable = { cbrn_external_aid_active_reservations_count > constant:cbrn_math.zero }
		# The defensive occupation and external-aid routes remain supported.
		AND = {
			cbrn_operations_decision_surface_is_unlocked = yes
			any_controlled_state = {
				NOT = { is_owned_by = ROOT }
				NOT = { is_core_of = ROOT }
			}
			OR = {
				has_country_flag = cbrn_occupation_protected_administration_authorized
				cbrn_occupation_protected_administration_authorization_requirements = yes
			}
		}
		AND = {
			cbrn_occupation_country_can_supply_external_protective_aid = yes
			any_state = { cbrn_occupation_state_can_receive_external_protective_aid = yes }
		}
	}
}

cbrn_occupation_ai_coercive_route_requirements = {
	has_country_flag = cbrn_armor_mastery_three
	has_tech = armored_agent_delivery
	has_tech = nerve_agent_suppression_formation
	cbrn_chaos_warfare_has_advanced_protection = yes
	cbrn_policy_allows_extreme_use = yes
	cbrn_occupation_country_has_non_payload_operation_costs = yes
	cbrn_occupation_country_has_adequate_protection = yes
	has_variable = chemical_readiness
	check_variable = {
		var = chemical_readiness
		value = constant:cbrn_occupation_target.minimum_chemical_readiness
		compare = greater_than_or_equals
	}
	OR = {
		cbrn_occupation_country_has_sarin_suppression_stock = yes
		cbrn_occupation_country_has_soman_suppression_stock = yes
	}
	any_country_division = {
		division_has_battalion_in_template = cbrn_gas_mask_decon_detachment
	}
	has_variable = condemnation_total
	check_variable = {
		var = condemnation_total
		value = constant:cbrn_occupation_target.ai_maximum_condemnation
		compare = less_than
	}
	has_variable = condemnation_import_vulnerability
	check_variable = {
		var = condemnation_import_vulnerability
		value = constant:condemnation_ai.import_vulnerability
		compare = less_than
	}
	has_variable = cbrn_mask_replacement_demand
	NOT = { cbrn_country_has_mask_replacement_backlog = yes }
	OR = {
		cbrn_ai_route_allows_unrestricted_use = yes
		AND = {
			cbrn_ai_route_allows_first_use = yes
			NOT = { cbrn_policy_is_defensive_preparation = yes }
			NOT = { condemnation_is_formally_censured = yes }
		}
	}
}

	has_country_flag = cbrn_program_established
	cbrn_country_has_any_mask_stock = yes

	}
}

	NOT = { has_capitulated = yes }
	has_country_flag = cbrn_program_established
}

cbrn_occupation_external_aid_payment_ready = {
	cbrn_occupation_country_can_supply_external_protective_aid = yes
	NOT = { has_political_power < constant:cbrn_occupation_decision_cost.protective_aid_pp }
	FROM = { cbrn_occupation_state_can_receive_external_protective_aid = yes }
}

cbrn_occupation_state_has_external_protective_aid_emergency = {
	OR = {
		has_state_flag = cbrn_chemical_raid_alert_active
		cbrn_state_has_chemical_contamination = yes
		is_biowarfare_contaminated_state = yes
		AND = {
			has_variable = cbrn_occupation_nerve_suppression_trauma
			check_variable = {
				cbrn_occupation_nerve_suppression_trauma > constant:cbrn_math.zero
			}
		}
	}
}

cbrn_occupation_state_can_receive_external_protective_aid = {
	cbrn_occupation_state_remains_external_protective_aid_target = yes
	NOT = { has_state_flag = cbrn_occupation_external_aid_project_active }
	NOT = { has_state_flag = cbrn_external_mask_escrow_active }
	set_temp_variable = { cbrn_mask_reservation_quote = constant:cbrn_math.zero }
	cbrn_quote_external_state_mask_reservation = yes
	check_variable = { cbrn_mask_reservation_quote > constant:cbrn_math.zero }
}

	var:cbrn_external_mask_escrow_donor = { tag = ROOT }

cbrn_occupation_state_remains_external_protective_aid_target = {
	state_population_k > constant:cbrn_math.zero
	controller = {
		exists = yes
		NOT = { tag = ROOT }
	}
	NOT = { is_core_of = controller }
	NOT = { has_state_flag = cbrn_protection_project_active }
	cbrn_occupation_state_has_external_protective_aid_emergency = yes
	set_temp_variable = { cbrn_reservation_quote_state_need = constant:cbrn_math.zero }
	cbrn_quote_external_state_mask_reservation = yes
	check_variable = { cbrn_reservation_quote_state_need > constant:cbrn_math.zero }
}

cbrn_occupation_state_is_occupied_non_core = {
	is_controlled_by = ROOT
	NOT = { is_owned_by = ROOT }
	NOT = { is_core_of = ROOT }
	state_population_k > constant:cbrn_occupation_target.minimum_population_k
	owner = {
		exists = yes
		NOT = { tag = ROOT }
	}
}

	has_state_flag = cbrn_occupation_delayed_backlash_pending
	has_variable = cbrn_occupation_backlash_due_day
	check_variable = {
		var = global.num_days
		value = cbrn_occupation_backlash_due_day
		compare = greater_than_or_equals
	}

	is_controlled_by = var:cbrn_occupation_responsible_country
