# ============================================================================
# CHAOS REDUX - CBRN OCCUPATION SCRIPTED TRIGGERS
# ============================================================================
# Overview:
# - Exact country/state predicates for occupation policy authorization,
#   protected/coercive policy state, nerve-agent lot selection, procurement,
#   protection, operation conditions, cooldown, trauma, and discovery.
# - The operation target is always the regular event target
#   cbrn_occupation_target_state. The caller must supply the state and the
#   explicit cbrn_occupation_requested_agent value (sarin or soman).
# - Unit presence and allied-force safety use direct state-division checks.
#   Selected-state ground operations use a bounded command receipt; this file
#   never estimates air weather, compliance agreements, or inspection status.
# - No trigger selects a state, iterates the world, changes an occupation law,
#   or opens an unrelated crisis surface.
# ============================================================================

cbrn_occupation_state_is_coercive_security = {
	occupation_law = cbrn_coercive_security_occupation
}

cbrn_occupation_state_is_protected_administration = {
	occupation_law = cbrn_protected_occupation_administration
}

cbrn_occupation_state_has_ready_nerve_detachment = {
	any_state_division = {
		OWNER = { tag = ROOT }
		division_has_battalion_in_template = cbrn_nerve_suppression_detachment
		unit_strength > constant:cbrn_occupation_target.minimum_detachment_strength
		unit_organization > constant:cbrn_occupation_target.minimum_detachment_organization
	}
}

cbrn_occupation_state_has_allied_force_risk = {
	any_state_division = {
		OWNER = {
			NOT = { tag = ROOT }
			is_in_faction_with = ROOT
		}
	}
}

cbrn_occupation_coercive_security_authorization_requirements = {
	has_country_flag = cbrn_armor_mastery_three
	has_tech = armored_agent_delivery
	OR = {
		is_special_project_completed = sp:sp_cw_sarin_program
		is_special_project_completed = sp:sp_cw_soman_program
	}
	cbrn_chaos_warfare_has_advanced_protection = yes
}

cbrn_occupation_protected_administration_authorization_requirements = {
	has_tech = basic_gas_masks
	cbrn_chaos_warfare_has_advanced_protection = yes
}

cbrn_occupation_category_visible = {
	OR = {
		has_country_flag = cbrn_occupation_nerve_suppression_history
		AND = {
			cbrn_operations_decision_surface_is_unlocked = yes
			any_controlled_state = {
				NOT = { is_owned_by = ROOT }
				NOT = { is_core_of = ROOT }
			}
			OR = {
				has_country_flag = cbrn_occupation_coercive_security_authorized
				has_country_flag = cbrn_occupation_protected_administration_authorized
				cbrn_occupation_coercive_security_authorization_requirements = yes
				cbrn_occupation_protected_administration_authorization_requirements = yes
			}
		}
	}
}

	has_variable = cbrn_occupation_requested_agent
	OR = {
		check_variable = {
			var = cbrn_occupation_requested_agent
			value = constant:cbrn_agent.sarin
			compare = equals
		}
		check_variable = {
			var = cbrn_occupation_requested_agent
			value = constant:cbrn_agent.soman

		}
	}
