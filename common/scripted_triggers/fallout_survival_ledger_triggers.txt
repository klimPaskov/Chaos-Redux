	has_variable = global.fallout_survival_ledger_state_count
	has_variable = global.fallout_survival_ledger_identity_staged_date
	has_variable = global.fallout_survival_ledger_identity_staged_day

	check_variable = { var = global.fallout_survival_ledger_generation value = global.fallout_transition_generation compare = equals }

	check_variable = { var = global.fallout_survival_ledger_source_allocation_generation value = global.fallout_successor_allocation_generation compare = equals }

	check_variable = { var = global.fallout_survival_ledger_country_count value = global.fallout_survival_ledger_source_country_count compare = equals }
	check_variable = { var = global.fallout_survival_ledger_country_count value = global.fallout_survival_ledger_countries^num compare = equals }
	check_variable = { var = global.fallout_survival_ledger_source_state_count value = global.fallout_successor_inventory_state_count compare = equals }
	check_variable = { var = global.fallout_survival_ledger_state_count value = global.fallout_survival_ledger_source_state_count compare = equals }
	check_variable = { var = global.fallout_survival_ledger_state_count value = global.fallout_survival_ledger_states^num compare = equals }
	NOT = { check_variable = { var = global.fallout_survival_ledger_identity_staged_date value = global.date compare = greater_than } }
	NOT = { check_variable = { var = global.fallout_survival_ledger_identity_staged_day value = global.num_days compare = greater_than } }
	fallout_successor_allocation_structure_is_current = yes
	fallout_survival_ledger_identity_arrays_are_aligned = yes
	all_of = {
		array = global.fallout_survival_ledger_country_index_entries
		value = fallout_survival_stored_country_index
		index = fallout_survival_country_payload_index
		set_temp_variable = { fallout_survival_country_context_loaded = constant:fallout_runtime.zero }
		check_variable = { var = fallout_survival_stored_country_index value = fallout_survival_country_payload_index compare = equals }
		check_variable = { var = global.fallout_survival_ledger_country_generation_entries^fallout_survival_country_payload_index value = global.fallout_transition_generation compare = equals }
		var:global.fallout_survival_ledger_countries^fallout_survival_country_payload_index = {
			set_temp_variable = { fallout_survival_country_context_loaded = constant:fallout_runtime.one }
			check_variable = { var = fallout_survival_country_index value = fallout_survival_country_payload_index compare = equals }
			fallout_survival_country_identity_row_is_current = yes
		}
		check_variable = { fallout_survival_country_context_loaded = constant:fallout_runtime.one }
	}
	all_of = {
		array = global.fallout_successor_assigned_countries
		value = fallout_survival_source_country
		set_temp_variable = { fallout_survival_source_country_context_loaded = constant:fallout_runtime.zero }
		var:fallout_survival_source_country = {
			set_temp_variable = { fallout_survival_source_country_context_loaded = constant:fallout_runtime.one }
			is_in_array = { array = global.fallout_survival_ledger_countries value = THIS }
		}
		check_variable = { fallout_survival_source_country_context_loaded = constant:fallout_runtime.one }
	}
	all_of = {
		array = global.fallout_survival_ledger_state_index_entries
		value = fallout_survival_stored_state_index
		index = fallout_survival_state_payload_index
		set_temp_variable = { fallout_survival_state_context_loaded = constant:fallout_runtime.zero }
		check_variable = { var = fallout_survival_stored_state_index value = fallout_survival_state_payload_index compare = equals }
		check_variable = { var = global.fallout_survival_ledger_state_generation_entries^fallout_survival_state_payload_index value = global.fallout_transition_generation compare = equals }

			set_temp_variable = { fallout_survival_state_context_loaded = constant:fallout_runtime.one }
			check_variable = { var = fallout_survival_state_index value = fallout_survival_state_payload_index compare = equals }
			fallout_survival_state_identity_row_is_current = yes
		}
		check_variable = { fallout_survival_state_context_loaded = constant:fallout_runtime.one }
	}
	NOT = {
		any_state = {
			NOT = { is_in_array = { array = global.fallout_survival_ledger_states value = THIS } }
		}
	}
	NOT = {
		any_country = {
			has_country_flag = fallout_survival_country_identity_staged
			NOT = { is_in_array = { array = global.fallout_survival_ledger_countries value = THIS } }
		}
	}
	NOT = {
		any_state = {
			has_state_flag = fallout_survival_state_identity_staged
			NOT = { is_in_array = { array = global.fallout_survival_ledger_states value = THIS } }
		}
	}
