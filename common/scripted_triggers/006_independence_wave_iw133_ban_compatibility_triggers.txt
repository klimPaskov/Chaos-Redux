# ============================================================================
# CHAOS REDUX - EVENT 006 IW-133 BAN COMPATIBILITY TRIGGERS
# ============================================================================
# Dormant, package-local preservation predicates for the registered vanilla
# BAN carrier. IW-133 remains formable-or-route-only: this file does not make
# BAN an automatic Event 006 candidate, create a country, or widen central
# adapter, attestation, preflight, or Join surfaces.
#
# The route owner must set independence_wave_iw_133_named_route_selected only
# after a named route has been accepted. Every predicate keeps the living BAN
# tag, Event 006 origin, exact package id, named route, and current-map anchor
# explicit before a preservation receipt can be recorded.
# ============================================================================

# COUNTRY scope. A living-tag witness for the registered vanilla carrier.
is_independence_wave_iw_133_ban_living_tag = {
	exists = yes
	original_tag = BAN
}

# COUNTRY scope. Explicit Event 006 origin and package provenance. This is
# deliberately stronger than registry membership: a living BAN from another
# system must never be treated as the IW-133 Event 006 counterpart.
is_independence_wave_iw_133_ban_event6_origin = {
	is_independence_wave_iw_133_ban_living_tag = yes
	is_independence_wave_active_origin_country = yes
	has_country_flag = independence_wave_active_origin
	NOT = { has_country_flag = independence_wave_origin_ended }
	check_variable = { liberation_origin = constant:liberation_origin.independence_wave }
	check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_133 }
}

# COUNTRY scope. Named vanilla route witness. The focus and partition paths
# are intentionally named instead of using a generic BAN-exists check:
# RAJ_united_bengal is the GOE Bengal branch, RAJ_GOE_partition_has_happened
# is the RAJ partition/pressure route, and ENG_the_three_nation_solution is
# the UK three-nation release route. The route selector marker is owned by a
# future source-backed caller and is never set by this package file.
is_independence_wave_iw_133_ban_named_vanilla_route = {
	has_country_flag = independence_wave_iw_133_named_route_selected
	OR = {
		RAJ = {
			has_completed_focus = RAJ_united_bengal
		}
		RAJ = {
			has_global_flag = RAJ_GOE_partition_has_happened
		}
		ENG = {
			has_completed_focus = ENG_the_three_nation_solution
		}
	}
}

# COUNTRY scope. The current installed binding uses state 430 as the BAN
# capital/core; state 431 remains the paired Bengal anchor in the reservation
# group and is accepted only as an additional registered core witness. This
# predicate never transfers, controls, or cores either state.
has_independence_wave_iw_133_ban_anchor_surface = {
	is_independence_wave_iw_133_ban_event6_origin = yes
	capital_scope = { state = 430 }
	430 = {
		is_owned_by = ROOT
		is_core_of = BAN
	}
	OR = {
		430 = { is_core_of = BAN }
		431 = { is_core_of = BAN }
	}
}

# COUNTRY scope. Strict preservation contract for the future named-route
# consumer. It is intentionally not a generic Bengal release trigger.
has_independence_wave_iw_133_ban_compatibility_contract = {
	is_independence_wave_iw_133_ban_event6_origin = yes
	has_independence_wave_iw_133_ban_anchor_surface = yes
	is_independence_wave_iw_133_ban_named_vanilla_route = yes
}

# COUNTRY scope. Exact RAJ GOE named-route guard for callers executing from
# the RAJ focus/decision side of the route.
is_independence_wave_iw_133_raj_bengal_route = {
	tag = RAJ
	BAN = {
		has_independence_wave_iw_133_ban_compatibility_contract = yes
	}
}

# COUNTRY scope. Exact RAJ partition route guard. The partition global flag
# is set before the timeout transfer loop, so callers can branch before BAN
# is considered for pressure-based state transfer.
is_independence_wave_iw_133_raj_partition_route = {
	tag = RAJ
	has_global_flag = RAJ_GOE_partition_has_happened
	BAN = {
		has_independence_wave_iw_133_ban_compatibility_contract = yes
	}
}

# COUNTRY scope. Exact UK three-nation named-route guard.
is_independence_wave_iw_133_uk_three_nation_route = {
	tag = ENG
	BAN = {
		has_independence_wave_iw_133_ban_compatibility_contract = yes
	}
}
