# ============================================================================
# CHAOS REDUX - EVENT 163 DOCTOR WU TRIGGERS
# ============================================================================
# Fire-once host selection and the stable public host API consumed by disease
# systems. Selection scans the world only when Event 163 is about to fire.
# ============================================================================

# COUNTRY scope.
doctor_wu_is_eligible_host = {
	exists = yes
	uses_normal_civilian_systems = yes
	has_capitulated = no
	is_subject = no
	capital_scope = { is_controlled_by = ROOT }
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { has_country_flag = doctor_wu_host_rejected }
	NOT = { has_country_flag = doctor_wu_host_history }
	NOT = { has_global_flag = world_end }
}

# COUNTRY scope.
doctor_wu_host_controls_severe_black_plague_state = {
	any_state_in = {
		array = global.black_plague_active_states
		is_controlled_by = ROOT
		black_plague_state_is_severe = yes
	}
}

# Any scope. Explicit availability query; never called from a periodic on_action.
doctor_wu_has_any_eligible_host = {
	any_country = { doctor_wu_is_eligible_host = yes }
}

doctor_wu_automatic_event_is_available = {
	NOT = { has_global_flag = doctor_wu_event_resolved }
	NOT = { has_global_flag = doctor_wu_active }
	NOT = { has_global_flag = world_end }
	doctor_wu_has_any_eligible_host = yes
}

# COUNTRY scope.
doctor_wu_is_current_host = {
	has_country_flag = doctor_wu_current_host
	has_global_flag = doctor_wu_active
}

# Any scope.
doctor_wu_has_current_host = {
	has_global_flag = doctor_wu_active
	has_event_target = doctor_wu_current_host
	event_target:doctor_wu_current_host = {
		exists = yes
		doctor_wu_is_current_host = yes
	}
}

# COUNTRY scope. Used for a validated transfer before the old host is mutated.
doctor_wu_can_receive_transfer = {
	doctor_wu_is_eligible_host = yes
	NOT = { doctor_wu_is_current_host = yes }
}

# COUNTRY scope. The requester must be a human neighbour or faction partner
# with a live Black Plague state selected on the shared crisis board.
doctor_wu_can_request_foreign_access = {
	exists = yes
	uses_normal_civilian_systems = yes
	NOT = { is_special_chaos_country = yes }
	NOT = { is_actual_nonhuman_country = yes }
	doctor_wu_has_current_host = yes
	NOT = { doctor_wu_is_current_host = yes }
	OR = {
		is_neighbor_of = event_target:doctor_wu_current_host
		is_in_faction_with = event_target:doctor_wu_current_host
	}
	NOT = { has_country_flag = doctor_wu_foreign_access_active }
	NOT = { has_country_flag = doctor_wu_foreign_request_pending }
	NOT = { has_country_flag = doctor_wu_foreign_request_cooldown }
	NOT = { has_global_flag = world_end }
	any_controlled_state = {
		OR = {
			black_plague_state_is_incubating = yes
			black_plague_state_is_infected = yes
			black_plague_state_is_severe = yes
			black_plague_state_is_collapsed = yes
			black_plague_state_is_contained = yes
			black_plague_state_is_recovering = yes
		}
	}
}

# STATE scope. ROOT is the requesting country.
doctor_wu_foreign_request_state_is_valid = {
	ROOT = { doctor_wu_can_request_foreign_access = yes }
	is_controlled_by = ROOT
	OR = {
		black_plague_state_is_incubating = yes
		black_plague_state_is_infected = yes
		black_plague_state_is_severe = yes
		black_plague_state_is_collapsed = yes
		black_plague_state_is_contained = yes
	black_plague_state_is_recovering = yes
	}
}

# COUNTRY scope while Doctor Wu is the responding host.
doctor_wu_can_pay_foreign_access_request = {
	has_equipment = { support_equipment > constant:doctor_wu_access.host_support_equipment }
	has_equipment = { motorized_equipment > constant:doctor_wu_access.host_motorized_equipment }
	has_fuel > constant:doctor_wu_access.host_fuel
}
