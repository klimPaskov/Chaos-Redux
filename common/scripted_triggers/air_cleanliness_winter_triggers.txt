air_winter_state_has_damageable_building = {
	OR = {
		infrastructure > constant:air_winter_runtime.zero
		industrial_complex > constant:air_winter_runtime.zero
		arms_factory > constant:air_winter_runtime.zero
		air_base > constant:air_winter_runtime.zero
		dockyard > constant:air_winter_runtime.zero
		supply_node > constant:air_winter_runtime.zero
		rail_way > constant:air_winter_runtime.zero
	}
}

# Supported scope: STATE
air_winter_building_damage_is_ready = {
	check_variable = {
		var = air_winter_phase
		value = constant:air_winter_phase.hard_freeze
		compare = greater_than_or_equals
	}
	check_variable = {
		var = air_winter_phase_months
		value = constant:air_winter_building_damage.minimum_phase_months
		compare = greater_than_or_equals
	}
	check_variable = {
		var = air_winter_building_damage_pressure
		value = constant:air_winter_building_damage.threshold
		compare = greater_than_or_equals
	}
	NOT = { check_variable = { air_winter_building_damage_cooldown > constant:air_winter_runtime.zero } }
	air_winter_state_has_damageable_building = yes
}

# Supported scope: STATE
air_winter_state_category_can_degrade = {
	check_variable = {
		var = air_winter_phase
		value = constant:air_winter_phase.black_harvest
		compare = greater_than_or_equals
	}
	check_variable = {
		var = air_winter_phase_months
		value = constant:air_winter_category_damage.minimum_phase_months
		compare = greater_than_or_equals
	}
	check_variable = {
		var = air_winter_category_damage
		value = constant:air_winter_category_damage.threshold
		compare = greater_than_or_equals
	}
	NOT = { check_variable = { air_winter_category_cooldown > constant:air_winter_runtime.zero } }
	NOT = {
		OR = {
			has_state_category = wasteland
			has_state_category = enclave
			has_state_category = tiny_island
			has_state_category = small_island
			has_state_category = large_island
		}
	}
	OR = {
		has_state_category = megalopolis
		has_state_category = metropolis
		has_state_category = large_city
		has_state_category = city
		has_state_category = large_town
		has_state_category = town
		has_state_category = rural
		has_state_category = pastoral
	}
}

	OR = {

		NOT = { check_variable = { air_winter_response_cooldown > constant:air_winter_runtime.zero } }
	}

# Supported scope: COUNTRY

	}
}

# Supported scope: COUNTRY. Both regular targets are saved by the scheduler.
air_winter_event_targets_are_valid = {
	has_event_target = air_winter_event_state
	has_event_target = air_winter_event_country
	event_target:air_winter_event_country = { exists = yes }
	event_target:air_winter_event_state = {
		air_winter_state_is_valid = yes
		is_owned_by = event_target:air_winter_event_country
	}

# Supported scope: COUNTRY. The response state is saved immediately before the
# result event and must still belong to the country resolving that event.

	has_event_target = air_winter_response_state
	event_target:air_winter_response_state = {
		air_winter_state_is_valid = yes
		is_owned_by = ROOT
	}
}

# Supported scope: COUNTRY. FROM must be the selected project STATE.
air_winter_response_project_should_cancel = {
	OR = {
		NOT = { air_winter_response_category_is_visible = yes }
		FROM = {
			OR = {
				NOT = { is_owned_by = ROOT }
				NOT = { is_controlled_by = ROOT }
				NOT = { air_winter_state_is_valid = yes }
				NOT = { air_winter_is_active = yes }
			}
		}
	}
}

# Supported scope: COUNTRY. This is a broad AI conservation signal, not a
# payment gate. Exact affordability remains action-specific below.

	OR = {
		NOT = { has_equipment = { support_equipment > constant:air_winter_response_cost.decontamination_support } }
		NOT = { has_equipment = { motorized_equipment > constant:air_winter_response_cost.decontamination_motorized } }
		NOT = { has_equipment = { train_equipment > constant:air_winter_response_cost.final_evacuation_trains } }
		NOT = { has_equipment = { convoy_1 > constant:air_winter_response_cost.final_evacuation_convoys } }
		NOT = { has_manpower > constant:air_winter_response_cost.shelter_manpower }
		NOT = { command_power > constant:air_winter_response_cost.decontamination_command }
		NOT = { has_fuel > constant:air_winter_response_cost.decontamination_fuel }
		has_stability < constant:air_winter_response_cost.minimum_stability_for_heavy_cost
	}
}
