# Supported scope: STATE. In scripted mapmode localisation ROOT is the viewing
# player country and the checked scope is the hovered state, matching the
# official scripted-mapmode scope contract and the existing plague precedent.
air_winter_mapmode_player_has_basic_sampling = {
	ROOT = { exists = yes }
	OR = {
		is_owned_by = ROOT
		is_controlled_by = ROOT
		ROOT = { has_country_flag = air_cleanliness_treaty_member }
	}
}

# Supported scope: STATE. Terminal modelling becomes public after ninety
# percent global contamination. Before then, a major power must have invested
# in an atmospheric office.
air_winter_mapmode_player_has_terminal_modelling = {
	ROOT = { exists = yes }
	OR = {
		check_variable = {
			var = global.air_contamination_bp
			value = constant:air_winter_monitoring.terminal_contamination_bp
			compare = greater_than_or_equals
		}
		ROOT = {
			is_major = yes
			has_country_flag = air_winter_atmospheric_office
		}
	}
}

# Supported scope: STATE. During the single monthly state pass, states already
# updated in the open cycle expose their previous phase. States not yet updated
# expose their current phase. Every neighbor therefore reads the same opening
# snapshot regardless of iterator order.
air_winter_was_severe_at_cycle_start = {
	OR = {
		AND = {
			has_global_flag = air_winter_cycle_open
			has_variable = global.air_winter_cycle_id
			has_variable = air_winter_last_update_cycle_id
			check_variable = {
				var = air_winter_last_update_cycle_id
				value = global.air_winter_cycle_id
				compare = equals
			}
			check_variable = {
				var = air_winter_previous_phase
				value = constant:air_winter_phase.black_harvest
				compare = greater_than_or_equals
			}
		}
		AND = {
			OR = {
				NOT = { has_global_flag = air_winter_cycle_open }
				NOT = { has_variable = global.air_winter_cycle_id }
				NOT = { has_variable = air_winter_last_update_cycle_id }
				NOT = {
					check_variable = {
						var = air_winter_last_update_cycle_id
						value = global.air_winter_cycle_id
						compare = equals
					}
				}
			}
			check_variable = {
				var = air_winter_phase
				value = constant:air_winter_phase.black_harvest
				compare = greater_than_or_equals
			}
		}
	}
}

# Supported scope: STATE. The five factory partitions exactly cover a combined
# operational military and civilian factory total of four or more.
air_winter_event_is_heavy_industry_state = {
	OR = {
		coal > constant:air_winter_event_runtime.zero
		non_damaged_building_level = { building = arms_factory level > constant:air_winter_event_runtime.three }
		AND = {
			non_damaged_building_level = { building = arms_factory level > constant:air_winter_event_runtime.two }
			non_damaged_building_level = { building = industrial_complex level > constant:air_winter_event_runtime.zero }
		}
		AND = {
			non_damaged_building_level = { building = arms_factory level > constant:air_winter_event_runtime.one }
			non_damaged_building_level = { building = industrial_complex level > constant:air_winter_event_runtime.one }
		}
		AND = {
			non_damaged_building_level = { building = arms_factory level > constant:air_winter_event_runtime.zero }
			non_damaged_building_level = { building = industrial_complex level > constant:air_winter_event_runtime.two }
		}
		non_damaged_building_level = { building = industrial_complex level > constant:air_winter_event_runtime.three }
	}
}

# Supported scope: COUNTRY. Both regular targets are saved by the scheduler.
air_winter_event_targets_are_valid = {
	NOT = { has_global_flag = fallout_transition_active }
	NOT = { has_global_flag = fallout_active }
	NOT = { has_global_flag = fallout_air_cleanliness_disabled }
	has_event_target = air_winter_event_state
	has_event_target = air_winter_event_country
	event_target:air_winter_event_country = {
		exists = yes
		tag = ROOT
	}
	event_target:air_winter_event_state = {
		air_winter_state_is_valid = yes
		is_owned_by = event_target:air_winter_event_country
		OR = {
			NOT = { has_state_flag = air_winter_pending_delayed_result }
			AND = {
				has_variable = air_winter_pending_event_owner
				check_variable = {
					var = air_winter_pending_event_owner
					value = event_target:air_winter_event_country
					compare = equals
				}
				air_winter_event_pending_owner_is_valid = yes
			}
		}
	}
}
