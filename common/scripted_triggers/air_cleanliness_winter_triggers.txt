# Supported scope: any
air_winter_system_enabled = {
	OR = {
		NOT = { has_global_flag = settings_air_cleanliness_disabled }
		has_global_flag = air_contamination_final_silence_locked
	}
}

# Supported scope: STATE

	impassable = no
	owner = { exists = yes }
	NOT = { has_state_flag = air_winter_excluded }
}

# Supported scope: STATE. In scripted mapmode localisation ROOT is the viewing
# player country and the checked scope is the hovered state, matching the
# official scripted-mapmode scope contract and the existing plague precedent.
air_winter_mapmode_player_has_basic_sampling = {
	ROOT = { exists = yes }
	OR = {
		is_owned_by = ROOT
		is_controlled_by = ROOT
		ROOT = { has_country_flag = air_cleanliness_treaty_member }
	}
}

# Supported scope: STATE. Completing one reviewed roof-sampler project creates
# a national atmospheric office. The capability is country-wide and persists
# across later state ownership changes until Air Winter is reset.
air_winter_mapmode_player_has_atmospheric_office = {
	ROOT = {
		exists = yes
		has_country_flag = air_winter_atmospheric_office
	}
}

# Supported scope: STATE. Terminal modelling becomes public after ninety
# percent global contamination. Before then, a major power must have invested
# in an atmospheric office.
air_winter_mapmode_player_has_terminal_modelling = {
	ROOT = { exists = yes }
	OR = {
		check_variable = {
			var = global.air_contamination_bp
			value = constant:air_winter_monitoring.terminal_contamination_bp
			compare = greater_than_or_equals
		}
		ROOT = {
			is_major = yes
			has_country_flag = air_winter_atmospheric_office
		}
	}
}

# Supported scope: any. One global generation is opened for each complete
# Fallout snapshot attempt, including retries on the same date.
air_winter_fallout_snapshot_generation_is_open = {
	has_variable = global.air_winter_fallout_snapshot_producer_schema_version
	has_variable = global.air_winter_fallout_snapshot_producer_generation
	check_variable = {
		var = global.air_winter_fallout_snapshot_producer_schema_version
		value = constant:air_winter_fallout_snapshot_schema.version
		compare = equals
	}
	check_variable = { global.air_winter_fallout_snapshot_producer_generation > constant:air_winter_runtime.zero }
}

# Supported scope: STATE. During the single monthly state pass, states already
# updated in the open cycle expose their previous phase. States not yet updated
# expose their current phase. Every neighbor therefore reads the same opening
# snapshot regardless of iterator order.
air_winter_was_severe_at_cycle_start = {
	OR = {
		AND = {
			has_global_flag = air_winter_cycle_open
			has_variable = global.air_winter_cycle_id
			has_variable = air_winter_last_update_cycle_id
			check_variable = {
				var = air_winter_last_update_cycle_id
				value = global.air_winter_cycle_id
				compare = equals
			}
			check_variable = {
				var = air_winter_previous_phase
				value = constant:air_winter_phase.black_harvest
				compare = greater_than_or_equals
			}
		}
		AND = {
			OR = {
				NOT = { has_global_flag = air_winter_cycle_open }
				NOT = { has_variable = global.air_winter_cycle_id }
				NOT = { has_variable = air_winter_last_update_cycle_id }
				NOT = {
					check_variable = {
						var = air_winter_last_update_cycle_id
						value = global.air_winter_cycle_id
						compare = equals
					}
				}
			}
			check_variable = {
				var = air_winter_phase
				value = constant:air_winter_phase.black_harvest
				compare = greater_than_or_equals
			}
		}
	}
}

# Supported scope: STATE. Peak-phase memory makes thaw eligibility persist
# through a multi-month recovery instead of depending on iterator timing.
air_winter_visual_thaw_is_eligible = {
	air_winter_visual_has_active_class_phase = yes
	air_winter_trend_is_recovering = yes
	OR = {
		AND = {
			OR = {
				air_winter_presentation_is_boreal = yes
				air_winter_presentation_is_maritime = yes
				air_winter_presentation_is_mediterranean = yes
				air_winter_presentation_is_arid = yes
				air_winter_presentation_is_oceanic = yes
			}
			check_variable = {
				var = air_winter_peak_phase_memory
				value = constant:air_winter_phase.hard_freeze
				compare = greater_than_or_equals
			}
		}
		AND = {
			OR = {
				air_winter_presentation_is_tropical = yes
				air_winter_presentation_is_equatorial = yes
				air_winter_presentation_is_highland = yes
				air_winter_presentation_is_polar = yes
			}
			check_variable = {
				var = air_winter_peak_phase_memory
				value = constant:air_winter_phase.crop_shock
				compare = greater_than_or_equals
			}
		}
	}
}
