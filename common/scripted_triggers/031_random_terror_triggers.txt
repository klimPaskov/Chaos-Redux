# ============================================================================
# CHAOS REDUX - EVENT 031 RANDOM TERROR TRIGGERS
# ============================================================================
# Country, state, actor, decision, mission, scenario, and terminal predicates
# for the Random Terror system.  Baseline target selection is deliberately
# mechanical: live state activity, pressure, response capacity, and prior
# incident memory are inputs; religion, ethnicity, nationality, refugee status,
# civilization, and ordinary ideology are never inputs.
# ============================================================================

random_terror_is_actor = {
	exists = yes
	has_country_flag = random_terror_actor
	NOT = { has_country_flag = random_terror_cleanup_complete }
}

random_terror_is_jihadist_actor = {
	random_terror_is_actor = yes
	OR = {
		check_variable = { random_terror_actor_profile = constant:random_terror_actor_profile.jihadist }
		check_variable = { random_terror_actor_profile = constant:random_terror_actor_profile.final }
	}
}

random_terror_is_final_actor = {
	random_terror_is_actor = yes
	check_variable = { random_terror_actor_profile = constant:random_terror_actor_profile.final }
	# The profile is not a transaction receipt: Final Jihad can precede revelation.
	hidden_trigger = {
		has_country_flag = random_terror_final_actor
		has_global_flag = random_terror_terminal_active
	}
}

random_terror_is_revelation_candidate = {
	random_terror_is_viable_actor = yes
	random_terror_is_jihadist_actor = yes
	has_global_flag = random_terror_evolution_5_unlocked
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = random_terror_terminal_active }
}

random_terror_is_viable_actor = {
	random_terror_is_actor = yes
	NOT = { has_capitulated = yes }
	num_of_controlled_states > constant:random_terror_value.zero
	OR = {
		check_variable = { random_terror_territorial_control > constant:random_terror_value.territorial_gate }
		check_variable = { random_terror_network_authority > constant:random_terror_value.unity_gate }
	}
}

random_terror_country_is_normal_target = {
	exists = yes
	NOT = { has_capitulated = yes }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { has_country_flag = random_terror_actor }
	NOT = { has_country_flag = random_terror_cleanup_complete }
	NOT = { has_country_flag = random_terror_target_reserved }
	# A recently cleared country rests before ordinary waves select it again.
	NOT = { has_country_flag = random_terror_recovery_cooldown }
	NOT = { has_country_flag = random_terror_incompatible_route }
	NOT = { has_global_flag = world_end }
	num_of_controlled_states > constant:random_terror_value.zero
	any_owned_state = {
		is_controlled_by = PREV
		state_population_k > constant:random_terror_value.zero
		OR = {
			NOT = { has_variable = random_terror_state_cooldown_until }
			check_variable = { var = random_terror_state_cooldown_until value = global.date compare = less_than_or_equals }
		}
	}
}

# STATE scope, intrinsic site checks with no country-owner relationship.
random_terror_state_is_eligible_incident_site = {
	state_population_k > constant:random_terror_value.zero
	OR = {
		NOT = { has_variable = random_terror_state_cooldown_until }
		check_variable = { var = random_terror_state_cooldown_until value = global.date compare = less_than_or_equals }
	}
	NOT = { has_state_flag = random_terror_state_lost_local_control }
}

# STATE scope for an owner-rooted action, retaining its explicit ROOT contract.
random_terror_state_is_valid_target = {
	random_terror_state_is_eligible_incident_site = yes
	OR = {
		is_controlled_by = ROOT
		is_owned_by = ROOT
	}
}

random_terror_state_is_active_crisis = {

		has_state_flag = random_terror_state_active
		has_state_flag = random_terror_state_entrenched
		has_state_flag = random_terror_state_armed_insurgency
		has_state_flag = random_terror_state_lost_local_control
	}
}

random_terror_country_has_active_package = {

# STATE scope, ROOT = the responding government.  The state is the subject of
# ROOT's running Hostage Deadline or Stop the Coordinated Uprising mission, so
# the emergency response actions stay usable on it whatever the phase and
# whichever mission slot that emergency holds.
random_terror_state_is_emergency_subject = {
	OR = {
		AND = {
			has_state_flag = random_terror_hostage_mission_subject_@ROOT
			ROOT = { has_country_flag = random_terror_mission_active_hostage }
		}
		AND = {
			has_state_flag = random_terror_uprising_subject_@ROOT
			ROOT = { has_country_flag = random_terror_mission_active_uprising }
		}
	}
}

# COUNTRY scope: a country that can sponsor an Event 31 actor.  The one
# shared check for Seek a Sponsor, Identify the Sponsor, the request commit
# and the consent event chaosx.nr31.42.  When ROOT is the actor (decision,
# focus and commit), the country must border it or share its faction and
# have no other request pending.  When ROOT is the asked country itself (the
# consent event, whose FROM is the requesting actor), the same relationship
# is read against FROM.  Either way it is never at war with the actor, never
# the government the actor rose against (random_terror_parent_of_<tag>),
# never an Event 14 cannibal country, and holds the sponsorship stock.
random_terror_sponsor_candidate = {
	exists = yes
	NOT = { random_terror_is_actor = yes }
	NOT = { has_capitulated = yes }
	NOT = { is_cannibalism_country = yes }
	OR = {
		AND = {
			NOT = { tag = ROOT }
			ROOT = { random_terror_is_actor = yes }
			NOT = { has_war_with = ROOT }
			NOT = { has_country_flag = random_terror_parent_of_@ROOT }
			NOT = { has_country_flag = random_terror_sponsor_contact_pending }
			OR = {
				is_neighbor_of = ROOT
				is_in_faction_with = ROOT
			}
		}
		AND = {
			tag = ROOT
			FROM = { random_terror_is_actor = yes }
			NOT = { has_war_with = FROM }
			NOT = { has_country_flag = random_terror_parent_of_@FROM }
			OR = {
				is_neighbor_of = FROM
				is_in_faction_with = FROM
			}
		}
	}
	has_political_power > constant:random_terror_cost.partner_sponsor_political_power
	has_equipment = { infantry_equipment > constant:random_terror_cost.partner_sponsor_infantry_equipment }
	has_equipment = { support_equipment > constant:random_terror_cost.partner_sponsor_support_equipment }
}

# STATE scope.  The state borders territory held by an Event 31 actor.

	}
}

# STATE scope.  A crisis state that can send its cell into a neighbouring
# state: Entrenched or worse with any reason, including bordering actor
# territory; Active with a recorded reason; or any state with remaining
# activity whose survivors were scattered.
random_terror_state_is_spread_source = {
	OR = {
		AND = {
			OR = {
				has_state_flag = random_terror_state_entrenched
				has_state_flag = random_terror_state_armed_insurgency
				has_state_flag = random_terror_state_lost_local_control
			}
			OR = {
				random_terror_state_has_recorded_spread_reason = yes
				random_terror_state_borders_actor_territory = yes
			}
		}
		AND = {
			has_state_flag = random_terror_state_active
			random_terror_state_has_recorded_spread_reason = yes
		}
		AND = {
			has_state_flag = random_terror_survivors_displaced
			check_variable = { random_terror_state_activity > constant:random_terror_value.zero }
		}
	}
}

# COUNTRY scope: one dormant carrier tag.  A carrier is free only when no live
# country uses the tag and its reservation flag is clear.  The spawner, the
# Global Jihad availability check and the scenario preflight share this rule,
# so a retained or occupied carrier never receives a second package.
random_terror_carrier_is_free = {
	exists = no
	OR = {
		AND = { original_tag = JHX NOT = { has_global_flag = random_terror_slot_jhx_used } }
		AND = { original_tag = JIX NOT = { has_global_flag = random_terror_slot_jix_used } }
		AND = { original_tag = JKX NOT = { has_global_flag = random_terror_slot_jkx_used } }
		AND = { original_tag = JLX NOT = { has_global_flag = random_terror_slot_jlx_used } }
		AND = { original_tag = JMX NOT = { has_global_flag = random_terror_slot_jmx_used } }
		AND = { original_tag = JNX NOT = { has_global_flag = random_terror_slot_jnx_used } }
		AND = { original_tag = JOX NOT = { has_global_flag = random_terror_slot_jox_used } }
		AND = { original_tag = JPX NOT = { has_global_flag = random_terror_slot_jpx_used } }
	}
}

# STATE scope.  An actor can form from a populated state only when its owner
# controls it, it is not the owner's capital, and the owner keeps its own
# controlled capital, so the parent always survives the secession (Part 5).

	state_population_k > constant:random_terror_value.zero
	NOT = { is_capital = yes }
	is_controlled_by = OWNER

		NOT = { has_country_flag = random_terror_actor }
		any_owned_state = {
			is_capital = yes
			is_controlled_by = PREV
		}
	}
}
