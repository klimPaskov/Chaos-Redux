# ============================================================================
# CHAOS REDUX - CHEMICAL BATTLEFIELD RAID TARGET AND EQUIPMENT TRIGGERS
# ============================================================================
# Overview:
# - Validates native chemical land raid state targets and route support stores.
# - Route delivery, payload identity, and exact target policy are checked by
#   the native raid definition and shared chemical outcome adapter.
# ============================================================================

@CR_SC_CBRN_MATH_ZERO = 0

cbrn_battlefield_common_equipment_is_full = {
	check_variable = { var = num_equipment@gas_mask_equipment value = constant:cbrn_battlefield_operation_cost.cylinder_mask_crates compare = greater_than_or_equals }
	check_variable = { var = num_equipment@decontamination_equipment value = constant:cbrn_battlefield_operation_cost.common_decontamination_equipment compare = greater_than_or_equals }
	check_variable = { var = num_equipment@cbrn_instrument_equipment value = constant:cbrn_battlefield_operation_cost.common_instrument_equipment compare = greater_than_or_equals }
	check_variable = { var = num_equipment@support_equipment value = constant:cbrn_battlefield_operation_cost.common_support_equipment compare = greater_than_or_equals }
}

cbrn_battlefield_common_equipment_is_shortage_ready = {
	check_variable = { var = num_equipment@gas_mask_equipment value = constant:cbrn_battlefield_operation_cost.cylinder_mask_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@decontamination_equipment value = constant:cbrn_battlefield_operation_cost.common_decontamination_equipment_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@cbrn_instrument_equipment value = constant:cbrn_battlefield_operation_cost.common_instrument_equipment_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@support_equipment value = constant:cbrn_battlefield_operation_cost.common_support_equipment_shortage_floor compare = greater_than_or_equals }
}

cbrn_battlefield_projector_equipment_is_full = {
	cbrn_battlefield_common_equipment_is_full = yes
	check_variable = { var = num_equipment@gas_mask_equipment value = constant:cbrn_battlefield_operation_cost.projector_mask_crates compare = greater_than_or_equals }
	check_variable = { var = num_equipment@cbrn_instrument_equipment value = constant:cbrn_battlefield_operation_cost.projector_instrument_equipment compare = greater_than_or_equals }
	check_variable = { var = num_equipment@livens_projector_chassis value = constant:cbrn_battlefield_operation_cost.projector_chassis compare = greater_than_or_equals }
}

cbrn_battlefield_projector_equipment_is_shortage_ready = {
	cbrn_battlefield_common_equipment_is_shortage_ready = yes
	check_variable = { var = num_equipment@gas_mask_equipment value = constant:cbrn_battlefield_operation_cost.projector_mask_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@cbrn_instrument_equipment value = constant:cbrn_battlefield_operation_cost.projector_instrument_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@livens_projector_chassis value = constant:cbrn_battlefield_operation_cost.projector_chassis_shortage_floor compare = greater_than_or_equals }
}

cbrn_battlefield_projector_equipment_is_ready = {
	has_tech = livens_projector_tech
	OR = {
		cbrn_battlefield_projector_equipment_is_full = yes
		cbrn_battlefield_projector_equipment_is_shortage_ready = yes
	}
}

cbrn_battlefield_artillery_equipment_is_shortage_ready = {
	cbrn_battlefield_common_equipment_is_shortage_ready = yes
	check_variable = { var = num_equipment@gas_mask_equipment value = constant:cbrn_battlefield_operation_cost.artillery_mask_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@cbrn_instrument_equipment value = constant:cbrn_battlefield_operation_cost.artillery_instrument_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@support_equipment value = constant:cbrn_battlefield_operation_cost.artillery_support_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@decontamination_equipment value = constant:cbrn_battlefield_operation_cost.artillery_decontamination_shortage_floor compare = greater_than_or_equals }
	check_variable = { var = num_equipment@chemical_shell_lot_1 value = constant:cbrn_battlefield_operation_cost.artillery_ammunition_shortage_floor compare = greater_than_or_equals }
}

cbrn_battlefield_common_target_state = {
	impassable = no
	state_population_k > constant:cbrn_battlefield_operation_target.minimum_population_k
	CONTROLLER = { exists = yes }
}

cbrn_battlefield_target_has_enemy_divisions = {
	CONTROLLER = {
		divisions_in_state = { state = PREV size > @CR_SC_CBRN_MATH_ZERO }
	}
}

cbrn_battlefield_cylinder_target_state = {
	cbrn_battlefield_common_target_state = yes
	OR = {
		supply_node > constant:cbrn_battlefield_operation_target.minimum_supply_node
		cbrn_battlefield_target_has_enemy_divisions = yes
	}
}

cbrn_battlefield_projector_target_state = {
	cbrn_battlefield_common_target_state = yes
	OR = {
		supply_node > constant:cbrn_battlefield_operation_target.minimum_supply_node
		check_variable = { var = fort value = constant:cbrn_battlefield_operation_target.minimum_fort_level compare = greater_than }
		has_state_category = city
		has_state_category = large_city
		has_state_category = metropolis
		has_state_category = megalopolis
	}
}

cbrn_battlefield_artillery_target_state = {
	cbrn_battlefield_common_target_state = yes
	cbrn_battlefield_target_has_enemy_divisions = yes
}

cbrn_battlefield_armored_target_state = {
	cbrn_battlefield_common_target_state = yes
	AND = {
		cbrn_battlefield_target_has_enemy_divisions = yes
		OR = {
			supply_node > constant:cbrn_battlefield_operation_target.minimum_supply_node
			has_state_category = city
			has_state_category = large_city
			has_state_category = metropolis
			has_state_category = megalopolis
		}
	}
}
