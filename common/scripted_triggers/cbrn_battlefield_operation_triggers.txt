# ============================================================================
# CHAOS REDUX - EXACT-STATE CBRN BATTLEFIELD OPERATION TRIGGERS
# ============================================================================
# Overview:
# - Provides the CBRN-specific actor, selected-agent, equipment, cooldown, and
#   exact-state target gates for cylinder, projector, artillery, and armored
#   chemical operations.
# - A target is always a state selected by the player or by the native timed
#   decision target surface. No random state, capital proxy, combat estimator,
#   aircraft-presence test, or periodic world scan can satisfy these gates.
# - The shared chemical action validator remains the final policy, payload,
#   protection, condition, evidence, attribution, and consequence gate.
# - Shortage gates permit a deliberately weaker operation when the committed
#   route has only its central shortage floor. The operation effect records that
#   receipt and scales command, release efficiency, and friendly risk.
# ============================================================================

# The current game build exposes weather and terrain combat triggers only from
# combat/unit-leader scopes. It does not expose a verified selected-state Army
# Headquarters receipt for these timed operations. Keep the route fail-closed
# until a build-specific hook is proven; the operation must not activate from a
# fixed condition estimate.

	always = no

cbrn_battlefield_state_relationship_is_valid = {
	CONTROLLER = {
		exists = yes
		NOT = { tag = ROOT }
		NOT = { is_subject_of = ROOT }
		ROOT = {
			NOT = { is_in_faction_with = PREV }
			NOT = { is_subject_of = PREV }
			PREV = { set_temp_variable = { cbrn_battlefield_target_id = id } }
			OR = {
				AND = {
					has_war_with = PREV
					cbrn_policy_allows_battlefield_use_against_target_id = yes
				}
				AND = {
					NOT = { has_war_with = PREV }
					cbrn_policy_allows_extreme_use = yes
				}
			}
		}
	}
}
