	OR = { original_tag = ENG original_tag = RAJ original_tag = USA original_tag = FRA tag = VIC original_tag = ITA original_tag = BEL original_tag = COG }
}

camp_administration_civil_british_authority = {
	original_tag = ENG
	date > 1940.5.15
	has_war = yes
	OR = { has_government = democratic has_government = neutrality }
}

camp_administration_civil_raj_authority = {
	original_tag = RAJ
	is_subject_of = ENG
	ENG = { exists = yes OR = { has_government = democratic has_government = neutrality } }
	date > 1939.9.2
	has_war = yes
}

camp_administration_civil_congo_authority = {
	OR = { original_tag = BEL AND = { original_tag = COG is_subject_of = BEL } }
	OR = { has_government = democratic has_government = neutrality }
	OR = {
		any_controlled_state = { is_bel_congo_concession_pool_state = yes }
		any_subject_country = { any_controlled_state = { is_bel_congo_concession_pool_state = yes } }
	}
}

camp_administration_civil_custody_authority = {
	NOT = { has_country_flag = camp_rework_expansion_frozen }
	NOT = { has_country_flag = camp_admin_reforming }
	OR = {
		camp_administration_civil_british_authority = yes
		camp_administration_civil_raj_authority = yes
		camp_administration_civil_usa_authority = yes
		camp_administration_civil_french_refugee_authority = yes
		camp_administration_civil_french_collaboration = yes
		camp_administration_civil_italian_authority = yes
		camp_administration_civil_congo_authority = yes
	}
}

camp_administration_civil_host_state = {
	OR = {
		is_controlled_by = ROOT
		AND = { ROOT = { original_tag = BEL } is_bel_congo_concession_pool_state = yes CONTROLLER = { original_tag = COG is_subject_of = ROOT } }
	}
	OR = {
		AND = { ROOT = { original_tag = ENG } is_core_of = ENG is_controlled_by = ROOT }
		AND = { ROOT = { original_tag = RAJ } is_core_of = RAJ is_controlled_by = ROOT }
		AND = { ROOT = { original_tag = USA } is_core_of = USA OR = { is_usa_interior_relocation_site_state = yes is_usa_wartime_security_zone_state = yes } }
		AND = { ROOT = { OR = { original_tag = FRA tag = VIC } } OR = { is_france_camp_legacy_pool_state = yes is_france_north_africa_labor_pool_state = yes } }
		AND = { ROOT = { original_tag = ITA } OR = { is_core_of = ITA is_italy_libya_repression_pool_state = yes is_italy_east_africa_repression_pool_state = yes AND = { NOT = { is_core_of = ROOT } OR = { is_core_of = YUG is_core_of = GRE is_core_of = ALB } } } }
		AND = { ROOT = { OR = { original_tag = BEL original_tag = COG } } is_bel_congo_concession_pool_state = yes }
	}
}
