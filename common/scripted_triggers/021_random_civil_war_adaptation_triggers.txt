# COUNTRY scope. Regional responses require a live war and the enabled stage.
event021_country_can_adapt = {
	has_global_flag = random_civil_war_evolution_ii_active
	NOT = { has_global_flag = events_log_disabled_evolution_21_21_2 }
	has_country_flag = random_civil_war_active
	has_war = yes
	NOT = { has_country_flag = random_civil_war_settlement_phase }
	NOT = { has_country_flag = random_civil_war_reconstruction_phase }
	NOT = { random_civil_war_has_terminal_lock = yes }
}

event021_adaptation_affordable = {
	check_variable = { var = army_experience value = constant:event021_regional_response.adaptation_xp compare = greater_than_or_equals }
	NOT = { has_equipment = { infantry_equipment < constant:event021_regional_response.adaptation_rifles } }
}

# One completed primary order frees the optional response slot.
event021_campaign_response_room = {
	OR = {
		AND = {
			event021_country_can_manage_fronts = yes
			OR = {
				has_country_flag = event021_capital_action_used
				has_country_flag = event021_loyalty_action_used
				has_country_flag = event021_integration_action_used
				has_country_flag = event021_priority_front_action_used
			}
		}
		AND = {
			NOT = { event021_country_can_manage_fronts = yes }
			OR = {
				has_country_flag = event021_arsenal_action_used
				has_country_flag = event021_capital_action_used
				has_country_flag = event021_loyalty_action_used
				has_country_flag = event021_depot_action_used
			}
		}
	}
}

event021_adaptation_offered = {
	event021_country_can_adapt = yes
	event021_campaign_response_room = yes
	NOT = { has_country_flag = event021_campaign_adapted }
	event021_adaptation_affordable = yes
	NOT = { event021_expose_intervention_offered = yes }
}

event021_communications_response_slot_free = {
	event021_campaign_response_room = yes
	NOT = { event021_adaptation_offered = yes }
	NOT = { event021_mass_mobilization_offered = yes }
	NOT = { event021_expose_intervention_offered = yes }
}

event021_mass_mobilization_offered = {
	event021_country_can_adapt = yes
	event021_campaign_response_room = yes
	NOT = { event021_expose_intervention_offered = yes }
	event021_mass_mobilization_affordable = yes
	has_country_flag = event021_campaign_adapted
	NOT = { has_country_flag = event021_mass_mobilization_used }
	OR = {
		has_global_flag = random_civil_war_stage_totalen_chaos
		has_global_flag = random_civil_war_stage_world_collapse
	}
}

event021_mass_mobilization_affordable = {
	NOT = { has_equipment = { infantry_equipment < constant:event021_regional_response.mobilization_rifles } }
	NOT = { has_equipment = { support_equipment < constant:event021_regional_response.mobilization_support } }
}

event021_can_expose_intervention = {
	event021_country_can_adapt = yes
	event021_campaign_response_room = yes
	has_country_flag = random_civil_war_government_side
	event021_foreign_intervention_live = yes
	NOT = { has_country_flag = event021_foreign_intervention_exposed }
}

event021_expose_intervention_affordable = {
	NOT = { has_political_power < constant:event021_regional_response.expose_political }
	NOT = { has_equipment = { support_equipment < constant:event021_regional_response.expose_support } }
}

# An unaffordable sponsor response must not block campaign or communications work.
event021_expose_intervention_offered = {
	event021_can_expose_intervention = yes
	event021_expose_intervention_affordable = yes
}
