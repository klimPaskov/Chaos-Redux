# ============================================================================
# CHAOS REDUX - EVENT 035 GREAT DEPRESSION DECISION AND MISSION TRIGGERS
# ============================================================================
# Player and AI gates for the ordinary Great Depression category. The public
# Severity value is the only exposed crisis number. Centre, exposure, social,
# and supplier details remain qualitative flags or bounded receipt rows.
# ============================================================================

great_depression_call_contract_is_valid = {
	has_variable = great_depression_call_contract_supplied
	has_variable = great_depression_call_caller_type
	has_variable = great_depression_call_source_event_id
	has_variable = great_depression_call_source_episode_id
	has_variable = great_depression_call_receipt_id
	has_variable = great_depression_call_pacing_mode
	has_variable = great_depression_call_history_mode
	has_variable = great_depression_call_report_policy
	has_variable = great_depression_call_news_policy
	has_variable = great_depression_call_evolution_floor
	has_variable = great_depression_call_evolution_proof
	has_variable = great_depression_call_severity_input_mode
	has_variable = great_depression_call_starting_severity
	has_variable = great_depression_call_starting_severity_proof
	has_variable = great_depression_call_deepen_amount
	has_variable = great_depression_call_deepen_amount_proof
	has_variable = great_depression_call_severity_profile
	has_variable = great_depression_call_severity_profile_proof
	has_variable = great_depression_call_opening_shock_profile
	check_variable = {
		var = great_depression_call_contract_supplied
		value = constant:great_depression_api.contract_version
		compare = equals
	}
	check_variable = {
		var = great_depression_call_receipt_id
		value = constant:great_depression_api.minimum_receipt_id
		compare = greater_than_or_equals
	}
}

great_depression_call_severity_input_is_valid = {
	OR = {
		AND = {
			check_variable = { var = great_depression_call_severity_input_mode value = constant:great_depression_severity_input.starting_severity compare = equals }
			check_variable = { var = great_depression_call_starting_severity_proof value = constant:great_depression_api.one compare = greater_than_or_equals }
			check_variable = { var = great_depression_call_starting_severity value = constant:great_depression_event.public_value_minimum compare = greater_than_or_equals }
			check_variable = { var = great_depression_call_starting_severity value = constant:great_depression_event.public_value_maximum compare = less_than_or_equals }
			check_variable = { var = great_depression_call_deepen_amount value = constant:great_depression_api.zero compare = equals }
		}
		AND = {
			check_variable = { var = great_depression_call_severity_input_mode value = constant:great_depression_severity_input.deepen_amount compare = equals }
			check_variable = { var = great_depression_call_deepen_amount_proof value = constant:great_depression_api.one compare = greater_than_or_equals }
			check_variable = { var = great_depression_call_deepen_amount value = constant:great_depression_event.deepening_minimum compare = greater_than_or_equals }
			check_variable = { var = great_depression_call_deepen_amount value = constant:great_depression_event.deepening_maximum compare = less_than_or_equals }
			check_variable = { var = great_depression_call_starting_severity value = constant:great_depression_api.zero compare = equals }
		}
		AND = {
			check_variable = { var = great_depression_call_severity_input_mode value = constant:great_depression_severity_input.severity_profile compare = equals }
			check_variable = { var = great_depression_call_severity_profile_proof value = constant:great_depression_api.one compare = greater_than_or_equals }
			great_depression_call_severity_profile_is_known = yes
			check_variable = { var = great_depression_call_starting_severity value = constant:great_depression_api.zero compare = equals }
			check_variable = { var = great_depression_call_deepen_amount value = constant:great_depression_api.zero compare = equals }
		}
	}
}

great_depression_has_selected_center = {
	has_variable = great_depression_selected_center_state
	var:great_depression_selected_center_state = {
		exists = yes
		is_owned_by = ROOT
		is_controlled_by = ROOT
		has_state_flag = great_depression_center
	}
}

great_depression_can_manage = {
	great_depression_active_crisis_is_coherent = yes
	uses_normal_civilian_systems = yes
	NOT = { has_country_flag = great_depression_transition_pending }
}

great_depression_can_select_center = {
	great_depression_can_manage = yes
	check_variable = { var = great_depression_center_count value = constant:great_depression_api.zero compare = greater_than }
	NOT = { has_country_flag = great_depression_center_selection_lock }
	any_controlled_state = {
		has_state_flag = great_depression_center
		NOT = { has_state_flag = great_depression_center_liquidated }
		state_population_k > constant:great_depression_api.minimum_viable_state_population_k
		OR = {
			industrial_complex > constant:great_depression_api.zero
			arms_factory > constant:great_depression_api.zero
			dockyard > constant:great_depression_api.zero
		}
	}
}
