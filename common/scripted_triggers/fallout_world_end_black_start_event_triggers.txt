fallout_event_310_country_is_current = {
	fallout_event_registry_country_row_is_current = yes
	fallout_survival_country_identity_row_is_current = yes
	fallout_survival_country_resource_row_is_durable = yes
	has_country_flag = fallout_event_303_memory_closed
	has_variable = fallout_generation_change_count
	has_variable = air_winter_reclamation
	check_variable = { fallout_generation_change_count value = constant:fallout_event_303_threshold.minimum_generation_count compare = greater_than_or_equals }
	check_variable = { var = global.num_days value = constant:fallout_event_310_threshold.minimum_campaign_day compare = greater_than_or_equals }
	check_variable = { var = global.num_days value = constant:fallout_event_310_threshold.maximum_campaign_day compare = less_than }
	check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_310_threshold.minimum_power compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_310_threshold.minimum_scrap compare = greater_than_or_equals }
	check_variable = { fallout_survival_cohesion_current value = constant:fallout_event_310_threshold.minimum_cohesion compare = greater_than_or_equals }
	check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_310_threshold.minimum_recognition compare = greater_than_or_equals }
	check_variable = { air_winter_reclamation value = constant:fallout_event_310_threshold.minimum_reclamation compare = greater_than_or_equals }
	NOT = { has_country_flag = fallout_event_310_chain_committed }
	NOT = { has_country_flag = fallout_event_310_memory_closed }
	OR = {
		AND = {
			check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_310_cost.central_power compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_310_cost.central_scrap compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_310_cost.cooperative_power compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^4 value = constant:fallout_event_310_cost.cooperative_scrap compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^1 value = constant:fallout_event_310_cost.cooperative_food compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_310_cost.military_power compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^5 value = constant:fallout_event_310_cost.military_fuel compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { fallout_survival_current_entries^6 value = constant:fallout_event_310_cost.foreign_power compare = greater_than_or_equals }
			check_variable = { fallout_survival_current_entries^9 value = constant:fallout_event_310_cost.foreign_recognition compare = greater_than_or_equals }
		}
	}
