# ============================================================================
# CHAOS REDUX - EVENT 014 CROSS-EVENT INTEGRATION TRIGGERS
# ============================================================================
# Shared classification for human special-country exceptions, shortages,
# disasters, detention sites, and contamination. The locust flags are an
# explicit ingress contract for any crop-swarm system; current crop-collapse
# pressure is supplied by the live famine and natural-disaster systems.
# ============================================================================

cannibalism_is_ineligible_special_country = {
	is_special_chaos_country = yes
	NOT = { has_country_flag = fury_actor }
}

cannibalism_state_has_famine_pressure = {
	OR = {
		has_state_flag = natural_disaster_famine_pressure
		has_state_flag = genocide_soviet_famine_pressure_site
		has_state_flag = camp_rework_soviet_famine_pressure
		has_dynamic_modifier = { modifier = sov_famine_pressure_state }
	}
}

cannibalism_state_has_locust_pressure = {
	OR = {
		has_state_flag = locust_pressure
		has_state_flag = locust_swarm_active
		has_state_flag = natural_disaster_locust_pressure
	}
}

cannibalism_state_has_disease_pressure = {
	OR = {
		has_state_flag = natural_disaster_disease_pressure
		cannibalism_state_has_bioweapon_contamination = yes
	}
}

cannibalism_state_has_disaster_supply_pressure = {
	OR = {
		has_state_flag = natural_disaster_supply_collapse
		has_state_flag = natural_disaster_aftermath_active
		has_state_flag = natural_disaster_recovery_unresolved_territory
		has_state_flag = natural_disaster_refugee_pressure
	}
}

cannibalism_state_has_external_hunger_pressure = {
	OR = {
		cannibalism_state_has_famine_pressure = yes
		cannibalism_state_has_locust_pressure = yes
		cannibalism_state_has_disease_pressure = yes
		cannibalism_state_has_disaster_supply_pressure = yes
	}
}

cannibalism_state_has_severe_external_hunger_pressure = {
	OR = {
		cannibalism_state_has_famine_pressure = yes
		cannibalism_state_has_locust_pressure = yes
		has_state_flag = natural_disaster_supply_collapse
		cannibalism_state_has_severe_bioweapon_contamination = yes
	}
}

cannibalism_country_has_external_hunger_pressure = {
	OR = {
		check_variable = { famine_pressure > 0 }
		has_country_flag = sov_famine_crisis_active
		has_country_flag = locust_outbreak_active
		has_country_flag = natural_disaster_locust_outbreak_active
		any_controlled_state = { cannibalism_state_has_external_hunger_pressure = yes }
	}
}

cannibalism_country_has_severe_external_hunger_pressure = {
	OR = {
		check_variable = {
			var = famine_pressure
			value = constant:camp_rework_soviet.famine_pressure_critical
			compare = greater_than_or_equals
		}
		has_country_flag = locust_outbreak_severe
		any_controlled_state = { cannibalism_state_has_severe_external_hunger_pressure = yes }
	}
}

cannibalism_state_has_chemical_contamination = {
	mapmode_state_has_chemical_contamination = yes
}

cannibalism_state_has_severe_chemical_contamination = {
	cannibalism_state_has_chemical_contamination = yes
	has_variable = chem_state_contamination_strength
	check_variable = {
		var = chem_state_contamination_strength
		value = constant:cannibalism_integration.severe_chemical_strength
		compare = less_than_or_equals
	}
}

cannibalism_state_has_any_larder_contamination = {
	OR = {
		cannibalism_state_has_chemical_contamination = yes
		cannibalism_state_has_bioweapon_contamination = yes
		has_dynamic_modifier = { modifier = nuclear_fallout_state }
	}
}

cannibalism_state_has_active_detention_exploitation_link = {
	has_state_flag = cannibalism_feeding_program_link
	cannibalism_state_has_prison_or_camp = yes
	cannibalism_state_has_active_cell = yes
}

cannibalism_country_has_active_detention_exploitation_link = {
	any_controlled_state = { cannibalism_state_has_active_detention_exploitation_link = yes }
}

cannibalism_can_pay_selected_prisoner_feeding_cost = {
	is_cannibalism_active_country = yes
	OR = {
		has_country_flag = cannibalism_policy_exploitation
		has_country_flag = cannibalism_ritual_response_appropriation
		has_country_flag = cannibalism_network_response_exploitation
	}
	command_power > constant:cannibalism_integration.prisoner_feeding_command_cost
	NOT = { has_equipment = { support_equipment_1 < constant:cannibalism_integration.prisoner_feeding_support_cost } }
}

cannibalism_state_can_feed_selected_prisoners = {
	is_controlled_by = ROOT
	impassable = no
	cannibalism_state_has_prison_or_camp = yes
	cannibalism_state_has_active_cell = yes
	NOT = { cannibalism_state_is_unusable_larder = yes }
	NOT = { has_state_flag = cannibalism_consumption_cooldown }
	check_variable = {
		var = state_population_k
		value = constant:cannibalism_integration.prisoner_feeding_minimum_population_k
		compare = greater_than_or_equals
	}
}

cannibalism_state_is_empty_or_contaminated_expansion_target = {
	OR = {
		cannibalism_state_is_unusable_larder = yes
		cannibalism_state_has_severe_chemical_contamination = yes
		has_global_flag = air_contamination_irreversible
	}
}
