# Shared score quote for all condemnation response payments. The optional
# condemnation_cost_score_override temporary value lets a caller quote a
# target-owned score while retaining the same formula used by the payment
# effect. Output: condemnation_cost_score and condemnation_cost_factor.
condemnation_prepare_cost_multiplier = {
	set_temp_variable = { condemnation_cost_score = constant:condemnation_cost.cost_multiplier_score_minimum }
	if = {
		limit = { has_variable = condemnation_cost_score_override }
		set_temp_variable = { condemnation_cost_score = condemnation_cost_score_override }
	}
	else_if = {
		limit = { has_variable = condemnation_total }
		set_temp_variable = { condemnation_cost_score = condemnation_total }
	}
	clamp_temp_variable = {
		var = condemnation_cost_score
		min = constant:condemnation_cost.cost_multiplier_score_minimum
		max = constant:condemnation_cost.cost_multiplier_score_maximum
	}
	set_temp_variable = { condemnation_cost_factor = constant:condemnation_cost.cost_multiplier_base }
	multiply_temp_variable = { condemnation_cost_factor = constant:condemnation_cost.cost_multiplier_score_divisor }
	add_to_temp_variable = { condemnation_cost_factor = condemnation_cost_score }
	always = yes
}

# Input: condemnation_unscaled_cost. Output: condemnation_scaled_cost. The
# rounded quotient is compared with its exact numerator so positive integer
# equipment and experience costs are rounded up, while exact whole multiples
# remain unchanged.
condemnation_prepare_scaled_integer_cost = {
	condemnation_prepare_cost_multiplier = yes
	set_temp_variable = { condemnation_scaled_cost_numerator = condemnation_unscaled_cost }
	multiply_temp_variable = { condemnation_scaled_cost_numerator = condemnation_cost_factor }
	set_temp_variable = { condemnation_scaled_cost = condemnation_scaled_cost_numerator }
	divide_temp_variable = { condemnation_scaled_cost = constant:condemnation_cost.cost_multiplier_score_divisor }
	round_temp_variable = condemnation_scaled_cost
	set_temp_variable = { condemnation_scaled_cost_check = condemnation_scaled_cost }
	multiply_temp_variable = { condemnation_scaled_cost_check = constant:condemnation_cost.cost_multiplier_score_divisor }
	if = {
		limit = { check_variable = { condemnation_scaled_cost_check < condemnation_scaled_cost_numerator } }
		add_to_temp_variable = { condemnation_scaled_cost = constant:condemnation_cost.cost_integer_step }
	}
	always = yes
}

# Input: condemnation_unscaled_cost. Output: condemnation_scaled_cost. Fuel
# fractions keep their fixed-point precision; no integer rounding is applied.
condemnation_prepare_scaled_fraction_cost = {
	condemnation_prepare_cost_multiplier = yes
	set_temp_variable = { condemnation_scaled_cost = condemnation_unscaled_cost }
	multiply_temp_variable = { condemnation_scaled_cost = condemnation_cost_factor }
	divide_temp_variable = { condemnation_scaled_cost = constant:condemnation_cost.cost_multiplier_score_divisor }
	always = yes
}

# Inspection burden is 5% at score 0 and 15% at score 1000. The existing
# inspection idea supplies the fixed 5% base; this variable supplies the
# score-driven increment through the inspection pulse.
condemnation_prepare_inspection_consumer_goods = {
	condemnation_prepare_cost_multiplier = yes
	set_temp_variable = { condemnation_inspection_consumer_goods_increment_temp = condemnation_cost_score }
	multiply_temp_variable = { condemnation_inspection_consumer_goods_increment_temp = constant:condemnation_cost.inspection_consumer_goods_per_score }
	set_temp_variable = { condemnation_inspection_consumer_goods_total = constant:condemnation_cost.inspection_consumer_goods_minimum }
	add_to_temp_variable = { condemnation_inspection_consumer_goods_total = condemnation_inspection_consumer_goods_increment_temp }
	clamp_temp_variable = {
		var = condemnation_inspection_consumer_goods_total
		min = constant:condemnation_cost.inspection_consumer_goods_minimum
		max = constant:condemnation_cost.inspection_consumer_goods_maximum
	}
	always = yes
}

# Targeted participant decisions run in the acting country scope while their
# quote and payment belong to FROM. Keep the available and custom-cost checks
# on the same target score before invoking a concrete payment predicate.
condemnation_set_target_cost_score = {
	set_temp_variable = { condemnation_cost_score_override = 0 }
	if = {
		limit = { FROM = { has_variable = condemnation_total } }
		set_temp_variable = { condemnation_cost_score_override = FROM.condemnation_total }
	}
	always = yes
}

	has_variable = condemnation_last_victim
	var:condemnation_last_victim = { exists = yes }
	set_temp_variable = { condemnation_unscaled_cost = condemnation_tier }
	multiply_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_convoys_per_tier }
	add_to_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_convoys_base }
	condemnation_prepare_scaled_integer_cost = yes
	NOT = { has_equipment = { convoy_1 < condemnation_scaled_cost } }
	set_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_infantry_equipment }
	condemnation_prepare_scaled_integer_cost = yes
	NOT = { has_equipment = { infantry_equipment < condemnation_scaled_cost } }
	set_temp_variable = { condemnation_unscaled_cost = constant:condemnation_cost.compensation_fuel_fraction }
	condemnation_prepare_scaled_fraction_cost = yes
	NOT = { fuel_ratio < condemnation_scaled_cost }
}
