# ============================================================================
# CHAOS REDUX - EVENT 006 READY FORMABLE STATE PUZZLE CONTRACT
# ============================================================================
# Overview:
# - Exposes exact anchor-owner proofs for the FORM-17, FORM-27, FORM-29,
#   FORM-31, FORM-32, FORM-36, and FORM-37 state puzzle rows, before formation
#   and after formation.
# - Pre-formation proofs use the current bounded Event 006 member contract;
#   post-formation proofs use the committed carrier/member contract and current
#   active-country registry generation.
# - FORM-32's Punjab row has two mutually exclusive choices, West Punjab 440 or
#   East Punjab 986, selected only by FDX's stored loader anchor.
# - No world scan, fallback member, state transfer, or new tag is introduced.
# ============================================================================

has_independence_wave_ready_formable_route_selector = {
	is_independence_wave_regional_power = yes
	is_independence_wave_active_country = yes
	NOT = { has_country_flag = independence_wave_formable_family_selected }
	NOT = { has_country_flag = independence_wave_formable_active }
	NOT = { has_country_flag = independence_wave_formable_founding_proposal_active }
	NOT = { has_independence_wave_formable_transaction_lock = yes }
	NOT = { has_valid_independence_wave_formable_founding_invitation = yes }
	OR = {
		AND = { is_independence_wave_form17_member = yes NOT = { has_global_flag = independence_wave_form17_identity_in_use } }
		AND = { is_independence_wave_form32_member = yes NOT = { has_global_flag = independence_wave_form32_identity_in_use } }
		AND = { is_independence_wave_form36_member = yes NOT = { has_global_flag = independence_wave_form36_identity_in_use } }
		AND = { is_independence_wave_form37_member = yes NOT = { has_global_flag = independence_wave_form37_identity_in_use } }

		AND = { is_independence_wave_form29_member = yes NOT = { has_global_flag = independence_wave_form29_identity_in_use } }
		AND = { is_independence_wave_form31_member = yes NOT = { has_global_flag = independence_wave_form31_identity_in_use } }
	}
}

# Candidate-country owner gates support either the unselected carrier's full
# live member roster or the selected family, then switch to exact committed
# carrier/member generations after formation.
# These owner helpers are called through fixed state -> owner scopes; PREV.PREV
# is the actual querying carrier even when ROOT belongs to a recipient event.
has_independence_wave_form17_puzzle_owner_from_root = {
	OR = {
		AND = {
			PREV.PREV = {
				NOT = { has_country_flag = independence_wave_formable_active }
				OR = {
					NOT = { has_country_flag = independence_wave_formable_family_selected }
					check_variable = { independence_wave_formable_profile_family = constant:independence_wave_formable_family.north_caucasian_federation }
				}
			}
			is_independence_wave_form17_member = yes
		}
		AND = {
			PREV.PREV = { is_independence_wave_form17_postformation_carrier = yes }
			OR = {
				is_independence_wave_form17_postformation_carrier = yes
				is_independence_wave_ready_formable_current_contract_member = yes
			}
		}
	}
	has_independence_wave_ready_formable_connection_to_state_query_carrier = yes
}
