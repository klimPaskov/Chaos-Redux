# ============================================================================
# CHAOS REDUX - EVENT 020 EVOLUTION TRIGGERS
# ============================================================================
# The five entries below are true evolutions. Threatened, Incubating, Infected,
# Severe Crisis, Collapsed, Contained, Recovery, Cured, and Rat-Controlled are
# ordinary disease phases and never satisfy an evolution-history check.
# ============================================================================

black_plague_state_is_infected_or_worse = {
	OR = {
		black_plague_state_is_infected = yes
		black_plague_state_is_severe = yes
		black_plague_state_is_collapsed = yes
		black_plague_state_is_contained = yes
		black_plague_state_is_recovering = yes
		black_plague_state_is_rat_controlled = yes
	}
}

black_plague_evolution_runtime_has_active_outbreak = {
	has_global_flag = black_plague_system_started
	has_global_flag = black_plague_system_active
	NOT = { has_global_flag = black_plague_world_end_complete }
}

black_plague_evolution_i_is_ready = {
	black_plague_evolution_runtime_has_active_outbreak = yes
	NOT = { has_global_flag = black_plague_evolution_i_active }
	NOT = { has_global_flag = black_plague_evolution_i_recorded }
	check_variable = {
		var = global.chaos_meter_value
		value = constant:black_plague_rat_evolution.evolution_i_chaos_gate
	compare = greater_than_or_equals
	}
	any_state = { black_plague_state_is_infected_or_worse = yes }
	OR = {
		check_variable = {
			var = global.black_plague_deaths_total
			value = constant:black_plague_evolution_runtime.minimum_deaths_for_i
			compare = greater_than_or_equals
		}
		check_variable = {
			var = global.black_plague_established_state_count
			value = constant:black_plague_rat_evolution.minimum_established_states
			compare = greater_than_or_equals
		}
		any_state = { black_plague_state_is_collapsed = yes }
	}
}
