# ============================================================================
# CHAOS REDUX - EVENT 021 LIVE EVIDENCE TRIGGERS
# ============================================================================
# A route is usable only while its source receipt is live. The parent planner
# owns the receipt refresh; these predicates are read-only and fail closed when
# an expiry, strength, or source proof is missing.
# ============================================================================

random_civil_war_live_evidence_receipt = {
	has_variable = random_civil_war_evidence_last_start_date
	has_variable = random_civil_war_evidence_cleanup_owner
}

random_civil_war_has_live_ideological_evidence = {
	has_country_flag = random_civil_war_ideological_evidence_active
	has_variable = random_civil_war_ideological_evidence_strength
	check_variable = { var = random_civil_war_ideological_evidence_strength value = constant:event021_evidence_tuning.minimum_live_strength compare = greater_than_or_equals }
	has_variable = random_civil_war_ideological_evidence_until
	check_variable = { var = global.date value = random_civil_war_ideological_evidence_until compare = less_than }
}
