	is_independence_wave_regional_framework_live = yes
	has_variable = independence_wave_framework_package_id
	check_variable = { var = independence_wave_framework_package_id value = independence_wave_package_id compare = equals }
	has_independence_wave_force_package_for_current_generation = yes
	has_independence_wave_regional_framework_anchor_capital = yes
	OR = {
		AND = { original_tag = DSX check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_097 } }
		AND = { original_tag = ATJ check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_150 } }
		AND = { original_tag = IAX check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_161 } }
	}
}
