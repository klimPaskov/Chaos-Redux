# State scope, exact normalized biological record before its commit.
national_policy_biological_seed_policy_is_valid = {
	event_target:bio_seed_victim = { exists = yes set_temp_variable = { cbrn_retaliation_target_id = id } }
	event_target:bio_seed_actor = {
		exists = yes
		has_war_with = event_target:bio_seed_victim
		if = {
			limit = { check_variable = { var = bio_seed_route value = constant:bio_lifecycle_route.battlefield_dissemination compare = equals } }
			national_policy_biological_allows_battlefield_use_against_target_id = yes
		}
		else = { national_policy_biological_allows_strategic_use_against_target_id = yes }
	}
}

		limit = { check_variable = { var = sov_nuclear_bombs_action_type value = constant:sov_nuclear_bombs_action.combat_strike compare = equals } }
		has_variable = sov_nuclear_bombs_action_target
		var:sov_nuclear_bombs_action_target = { exists = yes set_temp_variable = { cbrn_retaliation_target_id = id } }
		has_war_with = var:sov_nuclear_bombs_action_target
		national_policy_nuclear_allows_use_against_target_id = yes
	}
}
