# ============================================================================
# CHAOS REDUX - EVENT 006 FORM-08 DANUBIAN CONFEDERATION TRIGGERS
# ============================================================================
# FORM-08 is a negotiated Danube-corridor federation. The helper currently
# admits only the researched selectable Event 006 anchors (Transylvania and
# Banat). Vojvodina remains a vanilla HUN-origin overlay and Slavonia has no
# unique installed-map anchor, so neither is promoted into this ledger. Every
# member stays sovereign until an explicit consent row authorizes integration,
# and the family remains fail-closed until its minimum three members, consents,
# and unique anchors are present in the frozen registry ledger.
# ============================================================================

is_independence_wave_form08_eligible_member = {
	exists = yes
	is_independence_wave_active_country = yes
	is_subject = no
	NOT = { has_country_flag = independence_wave_form0124_autonomous_member }
	NOT = { has_country_flag = independence_wave_formable_active }
	NOT = { has_country_flag = independence_wave_formable_founding_proposal_active }
	NOT = { has_country_flag = soviet_collapse_active_origin }
	has_country_flag = independence_wave_formable_family_selected
	check_variable = { independence_wave_formable_family = constant:independence_wave_formable_family.danubian_confederation }
	has_variable = independence_wave_anchor_state
	OR = {
		AND = {
			original_tag = TRA
			check_variable = { independence_wave_anchor_state = 84 }
			owns_state = 84
			controls_state = 84
			capital_scope = { state = 84 }
		}
		AND = {
			original_tag = AXX
			check_variable = { independence_wave_anchor_state = 82 }
			owns_state = 82
			controls_state = 82
			capital_scope = { state = 82 }
		}
	}
}

has_independence_wave_form08_exact_carrier_anchor = {
	is_independence_wave_form08_eligible_member = yes
	has_independence_wave_form08_identity_clearance = yes
}

# Vanilla HUN, Habsburg, and Austria-Hungary routes can already claim the
# HUN_EMPIRE cosmetic identity. FORM-08 must not silently reuse that identity
# while another country is carrying it, so admission is fail-closed until the
# installed world has no live carrier and the vanilla formation flag is clear.
has_independence_wave_form08_identity_clearance = {
	NOT = { has_global_flag = independence_wave_form08_hun_empire_identity_in_use }
	NOT = { has_global_flag = form_austria_hungary_flag }
	NOT = { any_country = { has_cosmetic_tag = HUN_EMPIRE } }
}

		has_non_aggression_pact_with = var:independence_wave_formable_inviting_carrier
		has_military_access_to = var:independence_wave_formable_inviting_carrier
