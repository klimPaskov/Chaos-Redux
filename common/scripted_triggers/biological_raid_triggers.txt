# State scope with the acting country as PREV. ROOT may remain a raid instance
# when this helper is reached through the actor's stored staging-state scope.
bio_strategic_raid_state_is_valid_staging_candidate = {
	is_controlled_by = PREV
	impassable = no
	air_base > constant:bio_strategic_raid_staging.air_base_trigger_floor
	infrastructure > constant:bio_strategic_raid_staging.infrastructure_trigger_floor
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
}

# Country scope. The stored scope is the only accepted attacker-accident state.
bio_strategic_raid_actor_has_valid_staging_state = {
	has_variable = bio_strategic_raid_staging_state
	var:bio_strategic_raid_staging_state = {
		bio_strategic_raid_state_is_valid_staging_candidate = yes
	}
}

# Raid-instance scope after the native actor, victim, and target state have been
# saved as regular event targets by the resolver.
bio_strategic_raid_native_context_is_valid = {
	has_event_target = bio_raid_actor
	has_event_target = bio_raid_victim
	has_event_target = bio_raid_target_state
	bio_strategic_raid_agent_is_valid = yes
	bio_strategic_raid_engine_outcome_is_valid = yes
	event_target:bio_raid_actor = {
		exists = yes
		has_war_with = event_target:bio_raid_victim
		NOT = { is_in_faction_with = event_target:bio_raid_victim }
		NOT = { is_subject_of = event_target:bio_raid_victim }
		bio_strategic_raid_actor_has_agent_project = yes
		bio_strategic_raid_actor_has_valid_staging_state = yes
		event_target:bio_raid_victim = {
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}
		cbrn_policy_allows_strategic_use_against_target_id = yes
		has_variable = chemical_readiness
		has_variable = cbrn_biological_security
		has_variable = cbrn_attribution_control
	}
	event_target:bio_raid_victim = {
		exists = yes
		NOT = { is_subject_of = event_target:bio_raid_actor }
	}
	event_target:bio_raid_target_state = {
		bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
		CONTROLLER = { tag = event_target:bio_raid_victim }
	}
}

# Raid-instance scope. Outcome tooltips use only the native current-version
# raid variables because resolver-owned event targets and temporary agent codes
# do not exist while the result UI is being assembled. Immutable project and
# outcome identity remain enforced by the resolver itself.
bio_strategic_raid_native_tooltip_context_is_valid = {
	var:actor_country = {
		exists = yes
		has_war_with = var:ROOT.victim_country
		NOT = { is_in_faction_with = var:ROOT.victim_country }
		NOT = { is_subject_of = var:ROOT.victim_country }
		bio_strategic_raid_actor_has_valid_staging_state = yes
		var:ROOT.victim_country = {
			set_temp_variable = { cbrn_retaliation_target_id = id }
		}
		cbrn_policy_allows_strategic_use_against_target_id = yes
		has_variable = chemical_readiness
		has_variable = cbrn_biological_security
		has_variable = cbrn_attribution_control
	}
	var:victim_country = {
		exists = yes
		NOT = { is_subject_of = var:ROOT.actor_country }
	}
	var:target_state = {
		bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
		is_controlled_by = var:ROOT.victim_country
	}
}

# Country scope. This is the ordinary AI safety floor: one real handling tier
# and one real domestic containment tier. The explicit desperate route below
# is the only permission to cross this floor.
bio_strategic_raid_ai_has_safe_domestic_program = {
	has_tech = pathogen_handling_protocols
	has_tech = rapid_outbreak_response
}

# Country scope. Desperation never creates policy authority or payload. It only
# waives the ordinary AI safety floor after an unrestricted route already owns
# both the doctrine authorization and near-capitulation condition.
bio_strategic_raid_ai_is_desperate_route = {
	cbrn_ai_route_allows_unrestricted_use = yes
	cbrn_policy_allows_extreme_use = yes
	surrender_progress > constant:bio_strategic_raid_ai.desperate_surrender_threshold
}

# Country scope with FROM as the proposed victim. This is AI-only. Player raid
# availability continues to use the same project, policy, staging, war, target,
# aircraft, and native equipment-reservation gates without an ideology lock.

		bio_strategic_raid_ai_has_safe_domestic_program = yes

		}
		cbrn_ai_route_allows_first_use = yes
		bio_strategic_raid_ai_is_desperate_route = yes
	}
}
