# COUNTRY scope. Compatible Event 006 identity overlaps reuse their canonical
# carriers. Manden, Aksum, Nubia, and Great Zimbabwe remain on their existing
# vanilla carriers because their Event 012 identities do not have a compatible
# Event 006 row. Surveying any carrier records only Event 012 identity flags;
# it never transfers territory or grants cores.
africa_priority_member_has_supported_carrier_identity = {
	NOT = { is_independence_wave_registry_event6_origin = yes }
	NOT = { is_independence_wave_registry_soviet_origin = yes }
	OR = {
		africa_priority_member_origin_is_asante = yes
		africa_priority_member_origin_is_oyo = yes
		is_independence_wave_registry_africa_sokoto_carrier = yes
		africa_priority_member_origin_is_kanem_bornu = yes
		is_independence_wave_registry_africa_nonmatching_vanilla_carrier = yes
		africa_priority_member_origin_is_kongo = yes
		is_independence_wave_registry_africa_buganda_carrier = yes
		is_independence_wave_registry_africa_harar_carrier = yes
		africa_priority_member_origin_is_kilwa = yes
		africa_priority_member_origin_is_luba = yes
		africa_priority_member_origin_is_lunda = yes
		is_independence_wave_registry_africa_merina_carrier = yes
		africa_priority_member_origin_is_zulu = yes
	}
}

africa_priority_member_has_viable_compact_base = {
	exists = yes
	capital_scope = {
		is_on_continent = africa
		is_owned_by = ROOT
		is_controlled_by = ROOT
	}
	NOT = { has_capitulated = yes }
}

africa_priority_member_has_functioning_institution = {
	africa_priority_member_has_supported_carrier_identity = yes
	has_stability > constant:africa_priority_member_promotion.institution_stability_floor
	NOT = { has_country_flag = africa_priority_institution_destroyed }
}

africa_priority_member_has_documented_strategic_function = {
	africa_priority_member_has_supported_carrier_identity = yes
	OR = {
		num_of_factories > constant:africa_value.zero
		any_owned_state = {
			OR = {
				infrastructure > constant:africa_priority_member_promotion.infrastructure_floor
				state_population_k > constant:africa_priority_member_promotion.population_floor
			}
		}
	}
}

africa_priority_member_has_unresolved_league_problem = {
	NOT = { has_country_flag = africa_priority_member_package_active }
	NOT = { has_country_flag = africa_member_final_integration_consent }
	NOT = { africa_relationship_is_integrated_region = yes }
}

africa_priority_member_has_viable_population_and_infrastructure = {
	any_owned_state = {
		state_population_k > constant:africa_priority_member_promotion.population_floor
		infrastructure > constant:africa_priority_member_promotion.infrastructure_floor
	}
}

africa_priority_member_has_distinct_package_identity = {
	africa_priority_member_has_supported_carrier_identity = yes
	NOT = { has_country_flag = africa_priority_package_duplicates_existing_role }
}

africa_priority_member_promotion_survey_is_available = {
	exists = yes
	africa_priority_member_has_supported_carrier_identity = yes
	africa_priority_member_has_viable_compact_base = yes
	NOT = { africa_is_current_host = yes }
	NOT = { has_country_flag = africa_priority_member_package_active }
	NOT = { has_country_flag = africa_priority_promotion_requalification_pending }
	NOT = { has_country_flag = africa_action_record_active }
}

# COUNTRY scope after africa_priority_member_evaluate_promotion_candidate. The
# six condition flags and count are visible dossier evidence; support is a
# separate gate and cannot be replaced by opinion.
africa_priority_member_can_be_promoted = {
	has_country_flag = africa_priority_promotion_candidate
	africa_priority_member_has_valid_origin = yes
	check_variable = {
		var = africa_priority_promotion_criteria_count
		value = constant:africa_priority_member_promotion.required_criteria
		compare = greater_than_or_equals
	}
	check_variable = {
		var = africa_priority_local_support
		value = constant:africa_action_threshold.minimum_confidence
		compare = greater_than_or_equals
	}
	NOT = { has_country_flag = africa_priority_member_package_active }
	NOT = { has_country_flag = africa_priority_promotion_requalification_pending }
}

africa_priority_member_can_resolve_requalification = {
	has_country_flag = africa_priority_promotion_requalification_pending
	NOT = { has_country_flag = africa_priority_member_package_active }
	OR = {
		AND = {
			has_country_flag = africa_priority_requalification_access
			num_of_factories > constant:africa_value.zero
			any_owned_state = { infrastructure > constant:africa_priority_member_promotion.infrastructure_floor }
		}
		AND = {
			has_country_flag = africa_priority_requalification_overlap
			has_war = no
			has_event_target = africa_host
			event_target:africa_host = { has_country_flag = africa_league_congress_unlocked }
		}
		AND = {
			has_country_flag = africa_priority_requalification_local_ratification
			has_stability > constant:africa_priority_member_promotion.medium_support_stability
		}
	}
}

africa_priority_member_has_package = {
	has_country_flag = africa_priority_member_package_active
	has_variable = africa_priority_member_package_id
	check_variable = { africa_priority_member_package_id > constant:africa_priority_member_package.none }
	check_variable = {
		var = africa_priority_member_package_id
		value = constant:africa_priority_member_package.zulu
		compare = less_than_or_equals
	}
}

# COUNTRY scope. Only a full Action 102 result activates the complete package.
# Partial autonomy remains a protected or associated compact settlement until
# its recorded requalification problem is resolved and Action 102 succeeds.
africa_priority_member_can_register_package = {
	has_global_flag = africa_event_active
	has_event_target = africa_host
	event_target:africa_host = { has_country_flag = africa_host_commit_completed }
	NOT = { has_country_flag = africa_priority_member_package_active }
	# Event 006 origins keep their Event 006 package, tree, decisions, AI, and
	# formable routes. An identity match is not permission for Event 012 to
	# replace the creating event's package.
	NOT = { has_country_flag = independence_wave_active_origin }
	NOT = { check_variable = { var = liberation_origin value = constant:liberation_origin.independence_wave compare = equals } }
	NOT = { is_independence_wave_registry_soviet_origin = yes }
	has_country_flag = africa_priority_package_promotion_approved
	africa_priority_member_has_valid_origin = yes
}

africa_priority_member_can_open_political_settlement = {
	africa_priority_member_has_package = yes
	has_event_target = africa_host
	event_target:africa_host = { has_country_flag = africa_host_commit_completed }
	NOT = { has_country_flag = africa_priority_member_politics_settled }
}

africa_priority_member_can_advance_mechanic = {
	africa_priority_member_has_package = yes
	has_event_target = africa_host
	event_target:africa_host = { has_country_flag = africa_host_commit_completed }
	has_country_flag = africa_priority_member_politics_settled
	OR = {
		NOT = { has_variable = africa_priority_member_mechanic_progress }
		check_variable = { africa_priority_member_mechanic_progress < constant:africa_priority_member_progress.maximum }
	}
}

africa_priority_member_can_reinforce_force = {
	africa_priority_member_has_package = yes
	has_country_flag = africa_priority_member_politics_settled
	has_manpower > constant:africa_priority_member_manpower.force_gate
	OR = {
		NOT = { has_variable = africa_priority_member_security_readiness }
		check_variable = { africa_priority_member_security_readiness < constant:africa_priority_member_progress.maximum }
	}
}

africa_priority_member_can_negotiate_league_role = {
	africa_priority_member_has_package = yes
	has_event_target = africa_host
	event_target:africa_host = { has_country_flag = africa_host_commit_completed }
	NOT = { africa_relationship_is_occupied_settlement = yes }
	NOT = { has_country_flag = africa_priority_member_withdrawal_in_progress }
	NOT = { has_country_flag = africa_priority_member_preferred_relationship_reached }
	OR = {
		NOT = { has_country_flag = africa_priority_member_autonomy_review_pending }
		africa_member_can_enter_autonomous_federal = yes
	}
}
