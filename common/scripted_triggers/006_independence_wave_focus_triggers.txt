# ============================================================================
# CHAOS REDUX - EVENT 006 INDEPENDENCE WAVE FOCUS TRIGGERS
# ============================================================================
# Scope: country unless stated otherwise.
# These triggers gate the full Event 006 tree, the additive shared overlay,
# government route locks, former-host route families, regional ambitions,
# signature extension modules, league work, formables, and hidden high-chaos
# content. Package adapters publish explicit availability flags. No trigger
# scans every country or mutates state.
# ============================================================================

# The synchronized release transaction assigns focus ownership while a country
# still carries the prepared-origin receipt. Activation happens only after
# every selected package has passed setup, so assignment must accept both
# phases without weakening the player-facing focus gates below.
is_independence_wave_focus_assignment_scope = {
	OR = {
		is_independence_wave_active_country = yes
		AND = {
			exists = yes
			has_country_flag = independence_wave_origin_prepared
			check_variable = {
				var = liberation_origin
				value = constant:liberation_origin.independence_wave
				compare = equals
			}
			NOT = { has_country_flag = independence_wave_origin_ended }
		}
	}
}

can_use_independence_wave_full_focus_framework = {
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_full_focus_framework
	NOT = { has_country_flag = independence_wave_additive_focus_overlay }
}

can_use_independence_wave_additive_focus_overlay = {
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_additive_focus_overlay
	NOT = { has_country_flag = independence_wave_full_focus_framework }
}

has_valid_independence_wave_focus_assignment_input = {
	check_variable = {
		var = independence_wave_focus_assignment_input
		value = constant:independence_wave_focus_assignment.full_framework
		compare = greater_than_or_equals
	}
	check_variable = {
		var = independence_wave_focus_assignment_input
		value = constant:independence_wave_focus_assignment.post_formation_overlay
		compare = less_than_or_equals
	}
}

can_complete_independence_wave_survival_capstone = {
	can_use_independence_wave_full_focus_framework = yes
	has_country_flag = independence_wave_dm01_capital_secured
	capital_scope = { is_controlled_by = ROOT }
	has_independence_wave_established_legitimacy = yes
	has_independence_wave_functioning_capacity = yes
}

has_independence_wave_internal_power_struggle = {
	has_country_flag = independence_wave_internal_power_struggle_registered
	check_variable = {
		var = independence_wave_power_struggle_type
		value = constant:independence_wave_power_struggle.civilians_vs_army

	check_variable = {
		var = independence_wave_power_struggle_type
		value = constant:independence_wave_power_struggle.municipal_commission_vs_industrial_security
		compare = less_than_or_equals
	}
}
