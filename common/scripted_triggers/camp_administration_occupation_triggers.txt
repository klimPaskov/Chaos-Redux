camp_occ_authority_has_campaign = {
	camp_admin_country_regime_valid = yes
	NOT = { has_country_flag = camp_occ_country_closed }
	OR = {
		AND = { original_tag = JAP date > 1937.7.6 check_variable = { camp_admin_mandate > 1 } }
		AND = { camp_admin_country_nazi_authority = yes date > 1941.6.21 check_variable = { camp_admin_mandate = 3 } }
	}
}

	has_state_flag = camp_occ_registered

	var:camp_occ_responsible_country = { exists = yes camp_occ_authority_has_campaign = yes }
	is_controlled_by = var:camp_occ_responsible_country
}
