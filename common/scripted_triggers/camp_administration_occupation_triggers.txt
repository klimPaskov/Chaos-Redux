# STATE, row origin. In western, northern and southern Europe the German campaign
# reaches an origin only after it has had time to organize deportations and German
# authority over the state has consolidated (camp_occ_timing). Denmark's
# cooperating government shields Danish Jews while it still exists unbeaten and
# at peace with Germany. In the east the campaign strikes as soon as it opens,
# and home German origins and the China theater are not paced.
camp_occ_origin_is_paced_deportation_country = {
	OR = { is_core_of = FRA is_core_of = HOL is_core_of = BEL is_core_of = LUX is_core_of = NOR is_core_of = DEN is_core_of = GRE is_core_of = ITA }

		NOT = { camp_occ_state_authority_consolidated = yes }
