# ============================================================================
# CHAOS REDUX - EVENT 030 TIME TRAVELER TRIGGERS
# ============================================================================
# Scope notes: "country" triggers are read in a country scope, "global" ones
# in any scope. The custodian is the one country that physically holds the
# original traveler and carries the flag time_traveler_custodian. The three
# public values are Credibility and Exposure (country variables on the
# custodian) and Timeline Divergence (global.time_traveler_divergence).
# Triggers used inside available or visible blocks of decisions keep their
# bookkeeping in hidden_trigger so no raw variable name reaches a tooltip.
# ============================================================================

# ---------------------------------------------------------------------------
# Custody and status
# ---------------------------------------------------------------------------

# COUNTRY. The country that physically holds the original traveler.
time_traveler_is_custodian = {
	has_country_flag = time_traveler_custodian
}

# GLOBAL. The traveler is verified and available to the custodian.
time_traveler_status_is_active = {
	check_variable = { global.time_traveler_status = constant:time_traveler_status.active }
}

# GLOBAL. The traveler exists in the campaign and has not been lost.
time_traveler_is_present = {
	OR = {
		check_variable = { global.time_traveler_status = constant:time_traveler_status.arriving }
		check_variable = { global.time_traveler_status = constant:time_traveler_status.verifying }
		check_variable = { global.time_traveler_status = constant:time_traveler_status.choosing_access }
		check_variable = { global.time_traveler_status = constant:time_traveler_status.active }
	}
}

# COUNTRY. Direct access: the custodian can consult the traveler.
time_traveler_has_direct_access = {
	has_country_flag = time_traveler_custodian
	hidden_trigger = { check_variable = { global.time_traveler_status = constant:time_traveler_status.active } }
}

# COUNTRY. The main decision category is shown to the custodian.
time_traveler_category_is_visible = {
	has_country_flag = time_traveler_custodian
	hidden_trigger = {
		OR = {
			check_variable = { global.time_traveler_status = constant:time_traveler_status.choosing_access }
			check_variable = { global.time_traveler_status = constant:time_traveler_status.active }
		}
	}
}

# COUNTRY. A consultation mission is running.
time_traveler_mission_is_active = {
	OR = {
		has_active_mission = time_traveler_mission_garrison
		has_active_mission = time_traveler_mission_build
		has_active_mission = time_traveler_mission_research
		has_active_mission = time_traveler_mission_relation
	}
}

# COUNTRY. Legacy access held by a former custodian or an ally.
time_traveler_has_legacy_spirit = {
	OR = {
		has_idea = time_traveler_scattered_forecasts
		has_idea = time_traveler_scattered_forecasts_deep
		has_idea = time_traveler_future_fragment
	}
}

# Host selection

# COUNTRY. A valid host: an existing major country or a player country with
# civilian institutions, below the individual-crisis cap, not in a terminal
# state.
time_traveler_is_valid_host = {
	exists = yes
	hidden_trigger = {
		is_special_chaos_country = no
		is_actual_nonhuman_country = no
		uses_normal_civilian_systems = yes
		individual_crisis_load_is_below_cap = yes
		NOT = { has_capitulated = yes }
		NOT = { has_country_flag = time_traveler_custodian }
		OR = {
			is_major = yes
			is_ai = no
		}
	}
}

# GLOBAL. The traveler has not appeared and some country can host him. Used for
# the Not Available reason in the event list.
time_traveler_has_host = {
	time_traveler_has_not_appeared = yes
	any_country = { time_traveler_is_valid_host = yes }
}

# GLOBAL. The traveler appears once per campaign.
time_traveler_has_not_appeared = {
	hidden_trigger = { check_variable = { global.time_traveler_status = constant:time_traveler_status.dormant } }
}

# COUNTRY. The custodian can start a consultation: direct access, no running
# mission, the traveler not wounded or relocating.
time_traveler_can_consult = {
	time_traveler_has_direct_access = yes
	NOT = { has_country_flag = time_traveler_is_leader }
	NOT = { time_traveler_mission_is_active = yes }
	NOT = { has_country_flag = time_traveler_wounded }
	NOT = { has_country_flag = time_traveler_relocating }
}

# COUNTRY. A stored objective is live (country variables of the custodian).
time_traveler_has_objective = {
	hidden_trigger = { check_variable = { time_traveler_c_kind > constant:time_traveler_kind.none } }
}

# COUNTRY. Garrison objective: the named state is controlled and holds more
# than time_traveler_c_above divisions.
time_traveler_garrison_objective_met = {
	hidden_trigger = {
		var:time_traveler_c_state = { is_controlled_by = PREV }
		meta_trigger = {
			text = {
				divisions_in_state = { state = var:time_traveler_c_state size > [ABOVE] }
			}
			ABOVE = "[?time_traveler_c_above|.0]"
		}
	}
}

# COUNTRY. Research objective: the named technology is researched.
time_traveler_research_objective_met = {
	hidden_trigger = {
		OR = {
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.fuel_silos } has_tech = fuel_silos }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.improved_machine_tools } has_tech = improved_machine_tools }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.advanced_machine_tools } has_tech = advanced_machine_tools }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.computing_machine } has_tech = computing_machine }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.radio } has_tech = radio }
		}
	}
}

time_traveler_research_objective_partial = {
	hidden_trigger = {
		OR = {
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.fuel_silos } is_researching_technology = fuel_silos }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.improved_machine_tools } is_researching_technology = improved_machine_tools }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.advanced_machine_tools } is_researching_technology = advanced_machine_tools }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.computing_machine } is_researching_technology = computing_machine }
			AND = { check_variable = { time_traveler_c_tech = constant:time_traveler_tech.radio } is_researching_technology = radio }
		}
	}
}

# COUNTRY. A technology of the candidate list can be started now.
time_traveler_can_research_candidate = {
	OR = {
		AND = { NOT = { has_tech = fuel_silos } can_research = fuel_silos }
		AND = { NOT = { has_tech = improved_machine_tools } can_research = improved_machine_tools }
		AND = { NOT = { has_tech = advanced_machine_tools } can_research = advanced_machine_tools }
		AND = { NOT = { has_tech = computing_machine } can_research = computing_machine }
		AND = { NOT = { has_tech = radio } can_research = radio }
	}
}
