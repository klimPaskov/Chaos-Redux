# ============================================================================
# CHAOS REDUX - EVENT 026 BLACK FRIDAY TRIGGERS
# ============================================================================
# Global lifecycle predicates for Black Friday. The event is actorless: these
# triggers intentionally read global state rather than the selecting country.
# The Friday flag is derived by the shared daily pulse from global.num_days.
# ============================================================================

black_friday_event_system_is_running = {
	has_global_flag = event_system_initialized
	NOT = { has_global_flag = world_end }
	OR = {
		NOT = { has_variable = global.enabled_countries_count }
		check_variable = { global.enabled_countries_count > 0 }
	}
}

black_friday_is_disabled = {
	is_in_array = { global.disabled_events = constant:black_friday_event.id }
}

black_friday_is_fired = {
	is_in_array = { global.fired_events = constant:black_friday_event.id }
}

black_friday_is_reserved = {
	has_global_flag = black_friday_reserved
}

	has_global_flag = black_friday_active
}

# This trigger-side arithmetic is the owner bridge for custom decision costs.
# It uses the same exact producer and live aligned source registry as effect
# quotation. Owners supply current ordinary cost, a family mask, and any quantum
# override. The final quantizer runs once after eligible sources multiply.
black_friday_quote_custom_cost = {
	# The OUTER caller must predefine every black_friday_quote_* output it reads
	# after this API returns. These assignments own storage for nested calls only.
	set_temp_variable = { black_friday_quote_final_cost = black_friday_quote_ordinary_cost }
	set_temp_variable = { black_friday_quote_valid = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_quote_composed_ratio = constant:universal_cost_framework.zero }
	# Existing owners use family-default quanta. Generic owners may override via
	# black_friday_quote_requested_quantum; absence/non-positive means default.
	set_temp_variable = { black_friday_quote_rounding_quantum = constant:universal_cost_framework.default_rounding_quantum }
	if = {
		limit = { check_variable = { black_friday_quote_family_mask = constant:universal_cost_family.stability } }
		set_temp_variable = { black_friday_quote_rounding_quantum = constant:universal_cost_fixed_point_quantum.stability }
	}
	else_if = {
		limit = { check_variable = { black_friday_quote_family_mask = constant:universal_cost_family.war_support } }
		set_temp_variable = { black_friday_quote_rounding_quantum = constant:universal_cost_fixed_point_quantum.war_support }
	}
	if = {
		limit = { NOT = { has_variable = black_friday_quote_requested_quantum } }
		set_temp_variable = { black_friday_quote_requested_quantum = constant:universal_cost_framework.zero }
	}
	if = {
		limit = { check_variable = { black_friday_quote_requested_quantum > constant:universal_cost_framework.zero } }
		set_temp_variable = { black_friday_quote_rounding_quantum = black_friday_quote_requested_quantum }
	}
	set_temp_variable = { black_friday_quote_requested_quantum = constant:universal_cost_framework.zero }
	if = {
		limit = {
			black_friday_is_active = yes
			check_variable = { global.universal_cost_custom_quote_cache_valid = constant:universal_cost_framework.one }
			check_variable = { black_friday_quote_family_mask > constant:universal_cost_framework.zero }
		}
		set_temp_variable = { universal_cost_quote_ordinary_cost = black_friday_quote_ordinary_cost }
		set_temp_variable = { universal_cost_quote_family_mask = black_friday_quote_family_mask }
		set_temp_variable = { universal_cost_quote_rounding_quantum = black_friday_quote_rounding_quantum }
		# Initialize outputs in this enclosing trigger before the nested producer.
		set_temp_variable = { universal_cost_quote_final_cost = black_friday_quote_ordinary_cost }
		set_temp_variable = { universal_cost_quote_displayed_cost = black_friday_quote_ordinary_cost }
		set_temp_variable = { universal_cost_quote_saved_cost = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_quote_composed_ratio = constant:universal_cost_framework.basis }
		set_temp_variable = { universal_cost_quote_source_count = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_quote_primary_source_id = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_quote_valid = constant:universal_cost_framework.zero }
		hidden_trigger = { universal_cost_quote_exact = yes }
		set_temp_variable = { black_friday_quote_final_cost = universal_cost_quote_final_cost }
		set_temp_variable = { black_friday_quote_valid = universal_cost_quote_valid }
		set_temp_variable = { black_friday_quote_composed_ratio = universal_cost_quote_composed_ratio }
	}
	check_variable = { black_friday_quote_valid = constant:universal_cost_framework.one }
}

# ---------------------------------------------------------------------------
# Simple custom-cost sale API
# ---------------------------------------------------------------------------
# Drop-in sale pricing for an immediate custom cost paid in one native
# resource, with no refund or delayed settlement. Display, affordability, and
# payment read the same quote. Outside a sale the payable amount is the
# ordinary amount. Contract and conversion example:
# common/scripted_effects/026_black_friday_simple_cost_api.md.

# ANY scope. Input: temp black_friday_simple_cost_kind
# (constant:universal_cost_resource_kind.*). Output: temp
# black_friday_simple_cost_family, the achievement family and sale mask for
# that resource, or zero for a kind without a family. Always true.
black_friday_simple_cost_read_family = {
	set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_framework.zero }
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.political_power } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.political_power }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.command_power } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.command_power }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.manpower } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.manpower }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.fuel } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.fuel }
	}
	if = {
		limit = {
			OR = {
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.infantry_equipment }
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.support_equipment }
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.motorized_equipment }
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.support_equipment_1 }
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.motorized_equipment_0 }
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.motorized_equipment_1 }
				AND = {
					check_variable = { black_friday_simple_cost_kind > constant:universal_cost_resource_kind.war_support }
					check_variable = { black_friday_simple_cost_kind < constant:universal_cost_resource_kind.motorized_equipment_0 }
				}
			}
		}
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.equipment }
	}
	if = {
		limit = {
			OR = {
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.train_equipment }
				check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.train_equipment_1 }
			}
		}
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.trains }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.convoy } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.convoy }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.army_experience } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.army_experience }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.navy_experience } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.navy_experience }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.air_experience } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.air_experience }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.stability } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.stability }
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_kind = constant:universal_cost_resource_kind.war_support } }
		set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_family.war_support }
	}
	always = yes
}

# COUNTRY scope (payer). Inputs: temps black_friday_simple_cost_ordinary (the
# owner's current ordinary amount) and black_friday_simple_cost_kind. Outputs:
# temps black_friday_simple_cost_payable, black_friday_simple_cost_family,
# black_friday_simple_cost_quoted (1 when an active sale priced the cost),
# black_friday_simple_cost_on_sale (1 when the payable amount is below the
# ordinary amount), black_friday_simple_cost_valid (0 when an active sale could
# not price the cost, which blocks the purchase rather than charging a wrong
# price), and black_friday_simple_cost_affordable. Always true.
black_friday_simple_cost_quote = {
	set_temp_variable = { black_friday_simple_cost_payable = black_friday_simple_cost_ordinary }
	set_temp_variable = { black_friday_simple_cost_quoted = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_simple_cost_on_sale = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_simple_cost_valid = constant:universal_cost_framework.one }
	set_temp_variable = { black_friday_simple_cost_affordable = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_simple_cost_family = constant:universal_cost_framework.zero }
	black_friday_simple_cost_read_family = yes
	if = {
		limit = {
			black_friday_is_active = yes
			check_variable = { black_friday_simple_cost_ordinary > constant:universal_cost_framework.zero }
			check_variable = { black_friday_simple_cost_family > constant:universal_cost_framework.zero }
		}
		set_temp_variable = { black_friday_simple_cost_valid = constant:universal_cost_framework.zero }
		set_temp_variable = { universal_cost_source_query_id = constant:universal_cost_framework.black_friday_source_id }
		if = {
			limit = { universal_cost_source_is_active = yes }
			set_temp_variable = { black_friday_quote_ordinary_cost = black_friday_simple_cost_ordinary }
			set_temp_variable = { black_friday_quote_family_mask = black_friday_simple_cost_family }
			if = {
				limit = {
					# Caller-owned cost API outputs.
					set_temp_variable = { black_friday_quote_final_cost = black_friday_quote_ordinary_cost }
					black_friday_quote_custom_cost = yes
				}
				set_temp_variable = { black_friday_simple_cost_payable = black_friday_quote_final_cost }
				set_temp_variable = { black_friday_simple_cost_quoted = constant:universal_cost_framework.one }
				set_temp_variable = { black_friday_simple_cost_valid = constant:universal_cost_framework.one }
			}
		}
	}
	if = {
		limit = { check_variable = { black_friday_simple_cost_payable < black_friday_simple_cost_ordinary } }
		set_temp_variable = { black_friday_simple_cost_on_sale = constant:universal_cost_framework.one }
	}
	set_temp_variable = { universal_cost_affordability_resource_kind = black_friday_simple_cost_kind }
	set_temp_variable = { universal_cost_affordability_amount = black_friday_simple_cost_payable }
	if = {
		limit = {
			check_variable = { black_friday_simple_cost_valid = constant:universal_cost_framework.one }
			universal_cost_resource_kind_is_native = yes
			universal_cost_component_is_affordable = yes
		}
		set_temp_variable = { black_friday_simple_cost_affordable = constant:universal_cost_framework.one }
	}
	always = yes
}

	black_friday_simple_cost_quote = yes
	check_variable = { black_friday_simple_cost_affordable = constant:universal_cost_framework.one }
}
