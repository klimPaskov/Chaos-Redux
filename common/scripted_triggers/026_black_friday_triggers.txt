# ============================================================================
# CHAOS REDUX - EVENT 026 BLACK FRIDAY TRIGGERS
# ============================================================================
# Global lifecycle predicates for Black Friday. The event is actorless: these
# triggers intentionally read global state rather than the selecting country.
# The Friday flag is derived by the shared daily pulse from global.num_days.
# ============================================================================

black_friday_event_system_is_running = {
	has_global_flag = event_system_initialized
	NOT = { has_global_flag = world_end }
	OR = {
		NOT = { has_variable = global.enabled_countries_count }
		check_variable = { global.enabled_countries_count > 0 }
	}
}

black_friday_is_disabled = {
	is_in_array = { global.disabled_events = constant:black_friday_event.id }
}

black_friday_is_fired = {
	is_in_array = { global.fired_events = constant:black_friday_event.id }
}

black_friday_is_reserved = {
	has_global_flag = black_friday_reserved
}

black_friday_is_active = {
	has_global_flag = black_friday_active
}

# Country-scope owner adapters report a committed transaction through this
# temporary result. Fixed-price owner calls remain successful without it.
black_friday_owner_payment_succeeded = {
	OR = {
		NOT = { black_friday_is_active = yes }
		check_variable = {
			var = black_friday_owner_payment_success
			value = constant:universal_cost_framework.one
			compare = equals
		}
	}
}

# A payment adapter reports a failed active-sale charge through this temporary
# result. Ordinary owner calls remain successful without an adapter result.
black_friday_fury_payment_succeeded = {
	OR = {
		NOT = { black_friday_is_active = yes }
		check_variable = {
			var = black_friday_fury_payment_success
			value = constant:universal_cost_framework.one
			compare = equals
		}
	}
}

black_friday_is_friday = {
	has_global_flag = black_friday_friday
}

black_friday_can_reserve = {
	black_friday_event_system_is_running = yes
	NOT = { black_friday_is_disabled = yes }
	NOT = { black_friday_is_fired = yes }
	NOT = { black_friday_is_reserved = yes }
	NOT = { black_friday_is_active = yes }
	check_variable = {
		var = global.chaos_meter_value
		value = constant:black_friday_event.required_chaos
		compare = greater_than_or_equals
	}
}

black_friday_can_activate_naturally = {
	black_friday_event_system_is_running = yes
	black_friday_is_reserved = yes
	NOT = { black_friday_is_disabled = yes }
	NOT = { black_friday_is_active = yes }
	NOT = { black_friday_is_fired = yes }
	black_friday_is_friday = yes
	check_variable = {
		var = global.chaos_meter_value
		value = constant:black_friday_event.required_chaos
		compare = greater_than_or_equals
	}
}

black_friday_is_reserved_or_active = {
	OR = {
		black_friday_is_reserved = yes
		black_friday_is_active = yes
	}
}

# Achievement availability is campaign-scoped. It must not depend on the
# current sale, because the engine evaluates an achievement's possible block
# when the campaign starts and does not reopen a false result later.
black_friday_achievement_possible_campaign_is_valid = {
	exists = yes
	is_ai = no
	is_special_chaos_country = no
	NOT = { is_debug = yes }
	NOT = { has_country_flag = force_trigger_mode_enabled }
	NOT = { has_country_flag = black_friday_achievement_debug_or_scenario_disqualified }
	NOT = { has_global_flag = black_friday_achievement_debug_or_scenario_disqualified }
}

black_friday_achievement_is_valid_sale = {
	black_friday_achievement_possible_campaign_is_valid = yes
	has_global_flag = black_friday_natural_activation
	black_friday_is_active = yes
	NOT = { has_global_flag = black_friday_force_disqualified }
	NOT = { has_global_flag = black_friday_manual_disqualified }
	NOT = { has_country_flag = black_friday_achievement_awarded }

	black_friday_achievement_possible_campaign_is_valid = yes
	has_country_flag = black_friday_achievement_awarded
}

# This trigger-side arithmetic is the owner bridge for custom decision costs.
# HOI4 custom-cost triggers cannot call effects, so it reads the finite cache
# rebuilt by the shared source-composition effect. The owner still supplies one
# canonical family mask; the same composed ratio is used by the effect-side
# quote and payment/refund path.
black_friday_quote_custom_cost = {
	set_temp_variable = { black_friday_quote_final_cost = black_friday_quote_ordinary_cost }
	set_temp_variable = { black_friday_quote_valid = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_quote_composed_ratio = constant:universal_cost_framework.zero }
	set_temp_variable = { black_friday_quote_rounding_quantum = constant:universal_cost_framework.default_rounding_quantum }
	if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.stability compare = equals } }
		set_temp_variable = { black_friday_quote_rounding_quantum = constant:universal_cost_fixed_point_quantum.stability }
	}
	else_if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.war_support compare = equals } }
		set_temp_variable = { black_friday_quote_rounding_quantum = constant:universal_cost_fixed_point_quantum.war_support }
	}
	if = {
		limit = {
			black_friday_is_active = yes
			check_variable = {
				var = global.universal_cost_custom_quote_cache_valid
				value = constant:universal_cost_framework.one
				compare = equals
			}
			check_variable = {
				var = black_friday_quote_ordinary_cost
				value = constant:universal_cost_framework.zero
				compare = greater_than_or_equals
			}
			check_variable = {
				var = black_friday_quote_family_mask
				value = constant:universal_cost_framework.zero
				compare = greater_than
			}
		}
		if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.political_power compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_political_power }
		}
		else_if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.command_power compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_command_power }
		}
		else_if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.manpower compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_manpower }
		}
		else_if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.fuel compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_fuel }
		}
		else_if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.equipment compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_equipment }
		}
	else_if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.trains compare = equals } }
		set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_trains }
	}
	else_if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.convoy compare = equals } }
		set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_convoy }
	}
	else_if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.army_experience compare = equals } }
		set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_army_experience }
	}
	else_if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.navy_experience compare = equals } }
		set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_navy_experience }
	}
	else_if = {
		limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.air_experience compare = equals } }
		set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_air_experience }
	}
		else_if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.stability compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_stability }
		}
		else_if = {
			limit = { check_variable = { var = black_friday_quote_family_mask value = constant:universal_cost_family.war_support compare = equals } }
			set_temp_variable = { black_friday_quote_composed_ratio = global.universal_cost_custom_quote_ratio_war_support }
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_quote_composed_ratio
					value = constant:universal_cost_framework.zero
					compare = greater_than
				}
			}
			set_temp_variable = { black_friday_quote_valid = constant:universal_cost_framework.one }
		}
		if = {
			limit = {
				check_variable = {
					var = black_friday_quote_ordinary_cost
					value = constant:universal_cost_framework.zero
					compare = greater_than
				}
			}
			set_temp_variable = { black_friday_quote_numerator = black_friday_quote_ordinary_cost }
			divide_temp_variable = { black_friday_quote_numerator = black_friday_quote_rounding_quantum }
			multiply_temp_variable = { black_friday_quote_numerator = black_friday_quote_composed_ratio }
			set_temp_variable = { black_friday_quote_remainder = black_friday_quote_numerator }
			modulo_temp_variable = {
				var = black_friday_quote_remainder
				value = constant:black_friday_event.ratio_basis
			}
			subtract_from_temp_variable = {
				black_friday_quote_numerator = black_friday_quote_remainder
			}
			divide_temp_variable = {
				black_friday_quote_numerator = constant:black_friday_event.ratio_basis
			}
			if = {
				limit = { check_variable = { black_friday_quote_remainder > constant:universal_cost_framework.zero } }
				add_to_temp_variable = { black_friday_quote_numerator = constant:universal_cost_framework.one }
			}
			set_temp_variable = { black_friday_quote_final_cost = black_friday_quote_numerator }
			multiply_temp_variable = { black_friday_quote_final_cost = black_friday_quote_rounding_quantum }
		}
	}
	check_variable = { black_friday_quote_valid = constant:universal_cost_framework.one }
}
