# ============================================================================
# CHAOS REDUX - FAMINE AND MIGRATION SCRIPTED TRIGGERS
# ============================================================================
# Fail-closed predicates shared by food-security pressure, displacement,
# route, blockade, reception, and return adapters. These triggers validate
# explicit caller proof; they never scan the world or mutate runtime state.
# ============================================================================

famine_migration_country_is_valid = {
	hidden_trigger = {
		scope_exists = yes
		is_special_chaos_country = no
		uses_normal_civilian_systems = yes
	}
}

famine_migration_state_is_valid = {
	hidden_trigger = {
		scope_exists = yes
		check_variable = { var = state_population_k > constant:famine_migration_population.zero  }
		OWNER = {
			famine_migration_country_is_valid = yes
		}
		CONTROLLER = {
			famine_migration_country_is_valid = yes
		}
	}
}

}

famine_migration_destination_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		OR = {
			AND = {
				check_variable = { var = famine_migration_destination_food_safe_proven > constant:famine_migration_runtime.zero  }
				check_variable = { var = famine_migration_destination_reception_proven > constant:famine_migration_runtime.zero  }
			}
			famine_migration_forced_return_destination_is_valid = yes
		}
	}
}

# A forced return may target the persisted original state even when that state
# is not food- or reception-safe. The state-local proof is set only by the
# validated forced-return effect and is cleared after the one transfer attempt
# or route cleanup, so ordinary route requests cannot inherit unsafe access.
famine_migration_forced_return_destination_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		check_variable = { var = famine_migration_forced_return_destination_proven value = constant:famine_migration_runtime.one compare = equals }
	}
}

famine_migration_reception_delta_request_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		check_variable = { var = famine_migration_reception_delta_request_proven > constant:famine_migration_runtime.zero  }
		check_variable = { var = famine_migration_reception_delta_amount > constant:famine_migration_population.zero  }
		OR = {
			check_variable = { var = famine_migration_reception_delta_mode value = constant:famine_migration_reception_delta.credit compare = equals }
			check_variable = { var = famine_migration_reception_delta_mode value = constant:famine_migration_reception_delta.debit compare = equals }
		}
		OWNER = { famine_migration_country_is_valid = yes }
		OR = {
			check_variable = { var = famine_migration_reception_delta_mode value = constant:famine_migration_reception_delta.credit compare = equals }
			AND = {
				check_variable = { var = famine_migration_state_reception_load value = famine_migration_reception_delta_amount compare = greater_than_or_equals }
				OWNER = { check_variable = { var = famine_migration_reception_load value = famine_migration_reception_delta_amount compare = greater_than_or_equals } }
			}
		}
	}
}

# Every condition is binding. A caller must provide the explicit physical and
# political proof in addition to the engine's owner/controller war state.

	hidden_trigger = {

			check_variable = { var = famine_migration_blockade_route_disrupted_proven > constant:famine_migration_runtime.zero  }
			check_variable = { var = famine_migration_blockade_port_disrupted_proven > constant:famine_migration_runtime.zero  }
		}
		OR = {
			check_variable = { var = famine_migration_blockade_convoy_shortage_proven > constant:famine_migration_runtime.zero  }

	}

# A state is eligible for the scheduled evaluator only while an owner has
# deliberately submitted a context or a pressure request. This keeps the
# registry sparse and avoids a world scan hidden inside a trigger.
famine_migration_state_has_active_context = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		OR = {
			has_state_flag = famine_migration_food_security_active
			has_state_flag = famine_migration_surface_context_active
			has_state_flag = famine_migration_profile_context_active
			check_variable = { var = famine_migration_food_pressure > constant:famine_migration_population.zero  }
		}
	}
}

# Safe third-country resettlement rebinds an already received, nonterminal row
# from the actual new destination state. It requires the resolver's persisted
# host and owner targets plus explicit food, reception, route, and actor proof.
# The effect changes only aligned host/destination identity; it never creates,
# debits, credits, or duplicates population.
famine_migration_cohort_resettlement_rebind_request_is_valid = {
	hidden_trigger = {
		famine_migration_state_is_valid = yes
		check_variable = { var = famine_migration_cohort_id_request value = constant:famine_migration_runtime.zero compare = greater_than }
		check_variable = { var = famine_migration_cohort_resettlement_rebind_request_proven value = constant:famine_migration_runtime.zero compare = greater_than }
		check_variable = { var = famine_migration_cohort_resettlement_food_safe_proven value = constant:famine_migration_runtime.zero compare = greater_than }
		check_variable = { var = famine_migration_cohort_resettlement_reception_proven value = constant:famine_migration_runtime.zero compare = greater_than }
		check_variable = { var = famine_migration_cohort_resettlement_route_proven value = constant:famine_migration_runtime.zero compare = greater_than }
		check_variable = { var = famine_migration_cohort_resettlement_actor_proven value = constant:famine_migration_runtime.zero compare = greater_than }
		has_event_target = famine_migration_cohort_host
		event_target:famine_migration_cohort_host = {
			famine_migration_state_is_valid = yes
			OWNER = { famine_migration_country_is_valid = yes }
		}
		has_event_target = famine_migration_cohort_owner
		event_target:famine_migration_cohort_owner = { famine_migration_country_is_valid = yes }
		NOT = { state = event_target:famine_migration_cohort_host }
	}
}
