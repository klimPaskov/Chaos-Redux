# Event 012 Gods equipment capacity.
# Measures actual chassis stock and reinforcement need without changing countries.
# Temporary outputs are shared with the caller only when initialized there first.
# Mobile helpers measure one literal truck or towed-artillery archetype at a time.

gods_of_africa_armor_archetype_is_valid = {
	OR = {
		check_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.light }
		check_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.medium }
		check_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.heavy }
		check_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.modern }
		check_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.super_heavy }
		check_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.amphibious }
	}
}

# Input: caller-set gods_of_africa_armor_archetype. No scripted parameters.
gods_of_africa_measure_armor_archetype = {
	gods_of_africa_armor_archetype_is_valid = yes
	meta_trigger = {
		text = {
			set_temp_variable = { gods_of_africa_armor_type_stock = num_equipment@[EQUIPMENT] }
			set_temp_variable = { gods_of_africa_armor_type_target = num_target_equipment_in_armies_k@[EQUIPMENT] }
			set_temp_variable = { gods_of_africa_armor_type_fielded = num_equipment_in_armies_k@[EQUIPMENT] }
		}
		EQUIPMENT = "[This.GetGodsOfAfricaArmorEquipmentToken]"
	}
	multiply_temp_variable = { gods_of_africa_armor_type_target = constant:gods_of_africa_armor.units_per_thousand }
	multiply_temp_variable = { gods_of_africa_armor_type_fielded = constant:gods_of_africa_armor.units_per_thousand }
	set_temp_variable = { gods_of_africa_armor_type_deficit = gods_of_africa_armor_type_target }
	subtract_from_temp_variable = { gods_of_africa_armor_type_deficit = gods_of_africa_armor_type_fielded }
	clamp_temp_variable = { var = gods_of_africa_armor_type_deficit min = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_type_reserve = gods_of_africa_armor_type_target }
	multiply_temp_variable = { gods_of_africa_armor_type_reserve = constant:gods_of_africa_armor.operating_reserve_fraction }
	add_to_temp_variable = { gods_of_africa_armor_type_reserve = gods_of_africa_armor_type_deficit }
	clamp_temp_variable = { var = gods_of_africa_armor_type_reserve min = constant:gods_of_africa_armor.minimum_chassis_reserve }
	set_temp_variable = { gods_of_africa_armor_type_spare = gods_of_africa_armor_type_stock }
	subtract_from_temp_variable = { gods_of_africa_armor_type_spare = gods_of_africa_armor_type_reserve }
	clamp_temp_variable = { var = gods_of_africa_armor_type_spare min = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_fractional_spare = gods_of_africa_armor_type_spare }
	modulo_temp_variable = { gods_of_africa_armor_fractional_spare = constant:gods_of_africa_value.one }
	subtract_from_temp_variable = { gods_of_africa_armor_type_spare = gods_of_africa_armor_fractional_spare }
	add_to_temp_variable = { gods_of_africa_armor_stock = gods_of_africa_armor_type_stock }
	add_to_temp_variable = { gods_of_africa_armor_deficit = gods_of_africa_armor_type_deficit }
	add_to_temp_variable = { gods_of_africa_armor_spare = gods_of_africa_armor_type_spare }
	check_variable = { var = gods_of_africa_armor_type_spare value = constant:gods_of_africa_value.zero compare = greater_than_or_equals }
}

# COUNTRY scope. gods_of_africa_armor_required_amount is the caller-set input.
# The required shipment has per-chassis reserves
# already removed from the spare count. This recalculates at every use.
gods_of_africa_armor_capacity_is_sufficient = {
	set_temp_variable = { gods_of_africa_armor_stock = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_deficit = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_spare = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_type_stock = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_type_target = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_type_fielded = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.light }
	gods_of_africa_measure_armor_archetype = yes
	set_temp_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.medium }
	gods_of_africa_measure_armor_archetype = yes
	set_temp_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.heavy }
	gods_of_africa_measure_armor_archetype = yes
	set_temp_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.modern }
	gods_of_africa_measure_armor_archetype = yes
	set_temp_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.super_heavy }
	gods_of_africa_measure_armor_archetype = yes
	set_temp_variable = { gods_of_africa_armor_archetype = constant:gods_of_africa_armor_chassis.amphibious }
	gods_of_africa_measure_armor_archetype = yes
	custom_trigger_tooltip = {
		tooltip = gods_of_africa_armor_capacity_tt
		check_variable = { var = gods_of_africa_armor_spare value = gods_of_africa_armor_required_amount compare = greater_than_or_equals }
	}
}

# COUNTRY scope. Input: caller-set temporary gods_of_africa_mobile_type.
# Missing/unknown subtype has no default. All helper calls use = yes.
gods_of_africa_mobile_type_is_valid = {
	custom_trigger_tooltip = {
		tooltip = gods_of_africa_mobile_type_invalid_tt
		OR = {
			check_variable = { gods_of_africa_mobile_type = constant:gods_of_africa_mobile_type.trucks }
			check_variable = { gods_of_africa_mobile_type = constant:gods_of_africa_mobile_type.artillery }
		}
	}
}

# COUNTRY scope. Inputs: caller-set temporary type and required_amount.
# Zero is a measurement request. Payments require a positive amount upstream.
# Callers retaining measurement outputs must initialize them before this call.
gods_of_africa_mobile_capacity_is_sufficient = {
	set_temp_variable = { gods_of_africa_mobile_stock = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_mobile_target = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_mobile_fielded = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_mobile_reserve_floor = constant:gods_of_africa_value.zero }
	gods_of_africa_mobile_type_is_valid = yes
	check_variable = { var = gods_of_africa_mobile_required_amount value = constant:gods_of_africa_value.zero compare = greater_than_or_equals }
	set_temp_variable = { gods_of_africa_mobile_required_remainder = gods_of_africa_mobile_required_amount }
	modulo_temp_variable = { gods_of_africa_mobile_required_remainder = constant:gods_of_africa_value.one }
	check_variable = { gods_of_africa_mobile_required_remainder = constant:gods_of_africa_value.zero }
	if = {
		limit = { check_variable = { gods_of_africa_mobile_type = constant:gods_of_africa_mobile_type.trucks } }
		set_temp_variable = { gods_of_africa_mobile_stock = num_equipment@motorized_equipment }
		set_temp_variable = { gods_of_africa_mobile_target = num_target_equipment_in_armies_k@motorized_equipment }
		set_temp_variable = { gods_of_africa_mobile_fielded = num_equipment_in_armies_k@motorized_equipment }
		set_temp_variable = { gods_of_africa_mobile_reserve_floor = constant:gods_of_africa_mobile.truck_reserve_floor }
	}
	else_if = {
		limit = { check_variable = { gods_of_africa_mobile_type = constant:gods_of_africa_mobile_type.artillery } }
		set_temp_variable = { gods_of_africa_mobile_stock = num_equipment@artillery_equipment }
		set_temp_variable = { gods_of_africa_mobile_target = num_target_equipment_in_armies_k@artillery_equipment }
		set_temp_variable = { gods_of_africa_mobile_fielded = num_equipment_in_armies_k@artillery_equipment }
		set_temp_variable = { gods_of_africa_mobile_reserve_floor = constant:gods_of_africa_mobile.artillery_reserve_floor }
	}
	multiply_temp_variable = { gods_of_africa_mobile_target = constant:gods_of_africa_mobile.units_per_thousand }
	multiply_temp_variable = { gods_of_africa_mobile_fielded = constant:gods_of_africa_mobile.units_per_thousand }
	set_temp_variable = { gods_of_africa_mobile_deficit = gods_of_africa_mobile_target }
	subtract_from_temp_variable = { gods_of_africa_mobile_deficit = gods_of_africa_mobile_fielded }
	clamp_temp_variable = { var = gods_of_africa_mobile_deficit min = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_mobile_reserve = gods_of_africa_mobile_target }
	multiply_temp_variable = { gods_of_africa_mobile_reserve = constant:gods_of_africa_mobile.operating_reserve_fraction }
	add_to_temp_variable = { gods_of_africa_mobile_reserve = gods_of_africa_mobile_deficit }
	clamp_temp_variable = { var = gods_of_africa_mobile_reserve min = gods_of_africa_mobile_reserve_floor }
	set_temp_variable = { gods_of_africa_mobile_spare = gods_of_africa_mobile_stock }
	subtract_from_temp_variable = { gods_of_africa_mobile_spare = gods_of_africa_mobile_reserve }
	clamp_temp_variable = { var = gods_of_africa_mobile_spare min = constant:gods_of_africa_value.zero }
	set_temp_variable = { gods_of_africa_mobile_fractional_spare = gods_of_africa_mobile_spare }
	modulo_temp_variable = { gods_of_africa_mobile_fractional_spare = constant:gods_of_africa_value.one }
	subtract_from_temp_variable = { gods_of_africa_mobile_spare = gods_of_africa_mobile_fractional_spare }
	if = {
		limit = { check_variable = { gods_of_africa_mobile_type = constant:gods_of_africa_mobile_type.trucks } }
		custom_trigger_tooltip = {
			tooltip = gods_of_africa_mobile_truck_capacity_tt
			check_variable = { var = gods_of_africa_mobile_spare value = gods_of_africa_mobile_required_amount compare = greater_than_or_equals }
		}
	}
	else_if = {
		limit = { check_variable = { gods_of_africa_mobile_type = constant:gods_of_africa_mobile_type.artillery } }
		custom_trigger_tooltip = {
			tooltip = gods_of_africa_mobile_artillery_capacity_tt
			check_variable = { var = gods_of_africa_mobile_spare value = gods_of_africa_mobile_required_amount compare = greater_than_or_equals }
		}
	}
}
