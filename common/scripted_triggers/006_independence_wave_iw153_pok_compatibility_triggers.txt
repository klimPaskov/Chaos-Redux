# ============================================================================
# Dormant, package-local compatibility predicates for IW-153. The package is
# a specific-community variant only and remains unbound. POK is a registered
# vanilla carrier; these predicates never turn the broad Dayak label into a
# country candidate and never replace the Pontianak identity.
#
# Every predicate below is gated by all four runtime facts: the living country
# keeps original_tag = POK, the Event 006 origin is active, the exact IW-153
# package marker is present, and a future named-community selector has written
# independence_wave_iw_153_named_community_selected. The selector is reserved
# for a later, source-backed named polity and is not set in this file.
# ============================================================================

# COUNTRY scope. This is the only shared context gate for the dormant adapter.
# is_independence_wave_active_country already requires the Event 006 origin
# flag, the liberation_origin value, and an un-ended origin. The explicit
# origin checks remain here so this package-local boundary is self-documenting
# and cannot accidentally be weakened by a broader registry membership test.
is_independence_wave_iw_153_pok_compatibility_context = {
	exists = yes
	original_tag = POK
	is_independence_wave_active_country = yes
	has_country_flag = independence_wave_active_origin
	check_variable = { liberation_origin = constant:liberation_origin.independence_wave }
	check_variable = { independence_wave_package_id = constant:independence_wave_package_id.iw_153 }
	has_country_flag = independence_wave_iw_153_named_community_selected
}

# COUNTRY scope. Read-only witness for the vanilla character roster. The
# adapter does not recruit, remove, promote, or re-nationalise this character;
# vanilla indonesia_transfer_POK remains the sole owner of transfer behavior.
has_independence_wave_iw_153_pok_character_surface = {
	is_independence_wave_iw_153_pok_compatibility_context = yes
	has_character = INS_syarif_muhammad_alkadrie
}

# COUNTRY scope. Read-only witness for the installed vanilla POK core history.
# State 334 is the accepted research anchor; state 1022 is the other vanilla
# POK core and is included so the adapter cannot silently narrow the source
# country's core set.
has_independence_wave_iw_153_pok_core_surface = {
	is_independence_wave_iw_153_pok_compatibility_context = yes
	334 = { is_core_of = POK }
	1022 = { is_core_of = POK }
}

# COUNTRY scope. Read-only witness for the vanilla Indonesian releasable row.
# This checks source membership only; no adapter effect adds, removes, or
# rewrites INS_releasables.
has_independence_wave_iw_153_pok_releasable_surface = {
	is_independence_wave_iw_153_pok_compatibility_context = yes
	INS = {
		exists = yes
		is_in_array = { array = INS_releasables value = POK }
	}
}

# COUNTRY scope. Strict source-backed preservation contract for a future
# named-community consumer. It is intentionally not a generic Dayak route or
# a current-map selection predicate.
has_independence_wave_iw_153_pok_compatibility_contract = {
	is_independence_wave_iw_153_pok_compatibility_context = yes
	has_independence_wave_iw_153_pok_character_surface = yes
	has_independence_wave_iw_153_pok_core_surface = yes
	has_independence_wave_iw_153_pok_releasable_surface = yes
	owns_state = 334
	controls_state = 334
	capital_scope = { state = 334 }
}
