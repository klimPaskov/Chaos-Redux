# COUNTRY scope on an African government already present in one of Event 12's
# bounded relationship or selection arrays. This never scans the world.
africa_rsa_is_valid_exile_patron = {
	exists = yes
	is_subject = no
	has_focus_tree = generic_focus
	NOT = { has_capitulated = yes }
	NOT = { original_tag = SAF }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	NOT = { has_country_flag = africa_unifier_host }
	any_core_state = { is_on_continent = africa }
	capital_scope = {
		is_on_continent = africa
		is_controlled_by = PREV
	}
}

# COUNTRY scope on the current host. A later loyalist victory can transfer the
# suppressed project only to an actor chosen before the war from bounded stores.
africa_rsa_has_exile_patron_candidate = {
	OR = {
		any_of_scopes = {
			array = africa_relationship_countries
			africa_rsa_is_valid_exile_patron = yes
		}
		any_of_scopes = {
			array = africa_selected_targets
			africa_rsa_is_valid_exile_patron = yes
		}
		any_of_scopes = {
			array = africa_frozen_first_contact_targets
			africa_rsa_is_valid_exile_patron = yes
		}
	}
}

# COUNTRY scope during the one-shot host candidate build, after the
# prospective host's temporary contact pool exists. This prevents original
# SAF from entering the weighted pool when its frozen contact set contains no
# valid African exile patron in the temporary contact pool. Persistent Event 12
# relationship arrays do not exist before a host has been initialized and are
# intentionally not consulted during this prefire pass.
africa_rsa_prefire_contact_pool_has_patron = {
	any_of_scopes = {
		array = africa_prefire_contact_pool
		africa_rsa_is_valid_exile_patron = yes
	}
}

# COUNTRY scope on original SAF before the temporary contact pool exists. This
# mirrors the South African opening overlay and proves that at least one of its
# minimum valid contacts can also serve as the later exile patron.
africa_rsa_has_prefire_patron_contact = {
	any_country = {
		africa_is_valid_relationship_target = yes
		africa_rsa_is_valid_exile_patron = yes
		NOT = { tag = PREV }
		OR = {
			original_tag = SAF original_tag = ZIM original_tag = ZAM original_tag = MLW
			original_tag = MZB original_tag = BOT original_tag = HZX original_tag = EUX
			original_tag = ANG
		}
	}
}

# COUNTRY scope during the shared Event 12 fire-once selection pass. South
# Africa keeps its meaningful vanilla focus tree, so it enters through this
# narrow branch instead of the generic-focus host gate. Before the prefire
# roster is frozen, the patron check is deferred; after freezing, the same
# bounded roster must contain a valid African custodian before the entry event
# can fire. The public civil-war effect remains the final hard gate.

	exists = yes
	tag = SAF
	original_tag = SAF
	has_capitulated = no
	has_civil_war = no
	NOT = { has_global_flag = africa_event_active }
	NOT = { has_global_flag = africa_is_one }
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = africa_rsa_branch_used }
	NOT = { has_global_flag = africa_rsa_recovery_used }
	NOT = { has_global_flag = africa_rsa_civil_war_active }
	NOT = { has_country_flag = africa_unifier_host }
	NOT = { has_country_flag = africa_unified_identity }
	NOT = { has_country_flag = africa_terminal_world_actor }
	NOT = { has_country_flag = event_012_incompatible_terminal_actor }
	NOT = { is_actual_nonhuman_country = yes }
	NOT = { is_special_chaos_country = yes }
	capital_scope = {
		is_on_continent = africa
		is_controlled_by = PREV
	}
	any_core_state = { is_on_continent = africa }
	africa_has_prefire_opening_contact = yes
	africa_rsa_has_prefire_patron_contact = yes
	africa_rsa_has_supported_subject_autonomy = yes
	is_in_faction = yes
	OR = {
		ENG = {
			exists = yes
			is_in_faction_with = PREV

		has_country_flag = africa_rsa_allied_framework_confirmed
	}
	275 = {
		is_owned_by = PREV
		is_controlled_by = PREV
	}
	681 = {
		is_owned_by = PREV
		is_controlled_by = PREV
		naval_base > constant:africa_rsa_count.none
	}
	719 = {
		is_owned_by = PREV
		is_controlled_by = PREV
		naval_base > constant:africa_rsa_count.none
	}
	OR = {
		NOT = { has_country_flag = africa_prefire_contacts_frozen }
		africa_rsa_has_exile_patron_candidate = yes
	}
}

# COUNTRY scope on the original SAF host immediately after the civil war has
# been created. The Republican Nationalist side is deliberately fail-closed:
# it needs the documented high anti-war threshold, an absent ESX carrier, and
# one state selected by the pre-war SAF ownership/control scan. The saved state
# must still belong to either principal RSA side and cannot overlap the
# reserved EQX anchor or either principal capital/port corridor.
africa_rsa_republican_breakaway_can_start = {
	check_variable = {
		var = africa_rsa_prewar_neutrality_support
		value = constant:africa_rsa_pressure.high_anti_war
		compare = greater_than_or_equals
	}
	ESX = { exists = no }
	has_event_target = africa_rsa_republican_state
	event_target:africa_rsa_republican_state = {
		exists = yes
		is_on_continent = africa
		is_capital = no
		naval_base < constant:africa_rsa_count.one
		NOT = { is_core_of = EQX }
		NOT = { is_owned_by = EQX }
		OR = {
			is_owned_by = event_target:africa_rsa_continental_coalition
			is_owned_by = event_target:africa_rsa_allied_union_government
		}
		OR = {
			is_controlled_by = event_target:africa_rsa_continental_coalition
			is_controlled_by = event_target:africa_rsa_allied_union_government
		}
	}
}

africa_rsa_republican_breakaway_is_active = {

# COUNTRY scope on the Continental Coalition. The proof is physical: protected
# civilians, the Transvaal industrial base, a captured Cape or Natal port, a
# working allied-controlled railway, and two distinct African supporters.
africa_rsa_civil_war_corridor_is_secure = {
	africa_rsa_is_continental_coalition = yes
	has_global_flag = africa_rsa_civil_war_active
	has_country_flag = africa_rsa_civilians_protected
	check_variable = {
		var = africa_rsa_external_support_count
		value = constant:africa_rsa_count.required_external_supporters
		compare = greater_than_or_equals
	}
	275 = {
		is_controlled_by = PREV
		OR = {
			industrial_complex > constant:africa_rsa_count.none
			arms_factory > constant:africa_rsa_count.none
		}
	}
	OR = {
		AND = {
			681 = {
				is_controlled_by = PREV
				naval_base > constant:africa_rsa_count.none
			}
			has_railway_connection = {
				build_only_on_allied = yes
				start_state = 275
				target_state = 681
			}
		}
		AND = {
			719 = {
				is_controlled_by = PREV
				naval_base > constant:africa_rsa_count.none
			}
			has_railway_connection = {
				build_only_on_allied = yes
				start_state = 275
				target_state = 719
			}
		}
	}
}
