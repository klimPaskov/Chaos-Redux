# ============================================================================
# CHAOS REDUX - EVENT 020 BLACK PLAGUE WEAPONIZATION TRIGGERS
# ============================================================================
# Country and state gates for the abstract Black Plague weaponization program.
# The project reuses the native plague_bomb technology and stockpile while
# adding Event 020 attribution, accidents, countermeasure pressure, and the
# shared Biological Raids delivery route.
# ============================================================================

# COUNTRY scope. A live Kruger biological program and the reusable Directorate
# provider can expose this project without waiting for Event 020. They still
# use the same project, runtime, payload, delivery, and accident systems.
black_plague_weaponization_directorate_has_access = {
	OR = {
		AND = {
			brilliant_scientist_is_current_host = yes
			has_country_flag = brilliant_scientist_biological_weapons_theory_completed
		}
		AND = {
			is_mengele_clone_directorate_country = yes
			OR = {
				has_country_flag = directorate_special_projects_all_available
				has_country_flag = directorate_special_project_black_plague_available
			}
		}
	}
}

# COUNTRY scope.
black_plague_weaponization_actor_is_valid = {
	exists = yes
	NOT = { has_global_flag = world_end }
	OR = {
		AND = {
			uses_normal_civilian_systems = yes
			NOT = { is_actual_nonhuman_country = yes }
			NOT = { is_special_chaos_country = yes }
			has_global_flag = black_plague_system_active
			OR = {
				black_plague_country_has_active_severe_state = yes
				any_controlled_state = { black_plague_state_is_infected_or_worse = yes }
			}
		}
		black_plague_weaponization_directorate_has_access = yes
	}
}

# COUNTRY scope. Clinical access is earned through the shared countermeasure
# board or the native field-hospital / pathogen systems.
black_plague_weaponization_has_sample_access = {
	OR = {
		has_country_flag = black_plague_countermeasure_findings_available
		has_tech = pathogen_handling_protocols
		has_tech = bio_surveillance_networks
	}
}

black_plague_weaponization_project_is_available = {
	black_plague_weaponization_actor_is_valid = yes
	NOT = { is_special_project_completed = sp:black_plague_weaponization_program }
	NOT = { has_country_flag = black_plague_weaponization_condemned }
	NOT = { has_country_flag = black_plague_weaponization_defensive_conversion }
	OR = {
		AND = {
			black_plague_weaponization_has_sample_access = yes
			OR = {
				is_special_project_completed = sp:plague_bomb
				has_tech = plague_bomb_delivery_systems
			}
		}
		black_plague_weaponization_directorate_has_access = yes
	}
}

black_plague_weaponization_project_is_active = {
	black_plague_weaponization_actor_is_valid = yes
	is_special_project_being_researched = sp:black_plague_weaponization_program
	NOT = { has_country_flag = black_plague_weaponization_complete }
	NOT = { has_country_flag = black_plague_weaponization_defensive_conversion }
}

black_plague_weaponization_approach_can_continue = {
	black_plague_weaponization_project_is_active = yes
	NOT = { has_country_flag = black_plague_weaponization_approach_chosen }
}

# COUNTRY scope while the selected project posture occupies one civilian
# factory. The selected-approach and defensive-conversion flags are receipts of
# work already begun, so this continuation gate must not reuse the start gate.
black_plague_weaponization_approach_commitment_is_valid = {
	exists = yes
	NOT = { has_global_flag = world_end }
	has_global_flag = black_plague_system_active
	is_special_project_being_researched = sp:black_plague_weaponization_program
	has_country_flag = black_plague_weaponization_approach_chosen
}

black_plague_weaponization_can_choose_approach = {
	black_plague_weaponization_approach_can_continue = yes
	NOT = { has_equipment = { support_equipment < constant:black_plague_weaponization_program.approach_support_equipment } }
	NOT = { command_power < constant:black_plague_weaponization_program.approach_command_power }
}

# Pre-selection quote gate. The factory may already be reserved when
# complete_effect rechecks the material payment, so that effect calls the
# material helper above without rechecking free factories.
black_plague_weaponization_approach_is_affordable = {
	black_plague_weaponization_can_choose_approach = yes
	NOT = { num_of_civilian_factories_available_for_projects < constant:black_plague_weaponization_program.approach_civilian_factories }
	NOT = {
		OR = {
			has_decision = black_plague_weaponization_safety_first
			has_decision = black_plague_weaponization_military_acceleration
			has_decision = black_plague_weaponization_dual_use
			has_decision = black_plague_weaponization_defensive_conversion
		}
	}
}

black_plague_weaponization_can_begin = {
	black_plague_weaponization_project_is_available = yes
	NOT = { has_equipment = { support_equipment < constant:black_plague_weaponization_program.minimum_support_equipment } }
	NOT = { has_equipment = { motorized_equipment < constant:black_plague_weaponization_program.minimum_motorized_equipment } }
	NOT = { has_fuel < constant:black_plague_weaponization_program.minimum_fuel }
}

# COUNTRY scope. Delivery stays available after the actor's own outbreak has
# ended so a completed native payload can seed a hostile state through the
# ordinary Event 020 exposure lifecycle.
black_plague_weaponization_delivery_actor_is_valid = {
	exists = yes
	NOT = { has_global_flag = world_end }
	has_global_flag = black_plague_system_active
	OR = {
		AND = {
			uses_normal_civilian_systems = yes
			NOT = { is_actual_nonhuman_country = yes }
			NOT = { is_special_chaos_country = yes }
		}
		AND = {
			has_country_flag = directorate_special_project_black_plague_completed
			has_country_flag = black_plague_weaponization_complete
		}
	}
}

# COUNTRY scope while the expansion or destruction factory commitment runs.
# Payment stock and the absence of a cooldown are start-only checks. A present
# action-specific cooldown is a positive receipt that the click effect settled.
black_plague_weaponization_expansion_commitment_is_valid = {
	black_plague_weaponization_delivery_actor_is_valid = yes
	has_country_flag = black_plague_weaponization_complete
	has_country_flag = black_plague_weaponization_stockpile_expansion_cooldown
}

black_plague_weaponization_destruction_commitment_is_valid = {
	black_plague_weaponization_delivery_actor_is_valid = yes
	has_country_flag = black_plague_weaponization_complete
	has_country_flag = black_plague_weaponization_stockpile_destroy_cooldown
}

# COUNTRY scope. Post-project stockpile controls remain inside the shared
# disease category. Expansion creates one additional native payload; it does
# not grant instant delivery and remains exposed to the ordinary accident
# scheduler.
black_plague_weaponization_can_expand_stockpile = {
	black_plague_weaponization_delivery_actor_is_valid = yes
	has_country_flag = black_plague_weaponization_complete
	has_country_flag = black_plague_weaponized_delivery_ready
	NOT = { has_country_flag = black_plague_weaponization_defensive_conversion }
	NOT = { has_country_flag = black_plague_weaponization_stockpile_expansion_cooldown }
	NOT = { has_equipment = { support_equipment < constant:black_plague_weaponization_program.stockpile_expansion_support_equipment } }
	NOT = { has_fuel < constant:black_plague_weaponization_program.stockpile_expansion_fuel }
}

black_plague_weaponization_expansion_is_affordable = {
	black_plague_weaponization_can_expand_stockpile = yes
	NOT = { num_of_civilian_factories_available_for_projects < constant:black_plague_weaponization_program.stockpile_expansion_civilian_factories }
	NOT = { has_decision = black_plague_weaponization_expand_stockpile }
	NOT = { has_decision = black_plague_weaponization_destroy_stockpile }
}

# COUNTRY scope. Destruction is a deliberate safety action and can be used
# even when the payload is not currently deliverable. It removes the native
# payload, closes the accident scheduler, and leaves the project history intact.
black_plague_weaponization_can_destroy_stockpile = {
	black_plague_weaponization_delivery_actor_is_valid = yes
	has_country_flag = black_plague_weaponization_complete
	NOT = { has_country_flag = black_plague_weaponization_defensive_conversion }
	has_equipment = { plague_bomb_1 > 0 }
	NOT = { has_country_flag = black_plague_weaponization_stockpile_destroy_cooldown }
	NOT = { has_equipment = { support_equipment < constant:black_plague_weaponization_program.stockpile_destroy_support_equipment } }
	NOT = { command_power < constant:black_plague_weaponization_program.stockpile_destroy_command_power }
}

black_plague_weaponization_destruction_is_affordable = {
	black_plague_weaponization_can_destroy_stockpile = yes
	NOT = { num_of_civilian_factories_available_for_projects < constant:black_plague_weaponization_program.stockpile_destroy_civilian_factories }
	NOT = { has_decision = black_plague_weaponization_destroy_stockpile }
	NOT = { has_decision = black_plague_weaponization_expand_stockpile }
}

# COUNTRY scope. The native raid type stays visible after the project is
# complete even when its real payload or supporting crates are out of stock;
# essential_equipment and command_power own the native creation payment.
black_plague_weaponization_native_raid_is_visible = {
	black_plague_weaponization_delivery_actor_is_valid = yes
	has_country_flag = black_plague_weaponization_complete
	has_country_flag = black_plague_weaponized_delivery_ready
	NOT = { has_country_flag = black_plague_weaponization_defensive_conversion }
	OR = {
		is_special_project_completed = sp:plague_bomb
		has_tech = plague_bomb_delivery_systems
	}
}

# STATE scope. Country, war, policy, and stock are checked separately so the
# native raid's exact candidate state remains the single exposure target.
black_plague_weaponization_native_raid_state_is_valid = {
	black_plague_state_can_receive_exposure = yes
	NOT = { black_plague_state_is_rat_controlled = yes }
	NOT = { has_state_flag = black_plague_weaponized_provenance }
}

# COUNTRY scope with FROM as the exact proposed victim and var:target_state as
# the native raid's selected state. This gate has no country-wide iterator.
black_plague_weaponization_native_raid_target_is_visible = {
	FROM = {
		exists = yes
		NOT = { is_actual_nonhuman_country = yes }
	}
	has_war_with = FROM
	NOT = { is_in_faction_with = FROM }
	NOT = { is_subject_of = FROM }
	FROM = { NOT = { is_subject_of = ROOT } }
	var:target_state = {
		is_controlled_by = FROM
		black_plague_weaponization_native_raid_state_is_valid = yes
	}
}

# COUNTRY scope. Native essential_equipment and command_power are deliberately
# absent from these script gates; both are reserved once by raid creation.
black_plague_weaponization_native_raid_can_prepare = {
	black_plague_weaponization_native_raid_is_visible = yes
	black_plague_weaponization_native_raid_target_is_visible = yes
	cbrn_policy_allows_strategic_use_against_from = yes
}

black_plague_weaponization_native_raid_can_launch = {
	black_plague_weaponization_native_raid_can_prepare = yes
}

black_plague_weaponization_native_raid_ai_may_launch = {
	black_plague_weaponization_native_raid_can_prepare = yes
}

# COUNTRY scope in the fixed hidden actor outcome event. The native callback
# saved chain-local pointers before entering this country scope; the raid has
# already reserved its equipment and Command Power, so none is checked here.
black_plague_weaponization_native_raid_bridge_context_is_valid = {
	has_event_target = black_plague_weaponization_raid_actor
	has_event_target = black_plague_weaponization_raid_victim
	has_event_target = black_plague_weaponization_raid_target_state
	event_target:black_plague_weaponization_raid_actor = { tag = ROOT }
	event_target:black_plague_weaponization_raid_victim = { exists = yes }
	event_target:black_plague_weaponization_raid_target_state = {
		black_plague_weaponization_native_raid_state_is_valid = yes
		CONTROLLER = {
			tag = event_target:black_plague_weaponization_raid_victim
			has_war_with = ROOT
		}
	}
	black_plague_weaponization_native_raid_is_visible = yes
	event_target:black_plague_weaponization_raid_victim = { set_temp_variable = { cbrn_retaliation_target_id = id } }
	cbrn_policy_allows_strategic_use_against_target_id = yes
}
