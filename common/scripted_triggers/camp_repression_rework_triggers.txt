is_valid_camp_dormant_marker_state = {
	has_state_flag = genocide_historical_quiet_camp
	NOT = { has_state_flag = camp_rework_site_active }
}

camp_rework_state_can_accept_responsibility_from_root = {
	OR = {
		NOT = { has_variable = genocide_responsible_country }
		var:genocide_responsible_country = { tag = ROOT }
	}
}

is_valid_camp_active_site_state = {
	has_state_flag = camp_rework_site_active
	has_variable = genocide_responsible_country
	NOT = { has_state_flag = camp_rework_site_dismantled }
	OR = {
		concentration_camp > 0
		extermination_camp > 0
		gulag_labor_camp_network > 0
		has_state_flag = genocide_auschwitz_experiment_site
		has_state_flag = genocide_ss_laboratory_site
		has_state_flag = genocide_japanese_biowarfare_atrocity_site
		has_state_flag = genocide_soviet_mass_repression_site
		has_state_flag = genocide_soviet_famine_pressure_site
		has_state_flag = camp_rework_expanded_labor_site
		has_state_flag = camp_rework_experiment_site
		has_state_flag = camp_rework_radicalized_site
		has_state_flag = genocide_restricted_chemical_site_active
	}
}

is_valid_camp_discovery_state = {
	has_variable = genocide_responsible_country
	NOT = { has_state_flag = genocide_site_discovered }
	NOT = { has_state_flag = genocide_historical_quiet_camp }
	OR = {
		is_valid_camp_active_site_state = yes
		has_state_flag = genocide_destroyed_atrocity_site
		has_state_flag = genocide_evidence_destroyed
		has_state_flag = genocide_evidence_destroy_failed
		has_state_flag = camp_rework_liberated_evidence
	}
}

camp_rework_state_requires_unregister = {
	OR = {
		NOT = { has_variable = genocide_responsible_country }
		AND = {
			has_variable = genocide_responsible_country
			NOT = { var:genocide_responsible_country = { exists = yes } }
		}
		has_state_flag = camp_rework_site_dismantled
		AND = {
			has_variable = genocide_responsible_country
			NOT = {
				OR = {
					var:genocide_responsible_country = {
						PREV = { is_controlled_by = PREV }
					}
					CONTROLLER = {
						is_subject_of = var:PREV.genocide_responsible_country
					}
				}
			}
		}
		AND = {
			has_state_flag = camp_rework_site_active
			NOT = { is_valid_camp_active_site_state = yes }
		}
	}
}

	OR = {
		check_variable = { camp_active_site_count > 0 }
		check_variable = { camp_network_reach > 0 }
		genocide_country_has_registered_active_camps = yes
	}
}
