is_independence_wave_candidate_anchor_available = {
	owner = {
		exists = yes
		controls_state = PREV
		NOT = { is_soviet_collapse_active_origin_country = yes }
	}
	NOT = { controller = { is_soviet_collapse_active_origin_country = yes } }
	NOT = { is_liberation_release_current_reserved_state = yes }
	NOT = { is_liberation_release_current_protected_state = yes }
}

# COUNTRY scope. Shared dormant-carrier, reservation, and origin-system proof.
# This is intentionally separate from package readiness: reused vanilla tags
# cannot be granted build flags through duplicate dormant-country history files.
# A few custom Event 006 carrier shells are instantiated by startup history so
# their character definitions can be parsed. They are still dormant when they
# own and control no states. Treat that empty shell exactly like an absent tag;
# living countries remain excluded by the owned/control-state checks below.
# The reservation marker is intentionally not part of this predicate: the
# selected dormant target receives that marker before execution preflight, and
# the allocator checks duplicate reservations separately.
is_independence_wave_dormant_country_scope = {
	OR = {
		exists = no
		AND = {
			exists = yes
			check_variable = { var = num_owned_states value = 0 compare = equals }
			NOT = { num_of_controlled_states > 0 }
			NOT = { has_country_flag = independence_wave_active_origin }
			NOT = { has_country_flag = independence_wave_origin_prepared }
			NOT = { has_country_flag = independence_wave_origin_committed }
		}
	}
}

is_independence_wave_candidate_origin_available = {
	is_independence_wave_dormant_country_scope = yes
	# A frozen target is checked again after reservation during execution
	# preflight. Allow that same-plan reservation to pass the immutable package
	# identity trigger, while rejecting a stale reservation marker from another
	# plan. The reservation effect still rejects duplicate array entries.
	OR = {
		NOT = { has_country_flag = liberation_release_reserved_country }
		is_liberation_release_current_reserved_country = yes
	}
	NOT = {
		has_variable = independence_wave_last_rejected_plan_id
		check_variable = {
			var = independence_wave_last_rejected_plan_id
			value = global.liberation_plan_id
			compare = equals
		}
	}
	NOT = { has_country_flag = soviet_collapse_active_origin }
	NOT = { check_variable = { liberation_origin = constant:liberation_origin.soviet_collapse } }
	NOT = { has_country_flag = independence_wave_active_origin }
	# Event 012 carriers retain their creating event's package, tree, decisions,
	# AI, and formable routes even if the country later ceases to exist.
	NOT = { has_country_flag = africa_priority_member_package_active }
	NOT = { has_country_flag = africa_priority_member_focus_tree_loaded }
}
