# ============================================================================
# CHAOS REDUX - EVENT 016 MENGELE PROJECT STAGE TRIGGERS
# ============================================================================
# Private country-scope gates for the fifteen-family Mengele Event 016 stage
# receipt API. The selectors are temporary variables named
# mengele_event016_project_family and mengele_event016_requested_stage.
#
# Family IDs are one-based and the aligned receipt arrays use family ID minus
# one. These gates never read or write the Kruger project board, Capacity, or
# project-history variables.
# ============================================================================

# COUNTRY scope. The bridge file owns the shorter provider alias. This strict
# lifecycle gate is the single authorization source for paid stages and native
# provider callbacks.
brilliant_scientist_mengele_project_stage_provider_is_valid = {
	exists = yes
	is_mengele_clone_directorate_country = yes
	has_country_flag = mengele_clone_focus_directorate_project_registry
	NOT = { has_country_flag = brilliant_scientist_current_host }
	NOT = { has_country_flag = brilliant_scientist_project_incident_active }
	NOT = { has_country_flag = germany_mengele_defeated }
	NOT = { has_country_flag = germany_mengele_program_rejected }
	NOT = { has_country_flag = germany_mengele_program_closed }
	NOT = { has_country_flag = germany_mengele_program_expired }
	NOT = { has_country_flag = germany_mengele_program_recently_expired }
	NOT = { has_country_flag = brilliant_scientist_all_actions_locked_after_world_end }
	NOT = { has_global_flag = brilliant_scientist_singularity_terminal_fired }
	NOT = { has_global_flag = brilliant_scientist_lab_world_terminal_fired }
	NOT = { has_global_flag = world_end }
	OR = {
		AND = {
			germany_mengele_program_active = yes
			has_idea = germany_auschwitz_experiments
		}
		AND = {
			germany_mengele_restricted_program_active = yes
			has_idea = germany_auschwitz_experiments_restricted
		}
		AND = {
			has_country_flag = germany_mengele_faction
			has_idea = germany_mengele_laboratory_state
			NOT = { has_country_flag = germany_mengele_perfect_aryans_overthrew_mengele }
		}
		AND = {
			has_country_flag = germany_mengele_victorious
			has_idea = germany_mengele_victory_state
			NOT = { has_country_flag = germany_mengele_perfect_aryans_overthrew_mengele }
		}
		AND = {
			has_country_flag = mengele_clone_army_scenario_country
			has_idea = mengele_clone_army_scenario_state
			NOT = { check_variable = { triggerable_clone_aryan_variant > constant:brilliant_scientist_value.zero } }
		}
	}
}
