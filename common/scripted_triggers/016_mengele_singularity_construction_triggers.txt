# Event 016 Directorate Singularity construction gates and exact callback identity.
# COUNTRY scope unless entered through the selected STATE decision target.

brilliant_scientist_mengele_singularity_construction_command_node_is_unlocked = {
	brilliant_scientist_mengele_project_stage_provider_is_valid = yes
	OR = {
		has_country_flag = mengele_event016_singularity_command_core_completed
		AND = {
			NOT = { has_dlc = "Gotterdammerung" }
			has_country_flag = mengele_event016_singularity_prototype_completed
		}
	}
	brilliant_scientist_mengele_singularity_site_needs_command_node = yes
}

brilliant_scientist_mengele_singularity_construction_power_link_is_unlocked = {
	brilliant_scientist_mengele_project_stage_provider_is_valid = yes
	OR = {
		has_country_flag = mengele_event016_singularity_power_link_completed
		AND = {
			NOT = { has_dlc = "Gotterdammerung" }
			has_country_flag = mengele_event016_singularity_prototype_completed
		}
	}
	brilliant_scientist_mengele_singularity_site_needs_power_link = yes
}

# Country-scope cost rows deliberately exclude route, site, and active-timer
# requirements. The native begin callback rechecks direct costs without
# counting civilian factories already reserved by the timed decision.
brilliant_scientist_mengele_singularity_construction_can_afford_command_node = {
	NOT = { num_of_available_civilian_factories < constant:brilliant_scientist_project_board.singularity_node_civilian_factories }
	brilliant_scientist_mengele_singularity_construction_can_pay_direct_command_node = yes
}

brilliant_scientist_mengele_singularity_construction_can_pay_direct_command_node = {
	NOT = { has_equipment = { support_equipment < constant:brilliant_scientist_project_board.singularity_node_support_equipment } }
	NOT = { has_equipment = { motorized_equipment < constant:brilliant_scientist_project_board.singularity_node_trucks } }
	NOT = { has_fuel < constant:brilliant_scientist_project_board.singularity_node_fuel }
	check_variable = { var = manpower value = constant:brilliant_scientist_project_board.singularity_node_manpower compare = greater_than_or_equals }
	check_variable = { var = political_power value = constant:brilliant_scientist_project_board.singularity_node_political_power compare = greater_than_or_equals }
}

brilliant_scientist_mengele_singularity_construction_can_afford_power_link = {
	NOT = { num_of_available_civilian_factories < constant:brilliant_scientist_project_board.singularity_link_civilian_factories }
	brilliant_scientist_mengele_singularity_construction_can_pay_direct_power_link = yes
}

brilliant_scientist_mengele_singularity_construction_can_pay_direct_power_link = {
	NOT = { has_equipment = { support_equipment < constant:brilliant_scientist_project_board.singularity_link_support_equipment } }
	NOT = { has_equipment = { motorized_equipment < constant:brilliant_scientist_project_board.singularity_link_trucks } }
	NOT = { has_fuel < constant:brilliant_scientist_project_board.singularity_link_fuel }
	check_variable = { var = manpower value = constant:brilliant_scientist_project_board.singularity_link_manpower compare = greater_than_or_equals }
	check_variable = { var = political_power value = constant:brilliant_scientist_project_board.singularity_link_political_power compare = greater_than_or_equals }
}

# Once the native timed decision has begun, its civilian_factory_use may have
# reserved the same factories already. Recheck direct debits and physical
# requirements without counting that reservation a second time.
brilliant_scientist_mengele_singularity_construction_can_begin_command_node_after_native_reservation = {
	brilliant_scientist_mengele_singularity_construction_command_node_is_unlocked = yes
	NOT = { has_country_flag = mengele_event016_singularity_construction_receipt_active }
	NOT = { brilliant_scientist_mengele_singularity_site_pending_receipt_is_present = yes }
	brilliant_scientist_has_required_singularity_facilities = yes
	brilliant_scientist_mengele_singularity_construction_can_pay_direct_command_node = yes
}

brilliant_scientist_mengele_singularity_construction_can_begin_power_link_after_native_reservation = {
	brilliant_scientist_mengele_singularity_construction_power_link_is_unlocked = yes
	NOT = { has_country_flag = mengele_event016_singularity_construction_receipt_active }
	NOT = { brilliant_scientist_mengele_singularity_site_pending_receipt_is_present = yes }
	brilliant_scientist_has_required_singularity_facilities = yes
	brilliant_scientist_mengele_singularity_construction_can_pay_direct_power_link = yes
}

# The exact native callback must match the stored payment receipt. A missing
# physical pending row still allows this matching callback to refund its cost.
brilliant_scientist_mengele_singularity_construction_callback_matches_receipt = {
	has_country_flag = mengele_event016_singularity_construction_receipt_active
	brilliant_scientist_mengele_singularity_site_selected_state_scope_is_valid = yes
	has_variable = mengele_event016_singularity_construction_receipt_state
	has_variable = mengele_event016_singularity_construction_receipt_role
	has_variable = mengele_event016_singularity_construction_receipt_quantity
	has_variable = mengele_event016_singularity_construction_receipt_political_power
	has_variable = mengele_event016_singularity_construction_receipt_support_equipment
	has_variable = mengele_event016_singularity_construction_receipt_trucks
	has_variable = mengele_event016_singularity_construction_receipt_fuel
	has_variable = mengele_event016_singularity_construction_receipt_manpower
	check_variable = { var = mengele_event016_singularity_site_selected_role value = mengele_event016_singularity_construction_expected_role compare = equals }
	check_variable = { var = mengele_event016_singularity_construction_receipt_role value = mengele_event016_singularity_construction_expected_role compare = equals }
	check_variable = { var = mengele_event016_singularity_construction_receipt_state value = event_target:mengele_event016_singularity_site_selected_state compare = equals }
	check_variable = { var = mengele_event016_singularity_construction_receipt_quantity value = mengele_event016_singularity_site_selected_quantity compare = equals }
	check_variable = { var = mengele_event016_singularity_construction_receipt_quantity value = constant:brilliant_scientist_value.one compare = equals }
	OR = {
		NOT = { brilliant_scientist_mengele_singularity_site_pending_receipt_is_present = yes }
		brilliant_scientist_mengele_singularity_site_pending_callback_matches_receipt = yes
	}
}
