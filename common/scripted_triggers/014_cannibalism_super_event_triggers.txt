# ============================================================================
# CHAOS REDUX - EVENT 014 CANNIBALISM SUPER-EVENT TRIGGERS
# ============================================================================
# Guards the four public Event 014 super-events. Reveal-facing text is never
# reachable before cannibalism_reveal_complete. The defeat aftermath requires
# a year-long, multi-country, region-scale campaign and cumulative population
# loss; local containment can never satisfy it.
# ============================================================================

cannibalism_global_defeat_aftermath_is_eligible = {
	has_global_flag = cannibalism_global_victory
	has_global_flag = cannibalism_reveal_complete
	NOT = { has_global_flag = world_end }
	NOT = { has_global_flag = cannibalism_global_defeat_aftermath_emitted }
	check_variable = {
		var = global.cannibalism_consumed_population_k
		value = constant:cannibalism_evolution_threshold.major_defeat_consumed_population_k
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.cannibalism_max_controlled_states
		value = constant:cannibalism_evolution_threshold.major_defeat_controlled_states
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.cannibalism_global_defeat_duration_days
		value = constant:cannibalism_evolution_threshold.major_defeat_duration_days
		compare = greater_than_or_equals
	}
	check_variable = {
		var = global.cannibalism_global_defeat_contributor_count
		value = constant:cannibalism_evolution_threshold.major_defeat_contributor_countries
		compare = greater_than_or_equals
	}
}
