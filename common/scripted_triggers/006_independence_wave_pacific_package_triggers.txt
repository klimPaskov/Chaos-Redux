# HAW and FSM begin from island anchors without civilian industry. Their
# strategic delegation cost therefore consumes government cohesion, command
# attention, manpower, and convoy capacity rather than an impossible factory
# reservation.
can_pay_independence_wave_pacific_island_strategic_cost = {
	check_variable = { var = stability value = constant:independence_wave_decision_cost.stability_standard compare = greater_than }
	check_variable = { var = war_support value = constant:independence_wave_decision_cost.war_support_minor compare = greater_than }
	command_power > constant:independence_wave_decision_cost.command_power_standard
	has_manpower > constant:independence_wave_decision_cost.manpower_light
	has_equipment = { convoy > constant:independence_wave_decision_cost.convoy_light }
}
