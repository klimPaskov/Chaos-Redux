		exists = yes
		is_owned_by = ROOT
		is_controlled_by = ROOT
		if = {
			limit = { ROOT = { has_country_flag = brilliant_scientist_krg_temporal_rescue_target_was_capital } }
			is_capital = yes
		}
		else_if = {
			limit = { ROOT = { has_country_flag = brilliant_scientist_krg_temporal_rescue_target_was_singularity_site } }
			ROOT = { has_country_flag = brilliant_scientist_singularity_live_command_network }

				has_state_flag = brilliant_scientist_secondary_facility
			}
			NOT = { has_state_flag = brilliant_scientist_facility_destroyed }
		}
		else = { always = no }
	}
}
