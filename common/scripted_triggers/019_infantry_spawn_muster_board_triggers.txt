# These gates are the single availability contract for the movable Muster
# Board, ordinary-country decisions, and AI dispatch. Gameplay effects repeat
# their critical ledger/resource checks before payment so stale UI state cannot
# create a free action or mutate a resolved lot.

	infantry_spawn_country_is_active = yes
	infantry_spawn_has_evolution_iii = yes
	NOT = { is_infantry_spawn_derivative_country = yes }
	NOT = { has_country_flag = infantry_spawn_derivative_creation_in_progress }
	NOT = { has_global_flag = world_end }
}

	check_variable = { var = infantry_spawn_muster_gui_selected_family_view_index value = infantry_spawn_muster_gui_family_id_entries^num compare = less_than }
