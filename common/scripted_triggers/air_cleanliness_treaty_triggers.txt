# Supported scope: COUNTRY, the exact secretariat that owns the transaction.
air_cleanliness_treaty_verification_transaction_is_valid = {
	air_cleanliness_treaty_country_is_current_founder = yes
	NOT = { has_global_flag = fallout_transition_active }
	NOT = { has_global_flag = fallout_active }
	has_country_flag = air_cleanliness_treaty_verification_active
	has_variable = air_cleanliness_treaty_verification_transaction
	has_variable = air_cleanliness_treaty_verification_generation
	has_variable = air_cleanliness_treaty_verification_subject
	is_in_array = {
		array = global.air_cleanliness_treaty_active_inspectors
		value = THIS
	}
	check_variable = {
		var = air_cleanliness_treaty_verification_generation
		value = global.air_cleanliness_treaty_generation
		compare = equals
	}
	var:air_cleanliness_treaty_verification_subject = {
		exists = yes
		NOT = { tag = PREV }
		air_cleanliness_treaty_country_is_live_member = yes
		has_country_flag = air_cleanliness_treaty_inspection_reserved
		has_variable = air_cleanliness_treaty_inspection_inspector
		has_variable = air_cleanliness_treaty_inspection_transaction
		has_variable = air_cleanliness_treaty_inspection_generation
		check_variable = {
			var = air_cleanliness_treaty_inspection_inspector
			value = PREV
			compare = equals
		}
		check_variable = {
			var = air_cleanliness_treaty_inspection_transaction
			value = PREV.air_cleanliness_treaty_verification_transaction
			compare = equals
		}
		check_variable = {
			var = air_cleanliness_treaty_inspection_generation
			value = global.air_cleanliness_treaty_generation
			compare = equals
		}
	}
	NOT = { has_war_with = var:air_cleanliness_treaty_verification_subject }
	OR = {
		AND = {
			has_country_flag = air_cleanliness_treaty_verification_mission_active
			NOT = { has_country_flag = air_cleanliness_treaty_verification_awaiting_response }
			NOT = { has_country_flag = air_cleanliness_treaty_verification_result_pending }
			NOT = { has_variable = air_cleanliness_treaty_verification_outcome }
			var:air_cleanliness_treaty_verification_subject = {
				NOT = { has_country_flag = air_cleanliness_treaty_inspection_request_pending }
				NOT = { has_country_flag = air_cleanliness_treaty_inspection_response_pending }
				NOT = { has_variable = air_cleanliness_treaty_inspection_outcome }
			}
		}
		AND = {
			NOT = { has_country_flag = air_cleanliness_treaty_verification_mission_active }
			has_country_flag = air_cleanliness_treaty_verification_awaiting_response
			NOT = { has_country_flag = air_cleanliness_treaty_verification_result_pending }
			NOT = { has_variable = air_cleanliness_treaty_verification_outcome }
			var:air_cleanliness_treaty_verification_subject = {
				has_country_flag = air_cleanliness_treaty_inspection_request_pending
				NOT = { has_country_flag = air_cleanliness_treaty_inspection_response_pending }
				NOT = { has_variable = air_cleanliness_treaty_inspection_outcome }
			}
		}
		AND = {
			NOT = { has_country_flag = air_cleanliness_treaty_verification_mission_active }
			NOT = { has_country_flag = air_cleanliness_treaty_verification_awaiting_response }
			has_country_flag = air_cleanliness_treaty_verification_result_pending
			has_variable = air_cleanliness_treaty_verification_outcome
			OR = {
				check_variable = { var = air_cleanliness_treaty_verification_outcome value = constant:air_cleanliness_treaty_verification_outcome.full_access compare = equals }
				check_variable = { var = air_cleanliness_treaty_verification_outcome value = constant:air_cleanliness_treaty_verification_outcome.records_only compare = equals }
				check_variable = { var = air_cleanliness_treaty_verification_outcome value = constant:air_cleanliness_treaty_verification_outcome.refused compare = equals }
			}
			var:air_cleanliness_treaty_verification_subject = {
				NOT = { has_country_flag = air_cleanliness_treaty_inspection_request_pending }
				has_country_flag = air_cleanliness_treaty_inspection_response_pending
				has_variable = air_cleanliness_treaty_inspection_outcome
				check_variable = {
					var = air_cleanliness_treaty_inspection_outcome
					value = PREV.air_cleanliness_treaty_verification_outcome
					compare = equals
				}
			}
		}
	}
}
