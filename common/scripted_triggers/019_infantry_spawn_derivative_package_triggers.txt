	has_variable = infantry_spawn_derivative_claimant_uid
	check_variable = { infantry_spawn_derivative_claimant_uid > constant:infantry_spawn_event.invalid_uid }

		meta_trigger = {
			text = { has_id = [CLAIMANT_ID] }
			CLAIMANT_ID = "[?infantry_spawn_derivative_claimant_uid|.0]"
		}
		is_army_leader = yes
		is_female = no
	}
}
