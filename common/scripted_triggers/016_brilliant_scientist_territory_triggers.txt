# ============================================================================
# CHAOS REDUX - EVENT 016 KRUGER STATE TERRITORY TRIGGERS
# ============================================================================
# Overview:
# - Defines the live-state predicates used to build and revalidate a frozen
#   Kruger State territory plan.
# - Requires every selected state to remain owned and controlled by the former
#   host until the formation effect begins its first mutation.
# - Keeps charter, enclave, and rebellion contracts separate and fail-closed.
# - Institutional takeover is deliberately absent: territory failure never
#   grants or evaluates that route.
# ============================================================================

# STATE scope. ROOT must be the Event 016 host that is preparing the plan.
brilliant_scientist_territory_state_has_event_facility = {
	OR = {
		has_variable = brilliant_scientist_facility_type
		has_state_flag = brilliant_scientist_primary_facility
		has_state_flag = brilliant_scientist_secondary_facility
		has_state_flag = brilliant_scientist_joint_laboratory_site
		has_state_flag = brilliant_scientist_quantum_transit_terminal
		has_state_flag = brilliant_scientist_cloning_growth_site
		has_state_flag = brilliant_scientist_robotics_assembly_complex
		has_state_flag = brilliant_scientist_paleogenetic_reserve
		has_state_flag = brilliant_scientist_paleogenetic_hatchery
		has_state_flag = brilliant_scientist_xenobiological_vat_complex
		has_state_flag = brilliant_scientist_xenobiological_control_center
		has_state_flag = brilliant_scientist_alien_interface_chamber
		has_state_flag = brilliant_scientist_temporal_anchor
	}
	NOT = { has_state_flag = brilliant_scientist_facility_destroyed }
}

# STATE scope. This is the non-negotiable ownership and former-host survival
# boundary for every route. The host capital is never a territorial candidate.
brilliant_scientist_territory_state_is_host_candidate = {
	impassable = no
	is_capital = no
	is_owned_by = ROOT
	is_controlled_by = ROOT
	NOT = { has_state_flag = brilliant_scientist_facility_destroyed }
	OR = {
		ROOT = { NOT = { has_country_flag = brilliant_scientist_formation_territory_route_charter } }
		is_core_of = ROOT
	}
}

# STATE scope. A home-area connection is the preferred land supply proof.
# Remote states require an actual hub, railway, or serviced port.
brilliant_scientist_territory_state_has_logistics_access = {
	OR = {
		is_in_home_area = yes
		supply_node > constant:brilliant_scientist_territory_plan.zero
		rail_way > constant:brilliant_scientist_territory_plan.zero
		AND = {
			is_coastal = yes
			naval_base > constant:brilliant_scientist_territory_plan.zero
			infrastructure > constant:brilliant_scientist_territory_plan.zero
		}
	}
}

# STATE scope. Support territory must contribute a concrete industrial or
# transport asset; adjacency alone does not make an unrelated state eligible.
brilliant_scientist_territory_state_is_strategic_support = {
	OR = {
		supply_node > constant:brilliant_scientist_territory_plan.zero
		rail_way > constant:brilliant_scientist_territory_plan.zero
		industrial_complex > constant:brilliant_scientist_territory_plan.zero
		arms_factory > constant:brilliant_scientist_territory_plan.zero
		dockyard > constant:brilliant_scientist_territory_plan.zero
		AND = {
			is_coastal = yes
			naval_base > constant:brilliant_scientist_territory_plan.zero
		}
	}
}

# STATE scope. These are the only proofs accepted for a deliberately separated
# site: Event 016 transport hardware, a serviced air base, or a serviced port.
brilliant_scientist_territory_state_can_support_multi_site_plan = {
	OR = {
		has_state_flag = brilliant_scientist_quantum_transit_terminal
		has_state_flag = brilliant_scientist_temporal_anchor
		AND = {
			air_base > constant:brilliant_scientist_territory_plan.zero
			infrastructure > constant:brilliant_scientist_territory_plan.zero
		}
		AND = {
			is_coastal = yes
			naval_base > constant:brilliant_scientist_territory_plan.zero
			infrastructure > constant:brilliant_scientist_territory_plan.zero
		}
	}
}

			NOT = { has_state_flag = brilliant_scientist_formation_territory_capital }

	has_country_flag = brilliant_scientist_formation_territory_invalid
	NOT = { has_country_flag = brilliant_scientist_formation_territory_verified }
	NOT = { has_country_flag = brilliant_scientist_formation_territory_prepared }
	NOT = { has_country_flag = brilliant_scientist_formation_territory_revalidated }
	NOT = { has_country_flag = brilliant_scientist_formation_territory_planning }
	NOT = { has_event_target = brilliant_scientist_formation_capital_state }
	NOT = { has_event_target = brilliant_scientist_formation_territory_host }
	check_variable = {
		var = brilliant_scientist_formation_selected_state_count
		value = constant:brilliant_scientist_territory_plan.zero
		compare = equals
	}
	check_variable = {
		var = brilliant_scientist_formation_territory_invalid_reason
		value = constant:brilliant_scientist_territory_invalid_reason.none
		compare = greater_than
	}
	NOT = {
		any_state = {
			OR = {
				has_state_flag = brilliant_scientist_formation_territory_selected
				has_state_flag = brilliant_scientist_formation_territory_capital
				has_state_flag = brilliant_scientist_formation_territory_connected_facility
				has_state_flag = brilliant_scientist_formation_territory_multi_site_facility
