# Execution-side fixed-origin proofs. The shared executor supplies the
# reservation and package-id checks; these predicates only bind the immutable
# country identity and origin-system exclusions.
is_independence_wave_iw093_fixed_origin_preflight = {
	exists = no
	original_tag = DOX
	is_liberation_release_current_reserved_country = yes
	NOT = { has_country_flag = soviet_collapse_active_origin }
	NOT = { check_variable = { liberation_origin = constant:liberation_origin.soviet_collapse } }
	NOT = { has_country_flag = independence_wave_active_origin }
}

# `date` predicates do not accept script-constant tokens in the engine. The
# single canonical cutover is therefore kept here, while the corresponding
# day value is recorded in the package constants for cross-file tuning notes.
is_independence_wave_iw098_pre_cutover = {
	date < 1938.06.17
}

is_independence_wave_iw098_post_cutover = {
	NOT = { date < 1938.06.17 }
}

has_independence_wave_iw098_date_appropriate_role_attestation = {
	OR = {
		AND = {
			is_independence_wave_iw098_pre_cutover = yes
			has_country_flag = independence_wave_iw098_pre_cutover_role_attested
		}
		AND = {
			is_independence_wave_iw098_post_cutover = yes
			has_country_flag = independence_wave_iw098_post_cutover_role_attested
		}
	}
}
