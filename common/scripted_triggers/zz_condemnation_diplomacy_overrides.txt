# ============================================================================
# CHAOS REDUX - CONDEMNATION DIPLOMACY OVERRIDES (LATE LOAD)
# ============================================================================
# Preserves the vanilla lend-lease game-rule gates and adds the bilateral arms
# embargo rule. ROOT and FROM are the two diplomacy participants; the check is
# deliberately mirrored so either the outgoing or incoming gate catches a new
# lend-lease agreement from a sanction participant to its condemned target.
# ============================================================================

DIPLOMACY_LEND_LEASE_ENABLE_TRIGGER = {
	custom_trigger_tooltip = {
		tooltip = condemnation_lend_lease_blocked_tt
		FROM = {
			NOT = {
				is_in_array = {
					array = condemnation_arms_embargo_participants
					value = ROOT
				}
			}
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_lend_lease
				option = FREE
			}
		}
		always = yes
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_lend_lease
				option = LIMITED
			}
		}
		always = yes
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_lend_lease
				option = SAME_IDEOLOGY
			}
		}
		custom_trigger_tooltip = {
			tooltip = RULE_ALLOW_LEND_LEASE_SAME_IDEOLOGY_TT
			has_government = FROM
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_lend_lease
				option = SAME_FACTION
			}
		}
		custom_trigger_tooltip = {
			tooltip = RULE_ALLOW_LEND_LEASE_SAME_FACTION_TT
			is_in_faction_with = FROM
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_lend_lease
				option = BLOCKED
			}
		}
		custom_trigger_tooltip = {
			tooltip = RULE_ALLOW_LEND_LEASE_BLOCKED_TT
			always = no
		}
	}
}
