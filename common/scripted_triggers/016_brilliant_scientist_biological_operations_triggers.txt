brilliant_scientist_biological_agent_black_plague_is_unlocked = {
	OR = {
		has_country_flag = black_plague_weaponization_complete
		has_country_flag = brilliant_scientist_biological_agent_black_plague_authorized
		AND = {
			has_character = KRG_warren_kruger
			KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_black_plague }
		}
	}
}

# COUNTRY scope. Public query for scripted callers that have already run
# brilliant_scientist_calculate_selected_biological_payload_requirement. Reads
# the resulting temporary brilliant_scientist_biological_payload_required.
brilliant_scientist_selected_biological_stockpile_is_sufficient = {
	brilliant_scientist_selected_biological_agent_is_unlocked = yes
	check_variable = { brilliant_scientist_biological_payload_required > constant:brilliant_scientist_value.zero }
	OR = {
		AND = {
			check_variable = { brilliant_scientist_selected_biological_agent = constant:brilliant_scientist_biological_agent.anthrax }
			check_variable = { var = num_equipment@anthrax_bomb_1 value = brilliant_scientist_biological_payload_required compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_selected_biological_agent = constant:brilliant_scientist_biological_agent.plague }
			check_variable = { var = num_equipment@plague_bomb_1 value = brilliant_scientist_biological_payload_required compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_selected_biological_agent = constant:brilliant_scientist_biological_agent.tularemia }
			check_variable = { var = num_equipment@tularemia_bomb_1 value = brilliant_scientist_biological_payload_required compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_selected_biological_agent = constant:brilliant_scientist_biological_agent.smallpox }
			check_variable = { var = num_equipment@smallpox_bomb_1 value = brilliant_scientist_biological_payload_required compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_selected_biological_agent = constant:brilliant_scientist_biological_agent.weaponized_zombies }
			check_variable = { var = num_equipment@zombie_disease_bomb_1 value = brilliant_scientist_biological_payload_required compare = greater_than_or_equals }
		}
		AND = {
			check_variable = { brilliant_scientist_selected_biological_agent = constant:brilliant_scientist_biological_agent.black_plague }
			check_variable = { var = num_equipment@plague_bomb_1 value = brilliant_scientist_biological_payload_required compare = greater_than_or_equals }
		}
	}
}

# STATE scope with ROOT = actor country. Conventional access never charges
# transport equipment; the committed method cannot change during preparation.
brilliant_scientist_biological_conventional_battlefield_target_is_valid = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	OWNER = { exists = yes has_war_with = ROOT }
	is_controlled_by = OWNER
	OR = {
		any_neighbor_state = { is_controlled_by = ROOT }
		supply_node > 0
		bunker > 0
		coastal_bunker > 0
		air_base > 0
		arms_factory > 0
	}
}

# STATE scope with ROOT = actor country. Operational portals reach hostile
# rear areas; stockpile affordability belongs to the start-only query below.
brilliant_scientist_biological_portal_target_is_valid = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	OWNER = { exists = yes has_war_with = ROOT }
	is_controlled_by = OWNER
	ROOT = { has_tech = brilliant_scientist_portal_warfare_tech }
}

brilliant_scientist_biological_battlefield_target_is_valid = {
	OR = {
		brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes
		AND = {
			brilliant_scientist_biological_portal_target_is_valid = yes
			ROOT = {
				check_variable = { var = num_equipment@teleportation_equipment_1 value = constant:brilliant_scientist_biological_deployment.portal_transport_equipment compare = greater_than_or_equals }
			}
		}
	}
}

# STATE scope with ROOT = actor country.
brilliant_scientist_biological_covert_target_is_valid = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	OWNER = { exists = yes has_war_with = ROOT }
	is_controlled_by = OWNER
	is_core_of = OWNER
	OR = {
		industrial_complex > 0
		arms_factory > 0
		dockyard > 0
		supply_node > 0
		air_base > 0
		rocket_site > 0
		nuclear_reactor > 0
		commercial_nuclear_reactor > 0
		land_facility > 0
		air_facility > 0
		naval_facility > 0
		nuclear_facility > 0
	}
}

# COUNTRY scope. An in-flight operation cannot silently switch its victim when
# a target is transferred or occupied by another belligerent.
brilliant_scientist_biological_deployment_transaction_is_valid = {
	exists = yes
	has_capitulated = no
	NOT = { has_global_flag = world_end }
	has_country_flag = brilliant_scientist_biological_deployment_pending
	has_country_flag = brilliant_scientist_biological_deployment_payload_debited
	has_variable = brilliant_scientist_biological_deployment_target_state
	has_variable = brilliant_scientist_biological_deployment_victim
	check_variable = { brilliant_scientist_biological_deployment_payload_amount > constant:brilliant_scientist_value.zero }
	var:brilliant_scientist_biological_deployment_victim = { exists = yes has_war_with = ROOT }
	var:brilliant_scientist_biological_deployment_target_state = {
		OWNER = { tag = var:ROOT.brilliant_scientist_biological_deployment_victim }
		is_controlled_by = var:ROOT.brilliant_scientist_biological_deployment_victim
		OR = {
			AND = {
				ROOT = { check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield } }
				OR = {
					AND = {
						ROOT = { check_variable = { brilliant_scientist_biological_deployment_transport_amount > constant:brilliant_scientist_value.zero } }
						brilliant_scientist_biological_portal_target_is_valid = yes
					}
					AND = {
						ROOT = { NOT = { check_variable = { brilliant_scientist_biological_deployment_transport_amount > constant:brilliant_scientist_value.zero } } }
						brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes
					}
				}
			}
			AND = {
				ROOT = { check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.covert } }
				brilliant_scientist_biological_covert_target_is_valid = yes
			}
		}
	}
}
