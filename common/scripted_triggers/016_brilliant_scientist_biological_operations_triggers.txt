# ============================================================================
# CHAOS REDUX - EVENT 016 BIOLOGICAL RAID TRIGGERS
# ============================================================================
# Public agent unlocks and exact native-raid target guards. Each raid chooses
# its payload by ID. There is no selected arsenal or separate staging receipt.
# ============================================================================

brilliant_scientist_biological_agent_anthrax_is_unlocked = {
	OR = { has_tech = anthrax_bomb_delivery_systems is_special_project_completed = sp:anthrax_bomb }
}

brilliant_scientist_biological_agent_plague_is_unlocked = {
	OR = { has_tech = plague_bomb_delivery_systems is_special_project_completed = sp:plague_bomb }
}

brilliant_scientist_biological_agent_tularemia_is_unlocked = {
	OR = { has_tech = tularemia_bomb_delivery_systems is_special_project_completed = sp:tularemia_bomb }
}

brilliant_scientist_biological_agent_smallpox_is_unlocked = {
	OR = { has_tech = smallpox_bomb_delivery_systems is_special_project_completed = sp:smallpox_bomb }
}

brilliant_scientist_biological_agent_weaponized_zombies_is_unlocked = {
	OR = {
		has_tech = zombie_disease_bomb_delivery_systems
		has_country_flag = weaponized_zombies_completed
		is_special_project_completed = sp:weaponize_the_zombies
	}
}

brilliant_scientist_biological_agent_black_plague_is_unlocked = {
	OR = {
		has_country_flag = black_plague_weaponization_complete
		has_country_flag = brilliant_scientist_biological_agent_black_plague_authorized
		AND = {
			has_character = KRG_warren_kruger
			KRG_warren_kruger = { has_character_flag = brilliant_scientist_personal_biological_agent_black_plague }
		}
	}
}

brilliant_scientist_has_any_unlocked_biological_agent = {
	OR = {
		brilliant_scientist_biological_agent_anthrax_is_unlocked = yes
		brilliant_scientist_biological_agent_plague_is_unlocked = yes
		brilliant_scientist_biological_agent_tularemia_is_unlocked = yes
		brilliant_scientist_biological_agent_smallpox_is_unlocked = yes
		brilliant_scientist_biological_agent_weaponized_zombies_is_unlocked = yes
		brilliant_scientist_biological_agent_black_plague_is_unlocked = yes
	}
}

brilliant_scientist_biological_operations_country_is_active = {
	exists = yes
	has_capitulated = no
	NOT = { has_global_flag = world_end }
}

# STATE scope with ROOT = acting country. Conventional access never charges
# transport equipment; the committed method cannot change during preparation.
brilliant_scientist_biological_conventional_battlefield_target_is_valid = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	OWNER = { exists = yes has_war_with = ROOT }
	is_controlled_by = OWNER
	OR = {
		any_neighbor_state = { is_controlled_by = ROOT }
		supply_node > 0
		bunker > 0
		coastal_bunker > 0
		air_base > 0
		arms_factory > 0
	}
}

# STATE scope with ROOT = acting country. Operational portals reach hostile
# rear areas; stockpile affordability belongs to the start-only query below.
brilliant_scientist_biological_portal_target_is_valid = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	OWNER = { exists = yes has_war_with = ROOT }
	is_controlled_by = OWNER
	ROOT = { has_tech = brilliant_scientist_portal_warfare_tech }
}

# STATE scope with ROOT = acting country.
brilliant_scientist_biological_covert_target_is_valid = {
	bio_lifecycle_state_is_eligible_ordinary_pathogen_target = yes
	OWNER = { exists = yes has_war_with = ROOT }
	is_controlled_by = OWNER
	is_core_of = OWNER
	OR = {
		industrial_complex > 0
		arms_factory > 0
		dockyard > 0
		supply_node > 0
		air_base > 0
		rocket_site > 0
		nuclear_reactor > 0
		commercial_nuclear_reactor > 0
		land_facility > 0
		air_facility > 0
		naval_facility > 0
		nuclear_facility > 0
	}
}

# COUNTRY scope. Native raid actor predicates.
brilliant_scientist_biological_raid_actor_is_ready = {
	brilliant_scientist_biological_operations_country_is_active = yes
	bio_kruger_last_resort_authority_is_valid = yes
}

# COUNTRY scope, FROM is victim and var:target_state is the selected raid state.
brilliant_scientist_biological_raid_battlefield_target_is_valid = {
	has_war_with = FROM
	var:target_state = {
		brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes
	}
}

brilliant_scientist_biological_raid_portal_target_is_valid = {
	has_war_with = FROM
	has_tech = brilliant_scientist_portal_warfare_tech
	var:target_state = {
		brilliant_scientist_biological_portal_target_is_valid = yes
		NOT = { brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes }
	}
}

brilliant_scientist_biological_raid_covert_target_is_valid = {
	has_war_with = FROM
	var:target_state = {
		brilliant_scientist_biological_covert_target_is_valid = yes
	}
}

# COUNTRY scope after the immediate hidden raid event resets ROOT to the actor.
# Agent, route, and transport are reconstructed from the static event ID;
# regular event targets carry the exact raid-instance actor, state, and victim.
brilliant_scientist_native_biological_raid_receipt_is_valid = {
	brilliant_scientist_biological_raid_actor_is_ready = yes
	has_event_target = brilliant_scientist_native_raid_actor
	has_event_target = brilliant_scientist_native_raid_target_state
	has_event_target = brilliant_scientist_native_raid_victim
	tag = event_target:brilliant_scientist_native_raid_actor
	OR = {
		AND = {
			check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.anthrax }
			brilliant_scientist_biological_agent_anthrax_is_unlocked = yes
		}
		AND = {
			check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.plague }
			brilliant_scientist_biological_agent_plague_is_unlocked = yes
		}
		AND = {
			check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.tularemia }
			brilliant_scientist_biological_agent_tularemia_is_unlocked = yes
		}
		AND = {
			check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.smallpox }
			brilliant_scientist_biological_agent_smallpox_is_unlocked = yes
		}
		AND = {
			check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.weaponized_zombies }
			brilliant_scientist_biological_agent_weaponized_zombies_is_unlocked = yes
		}
		AND = {
			check_variable = { brilliant_scientist_biological_deployment_agent = constant:brilliant_scientist_biological_agent.black_plague }
			brilliant_scientist_biological_agent_black_plague_is_unlocked = yes
		}
	}
	event_target:brilliant_scientist_native_raid_target_state = {
		exists = yes
		OWNER = { tag = event_target:brilliant_scientist_native_raid_victim has_war_with = ROOT }
		is_controlled_by = OWNER
		OR = {
			AND = {
				check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield }
				check_variable = { brilliant_scientist_biological_deployment_transport_amount = constant:brilliant_scientist_value.zero }
				brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes
			}
			AND = {
				check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.battlefield }
				check_variable = { brilliant_scientist_biological_deployment_transport_amount = constant:brilliant_scientist_biological_deployment.portal_transport_equipment }
				brilliant_scientist_biological_portal_target_is_valid = yes
				NOT = { brilliant_scientist_biological_conventional_battlefield_target_is_valid = yes }
			}
			AND = {
				check_variable = { brilliant_scientist_biological_deployment_route = constant:brilliant_scientist_biological_route.covert }
				check_variable = { brilliant_scientist_biological_deployment_transport_amount = constant:brilliant_scientist_value.zero }
				brilliant_scientist_biological_covert_target_is_valid = yes
			}
		}
	}
}
