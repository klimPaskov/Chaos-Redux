# ============================================================================
# CHAOS REDUX - STATE CRISIS MAP MODES
# ============================================================================
# Scripted map modes for state-level crisis visualization.
# - contaminated_states_map_mode shows chemical, disease, and nuclear hazards.
# - deaths_state_map_mode shows cumulative civilian deaths by state.
# - famine_state_map_mode shows the five dynamic food-security stages and proof-bearing blockade/relief context.
# - migration_state_map_mode shows displacement origins, trapped populations, reception, resettlement, and return.
# Famine and migration remain two separate dedicated map modes; neither is an
# event surface or an event-pacing pulse.
# ============================================================================

scripted_map_modes = {
	deaths_state_map_mode = {
		top = {
			type = state
			color = {
				FROM = {
					mapmode_state_has_civilian_death_history = yes
					has_variable = global.chaos_state_civilian_deaths_map_max
					check_variable = { global.chaos_state_civilian_deaths_map_max > 0 }
				}
				FROM = {
					set_temp_variable = { red = 0 }
					set_temp_variable = { green = 0 }
					set_temp_variable = { blue = 0 }
					set_temp_variable = { alpha = 0 }
					set_temp_variable = { thickness = constant:state_map_modes_deaths.base_thickness }
					set_temp_variable = { mapmode_deaths_ratio = 0 }

					if = {
						limit = {
							mapmode_state_has_civilian_death_history = yes
							has_variable = global.chaos_state_civilian_deaths_map_max
							check_variable = { global.chaos_state_civilian_deaths_map_max > 0 }
						}
						set_temp_variable = { mapmode_deaths_ratio = chaos_state_civilian_deaths_total }
						divide_temp_variable = { mapmode_deaths_ratio = global.chaos_state_civilian_deaths_map_max }
						clamp_temp_variable = { var = mapmode_deaths_ratio min = 0 max = 1 }
						set_temp_variable = { red = constant:state_map_modes_deaths.gradient_min_red }
						set_temp_variable = { green = constant:state_map_modes_deaths.gradient_max_green }
						set_temp_variable = { blue = constant:state_map_modes_deaths.gradient_max_blue }
						set_temp_variable = { alpha = constant:state_map_modes_deaths.gradient_min_alpha }

						set_temp_variable = { mapmode_deaths_red_delta = mapmode_deaths_ratio }
						multiply_temp_variable = { mapmode_deaths_red_delta = constant:state_map_modes_deaths.gradient_red_span }
						add_to_temp_variable = { red = mapmode_deaths_red_delta }

						set_temp_variable = { mapmode_deaths_green_drop = mapmode_deaths_ratio }
						multiply_temp_variable = { mapmode_deaths_green_drop = constant:state_map_modes_deaths.gradient_green_drop }
						subtract_from_temp_variable = { green = mapmode_deaths_green_drop }

						set_temp_variable = { mapmode_deaths_blue_drop = mapmode_deaths_ratio }
						multiply_temp_variable = { mapmode_deaths_blue_drop = constant:state_map_modes_deaths.gradient_blue_drop }
						subtract_from_temp_variable = { blue = mapmode_deaths_blue_drop }

						set_temp_variable = { mapmode_deaths_alpha_delta = mapmode_deaths_ratio }
						multiply_temp_variable = { mapmode_deaths_alpha_delta = constant:state_map_modes_deaths.gradient_alpha_span }
						add_to_temp_variable = { alpha = mapmode_deaths_alpha_delta }
					}

					if = {
						limit = {
							check_variable = {
								var = mapmode_deaths_ratio
								value = constant:state_map_modes_deaths.extreme_threshold
								compare = greater_than_or_equals
							}
						}
						set_temp_variable = { red = constant:state_map_modes_deaths.extreme_red }
						set_temp_variable = { green = constant:state_map_modes_deaths.extreme_green }
						set_temp_variable = { blue = constant:state_map_modes_deaths.extreme_blue }
						set_temp_variable = { alpha = constant:state_map_modes_deaths.extreme_alpha }
						set_temp_variable = { thickness = constant:state_map_modes_deaths.extreme_thickness }
						set_temp_variable = { highlighted = 1 }
					}
					else_if = {
						limit = {
							check_variable = {
								var = mapmode_deaths_ratio
								value = constant:state_map_modes_deaths.high_threshold
								compare = greater_than_or_equals
							}
						}
						set_temp_variable = { thickness = constant:state_map_modes_deaths.high_thickness }
					}

					clamp_temp_variable = { var = red min = 0 max = 1 }
					clamp_temp_variable = { var = green min = 0 max = 1 }
					clamp_temp_variable = { var = blue min = 0 max = 1 }
					clamp_temp_variable = { var = alpha min = 0 max = 1 }
				}
				always = yes
			}
		}

		bottom = {
			type = country
			thickness = yes
			color = {
				set_temp_variable = { red = constant:state_map_modes_deaths.background_red }
				set_temp_variable = { green = constant:state_map_modes_deaths.background_green }
				set_temp_variable = { blue = constant:state_map_modes_deaths.background_blue }
				set_temp_variable = { alpha = constant:state_map_modes_deaths.background_alpha }
				always = yes
			}
		}

		far_text = country
		near_text = state
		update_daily = yes
	}
