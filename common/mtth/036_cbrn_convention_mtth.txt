# Ratify chance in percent before negotiation pressure.
# Input: temp cbrn_convention_family_id.
cbrn_convention_ai_ratify = {
	base = constant:cbrn_convention_ai.ratify_base
	modifier = { add = constant:cbrn_convention_ai.ratify_war_add has_war = yes }
	modifier = { add = constant:cbrn_convention_ai.ratify_victim_add has_country_flag = cbrn_convention_recent_cbrn_victim }
	modifier = {
		add = constant:cbrn_convention_ai.ratify_minor_protect_add
		is_major = no
		OR = {
			check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.compact }
			check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.protocol }
		}
	}
	modifier = {
		add = constant:cbrn_convention_ai.ratify_capability_add
		OR = {
			AND = {
				OR = {
					check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.chemical_articles }
					check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.first_use }
				}
				cbrn_convention_has_chemical_capability = yes
			}
			AND = {
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.bio_registry }
				cbrn_convention_has_biological_capability = yes
			}
			AND = {
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.nuclear }
				cbrn_convention_has_nuclear_research = yes
			}
			AND = {
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.delivery }
				cbrn_convention_has_delivery_capability = yes
			}
			AND = {
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.charter }
				cbrn_convention_has_two_usable_families = yes
			}
			AND = {
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.program_charter }
				cbrn_convention_participant_allows_biological = yes
				cbrn_convention_has_biological_capability = yes
			}
		}
	}
	modifier = {
		factor = constant:cbrn_convention_ai.ratify_democracy_first_use_factor
		has_government = democratic
		OR = {
			check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.first_use }
			check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.arsenal }
		}
	}
	modifier = {
		factor = constant:cbrn_convention_ai.ratify_reservation_offense_factor
		cbrn_convention_is_reservation = yes
		OR = {
			check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.arsenal }
			check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.nuclear }
		}
	}
}

# Agenda pool weight. Global; input temp cbrn_convention_family_id and the
# gate counts from cbrn_convention_evaluate_family_gate.
cbrn_convention_agenda_family_weight = {
	base = constant:cbrn_convention_agenda_weight.base
	modifier = {
		add = constant:cbrn_convention_agenda_weight.protection_after_attack
		has_global_flag = cbrn_convention_recent_member_attack
		check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.compact }
	}
	modifier = {
		add = constant:cbrn_convention_agenda_weight.inspection_after_dispute
		has_global_flag = cbrn_convention_inspection_dispute
		check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.protocol }
	}
	modifier = {
		add = constant:cbrn_convention_agenda_weight.newest_unlocked
		has_global_flag = cbrn_convention_newest_family_pending
		OR = {
			AND = {
				has_global_flag = cbrn_convention_evolution_2_recorded
				NOT = { has_global_flag = cbrn_convention_evolution_3_recorded }
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.charter }
			}
			AND = {
				has_global_flag = cbrn_convention_evolution_3_recorded
				check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.arsenal }
			}
		}
	}
	# The first agenda after Evolution I is the First-Use Revision.
	modifier = {
		factor = constant:cbrn_convention_ai.priority_family_factor
		has_global_flag = cbrn_convention_priority_first_use
		check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.first_use }
	}
	modifier = {
		add = constant:cbrn_convention_agenda_weight.delivery_gap
		check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.delivery }
		check_variable = { cbrn_convention_c_delivery < constant:cbrn_convention_ai.delivery_gap_members }
	}
	modifier = {
		add = constant:cbrn_convention_agenda_weight.nuclear_members
		check_variable = { cbrn_convention_family_id = constant:cbrn_convention_family.nuclear }
		NOT = { check_variable = { cbrn_convention_c_nuke_research < constant:cbrn_convention_ai.nuclear_research_members } }
	}
}
