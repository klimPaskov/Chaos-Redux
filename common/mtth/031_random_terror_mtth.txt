# COUNTRY scope: an affected government.  The band follows its most severe
# owned crisis state, then the track record and pressure shift it inside the
# band before the scheduler applies a random spread and the band clamp.
random_terror_incident_pulse_days = {
	base = constant:random_terror_mtth_days.pulse_active
	modifier = {
		add = constant:random_terror_mtth_days.pulse_entrenched_offset
		any_owned_state = {
			OR = {
				has_state_flag = random_terror_state_entrenched
				has_state_flag = random_terror_state_armed_insurgency
				has_state_flag = random_terror_state_lost_local_control
			}
		}
	}
	modifier = {
		add = constant:random_terror_mtth_days.pulse_armed_offset
		any_owned_state = {
			OR = {
				has_state_flag = random_terror_state_armed_insurgency
				has_state_flag = random_terror_state_lost_local_control
			}
		}
	}
	modifier = {
		factor = constant:random_terror_mtth_factor.moderate_delay
		OR = {
			has_country_flag = random_terror_containment_mission_success
			has_country_flag = random_terror_transport_mission_success
			has_country_flag = random_terror_capital_mission_success
		}
	}
	modifier = {
		factor = constant:random_terror_mtth_factor.moderate_acceleration
		OR = {
			has_country_flag = random_terror_failed_raid_memory
			has_country_flag = random_terror_containment_mission_failure
			has_country_flag = random_terror_transport_mission_failure
		}
	}
	modifier = {
		factor = constant:random_terror_mtth_factor.moderate_acceleration
		check_variable = { random_terror_pressure > constant:random_terror_value.high }
	}
	modifier = {
		factor = constant:random_terror_mtth_factor.moderate_acceleration
		check_variable = { global.random_terror_network_reach > constant:random_terror_value.high }
	}
}
