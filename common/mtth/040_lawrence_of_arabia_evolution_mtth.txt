# Days from the moment a stage is ready and enabled until it activates.
lawrence_evolution_interval = {
	base = constant:lawrence_evolution.base_days
	modifier = {
		factor = constant:lawrence_evolution.fast_factor
		ENG = { has_war_with_major = yes }
	}
	modifier = {
		factor = constant:lawrence_evolution.fast_factor
		check_variable = { var = global.chaos_meter_value value = constant:chaos_meter_tier_range.tier_3.min compare = greater_than_or_equals }
	}
	modifier = {
		factor = constant:lawrence_evolution.slow_factor
		check_variable = { var = global.lawrence_credibility value = constant:lawrence_drift.credibility_low compare = less_than }
	}
}
