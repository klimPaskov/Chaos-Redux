# These entries provide bounded, country-scoped delays for Event 35 evolution

great_depression_financial_contagion_activation_interval = {
	base = constant:great_depression_evolution_mtth.financial_contagion_days
	modifier = {
		factor = constant:great_depression_evolution_mtth.high_severity_factor
		check_variable = {
			var = great_depression_depression_severity
			value = constant:great_depression_event.deep_depression_threshold
			compare = greater_than_or_equals
		}
	}
	modifier = {
		factor = constant:great_depression_evolution_mtth.severe_social_pressure_factor
		has_stability < constant:great_depression_thresholds.social_peace_stability_floor
	}
	modifier = {
		factor = constant:great_depression_evolution_mtth.strong_link_factor
		OR = {
			is_in_faction = yes
			is_subject = yes
			has_country_flag = great_depression_event34_supplier_contract
		}
	}
	modifier = {
		factor = constant:great_depression_evolution_mtth.improving_trend_factor
		check_variable = { var = great_depression_severity_trend value = constant:great_depression_api.zero compare = less_than }
	}
	modifier = {
		factor = constant:great_depression_evolution_mtth.weak_link_factor
		NOT = { is_in_faction = yes }
		NOT = { is_subject = yes }
		NOT = { has_country_flag = great_depression_event34_supplier_contract }
	}
}
