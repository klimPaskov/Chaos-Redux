# Pacing of the order demands and the AI scores of every decision. Scores are
# weights, not click chances: a score of zero is never taken, and the engine's
# own AI roll decides among the rest. Each entry is evaluated for the actor
# only (the country already being evaluated), so no world scan is needed.
# Every score and factor lives in the malta_ai_score and malta_ai constant
# groups. Rows aimed at a target (regional crusade, ultimatum, landing,
# settlements, port projects) add their target factors inline in the decision,
# where FROM is available. Survival comes before spectacle: the escort, the air
# bridge, the rail and the harbour outrank banners and expansion while supply
# is short. The ranged company comes first among the banners, then the mounted
# column; the heavy Armored Knights and the Siege Host wait for war, and every
# banner halves in peace.
# ============================================================================

# Days between order demands. United councils ask less often, a council in open
# rivalry asks more often. Rotating offices lengthen only a stable council, so
# the factors never stack above the cadence band. The cadence floor lives in
# the demand constants.
malta_demand_interval = {
	base = constant:malta_demand.base_days
	modifier = {
		factor = constant:malta_demand.united_factor
		malta_cohesion_is_united = yes
	}
	modifier = {
		factor = constant:malta_demand.rivalry_factor
		malta_cohesion_is_open_rivalry = yes
	}
	modifier = {
		factor = constant:malta_demand.rotation_factor
		has_country_flag = malta_offices_rotating
		NOT = { malta_cohesion_is_united = yes }
		NOT = { malta_cohesion_is_open_rivalry = yes }
	}
}

# Accept when the price can be paid and the council needs calming. A calm
# council with a small grievance is still answered in full once the one
# compromise of that order is spent.
malta_ai_accept_demand = {
	base = 0
	modifier = {
		add = constant:malta_ai_score.accept_pressing
		malta_can_pay_open_demand = yes
		NOT = { malta_cohesion_is_working_or_better = yes }
	}
	modifier = {
		add = constant:malta_ai_score.accept_grievance
		malta_can_pay_open_demand = yes
		malta_cohesion_is_working_or_better = yes
		check_variable = { var = malta_order_grievance^malta_demand_order value = constant:malta_ai.grievance_pressing compare = greater_than_or_equals }
	}
	modifier = {
		add = constant:malta_ai_score.accept_after_compromise
		malta_can_pay_open_demand = yes
		malta_can_compromise_demand = no
		malta_cohesion_is_working_or_better = yes
		check_variable = { var = malta_order_grievance^malta_demand_order value = constant:malta_ai.grievance_pressing compare = less_than }
	}
	modifier = {
		add = constant:malta_ai_score.accept_influence
		malta_can_pay_open_demand = yes
		check_variable = { var = malta_order_influence^malta_demand_order value = constant:malta_influence.high compare = equals }
	}
}

# Compromise when the full price is out of reach or the order is not pressing,
# and only while the order has not compromised before.
malta_ai_compromise_demand = {
	base = 0
	modifier = {
		add = constant:malta_ai_score.compromise_half
		malta_can_compromise_demand = yes
		malta_can_pay_open_demand = no
		malta_can_pay_open_demand_half = yes
	}
	modifier = {
		add = constant:malta_ai_score.compromise_calm
		malta_can_compromise_demand = yes
		malta_can_pay_open_demand = yes
		malta_cohesion_is_working_or_better = yes
		check_variable = { var = malta_order_grievance^malta_demand_order value = constant:malta_ai.grievance_pressing compare = less_than }
	}
}

# Refuse when the full price cannot be paid and no compromise is left to
# offer, or the order is a minor voice.
malta_ai_refuse_demand = {
	base = 0
	modifier = {
		add = constant:malta_ai_score.refuse_empty
		malta_can_pay_open_demand = no
		OR = {
			malta_can_pay_open_demand_half = no
			malta_can_compromise_demand = no
		}
	}
	modifier = {
		add = constant:malta_ai_score.refuse_minor
		malta_cohesion_is_working_or_better = yes
		check_variable = { var = malta_order_influence^malta_demand_order value = constant:malta_influence.low compare = equals }
	}
}

# Readiness of the actor for a regional crusade: a supplied command and enough
# divisions. The target adds its own factors in the decision. A war already
# running halves the score, a war with a major halves it again, and a war that
# is being lost closes the row, so no new front opens on a failing one.
malta_ai_regional_crusade = {
	base = 0
	modifier = {
		add = constant:malta_ai_score.crusade_ready
		malta_can_pay_regional_crusade = yes
		malta_standing_is_recognized_or_better = yes
		check_variable = { var = num_divisions value = constant:malta_ai.crusade_divisions compare = greater_than_or_equals }
		malta_cohesion_is_working_or_better = yes
	}
	modifier = {
		add = constant:malta_ai_score.crusade_high
		malta_can_pay_regional_crusade = yes
		malta_standing_is_functional_or_better = yes
		check_variable = { var = num_divisions value = constant:malta_ai.crusade_divisions_high compare = greater_than_or_equals }
	}
	modifier = {
		factor = constant:malta_ai_score.crusade_war_factor
		has_war = yes
	}
	modifier = {
		factor = constant:malta_ai_score.crusade_major_war_factor
		has_war_with_major = yes
	}
	modifier = {
		factor = 0
		has_war = yes
		enemies_strength_ratio > constant:malta_ai.losing_ratio
	}
	modifier = {
		factor = 0
		check_variable = { var = malta_anchor_count value = constant:malta_standing.three_anchors compare = less_than }
	}
}

# A launch needs the rebuilt transports, the rebuilt force and a stock that is
# not reckless. A failed landing clears both, so the next launch waits for a
# new rebuild. After the failure cap the AI no longer attacks.
malta_ai_launch_eleventh = {
	base = 0
	modifier = {
		add = constant:malta_ai_score.launch_strong
		malta_can_pay_launch_eleventh = yes
		has_country_flag = malta_transports_rebuilt
		has_country_flag = malta_eleventh_force_ready
		NOT = { has_country_flag = malta_eleventh_force_weak }
	}
	modifier = {
		add = constant:malta_ai_score.launch_weak
		malta_can_pay_launch_eleventh = yes
		has_country_flag = malta_transports_rebuilt
		has_country_flag = malta_eleventh_force_ready
		has_country_flag = malta_eleventh_force_weak
		check_variable = { var = malta_eleventh_failures value = constant:malta_ai.weak_launch_failures compare = less_than }
	}
}

# Days between succession crises of one principality (COUNTRY scope: the
# principality). A Papal appointee and a federated principality rule longer, a
# local council is changed more often.
malta_pr_succession_interval = {
	base = constant:malta_pr_succession.interval_days
	modifier = {
		factor = constant:malta_pr_succession.papal_factor
		check_variable = { var = malta_pr_form_id value = constant:malta_pr_form.papal compare = equals }
	}
	modifier = {
		factor = constant:malta_pr_succession.local_factor
		check_variable = { var = malta_pr_form_id value = constant:malta_pr_form.local compare = equals }
	}
	modifier = {
		factor = constant:malta_pr_succession.federated_factor
		has_country_flag = malta_pr_federated
	}
}
