# CHAOS REDUX - EVENT 040 LAWRENCE OF ARABIA MTTH
# ============================================================================
# Incident pacing, the delay between targets and the AI willingness scores of
# every decision. Scores are weights, not click chances: a decision with a
# score of zero is never taken, and the engine's own AI roll decides among the
# rest. Each entry is read for the one country already being evaluated (the
# live target for target decisions, ENG for British decisions) so no world
# scan is needed. The target profile comes from lawrence_ai_profile and the
# British profile from the state of the war, the routes and the stockpile.
# ============================================================================

# Days between baseline incidents in a live target.
lawrence_incident_interval = {
	base = constant:lawrence_incident.base_days
	modifier = {
		factor = constant:lawrence_incident.fast_factor
		check_variable = { var = lawrence_influence value = constant:lawrence_influence.contested_edge compare = greater_than_or_equals }
	}
	modifier = {
		factor = constant:lawrence_incident.fast_factor
		check_variable = { var = lawrence_officers value = constant:lawrence_officers.capture_threshold compare = greater_than_or_equals }
	}
	modifier = {
		factor = constant:lawrence_incident.slow_factor
		has_country_flag = lawrence_file
	}
	modifier = {
		factor = constant:lawrence_incident.slow_factor
		has_country_flag = lawrence_escorted
	}
	modifier = {
		factor = constant:lawrence_incident.fast_factor
		has_country_flag = lawrence_delivery_pending
	}
}

# Days between one target's settlement and the next target.
lawrence_next_target_delay = {
	base = constant:lawrence_campaign.delay_base_days
	modifier = {
		factor = constant:lawrence_campaign.delay_fast_factor
		check_variable = { var = global.lawrence_clients^num value = constant:lawrence_campaign.delay_client_count compare = greater_than_or_equals }
	}
	modifier = {
		factor = constant:lawrence_campaign.delay_fast_factor
		check_variable = { var = global.lawrence_credibility value = constant:lawrence_drift.credibility_high compare = greater_than }
	}
	modifier = {
		factor = constant:lawrence_campaign.delay_slow_factor
		ENG = { has_war_with_major = yes }
	}
	modifier = {
		factor = constant:lawrence_campaign.delay_slow_factor
		check_variable = { var = global.lawrence_credibility value = constant:lawrence_drift.credibility_low compare = less_than }
	}
	modifier = {
		factor = constant:lawrence_campaign.delay_slow_factor
		check_variable = { var = global.lawrence_dismantled_count value = constant:lawrence_event.zero compare = greater_than }
	}
	modifier = {
		factor = constant:lawrence_campaign.delay_slow_factor
		OR = {
			lawrence_status_is_dead = yes
			lawrence_status_is_defected = yes
			has_global_flag = lawrence_agents_only
		}
	}
}
