secret_alliance_operation_interval = {
	base = constant:secret_alliance_mtth.base_days

	modifier = {
		factor = constant:secret_alliance_mtth.evolution_i_factor
		has_global_flag = secret_alliance_evolution_i
	}
	modifier = {
		factor = constant:secret_alliance_mtth.evolution_ii_factor
		has_global_flag = secret_alliance_evolution_ii
	}
	modifier = {
		factor = constant:secret_alliance_mtth.evolution_iii_factor
		has_global_flag = secret_alliance_evolution_iii
	}
	modifier = {
		factor = constant:secret_alliance_mtth.high_readiness_factor
		check_variable = { global.secret_alliance_readiness > constant:secret_alliance_band.readiness_operational }
	}
	modifier = {
		factor = constant:secret_alliance_mtth.high_alertness_factor
		check_variable = { global.secret_alliance_alertness > constant:secret_alliance_band.preparedness_guarded }
	}
	modifier = {
		factor = constant:secret_alliance_mtth.high_preparedness_factor
		check_variable = { global.secret_alliance_preparedness > constant:secret_alliance_band.preparedness_guarded }
	}
	modifier = {
		factor = constant:secret_alliance_mtth.recovery_factor
		has_global_flag = secret_alliance_operation_recovery
	}
	modifier = {
		factor = constant:secret_alliance_mtth.turned_channel_factor
		has_global_flag = secret_alliance_false_plan_channel_available
	}
}

secret_alliance_invitation_interval = {
	base = constant:secret_alliance_mtth.invitation_base_days

	modifier = {
		factor = constant:secret_alliance_mtth.evolution_ii_factor
		has_global_flag = secret_alliance_evolution_ii
	}
	modifier = {
		factor = constant:secret_alliance_mtth.high_alertness_factor
		check_variable = { global.secret_alliance_alertness > constant:secret_alliance_band.preparedness_guarded }
	}
}

secret_alliance_sponsor_interval = {
	base = constant:secret_alliance_mtth.sponsor_base_days

	modifier = {
		factor = constant:secret_alliance_mtth.second_sponsor_factor
		check_variable = { global.secret_alliance_major_count > constant:secret_alliance_event.evolution_ii_major_cap }
	}
	modifier = {
		factor = constant:secret_alliance_mtth.high_alertness_factor
		check_variable = { global.secret_alliance_alertness > constant:secret_alliance_band.preparedness_guarded }
	}
}
