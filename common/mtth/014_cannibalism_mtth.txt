# ============================================================================
# CHAOS REDUX - EVENT 014 CANNIBALISM MTTH VALUES
# ============================================================================
# These entries return evolution delays in days. Event 014 stores the result
# when a country first qualifies, then compares the elapsed candidate period on
# its targeted self-scheduling pulse. Successful containment therefore delays
# escalation while exploitation, spread, warlords, and high chaos accelerate it.
# ============================================================================

cannibalism_evolution_i_days = {
	base = constant:cannibalism_evolution_mtth.base_days

	modifier = {
		factor = constant:cannibalism_evolution_mtth.exploitation_factor
		has_country_flag = cannibalism_policy_exploitation
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.severe_factor
		check_variable = {
			var = cannibalism_field_hunger
			value = constant:cannibalism_meter.severe
			compare = greater_than_or_equals
		}
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.critical_factor
		check_variable = {
			var = cannibalism_cell_strength
			value = constant:cannibalism_meter.critical
			compare = greater_than_or_equals
		}
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.foreign_spread_factor
		has_country_flag = cannibalism_foreign_seeding_active
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.clean_response_factor
		has_country_flag = cannibalism_policy_open_emergency
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.high_integrity_factor
		check_variable = {
			var = cannibalism_command_integrity
			value = constant:cannibalism_meter.severe
			compare = greater_than_or_equals
		}
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.contained_factor
		has_variable = cannibalism_local_victory_due_date
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.high_chaos_factor
		check_variable = {
			var = global.chaos_meter_value
			value = constant:chaos_meter_tier_range.tier_3.min
			compare = greater_than_or_equals
		}
	}
	modifier = {
		factor = constant:cannibalism_evolution_mtth.final_chaos_factor
		check_variable = {
			var = global.chaos_meter_value
			value = constant:chaos_meter_tier_range.tier_final.min
			compare = greater_than_or_equals
		}
	}
}
