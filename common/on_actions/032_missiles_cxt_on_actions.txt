# ============================================================================
# CHAOS REDUX - EVENT 032 CXT REGISTRATION HOOKS
# ============================================================================
# Startup uses one bounded existing-country scope. The tag-scoped daily
# fallback repairs initialized test saves without a recurring world scan.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				missiles_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				missiles_register_cxt_test_content = yes
				if = {
					limit = {
						check_variable = {
							var = chaosx_test_country_extension_registration_added
							value = 0
							compare = greater_than
						}
					}
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}
}
