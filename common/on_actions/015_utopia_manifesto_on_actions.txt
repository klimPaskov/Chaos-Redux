# ============================================================================
# CHAOS REDUX - EVENT 015 UTOPIA MANIFESTO ON ACTIONS
# ============================================================================
# One-shot history hooks for Event 015. ROOT/FROM scopes follow the installed
# vanilla on-action documentation. No daily, weekly, or monthly world scan is
# used.
# ============================================================================

on_actions = {
	on_declare_war = {
		effect = {
			if = {
				limit = { ROOT = { has_country_flag = utopia_manifesto_accepted } }
				ROOT = { utopia_manifesto_record_achievement_offensive_war = yes }
			}
		}
	}

	# ROOT fires once when a country moves from peace into any war. War changes
	# the live Ledger, but it is not itself constitutional repeal or regime
	# collapse and must never enter the terminal aftermath chain.
	on_war = {
		effect = {
			if = {
				limit = { ROOT = { has_country_flag = utopia_manifesto_accepted } }
				ROOT = { utopia_manifesto_refresh_ledger = yes }
			}
		}
	}

	# ROOT fires once when a country leaves its final active war.
	on_peace = {
		effect = {
			if = {
				limit = { ROOT = { has_country_flag = utopia_manifesto_accepted } }
				ROOT = { utopia_manifesto_refresh_ledger = yes }
			}
		}
	}

	# Subject status changes alter the public Need and Concord accounting. These
	# hooks refresh only the two countries involved in the engine callback; no
	# recurring country or world scan is required.
	on_puppet = {
		effect = {
			if = {
				limit = { has_country_flag = utopia_manifesto_accepted }
				utopia_manifesto_refresh_ledger = yes
			}
			if = {
				limit = { FROM = { has_country_flag = utopia_manifesto_accepted } }
				FROM = { utopia_manifesto_refresh_ledger = yes }
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			if = {
				limit = { has_country_flag = utopia_manifesto_accepted }
				utopia_manifesto_refresh_ledger = yes
			}
			if = {
				limit = { FROM = { has_country_flag = utopia_manifesto_accepted } }
				FROM = { utopia_manifesto_refresh_ledger = yes }
			}
		}
	}

	on_release_as_free = {
		effect = {

				limit = { FROM = { has_country_flag = utopia_manifesto_accepted } }
				FROM = { utopia_manifesto_refresh_ledger = yes }
			}
		}
	}

	on_subject_free = {
		effect = {
			if = {

	on_subject_annexed = {
		effect = {
			if = {

	# ROOT is the attacking side and FROM is the defending side.
	on_war_relation_added = {
		effect = {
			ROOT = { save_event_target_as = utopia_manifesto_war_attacker }
			FROM = { save_event_target_as = utopia_manifesto_war_defender }
			if = {
				limit = {
					ROOT = {
						has_country_flag = utopia_manifesto_accepted
						has_country_flag = utopia_manifesto_case_enforced
					}
					FROM = { utopia_manifesto_is_exact_active_case_target_for_root = yes }
				}
				ROOT = { utopia_manifesto_mark_enforcement_war_active = yes }
			}
			if = {
				limit = { FROM = { has_country_flag = utopia_manifesto_accepted } }
				FROM = { utopia_manifesto_register_defensive_war_challenges = yes }
			}
			# An exact member-to-founder attack is a recorded obligation breach.
			if = {
				limit = {
					ROOT = {
						has_country_flag = utopia_manifesto_league_member
						is_in_array = { array = utopia_manifesto_league_founders value = FROM }
					}
					FROM = {
						has_country_flag = utopia_manifesto_accepted
						has_country_flag = utopia_manifesto_league_initialized
						is_in_array = { array = utopia_manifesto_league_members value = ROOT }
					}
				}
				ROOT = { set_country_flag = utopia_manifesto_league_obligation_breached }
				FROM = {
					utopia_manifesto_record_league_failure = yes
				}
			}
			# A founder attacking its exact member gives that member a one-shot
			# leave-or-remain answer. Pair arrays make the response bridge exact.
			if = {
				limit = {
					ROOT = {
						has_country_flag = utopia_manifesto_accepted
						has_country_flag = utopia_manifesto_league_initialized
						NOT = { has_country_flag = utopia_manifesto_league_collapsed }
						is_in_array = { array = utopia_manifesto_league_members value = FROM }
						NOT = { is_in_array = { array = utopia_manifesto_league_betrayal_pending_targets value = FROM } }
					}
					FROM = {
						has_country_flag = utopia_manifesto_league_member
						is_in_array = { array = utopia_manifesto_league_founders value = ROOT }
					}
				}
				ROOT = {
					set_country_flag = utopia_manifesto_league_founder_betrayal
					set_country_flag = utopia_manifesto_achievement_disqualifier_league_coercion
					set_country_flag = utopia_manifesto_league_betrayal_pending
					add_to_array = { array = utopia_manifesto_league_betrayal_pending_targets value = FROM }
					utopia_manifesto_reconcile_league_reverse_links = yes
				}
				FROM = {
					set_country_flag = utopia_manifesto_league_founder_betrayal_pending
					add_to_array = { array = utopia_manifesto_league_betrayal_pending_founders value = ROOT }
					country_event = { id = chaosx.nr15.217 }
				}
			}
		}
	}

	# ROOT is the capitulated country and FROM is the winner.
	on_capitulation = {
		effect = {
			ROOT = { save_event_target_as = utopia_manifesto_defeated_attacker }
			if = {
				limit = {
					ROOT = {
						check_variable = {
							var = utopia_manifesto_event15_defenders^num
							value = constant:utopia_manifesto_narrative_count.zero
							compare = greater_than
						}
					}
				}
				ROOT = { utopia_manifesto_resolve_recorded_attacker_defeat = yes }
			}
			if = {
				limit = {
					ROOT = {
						exists = yes
						has_capitulated = yes
						has_country_flag = utopia_manifesto_accepted
						has_country_flag = utopia_manifesto_identity_formed
					}
				}
				ROOT = { utopia_manifesto_begin_regime_collapse_aftermath = yes }
			}
			if = {
				limit = { ROOT = { has_country_flag = utopia_manifesto_accepted } }
				ROOT = { utopia_manifesto_refresh_ledger = yes }
			}
			if = {
				limit = { FROM = { has_country_flag = utopia_manifesto_accepted } }
				FROM = { utopia_manifesto_refresh_ledger = yes }
			}
		}
	}

	# ROOT is the country whose government changed. Formation itself cannot enter
	# this path because the identity flag is installed only after route politics.
	on_government_change = {
		effect = {

		}
	}
