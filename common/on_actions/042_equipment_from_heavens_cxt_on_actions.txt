# ============================================================================
# CHAOS REDUX - EVENT 042 CXT REGISTRATION HOOKS
# ============================================================================
# Startup registers the package through one bounded random existing country.
# The tag-scoped daily fallback repairs existing initialized test saves and
# never scans countries or states globally.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				equipment_from_heavens_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				set_temp_variable = { chaosx_test_country_extension_registration_added = 0 }
				equipment_from_heavens_register_cxt_test_content = yes
				if = {
					limit = { check_variable = { var = chaosx_test_country_extension_registration_added value = 0 compare = greater_than } }
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}
}
