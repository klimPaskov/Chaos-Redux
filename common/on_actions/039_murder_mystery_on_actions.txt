# ============================================================================
# CHAOS REDUX - EVENT 039 LIFECYCLE AND CXT HOOKS
# ============================================================================
# The startup registration enters one bounded existing-country scope. The
# daily fallback is tag-scoped to CXT and never scans the world.
# Capitulation and annexation hooks act only on countries that carry an
# Event 039 actor flag while the movement runtime is active, or on registered
# World of Anarchy campaign targets while the campaign is under way.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				murder_mystery_register_event19_assassin_provider = yes
				murder_mystery_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				murder_mystery_register_cxt_test_content = yes
				if = {
					limit = { check_variable = { chaosx_test_country_extension_registration_added > 0 } }
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}

	# ROOT is the capitulated country, FROM is the winner.
	on_capitulation = {
		effect = {
			if = {
				limit = {
					has_global_flag = murder_mystery_runtime_active
					murder_mystery_country_is_event_actor = yes
				}
				FROM = { save_event_target_as = murder_mystery_loss_winner }
				murder_mystery_on_capitulation = yes
			}
			# World of Anarchy: a campaign target that surrendered to the
			# movement waits for its successor administration.
			if = {
				limit = {
					has_global_flag = murder_mystery_terminal_war_active
					has_country_flag = murder_mystery_terminal_target_registered
				}
				FROM = { save_event_target_as = murder_mystery_loss_winner }
				murder_mystery_on_terminal_target_capitulated = yes
			}
		}
	}

	# ROOT is the winner, FROM is the annexed country.
	on_annex = {
		effect = {
			if = {
				limit = {
					has_global_flag = murder_mystery_runtime_active
					FROM = { murder_mystery_country_is_event_actor = yes }
				}
				save_event_target_as = murder_mystery_loss_winner
				FROM = { murder_mystery_on_annexed = yes }
			}
			# World of Anarchy: an annexed campaign target settles its row.
			if = {
				limit = {
					has_global_flag = murder_mystery_terminal_war_active
					FROM = { has_country_flag = murder_mystery_terminal_target_registered }
				}
				save_event_target_as = murder_mystery_loss_winner
				FROM = { murder_mystery_on_terminal_target_annexed = yes }
			}
		}
	}
}
