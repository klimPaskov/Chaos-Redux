# ============================================================================
# CHAOS REDUX - EVENT 012 SCRAMBLE AND CONTINENTAL WAR ON ACTIONS
# ============================================================================
# Pairwise war, capitulation, and peace hooks for the one-time Scramble crisis
# and prepared continent wars. No recurring country iteration is used.
# ROOT is the attacker in on_war_relation_added and the capitulated country in
# on_capitulation. ROOT and FROM are checked in both orders at peace.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				africa_register_event19_event12_providers = yes
				africa_register_strange_force_asset_manifests = yes
			}
		}
	}

	on_join_faction = {
		effect = {
			if = {
				limit = { has_country_flag = africa_achievement_peaceful_exit_started }
				africa_achievement_record_peaceful_reassociation = yes
			}
			# ROOT is the joining member. The helper resolves the current faction
			# leader and records only an external patron; internal League factions
			# remain outside the patronage state.
			africa_reconcile_foreign_patronage_state = yes
		}
	}

	on_leave_faction = {
		effect = {
			# The member has already left the faction when this callback runs, so
			# the same helper clears the derived receipt without a world scan.
			africa_reconcile_foreign_patronage_state = yes
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = { has_event_target = africa_host }
				event_target:africa_host = { africa_common_reserve_clear_expired_posture = yes }
			}
			africa_promoted_tiera_record_stoneborn_human_member_war_relation = yes
			if = {
				limit = {
					ROOT = { africa_is_current_host = yes }
					FROM = { has_country_flag = africa_scramble_expedition_leader }
				}
				ROOT = {
					set_country_flag = africa_scramble_defensive_war_active
					set_global_flag = africa_scramble_intervention_war_active
				}
				FROM = { set_country_flag = africa_scramble_intervention_war_actor }
			}
			else_if = {
				limit = {
					FROM = { africa_is_current_host = yes }
					ROOT = { has_country_flag = africa_scramble_expedition_leader }
				}
				FROM = {
					set_country_flag = africa_scramble_defensive_war_active
					set_global_flag = africa_scramble_intervention_war_active
				}
				ROOT = { set_country_flag = africa_scramble_intervention_war_actor }
			}
			else_if = {
				limit = {
					ROOT = { africa_is_current_host = yes }
					FROM = { has_country_flag = africa_continental_war_opponent }
				}
				ROOT = { set_country_flag = africa_continental_war_active }
				FROM = { set_country_flag = africa_continental_war_active }
			}
			else_if = {
				limit = {
					FROM = { africa_is_current_host = yes }
					ROOT = { has_country_flag = africa_continental_war_opponent }
				}
				FROM = { set_country_flag = africa_continental_war_active }
				ROOT = { set_country_flag = africa_continental_war_active }
			}
			if = {
				limit = {
					FROM = {
						africa_relationship_is_protected = yes
						has_country_flag = africa_achievement_counted_protected_partner
						africa_member_host_generation_is_current = yes
					}
				}
				FROM = { africa_achievement_record_protection_war_started = yes }
			}
			# ROOT is the attacker and FROM is the defender in this callback. A
			# live common-reserve posture therefore funds only the protected
			# partner's defensive war, one stockpile-backed deployment at a time.
			if = {
				limit = { FROM = { africa_common_reserve_can_cover_defensive_war = yes } }
				FROM = { africa_common_reserve_deploy_on_defensive_war = yes }
			}
			# A negotiated departure fails the no-war achievement at the exact
			# declaration callback, regardless of which side opened hostilities.
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = { africa_is_current_host = yes }
							FROM = { has_country_flag = africa_achievement_peaceful_exit_started }
						}
						AND = {
							FROM = { africa_is_current_host = yes }
							ROOT = { has_country_flag = africa_achievement_peaceful_exit_started }
						}
					}
				}
				set_global_flag = africa_achievement_exit_war_or_coup
			}
			# The Horn settlement requires the separately recognised Aksum and
			# Harar packages to remain at peace with one another.
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = {
								africa_member_host_generation_is_current = yes
								has_country_flag = africa_achievement_restoration_identity_aksum_recorded
							}
							FROM = {
								africa_member_host_generation_is_current = yes
								has_country_flag = africa_achievement_restoration_identity_harar_recorded
							}
						}
						AND = {
							ROOT = {
								africa_member_host_generation_is_current = yes
								has_country_flag = africa_achievement_restoration_identity_harar_recorded
							}
							FROM = {
								africa_member_host_generation_is_current = yes
								has_country_flag = africa_achievement_restoration_identity_aksum_recorded
							}
						}
					}
				}
				set_global_flag = africa_achievement_horn_package_war
				if = {
					limit = { has_event_target = africa_host }
					event_target:africa_host = { africa_achievement_record_horn_corridor_loss = yes }
				}
			}
			# A protected partner that starts an offensive war while the reserve
			# posture is open never receives reserve material, but the attempted
			# misuse remains a lifetime disqualifier for the achievement.
			if = {
				limit = {
					ROOT = {
						africa_relationship_is_protected = yes
						africa_member_host_generation_is_current = yes
						has_country_flag = africa_achievement_counted_protected_partner
					}
					has_event_target = africa_host
					event_target:africa_host = { has_country_flag = africa_common_reserve_active }
				}
				event_target:africa_host = { africa_achievement_record_reserve_offensive_abuse = yes }
			}
			# A grounded military sponsorship obligation is answered only when
			# the candidate is one side of the newly registered package war.
			if = {
				limit = {
					ROOT = { africa_world_sponsorship_military_obligation_is_due = yes }
					FROM = { africa_world_package_is_installed = yes }
				}
				ROOT = { africa_world_sponsorship_call_military_defence = yes }
			}
			if = {
				limit = {
					FROM = { africa_world_sponsorship_military_obligation_is_due = yes }
					ROOT = { africa_world_package_is_installed = yes }
				}
				FROM = { africa_world_sponsorship_call_military_defence = yes }
			}
			# W4 relation witness. The registered pair is already opened by the
			# prepare effect; this branch only confirms both scopes and marks the
			# protocol active without scanning or inferring partner membership.
			if = {
				limit = {
					has_global_flag = africa_world_continental_war_protocol_open
					has_event_target = africa_world_continental_war_protocol_attacker
					has_event_target = africa_world_continental_war_protocol_defender
					OR = {
						ROOT = { tag = event_target:africa_world_continental_war_protocol_attacker }
						ROOT = { tag = event_target:africa_world_continental_war_protocol_defender }
						FROM = { tag = event_target:africa_world_continental_war_protocol_attacker }
						FROM = { tag = event_target:africa_world_continental_war_protocol_defender }
					}
				}
				if = {
					limit = { ROOT = { tag = event_target:africa_world_continental_war_protocol_attacker } }
					ROOT = { set_country_flag = africa_world_continental_war_protocol_active }
				}
				if = {
					limit = { ROOT = { tag = event_target:africa_world_continental_war_protocol_defender } }
					ROOT = { set_country_flag = africa_world_continental_war_protocol_active }
				}
				if = {
					limit = { FROM = { tag = event_target:africa_world_continental_war_protocol_attacker } }
					FROM = { set_country_flag = africa_world_continental_war_protocol_active }
				}
				if = {
					limit = { FROM = { tag = event_target:africa_world_continental_war_protocol_defender } }
					FROM = { set_country_flag = africa_world_continental_war_protocol_active }
				}
			}
		}
	}

	# ROOT is the new puppet and FROM is its overlord. This is the exact
	# subject-creation witness for an Event 012 host/member that retains an
	# African core under an external capital; no current-control proxy is used.
	# A completed diaspora bond actor is classified separately from an unrelated
	# foreign overlord before the shared no-foreign-boot receipt is written.
	on_puppet = {
		effect = {
			if = {
				limit = {
					africa_priority_member_has_package = yes
					africa_member_host_generation_is_current = yes
					is_subject = yes
					FROM = { capital_scope = { NOT = { is_on_continent = africa } } }
				}
				set_global_flag = africa_achievement_priority_member_colonial_puppet
			}
			africa_world_record_sponsored_puppet_status = yes
			if = {
				limit = {
					any_core_state = { is_on_continent = africa }
					OR = {
						africa_is_current_host = yes
						africa_member_host_generation_is_current = yes
					}
					FROM = {
						capital_scope = {
							NOT = { is_on_continent = africa }
						}
					}
				}
				if = {
					limit = { africa_is_current_host = yes }
					if = {
						limit = { FROM = { has_country_flag = africa_diaspora_bond_issue_completed } }
						africa_achievement_record_diaspora_government_capture = yes
					}
					else = {
						africa_achievement_record_foreign_government_capture = yes
					}
				}
				africa_achievement_record_external_puppet_status = yes
			}
		}
	}

	# ROOT is the released subject and FROM is the overlord when the occupied
	# territories menu creates a puppet outside a peace conference. This is a
	# separate engine callback from on_puppet, but uses the same host-only
	# government-capture classification and the same no-foreign-boot receipt.
	on_release_as_puppet = {
		effect = {
			if = {
				limit = {
					africa_priority_member_has_package = yes
					africa_member_host_generation_is_current = yes
					is_subject = yes
					FROM = { capital_scope = { NOT = { is_on_continent = africa } } }
				}
				set_global_flag = africa_achievement_priority_member_colonial_puppet
			}
			africa_world_record_sponsored_puppet_status = yes
			if = {
				limit = {
					any_core_state = { is_on_continent = africa }
					africa_is_current_host = yes
					FROM = {
						capital_scope = {
							NOT = { is_on_continent = africa }
						}
					}
				}
				if = {
					limit = { FROM = { has_country_flag = africa_diaspora_bond_issue_completed } }
					africa_achievement_record_diaspora_government_capture = yes
				}
				else = {
					africa_achievement_record_foreign_government_capture = yes
				}
				africa_achievement_record_external_puppet_status = yes
			}
		}
	}

	# ROOT is the subject and FROM is its overlord when an autonomy effect
	# changes the subject level outside the two explicit release callbacks.
	# The external-capital and host-only guards make the state itself the exact
	# capture witness; ordinary African Charter-member autonomy cannot qualify.
	on_subject_autonomy_level_change = {
		effect = {
			if = {
				limit = {
					africa_priority_member_has_package = yes
					africa_member_host_generation_is_current = yes
					is_subject = yes
					FROM = { capital_scope = { NOT = { is_on_continent = africa } } }
				}
				set_global_flag = africa_achievement_priority_member_colonial_puppet
			}
			africa_world_record_sponsored_puppet_status = yes
			if = {
				limit = {
					any_core_state = { is_on_continent = africa }
					africa_is_current_host = yes
					is_subject = yes
					FROM = {
						capital_scope = {
							NOT = { is_on_continent = africa }
						}
					}
				}
				if = {
					limit = { FROM = { has_country_flag = africa_diaspora_bond_issue_completed } }
					africa_achievement_record_diaspora_government_capture = yes
				}
				else = {
					africa_achievement_record_foreign_government_capture = yes
				}
				africa_achievement_record_external_puppet_status = yes
			}
		}
	}

	# ROOT is the annexer and FROM is still the victim during this callback.
	# The victim's Event 012 generation or immutable achievement-counted flags
	# identify the exact member/partner transaction before the country is removed.
	on_annex = {
		effect = {
			# Preserve the direct claimant until the victim-owned achievement
			# callback has classified the annexation. The regular target clears with
			# this effect chain and cannot leak into a later transaction.
			ROOT = { save_event_target_as = africa_achievement_annexer }
			# The host must stage its ledger before the annexed country scope is
			# removed. A recorded constitutional designation can complete the
			# transfer in this callback; otherwise the annexer receives the explicit
			# congress/crisis resolver with the surviving Charter pool.
			if = {
				limit = {
					FROM = {
						africa_is_current_host = yes
						has_global_flag = africa_host_successor_consumed
					}
				}
				FROM = { africa_host_transfer_record_terminal_loss = yes }
			}
			else_if = {
				limit = { FROM = { africa_is_current_host = yes } }
				FROM = { africa_begin_host_transfer = yes }
			}
			if = {
				limit = { FROM = { has_country_flag = africa_common_reserve_deployment_active } }
				# The resolver owns the generation-gated capital-loss receipt and
				# always cleans the deployment callback, including stale transfers.
				FROM = { africa_common_reserve_resolve_deployment = yes }
			}
			if = {
				limit = {
					FROM = {
						OR = {
							has_variable = africa_member_host_generation
							has_country_flag = africa_achievement_counted_protected_partner
							has_country_flag = africa_achievement_counted_guarantee
							has_country_flag = africa_achievement_last_convoy_candidate
							has_country_flag = africa_achievement_archive_evacuated
							has_country_flag = africa_achievement_restoration_identity_luba_recorded
							has_country_flag = africa_achievement_restoration_identity_lunda_recorded
							has_country_flag = africa_achievement_restoration_identity_kuba_recorded
							has_country_flag = africa_achievement_restoration_identity_kush_recorded
							has_country_flag = africa_achievement_restoration_identity_nubia_recorded
							has_country_flag = africa_achievement_restoration_identity_makuria_recorded
							has_country_flag = africa_achievement_restoration_identity_alodia_recorded
							has_country_flag = africa_achievement_restoration_identity_aksum_recorded
							has_country_flag = africa_achievement_restoration_identity_harar_recorded
							has_country_flag = africa_achievement_restoration_identity_great_zimbabwe_recorded
							has_country_flag = africa_achievement_restoration_identity_mutapa_recorded
							has_country_flag = africa_achievement_restoration_identity_rozwi_recorded
						}
					}
				}
				FROM = { africa_achievement_record_coercive_annexation = yes }
			}
			if = {
				limit = { FROM = { africa_world_package_is_installed = yes } }
				FROM = { africa_world_handle_package_actor_loss = yes }
			}
			# An active action target can disappear before its targeted mission
			# evaluates the NOT-exists cancellation path. Resolve the victim-owned
			# ledger first; the helper refunds only a current host generation and
			# uses generation-matched state locks for stale records.
			if = {
				limit = { FROM = { has_country_flag = africa_action_record_active } }
				FROM = { africa_cleanup_annexed_action_target = yes }
			}
			# A priority-member country can be removed by an ordinary annexation
			# before a later relationship callback runs. Consume its transient
			# package mission and relationship runtime only after the achievement
			# and world-package loss owners above have read the immutable proof.
			if = {
				limit = { FROM = { has_country_flag = africa_priority_member_package_active } }
				FROM = { africa_priority_member_cleanup_runtime = yes }
			}
			# The victim still exists during this callback. Only a country that
			# actually supplied one of the live regional witnesses must be
			# excluded and cause the bounded current-host ledgers to rebuild.
			if = {
				limit = {
					FROM = {
						OR = {
							AND = {
								africa_member_host_generation_is_current = yes
								has_variable = africa_regional_overlay
							}
							has_country_flag = africa_achievement_live_overlap_settlement_proof
							has_country_flag = africa_achievement_live_continental_connection_proof
							has_country_flag = africa_achievement_live_worker_peasant_proof
							has_country_flag = africa_achievement_live_food_security_proof
						}
					}
				}
				FROM = { set_country_flag = africa_achievement_exclude_from_live_region_proof }
				for_each_scope_loop = {
					array = global.africa_achievement_current_hosts
					if = {
						limit = { exists = yes }
						africa_achievement_refresh_survival_windows = yes
					}
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM.FROM = {
						has_state_flag = africa_achievement_port_project_counted
						NOT = { has_state_flag = africa_achievement_port_loss_counted }
					}
					FROM = {
						OR = {
							africa_is_current_host = yes
							africa_member_host_generation_is_current = yes
						}
					}
					ROOT = {
						NOT = {
							OR = {
								africa_is_current_host = yes
								africa_member_host_generation_is_current = yes
							}
						}
					}
				}
				FROM.FROM = { set_state_flag = africa_achievement_port_loss_counted }
				add_to_variable = { global.africa_achievement_maritime_ports_lost = constant:africa_value.one }
				if = {
					limit = {
						check_variable = { var = global.africa_achievement_maritime_ports_lost value = constant:africa_value.two compare = greater_than_or_equals }
					}
					set_global_flag = africa_achievement_two_ports_lost
				}
			}
			ROOT = {
				if = {
					limit = { africa_world_package_is_installed = yes }
					africa_world_refresh_package_polity_proof = yes
				}
			}
			FROM = {
				if = {
					limit = { africa_world_package_is_installed = yes }
					africa_world_refresh_package_polity_proof = yes
				}
			}
			for_each_scope_loop = {
				array = global.africa_achievement_current_hosts
				if = {
					limit = { exists = yes }
					africa_achievement_refresh_survival_windows = yes
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = { ROOT = { has_country_flag = africa_common_reserve_deployment_active } }
				ROOT = { africa_common_reserve_resolve_deployment = yes }
			}
			# A capitulation never credits the row-24 rival-bloc victory. That owner
			# is deliberately restricted to on_peaceconference_ended, where ROOT is
			# the non-capitulation peace winner and FROM is the committed host.
			# A weather-marked hostile country is an army proof only when it
			# capitulates directly to the current Event 012 host. The target flag
			# is written by the accepted Event 013 selected-country call; ordinary
			# peace, a third-party capitulation, or a generic war relation cannot
			# satisfy the weather campaign owner.
			if = {
				limit = {
					ROOT = {
						has_country_flag = africa_natural_disaster_targeted
						has_capitulated = yes
						has_variable = africa_natural_disaster_target_host_generation
						check_variable = {
							var = africa_natural_disaster_target_host_generation
							value = FROM.africa_host_generation
							compare = equals
						}
					}
					FROM = {
						africa_is_current_host = yes
						has_country_flag = africa_host_commit_completed
					}
				}
				ROOT = { africa_achievement_record_weather_army_defeated = yes }
				FROM = { africa_achievement_record_weather_war_won = yes }
			}
			if = {
				limit = {
					ROOT = {
						africa_world_package_is_installed = yes
						has_capitulated = yes
					}
				}
				ROOT = { africa_world_handle_package_actor_loss = yes }
			}
			if = {
				limit = {
					has_global_flag = africa_world_continental_war_protocol_open
					ROOT = {
						africa_world_package_is_installed = yes
						has_country_flag = africa_world_package_actor
						has_country_flag = africa_world_continental_war_protocol_defender
						has_capitulated = yes
						NOT = { has_country_flag = africa_world_union_war_defeated_review_open }
					}
				}
				ROOT = { africa_world_union_war_open_defender_settlement = yes }
			}
			if = {
				limit = {
					ROOT = {
						africa_is_current_host = yes
						has_global_flag = africa_scramble_intervention_war_active
						has_capitulated = yes
					}
				}
				ROOT = { africa_scramble_record_continental_defeat = yes }
			}
			else_if = {
				limit = {
					ROOT = {
						has_country_flag = africa_scramble_expedition_leader
						has_capitulated = yes
					}
					FROM = { africa_is_current_host = yes }
				}
				ROOT = {
					set_country_flag = africa_scramble_expedition_defeated
					clr_country_flag = africa_scramble_intervention_war_actor
					if = {
						limit = { NOT = { has_country_flag = africa_achievement_foreign_expedition_counted } }
						set_country_flag = africa_achievement_foreign_expedition_counted
						africa_achievement_record_foreign_expedition_defeated = yes
					}
					if = {
						limit = { NOT = { has_country_flag = africa_achievement_major_intervention_counted } }
						set_country_flag = africa_achievement_major_intervention_counted
						africa_achievement_record_major_intervention_defeated = yes
					}
				}
				FROM = {
					africa_scramble_reconcile_intervention_war = yes
				}
			}
			else_if = {
				limit = {
					ROOT = {
						africa_world_package_is_installed = yes
						has_country_flag = africa_continental_war_opponent
						has_capitulated = yes
					}
					FROM = { africa_is_current_host = yes }
				}
				ROOT = {
					set_country_flag = africa_defeated_continent_power
					clr_country_flag = africa_continental_war_active
				}
				FROM = {
					clr_country_flag = africa_continental_war_active
					africa_achievement_record_continental_war_victory = yes
				}
			}
			if = {
				limit = {
					ROOT = {
						africa_relationship_is_cooperative = yes
						has_capitulated = yes
					}
				}
				ROOT = { africa_achievement_record_member_capitulation = yes }
			}
			for_each_scope_loop = {
				array = global.africa_achievement_current_hosts
				if = { limit = { exists = yes } africa_achievement_refresh_live_regional_proofs = yes }
			}
		}
	}

	on_peace = {
		effect = {
			if = {
				limit = { ROOT = { has_country_flag = africa_common_reserve_deployment_active } }
				ROOT = { africa_common_reserve_resolve_deployment = yes }
			}
			if = {
				limit = { FROM = { has_country_flag = africa_common_reserve_deployment_active } }
				FROM = { africa_common_reserve_resolve_deployment = yes }
			}
			# A peace callback is authoritative for negotiated independence or a
			# settled protection war even when no peace conference follows. These
			# target-owned writers are idempotent and retain the exact generation,
			# no-war, independence, and 180-day survival guards.
			if = {
				limit = { ROOT = { has_country_flag = africa_achievement_protection_war_unresolved } }
				ROOT = {
					africa_achievement_record_protection_war_settled = yes
					africa_achievement_record_last_convoy_war_settlement = yes
				}
			}
			if = {
				limit = { FROM = { has_country_flag = africa_achievement_protection_war_unresolved } }
				FROM = {
					africa_achievement_record_protection_war_settled = yes
					africa_achievement_record_last_convoy_war_settlement = yes
				}
			}
			if = {
				limit = { ROOT = { has_country_flag = africa_achievement_last_convoy_war_required } }
				ROOT = { africa_achievement_record_last_convoy_war_settlement = yes }
			}
			if = {
				limit = { FROM = { has_country_flag = africa_achievement_last_convoy_war_required } }
				FROM = { africa_achievement_record_last_convoy_war_settlement = yes }
			}
			# A negotiated end to an expedition war clears only the settling
			# actor. The host remains in the intervention phase until every
			# flagged expedition actor has left its war with the host.
			if = {
				limit = {
					ROOT = { has_country_flag = africa_scramble_intervention_war_actor }
					FROM = { africa_is_current_host = yes }
				}
				ROOT = { clr_country_flag = africa_scramble_intervention_war_actor }
				FROM = { africa_scramble_reconcile_intervention_war = yes }
			}
			else_if = {
				limit = {
					FROM = { has_country_flag = africa_scramble_intervention_war_actor }
					ROOT = { africa_is_current_host = yes }
				}
				FROM = { clr_country_flag = africa_scramble_intervention_war_actor }
				ROOT = { africa_scramble_reconcile_intervention_war = yes }
			}
			if = {
				limit = {
					has_global_flag = africa_world_continental_war_protocol_open
					has_event_target = africa_world_continental_war_protocol_attacker
					has_event_target = africa_world_continental_war_protocol_defender
					OR = {
						ROOT = { tag = event_target:africa_world_continental_war_protocol_attacker }
						ROOT = { tag = event_target:africa_world_continental_war_protocol_defender }
						FROM = { tag = event_target:africa_world_continental_war_protocol_attacker }
						FROM = { tag = event_target:africa_world_continental_war_protocol_defender }
					}
					event_target:africa_world_continental_war_protocol_attacker = {
						NOT = { has_war_with = event_target:africa_world_continental_war_protocol_defender }
					}
				}
				if = {
					limit = { event_target:africa_world_continental_war_protocol_attacker = { has_country_flag = africa_world_continental_war_protocol_settled } }
					event_target:africa_world_continental_war_protocol_attacker = { country_event = { id = africa_world_package.734 } }
				}
				else_if = {
					limit = { event_target:africa_world_continental_war_protocol_defender = { has_country_flag = africa_world_continental_war_protocol_settled } }
					event_target:africa_world_continental_war_protocol_defender = { country_event = { id = africa_world_package.734 } }
				}
				else = {
					event_target:africa_world_continental_war_protocol_attacker = {
						set_variable = { africa_world_continental_war_status = constant:africa_world_war_protocol.status.collapsed }
						set_country_flag = africa_world_continental_war_protocol_collapsed
						country_event = { id = africa_world_package.733 }
					}
					event_target:africa_world_continental_war_protocol_defender = {
						set_variable = { africa_world_continental_war_status = constant:africa_world_war_protocol.status.collapsed }
					}
					event_target:africa_world_continental_war_protocol_attacker = { africa_world_continental_war_protocol_cleanup = yes }
				}
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			# ROOT is the peace-conference winner and FROM the defeated party. A
			# rival priority member earns Row 24 only from this explicit winner
			# relationship after a non-capitulation peace; the separate
			# on_capitulation hook remains responsible for the host-lost branch.
			if = {
				limit = {
					ROOT = {
						africa_priority_member_has_package = yes
						africa_relationship_is_rival_bloc = yes
						has_country_flag = africa_priority_member_rival_departure_chosen
						NOT = { has_capitulated = yes }
					}
					FROM = {
						africa_is_current_host = yes
						has_country_flag = africa_host_commit_completed
					}
				}
				ROOT = { africa_priority_member_record_rival_bloc_victory = yes }
			}
			if = {
				limit = { ROOT = { africa_world_package_is_installed = yes has_capitulated = yes } }
				ROOT = { africa_world_handle_package_actor_loss = yes }
			}
			if = {
				limit = { FROM = { africa_world_package_is_installed = yes has_capitulated = yes } }
				FROM = { africa_world_handle_package_actor_loss = yes }
			}
			if = {
				limit = {
					ROOT = { africa_is_current_host = yes }
					FROM = { has_country_flag = africa_scramble_intervention_war_actor }
				}
				FROM = {
					clr_country_flag = africa_scramble_intervention_war_actor
					if = {
						limit = { NOT = { has_country_flag = africa_achievement_foreign_expedition_counted } }
						set_country_flag = africa_achievement_foreign_expedition_counted
						africa_achievement_record_foreign_expedition_defeated = yes
					}
					if = {
						limit = { NOT = { has_country_flag = africa_achievement_major_intervention_counted } }
						set_country_flag = africa_achievement_major_intervention_counted
						africa_achievement_record_major_intervention_defeated = yes
					}
				}
				ROOT = { africa_scramble_reconcile_intervention_war = yes }
			}
			else_if = {
				limit = {
					FROM = { africa_is_current_host = yes }
					ROOT = { has_country_flag = africa_scramble_intervention_war_actor }
				}
				FROM = {
					set_country_flag = africa_scramble_defensive_war_lost
					set_country_flag = africa_postwar_constitutional_review_pending
					set_country_flag = africa_fragmentation_crisis_active
					clr_country_flag = africa_scramble_defensive_war_active
					clr_global_flag = africa_scramble_intervention_war_active
					clr_global_flag = africa_scramble_response_active
					clr_country_flag = africa_scramble_response_open
					clr_country_flag = africa_scramble_recognition_phase
					clr_country_flag = africa_scramble_coalition_phase
					clr_country_flag = africa_scramble_intervention_phase
					set_variable = { africa_selected_action_family = constant:africa_action_family.post_unification }
					set_variable = { africa_selected_action_id = constant:africa_action.none }
					africa_clear_action_quote = yes
					africa_clear_selected_action_state_context = yes
				}
				ROOT = { clr_country_flag = africa_scramble_intervention_war_actor }
			}
			else_if = {
				limit = {
					ROOT = { africa_is_current_host = yes }
					FROM = { has_country_flag = africa_continental_war_opponent }
				}
				FROM = {
					set_country_flag = africa_defeated_continent_power
					clr_country_flag = africa_continental_war_active
				}
				ROOT = {
					clr_country_flag = africa_continental_war_active
					africa_achievement_record_continental_war_victory = yes
					africa_achievement_record_continental_settlement = yes
				}
			}
			else_if = {
				limit = {
					FROM = { africa_is_current_host = yes }
					ROOT = { has_country_flag = africa_continental_war_opponent }
				}
				ROOT = { clr_country_flag = africa_continental_war_active }
				FROM = {
					set_country_flag = africa_continental_campaign_defeated
					set_country_flag = africa_postwar_constitutional_review_pending
					clr_country_flag = africa_continental_war_active
				}
			}
			# The peace-conference callback can arrive after on_peace has already
			# cleared the actor flag. Re-run the host owner while the intervention
			# global is live so the final actor still opens aftermath deterministically.
			if = {
				limit = {
					has_global_flag = africa_scramble_intervention_war_active
					OR = {
						ROOT = { africa_is_current_host = yes }
						FROM = { africa_is_current_host = yes }
					}
				}
				if = {
					limit = { ROOT = { africa_is_current_host = yes } }
					ROOT = { africa_scramble_reconcile_intervention_war = yes }
				}
				else = { FROM = { africa_scramble_reconcile_intervention_war = yes } }
			}
			if = {
				limit = {
					ROOT = { has_country_flag = africa_achievement_protection_war_unresolved }
				}
				ROOT = {
					africa_achievement_record_protection_war_settled = yes
					africa_achievement_record_last_convoy_war_settlement = yes
				}
			}
			if = {
				limit = {
					FROM = { has_country_flag = africa_achievement_protection_war_unresolved }
				}
				FROM = {
					africa_achievement_record_protection_war_settled = yes
					africa_achievement_record_last_convoy_war_settlement = yes
				}
			}
			# Last Convoy is target-owned and may be an associate/chartered partner
			# rather than a protected-war participant. Its settlement writer runs on
			# the exact candidate after every colonial war has actually ended.
			if = {
				limit = { ROOT = { has_country_flag = africa_achievement_last_convoy_war_required } }
				ROOT = { africa_achievement_record_last_convoy_war_settlement = yes }
			}
			if = {
				limit = { FROM = { has_country_flag = africa_achievement_last_convoy_war_required } }
				FROM = { africa_achievement_record_last_convoy_war_settlement = yes }
			}
		}
	}
