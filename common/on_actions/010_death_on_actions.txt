# ============================================================================
# CHAOS REDUX - DEATH ON ACTIONS
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				chaos_unit_family_provider_502_register = yes
				death_register_event19_ghost_variant_provider = yes
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = { fallout_world_rewrite_callbacks_are_allowed = yes }
				if = {
					limit = {
						FROM.FROM = { has_state_flag = death_consumed_state }
					}
					FROM.FROM = {
						death_on_state_control_changed = yes
					}
				}
			}
		}
	}

	on_daily_DTH = {
		effect = {
			death_run_daily_public_pressure = yes
		}
	}

	on_puppet = {
		effect = {
			if = {
				limit = { original_tag = DTH }
				death_on_puppet = yes
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					NOT = { is_death_country = yes }
					FROM = { original_tag = DTH }
				}
				death_mark_living_war_pressure = yes
			}
			if = {
				limit = {
					original_tag = DTH
					FROM = { NOT = { is_death_country = yes } }
				}
				death_mark_living_war_pressure = yes
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = { original_tag = DTH }
				death_on_capitulation = yes
			}
		}
	}
}
