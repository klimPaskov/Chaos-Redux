# ============================================================================
# CHAOS REDUX - EVENT 021 CXT REGISTRATION HOOKS
# ============================================================================
# Startup uses one bounded existing-country scope to initialize the framework,
# publish its readiness gate, and register the CXT carrier. The tag-scoped
# daily hook repairs initialized test saves without adding a world iteration.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				event021_parent_initialize_global_runtime = yes
				random_civil_war_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				random_civil_war_register_cxt_test_content = yes
				if = {
					limit = { check_variable = { chaosx_test_country_extension_registration_added > 0 } }
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}

	# Event-driven receipts, not whole-world periodic reviews. State-control
	# changes interrupt the exact active capital/rail hold target by its saved
	# state pointer before the existing continuity receipt is updated.
	on_state_control_changed = {
		effect = {
			FROM.FROM = { save_event_target_as = event021_mission_control_changed_state }
			FROM = {
				if = {
					limit = {
						has_country_flag = event021_hold_capital_mission_active
						has_variable = event021_hold_capital_mission_target_state
						var:event021_hold_capital_mission_target_state = {
							state = event_target:event021_mission_control_changed_state
						}
					}
					event021_mission_hold_capital_timeout = yes
				}
				if = {
					limit = {
						has_country_flag = event021_secure_rail_mission_active
						has_variable = random_civil_war_rail_mission_state
						var:random_civil_war_rail_mission_state = {
							state = event_target:event021_mission_control_changed_state
						}
					}
					event021_mission_secure_rail_timeout = yes
				}
				event021_record_capital_continuity = yes
			}
			# State Authority follows the war. Only a live, unsettled crisis side
			# reacts: the old controller loses its crisis capital, bound supply
			# state, or territory once, and the new controller recovers them.
			FROM = {
				if = {
					limit = { event021_authority_follows_control = yes }
					event021_authority_review_lost_state = yes
				}
			}
			ROOT = {
				if = {
					limit = { event021_authority_follows_control = yes }
					event021_authority_review_gained_state = yes
				}
			}
		}
	}

	on_capitulation = {
		effect = { event021_record_external_capitulation = yes }
	}

	# Natural civil-war victory reaches this boundary before the engine removes
	# the losing country. Reuse the same explicit same-crisis successor proof so
	# the front roster is prepared before any annex callback or teardown.
	on_civil_war_end_before_annexation = {
		effect = {
			ROOT = { event021_parent_prepare_annex_successor = yes }
		}
	}

	# ROOT is the annexer and FROM is the annexed country. This transaction-bound
	# callback prevents removed countries from leaving stale bounded scheduler,
	# target-count, queue, theater, or front membership.
	on_annex = {
		effect = {
			event021_parent_prepare_annex_successor = yes
			FROM = {
				event021_treaty_rebind_successor = yes
				event021_handle_annexed_country = yes
			}
			event021_adopt_successor_roster = yes
			if = {
				limit = { has_event_target = event021_successor_roster_adoption_complete }
				event021_external_post_adoption = yes
			}
			else_if = {
				limit = { has_country_flag = random_civil_war_successor_roster_pending }
				event021_external_clear_queue = yes
			}
		}
	}
}
