# ============================================================================
# CHAOS REDUX - BIOWARFARE ON ACTIONS
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			every_country = {
				initialize_smallpox_vaccination_protection = yes
			}
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = {
					has_country_flag = bioweapon_target
				}
				check_all_states_for_contamination_cleanup = yes
			}
			# Process weekly vaccination progress for smallpox
			if = {
				limit = {
					has_idea = smallpox_vaccination_program_idea
				}
				progress_smallpox_vaccination = yes
			}
		}
	}
}
