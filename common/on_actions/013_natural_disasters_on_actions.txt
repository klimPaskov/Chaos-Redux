# ============================================================================
# CHAOS REDUX - EVENT 013 NATURAL DISASTERS ON ACTIONS
# ============================================================================
# The capitulated country is ROOT in on_capitulation. Event 013 only records a
# live Maximum Disaster Barrage attempt; no periodic world scan is used.
# ============================================================================

on_actions = {
	on_capitulation = {
		effect = {
			natural_disaster_achievement_on_capitulation = yes
		}
	}
	on_state_control_changed = {
		effect = {
			ROOT = { save_event_target_as = natural_disaster_new_responsible_country }
			FROM = { save_event_target_as = natural_disaster_former_responsible_country }
			FROM.FROM = {
				if = {
					limit = {
						OR = {
							has_state_flag = natural_disaster_impact_scheduled
							natural_disaster_state_has_open_card = yes
						}
					}
					save_event_target_as = natural_disaster_transfer_state
					natural_disaster_handle_state_control_change = yes
				}
			}
		}
	}
}
