# ============================================================================
# CHAOS REDUX - EVENT 028 ASTEROID INCOMING ON ACTIONS
# ============================================================================
# Event 028 reacts only to control changes on its registered crater and
# fragment states. No world-state scan is introduced here.
#
# on_state_control_changed uses ROOT as the new controller, FROM as the former
# controller, and FROM.FROM as the changed state.
# ============================================================================

on_actions = {
	on_annex = {
		effect = {
			if = {
				limit = { FROM = { has_country_flag = asteroid_incoming_original_target_country } }
				set_global_flag = asteroid_incoming_original_target_annexed
			}
			FROM = {
				if = {
					limit = {
						OR = {
							has_variable = asteroid_incoming_recovery_target_state
							has_country_flag = asteroid_incoming_recovery_project_pending
							asteroid_incoming_country_has_active_recovery_mission = yes
						}
					}
					asteroid_incoming_cancel_recovery_runtime = yes
				}
				# A later release starts without the lasting regional legacy.
				clr_country_flag = asteroid_incoming_regional_economy_complete
				if = {
					limit = { has_global_flag = asteroid_incoming_minerals_active }
					clear_variable = asteroid_incoming_collector_hold_start_day
					clear_variable = asteroid_incoming_main_capture_hold_start_day
					set_variable = { asteroid_incoming_collector_hold_days = constant:asteroid_incoming_runtime.zero }
					set_variable = { asteroid_incoming_main_capture_hold_days = constant:asteroid_incoming_runtime.zero }
					clr_country_flag = asteroid_incoming_main_captured_in_war
				}
			}
			if = {
				limit = { has_global_flag = asteroid_incoming_minerals_active }
				asteroid_incoming_refresh_mineral_control = yes
				# FROM is the annexed country. Clear it even if exists still reads
				# yes during the callback; a later release must not revive bonuses.
				FROM = { asteroid_incoming_clear_dead_mineral_recipient = yes }
			}
		}
	}

	# Country transitions can transfer registered sites without changing the
	# stored geographic targets. Bonuses are derived only from current control.
	on_release_as_free = {
		effect = {
			if = { limit = { has_global_flag = asteroid_incoming_minerals_active } asteroid_incoming_refresh_mineral_control = yes }
		}
	}

	on_release_as_puppet = {
		effect = {
			if = { limit = { has_global_flag = asteroid_incoming_minerals_active } asteroid_incoming_refresh_mineral_control = yes }
		}
	}

	on_liberate = {
		effect = {
			if = { limit = { has_global_flag = asteroid_incoming_minerals_active } asteroid_incoming_refresh_mineral_control = yes }
		}
	}

	on_civil_war_end = {
		effect = {
			if = { limit = { has_global_flag = asteroid_incoming_minerals_active } asteroid_incoming_refresh_mineral_control = yes }
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM.FROM = {
						OR = {
							has_state_flag = asteroid_incoming_main_crater
							has_state_flag = asteroid_incoming_fragment_site
						}
					}
				}
				FROM.FROM = { asteroid_incoming_on_state_control_changed = yes }
			}
		}
	}
}
