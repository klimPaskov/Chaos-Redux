# ============================================================================
# CHAOS REDUX - ZOMBIE CURE PAYLOAD CXT REGISTRATION HOOKS
# ============================================================================
# Startup registers through one existing country. The CXT-only daily fallback
# marks a newly discovered carrier pending without granting equipment.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				zombie_cure_payload_register_cxt_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				zombie_cure_payload_register_cxt_content = yes
				if = {
					limit = {
						check_variable = {
							var = chaosx_test_country_extension_registration_added
							value = 0
							compare = greater_than
						}
					}
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}
}
