# ============================================================================
# CHAOS REDUX - EVENT 040 LAWRENCE OF ARABIA ON ACTIONS
# ============================================================================
# One narrow transaction-boundary hook. When a country is annexed, FROM is the
# annexed country. If it was the core of the Arabian federation the federation
# state is cleared so a new congress can be called. The check is one flag test
# and no country or world iteration runs.
# ============================================================================

on_actions = {
	on_annex = {
		effect = {
			FROM = { lawrence_fed_core_lost = yes }
		}
	}
}
