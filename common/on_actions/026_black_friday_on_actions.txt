# ============================================================================
# CHAOS REDUX - EVENT 026 BLACK FRIDAY ON_ACTIONS
# ============================================================================
# The unit-leader creation hook extends an already-active sale to a commander
# recruited after activation. Activation and expiry still use bounded country
# scopes; this hook is event-driven and is not a recurring world scan.
# ============================================================================

on_actions = {
	on_unit_leader_created = {
		effect = {
			character = {
				black_friday_refresh_created_unit_leader = yes
			}
		}
	}
}
