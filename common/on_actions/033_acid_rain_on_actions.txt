# ============================================================================
# CHAOS REDUX - ACID RAIN ON ACTIONS
# ============================================================================

on_actions = {
	# Acid rain
	on_daily = {
		effect = {
			if = { limit = { any_owned_state = { has_dynamic_modifier = { modifier = acid_rain_state } } }
				every_owned_state = {
					limit = { has_dynamic_modifier = { modifier = acid_rain_state } }
					damage_building = {
						type = infrastructure
						damage = 0.15
					}damage_building = {
						type = industrial_complex
						damage = 0.15
					}damage_building = {
						type = arms_factory
						damage = 0.15
					}
					damage_units = {
						state = THIS
						limit = { always = no }
						org_damage = 0.15
						str_damage = 0.15
						ratio = yes
						army = yes
						navy = no
					}
				}
			}
		}
	}
}
