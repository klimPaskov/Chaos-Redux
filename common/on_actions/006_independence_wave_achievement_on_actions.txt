# ============================================================================
# CHAOS REDUX - EVENT 006 ACHIEVEMENT ON ACTIONS
# ============================================================================
# Narrow, event-driven history hooks for war, peace, annexation, subject status,
# and original-capital control. No daily, weekly, monthly, or world iteration is
# used.
# ============================================================================

on_actions = {
	on_war_relation_added = {
		effect = {
			save_event_target_as = independence_wave_achievement_war_attacker
			FROM = { save_event_target_as = independence_wave_achievement_war_defender }
			independence_wave_achievement_record_war = yes
			# ROOT is the attacker for this on-action. A live Event 006 member
			# starting an external war under the defensive charter is the factual
			# unauthorized-war ground. Civil-war sides are excluded so DM-61 does
			# not self-report the internal revolt it deliberately starts.
			if = {
				limit = {
					is_independence_wave_active_country = yes
					has_country_flag = independence_wave_league_member
					has_global_flag = independence_wave_charter_anti_puppetry
					has_global_flag = independence_wave_charter_mutual_defense
					check_variable = { global.independence_wave_league_route = constant:independence_wave_league_route.defensive_congress }
					has_civil_war = no
				}
				if = {
					limit = { has_country_flag = independence_wave_charter_war_authorized_@FROM }
					clr_country_flag = independence_wave_charter_war_authorized_@FROM
					set_variable = { independence_wave_last_charter_authorized_war_target = FROM }
					set_variable = { independence_wave_last_charter_authorized_war_date = global.date }
					clear_variable = independence_wave_last_charter_war_mandate_target
					clear_variable = independence_wave_last_charter_war_mandate_date
				}
				else = {
					set_temp_variable = { independence_wave_expulsion_ground_input = constant:independence_wave_expulsion_ground.unauthorized_war }
					set_temp_variable = { independence_wave_expulsion_has_target_input = constant:independence_wave_plan.loop_increment }
					set_temp_variable = { independence_wave_expulsion_evidence_target_input = event_target:independence_wave_achievement_war_defender }
					independence_wave_record_league_expulsion_ground = yes
				}
			}
		}
	}

	on_peace = {
		effect = {
			independence_wave_achievement_resolve_country_peace = yes
		}
	}

	on_peaceconference_ended = {
		effect = {
			independence_wave_achievement_resolve_country_peace = yes
			FROM = { independence_wave_achievement_resolve_country_peace = yes }
		}
	}

	on_state_control_changed = {
		effect = {
			FROM.FROM = { save_event_target_as = independence_wave_achievement_changed_state }
			FROM = { independence_wave_achievement_record_anchor_loss = yes }
			independence_wave_achievement_record_anchor_recovery = yes
		}
	}

	on_puppet = {
		effect = {
			independence_wave_achievement_refresh_host_remnant_peace = yes
			if = {
				limit = {
					has_country_flag = independence_wave_active_origin
					has_independence_wave_living_former_host = yes
					var:independence_wave_former_host = { tag = FROM }
				}
				independence_wave_achievement_disqualify_former_host_remnant = yes
			}
			if = {
				limit = { has_country_flag = independence_wave_active_origin }
				set_country_flag = independence_wave_achievement_subject_status_ever
				independence_wave_iw058_record_client_capture = yes
				independence_wave_achievement_refresh_scenario_survivor = yes
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			independence_wave_achievement_refresh_host_remnant_peace = yes
			if = {
				limit = {
					has_country_flag = independence_wave_active_origin
					has_independence_wave_living_former_host = yes
					var:independence_wave_former_host = { tag = FROM }
				}
				independence_wave_achievement_disqualify_former_host_remnant = yes
			}
			if = {
				limit = { has_country_flag = independence_wave_active_origin }
				set_country_flag = independence_wave_achievement_subject_status_ever
				independence_wave_iw058_record_client_capture = yes
				independence_wave_achievement_refresh_scenario_survivor = yes
			}
		}
	}

	on_subject_autonomy_level_change = {
		effect = {
			independence_wave_achievement_refresh_host_remnant_peace = yes
			if = {
				limit = {
					has_country_flag = independence_wave_active_origin
					is_subject = yes
					has_independence_wave_living_former_host = yes
					var:independence_wave_former_host = { tag = FROM }
				}
				independence_wave_achievement_disqualify_former_host_remnant = yes
			}
			if = {
				limit = {
					has_country_flag = independence_wave_active_origin
					is_subject = yes
				}
				set_country_flag = independence_wave_achievement_subject_status_ever
				independence_wave_iw058_record_client_capture = yes
			}
			independence_wave_achievement_refresh_scenario_survivor = yes
		}
	}

	on_subject_free = {
		effect = {
			independence_wave_achievement_refresh_host_remnant_peace = yes
			independence_wave_achievement_refresh_scenario_survivor = yes
		}
	}

	on_release_as_free = {
		effect = {
			independence_wave_achievement_refresh_host_remnant_peace = yes
			independence_wave_achievement_refresh_scenario_survivor = yes
		}
	}

	on_annex = {
		effect = {
			save_event_target_as = independence_wave_achievement_annexer
			# The annexation callback is the exact transaction witness. It runs
			# before the victim is removed, so both league-member flags are still
			# available. The victim pointer is intentionally not persisted because
			# the annexed country ceases to exist at the end of this action.
			if = {
				limit = {
					is_independence_wave_active_country = yes
					has_country_flag = independence_wave_league_member
					has_global_flag = independence_wave_charter_anti_puppetry
					FROM = {
						is_independence_wave_active_country = yes
						has_country_flag = independence_wave_league_member
					}
				}
				set_temp_variable = { independence_wave_expulsion_ground_input = constant:independence_wave_expulsion_ground.annexed_member }
				set_temp_variable = { independence_wave_expulsion_has_target_input = constant:independence_wave_value.minimum }
				independence_wave_record_league_expulsion_ground = yes
			}
			FROM = {
				independence_wave_achievement_remove_scenario_survivor = yes
				if = {
					limit = {
						has_country_flag = independence_wave_active_origin
						has_independence_wave_living_former_host = yes
						var:independence_wave_former_host = { tag = event_target:independence_wave_achievement_annexer }
					}
					independence_wave_achievement_disqualify_former_host_remnant = yes
				}
			}
		}
	}
}
