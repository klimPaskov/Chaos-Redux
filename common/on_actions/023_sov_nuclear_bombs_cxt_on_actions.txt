# ============================================================================
# CHAOS REDUX - EVENT 023 CXT REGISTRATION HOOKS
# ============================================================================
# Startup performs one bounded existing-country registration. The tag-scoped
# daily fallback repairs initialized CXT saves without adding a world loop.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				sov_nuclear_bombs_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				sov_nuclear_bombs_register_cxt_test_content = yes
				if = {
					limit = { check_variable = { var = chaosx_test_country_extension_registration_added value = constant:sov_nuclear_bombs_ledger.zero compare = greater_than } }
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}
}
