# ============================================================================
# CHAOS REDUX - GENOCIDE CRISIS ON ACTIONS
# ============================================================================
# Hooks camp discovery to state control changes and tribunal checks to
# capitulation. Monthly ticking is called from the existing host-only Chaos Meter
# monthly pulse to avoid adding another global country iteration.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				genocide_initialize_system_if_needed = yes
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = { fallout_world_rewrite_callbacks_are_allowed = yes }
				if = {
					limit = { exists = yes }
					camp_rework_initialize_country_variables = yes
				}
				genocide_on_state_control_changed = yes
			}
		}
	}

	on_release_as_free = {
		effect = {
			camp_rework_initialize_country_variables = yes
		}
	}

	on_release_as_puppet = {
		effect = {
			camp_rework_initialize_country_variables = yes
		}
	}

	on_subject_free = {
		effect = {
			if = {
				limit = { fallout_world_rewrite_callbacks_are_allowed = yes }
				remove_ideas = raj_colonial_labor_burden
				remove_ideas = fr_north_africa_labor_burden
				remove_ideas = congo_concession_labor_burden
				FROM = {
					if = {
						limit = {
							exists = yes
							OR = {
								has_active_camp_network = yes
								check_variable = { camp_evidence_states^num > 0 }
							}
						}
						set_country_flag = camp_rework_crisis_exposed
						camp_rework_freeze_expansion_for_reform = yes
						if = {
							limit = { original_tag = ENG }
							add_to_variable = { colonial_legitimacy_damage = constant:camp_rework_colonial.accountability_major }
						}
						else_if = {
							limit = { OR = { original_tag = FRA tag = VIC } }
							add_to_variable = { refugee_pressure = constant:camp_rework_colonial.accountability_major }
						}
						else_if = {
							limit = { original_tag = BEL }
							add_to_variable = { postwar_accountability_pressure = constant:camp_rework_colonial.accountability_major }
						}
						else = {
							add_to_variable = { camp_policy_evidence = constant:camp_rework_country_value.evidence_major }
						}
						camp_rework_expose_first_colonial_evidence = yes
						camp_rework_update_colonial_country_monthly_bridge = yes
					}
				}
			}
		}
	}

	on_puppet = {
		effect = {
			camp_rework_initialize_country_variables = yes
		}
	}

	on_annex = {
		effect = {
			FROM = {
				camp_rework_on_country_annexed = yes
			}
		}
	}

	on_government_change = {
		effect = {
			if = {
				limit = {
					fallout_world_rewrite_callbacks_are_allowed = yes
					OR = {
						has_active_camp_network = yes
						check_variable = { camp_evidence_states^num > 0 }
						has_visible_camp_reform_work = yes
					}
				}
				camp_rework_freeze_expansion_for_reform = yes
			}
			if = {
				limit = {
					fallout_world_rewrite_callbacks_are_allowed = yes
					has_country_flag = camp_repression_ledger_open
				}
				camp_rework_close_ledger = yes
			}
			if = {
				limit = {
					fallout_world_rewrite_callbacks_are_allowed = yes
					OR = {
						has_active_camp_network = yes
						has_visible_camp_reform_work = yes
					}
				}
				camp_rework_rebuild_display_values = yes
			}
		}
	}

	on_capitulation = {
		effect = {
			camp_rework_apply_ai_cap = yes
			genocide_try_trigger_tribunal_after_defeat = yes
		}
	}
}
