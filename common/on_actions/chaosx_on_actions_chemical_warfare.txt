# Chemical warfare on_actions.
# Leader-scoped callbacks maintain cylinder previews, wind forecasts, active
# dispersal and first-use surprise alongside equipment-backed mask snapshots.
# Passive battalion releases are not part of these callbacks.
on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				cbrn_register_event19_provider = yes
			}

			every_country = {
				chem_sync_chemical_tactic_preferred_weight_ideas = yes
				cbrn_migrate_legacy_chaos_warfare = yes

				if = {
					limit = {
						has_tech = basic_gas_masks
						NOT = { has_tactic = tactic_gas_mask_defense }
					}
					unlock_tactic = tactic_gas_mask_defense
				}

				if = {
					limit = {
						NOT = { has_tactic = tactic_chemical_shelling }
						OR = {
							has_tech = chlorine_gas
							has_tech = phosgene
							has_tech = mustard_gas
							has_tech = lewisite
							has_tech = sarin
							has_tech = soman
						}
					}
					unlock_tactic = tactic_chemical_shelling
				}
			}
		}
	}

	on_unit_leader_created = {
		effect = {
			chem_try_grant_chemical_operations_trait_from_academy_spirit = yes
		}
	}

	on_unit_leader_level_up = {
		effect = {
			chem_try_grant_chemical_operations_trait_from_academy_spirit = yes
		}
	}

	on_army_leader_daily = {
		effect = {
			FROM = {
				if = {
					limit = {
						NOT = { has_country_flag = cbrn_mask_snapshot_refreshed_today }
						OR = {
							cbrn_country_has_program = yes
							has_country_flag = cbrn_national_respirator_reserve
							has_tech = basic_gas_masks
							has_equipment = { gas_mask_equipment > constant:cbrn_math.zero }
							check_variable = { cbrn_military_mask_crates > constant:cbrn_math.zero }
						}
					}
					cbrn_refresh_country_mask_snapshot = yes
					set_temp_variable = { cbrn_mask_snapshot_refresh_guard_days = constant:cbrn_scheduler.mask_snapshot_refresh_guard_days }
					set_country_flag = {
						flag = cbrn_mask_snapshot_refreshed_today
						days = var:cbrn_mask_snapshot_refresh_guard_days
					}
				}
			}

			if = {
				limit = {
					OR = {
						has_trait = chemical_cylinder_active_chlorine
						has_trait = chemical_cylinder_active_phosgene
						has_trait = chemical_cylinder_active_mustard
						has_trait = chemical_cylinder_active_lewisite
					}
				}
				FROM = { save_event_target_as = chem_state_dispersal_owner_country }
				chem_apply_state_dispersal = yes
			}

			if = {
				limit = {
					FROM = {
						OR = {
							has_tech = chlorine_gas
							has_tech = phosgene
							has_tech = mustard_gas
							has_tech = lewisite
						}
					}
				}
				chem_update_cylinder_previews = yes
			}
		}
	}

	on_army_leader_won_combat = {
		effect = {
			if = {
				limit = {
					check_variable = { num_units_offensive_combats > 0 }
					OR = {
						has_trait = chemical_cylinder_active_chlorine
						has_trait = chemical_cylinder_active_phosgene
						has_trait = chemical_cylinder_active_mustard
						has_trait = chemical_cylinder_active_lewisite
					}
				}
				FROM = { chem_grant_first_use_surprise = yes }
			}
		}
	}

	on_army_leader_lost_combat = {
		effect = {
			if = {
				limit = {
					check_variable = { num_units_offensive_combats > 0 }
					OR = {
						has_trait = chemical_cylinder_active_chlorine
						has_trait = chemical_cylinder_active_phosgene
						has_trait = chemical_cylinder_active_mustard
						has_trait = chemical_cylinder_active_lewisite
					}
				}
				FROM = { chem_grant_first_use_surprise = yes }
			}
		}
	}
}
