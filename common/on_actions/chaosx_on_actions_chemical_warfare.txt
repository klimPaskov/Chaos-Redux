# Chemical warfare on_actions:
# - Relevant-country field-protection snapshots plus leader-daily tooltip
#   updates for the retained legacy cylinder UI.
# - Chaos Assault Battalions are protected formations and never release payload passively.
# - Exact-state chemical delivery remains adapter driven; this file does not
#   estimate a combat target, run periodic contamination, or register chemical
#   use from combat outcomes.
# - The old cylinder, Livens, and tank-shell state scans are intentionally not
#   called here. A future route must provide the selected state, payload debit,
#   protection proof, and verified conditions to the shared exposure pipeline.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				chem_sync_chemical_tactic_preferred_weight_ideas = yes
				cbrn_migrate_legacy_chaos_warfare = yes

				if = {
					limit = {
						has_tech = basic_gas_masks
						NOT = { has_tactic = tactic_gas_mask_defense }
					}
					unlock_tactic = tactic_gas_mask_defense
				}

				if = {
					limit = {
						NOT = { has_tactic = tactic_chemical_shelling }
						OR = {
							has_tech = chlorine_gas
							has_tech = phosgene
							has_tech = mustard_gas
							has_tech = lewisite
							has_tech = sarin
							has_tech = soman
						}
					}
					unlock_tactic = tactic_chemical_shelling
				}
			}
		}
	}

	on_army_leader_daily = {
		effect = {
			FROM = {
				if = {
					limit = {
						NOT = { has_country_flag = cbrn_mask_snapshot_refreshed_today }
						OR = {
							cbrn_country_has_program = yes
							has_country_flag = cbrn_national_respirator_reserve
							has_tech = basic_gas_masks
							has_equipment = { gas_mask_equipment > constant:cbrn_math.zero }
							check_variable = { cbrn_military_mask_crates > constant:cbrn_math.zero }
						}
					}
					cbrn_refresh_country_mask_snapshot = yes
					set_temp_variable = { cbrn_mask_snapshot_refresh_guard_days = constant:cbrn_scheduler.mask_snapshot_refresh_guard_days }
					set_country_flag = {
						flag = cbrn_mask_snapshot_refreshed_today
						days = var:cbrn_mask_snapshot_refresh_guard_days
					}
				}
			}

			if = {
				limit = {
					FROM = {
						OR = {
							has_tech = chlorine_gas
							has_tech = phosgene
							has_tech = mustard_gas
							has_tech = lewisite
						}
					}
				}
				chem_update_cylinder_previews = yes
			}

			# Combat hooks expose leader/combatant values but no verified selected
			# target state. No legacy support release is inferred from them.
		}
	}

}
