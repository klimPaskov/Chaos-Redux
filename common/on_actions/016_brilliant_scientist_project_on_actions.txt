# ============================================================================
# CHAOS REDUX - EVENT 016 PROJECT AND SOVEREIGNTY BRIDGES
# ============================================================================
# ROOT is the completing country and FROM is the completed project. This exact
# completion hook avoids any daily, weekly, monthly, or whole-world iteration.
# The state-control hook retries a deferred, one-time KRG opening force package
# after a fully occupied takeover first regains a valid controlled state.
# ============================================================================

on_actions = {
	on_project_completion = {
		effect = {
			ROOT = {
				if = {
					limit = { brilliant_scientist_is_current_host = yes }
					brilliant_scientist_sync_native_project_prototypes = yes
					brilliant_scientist_capture_biological_agent_history = yes
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			ROOT = {
				if = {
					limit = {
						has_country_flag = brilliant_scientist_kruger_state_active
						NOT = { has_country_flag = brilliant_scientist_project_force_package_from_history_applied }
						any_owned_state = { is_controlled_by = ROOT }
					}
					save_event_target_as = brilliant_scientist_kruger_state_carrier
					brilliant_scientist_apply_project_force_package_from_history = yes
				}
				if = {
					limit = { brilliant_scientist_is_kruger_sovereign_country = yes }
					brilliant_scientist_refresh_world_threat_source = yes
				}
			}
			FROM = {
				if = {
					limit = { brilliant_scientist_is_kruger_sovereign_country = yes }
					brilliant_scientist_refresh_world_threat_source = yes
				}
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					brilliant_scientist_is_kruger_sovereign_country = yes
					FROM = { exists = yes }
				}
				FROM = { save_event_target_as = brilliant_scientist_world_threat_opponent_input }
				brilliant_scientist_register_world_threat_opponent = yes
			}
			if = {
				limit = {
					FROM = { brilliant_scientist_is_kruger_sovereign_country = yes }
					exists = yes
				}
				save_event_target_as = brilliant_scientist_world_threat_opponent_input
				FROM = { brilliant_scientist_register_world_threat_opponent = yes }
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = { brilliant_scientist_is_kruger_sovereign_country = yes }
				FROM = { save_event_target_as = brilliant_scientist_defeat_victor_input }
				brilliant_scientist_evaluate_defeat_after_global_threat = yes
			}
		}
	}

	on_annex = {
		effect = {
			if = {
				limit = { FROM = { brilliant_scientist_is_kruger_sovereign_country = yes } }
				save_event_target_as = brilliant_scientist_defeat_victor_input
				FROM = { brilliant_scientist_evaluate_defeat_after_global_threat = yes }
			}
		}
	}
}
