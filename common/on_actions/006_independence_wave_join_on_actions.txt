# ============================================================================
# CHAOS REDUX - EVENT 006 JOIN-WAVE SCOPED CALLBACKS
# ============================================================================
# ROOT is the new controller, FROM is the old controller, and FROM.FROM is the
# changed state on on_state_control_changed. Every callback touches only those
# countries and never performs an all-country daily/weekly/monthly scan.
# ============================================================================

on_actions = {
	on_war_relation_added = {
		effect = {
			ROOT = { independence_wave_join_observe_war_entry = yes }
			FROM = { independence_wave_join_observe_war_entry = yes }
		}
	}

	on_state_control_changed = {
		effect = {
			# Controller changes do not prove that ownership changed. Re-evaluate
			# both affected country scopes without manufacturing an owned-state loss.
			# War entry remains the exact pre-loss baseline producer.
			FROM = { independence_wave_join_offer_if_eligible = yes }
			FROM.FROM = { owner = { independence_wave_join_offer_if_eligible = yes } }
		}
	}

	on_peaceconference_ended = {
		effect = {
			ROOT = { independence_wave_join_offer_if_eligible = yes }
			FROM = { independence_wave_join_offer_if_eligible = yes }
		}
	}

	on_capitulation_immediate = {
		effect = {
			ROOT = { independence_wave_join_offer_if_eligible = yes }
		}
	}

	on_release_as_free = {
		effect = {
			ROOT = { independence_wave_join_offer_if_eligible = yes }
			FROM = { independence_wave_join_offer_if_eligible = yes }
		}
	}

	on_release_as_puppet = {
		effect = {
			ROOT = { independence_wave_join_offer_if_eligible = yes }
			FROM = { independence_wave_join_offer_if_eligible = yes }
		}
	}
}
