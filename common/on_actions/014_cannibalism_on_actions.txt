# ============================================================================
# CHAOS REDUX - CANNIBALISM ON ACTIONS
# ============================================================================

on_actions = {
	on_state_control_changed = {
		effect = {
				if = {
					limit = {
						ROOT = { is_cannibal_commune_country = yes }
						FROM = {
							exists = yes
							NOT = { is_cannibal_commune_country = yes }
						}
						FROM.FROM = { is_capital = yes }
					}
					FROM = { set_country_flag = cannibalism_lost_capital_to_cannibals }
				}
		}
	}
}
