on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					has_global_flag = secret_alliance_active
					has_country_flag = secret_alliance_target
					NOT = { has_country_flag = secret_alliance_war_revealed }
					FROM = { has_country_flag = secret_alliance_member }
				}
				FROM = { save_event_target_as = secret_alliance_war_reveal_member }
				set_country_flag = secret_alliance_target_declared_reveal_war
				secret_alliance_reveal_pact_by_war = yes
			}
			if = {
				limit = {
					has_global_flag = secret_alliance_active
					has_country_flag = secret_alliance_member
					FROM = {
						has_country_flag = secret_alliance_target
						NOT = { has_country_flag = secret_alliance_war_revealed }
					}
				}
				save_event_target_as = secret_alliance_war_reveal_member
				FROM = { secret_alliance_reveal_pact_by_war = yes }
			}
		}
	}
	on_peaceconference_ended = {
		effect = {
			if = {
				limit = {
					has_country_flag = secret_alliance_target
					has_country_flag = secret_alliance_war_revealed
					FROM = { has_country_flag = secret_alliance_member }
				}
				secret_alliance_mark_reveal_war_victory = yes
			}
		}
	}
}
