# ============================================================================
# CHAOS REDUX - EVENT 011 SECRET ALLIANCE ON ACTIONS
# ============================================================================
# Narrow war, capitulation, annexation, and peace hooks. No recurring global
# daily, weekly, or monthly country iteration is used.
# ============================================================================

on_actions = {
	on_war_relation_added = {
		effect = { secret_alliance_handle_war_relation_added = yes }
	}

	on_capitulation = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_uncapitulation = {
		effect = {
			if = {
				limit = { fallout_world_rewrite_callbacks_are_allowed = yes }
				secret_alliance_handle_country_state_change = yes
			}
		}
	}

	on_government_change = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fallout_transition_active } }
				secret_alliance_handle_country_state_change = yes
			}
		}
	}

	on_annex = {
		effect = { secret_alliance_handle_annex = yes }
	}

	on_join_faction = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_leave_faction = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fallout_transition_active } }
				secret_alliance_handle_country_state_change = yes
			}
		}
	}

	on_assume_faction_leadership = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_puppet = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_liberate = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_release_as_free = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_release_as_puppet = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_subject_annexed = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_subject_free = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fallout_transition_active } }
				secret_alliance_handle_country_state_change = yes
			}
		}
	}

	on_subject_autonomy_level_change = {
		effect = { secret_alliance_handle_country_state_change = yes }
	}

	on_civil_war_end = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fallout_transition_active } }
				secret_alliance_handle_country_state_change = yes
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fallout_transition_active } }
				secret_alliance_handle_peaceconference_ended = yes
			}
		}
	}
}
