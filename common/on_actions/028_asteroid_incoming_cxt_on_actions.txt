# ============================================================================
# CHAOS REDUX - EVENT 028 CXT EXTENSION HOOKS
# ============================================================================
# Startup registers the inert Event 028 CXT fixture through one bounded
# existing-country scope. The tag-specific daily fallback repairs initialized
# test saves without adding a recurring world iteration.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				asteroid_incoming_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				asteroid_incoming_register_cxt_test_content = yes
				if = {
					limit = {
						check_variable = {
							var = chaosx_test_country_extension_registration_added
							value = 0
							compare = greater_than
						}
					}
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}
}
