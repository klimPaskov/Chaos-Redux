# Overview: Event 030 reacts when the custodian of the traveler is annexed. The callback runs only for
# the annexed country that holds the traveler and never scans countries on a timer.
# ROOT is the winner and FROM is the annexed country.

on_actions = {
	on_annex = {
		effect = {
			if = {
				limit = { FROM = { has_country_flag = time_traveler_custodian } }
				time_traveler_custody_after_annexation = yes
			}
		}
	}
}
