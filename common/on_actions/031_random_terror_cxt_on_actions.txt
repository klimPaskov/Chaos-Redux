# ============================================================================
# CHAOS REDUX - EVENT 031 CXT REGISTRATION HOOKS
# ============================================================================
# Startup registers through one bounded existing-country scope. The daily
# fallback repairs initialized test saves without adding a world iteration.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				random_terror_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				random_terror_register_cxt_test_content = yes
				if = {
					limit = { check_variable = { var = chaosx_test_country_extension_registration_added value = 1 compare = greater_than } }
					chaosx_test_country_sync_registered_content = yes
				}
			}
		}
	}
}
