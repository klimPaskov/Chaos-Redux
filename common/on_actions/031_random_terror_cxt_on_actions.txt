# ============================================================================
# CHAOS REDUX - EVENT 031 CXT REGISTRATION HOOKS
# ============================================================================
# Startup registers through one bounded existing-country scope. The daily
# fallback registers pending content for manual application without adding
# a world iteration or applying package grants.
# ============================================================================

on_actions = {
	on_startup = {
		effect = {
			random_country = {
				limit = { exists = yes }
				random_terror_register_cxt_test_content = yes
			}
		}
	}

	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				# Keep the registration receipt alive across both helper calls.
				set_temp_variable = { chaosx_test_country_extension_registration_added = constant:random_terror_value.zero }
				random_terror_register_cxt_test_content = yes
				if = {
					limit = { check_variable = { chaosx_test_country_extension_registration_added > constant:random_terror_value.zero } }
					set_country_flag = chaosx_test_country_registered_sync_pending
				}
			}
		}
	}
}
