# ============================================================================
# CHAOS REDUX - MENGELE CLONE EQUIPMENT PRODUCTION
# ============================================================================
# The user-authorized weekly country hook turns controlled biowarfare
# facilities into physical shared clone equipment after project completion.
# ============================================================================

on_actions = {
	on_weekly = {
		effect = {
			if = {
				limit = { has_country_flag = germany_mengele_cloning_project_completed }
				germany_mengele_produce_clone_equipment = yes
			}
		}
	}
}
