# ============================================================================
# CHAOS REDUX - FURY ON ACTIONS
# ============================================================================
# Hooks Fury capitulation settlements into the generic capitulation on action.
# ============================================================================

on_actions = {
	on_capitulation = {
		effect = {
			fury_on_capitulation = yes
			if = {
				limit = { is_fury_actor = yes }
				fury_cleanup_actor = yes
			}
		}
	}
	on_annex = {
		effect = {
			FROM = {
				if = {
					limit = { is_fury_actor = yes }
					fury_cleanup_actor = yes
				}
			}
		}
	}

}
