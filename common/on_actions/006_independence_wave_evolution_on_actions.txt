# ============================================================================
# CHAOS REDUX - EVENT 006 EVOLUTION FEEDBACK ON ACTIONS
# ============================================================================
# Narrow transaction-boundary hooks. No recurring country or world iteration is
# used. ROOT is the new subject in on_puppet, ROOT is the capitulated country in
# on_capitulation, and FROM is the annexed country in on_annex.
# ============================================================================

on_actions = {
	on_annex = {
		effect = {
			if = {
				limit = { FROM = { is_independence_wave_active_country = yes } }
				FROM = {
					set_temp_variable = { independence_wave_evolution_feedback_delta = constant:independence_wave_evolution_feedback.opening_confidence_origin_annexed }
					set_temp_variable = { independence_wave_evolution_feedback_type_input = constant:independence_wave_evolution_feedback_type.origin_annexed }
					independence_wave_record_origin_loss_feedback = yes
					set_temp_variable = { independence_wave_cleanup_reason = constant:independence_wave_origin_end_reason.annexed }
					independence_wave_end_active_origin = yes
				}
			}
		}
	}

	on_puppet = {
		effect = {
			if = {
				limit = { is_independence_wave_active_country = yes }
				set_temp_variable = { independence_wave_evolution_feedback_delta = constant:independence_wave_evolution_feedback.opening_confidence_origin_puppeted }
				set_temp_variable = { independence_wave_evolution_feedback_type_input = constant:independence_wave_evolution_feedback_type.origin_puppeted }
				independence_wave_record_origin_loss_feedback = yes
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = { is_independence_wave_active_country = yes }
				set_temp_variable = { independence_wave_evolution_feedback_delta = constant:independence_wave_evolution_feedback.opening_confidence_origin_capitulated }
				set_temp_variable = { independence_wave_evolution_feedback_type_input = constant:independence_wave_evolution_feedback_type.origin_capitulated }
				independence_wave_record_origin_loss_feedback = yes
			}
		}
	}
}
