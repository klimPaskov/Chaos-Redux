# ============================================================================
# CHAOS REDUX - TEST COUNTRY DYNAMIC HOOKS
# ============================================================================
# The CXT-specific hook avoids a global country iteration and is dormant until
# the console harness has initialized the test country. Content packages may
# add same-name blocks to use on_daily_CXT as the registration bus. A package
# that must be present before `e chaosx_test` may also run an idempotent
# registration helper from a bounded on_startup country scope such as
# random_country = { limit = { exists = yes } ... }, but must retain the daily
# block because on_startup does not run when a save is loaded.
# Daily requests share one pending receiver; additive hooks finish registration
# before that receiver consumes the bus. Console setup applies it immediately.
# ============================================================================

on_actions = {
	on_daily_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				if = {
					limit = { has_country_flag = chaosx_test_country_setup_trace_active }
					log = "CXT_SETUP_TRACE daily_begin country=[This.GetTag] date=[?global.date|.3]"
				}
				chaosx_test_country_refill_resources = yes
				chaosx_test_country_sync_registered_content = yes
				if = {
					limit = { has_country_flag = chaosx_test_country_setup_trace_active }
					log = "CXT_SETUP_TRACE daily_sync_queued date=[?global.date|.3]"
				}
			}
		}
	}
	on_weekly_CXT = {
		effect = {
			if = {
				limit = { has_country_flag = chaosx_test_country_initialized }
				chaosx_test_country_sync_dynamic_content = yes
				chaosx_test_country_fill_stockpile = yes
			}
		}
	}
}
