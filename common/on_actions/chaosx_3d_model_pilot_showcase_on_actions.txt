# Chaos Redux 3D model pilot showcase live consumer.
# on_startup covers a new game; on_daily_GER repairs a save that was already
# open when the standalone showcase was enabled. The latter is tag-specific,
# so it does not iterate over the world.

on_actions = {
	on_startup = {
		effect = {
			GER = {
				chaosx_3d_model_pilot_showcase_activate_country = yes
			}
		}
	}

	on_daily_GER = {
		effect = {
			GER = {
				chaosx_3d_model_pilot_showcase_activate_country = yes
			}
		}
	}
}
