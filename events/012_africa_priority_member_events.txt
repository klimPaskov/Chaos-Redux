# .1200 politics, .1210 League bargaining, .1220 overlap settlement, and

# shared atomic state machine.

# Internal synchronous character handoff for the nine accepted vanilla
# carriers. The seven niche Event 006 shells already own their matching
# sovereigns in country history, so this event never changes their
# ownership path. The completion flag and per-character guards make repeated
# package registration or recovery calls idempotent without touching vanilla
# country histories.
country_event = {
	id = africa_priority_member.1240
	hidden = yes
	is_triggered_only = yes
	trigger = {
		africa_priority_member_has_package = yes
		africa_priority_member_is_vanilla_sovereign_carrier = yes
		NOT = { has_country_flag = africa_priority_member_vanilla_sovereign_recruitment_complete }
	}

	immediate = {
		if = {
			limit = { africa_priority_member_package_is_sokoto = yes NOT = { has_character = africa_priority_sokoto_sovereign } }
