##############################################################################################################
# Entry event chaosx.nr38.1 is hidden and runs the opening release transaction
# (common/scripted_effects/038_malta_crusaders_setup_effects.txt). The pre-fire
# hook malta_prepare_random_event_fire freezes the state set before the Event
# History row is written, so the row names Malta as its actor.
#
#   .1   entry, opening transaction (hidden)
#   .2   Malta: The Crusade Is Underway, and the choice of the first command
#   .3   displaced owner: crusaders on our soil
#   .4   Catholic governments: the appeal from Malta
#   .5   weekly pulse of the actor (hidden)
#   .6   neighbouring governments: banners in the Levant
#   .7   world news
#   .8   an order's demand
#   .9   a rift in the council
#   .10  an unanswered demand
#   .14  the Holy Land is lost (opens the Eleventh Crusade)
#   .15  the cross returns (the recovery succeeds)
#   .16  Malta closes its gates (the island settlement)
#   .20  who keeps Jerusalem
#   .21  a charter for one order
#   .30  an ultimatum from Valletta
#   .31  the answer to the ultimatum
##############################################################################################################

country_event = {
	id = chaosx.nr38.1
	title = chaosx.nr38.1.t

		malta_commit_opening_transaction = yes

			limit = { check_variable = { var = global.malta_opening_result value = constant:malta_opening.result_success compare = equals } }
