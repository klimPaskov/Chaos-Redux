country_event = {
	id = chaosx.nr9.1
	title = chaosx.nr9.1.t

	hidden = yes
	is_triggered_only = yes

	immediate = {
		clr_country_flag = white_peace_current_invocation_has_human
		clear_white_peace_human_report_queue = yes
		if = {
			limit = {
				OR = {
					check_variable = { global.white_peace_context_ready < 1 }
					NOT = { has_event_target = white_peace_primary }
					NOT = { has_event_target = white_peace_partner }
				}
			}
			prepare_white_peace_runtime_context = yes
		}
		if = {
			limit = {
				check_variable = { global.white_peace_context_ready > 0 }
				has_event_target = white_peace_primary
				has_event_target = white_peace_partner
			}
			apply_white_peace_current_context = yes
			if = {
				limit = { check_variable = { global.white_peace_last_settled_pair_count > 0 } }
				fire_white_peace_report_event = yes
			}
			else = {
				set_variable = { global.white_peace_skip_reason = constant:white_peace_skip_reason.no_safe_pair }
				dispatch_white_peace_unavailable_report = yes
			}
		}
		else = {
			dispatch_white_peace_unavailable_report = yes
		}
	}
}
