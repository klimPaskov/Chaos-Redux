# The delayed router validates the frozen host receipt before sending a human
# response event or resolving an AI host without a weighted pool.
country_event = {
	id = chaosx.nr6.63
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_current_independence_wave_host_response_receipt = yes }
			if = {
				limit = { is_ai = yes }
				independence_wave_host_response_resolve_ai_action = yes
				independence_wave_host_response_apply_ai_action = yes
			}
			else_if = {
				limit = { check_variable = { var = independence_wave_host_response_tier value = constant:independence_wave_host_response_tier.limited compare = equals } }
				country_event = { id = chaosx.nr6.60 }
			}
			else_if = {
				limit = { check_variable = { var = independence_wave_host_response_tier value = constant:independence_wave_host_response_tier.severe compare = equals } }
				country_event = { id = chaosx.nr6.61 }
			}
			else_if = {
				limit = { check_variable = { var = independence_wave_host_response_tier value = constant:independence_wave_host_response_tier.remnant compare = equals } }
				country_event = { id = chaosx.nr6.62 }
			}
			else = { independence_wave_host_response_skip_invalid_current_host = yes }
		}
		else = { independence_wave_host_response_skip_invalid_current_host = yes }
	}
}
