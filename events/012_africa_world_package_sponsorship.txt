# ============================================================================
# CHAOS REDUX - EVENT 012 FOUR-MODE SPONSORSHIP DIPLOMACY
# ============================================================================
# The host selects one grounded sponsorship mode before the candidate answers.
# The candidate remains sovereign in every branch. These events only present
# the bounded offer, counterterms, fulfilment, default, and mode-specific duty
# responses. They never run a world scan or create a subject relationship.
# ============================================================================

country_event = {
	id = africa_world_package.20
	title = africa_world_package.20.t
	desc = africa_world_package.20.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	trigger = {
		africa_world_sponsorship_candidate_can_answer = yes
	}

	option = {
		name = africa_world_package.20.a
		custom_effect_tooltip = africa_world_package.20.a.tt
		ai_chance = {
			base = constant:africa_world_sponsorship_mode.ai_accept_base
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_accept_diplomatic
				check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.diplomatic }
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_accept_material
				check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.material }
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_accept_military
				check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.military }
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_accept_ideological
				check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.ideological }
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_accept_legitimacy_bonus
				has_variable = africa_world_package_legitimacy
				check_variable = {
					var = africa_world_package_legitimacy
					value = constant:africa_measure.medium
					compare = greater_than
				}
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_accept_grievance_penalty
				OR = {
					has_country_flag = africa_world_sponsorship_sovereignty_grievance
					has_country_flag = africa_world_sponsorship_defaulted
				}
			}
		}
		africa_world_sponsorship_accept_offer = yes
	}

	option = {
		name = africa_world_package.20.b
		custom_effect_tooltip = africa_world_package.20.b.tt
		ai_chance = {
			base = constant:africa_world_sponsorship_mode.ai_negotiate_base
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_negotiate_safeguard_bonus
				OR = {
					has_country_flag = africa_world_sponsorship_sovereignty_grievance
					has_country_flag = africa_world_package_sovereign_complete
				}
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_negotiate_material_bonus
				check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.material }
			}
		}
		africa_world_sponsorship_open_counterterms = yes
	}

	option = {
		name = africa_world_package.20.c
		custom_effect_tooltip = africa_world_package.20.c.tt
		ai_chance = {
			base = constant:africa_world_sponsorship_mode.ai_refuse_base
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_refuse_grievance_bonus
				OR = {
					has_country_flag = africa_world_sponsorship_sovereignty_grievance
					has_country_flag = africa_world_sponsorship_defaulted
					has_country_flag = africa_world_package_rival
				}
			}
			modifier = {
				add = constant:africa_world_sponsorship_mode.ai_refuse_ideological_bonus
				check_variable = { africa_world_sponsorship_mode = constant:africa_world_sponsorship_mode.ideological }
			}
		}
		africa_world_sponsorship_close_offer = yes
	}
}
