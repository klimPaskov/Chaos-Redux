country_event = {
	id = africa_world_package.101
	title = africa_world_package.101.t
	desc = africa_world_package.101.d
	picture = GFX_news_event_012_africa_scramble_response
	is_triggered_only = yes

	trigger = {
		has_country_flag = africa_world_middle_east_package
		has_country_flag = africa_crossroads_founding_crisis_pending
		OR = {
			has_country_flag = africa_crossroads_mandate_exit_board
			has_country_flag = africa_crossroads_water_pipeline_board
			has_country_flag = africa_crossroads_protected_routes_board
		}
	}

	option = {
		name = africa_world_package.101.a
		trigger = {
			OR = {
				AND = {
					has_country_flag = africa_crossroads_mandate_exit_board
					check_variable = { var = africa_crossroads_arab_compact value = constant:africa_measure.medium compare = greater_than }
					check_variable = { var = africa_crossroads_minority_guarantees value = constant:africa_measure.low compare = greater_than }
				}
				AND = {
					has_country_flag = africa_crossroads_water_pipeline_board
					check_variable = { var = africa_crossroads_water_security value = constant:africa_measure.medium compare = greater_than }
					check_variable = { var = africa_crossroads_oil_sovereignty value = constant:africa_measure.low compare = greater_than }
					check_variable = { var = africa_crossroads_persian_influence value = constant:africa_measure.low compare = greater_than }
				}
				AND = {
					has_country_flag = africa_crossroads_protected_routes_board
					check_variable = { var = africa_crossroads_holy_site_legitimacy value = constant:africa_measure.medium compare = greater_than }
					check_variable = { var = africa_crossroads_minority_guarantees value = constant:africa_measure.medium compare = greater_than }
					check_variable = { var = africa_crossroads_anatolian_power value = constant:africa_measure.low compare = greater_than }
				}
			}
		}
		custom_effect_tooltip = africa_world_package.101.a.tt
		ai_chance = {
			base = constant:africa_world_order.ai_high
			modifier = {
				factor = constant:africa_world_order.ai_modifier_quadruple
				OR = {
					check_variable = { var = africa_crossroads_arab_compact value = constant:africa_measure.high compare = greater_than }
					check_variable = { var = africa_crossroads_water_security value = constant:africa_measure.high compare = greater_than }
				}
			}
		}
		set_country_flag = africa_crossroads_success_pending
		country_event = { id = africa_world_package.102 days = 1 }
	}

	option = {
		name = africa_world_package.101.b
		custom_effect_tooltip = africa_world_package.101.b.tt
		ai_chance = {
			base = constant:africa_world_order.ai_normal
			modifier = {
				factor = constant:africa_world_order.ai_modifier_double
				check_variable = { var = africa_crossroads_holy_site_legitimacy value = constant:africa_measure.medium compare = less_than }
			}
		}
		set_country_flag = africa_crossroads_compromise_pending
		country_event = { id = africa_world_package.103 days = 1 }
	}

	option = {
		name = africa_world_package.101.c
		custom_effect_tooltip = africa_world_package.101.c.tt
		ai_chance = {
			base = constant:africa_world_order.ai_low
			modifier = {
				factor = constant:africa_world_order.ai_modifier_quadruple
				check_variable = { var = africa_crossroads_minority_guarantees value = constant:africa_measure.low compare = less_than_or_equals }
			}
		}
		set_country_flag = africa_crossroads_failure_pending
		country_event = { id = africa_world_package.104 days = 1 }
	}
}
