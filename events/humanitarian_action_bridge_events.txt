# Immediate country context bridges for humanitarian actions.
# The sparse country processor owns due dates. These events carry no delay.
add_namespace = humanitarian_actions

country_event = {
	id = humanitarian_actions.1
	hidden = yes
	is_triggered_only = yes
	immediate = {
	if = {
		limit = { NOT = { has_country_flag = humanitarian_action_dispatching } has_country_flag = humanitarian_action_request_pending }
		set_country_flag = humanitarian_action_dispatching
			set_temp_variable = { humanitarian_action_actor = ROOT }
			set_temp_variable = { humanitarian_action_context_proven = constant:humanitarian_runtime.one }
		set_temp_variable = { humanitarian_action_start_result = constant:humanitarian_runtime.zero }
		set_temp_variable = { humanitarian_action_target = humanitarian_action_request_state }
		set_temp_variable = { humanitarian_action_row_found = constant:humanitarian_runtime.zero }
		set_temp_variable = { humanitarian_action_row_index = constant:humanitarian_runtime.zero }
		set_temp_variable = { humanitarian_action_row_action = constant:humanitarian_runtime.zero }
		set_temp_variable = { humanitarian_action_row_id = constant:humanitarian_runtime.zero }
		set_temp_variable = { humanitarian_action_row_due = constant:humanitarian_runtime.zero }
		set_temp_variable = { humanitarian_action_target_id = constant:humanitarian_runtime.zero }
		famine_action_dispatch_begin = yes
		clr_country_flag = humanitarian_action_request_pending
		clr_country_flag = humanitarian_action_request_native
		clear_variable = humanitarian_action_request_action
		clear_variable = humanitarian_action_request_state
		clear_variable = humanitarian_action_callback_id
			clear_variable = humanitarian_action_callback_index
		clr_country_flag = humanitarian_action_dispatching
	}
	}
}
