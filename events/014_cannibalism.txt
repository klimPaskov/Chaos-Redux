add_namespace = chaosx.nr14

# ============================================================================
# CHAOS REDUX - EVENT 014 CANNIBALISM
# ============================================================================
# Minor fire-once outbreak. The entry event selects a country already under war
# pressure, then each spread country receives its own response window and can
# defeat or fail its local outbreak.
# ============================================================================

country_event = {
	id = chaosx.nr14.1
	title = chaosx.nr14.1.t

	hidden = yes
	is_triggered_only = yes

	immediate = {
		cannibalism_prepare_random_event_fire = yes
		if = {
			limit = { check_variable = { cannibalism_context_ready > 0 } }
			event_target:cannibalism_origin_country = {
				set_country_flag = cannibalism_event_dispatch_context
				country_event = { id = chaosx.nr14.2 days = 1 }
			}
		}
	}
}

country_event = {
	id = chaosx.nr14.2
	title = chaosx.nr14.2.t
	desc = chaosx.nr14.2.d
	picture = GFX_report_event_cannibalism

	is_triggered_only = yes

	trigger = {
		OR = {
			has_country_flag = cannibalism_event_dispatch_context
			NOT = { is_in_array = { global.fired_events = constant:cannibalism_event_log.event_id } }
		}
	}

	option = {
		name = chaosx.nr14.2.a
		ai_chance = {
			base = 45
			modifier = { factor = 2 OR = { has_government = fascism has_government = neutrality } }
		}
		cannibalism_start_discipline_posture = yes
		country_event = { id = chaosx.nr14.3 days = 3 }
	}

	option = {
		name = chaosx.nr14.2.b
		ai_chance = {
			base = 55
			modifier = { factor = 2 has_war = yes }
		}
		cannibalism_start_logistics_posture = yes
		country_event = { id = chaosx.nr14.3 days = 3 }
	}

	option = {
		name = chaosx.nr14.2.c
		ai_chance = {
			base = 15
			modifier = { factor = 2 has_stability < 0.35 }
			modifier = { factor = 0.25 has_government = democratic }
		}
		cannibalism_start_secrecy_posture = yes
		country_event = { id = chaosx.nr14.3 days = 3 }
	}
}

	picture = GFX_report_event_cannibalism

	is_triggered_only = yes

	option = {
		name = chaosx.nr14.3.a
		ai_chance = {
			base = 40
			modifier = { factor = 2 has_country_flag = cannibalism_posture_logistics }
		}
		subtract_from_variable = { cannibalism_hunger = constant:cannibalism_delta.small }
		add_to_variable = { cannibalism_containment = constant:cannibalism_delta.small }
		cannibalism_clamp_country_meters = yes
		cannibalism_try_country_resolution = yes
	}

	option = {
		name = chaosx.nr14.3.b
		ai_chance = {
			base = 35
			modifier = { factor = 2 has_country_flag = cannibalism_posture_discipline }
		}
		subtract_from_variable = { cannibalism_cult = constant:cannibalism_delta.small }
		add_to_variable = { cannibalism_public_fear = constant:cannibalism_delta.small }
		add_to_variable = { cannibalism_containment = constant:cannibalism_delta.small }
		cannibalism_clamp_country_meters = yes
		cannibalism_try_public_leak_news = yes
		cannibalism_try_country_resolution = yes
	}

	option = {
		name = chaosx.nr14.3.c
		ai_chance = {
			base = 20
			modifier = { factor = 2 has_country_flag = cannibalism_posture_secrecy }
			modifier = { factor = 0.5 has_government = democratic }
		}
		add_to_variable = { cannibalism_cult = constant:cannibalism_delta.small }
		add_to_variable = { cannibalism_hannibal_resonance = constant:cannibalism_delta.small }

		cannibalism_clamp_country_meters = yes
		cannibalism_try_public_leak_news = yes
		cannibalism_try_country_resolution = yes
	}
}
