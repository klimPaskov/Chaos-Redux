add_namespace = chaosx.nr33

# Event 033 is a single runtime. The root event is the only public launch
# surface; all later callbacks are hidden, generation-checked, and bounded by
# the registries created by acid_rain_start_runtime.
country_event = {
	id = chaosx.nr33.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		acid_rain_start_runtime = yes
		acid_rain_gui_refresh_read_only = yes
		if = {
			limit = {
				has_global_flag = acid_rain_runtime_active
				NOT = { has_global_flag = acid_rain_super_event_sent }
			}
			set_global_flag = acid_rain_super_event_sent
			set_global_flag = { flag = super_event_visible days = constant:acid_rain_event.super_event_visible_days value = constant:acid_rain_event.super_event_id }
			set_variable = { global.current_super_event_audio_id = constant:acid_rain_event.super_event_audio_id }
			every_country = {
				limit = { is_ai = no }
				play_current_super_event_audio = yes
			}
			news_event = { id = chaosx.nr33.201 days = 2 }
		}
		acid_rain_apply_formation_stage = yes
	}
}

# The only recurring callback. It runs on the stored host country, never on a
# world iterator, and the worker re-schedules itself only while the runtime is
# active.
country_event = {
	id = chaosx.nr33.20
	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = acid_rain_runtime_active
		has_country_flag = acid_rain_runtime_host
	}

	immediate = {
		acid_rain_host_worker = yes
	}
}

# Exact seven-day movement callbacks for the three stable front slots. The
# same guarded move effect is also checked by the bounded host worker after a
# reload, so neither callback loss nor duplicate delivery changes gameplay.
country_event = {
	id = chaosx.nr33.21
	hidden = yes
	is_triggered_only = yes
	trigger = {
		has_global_flag = acid_rain_runtime_active
		has_country_flag = acid_rain_runtime_host
		check_variable = { var = global.acid_rain_front_generation_entries^1 value = global.acid_rain_generation_id compare = equals }
		check_variable = { var = global.acid_rain_front_state_entries^1 value = constant:acid_rain_front_state.retiring compare = equals }
		check_variable = { var = global.acid_rain_front_next_region_entries^1 value = constant:acid_rain_runtime.zero compare = greater_than }
		check_variable = { var = global.date value = global.acid_rain_front_move_day_entries^1 compare = greater_than_or_equals }
	}
	immediate = {
		set_temp_variable = { acid_rain_front_id = constant:acid_rain_runtime.one }
		acid_rain_execute_front_move = yes
		acid_rain_gui_refresh_read_only = yes
	}
}

# Four prepared-project completion callbacks. The selected index and the
# generation are written by the decision transaction before this event is
# scheduled; stale callbacks therefore have no gameplay effect.
country_event = {
	id = chaosx.nr33.30
	hidden = yes
	is_triggered_only = yes
	trigger = {
		OR = {
			has_global_flag = acid_rain_runtime_active
			has_global_flag = acid_rain_aftermath_retained
		}
		check_variable = { var = acid_rain_project_active_entries^1 value = constant:acid_rain_runtime.one compare = equals }
		check_variable = { var = global.date value = acid_rain_project_end_day_entries^1 compare = greater_than_or_equals }
		check_variable = { var = acid_rain_project_generation_entries^1 value = global.acid_rain_generation_id compare = equals }
	}
	immediate = {
		set_variable = { acid_rain_selected_project = constant:acid_rain_project.shelter }
		acid_rain_complete_project = yes
		acid_rain_gui_refresh_read_only = yes
	}
}

# One callback serves all nine urgent/recovery actions. The country ledger
# stores the selected action and its generation, so a stale delayed event is a
# no-op even after a reload or a second Event 033 generation.
country_event = {
	id = chaosx.nr33.40
	hidden = yes
	is_triggered_only = yes
	trigger = {
		OR = {
			has_global_flag = acid_rain_runtime_active
			has_global_flag = acid_rain_aftermath_retained
		}
		has_country_flag = acid_rain_national_action_active
		check_variable = { var = global.date value = acid_rain_action_end_day compare = greater_than_or_equals }
		check_variable = { var = acid_rain_action_generation value = global.acid_rain_generation_id compare = equals }
	}
	immediate = {
		acid_rain_complete_action = yes
		acid_rain_gui_refresh_read_only = yes
	}
}
