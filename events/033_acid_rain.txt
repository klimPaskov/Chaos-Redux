# Event 033 is a single runtime. The root event is the only public launch
# surface; all later callbacks are hidden, generation-checked, and bounded by
# the registries created by acid_rain_start_runtime.
country_event = {
	id = chaosx.nr33.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		# Validate before changing provenance, resetting ledgers or presenting a
		# formation. An active duplicate only consumes markers and refreshes UI.
		if = {
			limit = { acid_rain_launch_preflight_is_valid = yes }
			# Latch provenance before formation initializes any participant ledger.
			# An active duplicate consumes markers without rewriting this generation.
			if = {
				limit = { NOT = { has_global_flag = acid_rain_runtime_active } }
				if = {
					limit = {
						OR = {
							has_country_flag = acid_rain_launch_forced
							NOT = { has_country_flag = acid_rain_launch_authorized }
							tag = CXT
						}
					}
					set_global_flag = acid_rain_forced_generation
				}
				else = { clr_global_flag = acid_rain_forced_generation }
			}
			clr_country_flag = acid_rain_launch_authorized
			clr_country_flag = acid_rain_launch_forced
			acid_rain_start_runtime = yes
			acid_rain_gui_refresh_read_only = yes
			if = {
				limit = {
					has_global_flag = acid_rain_runtime_active
					NOT = { has_global_flag = acid_rain_super_event_sent }
				}
				set_global_flag = acid_rain_super_event_sent
				set_temp_variable = { acid_rain_super_event_visible_days = constant:acid_rain_event.super_event_visible_days }
				set_global_flag = { flag = super_event_visible days = acid_rain_super_event_visible_days value = constant:acid_rain_event.super_event_id }
				set_variable = { global.current_super_event_audio_id = constant:acid_rain_event.super_event_audio_id }
				every_country = {
					limit = { is_ai = no }
					play_current_super_event_audio = yes
				}
				news_event = { id = chaosx.nr33.201 days = 2 }
			}
			acid_rain_apply_formation_stage = yes
			# Players whose territory is already warned or exposed see the world map
			# once, when the formation commits; a duplicate formation opens nothing.
			if = {
				limit = {
					has_global_flag = acid_rain_runtime_active
					NOT = { has_global_flag = acid_rain_formation_map_opened }
				}
				set_global_flag = acid_rain_formation_map_opened
				acid_rain_gui_open_at_formation = yes
			}
		}
		else = { acid_rain_gui_refresh_read_only = yes }
		clr_country_flag = acid_rain_launch_authorized
		clr_country_flag = acid_rain_launch_forced
	}
}

# reload, so neither callback loss nor duplicate delivery changes gameplay.

# Evacuations and recovery work complete through a state callback on the
# state they serve. The state stores its paying country, generation and due
# day, so a stale callback or one after a change of controller does nothing.
state_event = {
	id = chaosx.nr33.47
	hidden = yes
	is_triggered_only = yes
	trigger = {
		OR = {
			has_global_flag = acid_rain_runtime_active
			has_global_flag = acid_rain_aftermath_retained
		}
		has_state_flag = acid_rain_action_lock
		check_variable = { var = acid_rain_action_owner_generation value = global.acid_rain_generation_id compare = equals }
		check_variable = { var = global.date value = acid_rain_action_due_day compare = greater_than_or_equals }
	}
	immediate = {
		acid_rain_complete_state_operation = yes
	}
}
