country_event = {
	id = chaosx.nr8.20
	title = chaosx.nr8.20.t
	hidden = yes
	is_triggered_only = yes
	trigger = { has_country_flag = tensions_rising_thin_wire_callback_pending }

	immediate = {
		clr_country_flag = tensions_rising_thin_wire_callback_pending
		if = {
			limit = {
				is_ai = no
				has_country_flag = tensions_rising_thin_wire_watch
				NOT = { has_country_flag = tensions_rising_thin_wire_broken }
				NOT = { has_global_flag = world_end }
				has_war = no
				is_tensions_rising_world_tension_full = yes
				has_variable = tensions_rising_thin_wire_expiry_date
			}
			if = {
				limit = {
					check_variable = {
						var = global.date
						value = tensions_rising_thin_wire_expiry_date
						compare = greater_than_or_equals
					}
				}
				set_country_flag = achievement_tensions_thin_wire_ready
				clr_country_flag = tensions_rising_thin_wire_watch
				clr_country_flag = tensions_rising_thin_wire_broken
				clear_variable = tensions_rising_thin_wire_expiry_date
			}
			else = {
				set_country_flag = tensions_rising_thin_wire_callback_pending
				set_temp_variable = { tensions_rising_thin_wire_step_days = constant:tensions_rising_achievement.thin_wire_step_days }
				country_event = { id = chaosx.nr8.20 days = tensions_rising_thin_wire_step_days }
			}
		}
		else_if = {
			limit = { has_country_flag = tensions_rising_thin_wire_watch }
			clr_country_flag = tensions_rising_thin_wire_watch
			clr_country_flag = tensions_rising_thin_wire_broken
			clear_variable = tensions_rising_thin_wire_expiry_date
		}
		else_if = {
			limit = { has_variable = tensions_rising_thin_wire_expiry_date }
			clear_variable = tensions_rising_thin_wire_expiry_date
		}
	}
}
