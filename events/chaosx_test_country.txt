# ============================================================================
# CHAOS REDUX - TEST COUNTRY SETUP RECEIVER
# ============================================================================
# Re-enters the console harness in CXT country scope after activation and player
# transfer. The saved origin targets carry through the originating effect chain.
# The ordinary setup and extension bus therefore receive ROOT = CXT.
# ============================================================================

add_namespace = chaosx_test_country

country_event = {
	id = chaosx_test_country.1
	hidden = yes
	is_triggered_only = yes

	trigger = { tag = CXT }

	immediate = {
		if = {
			limit = { tag = CXT }
			# Temporary first-activation trace; removed once the freeze is resolved.
			set_country_flag = chaosx_test_country_setup_trace_active
			log = "CXT_SETUP_TRACE receiver_begin country=[This.GetTag] date=[?global.date|.3]"
			if = {
				limit = { has_event_target = chaosx_test_origin_country }
				if = {
					limit = {
						country_exists = event_target:chaosx_test_origin_country
						event_target:chaosx_test_origin_country = {
							NOT = { tag = CXT }
							any_owned_state = { always = yes }
						}
					}
					log = "CXT_SETUP_TRACE annex_begin origin=[chaosx_test_origin_country.GetTag] date=[?global.date|.3]"
					annex_country = {
						target = event_target:chaosx_test_origin_country
						transfer_troops = no
					}
					log = "CXT_SETUP_TRACE annex_complete date=[?global.date|.3]"
				}
			}
			if = {
				limit = { has_event_target = chaosx_test_origin_capital_state }
				event_target:chaosx_test_origin_capital_state = {
					if = {
						limit = { is_owned_by = CXT }
						CXT = {
							set_capital = {
								state = event_target:chaosx_test_origin_capital_state
								remember_old_capital = no
							}
						}
					}
				}
			}
			log = "CXT_SETUP_TRACE capital_restored date=[?global.date|.3]"
			if = {
				limit = {
					any_owned_state = { is_capital = yes is_controlled_by = PREV }
				}
				if = {
					limit = { NOT = { has_country_flag = chaosx_test_country_initialized } }
					chaosx_test_country_initial_setup = yes
				}
				else = { chaosx_test_country_refresh = yes }
			}
			else = { log = "CXT_SETUP_TRACE capital_gate_failed country=[This.GetTag] date=[?global.date|.3]" }
		}
	}
}

# Coalesced tag-specific daily repair; startup and console setup stay immediate.
country_event = {
	id = chaosx_test_country.2
	hidden = yes
	is_triggered_only = yes

	trigger = { tag = CXT }

	immediate = {
		if = {
			limit = {
				tag = CXT
				has_country_flag = chaosx_test_country_registered_sync_queued
			}
			# Clear before dispatch so a later registration can queue another pass.
			clr_country_flag = chaosx_test_country_registered_sync_queued
			if = {
				limit = { has_country_flag = chaosx_test_country_registered_sync_pending }
				clr_country_flag = chaosx_test_country_registered_sync_pending
				chaosx_test_country_sync_registered_content_apply = yes
				clr_country_flag = chaosx_test_country_setup_trace_active
			}
		}
	}
}
