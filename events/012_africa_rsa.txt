	is_triggered_only = yes

	option = {
		name = chaosx.nr12.1202.a
		trigger = {
			has_event_target = africa_rsa_continental_coalition
			event_target:africa_rsa_continental_coalition = {
				exists = yes
				has_global_flag = africa_rsa_civil_war_active
			}
		}
		ai_chance = {
			base = constant:africa_rsa_ai.option_normal
			modifier = {
				factor = constant:africa_rsa_ai.modifier_double
				has_opinion = { target = FROM value > @CR_SC_AFRICA_RSA_AI_SUPPORT_OPINION_GATE }
			}
		}
		custom_effect_tooltip = africa_rsa_regional_support_accept_tt
		hidden_effect = {
			send_equipment_fraction = {
				target = event_target:africa_rsa_continental_coalition
				value = constant:africa_rsa_ratio.regional_support_transfer
			}
			save_event_target_as = africa_rsa_support_actor
			event_target:africa_rsa_continental_coalition = { africa_rsa_register_regional_support = yes }
		}
	}

	option = {
		name = chaosx.nr12.1202.b
		ai_chance = { base = constant:africa_rsa_ai.option_minor }
		custom_effect_tooltip = africa_rsa_regional_support_refuse_tt
		hidden_effect = {
			set_country_flag = africa_rsa_regional_support_refused
			save_event_target_as = africa_rsa_support_actor
			event_target:africa_rsa_continental_coalition = { africa_rsa_register_regional_response = yes }
		}
