# ============================================================================
# CHAOS REDUX - EVENT 022 CONCENTRATION CAMPS
# ============================================================================
# Event 22 is the entry and escalation layer on top of the shared Camps and
# Repression system. A firing selects one country, registers camp sites in
# about a fifth of its eligible states and seeds their cohorts from a campaign
# marker. The country then manages the network through the shared Ledger and a
# short Event 22 decision category. Exposure and Resistance Pressure are the
# shared camp_foreign_visibility and camp_resistance_pressure values.
#
#   .1   hidden entry on the selected country
#   .2   opening report with the opening policy
#   .3   hidden monthly monitor (one per live network)
#   .101 escape networks        .102 sickness in the camps
#   .103 reports leave the camps
#   .201 Evolution I killing orders   .301 Evolution II deaths   .401 Evolution III half the country
#   .510 foreign report of credible camps
#   .800 closure after destroyed records   .801 closure with open records
#   .802 hidden closure proof
#   news chaosx.news.22 lives in events/_chaosx_news.txt
# ============================================================================

# Hidden entry on the selected country. A country without a live network gets
# the opening, a country with one gets the incident its shared values call for.
country_event = {
	id = chaosx.nr22.1
	title = chaosx.nr22.1.t

	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				concentration_camps_target_basics = yes
				NOT = { has_country_flag = concentration_camps_network_active }
			}
			concentration_camps_open = yes
			camp_admin_generic_reform_refresh_quote = yes
			set_country_flag = concentration_camps_report_pending
			country_event = {
				id = chaosx.nr22.2
				days = constant:concentration_camps_timing.report_delay_days
			}
		}
		else_if = {
			limit = { concentration_camps_has_incident_ready = yes }
			concentration_camps_set_cooldown = yes
			concentration_camps_fire_incident = yes
		}
	}
}
