# Hidden recipient boundary for the existing bounded monthly administration.
# No notification or choices; rebases ROOT to the country before legacy helpers.
add_namespace = camp_administration_country

country_event = {
	id = camp_administration_country.1
	hidden = yes
	is_triggered_only = yes
	immediate = { camp_admin_integration_monthly_country_recipient = yes }
}

# One timer per registered country keeps administration active when Chaos Meter is disabled.
country_event = {
	id = camp_administration_country.2
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				has_country_flag = camp_admin_dispatch_timer_active
				camp_rework_country_is_eligible = yes
				is_in_array = {
					array = global.camp_repression_active_countries
					value = THIS
				}
			}
			camp_admin_integration_monthly_country_recipient = yes
			if = {
				limit = {
					camp_rework_country_is_eligible = yes
					is_in_array = {
						array = global.camp_repression_active_countries
						value = THIS
					}
					camp_rework_country_has_registry_work = yes
				}
				country_event = {
					id = camp_administration_country.2
					days = constant:camp_admin_dispatch.period_days
				}
			}
			else = {
				if = { limit = { is_special_chaos_country = yes } camp_rework_exclude_special_country = yes }
				remove_from_array = {
					array = global.camp_repression_active_countries
					value = THIS
				}
				clr_country_flag = camp_admin_dispatch_timer_active
			}
		}
		else = {
			if = { limit = { is_special_chaos_country = yes } camp_rework_exclude_special_country = yes }
			clr_country_flag = camp_admin_dispatch_timer_active
		}
	}
}
