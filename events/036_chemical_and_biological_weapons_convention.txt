# ============================================================================
# CHAOS REDUX - EVENT 036 CHEMICAL AND BIOLOGICAL WEAPONS CONVENTION
# ============================================================================
# Minor Fire-Once diplomatic event. The entry runs the one bounded opening
# pass; every later event is an Event 036 follow-up and never enters the
# random-event pool. Logic lives in
# common/scripted_effects/036_cbrn_convention_effects.txt.
#
# .1 entry          .2 opening response    .3 opening news
# .10-.13 agendas   .20 evolution report   .40-.44 program reports
# .50-.53 exposure, breach, suspension, expulsion
# .55 covert family choice  .60 accession  .61 withdrawal  .62 revision
# .71-.78 news       .80 coordinator pulse
# ============================================================================

	desc = chaosx.nr36.11.d
	picture = GFX_report_event_cbrn_convention
	is_triggered_only = yes
	timeout_days = 14

	trigger = {
		cbrn_convention_agenda_is_open = yes
		NOT = { has_global_flag = cbrn_convention_agenda_negotiating }
		NOT = { is_in_array = { array = global.cbrn_convention_agenda_ratifiers value = THIS } }
	}

	immediate = {
		hidden_effect = { set_temp_variable = { cbrn_convention_family_id = global.cbrn_convention_agenda_family } }
	}

	option = {
		name = chaosx.nr36.11.c
		custom_effect_tooltip = cbrn_convention_response_decline_tt
	}
	option = {
		name = chaosx.nr36.11.a
		trigger = {
			set_temp_variable = { cbrn_convention_family_id = global.cbrn_convention_agenda_family }
			NOT = { cbrn_convention_posture_forces_reservation_clause = yes }
		}
		custom_effect_tooltip = cbrn_convention_response_ratify_tt
		hidden_effect = {
			if = {
				limit = { cbrn_convention_can_respond_to_agenda = yes }
				set_temp_variable = { cbrn_convention_response_reserved = constant:cbrn_convention.zero }
				cbrn_convention_record_agenda_ratification = yes
			}
		}
	}
	option = {
		name = chaosx.nr36.11.b
		trigger = {
			set_temp_variable = { cbrn_convention_family_id = global.cbrn_convention_agenda_family }
			cbrn_convention_family_has_reservation_clause = yes
		}
		custom_effect_tooltip = cbrn_convention_response_reserve_tt
		hidden_effect = {
			if = {
				limit = { cbrn_convention_can_respond_to_agenda = yes }
				set_temp_variable = { cbrn_convention_response_reserved = constant:cbrn_convention.one }
				cbrn_convention_record_agenda_ratification = yes
			}
		}
	}
}
