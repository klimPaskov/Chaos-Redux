# Crash: freeze live boom fields, end the package, deliver to Event 35.
country_event = {
	id = chaosx.nr34.3
	title = chaosx.nr34.3.t

	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				has_country_flag = industrial_boom_active
				NOT = { has_country_flag = industrial_boom_snapshot_frozen }
			}
			clr_country_flag = industrial_boom_crash_pending
			set_variable = { industrial_boom_last_outcome = constant:industrial_boom_outcome.crash }
			if = {
				limit = { NOT = { has_variable = industrial_boom_completed_count } }
				set_variable = { industrial_boom_completed_count = constant:industrial_boom_event.zero }
			}
			industrial_boom_prepare_collapse_fields = yes
			industrial_boom_freeze_collapse_snapshot = yes
			if = {
				limit = { has_variable = industrial_boom_crash_count }
				add_to_variable = { industrial_boom_crash_count = constant:industrial_boom_event.one }
			}
			else = {
				set_variable = { industrial_boom_crash_count = constant:industrial_boom_event.one }
			}
			set_temp_variable = { industrial_boom_flag_days = constant:industrial_boom_event.recent_crash_days }
			set_country_flag = { flag = industrial_boom_recent_crash days = industrial_boom_flag_days }
			industrial_boom_break_lasting_chain = yes
			industrial_boom_end_episode = yes
		}
		industrial_boom_deliver_collapse_snapshot = yes
	}
}
