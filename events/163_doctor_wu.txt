add_namespace = chaosx.nr163

# ============================================================================
# CHAOS REDUX - EVENT 163 DOCTOR WU
# ============================================================================
# The shared dispatcher resolves one weighted ordinary-country host before
# opening the visible choice. Acceptance commits one persistent global host;
# refusal resolves the fire-once event without inventing a substitute host.
# ============================================================================

country_event = {
	id = chaosx.nr163.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				doctor_wu_is_eligible_host = yes
				NOT = { has_global_flag = doctor_wu_event_resolved }
			}
			country_event = { id = chaosx.nr163.2 }
		}
	}
}

country_event = {
	id = chaosx.nr163.2
	title = chaosx.nr163.2.t
	desc = chaosx.nr163.2.d
	picture = GFX_report_event_020_doctor_wu
	is_triggered_only = yes

	option = {
		name = chaosx.nr163.2.a
		trigger = {
			doctor_wu_is_eligible_host = yes
			NOT = { has_global_flag = doctor_wu_event_resolved }
		}
		ai_chance = {
			base = constant:doctor_wu_opening_ai.public_mandate
			modifier = { factor = constant:doctor_wu_opening_ai.medical_need_factor doctor_wu_host_controls_severe_black_plague_state = yes }
			modifier = { factor = constant:doctor_wu_opening_ai.medical_need_factor has_active_biowarfare_outbreak = yes }
			modifier = { factor = constant:doctor_wu_opening_ai.democratic_public_factor has_government = democratic }
		}
		custom_effect_tooltip = doctor_wu_public_mandate_tt
		hidden_effect = { doctor_wu_accept_public_mandate = yes }
	}

	option = {
		name = chaosx.nr163.2.b
		trigger = {
			doctor_wu_is_eligible_host = yes
			NOT = { has_global_flag = doctor_wu_event_resolved }
		}
		ai_chance = {
			base = constant:doctor_wu_opening_ai.protected_service
			modifier = { factor = constant:doctor_wu_opening_ai.medical_need_factor doctor_wu_host_controls_severe_black_plague_state = yes }
			modifier = { factor = constant:doctor_wu_opening_ai.war_protected_factor has_war = yes }
			modifier = {
				factor = constant:doctor_wu_opening_ai.authoritarian_protected_factor
				OR = {
					has_government = fascism
					has_government = communism
					has_government = neutrality
				}
			}
		}
		custom_effect_tooltip = doctor_wu_protected_service_tt
		hidden_effect = { doctor_wu_accept_protected_service = yes }
	}

	option = {
		name = chaosx.nr163.2.c
		ai_chance = {
			base = constant:doctor_wu_opening_ai.refuse
			modifier = { factor = constant:doctor_wu_opening_ai.stable_refusal_factor has_stability > constant:doctor_wu_opening_ai.stable_stability_threshold }
		}
		custom_effect_tooltip = doctor_wu_refusal_tt
		hidden_effect = { doctor_wu_refuse_host = yes }
	}
}

				OR = {

	}
}
