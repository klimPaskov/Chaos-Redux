# Africa-side protection receipt, shown only when a paid protection action
# actually changes the selected participant's record.
country_event = {
	id = chaosx.nr12.610
	title = chaosx.nr12.610.t
	desc = chaosx.nr12.610.d
	picture = GFX_report_event_012_africa_gods_protection
	is_triggered_only = yes
	trigger = {
		africa_gods_of_africa_host_is_active = yes
		has_country_flag = gods_of_africa_protection_report_pending
	}
	option = {
		name = chaosx.nr12.610.a
		custom_effect_tooltip = gods_of_africa_protection_effect_tt
		if = {
			limit = { has_guaranteed = event_target:gods_of_africa_protected_participant }
			custom_effect_tooltip = gods_of_africa_protection_service_guarantee_tt
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_military_access_to = event_target:africa_host } }
			custom_effect_tooltip = gods_of_africa_protection_service_military_access_tt
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_market_access_with = event_target:africa_host } }
			custom_effect_tooltip = gods_of_africa_protection_service_market_access_tt
		}
		if = {
			limit = {
				NOT = {
					event_target:gods_of_africa_protected_participant = {
						has_market_access_with = event_target:africa_host
					}
				}
			}
			if = {
				limit = { NOT = { has_dlc = "Arms Against Tyranny" } }
				custom_effect_tooltip = gods_of_africa_protection_service_market_missing_dlc_tt
			}
			else_if = {
				limit = { NOT = { has_country_flag = gods_of_africa_focus_sanctuary_unlocked } }
				custom_effect_tooltip = gods_of_africa_protection_service_market_capability_locked_tt
			}
			else = {
				custom_effect_tooltip = gods_of_africa_protection_service_market_unconfirmed_tt
			}
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_country_flag = gods_of_africa_protection_intelligence_shared } }
			custom_effect_tooltip = gods_of_africa_protection_service_intelligence_tt
		}
		if = {
			limit = {
				NOT = {
					event_target:gods_of_africa_protected_participant = {
						has_country_flag = gods_of_africa_protection_intelligence_shared
					}
				}
			}
			if = {
				limit = { NOT = { has_dlc = "La Resistance" } }
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_missing_dlc_tt
			}
			else_if = {
				limit = { NOT = { has_country_flag = gods_of_africa_focus_security_unlocked } }
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_capability_locked_tt
			}
			else_if = {
				limit = {
					event_target:africa_host = {
						NOT = { has_intelligence_agency = yes }
					}
				}
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_host_agency_missing_tt
			}
			else_if = {
				limit = {
					event_target:gods_of_africa_protected_participant = {
						NOT = { has_intelligence_agency = yes }
					}
				}
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_participant_agency_missing_tt
			}
			else_if = {
				limit = { event_target:africa_host = { NOT = { has_war = yes } } }
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_no_war_tt
			}
			else_if = {
				limit = {
					event_target:africa_host = {
						NOT = {
							any_enemy_country = {
								exists = yes
								NOT = { has_capitulated = yes }
								event_target:gods_of_africa_protected_participant = {
									NOT = { has_war_with = PREV }
								}
							}
						}
					}
				}
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_no_eligible_enemy_tt
			}
			else = {
				custom_effect_tooltip = gods_of_africa_protection_service_intelligence_unconfirmed_tt
			}
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_country_flag = gods_of_africa_protection_convoys_delivered } }
			custom_effect_tooltip = gods_of_africa_protection_service_convoys_tt
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_country_flag = gods_of_africa_protection_trains_delivered } }
			custom_effect_tooltip = gods_of_africa_protection_service_trains_tt
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_country_flag = gods_of_africa_protection_capital_rebuilt } }
			custom_effect_tooltip = gods_of_africa_protection_service_capital_tt
		}
		if = {
			limit = { event_target:gods_of_africa_protected_participant = { has_country_flag = gods_of_africa_protection_reception_created } }
			custom_effect_tooltip = gods_of_africa_protection_service_reception_tt
		}
		if = {
			limit = { has_country_flag = gods_of_africa_focus_relief_unlocked }
			custom_effect_tooltip = gods_of_africa_protection_service_stability_tt
		}
		if = {
			limit = { has_country_flag = gods_of_africa_focus_expedition_unlocked }
			custom_effect_tooltip = gods_of_africa_protection_service_war_support_tt
		}
		custom_effect_tooltip = gods_of_africa_protection_service_conditions_tt
		hidden_effect = {
			event_target:gods_of_africa_protected_participant = { gods_of_africa_clear_protection_report_outcomes = yes }
		}
		clr_country_flag = gods_of_africa_protection_report_pending
	}
}
