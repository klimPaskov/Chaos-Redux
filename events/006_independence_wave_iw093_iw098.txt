# ============================================================================
# CHAOS REDUX - EVENT 006 IW-093 / IW-098 FORMER-HOST NEGOTIATIONS
# ============================================================================
# The released country pays and prepares the negotiation through its package
# decision. These events give the exact surviving former host the answer and
# then return a visible result to the released country. Regular event targets
# keep each bilateral chain isolated when several countries negotiate at once.
# ============================================================================

country_event = {
	id = chaosx.nr006.9301
	title = chaosx.nr006.9301.t
	desc = chaosx.nr006.9301.d
	picture = GFX_report_event_006_asset_002_host_crisis
	is_triggered_only = yes
	trigger = { is_independence_wave_iw093_negotiation_host = yes }
	immediate = { save_event_target_as = independence_wave_iw093_answering_host }

	option = {
		name = chaosx.nr006.9301.a
		diplomatic_relation = { country = event_target:independence_wave_iw093_negotiating_country relation = non_aggression_pact active = yes }
		event_target:independence_wave_iw093_negotiating_country = {
			independence_wave_iw093_apply_recognized_host_settlement = yes
			country_event = { id = chaosx.nr006.9302 }
		}
		ai_chance = {
			base = constant:independence_wave_iw093.ai_priority_high
			modifier = {
				factor = constant:independence_wave_decision_ai.modifier_double
				has_opinion = {
					target = event_target:independence_wave_iw093_negotiating_country
					value > constant:independence_wave_decision_gate.diplomatic_acceptance_opinion
				}
			}
			modifier = { factor = constant:independence_wave_decision_ai.modifier_double has_government = democratic }
		}
	}

	option = {
		name = chaosx.nr006.9301.b
		diplomatic_relation = { country = event_target:independence_wave_iw093_negotiating_country relation = non_aggression_pact active = yes }
		diplomatic_relation = { country = event_target:independence_wave_iw093_negotiating_country relation = guarantee active = yes }
		diplomatic_relation = { country = event_target:independence_wave_iw093_negotiating_country relation = military_access active = yes }
		event_target:independence_wave_iw093_negotiating_country = {
			independence_wave_iw093_apply_host_association_settlement = yes
			country_event = { id = chaosx.nr006.9303 }
		}
		ai_chance = {
			base = constant:independence_wave_iw093.ai_priority_medium
			modifier = { factor = constant:independence_wave_decision_ai.modifier_double is_major = yes }
		}
	}

	option = {
		name = chaosx.nr006.9301.c
		send_embargo = event_target:independence_wave_iw093_negotiating_country
		if = {
			limit = { NOT = { has_war_with = event_target:independence_wave_iw093_negotiating_country } }
			create_wargoal = { type = annex_everything target = event_target:independence_wave_iw093_negotiating_country }
		}
		event_target:independence_wave_iw093_negotiating_country = {
			independence_wave_iw093_apply_host_reclamation_crisis = yes
			country_event = { id = chaosx.nr006.9304 }
		}
		ai_chance = {
			base = constant:independence_wave_iw093.ai_priority_low
			modifier = {
				factor = constant:independence_wave_decision_ai.modifier_major
				has_opinion = {
					target = event_target:independence_wave_iw093_negotiating_country
					value < constant:independence_wave_value.minimum
				}
			}
			modifier = { factor = constant:independence_wave_decision_ai.modifier_double has_government = fascism }
		}
	}
}
