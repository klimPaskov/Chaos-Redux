# ============================================================================
# CHAOS REDUX - EVENT 011 SECRET ALLIANCE
# ============================================================================
# Minor fire-once diplomatic trap against the event target. The pact selects
# three eligible non-war founders, escalates through active evolution openings,
# opens target counterplay at Evo II, becomes a public crisis at Evo III, and
# reveals fully when any live member enters war with the target.
# ============================================================================

country_event = {
	id = chaosx.nr11.1
	title = chaosx.nr11.1.t
	desc = chaosx.nr11.1.d
	picture = GFX_report_event_secret_alliance_meeting

	is_triggered_only = yes

	trigger = {
		OR = {
			has_country_flag = secret_alliance_event_dispatch_context
			secret_alliance_can_be_target = yes
		}
	}

	immediate = {
		if = {
			limit = { NOT = { secret_alliance_target_has_active_pact = yes } }
			secret_alliance_initialize_pact = yes
		}
		if = {
			limit = { secret_alliance_target_has_active_pact = yes }
			secret_alliance_register_direct_fire_once_if_needed = yes
			secret_alliance_activate_opening_missions = yes
		}
	}

	option = {
		name = chaosx.nr11.1.a
		trigger = { secret_alliance_target_has_active_pact = yes }
		ai_chance = {
			base = 45
			modifier = { factor = 2 is_ai = yes }
			modifier = { factor = 2 check_variable = { var = secret_alliance_suspicion value = constant:secret_alliance_meter.low compare = greater_than_or_equals } }
		}
		secret_alliance_apply_entry_investigation = yes
		secret_alliance_activate_opening_missions = yes
	}

	option = {
		name = chaosx.nr11.1.b
		trigger = { secret_alliance_target_has_active_pact = yes }
		ai_chance = {
			base = 35
			modifier = { factor = 2 has_war = yes }
			modifier = { factor = 2 check_variable = { var = global.secret_alliance_readiness value = constant:secret_alliance_meter.low compare = greater_than_or_equals } }
		}
		secret_alliance_apply_entry_protection = yes
		secret_alliance_activate_opening_missions = yes
	}

	option = {
		name = chaosx.nr11.1.c
		trigger = { secret_alliance_target_has_active_pact = yes }
		ai_chance = {
			base = 20
			modifier = { factor = 2 NOT = { has_war = yes } }
		}
		secret_alliance_apply_entry_composure = yes
		secret_alliance_activate_opening_missions = yes
	}
}
