	id = chaosx.nr39.65
	title = chaosx.nr39.65.t
	desc = chaosx.nr39.65.d
	picture = GFX_news_event_039_murder_mystery_assassin_state_split
	is_triggered_only = yes
	trigger = {
		has_global_flag = murder_mystery_terminal_war_active
		has_country_flag = murder_mystery_assassin_state
		has_event_target = murder_mystery_terminal_choice
		event_target:murder_mystery_terminal_choice = { murder_mystery_terminal_target_can_be_administered = yes }
	}
	option = {
		name = chaosx.nr39.65.a
		ai_chance = {
			base = @murder_mystery_off_route_choice_weight
			modifier = {
				add = @murder_mystery_route_choice_weight
				NOT = { has_country_flag = murder_mystery_cells_without_masters_route }
				NOT = { has_country_flag = murder_mystery_necessary_mask_route }
			}
		}
		custom_effect_tooltip = murder_mystery_terminal_outcome_subject_tt
		if = {
			limit = {
				NOT = { has_country_flag = murder_mystery_cells_without_masters_route }

			}
			custom_effect_tooltip = murder_mystery_terminal_outcome_route_tt
		}
		else = { custom_effect_tooltip = murder_mystery_terminal_outcome_off_route_tt }
		hidden_effect = {
			set_temp_variable = { murder_mystery_terminal_outcome_input = constant:murder_mystery_terminal.administration_central_subject }
			event_target:murder_mystery_terminal_choice = { murder_mystery_dismantle_terminal_target = yes }
		}
	}
	option = {
		name = chaosx.nr39.65.b
