# ============================================================================
# CHAOS REDUX - EVENT 039 MURDER MYSTERY
# ============================================================================
# Overview:
# - The root is a hidden, fire-once entry point selected by the shared event
#   pool or by the bounded Assassin Network scenario.
# - The opening leader removal and successor installation are performed only by
#   the validated integration transaction; no event option deletes a character.
# - Visible reports describe the six investigation phases, evolution reveals,
#   scheduled incidents, the country split, inheritance, defeat outcomes, the
#   terminal branch, and permanent outcomes. Decisions and missions own
#   repeatable actions and costs.
# - chaosx.nr39.90 is the hidden movement pulse and chaosx.nr39.64 absorbs a
#   capitulated movement country into its winner.
# ============================================================================

	title = chaosx.nr39.1.t

	is_triggered_only = yes
	trigger = {
		murder_mystery_prefire_host_is_valid = yes
	}
	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = murder_mystery_scenario_launched }
				NOT = { has_global_flag = murder_mystery_scenario_launch_in_progress }
			}
			set_global_flag = murder_mystery_natural_run
		}
		murder_mystery_prepare_opening_package = yes
		murder_mystery_commit_opening_transaction = yes
		if = {
			limit = { NOT = { has_global_flag = murder_mystery_opening_transaction_committed } }
			murder_mystery_abort_opening_transaction = yes
		}
		if = {
			limit = {
				has_global_flag = murder_mystery_opening_transaction_committed
				has_global_flag = murder_mystery_scenario_launch_in_progress
			}
			set_global_flag = murder_mystery_scenario_launched
			murder_mystery_apply_scenario_setup = yes
		}
		clr_global_flag = murder_mystery_scenario_launch_in_progress
		clear_variable = global.murder_mystery_scenario_launch_intensity
	}
}

# Hidden movement pulse. A pulse that fires before the recorded due date was
# queued before the pulse was rescheduled and is ignored.
country_event = {
	id = chaosx.nr39.90
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				has_variable = global.murder_mystery_pulse_due_date
				check_variable = { var = global.date value = global.murder_mystery_pulse_due_date compare = greater_than_or_equals }
			}
			clr_global_flag = murder_mystery_pulse_scheduled
			if = {
				limit = {
					murder_mystery_runtime_is_active = yes
					NOT = { has_global_flag = murder_mystery_cleanup_in_progress }
				}
				murder_mystery_run_pulse = yes
			}
			murder_mystery_schedule_pulse = yes
		}
	}
}
