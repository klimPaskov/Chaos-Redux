country_event = {
	id = chaosx.nr1.1
	title = chaosx.nr1.1.t
	desc = chaosx.nr1.1.d
	picture = GFX_report_event_communist_insurgency_start

	is_triggered_only = yes

	trigger = {
		hidden_trigger = { NOT = { is_special_chaos_country = yes } }
		has_communism_spread_minimum_country_size = yes
	}

	immediate = {
		if = {
			limit = {
				hidden_trigger = { NOT = { is_special_chaos_country = yes } }
				has_communism_spread_minimum_country_size = yes
			}
			hidden_effect = {
				if = {
					limit = { NOT = { has_global_flag = communism_spread } }
					set_global_flag = communism_spread
					every_other_country = {
						limit = {
							hidden_trigger = { NOT = { is_special_chaos_country = yes } }
							has_communism_spread_minimum_country_size = yes
						}
						country_event = { id = chaosx.nr1.1 }
					}
				}

				if = {
					limit = { NOT = { has_government = communism } }
					if = {
						limit = { NOT = { has_idea = communism_spread_idea } }
						add_ideas = communism_spread_idea
					}
					queue_communism_spread_defeat_grace_timeout = yes
					apply_communism_spread_initial_support = yes
					seed_initial_communist_control_state = yes
					refresh_communism_spread_pressure = yes
					queue_initial_communism_spread_maintenance = yes
				}
			}
		}
	}

	option = {
		trigger = {
			hidden_trigger = { NOT = { is_special_chaos_country = yes } }
			has_communism_spread_minimum_country_size = yes
			NOT = { has_government = communism }
		}
		name = chaosx.nr1.1.a
		effect_tooltip = {
			add_ideas = communism_spread_idea
			add_popularity = {
				ideology = communism
				popularity = var:communism_spread_initial_support_effect
			}
		}
	}
	option = {
		trigger = {
			hidden_trigger = { NOT = { is_special_chaos_country = yes } }
			has_communism_spread_minimum_country_size = yes
			has_government = communism
		}
		name = chaosx.nr1.1.b
	}
	option = {
		trigger = {
			OR = {
				is_special_chaos_country = yes
				NOT = { has_communism_spread_minimum_country_size = yes }
			}
		}
		name = chaosx.nr1.1.invalid
	}
}
