# Direct-caller confirmation. The requesting country persists the selected
# type and intensity so this event never depends on temporary values.
country_event = {
	id = chaosx.nr19.951
	title = infantry_spawn_scenario_confirmation.t
	desc = infantry_spawn_scenario_confirmation.d
	picture = GFX_report_event_infantry_spawn
	is_triggered_only = yes

	option = {
		name = infantry_spawn_scenario_confirmation.confirm
		trigger = {
			has_country_flag = infantry_spawn_scenario_confirmation_pending
			infantry_spawn_scenario_pending_inputs_are_valid = yes
		}
		custom_effect_tooltip = infantry_spawn_scenario_launch_unregistered_tt
		ai_chance = { factor = 1 }
		hidden_effect = {
			set_temp_variable = {
				infantry_spawn_scenario_launch_type_input = infantry_spawn_scenario_pending_type
			}
			set_temp_variable = {
				infantry_spawn_scenario_launch_intensity_input = infantry_spawn_scenario_pending_intensity
			}
			clr_country_flag = infantry_spawn_scenario_confirmation_pending
			clear_variable = infantry_spawn_scenario_pending_type
			clear_variable = infantry_spawn_scenario_pending_intensity
			infantry_spawn_scenario_launch_unregistered = yes
		}
	}

	option = {
		name = infantry_spawn_scenario_confirmation.cancel
		ai_chance = { factor = 0 }
		hidden_effect = {
			clr_country_flag = infantry_spawn_scenario_confirmation_pending
			clear_variable = infantry_spawn_scenario_pending_type
			clear_variable = infantry_spawn_scenario_pending_intensity
		}
	}
}
