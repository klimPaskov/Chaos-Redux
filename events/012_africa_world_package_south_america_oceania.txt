	id = africa_world_package.505
	title = africa_world_package.505.t
	desc = africa_world_package.505.d
	is_triggered_only = yes
	trigger = {
		africa_world_package_is_installed = yes
		has_country_flag = africa_south_america_south_atlantic_partnership_proof
	}

	immediate = {
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.shared_lanes }
	}

	option = {
		name = africa_world_package.505.a
		trigger = {
			has_country_flag = africa_south_america_resource_debt_law_proof
			has_country_flag = africa_south_america_defence_corridor_proof
			has_country_flag = africa_south_america_south_atlantic_partnership_proof
			OR = {
				has_country_flag = africa_south_america_founding_crisis_success
				has_country_flag = africa_south_america_founding_crisis_compromise
			}
			africa_south_america_three_regions_balance_is_settled = yes
			africa_world_package_heartland_is_proven = yes
			africa_world_package_constituent_settlement_is_proven = yes
		}
		custom_effect_tooltip = africa_world_package.505.a.tt
		ai_chance = {
			base = constant:africa_world_order.response_event_major_chance
			modifier = { factor = constant:africa_world_order.ai_modifier_double check_variable = { africa_world_package_legitimacy > constant:africa_measure.medium } }
		}
		africa_south_america_mark_shared_lane_ratification = yes
		if = {
			limit = { NOT = { has_country_flag = africa_south_america_lifecycle_crisis_resolved } }
			set_country_flag = africa_south_america_lifecycle_crisis_open
			country_event = { id = africa_world_package.506 }
		}
	}

	option = {
		name = africa_world_package.505.b
		custom_effect_tooltip = africa_world_package.505.b.tt
		ai_chance = {
			base = constant:africa_world_order.response_event_base_chance
			modifier = { factor = constant:africa_world_order.ai_modifier_double check_variable = { africa_world_package_constituent_consent_count < constant:africa_world_roster.minimum_constituents } }
		}
		clr_country_flag = africa_south_america_shared_lane_ratification_event_seen
		set_country_flag = africa_south_america_ratification_review_pending
		set_variable = { africa_world_package_phase = constant:africa_world_package_phase.route_committed }
	}
}
