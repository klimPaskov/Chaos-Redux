country_event = {
	id = chaosx_contamination.9
	title = chaosx_contamination.9.t
	desc = chaosx_contamination.9.d
	picture = GFX_report_event_generic_sign_treaty2

	is_triggered_only = yes
	trigger = { air_cleanliness_treaty_pending_invitation_is_valid = yes }
	immediate = { air_cleanliness_treaty_open_invitation = yes }

	option = {
		name = chaosx_contamination.9.a
		trigger = { air_cleanliness_treaty_can_accept_invitation = yes }
		ai_chance = {
			base = constant:air_cleanliness_treaty_lifecycle_ai.join_base
			modifier = {
				factor = constant:air_cleanliness_treaty_lifecycle_ai.join_democratic_factor
				has_government = democratic
			}
			modifier = {
				factor = constant:air_cleanliness_treaty_lifecycle_ai.join_minor_without_stockpile_factor
				is_major = no
				NOT = { has_unconventional_weapon_stockpile = yes }
			}
			modifier = {
				factor = constant:air_cleanliness_treaty_lifecycle_ai.join_stockpile_factor
				has_unconventional_weapon_stockpile = yes
			}
			modifier = {
				factor = constant:air_cleanliness_treaty_lifecycle_ai.join_used_unconventional_factor
				has_used_unconventional_weapons = yes
			}
			modifier = {
				factor = constant:air_cleanliness_treaty_lifecycle_ai.join_war_with_founder_factor
				has_war_with = event_target:air_cleanliness_treaty_founder
			}
			modifier = {
				factor = constant:air_cleanliness_treaty_lifecycle_ai.join_major_factor
				is_major = yes
			}
		}
		air_cleanliness_treaty_accept_invitation = yes
	}

	option = {
		name = chaosx_contamination.9.b
		ai_chance = {
			base = constant:air_cleanliness_treaty_lifecycle_ai.decline_base
		}
		air_cleanliness_treaty_decline_invitation = yes
	}
}
