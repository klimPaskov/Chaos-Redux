add_namespace = chaosx.nr18

# ============================================================================
# CHAOS REDUX - EVENT 018: RESOURCES FOUND
# ============================================================================
# Persistent resource-field discovery, enrichment, diplomacy, border crises,
# evolved incidents, exact closure, cave emergence, continental escalation,
# defeat, and reconstruction callbacks. The canonical entry consumes the
# prepared resources_found_prefire_state/resources_found_prefire_owner targets.
# Direct firing performs one defensive field selection and never reselects
# after the opening package has run.
# ============================================================================

# ---------------------------------------------------------------------------
# Canonical discovery and baseline field lifecycle
# ---------------------------------------------------------------------------

country_event = {
	id = chaosx.nr18.1
	title = chaosx.nr18.1.t
	desc = chaosx.nr18.1.d
	picture = GFX_report_event_018_resource_discovery
	is_triggered_only = yes

	trigger = {
		OR = {
			AND = {
				check_variable = { resources_found_prefire_ready = constant:resources_found_value.one }
				has_event_target = resources_found_prefire_owner
				has_event_target = resources_found_prefire_state
				event_target:resources_found_prefire_owner = { tag = ROOT }
				event_target:resources_found_prefire_state = {
					is_owned_and_controlled_by = ROOT
					resources_found_is_valid_new_field_state = yes
				}
			}
			AND = {
				NOT = { has_event_target = resources_found_prefire_owner }
				NOT = { has_event_target = resources_found_prefire_state }
				any_owned_state = { resources_found_is_valid_new_field_state = yes }
			}
		}
	}

	immediate = {
		hidden_effect = {
		# The generic dispatcher prepares these targets. This block exists only
		# for a deliberate direct fire and performs a single bounded selection.
		if = {
			limit = {
				NOT = { has_event_target = resources_found_prefire_owner }
				NOT = { has_event_target = resources_found_prefire_state }
			}
			save_event_target_as = resources_found_prefire_owner
			if = {
				limit = { any_owned_state = { resources_found_is_valid_new_field_state = yes } }
				random_owned_controlled_state = {
					limit = { resources_found_is_valid_new_field_state = yes }
					save_event_target_as = resources_found_prefire_state
				}
			}
			if = {
				limit = { has_event_target = resources_found_prefire_state }
				set_temp_variable = { resources_found_prefire_ready = constant:resources_found_value.one }
			}
		}
		if = {
			limit = {
				check_variable = { resources_found_prefire_ready = constant:resources_found_value.one }
				has_event_target = resources_found_prefire_owner
				has_event_target = resources_found_prefire_state
				event_target:resources_found_prefire_owner = { tag = ROOT }
				event_target:resources_found_prefire_state = {
					is_owned_and_controlled_by = ROOT
					resources_found_is_valid_new_field_state = yes
				}
			}
			event_target:resources_found_prefire_state = {
				save_event_target_as = resources_found_field_state
				if = {
					limit = { NOT = { resources_found_is_active_field = yes } }
					resources_found_initialize_field_record = yes
				}
				resources_found_apply_prefire_opening_package = yes
				set_state_flag = resources_found_opening_notice_pending
			}
		}
		}
	}

	option = {
		name = chaosx.nr18.1.a
		custom_effect_tooltip = chaosx.nr18.1.a.tt
		hidden_effect = {
			event_target:resources_found_field_state = {
				set_variable = { resources_found_field_posture = constant:resources_found_posture.national_authority }
				set_variable = { resources_found_field_stage = constant:resources_found_field_stage.appraisal }
				set_state_flag = resources_found_national_authority
				resources_found_refresh_field_record = yes
			}
		}
		ai_chance = {
			base = constant:resources_found_ai_weight.national_authority
			modifier = {
				add = constant:resources_found_ai_weight.strong_country_national_add
				num_of_factories > constant:resources_found_country_scale.large_factory_floor
			}
		}
	}

	option = {
		name = chaosx.nr18.1.b
		custom_effect_tooltip = chaosx.nr18.1.b.tt
		hidden_effect = {
			event_target:resources_found_field_state = {
				set_variable = { resources_found_field_posture = constant:resources_found_posture.domestic_charter }
				set_variable = { resources_found_field_stage = constant:resources_found_field_stage.appraisal }
				set_state_flag = resources_found_domestic_charter
				resources_found_refresh_field_record = yes
			}
		}
		ai_chance = {
			base = constant:resources_found_ai_weight.domestic_charter
			modifier = {
				add = constant:resources_found_ai_weight.commercial_government_add
				event_target:resources_found_field_state = {
					check_variable = { resources_found_foreign_pressure > constant:resources_found_threshold.commercial_interest_pressure }
				}
			}
		}
	}

	option = {
		name = chaosx.nr18.1.c
		custom_effect_tooltip = chaosx.nr18.1.c.tt
		hidden_effect = {
			event_target:resources_found_field_state = {
				set_variable = { resources_found_field_posture = constant:resources_found_posture.foreign_concession }
				set_variable = { resources_found_field_stage = constant:resources_found_field_stage.appraisal }
				set_state_flag = resources_found_foreign_concession_open
				resources_found_refresh_field_record = yes
			}
		}
		ai_chance = {
			base = constant:resources_found_ai_weight.foreign_concession
			modifier = {
				add = constant:resources_found_ai_weight.weak_country_concession_add
				NOT = { num_of_factories > constant:resources_found_country_scale.large_factory_floor }
			}
		}
	}

	option = {
		name = chaosx.nr18.1.e
		custom_effect_tooltip = chaosx.nr18.1.e.tt
		hidden_effect = {
			event_target:resources_found_field_state = {
				set_variable = { resources_found_field_posture = constant:resources_found_posture.strategic_reserve }
				set_variable = { resources_found_field_stage = constant:resources_found_field_stage.appraisal }
				set_state_flag = resources_found_strategic_reserve
				resources_found_refresh_field_record = yes
			}
		}
		ai_chance = {
			base = constant:resources_found_ai_weight.strategic_reserve
			modifier = {
				add = constant:resources_found_ai_weight.critical_disturbance_closure_add
				event_target:resources_found_field_state = {
					check_variable = { resources_found_subsurface_disturbance > constant:resources_found_field_value.band_5 }
				}
			}
		}
	}

	after = {
		if = {
			limit = { has_event_target = resources_found_field_state }
			event_target:resources_found_field_state = {
				clr_state_flag = resources_found_opening_notice_pending
				if = {
					limit = { has_state_flag = resources_found_compressed_opening }
					OWNER = { country_event = { id = chaosx.nr18.60 } }
				}
				else_if = {
					limit = { check_variable = { resources_found_last_evolution_stage = constant:resources_found_event.evolution_iii_stage } }
					OWNER = { country_event = { id = chaosx.nr18.60 } }
				}
				else_if = {
					limit = { check_variable = { resources_found_last_evolution_stage = constant:resources_found_event.evolution_ii_stage } }
					OWNER = { country_event = { id = chaosx.nr18.50 } }
				}
				else_if = {
					limit = { check_variable = { resources_found_last_evolution_stage = constant:resources_found_event.evolution_i_stage } }
					OWNER = { country_event = { id = chaosx.nr18.41 } }
				}
			}
		}
		resources_found_schedule_evolution_paths = yes
		resources_found_refresh_decision_cost_previews = yes
	}
}

country_event = {
	id = chaosx.nr18.2

	desc = chaosx.nr18.2.d
	picture = GFX_report_event_018_resource_discovery
	is_triggered_only = yes

	trigger = {

				var:resources_found_selected_field = { resources_found_is_valid_enrichment_field = yes }

		}
	}

	immediate = {
		hidden_effect = {
		if = {
			limit = {
				NOT = { has_event_target = resources_found_prefire_owner }
				NOT = { has_event_target = resources_found_prefire_state }
				resources_found_country_has_valid_selected_field = yes
				var:resources_found_selected_field = { resources_found_is_valid_enrichment_field = yes }
			}
			save_event_target_as = resources_found_prefire_owner
			var:resources_found_selected_field = { save_event_target_as = resources_found_prefire_state }
			set_temp_variable = { resources_found_prefire_ready = constant:resources_found_value.one }
		}
		if = {
			limit = {
				check_variable = { resources_found_prefire_ready = constant:resources_found_value.one }
				has_event_target = resources_found_prefire_owner
				has_event_target = resources_found_prefire_state
				event_target:resources_found_prefire_owner = { tag = ROOT }
				event_target:resources_found_prefire_state = {
					is_owned_and_controlled_by = ROOT
					resources_found_is_valid_enrichment_field = yes
				}
			}
			event_target:resources_found_prefire_state = {
				save_event_target_as = resources_found_field_state
				clr_state_flag = resources_found_evolution_i_active_enrichment
				clr_state_flag = resources_found_evolution_ii_active_enrichment
				clr_state_flag = resources_found_evolution_iii_active_enrichment
				clr_state_flag = resources_found_evolution_iv_active_enrichment
				if = {
					limit = { resources_found_is_evolution_iv_enabled = yes }
					resources_found_apply_evolution_iii_all_resources_package = yes
					set_state_flag = resources_found_evolution_iv_active_enrichment
					set_state_flag = resources_found_compressed_opening
					set_state_flag = resources_found_breach_prone_opening
					set_variable = { resources_found_compressed_incident_minimum_date = global.date }
					set_temp_variable = { resources_found_compressed_incident_delay = constant:resources_found_project_duration.compressed_incident_minimum_days }
					add_to_variable = { resources_found_compressed_incident_minimum_date = resources_found_compressed_incident_delay }
				}
				else_if = {
					limit = { resources_found_is_evolution_iii_enabled = yes }
					resources_found_apply_evolution_iii_all_resources_package = yes
					set_state_flag = resources_found_evolution_iii_active_enrichment
				}
				else_if = {
					limit = { resources_found_is_evolution_ii_enabled = yes }
					resources_found_apply_evolution_ii_active_package = yes
					set_state_flag = resources_found_evolution_ii_active_enrichment
				}
				else_if = {
					limit = { resources_found_is_evolution_i_enabled = yes }
					resources_found_apply_evolution_i_active_package = yes
					set_state_flag = resources_found_evolution_i_active_enrichment
				}
				else = { resources_found_apply_baseline_package = yes }
			}
		}
		}
	}

	option = {
		name = chaosx.nr18.2.a
		custom_effect_tooltip = chaosx.nr18.2.a.tt
		hidden_effect = {
			event_target:resources_found_field_state = {
				set_variable = { resources_found_field_stage = constant:resources_found_field_stage.development }
				set_state_flag = resources_found_enrichment_integrated
				resources_found_refresh_field_record = yes
			}
		}
		ai_chance = {
			base = constant:resources_found_ai_weight.base
			modifier = {
				add = constant:resources_found_ai_weight.high_safety_exploitation_add
				event_target:resources_found_field_state = {
					check_variable = { resources_found_workforce_safety > constant:resources_found_field_value.band_4 }
				}
			}
		}
	}

	option = {
		name = chaosx.nr18.2.b
		custom_effect_tooltip = chaosx.nr18.2.b.tt
		hidden_effect = {
			event_target:resources_found_field_state = {
				set_variable = { resources_found_field_posture = constant:resources_found_posture.strategic_reserve }
				set_state_flag = resources_found_enrichment_reserved
				resources_found_suspend_field = yes
			}
		}
		ai_chance = {
			base = constant:resources_found_ai_weight.cautious_appraisal
			modifier = {
				add = constant:resources_found_ai_weight.low_safety_caution_add
				event_target:resources_found_field_state = {
					check_variable = { resources_found_workforce_safety < constant:resources_found_field_value.band_3 }
				}
			}
		}
	}

	after = {
		if = {
			limit = { has_event_target = resources_found_field_state }
			event_target:resources_found_field_state = {
				if = {
					limit = { has_state_flag = resources_found_evolution_iv_active_enrichment }
					OWNER = { country_event = { id = chaosx.nr18.60 } }
				}
				else_if = {
					limit = { has_state_flag = resources_found_evolution_iii_active_enrichment }
					OWNER = { country_event = { id = chaosx.nr18.60 } }
				}
				else_if = {
					limit = { has_state_flag = resources_found_evolution_ii_active_enrichment }
					OWNER = { country_event = { id = chaosx.nr18.50 } }
				}
				else_if = {
					limit = { has_state_flag = resources_found_evolution_i_active_enrichment }
					OWNER = { country_event = { id = chaosx.nr18.40 } }
				}
				clr_state_flag = resources_found_evolution_i_active_enrichment
				clr_state_flag = resources_found_evolution_ii_active_enrichment
				clr_state_flag = resources_found_evolution_iii_active_enrichment
				clr_state_flag = resources_found_evolution_iv_active_enrichment
			}
		}
		resources_found_schedule_evolution_paths = yes
		resources_found_refresh_decision_cost_previews = yes
	}
