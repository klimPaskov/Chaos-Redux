# ============================================================================
# CHAOS REDUX - ORDINARY BIOLOGICAL NOTIFICATION EVENTS
# ============================================================================
# Overview:
# - Field-test events only report the outcome already resolved in the exact
#   native special-project facility state.
# - Arsenal-incident events only report the stockpile-safety resolver's exact
#   state, stock loss, severity, and lifecycle dispatch.
# - Smallpox public-use and first-detection ids remain stable notifications.
# - Incubation, detection, spread, deaths, treatment, and cleanup are owned by
#   cbrn_bio_lifecycle and never by MTTH events in this file.
# ============================================================================

# Anthrax stockpile accident notification. Accounting and lifecycle dispatch
# are completed once by biological_stockpile_safety_effects.txt.
country_event = {
	id = chaosx_bioweapon.4
	title = chaosx_bioweapon.4.t
	desc = chaosx_bioweapon.4.d
	picture = GFX_report_event_generic_bioweapon
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_variable = bio_stockpile_safety_arsenal_state }
			var:bio_stockpile_safety_arsenal_state = { save_event_target_as = lab_accident_state }
		}
	}

	option = {
		name = chaosx_bioweapon.4.a
		custom_effect_tooltip = bio_stockpile_safety_accident_resolved_tt
		hidden_effect = { bio_stockpile_safety_cleanup_runtime_context = yes }
	}
}

	id = biowarfare.1
	title = biowarfare.1.t
	desc = biowarfare.1.d
	picture = GFX_report_event_generic_biowarfare
	is_triggered_only = yes

	option = {
		name = biowarfare.1.a
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				has_government = fascism
			}
		}
		random_controlled_state = {
			limit = { state = 88 }
			add_building_construction = {
				type = biowarfare_facility
				level = 1
				instant_build = yes
				province = { id = 9412 }
			}
		}
	}

	option = {
		name = biowarfare.1.b
		add_stability = 0.05
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_government = fascism
			}
		}
	}
}
