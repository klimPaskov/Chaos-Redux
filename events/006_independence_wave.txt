# Canonical repeatable entry. A joint-cluster presentation always takes
# priority and cannot allocate a second wave. Standalone calls present only
# after the frozen plan has reached the committed phase.
country_event = {
	id = chaosx.nr6.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		clr_global_flag = independence_wave_event_log_actor_ready
		if = {
			# The pending flag is only a delivery receipt for a committed joint
			# Event 005 + Event 006 plan. A stale flag must never swallow a fresh
			# standalone invocation before the allocator has had a chance to run.
			limit = {
				has_global_flag = independence_wave_joint_presentation_pending
				has_global_flag = liberation_release_joint_plan_executed
				check_variable = { global.liberation_plan_phase = constant:liberation_plan_phase.committed }
				check_variable = { global.liberation_plan_owner = constant:liberation_plan_owner.joint }
				check_variable = {
					var = global.independence_wave_presentation_country_count
					value = constant:independence_wave_planner.no_candidates
					compare = greater_than
				}
			}
			clr_global_flag = independence_wave_joint_presentation_pending
			set_global_flag = independence_wave_joint_presentation_delivered
			if = {
				limit = {
					has_event_target = independence_wave_latest_actor
					event_target:independence_wave_latest_actor = { exists = yes }
				}
				set_global_flag = independence_wave_event_log_actor_ready
			}
			country_event = { id = chaosx.nr6.2 }
		}
		else = {
			# Clear an orphaned joint-delivery marker before opening a new
			# standalone transaction. This is state cleanup only; it creates no
			# player-facing pre-event surface.
			clr_global_flag = independence_wave_joint_presentation_pending
			independence_wave_prepare_and_execute_standalone_incident = yes
			if = {
				limit = { has_global_flag = independence_wave_standalone_incident_committed }
				if = {
					limit = {
						has_event_target = independence_wave_latest_actor
						event_target:independence_wave_latest_actor = { exists = yes }
					}
					set_global_flag = independence_wave_event_log_actor_ready
				}
				country_event = { id = chaosx.nr6.2 }
			}
		}
	}
}

	title = chaosx.nr6.311.t
	desc = chaosx.nr6.311.d
	picture = GFX_report_event_006_asset_001_wave_summary
	is_triggered_only = yes
	trigger = {
		is_independence_wave_active_country = yes
		has_country_flag = independence_wave_dm01_relocation_required
	}

	option = {
		name = chaosx.nr6.311.a
		random_owned_controlled_state = {
			limit = {
				is_controlled_by = ROOT
				NOT = { is_capital = yes }
			}
			ROOT = {
				set_capital = { state = PREV remember_old_capital = no }
				set_country_flag = independence_wave_dm01_relocation_complete
			}
		}
		if = {
			limit = { NOT = { has_country_flag = independence_wave_dm01_relocation_complete } }
			set_country_flag = independence_wave_dm01_relocation_complete
		}
		clr_country_flag = independence_wave_dm01_relocation_required
		independence_wave_refresh_country_state = yes
	}

	option = {
		name = chaosx.nr6.311.b
		set_country_flag = independence_wave_dm01_relocation_complete
		clr_country_flag = independence_wave_dm01_relocation_required
		independence_wave_refresh_country_state = yes
	}
