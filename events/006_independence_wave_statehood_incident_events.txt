# ============================================================================
# CHAOS REDUX - EVENT 006 STATEHOOD INCIDENTS
# ============================================================================
# The recurring incident layer of every active Event 006 country.
# - chaosx.nr6.150 is the hidden pulse. The country queues it when it resolves
#   its founding posture, and each pulse queues the next one.
# - chaosx.nr6.151 to .170 are the baseline incidents: founding (.151-.154),
#   recognition (.155-.157), security (.158-.160), government (.161-.164),
#   League (.165-.167) and formable (.168-.170).
# - chaosx.nr6.171 to .181 are evolution incidents: Replicable Independence
#   (.171-.172), Dormant Nations (.173-.174), Armed Birth (.175-.176),
#   Sovereign Congress (.177-.179) and Open Sovereignty (.180-.181).
# Each incident fires only while it is the open incident of the live
# generation. Options trade the five Statehood values against the former-host
# relation, the League values, Network Standing or a resource cost.
# ============================================================================

add_namespace = chaosx.nr6

country_event = {
	id = chaosx.nr6.150

	is_triggered_only = yes

	immediate = { independence_wave_run_statehood_incident_pulse = yes }
