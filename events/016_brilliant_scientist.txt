# ============================================================================
# CHAOS REDUX - EVENT 016 BRILLIANT SCIENTIST OPENING
# ============================================================================
# The shared random-event dispatcher resolves one weighted eligible country
# before firing .1. The visible appointment then recruits exactly one Doctor
# Warren Kruger through a guarded scripted transaction. A human may redirect
# the still-unrecruited character once to a prevalidated weighted recipient;
# AI countries always choose one of the two appointment routes.
# ============================================================================

# Dispatcher entry. ROOT is already the validated prefire host.
country_event = {
	id = chaosx.nr16.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				has_country_flag = brilliant_scientist_opening_chain_active
				brilliant_scientist_is_eligible_host = yes
				NOT = { has_global_flag = brilliant_scientist_event_resolved }
			}
			country_event = { id = chaosx.nr16.2 }
		}
	}
}

					has_government = fascism
					has_government = communism
					has_government = neutrality
				}
			}
			modifier = { factor = constant:brilliant_scientist_opening_ai.secret_unstable_factor has_stability < constant:brilliant_scientist_opening_ai.low_stability }
			modifier = {
				factor = constant:brilliant_scientist_opening_ai.secret_strategic_facility_factor
				OR = {
					has_facility_specialization = specialization_nuclear
					has_facility_specialization = specialization_air
					has_facility_specialization = specialization_land
					has_facility_specialization = specialization_naval
					has_facility_specialization = specialization_biowarfare
					has_facility_specialization = specialization_cw
				}
			}
			modifier = { factor = constant:brilliant_scientist_opening_ai.secret_technology_gap_factor NOT = { has_tech = computing_machine } }
		}
		custom_effect_tooltip = brilliant_scientist_secret_appointment_effect_tt
		hidden_effect = {
			set_temp_variable = { brilliant_scientist_recruitment_posture = constant:brilliant_scientist_recruitment_posture.secret_directorate }
			brilliant_scientist_appoint_kruger_from_opening = yes
		}
	}

	option = {
		name = chaosx.nr16.2.c
		trigger = {
			is_ai = no
			NOT = { has_global_flag = brilliant_scientist_event_resolved }

			}

		custom_effect_tooltip = brilliant_scientist_send_away_effect_tt
		hidden_effect = {
			brilliant_scientist_forward_opening_to_selected_recipient = yes
		}
	}
}

# The country that declined Kruger receives one bounded intelligence report
# after the referred appointment commits. This is a consequence of the
# opening choice only; it never reopens recruitment or creates a second actor.
country_event = {
	id = chaosx.nr16.15
	title = chaosx.nr16.15.t
	desc = {
		trigger = {
			has_event_target = brilliant_scientist_send_away_recipient
			event_target:brilliant_scientist_send_away_recipient = {
				has_country_flag = brilliant_scientist_public_compact
			}
		}
		text = chaosx.nr16.15.public.d
	}
	desc = {
		trigger = {
			has_event_target = brilliant_scientist_send_away_recipient
			event_target:brilliant_scientist_send_away_recipient = {
				has_country_flag = brilliant_scientist_secret_directorate
			}
		}
		text = chaosx.nr16.15.secret.d
	}
	desc = chaosx.nr16.15.default.d
	picture = GFX_report_event_016_brilliant_scientist_directorate_dossier
	is_triggered_only = yes
	trigger = {
		has_country_flag = brilliant_scientist_rejection_reaction_pending
		has_country_flag = brilliant_scientist_rejection_memory
		NOT = { has_country_flag = brilliant_scientist_rejection_reaction_seen }
		has_event_target = brilliant_scientist_send_away_recipient
		event_target:brilliant_scientist_send_away_recipient = {
			has_country_flag = brilliant_scientist_appointment_completed
		}
	}

	option = {
		name = chaosx.nr16.15.a
		custom_effect_tooltip = brilliant_scientist_rejection_reaction_effect_tt
		hidden_effect = {
			clr_country_flag = brilliant_scientist_rejection_memory
			clr_country_flag = brilliant_scientist_rejection_reaction_pending
			set_country_flag = brilliant_scientist_rejection_reaction_seen
		}
	}
}
