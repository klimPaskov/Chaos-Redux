# ============================================================================
# CHAOS REDUX - EVENT 006 PACKAGE INCIDENTS
# ============================================================================
# Two country incidents for each Level 2 package that had none: SCO, WLS, BRI,
# CAT, TRA, MAC, BOS, KAR, CRI, RUT, KUB, TAT, BSK, ARM, GEO, AZR, HAW, HBX, NAV,
# GLC and DON. The first is scheduled when the country settles its founding
# posture; either answer schedules the second. Each option trades the five
# Statehood values against the package's own ledgers, the former-host relation
# or a resource cost. Triggers require the same Event 006 generation that
# scheduled them, so a later release of the tag never receives a stale event.
# ============================================================================

add_namespace = chaosx.nr6

country_event = {
	id = chaosx.nr6.101
	title = chaosx.nr6.101.t
	desc = chaosx.nr6.101.d
	picture = GFX_report_event_merchant_ship_01
	is_triggered_only = yes
	trigger = {
		is_independence_wave_sco_package = yes
		can_fire_independence_wave_package_incident = yes
		has_country_flag = independence_wave_package_incident_a_scheduled
		NOT = { has_country_flag = independence_wave_package_incident_a_resolved }
	}

	option = {
		name = chaosx.nr6.101.a
		custom_effect_tooltip = chaosx.nr6.101.a.tt
		hidden_effect = {
			set_country_flag = independence_wave_package_incident_a_resolved
			independence_wave_package_incident_clear_deltas = yes
			set_temp_variable = { independence_wave_decision_legitimacy_delta = constant:independence_wave_value.minor_loss }
			set_temp_variable = { independence_wave_decision_recognition_delta = constant:independence_wave_value.minor_gain }
			independence_wave_decision_apply_country_deltas = yes
			set_temp_variable = { independence_wave_sco_shipping_delta = constant:independence_wave_scotland_wales_pressure.minor_gain }
			independence_wave_change_sco_shipping_authority = yes
			if = {
				limit = { has_independence_wave_living_former_host = yes }
				set_temp_variable = { independence_wave_decision_hostility_delta = constant:independence_wave_decision_effect.bilateral_minor_loss }
				independence_wave_decision_apply_host_deltas = yes
			}
			independence_wave_schedule_package_incident_b = yes
		}
		ai_chance = {
			base = constant:independence_wave_decision_ai.standard
			modifier = { factor = constant:independence_wave_decision_ai.modifier_double has_independence_wave_severe_host_threat = yes }
		}
	}

	option = {
		name = chaosx.nr6.101.b
		custom_effect_tooltip = chaosx.nr6.101.b.tt
		hidden_effect = {
			set_country_flag = independence_wave_package_incident_a_resolved
			independence_wave_package_incident_clear_deltas = yes
			set_temp_variable = { independence_wave_decision_recognition_delta = constant:independence_wave_value.minor_loss }
			set_temp_variable = { independence_wave_decision_security_delta = constant:independence_wave_value.minor_gain }
			set_temp_variable = { independence_wave_decision_instability_delta = constant:independence_wave_value.minor_gain }
			independence_wave_decision_apply_country_deltas = yes
			set_temp_variable = { independence_wave_sco_shipping_delta = constant:independence_wave_scotland_wales_pressure.standard_gain }
			independence_wave_change_sco_shipping_authority = yes
			if = {
				limit = { has_independence_wave_living_former_host = yes }
				set_temp_variable = { independence_wave_decision_hostility_delta = constant:independence_wave_decision_effect.bilateral_minor }
				independence_wave_decision_apply_host_deltas = yes
			}
			independence_wave_schedule_package_incident_b = yes
		}
		ai_chance = {
			base = constant:independence_wave_decision_ai.standard
			modifier = { factor = constant:independence_wave_decision_ai.modifier_double OR = { has_independence_wave_constitutional_route = yes has_independence_wave_popular_council_route = yes } }
