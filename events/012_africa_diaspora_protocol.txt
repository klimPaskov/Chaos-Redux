add_namespace = chaosx.nr12

# ============================================================================
# CHAOS REDUX - EVENT 012 DIASPORA OWNER-PROTOCOL EVENTS
# ============================================================================
# The selected target owns every consent, counterterm, refusal, and withdrawal
# answer. Event targets retain the host/target pair for this one request chain;
# scripted effects clear those targets after resume, refusal, or host closure.
# ============================================================================

# Target receives a voluntary diaspora request.
country_event = {
	id = chaosx.nr12.310
	title = chaosx.nr12.310.t
	desc = chaosx.nr12.310.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	trigger = {
		has_event_target = africa_diaspora_request_host
		event_target:africa_diaspora_request_host = {
			has_country_flag = africa_diaspora_request_pending
			africa_diaspora_request_action_is_valid = yes
		}
	}

	option = {
		name = chaosx.nr12.310.a
		custom_effect_tooltip = chaosx.nr12.310.a.tt
		ai_chance = { base = constant:africa_diaspora_ai.owner_accept }
		africa_diaspora_accept_target_request = yes
	}
	option = {
		name = chaosx.nr12.310.b
		custom_effect_tooltip = chaosx.nr12.310.b.tt
		ai_chance = { base = constant:africa_diaspora_ai.owner_negotiate }
		africa_diaspora_request_counterterms = yes
	}
	option = {
		name = chaosx.nr12.310.c
		custom_effect_tooltip = chaosx.nr12.310.c.tt
		ai_chance = { base = constant:africa_diaspora_ai.owner_refuse }
		africa_diaspora_refuse_target_request = yes
	}
}

		event_target:africa_diaspora_request_target = { has_country_flag = africa_diaspora_terms_requested }

	}

		name = chaosx.nr12.311.b

		event_target:africa_diaspora_request_target = { africa_diaspora_close_host_offer = yes }
	}
}

# Delayed target-owned review while a diaspora action remains active.
