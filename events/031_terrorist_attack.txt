# Event 31 is a single global pacing event. The root is hidden and fans one
# bounded wave into eligible countries and states; the report callback is

	immediate = {
		if = {
			limit = { has_global_flag = random_terror_automatic_wave_pending }
			set_global_flag = random_terror_automatic_wave_active
		}
		random_terror_run_global_wave = yes
		random_terror_refresh_global_values = yes
		random_terror_schedule_false_revelation = yes
		random_terror_check_for_defeat = yes
		if = {
			limit = { has_global_flag = random_terror_automatic_wave_pending }
			random_terror_record_automatic_wave_history = yes
			clr_global_flag = random_terror_automatic_wave_active
			clr_global_flag = random_terror_automatic_wave_pending
		}
		if = {
			limit = { NOT = { has_global_flag = random_terror_wave_news_sent } }
			news_event = { id = chaosx.news.310 days = 2 }
			set_global_flag = random_terror_wave_news_sent
		}
		set_temp_variable = { random_terror_dispatch_cleanup_days = constant:random_terror_tuning.wave_cooldown_days }
		country_event = { id = chaosx.nr31.99 days = random_terror_dispatch_cleanup_days }
	}
}
