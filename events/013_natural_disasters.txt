country_event = {
	id = chaosx.nr13.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		set_temp_variable = { natural_disaster_call_caller_type = constant:natural_disaster_caller.random }
		set_temp_variable = { natural_disaster_call_caller_event_id = constant:natural_disaster_event.id }
		if = {
			limit = {
				NOT = {
					AND = {
						has_event_target = natural_disaster_call_target_state
						event_target:natural_disaster_call_target_state = { has_variable = natural_disaster_prefire_family }
					}
				}
			}
			natural_disaster_prepare_random_event_fire = yes
		}
		if = {
			limit = {
				has_event_target = natural_disaster_call_target_state
				event_target:natural_disaster_call_target_state = { has_variable = natural_disaster_prefire_family }
			}
			event_target:natural_disaster_call_target_state = {
				ROOT = { set_temp_variable = { natural_disaster_call_family = PREV.natural_disaster_prefire_family } }
				clear_variable = natural_disaster_prefire_family
			}
			set_temp_variable = { natural_disaster_call_target_state_supplied = 1 }
			set_temp_variable = { natural_disaster_call_target_mode = constant:natural_disaster_target_mode.selected_state }
		}
		else = {
			set_temp_variable = { natural_disaster_call_family = constant:natural_disaster_family.random }
			save_event_target_as = natural_disaster_call_target_country
			set_temp_variable = { natural_disaster_call_target_country_supplied = 1 }
			set_temp_variable = { natural_disaster_call_target_mode = constant:natural_disaster_target_mode.selected_country }
		}
		set_temp_variable = { natural_disaster_call_sequence_mode = constant:natural_disaster_sequence_mode.local_season }
		set_temp_variable = { natural_disaster_call_news_policy = constant:natural_disaster_news_policy.meaningful }
		set_temp_variable = { natural_disaster_call_report_policy = constant:natural_disaster_report_policy.caller }
		set_temp_variable = { natural_disaster_call_aftermath_policy = constant:natural_disaster_aftermath_policy.normal }
		set_temp_variable = { natural_disaster_call_chain_policy = constant:natural_disaster_chain_policy.family }
		set_temp_variable = { natural_disaster_call_log_mode = constant:natural_disaster_log_mode.event_system }
		call_natural_disaster = yes
	}
}

country_event = {
	id = chaosx.nr13.3
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				has_country_flag = achievement_nd_barrage_maximum_started
				NOT = { has_country_flag = achievement_nd_barrage_maximum_ended }
				check_variable = {
					var = achievement_nd_barrage_expiry_date
					value = global.date
					compare = less_than_or_equals
				}
			}
			set_country_flag = achievement_nd_barrage_deadline_reached
			natural_disaster_achievement_check_operational_state = yes
			if = {
				limit = {
					OR = {
						NOT = { has_country_flag = achievement_nd_barrage_impact_sequence_ended }
						natural_disaster_country_has_open_aftermath = yes
						has_country_flag = natural_disaster_chain_mission_active
						check_variable = {
							var = achievement_nd_barrage_recovery_cards
							value = constant:natural_disaster_achievement.barrage_maximum_recovery_cards
							compare = less_than
						}
						NOT = { has_country_flag = achievement_nd_barrage_capital_route_preserved }
						AND = {
							NOT = { has_country_flag = achievement_nd_barrage_port_corridor_preserved }
							NOT = { has_country_flag = achievement_nd_barrage_rail_corridor_preserved }
						}
						has_country_flag = achievement_nd_barrage_country_collapsed
					}
				}
				set_country_flag = achievement_nd_barrage_aftermath_unresolved
			}
			else = {
				set_country_flag = achievement_nd_barrage_maximum_ended
			}
		}
	}
}
