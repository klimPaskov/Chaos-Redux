# Event 013 - Natural Disasters

country_event = {
	id = chaosx.nr13.1
	title = chaosx.nr13.1.t

	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_event_target = natural_disaster_random_event_actor } }
			natural_disaster_prepare_random_event_fire = yes
		}
		if = {
			limit = { has_event_target = natural_disaster_random_event_actor }
			event_target:natural_disaster_random_event_actor = {
				country_event = { id = chaosx.nr13.100 days = 1 }
			}
		}
	}
}

country_event = {
	id = chaosx.nr13.100
	title = chaosx.nr13.100.t

	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_event_target = natural_disaster_current_target_state } }
			natural_disaster_prepare_random_event_context = yes
		}
		if = {
			limit = { has_event_target = natural_disaster_current_target_state }
			natural_disaster_store_pending_impact_context = yes
			natural_disaster_record_evolution_stage = yes
			natural_disaster_emit_abnormal_age_super_event = yes
			set_country_flag = { flag = natural_disaster_primary_incident_for_burst days = 60 }

				limit = { check_variable = { natural_disaster_should_warn > 0 } }
				natural_disaster_setup_warning = yes
				country_event = { id = chaosx.nr13.10 days = 1 }
				country_event = { id = chaosx.nr13.20 days = 14 random_days = 14 }
			}
			else = {
				country_event = { id = chaosx.nr13.20 days = 1 }
			}
		}
	}
}

country_event = {
	id = chaosx.nr13.20
	title = chaosx.nr13.20.t
	desc = chaosx.nr13.20.d
	picture = GFX_report_event_natural_disaster_impact

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_event_target = natural_disaster_current_target_state } }
			random_controlled_state = {
				limit = { has_state_flag = natural_disaster_pending_impact }
				save_event_target_as = natural_disaster_current_target_state
				CONTROLLER = { save_event_target_as = natural_disaster_current_controller }
			}
		}
		if = {
			limit = { has_variable = natural_disaster_current_stage }
			set_temp_variable = { natural_disaster_stage = natural_disaster_current_stage }
		}
	}

	option = {
		name = chaosx.nr13.20.a
		custom_effect_tooltip = chaosx.nr13.20.tt
		hidden_effect = {
			if = {
				limit = { has_event_target = natural_disaster_current_target_state }
				event_target:natural_disaster_current_target_state = {
					natural_disaster_apply_impact_to_current_state = yes
				}
			}
			if = {
				limit = { has_variable = natural_disaster_current_stage }
				set_temp_variable = { natural_disaster_stage = natural_disaster_current_stage }
			}
			if = {
				limit = {
					has_country_flag = natural_disaster_primary_incident_for_burst
					check_variable = { natural_disaster_stage > constant:natural_disaster_event.stage_baseline }
				}
				natural_disaster_queue_evolved_burst = yes
				clr_country_flag = natural_disaster_primary_incident_for_burst
			}
			if = {
				limit = {
					has_variable = natural_disaster_current_stage
					check_variable = { natural_disaster_current_stage > constant:natural_disaster_event.stage_regional_systems }
				}
				country_event = { id = chaosx.nr13.30 days = 12 random_days = 18 }
			}
			if = {
				limit = { check_variable = { natural_disaster_stage = constant:natural_disaster_event.stage_abnormal_age } }
				set_global_flag = natural_disaster_abnormal_disaster_age_without_world_end
			}
		}
		ai_chance = { base = 100 }
	}
}

country_event = {
	id = chaosx.nr13.30
	title = chaosx.nr13.30.t
	desc = chaosx.nr13.30.d
	picture = GFX_report_event_natural_disaster_impact

	is_triggered_only = yes

	trigger = {
		natural_disaster_country_has_active_aftermath = yes
	}

	option = {
		name = chaosx.nr13.30.a
		custom_effect_tooltip = chaosx.nr13.30.tt
		hidden_effect = {
			if = {
				limit = {
					any_controlled_state = {
						natural_disaster_state_has_active_aftermath = yes
						has_variable = natural_disaster_state_family
					}
				}
				random_controlled_state = {
					limit = {
						natural_disaster_state_has_active_aftermath = yes
						has_variable = natural_disaster_state_family
					}
					natural_disaster_apply_family_chain_follow_up_from_state = yes
				}
			}
			else = {
				set_country_flag = natural_disaster_chained_aftermath_escalated
				if = {
					limit = { has_variable = natural_disaster_recovery_stability_risk }
					add_stability = natural_disaster_recovery_stability_risk
				}
				if = {
					limit = { has_variable = natural_disaster_recovery_war_support_risk }
					add_war_support = natural_disaster_recovery_war_support_risk
				}
			}
		}
		ai_chance = { base = 100 }
	}
}
