	title = {

	}
	title = chaosx.nr16.49.t
	desc = {
		text = chaosx.nr16.49.partner.imperial.d
		trigger = {
			event_target:dhrondan_diplomatic_actor = {
				has_country_flag = dhrondan_diplomatic_offer_partner_accord
				dhrondan_is_imperial_regime = yes
			}
		}
	}
	desc = {
		text = chaosx.nr16.49.partner.synod_market.d
		trigger = {
			event_target:dhrondan_diplomatic_actor = {
				has_country_flag = dhrondan_diplomatic_offer_partner_accord
				dhrondan_is_synod_regime = yes
				has_dlc = "Arms Against Tyranny"
			}
		}
	}
	desc = {
		text = chaosx.nr16.49.partner.synod_military.d
		trigger = {
			event_target:dhrondan_diplomatic_actor = {
				has_country_flag = dhrondan_diplomatic_offer_partner_accord
				dhrondan_is_synod_regime = yes
				NOT = { has_dlc = "Arms Against Tyranny" }
			}
		}
	}
	desc = {
		text = chaosx.nr16.49.partner.covenant.d
		trigger = {
			event_target:dhrondan_diplomatic_actor = {
				has_country_flag = dhrondan_diplomatic_offer_partner_accord
				dhrondan_is_covenant_regime = yes
			}
		}
	}
	desc = chaosx.nr16.49.d
	picture = GFX_report_event_016_dhrondan_diplomatic_compact
	is_triggered_only = yes
	timeout_days = @DHR_COMPACT_RESPONSE_TIMEOUT_DAYS
	trigger = {
		exists = yes
		dhrondan_compact_receipt_matches_recipient = yes
		event_target:dhrondan_diplomatic_actor = {
			exists = yes
			NOT = { has_country_flag = dhrondan_diplomatic_offer_delivered }
			has_variable = dhrondan_diplomatic_offer_expiry_num_days
			check_variable = { dhrondan_diplomatic_offer_expiry_num_days > global.num_days }
		}
	}
	immediate = {
		event_target:dhrondan_diplomatic_actor = {
			set_country_flag = dhrondan_diplomatic_offer_delivered
			dhrondan_refresh_compact_expiry = yes
		}
	}

	option = {
		name = chaosx.nr16.49.partner.b
		trigger = {
			dhrondan_compact_response_is_valid = yes
			event_target:dhrondan_diplomatic_actor = { has_country_flag = dhrondan_diplomatic_offer_partner_accord }
		}
		ai_chance = {
			base = constant:dhrondan_event_ai.refuse_base
			modifier = {
				factor = constant:dhrondan_event_ai.hostile_factor
				has_opinion = {
					target = event_target:dhrondan_diplomatic_actor
					value < @DHR_NEGATIVE_OPINION_THRESHOLD
				}
			}
		}
		custom_effect_tooltip = dhrondan_partner_accord_refuse_effect_tt
		hidden_effect = {
			if = {
				limit = {
					dhrondan_compact_response_can_commit = yes
					event_target:dhrondan_diplomatic_actor = { has_country_flag = dhrondan_diplomatic_offer_partner_accord }
				}
				event_target:dhrondan_diplomatic_actor = { dhrondan_clear_diplomatic_offer = yes }
			}
			else = { dhrondan_close_owned_compact_response = yes }
		}
	}

	option = {
		name = chaosx.nr16.49.response.a
		trigger = { dhrondan_compact_response_is_valid = yes }
		ai_chance = {
			base = constant:dhrondan_event_ai.accept_base
			modifier = {
				factor = constant:dhrondan_event_ai.friendly_factor
				has_opinion = {
					target = event_target:dhrondan_diplomatic_actor
					value > @DHR_POSITIVE_OPINION_THRESHOLD
				}
			}
			modifier = { factor = constant:dhrondan_event_ai.covenant_factor has_government = democratic }
		}
		if = {
			limit = { event_target:dhrondan_diplomatic_actor = { hidden_trigger = { has_country_flag = dhrondan_diplomatic_offer_partner_accord } } }
			if = {
				limit = { event_target:dhrondan_diplomatic_actor = { dhrondan_is_imperial_regime = yes } }
				custom_effect_tooltip = dhrondan_partner_accord_imperial_accept_effect_tt
			}
			if = {
				limit = { event_target:dhrondan_diplomatic_actor = { dhrondan_is_synod_regime = yes } }
				custom_effect_tooltip = dhrondan_partner_accord_synod_accept_effect_tt
			}
			if = {
				limit = { event_target:dhrondan_diplomatic_actor = { dhrondan_is_covenant_regime = yes } }
				custom_effect_tooltip = dhrondan_partner_accord_covenant_accept_effect_tt
			}
		}
		else = {
			custom_effect_tooltip = dhrondan_compact_accept_effect_tt
			if = {
				limit = { event_target:dhrondan_diplomatic_actor = { hidden_trigger = { has_country_flag = dhrondan_right_to_depart_guaranteed } } }
				custom_effect_tooltip = dhrondan_compact_departure_terms_tt
			}
			if = {
				limit = {
					is_in_faction = no
					event_target:dhrondan_diplomatic_actor = {
						hidden_trigger = { has_country_flag = dhrondan_two_world_federation_ready }
						is_faction_leader = yes
					}
				}
				custom_effect_tooltip = dhrondan_compact_federation_terms_tt
			}
		}
		hidden_effect = {
			if = {
				limit = { dhrondan_compact_response_can_commit = yes }
				if = {
					limit = { event_target:dhrondan_diplomatic_actor = { has_country_flag = dhrondan_diplomatic_offer_partner_accord } }
					dhrondan_complete_terrestrial_partner_accord = yes
				}
				else = {
					diplomatic_relation = { country = event_target:dhrondan_diplomatic_actor relation = non_aggression_pact active = yes }
					add_opinion_modifier = { target = event_target:dhrondan_diplomatic_actor modifier = dhrondan_two_world_compact }
					set_country_flag = dhrondan_two_world_compact_partner
					dhrondan_apply_compact_focus_terms = yes
					event_target:dhrondan_diplomatic_actor = {
						add_opinion_modifier = { target = event_target:dhrondan_diplomatic_recipient modifier = dhrondan_two_world_compact }
						set_country_flag = dhrondan_diplomatic_compact_concluded
						add_stability = constant:dhrondan_decision_reward.stability_small
						dhrondan_clear_diplomatic_offer = yes
						country_event = { id = chaosx.nr16.50 }
					}
				}
			}
			else = { dhrondan_close_owned_compact_response = yes }
		}
	}

	option = {
		name = chaosx.nr16.49.b
		trigger = {
			dhrondan_compact_response_is_valid = yes
			event_target:dhrondan_diplomatic_actor = { NOT = { has_country_flag = dhrondan_diplomatic_offer_partner_accord } }
		}
		ai_chance = {
			base = constant:dhrondan_event_ai.refuse_base
			modifier = {
				factor = constant:dhrondan_event_ai.hostile_factor
				has_opinion = {
					target = event_target:dhrondan_diplomatic_actor
					value < @DHR_NEGATIVE_OPINION_THRESHOLD
				}
			}
		}
		if = {
			limit = { event_target:dhrondan_diplomatic_actor = { hidden_trigger = { has_country_flag = dhrondan_diplomatic_offer_partner_accord } } }
			custom_effect_tooltip = dhrondan_partner_accord_refuse_effect_tt
		}
		else = { custom_effect_tooltip = dhrondan_compact_refuse_effect_tt }
		hidden_effect = {
			if = {
				limit = { dhrondan_compact_response_can_commit = yes }
				if = {
					limit = { event_target:dhrondan_diplomatic_actor = { has_country_flag = dhrondan_diplomatic_offer_partner_accord } }
					event_target:dhrondan_diplomatic_actor = {
						dhrondan_clear_diplomatic_offer = yes
					}
				}
				else = {
					add_opinion_modifier = { target = event_target:dhrondan_diplomatic_actor modifier = dhrondan_compact_refused }
					event_target:dhrondan_diplomatic_actor = {
						dhrondan_clear_diplomatic_offer = yes
						country_event = { id = chaosx.nr16.51 }
					}
				}
			}
			else = { dhrondan_close_owned_compact_response = yes }
		}
	}

		custom_effect_tooltip = dhrondan_compact_expired_effect_tt
		hidden_effect = {
			dhrondan_close_owned_compact_response = yes
		}
	}
