country_event = {
	id = chaosx.fallout.1
	title = chaosx.fallout.1.t
	desc = chaosx.fallout.1.d
	picture = GFX_report_event_air_winter_phase_1
	is_triggered_only = yes
	trigger = { air_winter_event_targets_are_valid = yes }

	option = {
		name = chaosx.fallout.1.a
		ai_chance = {
			base = constant:air_winter_event_ai.primary
			modifier = {
				factor = constant:air_winter_event_ai.strong_modifier
				event_target:air_winter_event_state = { check_variable = { air_winter_adaptation < constant:air_winter_event_ai.low_ledger } }
			}
		}
		hidden_effect = {
			event_target:air_winter_event_state = {
				add_to_variable = { air_winter_adaptation = constant:air_winter_event_ledger.adaptation_moderate }
				subtract_from_variable = { air_winter_food_reserve = constant:air_winter_event_ledger.food_minor }
				set_state_flag = air_winter_memory_boreal_shift_network
				air_winter_event_refresh_state = yes
			}
			add_command_power = constant:air_winter_event_country.command_power_minor_cost
		}
		custom_effect_tooltip = chaosx.fallout.1.a.tt
	}

	option = {
		name = chaosx.fallout.1.b
		ai_chance = {
			base = constant:air_winter_event_ai.secondary
			modifier = {
				factor = constant:air_winter_event_ai.strong_modifier
				has_war = yes
			}
		}
		hidden_effect = {
			event_target:air_winter_event_state = {
				add_to_variable = { air_winter_exposure = constant:air_winter_event_ledger.exposure_moderate }
				add_to_variable = { air_winter_disease_pressure = constant:air_winter_event_ledger.disease_minor }
