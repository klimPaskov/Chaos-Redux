# ============================================================================
# CHAOS REDUX - EVENT 042 EQUIPMENT FROM HEAVENS
# ============================================================================
# chaosx.nr42.1 is the canonical entry and the only delivery transaction. The
# shared dispatcher prepares one uniformly random ordinary recipient before
# this event fires. The hidden immediate block builds the manifest, applies
# the stockpile grant once, and opens the discovery popup (.2).
#
# chaosx.nr42.9 performs sequential recovery visits; .13 or .16 summarises
# them once. The other report ids remain dormant compatibility events. None grants
# equipment, records Event 42 history, advances pacing, or changes weight.
# chaosx.nr42.19 closes a stale chain. chaosx.nr42.30 and .31 are hidden
# achievement pulses.
# ============================================================================

# ---------------------------------------------------------------------------
# Delivery
# ---------------------------------------------------------------------------

country_event = {
	id = chaosx.nr42.1
	hidden = yes
	is_triggered_only = yes
	immediate = { equipment_from_heavens_enter_delivery = yes }
}

# The visible discovery reads the grant prepared by the canonical entry.
country_event = {
	id = chaosx.nr42.2
	title = chaosx.nr42.1.t
	desc = {
		text = chaosx.nr42.1.desc.impossible
		trigger = { check_variable = { efh_manifest_magnitude = constant:equipment_from_heavens_magnitude.impossible } }
	}
	desc = {
		text = chaosx.nr42.1.desc.colossal
		trigger = { check_variable = { efh_manifest_magnitude = constant:equipment_from_heavens_magnitude.colossal } }
	}
	desc = {
		text = chaosx.nr42.1.desc.enormous
		trigger = {
			NOT = { check_variable = { efh_manifest_magnitude = constant:equipment_from_heavens_magnitude.colossal } }
			NOT = { check_variable = { efh_manifest_magnitude = constant:equipment_from_heavens_magnitude.impossible } }
		}
	}
	picture = "[Root.GetEquipmentFromHeavensDiscoveryPicture]"
	is_triggered_only = yes

	trigger = { has_country_flag = efh_chain_active }

	option = {
		name = chaosx.nr42.1.a
		ai_chance = { base = 1 }
		custom_effect_tooltip = chaosx.nr42.1.magnitude_tt
		equipment_from_heavens_show_manifest_tooltips = yes
		hidden_effect = { equipment_from_heavens_start_recovery = yes }
	}
}

# Closes a chain whose final report never resolved. Only a chain at least
# forty-four days old is touched, so a newer delivery is never cleared.
country_event = {
	id = chaosx.nr42.19
	title = chaosx.nr42.9.t
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				has_country_flag = efh_chain_active
				has_variable = efh_recovery_date
			}
			set_temp_variable = { chaosx_date_days = 0 }
			set_temp_variable = { chaosx_date_offset = global.date }
			subtract_from_temp_variable = { chaosx_date_offset = efh_recovery_date }
			chaosx_convert_date_offset_to_days = yes
			set_temp_variable = { efh_chain_age = chaosx_date_days }
			if = {
				limit = { check_variable = { var = efh_chain_age value = constant:equipment_from_heavens_report.safety_close_days compare = greater_than_or_equals } }
				equipment_from_heavens_cleanup_chain = yes
			}
		}
	}
}
