	option = {
		name = chaosx.nr40.120.a
		custom_effect_tooltip = chaosx.nr40.120.a_tt
		ai_chance = {
			base = 35
			modifier = { add = 30 lawrence_fed_outcome_is_british = yes is_subject_of = ENG }
			modifier = { add = 25 lawrence_fed_outcome_is_sovereign = yes has_country_flag = lawrence_bloc_member }
			modifier = { add = 25 has_country_flag = lawrence_bloc_member NOT = { lawrence_fed_outcome_is_british = yes } }
			modifier = { add = 20 has_variable = lawrence_settlement }
			modifier = { add = 10 has_idea = lawrence_counter_bloc_member }
			modifier = { factor = 0.5 is_subject_of = ENG NOT = { lawrence_fed_outcome_is_british = yes } }
			modifier = { factor = 0.5 has_war = yes has_defensive_war = yes }
		}
		hidden_effect = { lawrence_ratify_charter = yes }
	}
	option = {
		name = chaosx.nr40.120.b
		custom_effect_tooltip = chaosx.nr40.120.b_tt

			base = 40
			modifier = { add = 20 is_in_faction = yes }

		}
		hidden_effect = { lawrence_refuse_charter = yes }
	}
}
