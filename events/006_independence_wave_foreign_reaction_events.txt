# ============================================================================
# CHAOS REDUX - EVENT 006 FOREIGN REACTION EVENTS
# Overview:
# - .64 hidden router: dispatches an actor's one reaction per wave, answers a
#   patron offer for an AI released country, and answers a remnant petition
#   for an AI recipient.
# - .69 remnant petition: the chosen government decides whether to guarantee
#   a remnant former host. The first option declines, which the 13-day
#   timeout selects. .70 tells the host the result.
# - .72 hidden deadline: closes a queue whose receipts outlived every answer.
# - .65 neighbor, .66 regional power, .67 great power and .68 earlier-wave
#   state: one top-target choice per actor. The first option is the
#   noncommittal stance, which the 13-day timeout selects.
# - .71 consent: the released country answers a patron offer. The first
#   option refuses, which the 13-day timeout selects.
# - Every option applies its stance through the shared foreign reaction
#   effects, which validate the receipt again before any cost or reward.
# - The ordinary windows reuse the accepted ASSET-001 wave-summary report.
# ============================================================================

add_namespace = chaosx.nr6

# `timeout_days` rejects script constants; these mirror
# `independence_wave_foreign_reaction_timing.response_timeout_days` and
# `independence_wave_remnant_petition.response_timeout_days`.
@CR_SC_INDEPENDENCE_WAVE_FOREIGN_REACTION_TIMEOUT_DAYS = 13
@CR_SC_INDEPENDENCE_WAVE_REMNANT_PETITION_TIMEOUT_DAYS = 13

country_event = {
	id = chaosx.nr6.64
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				is_ai = yes
				has_country_flag = independence_wave_foreign_offer_pending
			}
			# An invalid offer still closes once and frees the waiting actor.
			independence_wave_foreign_reaction_resolve_ai_offer = yes
		}
		if = {
			limit = {
				is_ai = yes
				has_country_flag = independence_wave_remnant_petition_received
			}
			independence_wave_remnant_petition_resolve_ai = yes
		}
		if = {
			limit = {
				has_country_flag = independence_wave_foreign_reaction_pending
				NOT = { has_country_flag = independence_wave_foreign_reaction_routed }
			}
			independence_wave_foreign_reaction_route_actor = yes
		}
	}
}
