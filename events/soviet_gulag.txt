# ============================================================================
# CHAOS REDUX - SOVIET GULAG, PARANOIA, AND FAMINE EVENTS
# ============================================================================
# Overview:
# - Provides threshold-only famine, administrative-breakdown, relief, and
#   post-collapse evidence events for the Soviet repression network.
# - Recurring population harm remains in the shared registered-state monthly
#   processor, so these events do not add a second famine death tick.
# - Choices use the vanilla-paranoia projection and existing Union Crisis bridge.
# ============================================================================

add_namespace = soviet_gulag

country_event = {
	id = soviet_gulag.1
	trigger = { original_tag = SOV }
	title = soviet_gulag.1.t
	desc = soviet_gulag.1.d
	picture = GFX_report_event_soviet_famine_warning
	is_triggered_only = yes

	option = {
		name = soviet_gulag.1.a
		trigger = { camp_rework_soviet_relief_order_ready = yes }
		set_country_flag = sov_famine_warning_fired
		if = {
			limit = { camp_rework_soviet_relief_order_ready = yes }
			custom_effect_tooltip = soviet_gulag_relief_funded_tt
			hidden_effect = { camp_rework_soviet_emergency_famine_relief = yes }
		}
		else_if = {
			limit = { camp_rework_soviet_relief_mobilization_pending = yes }
			custom_effect_tooltip = soviet_gulag_relief_shortfall_tt
			hidden_effect = { camp_rework_soviet_relief_mobilization_failed = yes }
		}
		else_if = {
			limit = { camp_rework_soviet_relief_underway = yes }
			custom_effect_tooltip = soviet_gulag_relief_underway_tt
		}
		else = { custom_effect_tooltip = soviet_gulag_famine_eased_tt }
		clr_country_flag = camp_rework_soviet_relief_order_accepted
		clear_variable = sov_famine_warning_state_id
	}
	option = {
		name = soviet_gulag.1.c
		trigger = {
			camp_rework_soviet_relief_mobilization_pending = yes
			NOT = { camp_rework_soviet_relief_resources_ready = yes }
		}
		set_country_flag = sov_famine_warning_fired
		custom_effect_tooltip = soviet_gulag_relief_shortfall_tt
		camp_rework_soviet_relief_mobilization_failed = yes
		clear_variable = sov_famine_warning_state_id
	}
	option = {
		name = soviet_gulag.1.relief_underway
		trigger = { camp_rework_soviet_relief_underway = yes }
		custom_effect_tooltip = soviet_gulag_relief_underway_tt
		clear_variable = sov_famine_warning_state_id
	}
	option = {
		name = soviet_gulag.1.famine_eased
		trigger = { camp_rework_soviet_famine_eased = yes }
		custom_effect_tooltip = soviet_gulag_famine_eased_tt
		clear_variable = sov_famine_warning_state_id
	}
	option = {
		name = soviet_gulag.1.b
		trigger = {
			has_variable = sov_famine_warning_state_id
			var:sov_famine_warning_state_id = {
				camp_rework_state_is_actor_responsible_controlled = yes
				has_state_flag = genocide_soviet_famine_pressure_site
				NOT = { has_state_flag = camp_rework_soviet_records_concealed }
			}
		}
		set_country_flag = sov_famine_warning_fired
		set_variable = { sov_famine_event_state_id = sov_famine_warning_state_id }
		camp_rework_soviet_conceal_famine_mortality = yes
		clear_variable = sov_famine_event_state_id
		clear_variable = sov_famine_warning_state_id
	}
}
