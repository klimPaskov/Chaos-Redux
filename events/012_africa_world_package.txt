add_namespace = africa_world_package

# ============================================================================
# CHAOS REDUX - EVENT 012 CONSTITUENT AGENCY AND PACKAGE RATIFICATION PROTOCOL
# ============================================================================
# The package actor can invite, negotiate, or contest a response, but the
# constituent owns every acceptance, refusal, and withdrawal answer. Event
# targets preserve the exact package/constituent pair for this bounded chain.
# No event creates a tag, subject, territorial transfer, recurring scan, or
# final package readiness flag.
# ============================================================================

# Constituent receives a package invitation.
country_event = {
	id = africa_world_package.1
	title = africa_world_package.1.t
	desc = africa_world_package.1.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes
	trigger = {
		has_event_target = africa_world_protocol_package
		event_target:africa_world_protocol_package = { africa_world_package_is_installed = yes }
		africa_world_constituent_invitation_is_open = yes
	}
	immediate = { save_event_target_as = africa_world_protocol_constituent }

	option = {
		name = africa_world_package.1.a
		custom_effect_tooltip = africa_world_package.1.a.tt
		ai_chance = { base = constant:africa_world_package_protocol.ai_accept }
		set_country_flag = africa_world_constituent_protocol_accept
		event_target:africa_world_protocol_package = { country_event = { id = africa_world_package.2 } }
	}
	option = {
		name = africa_world_package.1.b
		custom_effect_tooltip = africa_world_package.1.b.tt
		ai_chance = { base = constant:africa_world_package_protocol.ai_negotiate }
		africa_world_open_constituent_counterterms = yes
		event_target:africa_world_protocol_package = { country_event = { id = africa_world_package.3 } }
	}
	option = {
		name = africa_world_package.1.c
		custom_effect_tooltip = africa_world_package.1.c.tt
		ai_chance = { base = constant:africa_world_package_protocol.ai_refuse }
		set_country_flag = africa_world_constituent_protocol_refuse
		event_target:africa_world_protocol_package = { country_event = { id = africa_world_package.4 } }
	}
}
